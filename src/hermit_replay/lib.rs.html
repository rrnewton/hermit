9//! Hermit's ptrace replay tool, [`Replayer`], and the one place its Detcore
10//! instantiation is rooted.
11//!
12//! `detcore::Detcore<Replayer>` drives `reverie_ptrace`'s `TracedTask`, and
13//! that instantiation is large. In its own crate it builds in a separate
14//! rustc process, beside the other backends' instantiations. The entry point,
15//! [`spawn_tracer`], is not generic and returns a boxed future, so `hermit`
16//! instantiates nothing of it (see `hermit-record` for why it is boxed).
17

38/// The replay tracer: `detcore::Detcore<Replayer>` under `reverie_ptrace`.
39pub type Tracer = reverie_ptrace::Tracer<detcore::GlobalState>;
40
41/// The future [`spawn_tracer`] returns.
42pub type SpawnTracer = Pin<Box<dyn Future<Output = Result<Tracer, reverie::Error>>>>;
43
44/// Spawns COMMAND under the replay tool with CONFIG, as `hermit replay` does:
45/// serving a gdb remote on the GDBSERVER connection when given, and telling
46/// gdbserver the guest is already sequentialized when SEQUENTIALIZED_GUEST.
47///
48/// The future is boxed so the async body, and everything it instantiates, is
49/// compiled in this crate rather than in its caller.
50pub fn spawn_tracer(
51    command: Command,
52    config: detcore::Config,
53    gdbserver: Option<GdbConnection>,
54    sequentialized_guest: bool,
55) -> SpawnTracer {
56    Box::pin(async move {
57        let mut builder = reverie_ptrace::TracerBuilder::<detcore::Detcore<Replayer>>::new(command)
58            .config(config);
59        if let Some(connection) = gdbserver {
60            builder = builder.gdbserver(connection);
61        }
62        if sequentialized_guest {
63            builder = builder.sequentialized_guest();
64        }
65        builder.spawn().await
66    })
67}
