9use std::cell::UnsafeCell;
10use std::env;
11use std::sync::Arc;
12use std::sync::atomic::AtomicUsize;
13use std::sync::atomic::Ordering;
14use std::thread;
15const NUM_ELEMENTS: usize = 20_000_000;
16
17/// A shared array the workers mutate concurrently through `&self`.
18///
19/// WHY `UnsafeCell` AND NOT A RAW POINTER. `Arc` hands out shared access only, so
20/// a pointer derived from it carries read-only provenance and no cast adds write
21/// provenance to it -- that is a language rule, not a lint, and miri rejects the
22/// write. `UnsafeCell` is the sanctioned way to mutate through a shared
23/// reference. Two earlier spellings of this loop were both unsound:
24///
25/// ```text
26/// Arc::get_mut_unchecked(&mut data)                   UB: data race between the
27///                                                         non-atomic write and the
28///                                                         other thread's retag
29/// from_raw_parts_mut(data.as_ptr() as *mut u64, len)  UB: retag for Unique from a
30///                                                         tag granting only
31///                                                         SharedReadOnly
32/// ```
33///
34/// WHY `Arc<[UnsafeCell<u64>]>` AND NOT `Arc<Vec<UnsafeCell<u64>>>`. The elements
35/// of an `Arc<[T]>` live inside the `Arc` allocation, so reaching one is a single
36/// indirection -- exactly what `Arc<[u64]>` cost before. An `Arc<Vec<T>>` stores a
37/// `Vec` header in that slot and the elements elsewhere, adding a load to a loop
38/// that runs ten million times per thread. This array is the subject of the weekly
39/// `mem_race` PMU gates, whose premise is that preemption is driven by retired
40/// conditional branches, so an extra load or a dropped bounds check moves the
41/// numbers they pin.
42struct SharedCells(Arc<[UnsafeCell<u64>]>);
43
44// SAFETY: the workers only ever write cells whose index came from the shared
45// atomic counter, so no two threads touch the same cell, and every read happens
46// after `join()`. Sharing the handle is therefore sound; the type cannot enforce
47// that discipline, which is why this impl is unsafe and why the invariant is
48// stated here rather than assumed.
49unsafe impl Send for SharedCells {}
50unsafe impl Sync for SharedCells {}
51
52impl Clone for SharedCells {
53    fn clone(&self) -> Self {
54        SharedCells(Arc::clone(&self.0))
55    }
56}
57
58impl SharedCells {
59    fn zeroed(len: usize) -> Self {
60        let cells: Vec<UnsafeCell<u64>> = (0..len).map(|_| UnsafeCell::new(0)).collect();
61        SharedCells(cells.into())
62    }
63
64    fn len(&self) -> usize {
65        self.0.len()
66    }
67
68    /// Read a cell. Callers must have established happens-before with the writers.
69    fn get(&self, i: usize) -> u64 {
70        // SAFETY: indexing keeps its bounds check; the caller has joined the workers.
71        unsafe { *self.0[i].get() }
72    }
73
74    /// Write a cell. Callers must own `i` exclusively.
75    fn set(&self, i: usize, v: u64) {
76        // SAFETY: `i` came from the shared counter, so this worker owns it. `self.0[i]`
77        // is a bounds-checked index -- do NOT replace it with `get_unchecked`, the
78        // branch it emits is part of what the RCB preemption timer measures.
79        unsafe { *self.0[i].get() = v }
80    }
81}
82
83/// Calculate the number of switch points. E.g. the number of times we observed interleaved
84/// writes between the threads.
85fn count_switch_points(shared_data: &SharedCells) -> u64 {
86    // Calculate the number of switch points. E.g. the number of times we observed interleaved
87    // writes between the threads.
88    let mut switch_points = 0;
89    let mut prev = shared_data.get(0);
90    for i in 1..shared_data.len() {
91        if prev != shared_data.get(i) {
92            prev = shared_data.get(i);
93            switch_points += 1;
94        }
95    }
96    switch_points
97}

99/// In guest mode two threads will try to fill up half of the data array with their thread id as
100/// value. The threads grab indices through an atomic int. For sufficiently large arrays we expect
101/// the thread ids to show up interleaved.
102fn run_test() -> u64 {
103    let shared_data = SharedCells::zeroed(NUM_ELEMENTS);
104    let shared_idx = Arc::new(AtomicUsize::new(0));
105
106    // `tag` is the distinct per-worker value written into the array. It used to be
107    // the thread id, but `ThreadId` has no stable numeric accessor and the test only
108    // needs the two workers to write DIFFERENT values -- count_switch_points compares
109    // adjacent elements and never inspects the value itself. They must nevertheless
110    // stay DISTINCT and nonzero: give both workers the same tag and the switch-point
111    // count collapses to zero, which is the test measuring nothing while still
112    // passing its compile.
113    fn worker(idx: Arc<AtomicUsize>, data: SharedCells, tag: u64) {
114        // The contention is the point of this test and must survive any tidy-up: two
115        // threads interleaving writes into ONE array, ordered only by the shared
116        // counter, is what produces switch points at all. What must NOT survive is
117        // undefined behaviour -- the elements never raced (the counter hands out
118        // disjoint indices); what raced was the construction of the `&mut`. So do NOT
119        // "fix" this with atomics: that changes the memory operations the test exists
120        // to exercise, and it is not what was ever wrong here.
121        for _ in 0..(NUM_ELEMENTS / 2) {
122            let idx = idx.fetch_add(1, Ordering::SeqCst);
123            data.set(idx, tag);
124        }
125    }
126
127    // One-armed variant: spawn one child:
128    let handle = {
129        let (idx, data) = (shared_idx.clone(), shared_data.clone());
130        thread::spawn(move || {
131            // This exercises the futex_wait-called-by-kernel behavior, even on "bottom":
132            worker(idx, data, 1)
133        })
134    };
135    println!("Parent done spawning child thread and starting own work...");
136    worker(shared_idx, shared_data.clone(), 2);
137
138    println!("Parent done with work and joining child thread..");
139    handle.join().unwrap();
140
141    let switch_points = count_switch_points(&shared_data);
142    let s: String = format!("Switch points: {}\n", switch_points);
143    println!("{}", s); // Print a bit more atomically.
144    // Only when running deterministically:
145    switch_points
146}

148fn main() {
149    if env::var("HERMIT_MODE") == Ok("strict".to_string()) {
150        eprintln!("Running sequentialized, deterministically.");
151        let switches = run_test();
152        assert!(
153            switches > 1,
154            "Expecting deterministic preemptions when using RCB timers"
155        );
156    } else {
157        eprintln!("Running mem_race, but not expecting determinism.");
158        let _switches = run_test();
159    }
160}
