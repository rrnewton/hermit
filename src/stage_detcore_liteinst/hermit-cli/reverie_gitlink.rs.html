9// The gitlink half of the build scripts' Reverie pin reader, kept apart
10// from `reverie_pin.rs` on purpose: hermit-cli's library test module
11// includes `reverie_pin.rs` alone to bracket `parse_reverie_pin`, and
12// these functions — used only by build-side consumers — would be dead
13// code there (and an `expect(dead_code)` here would be unfulfilled in
14// the build scripts that do call them; the fbsource cargo clippy job
15// compiles both shapes and refuses either mismatch). Build-side
16// consumers include `reverie_pin.rs` first, then this file, in one
17// scope: `reverie_pin_at` calls `parse_reverie_pin` from it.
18//
19// Read and watch the SAME source (seat round 4): the embedded pin is
20// the submodule CHECKOUT's HEAD — the sources Cargo actually compiles
21// through the path dependencies — and `reverie_pin_watch_paths_in`
22// watches exactly that file (plus the Eden marker). V0.7 read the
23// superproject index while watching the submodule HEAD, so the
24// documented advance sequence (checkout, build, stage) embedded the
25// pre-advance pin. Where there is no submodule checkout (Eden), the
26// marker is read instead; the recorded gitlink and the manifest parse
27// remain as fallbacks for trees without either.
28
29fn pin_git(root: &std::path::Path, args: &[&str]) -> Option<String> {
30    let output = std::process::Command::new("git")
31        .arg("-C")
32        .arg(root)
33        .args(args)
34        .output()
35        .ok()?;
36    if !output.status.success() {
37        return None;
38    }
39    Some(String::from_utf8_lossy(&output.stdout).into_owned())
40}

42fn pin_full_sha(value: &str) -> Option<String> {
43    (value.len() == 40 && value.chars().all(|c| c.is_ascii_hexdigit())).then(|| value.to_owned())
44}

46/// The submodule checkout's HEAD: the Reverie sources a build in this
47/// tree actually compiles. Requires an initialized submodule
48/// (`reverie/.git` present, so Git cannot discover the superproject
49/// instead and answer for the wrong repository).
50fn reverie_checkout_head(root: &std::path::Path) -> Option<String> {
51    let submodule = root.join("reverie");
52    if !submodule.join(".git").exists() {
53        return None;
54    }
55    pin_git(&submodule, &["rev-parse", "HEAD"])
56        .and_then(|text| pin_full_sha(text.trim()))
57}

59/// The pin recorded by the Eden/Sapling marker, `reverie.submodule.txt`
60/// holding exactly `Subproject commit <40-hex>` on one line — the only
61/// spelling CodeSync produces or consumes. Anything else yields `None`
62/// (fall through), never a partial read.
63fn reverie_marker_pin(root: &std::path::Path) -> Option<String> {
64    let text = std::fs::read_to_string(root.join("reverie.submodule.txt")).ok()?;
65    let line = text.strip_suffix('\n').unwrap_or(&text);
66    if line.contains('\n') {
67        return None;
68    }
69    line.strip_prefix("Subproject commit ")
70        .and_then(pin_full_sha)
71}

73/// The Reverie revision recorded by the `reverie` submodule gitlink of the
74/// checkout rooted at `root`, or `None` when it cannot be read: the index
75/// entry when an advance is staged, else HEAD's tree entry — the same
76/// pair `scripts/check-reverie-pin.rs` reads. Fallback only (see
77/// `reverie_pin_at`): provenance for trees with no submodule checkout
78/// and no marker (tarballs, uninitialized submodules), where the value
79/// degrades to `"unknown"` at callers exactly as an unparseable
80/// manifest did.
81fn reverie_gitlink_pin(root: &std::path::Path) -> Option<String> {
82    // Index entry: `160000 <sha> <stage>\treverie`. No let-chain here:
83    // version_provenance's fixture crates compile this file at edition
84    // 2021, where let-chains are rejected.
85    if let Some(text) = pin_git(root, &["ls-files", "--stage", "--", "reverie"]) {
86        for line in text.lines() {
87            let mut fields = line.split_whitespace();
88            let sha = (fields.next() == Some("160000"))
89                .then(|| fields.next().and_then(pin_full_sha))
90                .flatten();
91            if let Some(sha) = sha {
92                return Some(sha);
93            }
94        }
95    }
96    // HEAD tree entry: `160000 commit <sha>\treverie`.
97    let text = pin_git(root, &["ls-tree", "HEAD", "--", "reverie"])?;
98    let mut fields = text.split_whitespace();
99    if fields.next() != Some("160000") || fields.next() != Some("commit") {
100        return None;
101    }
102    fields.next().and_then(pin_full_sha)
103}

105/// The Reverie pin of the checkout rooted at `root`, in the order the
106/// build actually sees the sources: submodule checkout HEAD, Eden
107/// marker, recorded gitlink, pre-migration manifest. `None` renders
108/// as `"unknown"` at callers.
109fn reverie_pin_at(root: &std::path::Path) -> Option<String> {
110    if let Some(pin) = reverie_checkout_head(root) {
111        return Some(pin);
112    }
113    if let Some(pin) = reverie_marker_pin(root) {
114        return Some(pin);
115    }
116    if let Some(pin) = reverie_gitlink_pin(root) {
117        return Some(pin);
118    }
119    let text = std::fs::read_to_string(root.join("detcore/Cargo.toml")).ok()?;
120    parse_reverie_pin(&text)
121}
