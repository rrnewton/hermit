9// The Reverie pin reader shared by the two build scripts that need it.
10//
11// `include!`d rather than imported because both consumers are build scripts in
12// different crates (`hermit-cli/build_support.rs` and `hermit-install/build.rs`)
13// and cannot depend on each other. Two hand-copied readers previously existed
14// with nothing asserting they agreed.
15//
16// The rule here is the one `scripts/check-reverie-pin.rs::unique_pin` enforces:
17// **every** Reverie revision named in the manifest must agree, and a manifest
18// naming more than one is refused rather than resolved. First-match-wins is the
19// grep hazard this project has been bitten by -- a recursive search returns
20// several answers and the true one is not the most common, nor the first.
21//
22// Two concrete inputs that first-match-wins gets wrong, both reachable in a
23// locally-dirty tree mid-bump:
24//   * a commented-out dependency line above the live one, whose revision is
25//     read as though it were the pin;
26//   * a manifest halfway through a bump, naming the old revision on one line
27//     and the new one on the next.
28//
29// Ambiguity yields `None`, which callers render as `"unknown"`. The pin guard
30// SKIPS on `"unknown"` rather than refusing, so an unreadable manifest degrades
31// to the pre-guard behaviour instead of refusing a correctly staged runtime.
32// That is the deliberate choice between the two available errors: failing to
33// catch a stale runtime is recoverable, while refusing a good one while naming
34// a revision that was never the pin sends the reader somewhere that does not
35// exist.
36
37/// Extract the single Reverie revision a manifest pins, or `None` if the
38/// manifest names zero revisions, more than one distinct revision, or names
39/// Reverie on a line whose revision cannot be parsed.
40fn parse_reverie_pin(text: &str) -> Option<String> {
41    let mut found: Option<String> = None;
42    for line in text.lines() {
43        // A commented-out dependency is not the pin. Skipping these is what
44        // makes the all-must-agree rule safe to apply to a dirty tree.
45        if line.trim_start().starts_with('#') {
46            continue;
47        }
48        if !line.contains("rrnewton/reverie") {
49            continue;
50        }
51        let rev = line
52            .split("rev = \"")
53            .nth(1)
54            .and_then(|rest| rest.split('"').next())
55            .filter(|rev| rev.len() == 40 && rev.chars().all(|c| c.is_ascii_hexdigit()));
56        // A Reverie line we cannot parse makes the manifest ambiguous. Treating
57        // it as absent would let a malformed line hide a disagreement.
58        let rev = rev?;
59        match &found {
60            Some(seen) if seen != rev => return None,
61            Some(_) => {}
62            None => found = Some(rev.to_owned()),
63        }
64    }
65    found
66}
