53    std::env::var("DETCORE_TEST_RUST_LOG").unwrap_or_else(|_| {

60fn install_global_test_subscriber<W>(trace_level: &str, writer: W) -> bool
61where
62    W: for<'writer> MakeWriter<'writer> + Send + Sync + 'static,
63{
64    let collector = tracing_subscriber::fmt()
65        .with_env_filter(trace_level)
66        .with_writer(writer)
67        .finish();
68    tracing::subscriber::set_global_default(collector).is_ok()
69}

348/// A basic oracle, which expects a success exit code.
349pub fn expect_success(o: &Output, _s: <Detcore as Tool>::GlobalState) {
350    if o.status != reverie::ExitStatus::Exited(0) {
351        eprintln!("Expected successful exit code, instead tracee output was:");
352        print_tracee_output(o);
353        panic!("Guest exited with non-zero status.",);
354    }
355}

358macro_rules! make_det_test_variants {
359    ( $fn:path ) => {
360        $crate::make_det_test_variants!($fn, "all");
361    };
362    ( $fn:path, "all" ) => {
363        $crate::make_det_test_variants!(@variants ["bottom" "middle" "default" "top"] $fn);
364    };
365    ( $fn:path, $($various:tt),*  ) => {
366        $crate::make_det_test_variants!(@variants [$($various)*] $fn);
367    };
368
369    (@variants [ ] $fn:path ) => {
370    };
371
372    // Here we use good-old Lisp-style lists with no commas:
373    (@variants [ $first:tt $($rest:tt)* ] $fn:path ) => {
374        $crate::make_det_test_variants!(@one_variant $first, $fn);
375        $crate::make_det_test_variants!(@variants [ $($rest)* ] $fn);
376    };
377
378    ( @one_variant "bottom", $fn:path ) => {
379        #[test]
380        fn bottom_detcore() {
381            $fn(& $crate::BOTTOM_CFG);
382        }
383    };
384    ( @one_variant "middle", $fn:path ) => {
385        #[test]
386        fn middle_detcore() {
387            $fn(& $crate::MIDDLE_CFG);
388        }
389    };
390    ( @one_variant "default", $fn:path ) => {
391        #[test]
392        fn default_detcore() {
393            $fn(& ::core::default::Default::default());
394        }
395    };
396    ( @one_variant "top", $fn:path ) => {
397        #[test]
398        fn top_detcore() {
399            $fn(& $crate::TOP_CFG);
400        }
401    };

404/// A convenient way to wrap a function to test multiple detcore execution variants, each
405/// as a separate unit test. This is in contrast with `det_test_all_configs` which runs
406/// multiple modes sequentially under one test target.
407///
408/// Arguments are `basic_det_test(function, predicate, modes...)`, where:
409///   - `function` is the procedure under test.
410///   - `predicate` is a function that accepts a Config and returns true if the test should
411///     run deterministically in that configuration.
412///
413/// This generates calls to `det_test_fn_with_config`.
414#[macro_export]
415macro_rules! basic_det_test {
416    ( $fn:path ) => {
417        $crate::basic_det_test!($fn, |_| true);
418    };
419    ( $fn:path, $f:expr ) => {
420        $crate::basic_det_test!(@gendef $fn, $f);
421        $crate::make_det_test_variants!(detcore, "all");
422    };
423    ( $fn:path, $f:expr, $($variants:tt),+ ) => {
424        $crate::basic_det_test!(@gendef $fn, $f);
425        $crate::make_det_test_variants!(detcore $(,$variants)* );
426    };
427
428    (@gendef $fn:path, $f:expr ) => {
429        fn detcore(cfg: & ::detcore::Config) {
430            $crate::det_test_fn_with_config(
431                $f(cfg),
432                $fn,
433                cfg.clone(),
434                $crate::expect_success,
435            );
436        }
437    }
438}

440/// Runs a test across MULTIPLE configurations.  This combines the multiple variants into
441/// a single test execution (running them one after another).  This can be convenient at
442/// times, but, when debugging, you may want to split these apart into their own test
443/// targets.
444///
445/// The first function argument returns `true` iff the test *should* be deterministic
446/// under a given configuration.
447pub fn det_test_all_configs<C, T, O>(check: C, test: T, oracle: O)
448where
449    C: Fn(&Config) -> bool,
450    T: Fn(&Config) + Sync,
451    O: Fn(&Output, <Detcore as Tool>::GlobalState) + Clone,
452{
453    let do_cfg = |cfg: Config| {
454        let cfg2 = cfg.clone();
455        let test2 = || test(&cfg);
456        println!("Full config: {:?}", cfg);
457        println!(
458            "================================================================================"
459        );
460        let oracle2 = |output: &Output, state: <Detcore as Tool>::GlobalState| {
461            (oracle.clone())(output, state);
462        };
463        det_test_fn_with_config(check(&cfg), test2, cfg2, oracle2);
464    };
465    println!("\nDEFAULT mode");
466    do_cfg(DEFAULT_CFG.clone());
467    println!("\n\"Bottom\", least-strict configuration");
468    do_cfg(BOTTOM_CFG.clone());
469    println!("\n\"Middle\", medium-strict configuration");
470    do_cfg(MIDDLE_CFG.clone());
471    println!("\n\"Top\", most-strict configuration");
472    do_cfg(TOP_CFG.clone());
473}

475/// The log messages produced by the tracer while executing the tracee, split into lines.

478struct BufWriter {
479    buf: Arc<Mutex<Vec<u8>>>,
480}
481
482impl Clone for BufWriter {
483    fn clone(&self) -> Self {
484        BufWriter {
485            buf: self.buf.clone(),
486        }
487    }
488}
489
490impl BufWriter {
491    fn new() -> BufWriter {
492        BufWriter {
493            buf: Arc::new(Mutex::new(Vec::new())),
494        }
495    }
496
497    fn buf(&self) -> MutexGuard<'_, Vec<u8>> {
498        self.buf.lock().unwrap()
499    }
500
501    fn get_strings(&self) -> Vec<String> {
502        let mut b = self.buf();
503        let s = String::from_utf8_lossy(&b[..]).to_string();
504        b.clear();
505        s.lines().map(String::from).collect()
506    }
507}
508
509impl io::Write for BufWriter {
510    fn write(&mut self, msg: &[u8]) -> io::Result<usize> {
511        let _ = std::io::stderr().write(msg);
512        self.buf().write(msg)
513    }
514
515    fn flush(&mut self) -> std::io::Result<()> {
516        let _ = std::io::stderr().flush();
517        self.buf().flush()
518    }
519}
520
521impl MakeWriter<'_> for BufWriter {
522    type Writer = BufWriter;
523
524    fn make_writer(&self) -> Self::Writer {
525        BufWriter {
526            buf: self.buf.clone(),
527        }
528    }
529}

531/// Similar to Reverie's `test_fn_with_config` that captures and returns the tracer's
532/// logs as well.
533fn test_fn_with_logs<T, F>(
534    f: F,
535    config: <T::GlobalState as GlobalTool>::Config,
536    capture_output: bool,
537) -> Result<(Output, T::GlobalState, TracerLogs), Error>
538where
539    T: Tool + 'static,
540    F: FnOnce() + Send,
541{
542    let marker = std::env::var_os(ISOLATED_WORKDIR_ENV);
543    let workdir = requested_test_workdir(marker.as_deref())
544        .map_err(|error| io::Error::new(io::ErrorKind::InvalidInput, error))?;
545    LazyLock::force(&GLOBAL_TEST_SUBSCRIBER);
546    let trace_level = test_trace_level();
547    let bufwriter = BufWriter::new();
548    let collector = tracing_subscriber::fmt()
549        .with_env_filter(trace_level)
550        .with_writer(bufwriter.clone())
551        .finish();
552
553    // Wrap function with an allocator reset for deterministic in-guest
554    // allocations. This only has an effect if the global allocator has been set
555    // to test_allocator.
556    let f = move || {
557        // The host thread mounted this run's /test before constructing the
558        // runtime and forking. Only the tracee changes its working directory.
559        if let Some(workdir) = workdir {
560            std::env::set_current_dir(workdir).unwrap_or_else(|error| {
561                panic!(
562                    "cannot enter requested test workdir {}: {error}",
563                    workdir.display()
564                )
565            });
566        }
567        test_allocator::GLOBAL
568            // Try to skip to an arbitrary fixed offset that's likely to be far
569            // past all the memory we've allocated so far.
570            .skip_to_offset_if_init(test_allocator::GLOBAL_MAX_OFFSET / 2)
571            .unwrap();
572        f()
573    };
574
575    // Here we have to keep the collector tightly scoped to this test, because
576    // we want to recapture the output of this test and no other.
577    let run = move || {
578        tracing::subscriber::with_default(collector, || {
579            let rt = tokio::runtime::Builder::new_current_thread()
580                .enable_io()
581                .build()
582                .unwrap();
583            rt.block_on(async move {
584                let tracee = spawn_fn_with_config::<T, _>(f, config, capture_output).await?;
585                tracee.wait_with_output().await
586            })
587            // Drop joins this runtime's blocking workers before the namespace
588            // thread returns. No external executor launches this tracee.
589        })
590    };
591    let (out, state) = if workdir.is_some() {
592        hermit_test_workdir::with_isolated_workdir(run)?
593    } else {
594        run()
595    }?;
596    Ok((out, state, bufwriter.get_strings()))
597}

599/// Runs a function as a guest with Detcore's test log filter applied both to
600/// the calling thread and to tracer work performed on other threads.
601pub fn test_fn_with_config<T, F>(
602    f: F,
603    config: <T::GlobalState as GlobalTool>::Config,
604    capture_output: bool,
605) -> Result<(Output, T::GlobalState), Error>
606where
607    T: Tool + 'static,
608    F: FnOnce() + Send,
609{
610    test_fn_with_logs::<T, F>(f, config, capture_output)
611        .map(|(output, state, _logs)| (output, state))
612}

619    capture_output: bool,
620) -> T::GlobalState
621where
622    T: Tool + 'static,
623    F: FnOnce() + Send,
624{

647where

654/// Like [`det_test_fn_with_config`], but runs and compares exactly `repetitions` times.
655///
656/// Each repetition includes the tracer and guest in a fresh PID namespace with
657/// matching procfs. Container maps the caller's effective uid/gid to the same
658/// numeric values. A fresh user namespace still changes the capability/group
659/// context; this is not transparent isolation for arbitrary credential tests.
660/// The oracle consumes the actual global state in that child;
661/// only the original output and complete captured logs return to the repetition
662/// driver. The general [`test_fn_with_config`] state-returning API is unchanged.
663///
664/// The oracle therefore has a process boundary: mutations of captured Rust
665/// memory do not reach the caller or later repetitions. External side effects
666/// (files, inherited descriptors, diagnostics) remain real and require their own
667/// ownership/cleanup. Callbacks must not depend on parent-memory communication
668/// or locks held by other parent threads. This inherits `spawn_fn_with_config`'s
669/// best-effort Rust-after-fork limitation; it is not a general fork-safety promise.
670pub fn det_test_fn_with_config_repetitions<F, O>(
671    repetitions: u64,
672    isdet: bool,
673    f: F,
674    config: Config,
675    oracle: O,
676) where
677    F: Fn() + Send + Sync,
678    O: Fn(&Output, <Detcore as Tool>::GlobalState),
679{
680    assert!(repetitions > 0, "at least one test repetition is required");
681    // Borrow the caller's values: only child-created configs/runtime/state are
682    // consumed in children, and the caller drops its originals after the wait.
683    run_function_test_driver(|| {
684        compare_function_repetitions(repetitions, isdet, &f, &config, &oracle);
685    });
686}

688fn compare_function_repetitions<F, O>(
689    repetitions: u64,
690    isdet: bool,
691    f: &F,
692    config: &Config,
693    oracle: &O,
694) where
695    F: Fn() + Send + Sync,
696    O: Fn(&Output, <Detcore as Tool>::GlobalState),
697{
698    if isdet {
699        println!("Expecting determinism:");
700    } else {
701        println!("Not expecting determinism, but still performing multiple test runs:");
702    }
703    let mut dts = DetTestState::default();
704    for ix in 1..repetitions {
705        println!("Test Run {}:", ix);
706        let (output, logs) = isolated_function_repetition(f, config, oracle).unwrap();
707        println!("Oracle passed.");
708        if isdet {
709            println!("Comparing against prior run, if any.");
710            check_output(&output, logs, &mut dts);
711        }
712    }
713    println!("Test Run {}:", repetitions);
714    let (output, logs) = isolated_function_repetition(f, config, oracle).unwrap();
715    println!("Oracle passed. Full stderr:");
716    println!("{}", String::from_utf8_lossy(&output.stderr));
717    println!("Full stdout:");
718    println!("{}", String::from_utf8_lossy(&output.stdout));
719    if isdet {
720        println!("Comparing against prior run, if any.");
721        check_output(&output, logs, &mut dts);
722    }
723}

725// Use libc's fork path, as spawn_fn_with_config does, before constructing a
726// Container or Tokio runtime. Libtest can have a waiting main thread even with
727// --test-threads=1; cloning a Container directly from its worker would bypass
728// libc's atfork handling. The forked driver has one surviving thread, and each
729// Container starts before that driver has created any runtime/workdir threads.
730// This does not repair arbitrary inherited Rust locks; see the public contract.
731fn run_function_test_driver(run: impl Fn()) {
732    // A concurrent generic function test may be initializing this LazyLock.
733    // Complete it in the original process, while that thread still exists;
734    // otherwise the child could inherit an in-progress Once and wait forever.
735    LazyLock::force(&GLOBAL_TEST_SUBSCRIBER);
736
737    // Match spawn_fn's handling of libtest's thread-local output capture. Child
738    // diagnostics go to the inherited streams, while the parent regains its
739    // original capture even when fork fails. Do not drop the parent's capture
740    // allocation in the child.
741    let output_capture = std::io::set_output_capture(None);
742    let pid = unsafe { libc::fork() };
743    let fork_error = (pid < 0).then(io::Error::last_os_error);
744    if pid == 0 {
745        let result = std::panic::catch_unwind(std::panic::AssertUnwindSafe(&run));
746        let status = if result.is_ok() { 0 } else { 101 };
747        // Catching the panic keeps its original hook diagnostics and prevents
748        // unwinding into the inherited libtest stack. Flush diagnostics before
749        // _exit; flushing failures also make this driver fail.
750        for result in [std::io::stdout().flush(), std::io::stderr().flush()] {
751            if let Err(error) = result {
752                eprintln!("function-test driver could not flush diagnostics: {error}");
753                unsafe { libc::_exit(101) };
754            }
755        }
756        unsafe { libc::_exit(status) };
757    }
758    std::io::set_output_capture(output_capture);
759    if let Some(error) = fork_error {
760        panic!("cannot fork function-test driver: {error}");
761    }
762
763    let mut status = 0;
764    loop {
765        let waited = unsafe { libc::waitpid(pid, &mut status, 0) };
766        if waited == pid {
767            break;
768        }
769        let error = io::Error::last_os_error();
770        if waited < 0 && error.raw_os_error() == Some(libc::EINTR) {
771            continue;
772        }
773        panic!("cannot wait for function-test driver {pid}: {error}");
774    }
775    let status = ExitStatus::from_raw(status);
776    assert_eq!(
777        status,
778        ExitStatus::Exited(0),
779        "isolated function-test driver failed; see its original diagnostics above",
780    );
781}

783// The fixed return type is the real Output and complete logs, never a stand-in
784// GlobalState. Keep the deferred result inseparable from its checked child exit,
785// even if serialization succeeds before child cleanup fails.
786fn in_function_pid_namespace<F, D>(run: F) -> Result<(Output, TracerLogs), RunError>
787where
788    F: FnMut() -> ((Output, TracerLogs), D),
789{
790    let uid = unsafe { libc::geteuid() };
791    let gid = unsafe { libc::getegid() };
792    Container::new()
793        .unshare(Namespace::PID)
794        .map_uid(uid, uid)
795        .map_gid(gid, gid)
796        .mount(Mount::proc())
797        .run_with_deferred_drop(run)?
798        .finalize()
799}
800
801fn isolated_function_repetition<F, O>(
802    f: &F,
803    config: &Config,
804    oracle: &O,
805) -> Result<(Output, TracerLogs), RunError>
806where
807    F: Fn() + Send + Sync,
808    O: Fn(&Output, <Detcore as Tool>::GlobalState),
809{
810    in_function_pid_namespace(|| {
811        // The Container clone callback is an extern-C boundary. Keep a Rust
812        // panic's diagnostics, then exit unsuccessfully instead of unwinding
813        // through that boundary or publishing a substitute successful value.
814        let result = std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| {
815            let (output, state, logs) =
816                test_fn_with_logs::<Detcore, _>(f, config.clone(), true).unwrap();
817            println!("({} log lines captured.)", logs.len());
818            oracle(&output, state);
819            (output, logs)
820        }));
821        match result {
822            Ok(result) => (result, ()),
823            Err(_panic) => unsafe { libc::_exit(101) },
824        }
825    })
826}

858/// Checks the output of a run to ensure there is no difference from the last
859/// run.
860fn check_output(output: &Output, logs: Vec<String>, dts: &mut DetTestState) {
861    dts.test_run_num += 1;
862
863    match dts.last_exit {
864        None => {}
865        Some(x) => assert_eq!(
866            x, output.status,
867            "\n  Consecutive runs of test had different exit status: {:?} {:?}",
868            x, output.status
869        ),
870    }
871    dts.last_exit = Some(output.status);
872
873    let stdout_str = String::from_utf8(output.stdout.clone()).unwrap_or_else(|_| {
874        panic!(
875            "Test produced stdout that is not valid utf8: {:?}",
876            output.stdout
877        );
878    });
879    let stderr_str = String::from_utf8(output.stderr.clone()).unwrap_or_else(|_| {
880        panic!(
881            "Test produced stderr that is not valid utf8: {:?}",
882            output.stderr
883        );
884    });
885
886    // TODO: use some kind of diffing framework to concisely print only the
887    // diffs, as well as some quantitative info about the extent of the diff.
888    match &dts.last_stdout {
889        None => {}
890        Some(x) => assert_eq!(
891            x, &stdout_str,
892            "\n  Consecutive runs of test had different stdout, run1:\n{:?}\n  Run2:\n{:?}",
893            x, &stdout_str
894        ),
895    }
896    dts.last_stdout = Some(stdout_str);
897
898    match &dts.last_stderr {
899        None => {}
900        Some(x) => assert_eq!(
901            x, &stderr_str,
902            "\n  Consecutive runs of test had different stderr, run1:\n{:?}\n  Run2:\n{:?}",
903            x, &stderr_str
904        ),
905    }
906    dts.last_stderr = Some(stderr_str);
907
908    let filtered: Vec<String> = logs
909        .iter()
910        .filter(|l| l.contains("COMMIT turn"))
911        .map(|s| {
912            let vec: Vec<&str> = s.split("COMMIT").collect();
913            if let [_pref, suffix] = vec[..] {
914                suffix.to_string()
915            } else {
916                panic!("Unexpected form to COMMIT log line: {}", s);
917            }
918        })
919        .collect();
920
921    match &dts.last_log {
922        None => {}
923        Some(x) => {
924            if x.len() != filtered.len() {
925                eprintln!(
926                    "Differing number of commit lines! ({} vs {})",
927                    x.len(),
928                    filtered.len()
929                )
930            }
931            for ix in 0..x.len().min(filtered.len()) {
932                let str_a = detcore::logdiff::strip_log_entry(&x[ix]);
933                let str_b = detcore::logdiff::strip_log_entry(&filtered[ix]);
934                assert_eq!(
935                    str_a, str_b,
936                    "\n  Consecutive runs of test had different COMMITs #{}, run1:\n{:?}\n  Run2:\n{:?}",
937                    ix, str_a, str_b,
938                )
939            }
940            if x.len() != filtered.len() {
941                panic!(
942                    "All present lines matched, but different number of COMMIT lines across two runs."
943                )
944            }
945        }
946    }
947    dts.last_log = Some(filtered);
948}
