9//! Hermit's KVM run, and the one place its Detcore instantiation is rooted.
10//!
11//! `detcore::Detcore` under `reverie_kvm`'s tool runtime is a large
12//! instantiation. In its own crate it builds in a separate rustc process,
13//! beside the other backends' instantiations. The entry point,
14//! [`run_detcore`], is not generic and returns a boxed future, so its callers
15//! instantiate nothing of it (see `hermit-record` for why it is boxed).
16//!
17//! `reverie_kvm` is an x86-64 VMM and empty elsewhere, and so is this crate.
18#![cfg(target_arch = "x86_64")]
19
20use std::future::Future;
21use std::pin::Pin;
22
23use reverie_kvm::KvmBackend;
24use reverie_kvm::ToolRunCompletion;
25
26/// The future [`run_detcore`] returns. It borrows the backend it runs on.
27pub type RunDetcore<'a> = Pin<
28    Box<dyn Future<Output = reverie_kvm::Result<ToolRunCompletion<detcore::GlobalState>>> + 'a>,
29>;
30
31/// Runs the static ELF already installed in BACKEND to completion under
32/// Detcore with CONFIG. With CAPTURE_OUTPUT the guest's stdout and stderr are
33/// returned in the completion instead of written through.
34///
35/// The future is boxed so the async body, and everything it instantiates, is
36/// compiled in this crate rather than in its caller.
37pub fn run_detcore(
38    backend: &mut KvmBackend,
39    config: detcore::Config,
40    capture_output: bool,
41) -> RunDetcore<'_> {
42    Box::pin(
43        backend.run_static_elf_with_tool_completion::<detcore::Detcore>(config, capture_output),
44    )
45}
