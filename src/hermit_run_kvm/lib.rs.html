9//! Hermit's KVM run, and the one place its Detcore instantiation is rooted.
10//!
11//! `detcore::Detcore` under `reverie_kvm`'s tool runtime is a large
12//! instantiation. In its own crate it builds in a separate rustc process,
13//! beside the other backends' instantiations. The entry point,
14//! [`run_detcore`], is not generic and returns a boxed future, so its callers
15//! instantiate nothing of it (see `hermit-record` for why it is boxed).
16//!
17//! `reverie_kvm` is an x86-64 VMM and empty elsewhere, and so is this crate.
18#![cfg(target_arch = "x86_64")]
19
20use std::future::Future;
21use std::pin::Pin;
22
23use reverie_kvm::KvmBackend;
24use reverie_kvm::ToolRunCompletion;
25
26/// The future [`run_detcore`] returns. It borrows the backend it runs on.
27pub type RunDetcore<'a> = Pin<
28    Box<dyn Future<Output = reverie_kvm::Result<ToolRunCompletion<detcore::GlobalState>>> + 'a>,
29>;
30
31/// Runs the static ELF already installed in BACKEND to completion under
32/// Detcore with CONFIG. With CAPTURE_OUTPUT the guest's stdout and stderr are
33/// returned in the completion instead of written through.
34///
35/// The future is boxed so the async body, and everything it instantiates, is
36/// compiled in this crate rather than in its caller.
37///
38/// A CONFIG that leaves the pass-through syscalls untrapped
39/// (`untrap_passthrough_syscalls`) is refused before the backend is touched,
40/// as `detcore::UntrapPassthroughRefusal::Backend("kvm")`: untrapping is the
41/// ptrace backend's alone (`detcore::untrap_passthrough_off_ptrace`). This
42/// entry is public, so it asks that rule itself (`admitted`) rather than
43/// relying on the hermit library's launch dispatchers, which ask it too.
44pub fn run_detcore(
45    backend: &mut KvmBackend,
46    config: detcore::Config,
47    capture_output: bool,
48) -> RunDetcore<'_> {
49    admitted(config, |config| {
50        Box::pin(
51            backend.run_static_elf_with_tool_completion::<detcore::Detcore>(config, capture_output),
52        )
53    })
54}
55
56/// RUN's future for CONFIG, unless KVM refuses CONFIG first: RUN, which
57/// holds the backend, is called only once CONFIG is admitted, so a refused
58/// run never reaches the backend or the guest.
59fn admitted<'a>(
60    config: detcore::Config,
61    run: impl FnOnce(detcore::Config) -> RunDetcore<'a>,
62) -> RunDetcore<'a> {
63    match detcore::untrap_passthrough_off_ptrace(&config, "kvm") {
64        Ok(()) => run(config),
65        Err(refusal) => Box::pin(std::future::ready(Err(reverie_kvm::Error::Reverie(
66            refusal.into(),
67        )))),
68    }
69}

71#[cfg(test)]
72mod tests {
73    use std::cell::Cell;
74    use std::task::Context;
75    use std::task::Poll;
76    use std::task::Waker;
77
78    use super::*;
79
80    /// The result of RUN, which must be ready at once.
81    fn ready(
82        mut run: RunDetcore<'_>,
83    ) -> reverie_kvm::Result<ToolRunCompletion<detcore::GlobalState>> {
84        match run.as_mut().poll(&mut Context::from_waker(Waker::noop())) {
85            Poll::Ready(result) => result,
86            Poll::Pending => panic!("the run was not decided at once"),
87        }
88    }
89
90    /// REFUSAL as the tool error a KVM run returns, written out.
91    fn as_tool_error<E>(_: &E, refusal: detcore::UntrapPassthroughRefusal) -> String
92    where
93        E: std::fmt::Display + From<detcore::UntrapPassthroughRefusal>,
94    {
95        E::from(refusal).to_string()
96    }
97
98    /// `run_detcore` refuses a configuration that leaves the pass-through
99    /// syscalls untrapped before its run, which holds the backend, is called,
100    /// though the configuration is otherwise admitted (RCB time, no record)
101    /// and has KVM's declared capabilities. Here the run is a stand-in that
102    /// fails the test if called, so the test needs no /dev/kvm and opens no
103    /// VM. The same configuration without the mode reaches the run, whose
104    /// result is returned unchanged.
105    #[test]
106    fn untrapping_is_refused_before_the_backend_runs() {
107        let qualified = detcore::Config {
108            untrap_passthrough_syscalls: true,
109            sequentialize_threads: true,
110            max_timeslice: std::num::NonZeroU64::new(10_000),
111            backend: KvmBackend::capabilities(),
112            ..Default::default()
113        };
114        assert_eq!(detcore::untrap_passthrough_admission(&qualified), Ok(()));
115        match ready(admitted(qualified.clone(), |_| {
116            panic!("the refused run reached the backend")
117        })) {
118            Err(reverie_kvm::Error::Reverie(error)) => assert_eq!(
119                error.to_string(),
120                as_tool_error(&error, detcore::UntrapPassthroughRefusal::Backend("kvm"))
121            ),
122            Err(other) => panic!("not the untrap refusal: {other}"),
123            Ok(_) => panic!("the untrapped calls were admitted on KVM"),
124        }
125
126        let ran = Cell::new(false);
127        let without = detcore::Config {
128            untrap_passthrough_syscalls: false,
129            ..qualified
130        };
131        match ready(admitted(without, |config| {
132            assert!(!config.untrap_passthrough_syscalls);
133            ran.set(true);
134            Box::pin(std::future::ready(Err(reverie_kvm::Error::RunAborted)))
135        })) {
136            Err(reverie_kvm::Error::RunAborted) => {}
137            _ => panic!("the admitted run's own result was not returned"),
138        }
139        assert!(ran.get(), "the admitted run was not called");
140    }
141
142    /// The public entry itself, on a real backend: `run_detcore` answers a
143    /// configuration that leaves the pass-through syscalls untrapped with the
144    /// refusal at once, before the backend runs anything. No program is
145    /// installed, so a run that reached the backend would not end with this
146    /// refusal. Requires /dev/kvm, as the backend's own unit tests do.
147    #[test]
148    fn run_detcore_refuses_untrapping_before_the_backend_runs() {
149        let mut backend = KvmBackend::new(0x10000).expect("this refusal test requires /dev/kvm");
150        let config = detcore::Config {
151            untrap_passthrough_syscalls: true,
152            sequentialize_threads: true,
153            max_timeslice: std::num::NonZeroU64::new(10_000),
154            backend: KvmBackend::capabilities(),
155            ..Default::default()
156        };
157        assert_eq!(detcore::untrap_passthrough_admission(&config), Ok(()));
158        match ready(run_detcore(&mut backend, config, false)) {
159            Err(reverie_kvm::Error::Reverie(error)) => assert_eq!(
160                error.to_string(),
161                as_tool_error(&error, detcore::UntrapPassthroughRefusal::Backend("kvm"))
162            ),
163            Err(other) => panic!("not the untrap refusal: {other}"),
164            Ok(_) => panic!("the untrapped calls were admitted on KVM"),
165        }
166    }
