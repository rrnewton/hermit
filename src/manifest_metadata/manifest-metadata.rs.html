1//! Export the current manifest policy as deterministic, typed JSON.
2
3use std::io::BufWriter;
4use std::io::Write;
5use std::path::PathBuf;
6use std::process::ExitCode;
7
8use hermit_manifest_plan::manifest_metadata::build_export;
9
10const HELP: &str = "Usage: manifest-metadata
11
12Export current E2E manifest policy as JSON.
13
14The JSON contains current test metadata, every comparable cell in the manifest,
15and custom commands selected by full validation. Each comparable cell records
16whether full validation selects it. It contains no run result or measurement
17state. Schema 2 includes both inherited or overridden manifest CPU and wall
18limits; execution-time environment multipliers are not applied to this policy.
19";
20
21fn main() -> ExitCode {
22    match real_main() {
23        Ok(()) => ExitCode::SUCCESS,
24        Err(error) => {
25            eprintln!("manifest-metadata: REFUSED: {error}");
26            ExitCode::from(2)
27        }
28    }
29}
30
31fn real_main() -> Result<(), String> {
32    let mut args = std::env::args().skip(1);
33    if let Some(arg) = args.next() {
34        if matches!(arg.as_str(), "-h" | "--help") && args.next().is_none() {
35            print!("{HELP}");
36            return Ok(());
37        }
38        return Err(format!("unexpected argument {arg:?}\n\n{HELP}"));
39    }
40
41    let root = PathBuf::from(env!("CARGO_MANIFEST_DIR"))
42        .join("../..")
43        .canonicalize()
44        .map_err(|error| format!("cannot resolve repository root: {error}"))?;
45    let export = build_export(&root)?;
46    let stdout = std::io::stdout();
47    let mut output = BufWriter::new(stdout.lock());
48    serde_json::to_writer(&mut output, &export)
49        .map_err(|error| format!("cannot encode manifest metadata: {error}"))?;
50    output
51        .write_all(b"\n")
52        .and_then(|()| output.flush())
53        .map_err(|error| format!("cannot write manifest metadata: {error}"))?;
54    Ok(())
55}
