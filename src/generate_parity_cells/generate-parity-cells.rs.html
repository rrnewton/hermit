18fn usage() -> &'static str {
19    "Usage: generate-parity-cells [--check | --write]\n\
20     Derive every cross-backend parity cell (each manifest test's verify mode on\n\
21     kvm, liteinst, in-guest-trap, sabre and dbt, against ptrace) and the parity\n\
22     population (every test whose ptrace verify cell full validation selects,\n\
23     on every candidate backend) from tests/e2e/manifests, and compare it with, or write\n\
24     it to, ci/compat-envelope/parity-cells.json.\n\
25     --check is the default. It refuses a stale snapshot and prints the counts."
26}

28fn summary(snapshot: &ParityCells) -> String {
29    let all = &snapshot.counts.all;
30    let mut text = format!(
31        "{} cells, {} applicable, {} selectable; population {} ({} candidate selected, {} \
32         candidate not selected, {} candidate disabled)",
33        all.cells,
34        all.applicable,
35        all.selectable,
36        all.population,
37        all.population_candidate_selected,
38        all.population_candidate_not_selected,
39        all.population_candidate_disabled
40    );
41    for (backend, count) in &snapshot.counts.by_backend {
42        text.push_str(&format!(
43            "\n  {backend}: {} applicable, {} selectable; population {} ({} candidate selected, \
44             {} not selected, {} disabled)",
45            count.applicable,
46            count.selectable,
47            count.population,
48            count.population_candidate_selected,
49            count.population_candidate_not_selected,
50            count.population_candidate_disabled
51        ));
52    }
53    text
54}

56fn run() -> Result<(), String> {
57    // One mode flag at most. `--check --write` used to write, while
58    // `--write --check` was refused: the same two flags gave opposite results
59    // depending on their order.
60    let mut mode: Option<&'static str> = None;
61    for arg in std::env::args().skip(1) {
62        let flag = match arg.as_str() {
63            help if is_help_flag(help) => {
64                println!("{}", usage());
65                return Ok(());
66            }
67            "--check" => "--check",
68            "--write" => "--write",
69            _ => {
70                return Err(format!("unrecognized argument {arg:?}\n{}", usage()));
71            }
72        };
73        match mode.replace(flag) {
74            None => {}
75            Some(first) if first == flag => {
76                return Err(format!("{flag} may be given only once\n{}", usage()));
77            }
78            Some(_) => {
79                return Err(format!(
80                    "--check and --write are mutually exclusive\n{}",
81                    usage()
82                ));
83            }
84        }
85    }
86    let write = mode == Some("--write");
87    let root = repo_root()?;
88    let path = root.join(PARITY_CELLS_PATH);
89    let snapshot = generate(&root)?;
90    let generated = canonical_text(&snapshot)?;
91    if write {
92        fs::write(&path, &generated)
93            .map_err(|error| format!("cannot write {}: {error}", path.display()))?;
94        println!("wrote {} ({} bytes)", path.display(), generated.len());
95    } else {
96        let committed = fs::read_to_string(&path)
97            .map_err(|error| format!("cannot read {}: {error}", path.display()))?;
98        require_fresh(&committed, &generated)?;
99        println!("{PARITY_CELLS_PATH} is canonical and fresh");
100    }
101    println!("{}", summary(&snapshot));
102    Ok(())
103}
104
105fn main() -> ExitCode {
106    match run() {
107        Ok(()) => ExitCode::SUCCESS,
108        Err(error) => {
109            eprintln!("generate-parity-cells: {error}");
110            ExitCode::from(2)
111        }
112    }
113}
