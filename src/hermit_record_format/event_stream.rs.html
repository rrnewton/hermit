30/// Stable event-stream identity derived from the guest process tree.
31///
32/// Each component after the root is the parent's deterministic child ordinal.
33/// Unlike a process-local counter, this remains collision-free when Reverie
34/// forks the tool itself and each process receives its own tool instance.
35#[derive(Clone, Debug, Default, Eq, PartialEq, Serialize, Deserialize)]
36pub struct EventStreamId(Vec<u64>);
37
38impl EventStreamId {
39    pub fn root() -> Self {
40        Self(Vec::new())
41    }
42
43    pub fn child(&self, ordinal: u64) -> Self {
44        let mut path = self.0.clone();
45        path.push(ordinal);
46        Self(path)
47    }
48
49    fn file_name(&self) -> String {
50        let mut identity = b"hermit-event-stream-v1\0".to_vec();
51        for ordinal in &self.0 {
52            identity.extend_from_slice(&ordinal.to_be_bytes());
53        }
54        format!("stream-{}", detcore::Digest::new(&identity))
55    }
56}
57
58impl fmt::Display for EventStreamId {
59    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
60        write!(f, "{}", detcore::ROOT_DETPID.as_raw())?;
61        for ordinal in &self.0 {
62            write!(f, ".{ordinal}")?;
63        }
64        Ok(())
65    }
66}

68/// Allocates child ordinals within one deterministic parent stream.
69///
70/// The counter belongs to the parent thread state rather than the Recorder or
71/// Replayer process instance. When Reverie forks a tool process, the child
72/// receives its already-assigned stream identity and a fresh descendant
73/// counter, while the continuing parent retains the incremented counter.
74#[derive(Default)]
75pub struct ChildEventStreamIds(AtomicU64);
76
77impl ChildEventStreamIds {
78    pub fn next(&self, parent: &EventStreamId) -> EventStreamId {
79        let ordinal = self
80            .0
81            .try_update(Ordering::Relaxed, Ordering::Relaxed, |value| {
82                value.checked_add(1)
83            })
84            .unwrap_or_else(|_| panic!("record/replay child stream ordinal overflow"));
85        parent.child(ordinal)
86    }
87}

89#[derive(Serialize, Deserialize)]
90struct EventStreamHeader {
91    stream_id: EventStreamId,
92}
93
94fn decode_stream_header(
95    reader: &mut io::BufReader<fs::File>,
96    expected: &EventStreamId,
97) -> io::Result<()> {
98    let header: EventStreamHeader =
99        bincode::serde::decode_from_std_read(reader, bincode::config::legacy()).map_err(
100            |error| {
101                io::Error::new(
102                    io::ErrorKind::InvalidData,
103                    format!("invalid event-stream identity header: {error}"),
104                )
105            },
106        )?;
107    if header.stream_id != *expected {
108        return Err(io::Error::new(
109            io::ErrorKind::InvalidData,
110            format!(
111                "event-stream identity mismatch: expected {expected}, found {}",
112                header.stream_id
113            ),
114        ));
115    }
116    Ok(())
117}
118
119fn encode_stream_header(
120    writer: &mut io::BufWriter<fs::File>,
121    stream_id: &EventStreamId,
122) -> io::Result<()> {
123    bincode::serde::encode_into_std_write(
124        EventStreamHeader {
125            stream_id: stream_id.clone(),
126        },
127        writer,
128        bincode::config::legacy(),
129    )
130    .map(|_| ())
131    .map_err(|error| {
132        io::Error::new(
133            io::ErrorKind::InvalidData,
134            format!("failed to encode event-stream identity header: {error}"),
135        )
136    })
137}

139impl Serialize for ChildEventStreamIds {
140    fn serialize<S>(&self, serializer: S) -> Result<S::Ok, S::Error>
141    where
142        S: serde::Serializer,
143    {
144        serializer.serialize_u64(self.0.load(Ordering::Relaxed))
145    }
146}

148impl<'de> Deserialize<'de> for ChildEventStreamIds {
149    fn deserialize<D>(deserializer: D) -> Result<Self, D::Error>
150    where
151        D: serde::Deserializer<'de>,
152    {
153        Ok(Self(AtomicU64::new(u64::deserialize(deserializer)?)))
154    }

157/// An event to help with debugging, but is not actually necessary for the
158/// functionality of record/replay.
159#[derive(Debug, Serialize, Deserialize)]
160pub struct DebugEvent {
161    /// The raw syscall.
162    syscall: (Sysno, SyscallArgs),
163
164    /// The pretty, displayable version of the syscall.
165    pretty: String,
166
167    /// Exec pathname bytes and lookup context. Raw syscall registers only hold
168    /// a pointer, so this snapshot prevents a changed failed exec request from
169    /// silently consuming an event that happened to return the same errno.
170    exec_request: Option<DebugExecRequest>,
171}
172
173#[derive(Debug, Eq, PartialEq, Serialize, Deserialize)]
174struct DebugExecRequest {
175    dirfd: libc::c_int,
176    path: Result<Vec<u8>, Errno>,
177    flags: libc::c_int,
178}
179
180impl DebugEvent {
181    /// Constructs a new `DebugEvent`.
182    pub fn new<M: MemoryAccess>(syscall: Syscall, memory: &M) -> Self {
183        let exec_request = match syscall {
184            Syscall::Execve(call) => {
185                let call: reverie::syscalls::Execveat = call.into();
186                Some(DebugExecRequest {
187                    dirfd: call.dirfd(),
188                    path: call
189                        .path()
190                        .ok_or(Errno::EFAULT)
191                        .and_then(|path| path.read(memory))
192                        .map(|path| path.as_os_str().as_bytes().to_vec()),
193                    flags: call.flags(),
194                })
195            }
196            Syscall::Execveat(call) => Some(DebugExecRequest {
197                dirfd: call.dirfd(),
198                path: call
199                    .path()
200                    .ok_or(Errno::EFAULT)
201                    .and_then(|path| path.read(memory))
202                    .map(|path| path.as_os_str().as_bytes().to_vec()),
203                flags: call.flags(),
204            }),
205            _ => None,
206        };
207        Self {
208            syscall: syscall.into_parts(),
209            pretty: format!("{}", syscall.display(memory)),
210            exec_request,
211        }
212    }
213
214    /// Returns the syscall associated with this debug event.
215    pub fn syscall(&self) -> Syscall {
216        Syscall::from_raw(self.syscall.0, self.syscall.1)
217    }
218
219    pub fn exec_request_matches(&self, other: &Self) -> bool {
220        self.exec_request == other.exec_request
221    }
222
223    /// A debug event naming SYSNO with no arguments. Test support only, for the
224    /// tests here and in the hermit library.
225    #[doc(hidden)]
226    pub fn for_test(sysno: Sysno, pretty: &str) -> Self {
227        Self {
228            syscall: (sysno, SyscallArgs::new(0, 0, 0, 0, 0, 0)),
229            pretty: pretty.to_owned(),
230            exec_request: None,
231        }
232    }
233}

235impl fmt::Display for DebugEvent {
236    fn fmt(&self, f: &mut fmt::Formatter) -> fmt::Result {
237        f.write_str(&self.pretty)
238    }

241/// The number of argument registers the x86-64 kernel actually reads for
242/// `sysno`. Registers beyond this are not part of the syscall's ABI and can hold
243/// arbitrary leftover values in the guest, so they must be excluded when
244/// comparing a replayed syscall against its recorded counterpart.
245///
246/// Reverie stores all six raw registers in every typed syscall and derives
247/// `PartialEq` over them, so a naive `Syscall == Syscall` compares unused
248/// registers too. That produces false desyncs for any syscall with fewer than
249/// six arguments (e.g. `statfs`, which uses two).
250///
251/// Returns `None` for syscalls without an entry; callers then fall back to
252/// comparing all six registers (the conservative pre-existing behavior). Only
253/// syscalls that record/replay subscribes to (see [`crate::subscriptions::record_and_replay`])
254/// flow through the comparator, so those are the ones covered here.
255///
256/// These are true kernel arities, which for a few syscalls exceed the number of
257/// typed fields reverie exposes (`open`, `openat`, `ioctl`, `socket`, and
258/// `fcntl` fold meaningful arguments into typed enums). Using the kernel arity
259/// -- not the typed field count -- guarantees we never zero a meaningful
260/// argument and therefore never mask a genuine divergence.
261fn kernel_arg_count(sysno: Sysno) -> Option<u8> {
262    // Every name is qualified: a bare name that the target's table lacks
263    // would be a catch-all binding, not a pattern.
264    Some(match sysno {
265        Sysno::close | Sysno::chdir | Sysno::fchdir | Sysno::dup => 1,
266        Sysno::getcwd
267        | Sysno::set_robust_list
268        | Sysno::fstat
269        | Sysno::clock_gettime
270        | Sysno::clock_getres
271        | Sysno::gettimeofday
272        | Sysno::settimeofday
273        | Sysno::statfs
274        | Sysno::fstatfs
275        | Sysno::ftruncate
276        | Sysno::kill
277        | Sysno::listen
278        | Sysno::shutdown
279        | Sysno::rt_sigpending => 2,
280        Sysno::mprotect
281        | Sysno::read
282        | Sysno::readv
283        | Sysno::write
284        | Sysno::writev
285        | Sysno::lseek
286        | Sysno::getdents64
287        | Sysno::dup3
288        | Sysno::ioctl
289        | Sysno::socket
290        | Sysno::fcntl
291        | Sysno::connect
292        | Sysno::sendmsg
293        | Sysno::getpeername
294        | Sysno::getsockname
295        | Sysno::getrandom
296        | Sysno::unlinkat
297        | Sysno::execve
298        | Sysno::close_range
299        | Sysno::tgkill
300        | Sysno::faccessat
301        | Sysno::accept
302        | Sysno::bind => 3,
303        Sysno::pread64
304        | Sysno::pwrite64
305        | Sysno::fadvise64
306        | Sysno::openat
307        | Sysno::faccessat2
308        | Sysno::accept4 => 4,
309        Sysno::statx
310        | Sysno::pwritev
311        | Sysno::preadv
312        | Sysno::ppoll
313        | Sysno::setsockopt
314        | Sysno::getsockopt
315        | Sysno::execveat
316        | Sysno::prctl => 5,
317        Sysno::recvfrom | Sysno::sendto | Sysno::pwritev2 | Sysno::preadv2 | Sysno::mmap => 6,
318        #[cfg(target_arch = "x86_64")]
319        Sysno::time | Sysno::unlink | Sysno::rmdir => 1,
320        #[cfg(target_arch = "x86_64")]
321        Sysno::access
322        | Sysno::stat
323        | Sysno::lstat
324        | Sysno::dup2
325        | Sysno::mkdir
326        | Sysno::rename
327        | Sysno::link
328        | Sysno::symlink
329        | Sysno::chmod => 2,
330        #[cfg(target_arch = "x86_64")]
331        Sysno::getdents
332        | Sysno::poll
333        | Sysno::readlink
334        | Sysno::open
335        | Sysno::chown
336        | Sysno::lchown
337        | Sysno::mknod => 3,
338        #[cfg(target_arch = "x86_64")]
339        Sysno::newfstatat => 4,
340        #[cfg(target_arch = "aarch64")]
341        Sysno::fstatat => 4,
342        _ => return None,
343    })
344}

346/// Returns `syscall` with any argument registers beyond its kernel arity zeroed.
347/// This makes the record/replay desync comparison ignore unused registers, which
348/// are not part of the syscall and may legitimately differ between record and
349/// replay. Syscalls without a known arity are returned unchanged (all six
350/// registers still compared).
351pub fn normalize_unused_args(syscall: Syscall) -> Syscall {
352    let (sysno, args) = syscall.into_parts();
353    let Some(used) = kernel_arg_count(sysno) else {
354        return syscall;
355    };
356    let mut raw = [
357        args.arg0, args.arg1, args.arg2, args.arg3, args.arg4, args.arg5,
358    ];
359    for reg in raw.iter_mut().skip(usize::from(used)) {
360        *reg = 0;
361    }
362    Syscall::from_raw(
363        sysno,
364        SyscallArgs::new(raw[0], raw[1], raw[2], raw[3], raw[4], raw[5]),
365    )
366}

368/// A stream of syscall events.
369#[derive(Serialize, Deserialize)]
370pub struct EventReader {
371    // The file where events are stored.
372    //
373    // NOTE: This field isn't serializable/deserializable, so we have to skip it
374    // for now. With an in-guest backend, we'd need to implement this manually
375    // to support state migration.
376    #[serde(skip, default = "default_reader")]
377    reader: io::BufReader<fs::File>,
378
379    // The file where raw syscalls are stored. This is used for detecting
380    // desynchronization bugs. This is stored in a separate file so that we can
381    // easily turn this on or off to shift the balance on debuggability and
382    // performance.
383    #[serde(skip, default = "default_reader")]
384    debug_events: io::BufReader<fs::File>,
385
386    // The number of events read so far. Useful for debugging purposes.
387    pub count: u64,
388}
389
390fn default_reader() -> io::BufReader<fs::File> {
391    unimplemented!("Serialization is not yet implemented")
392}
393
394impl EventReader {
395    /// Opens an existing event stream.
396    pub fn open(path: &Path, stream_id: &EventStreamId) -> io::Result<Self> {
397        let file_name = stream_id.file_name();
398        let mut reader = io::BufReader::new(fs::File::open(path.join("thread").join(&file_name))?);
399        let mut debug_events = io::BufReader::new(fs::File::open(
400            path.join("thread").join(format!("{file_name}.debug")),
401        )?);
402        decode_stream_header(&mut reader, stream_id)?;
403        decode_stream_header(&mut debug_events, stream_id)?;
404        Ok(Self {
405            reader,
406            debug_events,
407            count: 0,
408        })
409    }
410
411    /// Reads the next event from the stream. Returns an error if there are no
412    /// more events to consume.
413    pub fn next_event(&mut self) -> Result<Event, bincode::error::DecodeError> {
414        bincode::serde::decode_from_std_read(&mut self.reader, bincode::config::legacy())
415    }
416
417    /// Reads the next syscall from the syscall stream.
418    pub fn next_debug_event(&mut self) -> Result<DebugEvent, bincode::error::DecodeError> {
419        let debug_event = bincode::serde::decode_from_std_read(
420            &mut self.debug_events,
421            bincode::config::legacy(),
422        )?;
423        self.count += 1;
424        Ok(debug_event)
425    }
426}
427
428impl Default for EventReader {
429    fn default() -> Self {
430        panic!("Thread state should be explicitly initialized in init_thread_state")
431    }
432}

434/// A stream of syscall events.
435#[derive(Serialize, Deserialize)]
436pub struct EventWriter {
437    // The file where events are stored.
438    //
439    // NOTE: This field isn't serializable/deserializable, so we have to skip it
440    // for now. With an in-guest backend, we'd need to implement this manually
441    // to support state migration.
442    #[serde(skip, default = "default_writer")]
443    writer: io::BufWriter<fs::File>,
444
445    // The file where syscalls are stored. This is used for debugging purposes.
446    #[serde(skip, default = "default_writer")]
447    debug_events: io::BufWriter<fs::File>,
448}
449
450fn default_writer() -> io::BufWriter<fs::File> {
451    unimplemented!("Serialization is not yet implemented")
452}
453
454impl EventWriter {
455    /// Creates a new event stream.
456    pub fn create(path: &Path, stream_id: &EventStreamId) -> io::Result<Self> {
457        let path = path.join("thread");
458
459        fs::create_dir_all(&path)?;
460        let file_name = stream_id.file_name();
461
462        let create = |path| {
463            fs::OpenOptions::new()
464                .write(true)
465                .create_new(true)
466                .open(path)
467        };
468
469        let mut writer = io::BufWriter::new(create(path.join(&file_name))?);
470        let mut debug_events = io::BufWriter::new(create(path.join(format!("{file_name}.debug")))?);
471        encode_stream_header(&mut writer, stream_id)?;
472        encode_stream_header(&mut debug_events, stream_id)?;
473        Ok(Self {
474            writer,
475            debug_events,
476        })
477    }
478
479    /// Writes an event to the end of the stream.
480    pub fn push_event(&mut self, event: Event) -> Result<(), bincode::error::EncodeError> {
481        bincode::serde::encode_into_std_write(&event, &mut self.writer, bincode::config::legacy())
482            .map(|_| ())
483    }
484
485    /// Writes a debug event to the end of the stream.
486    pub fn push_debug_event(
487        &mut self,
488        event: DebugEvent,
489    ) -> Result<(), bincode::error::EncodeError> {
490        bincode::serde::encode_into_std_write(
491            &event,
492            &mut self.debug_events,
493            bincode::config::legacy(),
494        )
495        .map(|_| ())
496    }
497
498    /// Writes out what both streams still buffer and closes their files,
499    /// reporting the first error. Dropping the writer instead would flush and
500    /// close them too, but would ignore every error, so a stream cut short by a
501    /// full disk (ENOSPC), a file-size limit or an I/O error at its last write,
502    /// or by an error close(2) reports for writes it could not complete, went
503    /// unnoticed until replay failed on it.
504    pub fn finish(self) -> io::Result<()> {
505        let Self {
506            writer,
507            debug_events,
508        } = self;
509        let events = finish_file(writer);
510        let debug_events = finish_file(debug_events);
511        events.and(debug_events)
512    }
513
514    /// An event writer over `events` and `debug_events`, with no stream
515    /// header. Test support only, for the tests here and in the recorder.
516    #[doc(hidden)]
517    pub fn for_test(events: fs::File, debug_events: fs::File) -> Self {
518        Self {
519            writer: io::BufWriter::new(events),
520            debug_events: io::BufWriter::new(debug_events),
521        }
522    }
523}
524
525/// Flushes `writer` and closes its file, reporting an error from either. A
526/// failed flush leaves the rest unwritten: the writer is taken apart rather
527/// than dropped, since its drop would write the rest again and could leave a
528/// complete file behind a reported error.
529fn finish_file(writer: io::BufWriter<fs::File>) -> io::Result<()> {
530    use std::os::fd::IntoRawFd as _;
531    let file = match writer.into_inner() {
532        Ok(file) => file,
533        Err(error) => {
534            let (error, writer) = error.into_parts();
535            let (_file, _unwritten) = writer.into_parts();
536            return Err(error);
537        }
538    };
539    let fd = file.into_raw_fd();
540    // SAFETY: `fd` came from `into_raw_fd`, so it is open and nothing else
541    // owns it. Linux releases the descriptor before it flushes the file, so it
542    // is released even when close fails, and an EINTR must not be retried
543    // (`file_close_fd` before `filp_flush` in fs/open.c).
544    if unsafe { libc::close(fd) } == 0 {
545        Ok(())
546    } else {
547        Err(io::Error::last_os_error())
548    }
549}

551impl Default for EventWriter {
552    fn default() -> Self {
553        panic!("Thread state should be explicitly initialized in init_thread_state")
554    }
