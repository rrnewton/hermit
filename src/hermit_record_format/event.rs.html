26/// Returns the stable error used for legacy ioctls whose nested output cannot
27/// be represented by the currently pinned Reverie decoder.
28pub fn deterministic_ioctl_error(request: &ioctl::Request<'_>) -> Option<Errno> {
29    match request {
30        // SIOCETHTOOL stores its output behind the data pointer nested in an
31        // ifreq. Treating it as an opaque request would lose guest-visible
32        // memory updates, so reject it identically during record and replay.
33        ioctl::Request::SIOCETHTOOL(_) | ioctl::Request::Other(SIOCETHTOOL, _) => {
34            Some(Errno::ENODEV)
35        }
36        _ => None,
37    }
38}

41/// Returns the output buffer and its size for an xattr query (`getxattr`,
42/// `lgetxattr`, `fgetxattr` and the `listxattr` family), or `None` for any
43/// other syscall. A zero size only asks for the length the value needs: the
44/// kernel copies nothing out and returns that length.
45pub fn xattr_query_output(syscall: &Syscall) -> Option<(Option<AddrMut<'_, u8>>, usize)> {
46    match syscall {
47        Syscall::Getxattr(call) => Some((call.value().map(|a| a.cast()), call.size())),
48        Syscall::Lgetxattr(call) => Some((call.value().map(|a| a.cast()), call.size())),
49        Syscall::Fgetxattr(call) => Some((call.value().map(|a| a.cast()), call.size())),
50        Syscall::Listxattr(call) => Some((call.list().map(|a| a.cast()), call.size())),
51        Syscall::Llistxattr(call) => Some((call.list().map(|a| a.cast()), call.size())),
52        Syscall::Flistxattr(call) => Some((call.list().map(|a| a.cast()), call.size())),
53        _ => None,
54    }
55}

57/// Returns the output buffer and its size for an `fcntl` command that writes a
58/// result through its argument instead of returning it: `F_GETOWN_EX`, the
59/// record-lock queries `F_GETLK`, `F_GETLK64` and `F_OFD_GETLK`, and the
60/// write-hint queries. Recording keeps those bytes, and replay writes them
61/// back; returning only the recorded status would leave the guest's struct as
62/// it was (<https://github.com/rrnewton/hermit/issues/2289>). glibc implements
63/// `fcntl(F_GETOWN)` with `F_GETOWN_EX`, so this also covers `F_GETOWN`.
64pub fn fcntl_output(
65    call: &reverie::syscalls::Fcntl,
66) -> Option<(Option<AddrMut<'static, u8>>, usize)> {
67    use reverie::syscalls::FcntlCmd;
68    let out = |addr: Option<usize>, len: usize| Some((addr.and_then(AddrMut::from_raw), len));
69    match call.cmd() {
70        // `struct f_owner_ex { int type; pid_t pid; }`.
71        FcntlCmd::F_GETOWN_EX(addr) => out(
72            addr.map(|a| a.as_raw()),
73            std::mem::size_of::<[libc::c_int; 2]>(),
74        ),
75        FcntlCmd::F_GETLK(addr) | FcntlCmd::F_OFD_GETLK(addr) => {
76            out(addr.map(|a| a.as_raw()), std::mem::size_of::<libc::flock>())
77        }
78        FcntlCmd::F_GETLK64(addr) => out(
79            addr.map(|a| a.as_raw()),
80            std::mem::size_of::<libc::flock64>(),
81        ),
82        FcntlCmd::F_GET_RW_HINT(addr) | FcntlCmd::F_GET_FILE_RW_HINT(addr) => {
83            out(addr.map(|a| a.as_raw()), std::mem::size_of::<u64>())
84        }
85        _ => None,
86    }
87}

89/// An event. This contains everything needed to verify and reproduce the
90/// execution of a syscall.
91#[derive(Debug, Serialize, Deserialize)]
92pub struct Event {
93    /// The event that we use to reconstruct the outputs of the original syscall.
94    /// This is `Some` if need to record this syscall. If the syscall is already
95    /// deterministic, then this is `None`.
96    ///
97    /// If a recorded syscall failed, then this is `Some(Err(Errno))`. That is,
98    /// the failure should be reproduced during replay.
99    pub event: Result<SyscallEvent, Errno>,
100}
101
102/// A `SyscallEvent` contains all the necessary information to replay a system
103/// call.
104///
105/// Note that we only need a small amount of information to replay a syscall. The
106/// only side effects observable by the user are:
107///  1. Mutable pointers
108///  2. Return values.
109///
110/// No registers are modified by the kernel except for `rax` (the return value).
111/// Therefore, registers themselves do not need to be recorded since they are
112/// strictly inputs. However, any arguments that are pointers that point to
113/// mutable data expected to be modified by the kernel need to be recorded. If
114/// this rule is applied uniformly for all syscalls, then we should be able to
115/// implement full record and replay.
116#[derive(Debug, Serialize, Deserialize)]
117pub enum SyscallEvent {
118    Bytes(Vec<u8>),
119    /// The flattened output bytes of a vectored read (`readv`/`preadv`/
120    /// `preadv2`). The bytes are stored contiguously in read order; on replay
121    /// they are scattered back across the guest's `iovec` buffers. The length of
122    /// the vector is exactly the return value of the syscall.
123    Readv(Vec<u8>),
124    Write(i64),
125    Mmap(MmapEvent),
126    Recvmsg(RecvmsgEvent),
127    /// A syscall whose only value we care about is the return value. For many
128    /// syscalls, this is often the only output of the syscall and thus it is the
129    /// only piece of information that needs to be recorded.
130    Return(i64),
131    Stat(StatEvent),
132    Statfs(Vec<u8>),
133    Statx(StatxBuf),
134    /// x86 only: aarch64 has no `rdtsc`, so its recordings number every later
135    /// variant one lower, and a recording made on one architecture does not
136    /// decode on the other. Recordings do not yet carry their architecture, so
137    /// nothing refuses such a replay by name.
138    #[cfg(target_arch = "x86_64")]
139    Rdtsc(RdtscResult),
140    Ioctl(ioctl::Output),
141    Timespec(TimespecEvent),
142    Timeofday((Timeval, Timezone)),
143    Poll(PollEvent),
144    SockOpt(SockOptEvent),
145    EpollWait(EpollWaitEvent),
146    // AUTONOMOUS-BOT-IMPLEMENTED
147    // TODO-HUMAN-REVIEW(#662): Audit the replay filesystem event schema.
148    Open(OpenEvent),
149    /// Result of `mkdir`/`mkdirat` plus the physical side effect replay must
150    /// reconstruct when the recorded call found an existing directory.
151    Mkdir(MkdirEvent),
152    // TODO-HUMAN-REVIEW(#557): Audit the V2 record/replay event schema.
153    WriteV2(WriteEvent),
154    ReadV2(ReadEvent),
155    ReadvV2(ReadEvent),
156    FtruncateV2(FtruncateEvent),
157    /// Destination length after a successful clone ioctl.
158    FileClone(FileCloneEvent),
159    /// A successful exec and the executable images required immediately before
160    /// replaying it. Failed execs use the enclosing [`Event::event`] error.
161    Exec(ExecEvent),
162    /// The result and mutable output fields of a raw `ppoll` call.
163    Ppoll(PpollEvent),
164    /// The result and mutable output fields of a raw `select` or `pselect6`
165    /// call.
166    Select(SelectEvent),
167    /// The result of a guest-semantic `madvise` and the file-backed contents
168    /// replay must restore over its anonymous replacement mappings.
169    Madvise(MadviseEvent),
170    /// The outcome and peer-address copy-out of an `accept4` that the kernel
171    /// completed.
172    Accept(AcceptEvent),
173    /// What a `sendfile` copied and where its input stopped, with its output
174    /// side recorded as a write's is
175    /// (<https://github.com/rrnewton/hermit/issues/4416>).
176    Sendfile(SendfileEvent),
177}

483/// A `poll` that a signal ended in the kernel with `ERESTART_RESTARTBLOCK`: its
484/// pollfd array and length. With no handler to run, Linux restarts such a poll
485/// through the guest's own `restart_syscall` (`do_restart_poll`), which writes
486/// the array's `revents` and returns as the poll would. The recorder runs that
487/// `restart_syscall` in place and records its outputs as a [`PollEvent`];
488/// replay serves them back. Each thread keeps one until its next syscall.
489#[derive(Clone, Copy, Debug, Eq, PartialEq, Serialize, Deserialize)]
490pub struct InterruptedPoll {
491    /// The guest's pollfd array address, 0 for none.
492    pub fds: u64,
493    /// Its length.
494    pub nfds: u64,
495}
496
497impl InterruptedPoll {
498    /// The poll `fds`/`nfds` ended with `result`, if a signal ended it in the
499    /// kernel with `ERESTART_RESTARTBLOCK`.
500    pub fn after(
501        fds: Option<AddrMut<'_, PollFd>>,
502        nfds: u64,
503        result: &Result<i64, Errno>,
504    ) -> Option<Self> {
505        matches!(result, Err(Errno::ERESTART_RESTARTBLOCK)).then(|| Self {
506            fds: fds.map_or(0, |fds| fds.as_raw() as u64),
507            nfds,
508        })
509    }
510
511    /// Whether `syscall` is the kernel's restart of this poll: a
512    /// `restart_syscall` whose argument registers still hold the poll's
513    /// array and length, as Linux leaves them when it restarts a call.
514    pub fn restarted_by(&self, syscall: &Syscall) -> bool {
515        use reverie::syscalls::SyscallInfo;
516        let (number, args) = syscall.into_parts();
517        number == reverie::syscalls::Sysno::restart_syscall
518            && args.arg0 as u64 == self.fds
519            && args.arg1 as u64 == self.nfds
520    }
521
522    /// The pollfd array address.
523    pub fn fds_address(&self) -> Option<AddrMut<'static, PollFd>> {
524        AddrMut::from_raw(self.fds as usize)
525    }
526}

528/// Records every guest-visible output of a raw `ppoll` call.
529///
530/// Unlike `poll`, Linux treats the timeout as an in-out parameter. The pollfd
531/// and timeout copyouts are independent of the return value. Linux writes
532/// `revents` before attempting the remaining-time copyout and preserves the
533/// syscall result when that later write faults. Keep both output snapshots
534/// alongside the nested result so replay can restore the writes in kernel order
535/// before returning either success or the error.
536#[derive(Serialize, Deserialize, Debug)]
537pub struct PpollEvent {
538    /// The exact return value or errno observed while recording.
539    pub result: Result<i64, Errno>,
540
541    /// Whether the guest supplied a non-null pollfd pointer.
542    pub fds_pointer_present: bool,
543
544    /// Complete output on success or the readable output prefix on EFAULT.
545    pub fds: Option<Vec<PollFd>>,
546
547    /// Whether the guest supplied a non-null raw timeout pointer.
548    pub timeout_pointer_present: bool,
549
550    /// The exact remaining timeout after the kernel call. Raw `ppoll` mutates
551    /// this continuously; replay must restore the captured value without
552    /// rounding, freezing, or synthesizing it.
553    pub timeout: Option<Timespec>,
554}

556/// Records every guest-visible output of a raw `select` or `pselect6` call.
557///
558/// Linux copies the three descriptor sets out only when the call succeeds, or
559/// partially before an EFAULT. It may then write the remaining time into a
560/// non-null timeout (a `struct timeval` for `select`, a `struct timespec` for
561/// `pselect6`); the recorder keeps the post-call bytes whether or not it did.
562/// Replay restores the writes in that order before returning the recorded
563/// result.
564#[derive(Serialize, Deserialize, Debug)]
565pub struct SelectEvent {
566    /// The exact return value or errno observed while recording.
567    pub result: Result<i64, Errno>,
568
569    /// Post-kernel bytes of the read, write and exception sets, in argument
570    /// order. Each holds the readable prefix of the set's `FDS_BYTES(nfds)`
571    /// range, with `nfds` clamped to the guest's descriptor table, or `None`
572    /// when the pointer was null or the kernel did not copy the sets out.
573    pub fd_sets: [Option<Vec<u8>>; 3],
574
575    /// Post-kernel bytes of the timeout, when its pointer was non-null and
576    /// readable. Replay restores them exactly, without rounding or freezing.
577    pub timeout: Option<Vec<u8>>,
578}
579
580/// Bytes Linux copies in and out for each descriptor set of `nfds` bits:
581/// whole `long`s, as `FDS_BYTES` in fs/select.c.
582pub fn fd_set_bytes(nfds: i32) -> usize {
583    const LONG_BYTES: usize = std::mem::size_of::<libc::c_long>();
584    usize::try_from(nfds).map_or(0, |nfds| nfds.div_ceil(8 * LONG_BYTES) * LONG_BYTES)
585}

611    /// Only a poll a signal ended with `ERESTART_RESTARTBLOCK` is kept, and
612    /// only a `restart_syscall` that still names its array and length is its
613    /// restart.
614    #[test]
615    fn only_a_restart_naming_an_interrupted_poll_restarts_it() {
616        use reverie::syscalls::SyscallArgs;
617        use reverie::syscalls::Sysno;
618        let fds = AddrMut::<PollFd>::from_raw(0x7000);
619        assert_eq!(InterruptedPoll::after(fds, 2, &Ok(1)), None);
620        assert_eq!(InterruptedPoll::after(fds, 2, &Err(Errno::EINTR)), None);
621        let poll = InterruptedPoll::after(fds, 2, &Err(Errno::ERESTART_RESTARTBLOCK))
622            .expect("an interrupted poll is kept");
623        assert_eq!(
624            poll,
625            InterruptedPoll {
626                fds: 0x7000,
627                nfds: 2
628            }
629        );
630        assert_eq!(poll.fds_address().map(|fds| fds.as_raw()), Some(0x7000));
631        let call = |number, fds, nfds| {
632            Syscall::from_raw(number, SyscallArgs::new(fds, nfds, 300, 0, 0, 0))
633        };
634        assert!(poll.restarted_by(&call(Sysno::restart_syscall, 0x7000, 2)));
635        assert!(!poll.restarted_by(&call(Sysno::restart_syscall, 0x7000, 3)));
636        assert!(!poll.restarted_by(&call(Sysno::restart_syscall, 0x8000, 2)));
637        assert!(!poll.restarted_by(&call(Sysno::poll, 0x7000, 2)));
638        // A poll of no array keeps 0 for it.
639        let none = InterruptedPoll::after(None, 0, &Err(Errno::ERESTART_RESTARTBLOCK))
640            .expect("an interrupted poll is kept");
641        assert!(none.restarted_by(&call(Sysno::restart_syscall, 0, 0)));
642        assert_eq!(none.fds_address(), None);
643    }
