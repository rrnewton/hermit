9//! Syscall-argument decoding that both recording and replay read the same way.
10
11/// The iovecs a guest `msghdr` points at, read from guest MEMORY.
12pub fn read_iovecs<M: reverie::syscalls::MemoryAccess>(
13    memory: &M,
14    message: &libc::msghdr,
15) -> Result<Vec<libc::iovec>, reverie::Errno> {
16    if message.msg_iovlen == 0 {
17        return Ok(Vec::new());
18    }
19    let address = reverie::syscalls::Addr::from_raw(message.msg_iov as usize)
20        .ok_or(reverie::Errno::EFAULT)?;
21    let mut iovecs = vec![
22        libc::iovec {
23            iov_base: std::ptr::null_mut(),
24            iov_len: 0,
25        };
26        message.msg_iovlen
27    ];
28    memory.read_values(address, &mut iovecs)?;
29    Ok(iovecs)
30}

32/// The file offset of a vectored positional syscall (`preadv`, `pwritev`, ...)
33/// from its split LOW and HIGH arguments.
34pub fn vectored_offset(low: u64, high: u64) -> i64 {
35    if std::mem::size_of::<usize>() == 8 {
36        low as i64
37    } else {
38        ((high << 32) | (low & u32::MAX as u64)) as i64
39    }
40}
