1//! In-guest Detcore runtime for the LiteInst backend.
2//!
3//! Hermit preloads this library into the guest when `HERMIT_LITEINST_IN_GUEST=1`
4//! selects LiteInst's in-guest runtime. When the dynamic loader loads it, its
5//! constructor installs Detcore's `Tool` inside the guest process before the
6//! program's `main`. The loader, the C library's initialization, `IFUNC`
7//! resolvers and the constructors of the program's own shared libraries run
8//! earlier, unmonitored. A loader that fails to load the library runs the
9//! guest without it; Hermit reports that the guest never connected only after
10//! the guest exits. Once Detcore is installed, patched system calls and the
11//! SIGSYS fallback dispatch to Detcore in the guest, and Detcore's global
12//! state stays in Hermit, reached over the coordinator RPC socket that
13//! `reverie_liteinst::LiteinstBackend::run_with_preload` creates. No ptrace
14//! tracer sits on the system-call path.
15#![deny(missing_docs)]
16
17// AUTONOMOUS-BOT-IMPLEMENTED
18// TODO-HUMAN-REVIEW(PR-3635): Review the
19// in-guest Detcore constructor boundary.

21/// Hermit validates that this constructor is registered in `.init_array`
22/// before it preloads the library, so a DSO that would load without
23/// installing Detcore is refused instead of running the guest unmonitored.
24#[used]
25#[unsafe(link_section = ".init_array")]
26static DETCORE_LITEINST_INIT: unsafe extern "C" fn() = detcore_liteinst_initialize;

28/// Installs Detcore in the current guest process.
29///
30/// The `.init_array` entry above runs this once per process image, before
31/// `main`. It exits the process with status 127 when the coordinator socket is
32/// missing or Detcore cannot be installed, so a guest never runs past a failed
33/// installation.
34///
35/// # Safety
36///
37/// Only the dynamic loader may call this, from `.init_array`, while the process
38/// is still single-threaded and before any seccomp filter is active.
39#[unsafe(no_mangle)]
40pub unsafe extern "C" fn detcore_liteinst_initialize() {
41    let Some(socket) = std::env::var_os(reverie_liteinst::COORDINATOR_ENV) else {
42        fail("the coordinator socket environment variable is missing");
43    };
44    // SAFETY: the loader runs constructors before any application thread
45    // exists and before the application can install a seccomp filter, which is
46    // the window `install_tool` requires.
47    if let Err(error) = unsafe { reverie_liteinst::install_tool::<detcore::Detcore>(socket) } {
48        fail(&error.to_string());
49    }
50}
51
52fn fail(message: &str) -> ! {
53    eprintln!("detcore-liteinst: initialization failed: {message}");
54    // SAFETY: `_exit` takes no pointers and does not return.
55    unsafe { libc::_exit(127) }
56}
