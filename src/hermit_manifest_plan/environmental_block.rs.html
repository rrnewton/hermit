7//! Shared classification of captured output that shows an environmental block.
8
9use serde::Deserialize;
10use serde::Serialize;
11
12/// The closed environmental-block values emitted by this classifier.
13///
14/// Consumers that turn an environmental observation into a retry decision use
15/// this enum rather than matching the rendered string. Adding a classifier arm
16/// therefore requires every typed consumer to handle the new value.
17#[derive(
18    Clone,
19    Copy,
20    Debug,
21    Deserialize,
22    Eq,
23    Ord,
24    PartialEq,
25    PartialOrd,
26    Serialize
27)]
28#[serde(rename_all = "kebab-case")]
29pub enum EnvBlockClass {
30    BpfjailerBanner,
31    ProxyEgress,
32    ThirdPartyBuild,
33    ToolchainEperm,
34    VcsFsDenial,
35}
36
37impl EnvBlockClass {
38    pub fn as_str(self) -> &'static str {
39        match self {
40            Self::BpfjailerBanner => "bpfjailer-banner",
41            Self::ProxyEgress => "proxy-egress",
42            Self::ThirdPartyBuild => "third-party-build",
43            Self::ToolchainEperm => "toolchain-eperm",
44            Self::VcsFsDenial => "vcs-fs-denial",
45        }
46    }
47}

49/// Phrases that mean "the kernel/sandbox said no", in lowercase.
50const DENIALS: &[&str] = &[
51    "operation not permitted",
52    "permission denied",
53    "(os error 1)",
54];
55
56fn has_denial(line: &str) -> bool {
57    DENIALS.iter().any(|d| line.contains(d))
58}
59
60fn has_any(line: &str, needles: &[&str]) -> bool {
61    needles.iter().any(|n| line.contains(n))
62}

64/// Build-tool / compiler / linker phrasing for a Form-2 denial, in lowercase.
65///
66/// Split out of the old inline test so the same anchors can be applied per BLOCK
67/// (see [`diagnostic_blocks`]) instead of per physical line.
68fn has_toolchain_phrase(line: &str) -> bool {
69    let trimmed = line.trim_start();
70    trimmed.starts_with("error: failed to write ")
71        || trimmed.starts_with("rm: cannot remove ")
72        || (line.contains("fatal error: ") && line.matches(':').count() >= 2)
73        || line.contains("cmake error")
74        || has_any(
75            line,
76            &[
77                "cannot open",
78                "error opening",
79                "failed to open",
80                "could not open",
81                "opening dependency file",
82                "could not write output to",
83                "couldn't create the temp file",
84                "can't create",
85                "cannot execute",
86                "could not create temporary file",
87                "failed to build archive at",
88                // MEASURED 2026-08-17, five phrasings of one host condition that
89                // reached a node's output and were classified as product reds:
90                // rustc's incremental link_or_copy step, which says "unable to
91                // copy" where the sibling path says "could not write output to";
92                "unable to copy",
93                // A same-line cargo spawn denial retains the historical
94                // classification. Cross-line matching is narrower and handled
95                // by `has_cross_line_toolchain_phrase` below.
96                "could not execute process",
97                // `cp -a`/`ln -s` refusing to make a link, which "can't create"
98                // above does not spell the same way.
99                "cannot create symbolic link",
100                // The rustc and coreutils anchors are handled above as
101                // line-prefix checks. They MUST NOT use this substring matcher:
102                // guest prose can quote the complete diagnostic text next to a
103                // real EPERM, and that must remain a product red.
104            ],
105        )
106}

108/// Tool syntax proven safe to join with a denial elsewhere in the same Cargo
109/// diagnostic block. Generic phrases such as "could not open" stay same-line
110/// only: product prose can contain them above an indented guest EPERM.
111fn has_cross_line_toolchain_phrase(line: &str) -> bool {
112    line.trim_start().starts_with("could not execute process ")
113}

115/// Group `lower` into diagnostic blocks: one leading line plus its continuations.
116///
117/// WHY THIS EXISTS. Form 2 used to require the denial phrase and the build-tool
118/// phrase on the SAME physical line. Cargo does not oblige. Measured verbatim:
119///
120/// ```text
121/// error: failed to run custom build command for `libm v0.2.16`
122///
123/// Caused by:
124///   could not execute process .../build-script-build (never executed)
125///
126/// Caused by:
127///   Permission denied (os error 13)
128/// ```
129///
130/// The line carrying the tool phrase has no denial and the line carrying the
131/// denial has no tool phrase, so neither line satisfied both and a host denial
132/// was recorded as a product failure.
133///
134/// WHY THIS DOES NOT WEAKEN THE SCOPING RULE. This does NOT widen the search to
135/// the whole region — that is what would let an unrelated concurrent denial
136/// excuse a real red, and it stays forbidden. A block ends at the next
137/// UNINDENTED line, so a jail banner (`An action was blocked on this server…`,
138/// unindented) always starts its own block and can never be absorbed into a
139/// genuine compiler error's block. A gcc error keeps its indented source excerpt
140/// and nothing else. Continuations are: blank lines, indented lines, and the
141/// bare `Caused by:` header, which is exactly cargo's own error grouping.
142fn diagnostic_blocks(lower: &str) -> Vec<Vec<&str>> {
143    let mut blocks: Vec<Vec<&str>> = Vec::new();
144    for line in lower.lines() {
145        let trimmed = line.trim_start();
146        let continues = trimmed.is_empty()
147            || line.starts_with(' ')
148            || line.starts_with('\t')
149            || trimmed.starts_with("caused by:");
150        if continues && !blocks.is_empty() {
151            blocks.last_mut().expect("checked non-empty").push(line);
152        } else {
153            blocks.push(vec![line]);
154        }
155    }
156    blocks
157}

159/// Classify a failed gate's output as an ENVIRONMENTAL block, naming the class.
160///
161/// Returns `None` for a genuine product/test failure. Misclassifying a real
162/// failure as environmental is as harmful as the reverse, so every form-2
163/// (banner-less) anchor is pinned to build-tool / VCS phrasing: ordinary GUEST
164/// output that legitimately produces `EPERM` - `DETLOG ... madvise ... EPERM
165/// (Operation not permitted)`, the `kcmp-eperm` fixture, a `context: Mount`
166/// EPERM - must never trip it. The self-test brackets both directions with
167/// counts.
168///
169/// The classes, and what each is evidenced by:
170///
171/// * `bpfjailer-banner` - the canonical jail banner. The agent sandbox is a
172///   BPFJailer jail inherited by every descendant (`validate -> cargo ->
173///   rustc/cmake/cc1/ld`), and its FS/EXEC/NET enforcers transiently deny an
174///   `open`/`execve` for reasons unrelated to the code under test.
175/// * `toolchain-eperm` - a raw `EPERM`/`EACCES` leaked to a build tool with NO
176///   banner: `cc1 fatal error: .../stddef.h: Operation not permitted`, a CMake or
177///   linker denial on a system path, `rustc error: could not write output to ...`.
178///   Those files are world-readable `root:root -rw-r--r--` on this host, so a
179///   *compiler* reporting it cannot open a header for a permission reason is
180///   never legitimate product behaviour.
181/// * `third-party-build` - the vendored DynamoRIO/elfutils build under
182///   `reverie-dbi`. At `nproc=316` an unbounded dependency scan drives elfutils
183///   into a concurrency-exposed `SIGABRT`; that is a HOST build flake, not a
184///   Hermit defect (Hermit source is not what failed to compile).
185/// * `proxy-egress` - **NEW.** Egress through `fwdproxy` failed, so a networked
186///   gate could not reach GitHub. Verbatim from
187///   `/tmp/hermit-validate.WUrHlJ.log` (2026-08-07 09:37): `Lookup error: git
188///   ls-remote https://github.com/rrnewton/reverie.git refs/heads/main failed:
189///   fatal: unable to access '...': Could not resolve proxy: fwdproxy`, which the
190///   old regex did NOT match - that run recorded a PRODUCT red for the Reverie
191///   pin gate (the log contains no `ENVIRONMENTAL block` line) when nothing about
192///   the tree was wrong. `CONNECT tunnel failed, response 403` is the same class:
193///   the proxy is a per-destination allowlist, so a 403 is an egress verdict, not
194///   a Hermit result.
195/// * `vcs-fs-denial` - **NEW, defence in depth.** A banner-less git FS denial,
196///   e.g. a jail denying `git init`/`git config` inside
197///   `check-reverie-pin.rs`'s `/tmp` fixture repository. NOTE, honestly: the one
198///   occurrence found on disk (`/tmp/hermit-validate.H61gJP.log:1400-1419`,
199///   `Enforcer: FS, Reason: FILE_OPEN` while running `git -C
200///   /tmp/check-reverie-pin-stale-lock-... config user.name`) DID carry the
201///   jail banner and WAS already classified and retried successfully
202///   (`:1499 ENVIRONMENTAL block on attempt 1/3`). So this anchor is not fixing a
203///   measured miss; it covers the same denial arriving without the banner, which
204///   is how the FS enforcer surfaces when it denies a child that captures its own
205///   stderr.
206///
207/// What ONE output showed about an environmental block, in THREE STATES.
208///
209/// ⚠️ NOT `EnvBlockVerdict`, WHICH IS A DIFFERENT AXIS. That type settles whether
210/// a classification HYPOTHESIS survived a later re-execution (confirmed /
211/// refuted / unconfirmed). This one reports what a single captured region
212/// ACTUALLY CONTAINED. A run can be `Denied` here and `Refuted` there — observed
213/// a denial, then passed on rerun — and the two must not be read as the same fact.
214///
215/// ⚠️ TWO STATES CANNOT CARRY THREE FACTS, AND THIS IS WHERE THE THIRD GOT LOST.
216/// The old `Option<&str>` said `Some(class)` for a denial and `None` for
217/// everything else — collapsing "we looked and there was no denial" together with
218/// "there was nothing to look at". Those need opposite responses: the first is a
219/// real product failure, the second is a run that produced no evidence either way.
220#[derive(Debug, Clone, Copy, PartialEq, Eq)]
221pub enum EnvBlockObservation {
222    /// A denial FIRED, and this names its form.
223    Denied(EnvBlockClass),
224    /// Evidence was present and carried no denial. A product failure here is real.
225    NoDenial,
226    /// There was nothing to evaluate. NOT a statement that no denial occurred.
227    NothingObserved,
228}
229
230impl EnvBlockObservation {
231    pub fn block_class(self) -> Option<EnvBlockClass> {
232        match self {
233            Self::Denied(class) => Some(class),
234            Self::NoDenial | Self::NothingObserved => None,
235        }
236    }
237
238    /// The legacy two-state view, for callers that genuinely only ask "was it
239    /// blocked". `NothingObserved` maps to `None` — correct for that question,
240    /// and the reason the three-state form exists for callers asking anything else.
241    pub fn class(self) -> Option<&'static str> {
242        self.block_class().map(EnvBlockClass::as_str)
243    }
244}
245
246/// Preserved two-state entry point. Every existing caller keeps working; anything
247/// that must tell "no denial" from "no evidence" calls
248/// [`environmental_block_observation`] instead.
249pub fn environmental_block_class(output: &str) -> Option<&'static str> {
250    environmental_block_observation(output).class()
251}
252
253pub fn environmental_block_observation(output: &str) -> EnvBlockObservation {
254    // NOTHING TO EVALUATE IS ITS OWN ANSWER. An empty region means the step
255    // produced no output, which is not evidence that it ran unblocked.
256    if output.trim().is_empty() {
257        return EnvBlockObservation::NothingObserved;
258    }
259    let lower = output.to_ascii_lowercase();
260    // Form 1: the canonical jail banner, anywhere in the region.
261    //
262    // ⚠️ `lower.contains("bpfjailer")` IS DELIBERATELY UNANCHORED, AND I TRIED TO
263    // ANCHOR IT AND WAS WRONG. A bare mention does read as DENIED under it —
264    // measured on three shapes, including a diagnostic that says "no bpfjailer
265    // denial was observed this run" — and a `Denied` verdict is EXCUSING, so a
266    // false positive is silent absolution rather than a loud false red. That
267    // argument is real but it is not new information: requiring a denial word on
268    // the same line breaks a case `self_test` ASSERTS must be accepted,
269    //     "[e2e.metadata] Bunnylol `scuba bpfjailer_enforce` for more details"
270    // which is a genuine jailer output line carrying no denial word of its own.
271    // The unanchored form is a DECLARED TRADE, recorded in the self-test: accept
272    // some mentions rather than miss the jailer's own pointer line. Narrowing it
273    // needs a replacement that keeps that case, and that is an owner decision
274    // about which error to prefer, not a repair.
275    if lower.contains("blocked on this server based on a security policy")
276        || lower.contains("enforcer: fs, reason:")
277        || lower.contains("enforcer: exec, reason:")
278        || lower.contains("enforcer: net, reason:")
279        || lower.contains("bpfjailer")
280    {
281        return EnvBlockObservation::Denied(EnvBlockClass::BpfjailerBanner);
282    }
283    // Form 4 (checked before the per-line scan because it is a whole-region
284    // signature): the vendored third-party build script.
285    if (lower.contains("failed to run custom build command for") && lower.contains("reverie-dbi"))
286        || (lower.contains("panicked at") && lower.contains("reverie-dbi/build.rs"))
287    {
288        return EnvBlockObservation::Denied(EnvBlockClass::ThirdPartyBuild);
289    }
290    let mut vcs_hit = false;
291    for line in lower.lines() {
292        // Form 3 (NEW): egress through the forward proxy failed.
293        if has_any(
294            line,
295            &[
296                "could not resolve proxy",
297                "could not resolve host",
298                "connect tunnel failed, response 403",
299                "proxy connect aborted",
300                "failed to connect to fwdproxy",
301            ],
302        ) {
303            return EnvBlockObservation::Denied(EnvBlockClass::ProxyEgress);
304        }
305        // Form 2a: legacy same-line toolchain denial. The conjunction remains
306        // same-line so generic product prose cannot borrow a denial from a later
307        // indented line in its block.
308        if has_denial(line) && has_toolchain_phrase(line) {
309            return EnvBlockObservation::Denied(EnvBlockClass::ToolchainEperm);
310        }
311        // Form 5 (NEW): a banner-less git FS denial. Requires BOTH a git-fatal
312        // shape and a denial on the same line, so a guest test that merely prints
313        // "permission denied" cannot trip it.
314        if (line.starts_with("fatal:") || line.contains(" fatal: ") || line.contains(".git/"))
315            && has_any(
316                line,
317                &[
318                    "cannot mkdir",
319                    "could not create work tree dir",
320                    "could not create leading directories",
321                    "unable to create",
322                    "unable to write",
323                    "unable to access",
324                    "could not lock",
325                    "chmod on",
326                    ".git/",
327                ],
328            )
329        {
330            vcs_hit = true;
331        }
332    }
333    // Form 2b: the measured Cargo spawn diagnostic puts its denial in a
334    // continuation line. Only that structural phrase may join evidence across
335    // one diagnostic block; generic legacy phrases remain same-line above.
336    if diagnostic_blocks(&lower).iter().any(|block| {
337        block.iter().any(|line| has_denial(line))
338            && block
339                .iter()
340                .any(|line| has_cross_line_toolchain_phrase(line))
341    }) {
342        return EnvBlockObservation::Denied(EnvBlockClass::ToolchainEperm);
343    }
344    if vcs_hit {
345        return EnvBlockObservation::Denied(EnvBlockClass::VcsFsDenial);
346    }
347    EnvBlockObservation::NoDenial
348}
