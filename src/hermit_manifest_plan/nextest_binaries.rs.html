44/// Separate only the Cargo selectors that validation actually supports.
45/// Everything following `--` belongs to the test filter, even if it resembles
46/// a Cargo selector. Never reconstruct argv by joining it into shell text.
47pub fn split_arguments(args: &[String]) -> Result<Arguments, String> {
48    let mut result = Arguments {
49        build: Vec::new(),
50        runtime: Vec::new(),
51    };
52    let mut index = 0;
53    while index < args.len() {
54        let arg = &args[index];
55        if arg == "--" {
56            result.runtime.extend_from_slice(&args[index..]);
57            break;
58        }
59        match arg.as_str() {
60            "--workspace"
61            | "--all"
62            | "--lib"
63            | "--bins"
64            | "--tests"
65            | "--benches"
66            | "--examples"
67            | "--all-targets"
68            | "--all-features"
69            | "--no-default-features" => {
70                result.build.push(arg.clone());
71            }
72            "-p" | "--package" | "--exclude" | "-F" | "--features" | "--bin" | "--example"
73            | "--test" | "--bench" | "--target" => {
74                let value = args
75                    .get(index + 1)
76                    .filter(|value| !value.is_empty() && !value.starts_with('-'))
77                    .ok_or_else(|| format!("Cargo selector {arg} requires a value"))?;
78                result.build.extend([arg.clone(), value.clone()]);
79                index += 1;
80            }
81            "--release" | "--cargo-profile" | "--manifest-path" | "--target-dir" => {
82                return Err(format!("unrecorded Cargo build option {arg}"));
83            }
84            "--profile"
85            | "-j"
86            | "--test-threads"
87            | "-E"
88            | "--filter-expr"
89            | "--message-format"
90            | "--message-format-version"
91            | "--color" => {
92                let value = args
93                    .get(index + 1)
94                    .ok_or_else(|| format!("Nextest runtime option {arg} requires a value"))?;
95                result.runtime.extend([arg.clone(), value.clone()]);
96                index += 1;
97            }
98            "--no-capture" | "--nocapture" | "--no-fail-fast" | "--fail-fast" => {
99                result.runtime.push(arg.clone());
100            }
101            _ => {
102                if [
103                    "--package=",
104                    "--exclude=",
105                    "--features=",
106                    "--bin=",
107                    "--example=",
108                    "--test=",
109                    "--bench=",
110                    "--target=",
111                ]
112                .iter()
113                .any(|prefix| arg.starts_with(prefix))
114                {
115                    let (option, value) = arg.split_once('=').expect("matched equals prefix");
116                    if value.is_empty() {
117                        return Err(format!("Cargo selector {option} requires a value"));
118                    }
119                    result.build.extend([option.into(), value.into()]);
120                } else if arg.starts_with("--cargo-profile=")
121                    || arg.starts_with("--manifest-path=")
122                    || arg.starts_with("--target-dir=")
123                {
124                    return Err(format!("unrecorded Cargo build option {arg}"));
125                } else if ["--profile=", "--test-threads=", "--filter-expr="]
126                    .iter()
127                    .any(|prefix| arg.starts_with(prefix))
128                    || arg.strip_prefix("-j").is_some_and(|value| {
129                        !value.is_empty() && value.bytes().all(|b| b.is_ascii_digit())
130                    })
131                {
132                    result.runtime.push(arg.clone());
133                } else if arg.starts_with('-') {
134                    return Err(format!(
135                        "unrecognized option before the test-filter separator: {arg}"
136                    ));
137                } else {
138                    result.runtime.push(arg.clone());
139                }
140            }
141        }
142        index += 1;
143    }
144    Ok(result)
145}

163fn file_identity(path: &Path, executable: bool) -> Result<FileIdentity, String> {
164    let before = fs::symlink_metadata(path)
165        .map_err(|error| format!("cannot inspect prepared input {}: {error}", path.display()))?;
166    if !before.is_file() || executable && before.mode() & 0o111 == 0 {
167        return Err(format!(
168            "prepared input is not a regular {}file: {}",
169            if executable { "executable " } else { "" },
170            path.display()
171        ));
172    }
173    let mut file = fs::File::open(path)
174        .map_err(|error| format!("cannot open prepared input {}: {error}", path.display()))?;
175    let opened = file
176        .metadata()
177        .map_err(|error| format!("cannot inspect open input {}: {error}", path.display()))?;
178    if before.dev() != opened.dev() || before.ino() != opened.ino() {
179        return Err(format!(
180            "prepared input changed while opening {}",
181            path.display()
182        ));
183    }
184    let mut hasher = Sha256::new();
185    let mut buffer = [0; 128 * 1024];
186    loop {
187        let count = file
188            .read(&mut buffer)
189            .map_err(|error| format!("cannot hash prepared input {}: {error}", path.display()))?;
190        if count == 0 {
191            break;
192        }
193        hasher.update(&buffer[..count]);
194    }
195    let after = fs::symlink_metadata(path)
196        .map_err(|error| format!("prepared input disappeared {}: {error}", path.display()))?;
197    if before.dev() != after.dev()
198        || before.ino() != after.ino()
199        || before.len() != after.len()
200        || before.mode() != after.mode()
201        || before.mtime() != after.mtime()
202        || before.mtime_nsec() != after.mtime_nsec()
203        || before.ctime() != after.ctime()
204        || before.ctime_nsec() != after.ctime_nsec()
205    {
206        return Err(format!(
207            "prepared input changed while hashing {}",
208            path.display()
209        ));
210    }
211    Ok(FileIdentity {
212        path: path.to_owned(),
213        size: before.len(),
214        mode: before.mode(),
215        sha256: format!("{:x}", hasher.finalize()),
216    })
217}

219fn git_bytes(root: &Path, args: &[&str]) -> Result<Vec<u8>, String> {
220    let result = Command::new("git")
221        .args(args)
222        .current_dir(root)
223        .output()
224        .map_err(|error| format!("cannot inspect preparation source: {error}"))?;
225    if !result.status.success() {
226        return Err(format!(
227            "cannot inspect preparation source with git {args:?}: {}",
228            String::from_utf8_lossy(&result.stderr).trim()
229        ));
230    }
231    Ok(result.stdout)
232}

234fn source_identity(root: &Path) -> Result<String, String> {
235    let mut hash = Sha256::new();
236    for args in [
237        &["rev-parse", "HEAD", "HEAD^{tree}"][..],
238        &[
239            "diff",
240            "--binary",
241            "--no-ext-diff",
242            "--submodule=diff",
243            "--no-textconv",
244            "HEAD",
245            "--",
246        ][..],
247    ] {
248        let bytes = git_bytes(root, args)?;
249        hash.update((bytes.len() as u64).to_le_bytes());
250        hash.update(bytes);
251    }
252    let untracked = git_bytes(root, &["ls-files", "--others", "--exclude-standard", "-z"])?;
253    for name in untracked
254        .split(|byte| *byte == 0)
255        .filter(|name| !name.is_empty())
256    {
257        let name = std::str::from_utf8(name)
258            .map_err(|_| "untracked preparation source path is not UTF-8")?;
259        let identity = file_identity(&root.join(name), false)?;
260        hash.update((name.len() as u64).to_le_bytes());
261        hash.update(name.as_bytes());
262        hash.update(identity.sha256.as_bytes());
263        hash.update(identity.mode.to_le_bytes());
264    }
265    Ok(format!("{:x}", hash.finalize()))
266}

283fn target_path(target: &Path, raw: &str) -> Result<PathBuf, String> {
284    let input = Path::new(raw);
285    if input
286        .components()
287        .any(|part| matches!(part, Component::ParentDir))
288    {
289        return Err(format!("prepared input contains a parent traversal: {raw}"));
290    }
291    let path = if input.is_absolute() {
292        input.to_owned()
293    } else {
294        target.join(input)
295    };
296    if !path.starts_with(target) {
297        return Err(format!(
298            "prepared input is outside target {}: {raw}",
299            target.display()
300        ));
301    }
302    let canonical = path
303        .canonicalize()
304        .map_err(|error| format!("missing prepared input {}: {error}", path.display()))?;
305    if canonical != path {
306        return Err(format!(
307            "prepared input uses a symlink or noncanonical path: {raw}"
308        ));
309    }
310    Ok(path)
311}

352fn cpu_wrapper_artifact(
353    events: &str,
354    cargo: &Value,
355    target: &Path,
356) -> Result<FileIdentity, String> {
357    let packages = cargo["packages"]
358        .as_array()
359        .ok_or("missing Cargo packages")?;
360    let packages = packages
361        .iter()
362        .filter(|p| p["name"] == CPU_WRAPPER_PACKAGE)
363        .collect::<Vec<_>>();
364    if packages.len() != 1 {
365        return Err("CPU wrapper package is missing or ambiguous".into());
366    }
367    let package = packages[0];
368    let package_id = string(package, "id")?;
369    let targets = package["targets"]
370        .as_array()
371        .ok_or("missing CPU wrapper targets")?;
372    if targets
373        .iter()
374        .filter(|t| t["name"] == CPU_WRAPPER_TARGET && t["kind"] == serde_json::json!(["bin"]))
375        .count()
376        != 1
377    {
378        return Err("CPU wrapper Cargo target is missing or ambiguous".into());
379    }
380    let mut found = None;
381    for line in events.lines() {
382        let event: Value = serde_json::from_str(line)
383            .map_err(|e| format!("invalid CPU wrapper Cargo event: {e}"))?;
384        if event["reason"] != "compiler-artifact" || event["target"]["name"] != CPU_WRAPPER_TARGET {
385            continue;
386        }
387        if event["package_id"] != package_id
388            || event["target"]["kind"] != serde_json::json!(["bin"])
389            || event["profile"]["test"] != false
390        {
391            return Err("CPU wrapper artifact has the wrong package, target, or profile".into());
392        }
393        let path = target_path(target, string(&event, "executable")?)?;
394        if found.replace(file_identity(&path, true)?).is_some() {
395            return Err("Cargo emitted ambiguous CPU wrapper artifacts".into());
396        }
397    }
398    found.ok_or_else(|| "Cargo did not report the normal CPU wrapper executable".into())
399}

401fn verify_cpu_wrapper(record: &PreparedRecord, cargo: &Value) -> Result<(), String> {
402    check_identity(&record.cpu_wrapper.metadata, false)?;
403    let events =
404        fs::read_to_string(&record.cpu_wrapper.metadata.path).map_err(|e| e.to_string())?;
405    if cpu_wrapper_artifact(&events, cargo, &record.target)? != record.cpu_wrapper.executable {
406        return Err("prepared CPU wrapper executable changed".into());
407    }
408    if let Some(configured) = std::env::var_os(CPU_WRAPPER_ENV) {
409        if Path::new(&configured) != record.cpu_wrapper.executable.path {
410            return Err("configured CPU wrapper differs from the prepared executable".into());
411        }
412    }
413    Ok(())
414}

433/// Explicit standalone preparation. Official test consumers never call this.
434pub fn build_cpu_wrapper(root: &Path) -> Result<PathBuf, PreparationError> {
435    if std::env::var(REQUIRED_ENV).as_deref() == Ok("1") {
436        return Err("an official consumer cannot build a CPU wrapper".into());
437    }
438    let root = root.canonicalize().map_err(|e| e.to_string())?;
439    let artifacts = LockedArtifacts::open(&root, true)?;
440    let cargo_path = artifacts.root.join("standalone-cargo.json");
441    cargo_output(
442        &root,
443        &["metadata", "--locked", "--format-version", "1"].map(String::from),
444        &cargo_path,
445    )?;
446    let cargo = read_json(&cargo_path)?;
447    let events = artifacts.root.join("standalone-cpu-wrapper.jsonl");
448    prepare_cpu_wrapper(&root, &events)?;
449    Ok(cpu_wrapper_artifact(
450        &fs::read_to_string(events).map_err(|e| e.to_string())?,
451        &cargo,
452        Path::new(string(&cargo, "target_directory")?),
453    )?
454    .path)
455}

466    std::env::vars()
467        .filter(|(name, _)| {
468            matches!(
469                name.as_str(),
470                "RUSTFLAGS"
471                    | "SOURCE_DATE_EPOCH"
472                    | "CARGO_ENCODED_RUSTFLAGS"
473                    | "RUSTDOCFLAGS"
474                    | "RUSTC"
475                    | "RUSTC_WRAPPER"
476                    | "RUSTC_WORKSPACE_WRAPPER"
477                    | "RUSTUP_TOOLCHAIN"
478                    | "CARGO_BUILD_TARGET"
479                    | "CARGO_TARGET_DIR"
480                    | "CARGO_BUILD_BUILD_DIR"
481                    | "CC"
482                    | "CXX"
483                    | "AR"
484                    | "CFLAGS"
485                    | "CXXFLAGS"
486                    | "LDFLAGS"
487                    | "LIBRARY_PATH"
488            ) || name.starts_with("CARGO_PROFILE_")
489                || (name.starts_with("CARGO_TARGET_") && name != "CARGO_TARGET_TMPDIR")
490                || ["CC_", "CXX_", "AR_", "CFLAGS_", "CXXFLAGS_"]
491                    .iter()
492                    .any(|prefix| name.starts_with(prefix))
493        })
494        .collect()
495}

508fn metadata_binaries(
509    metadata: &Value,
510    cargo: &Value,
511    target: &Path,
512) -> Result<(Vec<BinaryIdentity>, Vec<FileIdentity>), String> {
513    let build = metadata
514        .get("rust-build-meta")
515        .ok_or("Nextest metadata has no rust-build-meta")?;
516    if Path::new(string(build, "target-directory")?) != target {
517        return Err("Nextest metadata names a different Cargo target directory".into());
518    }
519    let packages = cargo
520        .get("packages")
521        .and_then(Value::as_array)
522        .ok_or("Cargo metadata has no package list")?;
523    let mut package_ids = BTreeMap::new();
524    for package in packages {
525        let id = string(package, "id")?;
526        if package_ids.insert(id, package).is_some() {
527            return Err(format!("Cargo metadata repeats package identity {id}"));
528        }
529    }
530    let binaries = metadata
531        .get("rust-binaries")
532        .and_then(Value::as_object)
533        .filter(|binaries| !binaries.is_empty())
534        .ok_or("Nextest metadata has no test executables")?;
535    let mut result = Vec::new();
536    let mut paths = BTreeSet::new();
537    let mut seen_targets = BTreeSet::new();
538    for (id, binary) in binaries {
539        if string(binary, "binary-id")? != id {
540            return Err(format!(
541                "Nextest binary key disagrees with its identity: {id}"
542            ));
543        }
544        let package_id = string(binary, "package-id")?;
545        let package = package_ids
546            .get(package_id)
547            .ok_or_else(|| format!("Nextest binary {id} names an unknown package {package_id}"))?;
548        let name = string(binary, "binary-name")?;
549        let kind = string(binary, "kind")?;
550        let targets = package
551            .get("targets")
552            .and_then(Value::as_array)
553            .ok_or_else(|| format!("Cargo package {package_id} has no targets"))?;
554        if !targets.iter().any(|target| {
555            target.get("name").and_then(Value::as_str) == Some(name)
556                && target
557                    .get("kind")
558                    .and_then(Value::as_array)
559                    .is_some_and(|kinds| {
560                        kinds.iter().any(|value| value.as_str() == Some(kind))
561                            || (kind == "lib"
562                                && !kinds.is_empty()
563                                && kinds.iter().all(|value| {
564                                    matches!(
565                                        value.as_str(),
566                                        Some(
567                                            "rlib"
568                                                | "dylib"
569                                                | "cdylib"
570                                                | "staticlib"
571                                                | "proc-macro"
572                                        )
573                                    )
574                                }))
575                    })
576        }) {
577            return Err(format!(
578                "Nextest executable {id} is not the declared {kind} target {name} in {package_id}"
579            ));
580        }
581        if !seen_targets.insert((package_id, name, kind, string(binary, "build-platform")?)) {
582            return Err(format!(
583                "Nextest metadata has ambiguous executables for {package_id} {kind} {name}"
584            ));
585        }
586        let path = target_path(target, string(binary, "binary-path")?)?;
587        if !paths.insert(path.clone()) {
588            return Err(format!(
589                "Nextest identities share one executable path: {}",
590                path.display()
591            ));
592        }
593        result.push(BinaryIdentity {
594            binary_id: id.clone(),
595            package_id: package_id.into(),
596            binary_name: name.into(),
597            kind: kind.into(),
598            build_platform: string(binary, "build-platform")?.into(),
599            executable: file_identity(&path, true)?,
600        });
601    }
602    let mut runtime_paths = BTreeSet::new();
603    for (package, files) in build
604        .get("non-test-binaries")
605        .and_then(Value::as_object)
606        .ok_or("Nextest metadata has no non-test-binaries map")?
607    {
608        if !package_ids.contains_key(package.as_str()) {
609            return Err(format!(
610                "Nextest runtime files name unknown Cargo package {package}"
611            ));
612        }
613        for file in files.as_array().ok_or("invalid non-test-binaries entry")? {
614            runtime_paths.insert(target_path(target, string(file, "path")?)?);
615        }
616    }
617    let runtime_files = runtime_paths
618        .iter()
619        .map(|path| file_identity(path, false))
620        .collect::<Result<Vec<_>, _>>()?;
621    Ok((result, runtime_files))
622}

624fn sources(cargo: &Value, root: &Path) -> Result<BTreeMap<PathBuf, String>, String> {
625    let mut roots = BTreeSet::from([root.to_owned()]);
626    for package in cargo["packages"]
627        .as_array()
628        .ok_or("Cargo packages are missing")?
629    {
630        // Registry sources have checksum-bound entries in Cargo.lock. Local
631        // and git packages may have mutable checkouts outside this repository.
632        if package["source"]
633            .as_str()
634            .is_some_and(|source| source.starts_with("registry+"))
635        {
636            continue;
637        }
638        let manifest = Path::new(string(package, "manifest_path")?);
639        let directory = manifest.parent().ok_or("package manifest has no parent")?;
640        let git_root = git_bytes(directory, &["rev-parse", "--show-toplevel"])?;
641        let git_root = std::str::from_utf8(&git_root)
642            .map_err(|_| "non-UTF-8 dependency root")?
643            .trim();
644        roots.insert(PathBuf::from(git_root));
645    }
646    roots
647        .into_iter()
648        .map(|root| source_identity(&root).map(|identity| (root, identity)))
649        .collect()
650}

652fn cargo_config(root: &Path) -> Result<BTreeMap<PathBuf, Option<FileIdentity>>, String> {
653    let mut paths = BTreeSet::new();
654    for parent in root.ancestors() {
655        for name in ["config", "config.toml"] {
656            paths.insert(parent.join(".cargo").join(name));
657        }
658    }
659    let home = std::env::var_os("CARGO_HOME")
660        .map(PathBuf::from)
661        .or_else(|| std::env::var_os("HOME").map(|home| PathBuf::from(home).join(".cargo")))
662        .ok_or("cannot determine Cargo home")?;
663    for name in ["config", "config.toml"] {
664        paths.insert(home.join(name));
665    }
666    paths
667        .into_iter()
668        .map(|path| {
669            let identity = match fs::symlink_metadata(&path) {
670                Ok(_) => Some(file_identity(&path, false)?),
671                Err(error) if error.kind() == std::io::ErrorKind::NotFound => None,
672                Err(error) => {
673                    return Err(format!(
674                        "cannot inspect Cargo configuration {}: {error}",
675                        path.display()
676                    ));
677                }
678            };
679            Ok((path, identity))
680        })
681        .collect()
682}

695pub fn config_selections(
696    graph: &dagrun::model::DagConfig,
697    profile: &str,
698) -> Result<BTreeMap<String, Vec<String>>, String> {
699    let selected =
700        dagrun::select_steps_by_labels(graph, &[profile.into()]).map_err(|e| e.to_string())?;
701    let mut selections = BTreeMap::new();
702    for step in selected.steps {
703        if let Some(args) = step.env.get(SELECTION_ENV) {
704            if step.env.get(REQUIRED_ENV).map(String::as_str) != Some("1") {
705                return Err(format!(
706                    "{} does not require prepared executables",
707                    step.tag()
708                ));
709            }
710            let args: Vec<String> =
711                serde_json::from_str(args).map_err(|e| format!("{}: {e}", step.tag()))?;
712            let parsed = split_arguments(&args)?;
713            if parsed.build != args || !parsed.runtime.is_empty() || args.is_empty() {
714                return Err(format!(
715                    "{} has an invalid prepared Cargo selection",
716                    step.tag()
717                ));
718            }
719            selections.insert(selection_key(&args), args);
720        }
721    }
722    if selections.is_empty() {
723        return Err(format!(
724            "profile {profile} has no prepared Nextest selections"
725        ));
726    }
727    Ok(selections)
728}

730struct LockedArtifacts {
731    root: PathBuf,
732    _lock: fs::File,
733}
734
735impl LockedArtifacts {
736    fn open(root: &Path, exclusive: bool) -> Result<Self, String> {
737        let root = root.join("target/ci/nextest-binaries");
738        fs::create_dir_all(&root).map_err(|e| e.to_string())?;
739        let lock = fs::OpenOptions::new()
740            .create(true)
741            .truncate(false)
742            .read(true)
743            .write(true)
744            .open(root.join("lock"))
745            .map_err(|e| e.to_string())?;
746        // Keep this file open across verification and the entire Nextest child.
747        // A later producer cannot replace a generation beneath a running test.
748        if unsafe {
749            libc::flock(
750                lock.as_raw_fd(),
751                if exclusive {
752                    libc::LOCK_EX
753                } else {
754                    libc::LOCK_SH
755                },
756            )
757        } != 0
758        {
759            return Err(format!(
760                "cannot lock Nextest artifacts: {}",
761                std::io::Error::last_os_error()
762            ));
763        }
764        Ok(Self { root, _lock: lock })
765    }
766
767    fn current(&self) -> Result<PreparedRecord, String> {
768        let path = self.root.join("current.json");
769        let _: FileIdentity = file_identity(&path, false)?;
770        serde_json::from_slice(&fs::read(&path).map_err(|e| e.to_string())?)
771            .map_err(|e| format!("invalid prepared record {}: {e}", path.display()))
772    }
773}

803fn cargo_output(root: &Path, args: &[String], destination: &Path) -> Result<(), PreparationError> {
804    let output = fs::File::create(destination).map_err(|e| e.to_string())?;
805    let status = Command::new(root.join("ci/run-with-reverie-dbt-budget.sh"))
806        .arg("cargo")
807        .args(args)

809        .current_dir(root)
810        .stdout(output)
811        .status()
812        .map_err(|e| e.to_string())?;
813    if !status.success() {
814        return Err(PreparationError {
815            message: format!("Cargo preparation failed ({status}): {args:?}"),
816            status: u8::try_from(exit_code(status)).unwrap_or(1),
817        });
818    }
819    Ok(())
820}

833fn verify_record(root: &Path, record: &PreparedRecord) -> Result<Value, String> {
834    if record.schema != RECORD_SCHEMA || record.repository != root {
835        return Err("prepared record has the wrong schema or repository".into());
836    }
837    check_identity(&record.cargo_metadata, false)?;
838    let cargo = read_json(&record.cargo_metadata.path)?;
839    let target = Path::new(string(&cargo, "target_directory")?);
840    if target != record.target || target.canonicalize().map_err(|e| e.to_string())? != target {
841        return Err("prepared record has the wrong Cargo target directory".into());
842    }
843    if record.sources != sources(&cargo, root)? {
844        return Err("prepared executables are stale: source identity changed".into());
845    }
846    if record.rustc != rustc_identity()?
847        || record.build_environment != build_environment()
848        || record.cargo_config != cargo_config(root)?
849    {
850        return Err("prepared executables are stale: compiler, Cargo configuration, or build environment changed".into());
851    }
852    verify_cpu_wrapper(record, &cargo)?;
853    Ok(cargo)
854}

856fn verify_selection(
857    record: &PreparedRecord,
858    cargo: &Value,
859    args: &[String],
860) -> Result<SelectionRecord, String> {
861    let selected = record
862        .selections
863        .get(&selection_key(args))
864        .ok_or_else(|| format!("no prepared executable selection for {args:?}"))?;
865    if selected.build_args != args {
866        return Err("prepared Cargo selection does not match its key".into());
867    }
868    check_identity(&selected.metadata, false)?;
869    let metadata = read_json(&selected.metadata.path)?;
870    let (binaries, runtime_files) = metadata_binaries(&metadata, cargo, &record.target)?;
871    if selected.binaries != binaries || selected.runtime_files != runtime_files {
872        return Err("prepared executable, package identity, or runtime file changed".into());
873    }
874    Ok(selected.clone())
875}

877fn verify_guests(record: &PreparedRecord) -> Result<BTreeMap<String, String>, String> {
878    if !record.selections.values().any(|selection| {
879        selection
880            .build_args
881            .windows(2)
882            .any(|args| args == ["--test", "hermit_modes"])
883    }) {
884        if !record.guests.is_empty() {
885            return Err(
886                "unexpected Cargo guest records for a selection without hermit_modes".into(),
887            );
888        }
889        return Ok(BTreeMap::new());
890    }
891    let mut result = BTreeMap::new();
892    for guest in &record.guests {
893        check_identity(guest, true)?;
894        let name = guest
895            .path
896            .file_name()
897            .and_then(|s| s.to_str())
898            .ok_or("invalid guest path")?;
899        if !guest.path.starts_with(&record.target)
900            || result
901                .insert(name.into(), guest.path.to_string_lossy().into())
902                .is_some()
903        {
904            return Err("prepared guest identities are outside the target or ambiguous".into());
905        }
906    }
907    if result.keys().map(String::as_str).collect::<BTreeSet<_>>()
908        != cargo_guests::CARGO_GUEST_BINARIES.into_iter().collect()
909    {
910        return Err("prepared Cargo guest population differs from hermit_modes fixtures".into());
911    }
912    Ok(result)
913}

915fn guest_executables(
916    events: &str,
917    cargo: &Value,
918    target: &Path,
919) -> Result<Vec<FileIdentity>, String> {
920    let package = cargo["packages"]
921        .as_array()
922        .ok_or("missing Cargo packages")?
923        .iter()
924        .filter(|package| package["name"].as_str() == Some("hermetic_infra_hermit_tests"))
925        .collect::<Vec<_>>();
926    if package.len() != 1 {
927        return Err("Cargo guest package is missing or ambiguous".into());
928    }
929    let package = string(package[0], "id")?;
930    let mut by_name = BTreeMap::<String, PathBuf>::new();
931    for line in events.lines() {
932        let event: Value =
933            serde_json::from_str(line).map_err(|e| format!("invalid Cargo guest event: {e}"))?;
934        if event["reason"] != "compiler-artifact"
935            || event["package_id"] != package
936            || event["executable"].is_null()
937        {
938            continue;
939        }
940        let name = string(&event["target"], "name")?;
941        if !cargo_guests::CARGO_GUEST_BINARIES.contains(&name) {
942            continue;
943        }
944        if event["target"]["kind"] != serde_json::json!(["bin"])
945            || event["profile"]["test"] != false
946        {
947            return Err(format!("Cargo guest {name} has the wrong target/profile"));
948        }
949        let path = target_path(target, string(&event, "executable")?)?;
950        if by_name.insert(name.into(), path).is_some() {
951            return Err(format!("Cargo emitted ambiguous guest {name}"));
952        }
953    }
954    if by_name.keys().map(String::as_str).collect::<BTreeSet<_>>()
955        != cargo_guests::CARGO_GUEST_BINARIES.into_iter().collect()
956    {
957        return Err("Cargo did not report all 21 hermit_modes guest executables".into());
958    }
959    by_name
960        .values()
961        .map(|path| file_identity(path, true))
962        .collect()
963}

965pub fn prepare(root: &Path, profile: &str) -> Result<(), PreparationError> {
966    let root = root.canonicalize().map_err(|e| e.to_string())?;
967    let selections = profile_selections(&root, profile)?;
968    let artifacts = LockedArtifacts::open(&root, true)?;
969    let generation = artifacts.root.join(format!(
970        "generation-{}-{}",
971        std::process::id(),
972        std::time::SystemTime::now()
973            .duration_since(std::time::UNIX_EPOCH)
974            .map_err(|e| e.to_string())?
975            .as_nanos()
976    ));
977    fs::create_dir(&generation).map_err(|e| e.to_string())?;
978    let cargo_path = generation.join("cargo.json");
979    cargo_output(
980        &root,
981        &[
982            "metadata".into(),
983            "--format-version".into(),
984            "1".into(),
985            "--locked".into(),
986        ],
987        &cargo_path,
988    )?;
989    let cargo = read_json(&cargo_path)?;
990    let before_sources = sources(&cargo, &root)?;
991    let before_rustc = rustc_identity()?;
992    let before_environment = build_environment();
993    let before_config = cargo_config(&root)?;
994    let target = PathBuf::from(string(&cargo, "target_directory")?);
995    fs::create_dir_all(&target).map_err(|e| e.to_string())?;
996    if !target.is_absolute() || target.canonicalize().map_err(|e| e.to_string())? != target {
997        return Err("Cargo reported a noncanonical target directory".into());
998    }
999    for (key, selection) in &selections {
1000        let mut args = vec![
1001            "nextest".into(),
1002            "list".into(),
1003            "--locked".into(),
1004            "--message-format".into(),
1005            "json".into(),
1006            "--list-type".into(),
1007            "binaries-only".into(),
1008        ];
1009        args.extend(selection.iter().cloned());
1010        cargo_output(&root, &args, &generation.join(format!("{key}.json")))?;
1011    }
1012    let needs_guests = selections.values().any(|args| {
1013        args.windows(2)
1014            .any(|args| args == ["--test", "hermit_modes"])
1015    });
1016    let guest_path = generation.join("guests.jsonl");
1017    if needs_guests {
1018        cargo_output(
1019            &root,
1020            &[
1021                "build",
1022                "--locked",
1023                "-p",
1024                "hermetic_infra_hermit_tests",
1025                "--bins",
1026                "--message-format=json",
1027            ]
1028            .map(String::from),
1029            &guest_path,
1030        )?;
1031    }
1032    let cpu_wrapper_path = generation.join("cpu-wrapper.jsonl");
1033    prepare_cpu_wrapper(&root, &cpu_wrapper_path)?;
1034    // Hash only after every Cargo selection has completed. Shared paths may be
1035    // rebuilt by another selection; every published record names the final files.
1036    let mut recorded_selections = BTreeMap::new();
1037    for (key, build_args) in selections {
1038        let path = generation.join(format!("{key}.json"));
1039        let (binaries, runtime_files) = metadata_binaries(&read_json(&path)?, &cargo, &target)?;
1040        let metadata = file_identity(&path, false)?;
1041        let args = [
1042            "nextest",
1043            "list",
1044            "--cargo-metadata",
1045            cargo_path.to_str().ok_or("non-UTF-8 Cargo metadata path")?,
1046            "--binaries-metadata",
1047            path.to_str().ok_or("non-UTF-8 Nextest metadata path")?,
1048            "--message-format",
1049            "json",
1050        ]
1051        .map(String::from);
1052        // Nextest itself must accept every complete metadata pair after the
1053        // final build; this enumerates tests and never invokes the compiler.
1054        cargo_output(&root, &args, &generation.join(format!("{key}.tests.json")))?;
1055        recorded_selections.insert(
1056            key,
1057            SelectionRecord {
1058                build_args,
1059                metadata,
1060                binaries,
1061                runtime_files,
1062            },
1063        );
1064    }
1065    let guests = if needs_guests {
1066        guest_executables(
1067            &fs::read_to_string(&guest_path).map_err(|e| e.to_string())?,
1068            &cargo,
1069            &target,
1070        )?
1071    } else {
1072        vec![]
1073    };
1074    let record = PreparedRecord {
1075        schema: RECORD_SCHEMA,
1076        repository: root.clone(),
1077        target: target.clone(),
1078        sources: before_sources,
1079        rustc: before_rustc,
1080        cargo_config: before_config,
1081        build_environment: before_environment,
1082        cargo_metadata: file_identity(&cargo_path, false)?,
1083        selections: recorded_selections,
1084        guests,
1085        cpu_wrapper: CpuWrapperRecord {
1086            executable: cpu_wrapper_artifact(
1087                &fs::read_to_string(&cpu_wrapper_path).map_err(|e| e.to_string())?,
1088                &cargo,
1089                &target,
1090            )?,
1091            metadata: file_identity(&cpu_wrapper_path, false)?,
1092        },
1093    };
1094    let checked_cargo = verify_record(&root, &record)?;
1095    for selection in record.selections.values() {
1096        verify_selection(&record, &checked_cargo, &selection.build_args)?;
1097    }
1098    verify_guests(&record)?;
1099    let staging = generation.join("record.json");
1100    fs::write(
1101        &staging,
1102        serde_json::to_vec_pretty(&record).map_err(|e| e.to_string())?,
1103    )
1104    .map_err(|e| e.to_string())?;
1105    // A single rename publishes all selections together. Failure leaves the
1106    // previous record in place; it remains usable only while its recorded
1107    // inputs still match, since earlier Cargo commands may have changed them.
1108    fs::rename(staging, artifacts.root.join("current.json")).map_err(|e| e.to_string())?;
1109    eprintln!(
1110        "prepared-nextest: published {} selections and {} Cargo guests for {profile}",
1111        record.selections.len(),
1112        record.guests.len()
1113    );
1114    Ok(())
1115}

1117pub fn run(
1118    root: &Path,
1119    operation: &str,
1120    config: Option<&Path>,
1121    args: &[String],
1122) -> Result<i32, String> {
1123    if !matches!(operation, "run" | "list") {
1124        return Err(format!("unsupported Nextest operation {operation}"));
1125    }
1126    let root = root.canonicalize().map_err(|e| e.to_string())?;
1127    let parsed = split_arguments(args)?;
1128    if let Ok(declared) = std::env::var(SELECTION_ENV) {
1129        let declared: Vec<String> =
1130            serde_json::from_str(&declared).map_err(|e| format!("invalid {SELECTION_ENV}: {e}"))?;
1131        if declared != parsed.build {
1132            return Err("actual Cargo selectors differ from the committed node's selection".into());
1133        }
1134    } else if std::env::var(REQUIRED_ENV).as_deref() == Ok("1") {
1135        return Err(format!("official Nextest consumer has no {SELECTION_ENV}"));
1136    }
1137    let artifacts = LockedArtifacts::open(&root, false)?;
1138    let record = artifacts.current()?;
1139    let cargo = verify_record(&root, &record)?;
1140    let selection = verify_selection(&record, &cargo, &parsed.build)?;
1141    let guests = verify_guests(&record)?;
1142    let mut command = Command::new("cargo");
1143    command.arg("nextest");
1144    if let Some(config) = config {
1145        command.arg("--config-file").arg(config);
1146    }
1147    command
1148        .arg(operation)
1149        .arg("--cargo-metadata")
1150        .arg(&record.cargo_metadata.path)
1151        .arg("--binaries-metadata")
1152        .arg(&selection.metadata.path)
1153        .args(&parsed.runtime)
1154        .env(
1155            GUESTS_ENV,
1156            serde_json::to_string(&guests).map_err(|e| e.to_string())?,
1157        )
1158        .current_dir(&root);
1159    let status = command
1160        .status()
1161        .map_err(|e| format!("cannot run prepared Nextest: {e}"))?;
1162    // Keep the shared lock until the child has joined. Preserve Nextest's
1163    // status; a refusal never becomes an empty or successful test report.
1164    drop(artifacts);
1165    Ok(exit_code(status))
1166}

1180pub fn executable(root: &Path, package: &str, name: &str) -> Result<PathBuf, String> {
1181    let root = root.canonicalize().map_err(|e| e.to_string())?;
1182    let artifacts = LockedArtifacts::open(&root, false)?;
1183    let record = artifacts.current()?;
1184    let cargo = verify_record(&root, &record)?;
1185    let ids = cargo["packages"]
1186        .as_array()
1187        .ok_or("missing Cargo packages")?
1188        .iter()
1189        .filter(|item| item["name"].as_str() == Some(package))
1190        .map(|item| string(item, "id"))
1191        .collect::<Result<BTreeSet<_>, _>>()?;
1192    if ids.len() != 1 {
1193        return Err(format!(
1194            "prepared package {package} is missing or ambiguous"
1195        ));
1196    }
1197    let required_selection = if std::env::var(REQUIRED_ENV).as_deref() == Ok("1") {
1198        let raw = std::env::var(SELECTION_ENV)
1199            .map_err(|_| "prepared executable lookup has no declared build selection")?;
1200        let args: Vec<String> = serde_json::from_str(&raw)
1201            .map_err(|e| format!("invalid executable build selection: {e}"))?;
1202        Some(verify_selection(&record, &cargo, &args)?)
1203    } else {
1204        None
1205    };
1206    let selections = match required_selection.as_ref() {
1207        Some(selection) => vec![selection],
1208        None => record.selections.values().collect(),
1209    };
1210    let mut matches = BTreeSet::new();
1211    for selection in selections {
1212        for binary in &selection.binaries {
1213            if ids.contains(binary.package_id.as_str())
1214                && binary.binary_name == name
1215                && binary.kind == "test"
1216            {
1217                verify_selection(&record, &cargo, &selection.build_args)?;
1218                matches.insert(binary.executable.path.clone());
1219            }
1220        }
1221    }
1222    if matches.len() != 1 {
1223        return Err(format!(
1224            "expected one prepared {package} test target {name}, found {}",
1225            matches.len()
1226        ));
1227    }
1228    Ok(matches.into_iter().next().unwrap())
1229}
