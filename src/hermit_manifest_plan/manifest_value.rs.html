7/// The deliberately small data model accepted by the E2E manifest schema.
8///
9/// Keeping this independent of the serialization format lets the validator and
10/// execution code consume one representation while the checked-in manifests
11/// use YAML for readable nested mode/backend policy.
12#[derive(Clone, Debug, PartialEq, Serialize)]
13#[serde(untagged)]
14pub enum Value {
15    String(String),
16    Integer(i64),
17    Float(f64),
18    Boolean(bool),
19    Array(Vec<Value>),
20    Table(BTreeMap<String, Value>),
21    Null,
22}
23
24impl Value {
25    pub fn get(&self, key: &str) -> Option<&Self> {
26        self.as_table()?.get(key)
27    }
28
29    pub fn as_table(&self) -> Option<&BTreeMap<String, Value>> {
30        match self {
31            Self::Table(table) => Some(table),
32            _ => None,
33        }
34    }
35
36    pub fn as_array(&self) -> Option<&Vec<Value>> {
37        match self {
38            Self::Array(array) => Some(array),
39            _ => None,
40        }
41    }
42
43    pub fn as_str(&self) -> Option<&str> {
44        match self {
45            Self::String(value) => Some(value),
46            _ => None,
47        }
48    }
49
50    pub fn as_integer(&self) -> Option<i64> {
51        match self {
52            Self::Integer(value) => Some(*value),
53            _ => None,
54        }
55    }
56
57    #[allow(dead_code)]
58    pub fn as_float(&self) -> Option<f64> {
59        match self {
60            Self::Float(value) => Some(*value),
61            _ => None,
62        }
63    }
64
65    pub fn as_bool(&self) -> Option<bool> {
66        match self {
67            Self::Boolean(value) => Some(*value),
68            _ => None,
69        }
70    }
71}

73impl Index<&str> for Value {
74    type Output = Value;

76    fn index(&self, key: &str) -> &Self::Output {
77        self.get(key)
78            .unwrap_or_else(|| panic!("missing manifest key `{key}`"))
79    }

82impl FromStr for Value {
83    type Err = String;
84
85    fn from_str(source: &str) -> Result<Self, Self::Err> {
86        let yaml: serde_yaml::Value =
87            serde_yaml::from_str(source).map_err(|error| error.to_string())?;
88        convert(yaml)
89    }
90}
91
92fn convert(value: serde_yaml::Value) -> Result<Value, String> {
93    match value {
94        serde_yaml::Value::Null => Ok(Value::Null),
95        serde_yaml::Value::Bool(value) => Ok(Value::Boolean(value)),
96        serde_yaml::Value::Number(value) => {
97            if let Some(integer) = value.as_i64() {
98                Ok(Value::Integer(integer))
99            } else if let Some(float) = value.as_f64() {
100                Ok(Value::Float(float))
101            } else {
102                Err(format!("unsupported YAML number: {value}"))
103            }
104        }
105        serde_yaml::Value::String(value) => Ok(Value::String(value)),
106        serde_yaml::Value::Sequence(values) => values
107            .into_iter()
108            .map(convert)
109            .collect::<Result<Vec<_>, _>>()
110            .map(Value::Array),
111        serde_yaml::Value::Mapping(values) => {
112            let mut table = BTreeMap::new();
113            for (key, value) in values {
114                let serde_yaml::Value::String(key) = key else {
115                    return Err("manifest mapping keys must be strings".to_string());
116                };
117                if table.insert(key.clone(), convert(value)?).is_some() {
118                    return Err(format!("duplicate manifest key `{key}`"));
119                }
120            }
121            Ok(Value::Table(table))
122        }
123        serde_yaml::Value::Tagged(_) => Err("YAML tags are not allowed in manifests".to_string()),
124    }
125}

127#[cfg(test)]
128#[allow(dead_code)]
129pub fn from_toml(value: toml::Value) -> Value {
130    match value {
131        toml::Value::String(value) => Value::String(value),
132        toml::Value::Integer(value) => Value::Integer(value),
133        toml::Value::Float(value) => Value::Float(value),
134        toml::Value::Boolean(value) => Value::Boolean(value),
135        toml::Value::Datetime(value) => Value::String(value.to_string()),
136        toml::Value::Array(values) => Value::Array(values.into_iter().map(from_toml).collect()),
137        toml::Value::Table(values) => Value::Table(
138            values
139                .into_iter()
140                .map(|(key, value)| (key, from_toml(value)))
141                .collect(),
142        ),
143    }
144}

150    #[test]
151    fn parses_nested_backend_reasons() {
152        let value: Value = r#"
153test:
154  modes:
155    verify:
156      ci: false
157      backends_enabled: [ptrace, sabre]
158      backends_disabled:
159        dbt: reason
160        kvm: reason
161        liteinst: reason
162"#
163        .parse()
164        .unwrap();
165        assert_eq!(
166            value
167                .get("test")
168                .and_then(|value| value.get("modes"))
169                .and_then(|value| value.get("verify"))
170                .and_then(|value| value.get("backends_disabled"))
171                .and_then(|value| value.get("kvm"))
172                .and_then(Value::as_str),
173            Some("reason")
174        );
175    }
