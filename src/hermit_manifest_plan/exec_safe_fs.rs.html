1//! Create files that this process, or a child of it, will execute.
2//!
3//! Linux refuses to execute a file while any process holds it open for
4//! writing (ETXTBSY). Rust opens files close-on-exec, but that does not close
5//! the window: a sibling thread that spawns while this thread still holds the
6//! write descriptor forks a child that owns a duplicate until the child itself
7//! execs. The parallel libtest process for this crate spawns constantly, so a
8//! fixture written and then run in-process is refused intermittently
9//! (dh-lnslwh: `apply_harvest_refuses_case_set_and_name_set_violations` failed
10//! listing its freshly written Cargo twin). scripts/lib/exec_safe_fs.rs
11//! measured the same hazard: with eight threads spawning `true`, 79, 95 and
12//! 134 of 2000 freshly written executables were busy.
13//!
14//! Writing to a temporary name and renaming does not help: the rename keeps
15//! the inode, and the stray child's descriptor names the inode. Nothing here
16//! opens an executable target for writing in this process. A short-lived
17//! `install` child, which never forks, creates it instead, so no descriptor
18//! any other child can inherit ever refers to it. GNU `install` unlinks an
19//! existing destination first, so replacing an executable also yields a new
20//! inode that no earlier descriptor names.
21
22use std::io;
23use std::os::unix::fs::PermissionsExt;
24use std::path::Path;
25use std::process::Command;
26use std::sync::atomic::AtomicU64;
27use std::sync::atomic::Ordering;
28
29/// Write `contents` to `path` as a file with `mode`, creating or replacing it
30/// without this process ever holding the final inode open for writing.
31pub fn write_executable(
32    path: impl AsRef<Path>,
33    contents: impl AsRef<[u8]>,
34    mode: u32,
35) -> io::Result<()> {
36    static NEXT: AtomicU64 = AtomicU64::new(0);
37    let path = path.as_ref();
38    let name = path
39        .file_name()
40        .ok_or_else(|| io::Error::other(format!("{} names no file", path.display())))?;
41    // The staging inode is written here but never executed.
42    let staging = path.with_file_name(format!(
43        ".{}.staging.{}.{}",
44        name.to_string_lossy(),
45        std::process::id(),
46        NEXT.fetch_add(1, Ordering::Relaxed)
47    ));
48    std::fs::write(&staging, contents)?;
49    let installed = install(&staging, path, mode);
50    let removed = std::fs::remove_file(&staging);
51    installed?;
52    removed
53}
54
55/// `std::fs::copy`, except that an executable source is copied by a child.
56pub fn copy(from: impl AsRef<Path>, to: impl AsRef<Path>) -> io::Result<u64> {
57    let (from, to) = (from.as_ref(), to.as_ref());
58    let metadata = std::fs::metadata(from)?;
59    let mode = metadata.permissions().mode() & 0o7777;
60    if mode & 0o111 == 0 {
61        return std::fs::copy(from, to);
62    }
63    install(from, to, mode)?;
64    Ok(metadata.len())
65}
66
67fn install(from: &Path, to: &Path, mode: u32) -> io::Result<()> {
68    let output = Command::new("install")
69        .arg("-m")
70        .arg(format!("{mode:o}"))
71        .arg("-T")
72        .arg("--")
73        .arg(from)
74        .arg(to)
75        .output()?;
76    if output.status.success() {
77        return Ok(());
78    }
79    Err(io::Error::other(format!(
80        "install {} -> {} failed ({}): {}",
81        from.display(),
82        to.display(),
83        output.status,
84        String::from_utf8_lossy(&output.stderr).trim()
85    )))
86}

101    const SCRIPT: &str = "#!/bin/sh\nexit 0\n";
102
103    /// A fresh directory under the temporary root, removed when dropped.
104    struct Scratch(PathBuf);
105
106    impl Scratch {
107        fn new(label: &str) -> Self {
108            static NEXT: AtomicU64 = AtomicU64::new(0);
109            let path = std::env::temp_dir().join(format!(
110                "exec-safe-fs-{label}-{}-{}",
111                std::process::id(),
112                NEXT.fetch_add(1, Ordering::Relaxed)
113            ));
114            std::fs::create_dir(&path).unwrap();
115            Self(path)
116        }
117    }
118
119    impl Drop for Scratch {
120        fn drop(&mut self) {
121            let _ = std::fs::remove_dir_all(&self.0);
122        }
123    }

125    /// Negative control: the hazard is real on this host. A write descriptor
126    /// that a live child holds, exactly what a sibling's fork leaves behind,
127    /// makes the file unexecutable until that child is gone.
128    #[test]
129    fn a_write_descriptor_held_by_a_child_makes_the_file_busy() {
130        let dir = Scratch::new("busy");
131        let path = dir.0.join("busy");
132        let file = std::fs::File::create(&path).unwrap();
133        (&file).write_all(SCRIPT.as_bytes()).unwrap();
134        file.set_permissions(std::fs::Permissions::from_mode(0o700))
135            .unwrap();
136        let mut holder = Command::new("sleep")
137            .arg("30")
138            .stdin(Stdio::from(file))
139            .spawn()
140            .unwrap();
141        let busy = Command::new(&path).status();
142        holder.kill().unwrap();
143        holder.wait().unwrap();
144        assert_eq!(
145            busy.map_err(|error| error.kind()).err(),
146            Some(io::ErrorKind::ExecutableFileBusy)
147        );
148        // With the holder gone the file runs. This test wrote it in-process,
149        // so a sibling test's spawn may still hold a copy for the instant
150        // before that child execs; that is the hazard itself, so retry briefly.
151        let mut last = None;
152        for _ in 0..200 {
153            match Command::new(&path).status() {
154                Err(error) if error.kind() == io::ErrorKind::ExecutableFileBusy => {
155                    last = Some(error);
156                    std::thread::sleep(std::time::Duration::from_millis(5));
157                }
158                other => {
159                    assert!(other.unwrap().success());
160                    return;
161                }
162            }
163        }
164        panic!("still busy after the holder exited: {last:?}");
165    }

167    #[test]
168    fn helpers_keep_mode_and_bytes_and_leave_nothing_beside_the_target() {
169        let dir = Scratch::new("shape");
170        let executable = dir.0.join("executable");
171        write_executable(&executable, SCRIPT, 0o750).unwrap();
172        assert_eq!(std::fs::read(&executable).unwrap(), SCRIPT.as_bytes());
173        assert_eq!(
174            std::fs::metadata(&executable).unwrap().mode() & 0o7777,
175            0o750
176        );
177        // Replacing an executable yields a new inode, so no descriptor to the
178        // old one, held here or inherited by any child, names the new file.
179        let first_inode = std::fs::metadata(&executable).unwrap().ino();
180        write_executable(&executable, "#!/bin/sh\nexit 3\n", 0o755).unwrap();
181        let replaced = std::fs::metadata(&executable).unwrap();
182        assert_ne!(replaced.ino(), first_inode, "replacement reused the inode");
183        assert_eq!(replaced.mode() & 0o7777, 0o755);
184        write_executable(&executable, SCRIPT, 0o750).unwrap();
185        let copied = dir.0.join("copied");
186        assert_eq!(copy(&executable, &copied).unwrap(), SCRIPT.len() as u64);
187        assert_eq!(std::fs::read(&copied).unwrap(), SCRIPT.as_bytes());
188        assert_eq!(std::fs::metadata(&copied).unwrap().mode() & 0o7777, 0o750);
189        // Copying over an existing executable yields a new inode too.
190        let copied_inode = std::fs::metadata(&copied).unwrap().ino();
191        copy(&executable, &copied).unwrap();
192        assert_ne!(
193            std::fs::metadata(&copied).unwrap().ino(),
194            copied_inode,
195            "copy over an executable reused the inode"
196        );
197        let plain = dir.0.join("plain");
198        std::fs::write(&plain, "data\n").unwrap();
199        std::fs::set_permissions(&plain, std::fs::Permissions::from_mode(0o640)).unwrap();
200        let plain_copy = dir.0.join("plain-copy");
201        assert_eq!(copy(&plain, &plain_copy).unwrap(), 5);
202        assert_eq!(
203            std::fs::metadata(&plain_copy).unwrap().mode() & 0o7777,
204            0o640
205        );
206        // A failed write is an error, not a silent partial file.
207        let missing = dir.0.join("no/such/dir/target");
208        assert!(write_executable(&missing, SCRIPT, 0o700).is_err());
209        assert!(!missing.exists());
210        // Only the final files remain: staging is removed.
211        let names = std::fs::read_dir(&dir.0)
212            .unwrap()
213            .map(|entry| entry.unwrap().file_name().into_string().unwrap())
214            .collect::<std::collections::BTreeSet<_>>();
215        assert_eq!(
216            names,
217            ["copied", "executable", "plain", "plain-copy"]
218                .into_iter()
219                .map(String::from)
220                .collect()
221        );
222    }
