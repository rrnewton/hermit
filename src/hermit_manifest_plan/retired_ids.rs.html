7//! Retired E2E manifest test ids and the live ids that replaced them.
8//!
9//! `tests/e2e/manifests/inventory/retired-ids.json` records every test id that
10//! no manifest declares any more because its program moved to a different id,
11//! for example when the backend-parity-c bucket was folded into c-programs
12//! (<https://github.com/rrnewton/hermit/issues/3301>, slice S6). History is
13//! keyed by test id, so a reader that must join a row recorded under an old id
14//! with a row recorded under the new one resolves the old id through this file
15//! rather than through a string rewrite of its own.
16//!
17//! The file must stay a bijection from its retired ids onto live ids: every
18//! retired id is absent from the manifests, every successor is declared by one,
19//! and no two retired ids share a successor. [`RetiredIds::check_live`] is the
20//! check; the manifest expander runs it on every metadata validation.
21//!
22//! A successor must also be the retired id's own test: either the plain
23//! bucket-prefix rename (`old-bucket/name` becomes `new-bucket/name`) or a
24//! rename listed, with its reason, in the retirement's `collisions`.
25//! [`RetiredIds::parse`] refuses any other successor, so a map that points a
26//! retired id at an unrelated live test is refused even though that test is
27//! live and unshared.
28
29use std::collections::BTreeMap;
30use std::collections::BTreeSet;
31use std::path::Path;
32
33use serde::Deserialize;

35/// Repository-relative path of the retired-id map.
36pub const RETIRED_IDS_FILE: &str = "tests/e2e/manifests/inventory/retired-ids.json";
37
38/// The only schema this reader accepts.
39pub const RETIRED_IDS_SCHEMA: u64 = 1;
40
41#[derive(Clone, Debug, Deserialize, PartialEq, Eq)]
42#[serde(deny_unknown_fields)]
43pub struct RetiredIds {
44    pub schema: u64,
45    pub description: String,
46    pub retirements: Vec<Retirement>,
47}
48
49/// One retired bucket and the id each of its tests now carries.
50#[derive(Clone, Debug, Deserialize, PartialEq, Eq)]
51#[serde(deny_unknown_fields)]
52pub struct Retirement {
53    /// The manifest bucket that no longer exists.
54    pub retired_bucket: String,
55    /// The live bucket whose manifest now declares every successor id.
56    pub successor_bucket: String,
57    /// The last Hermit commit whose manifests declared the retired ids.
58    pub last_live_commit: String,
59    /// Why the ids moved, including every successor that is not a plain
60    /// bucket-prefix rename.
61    pub reason: String,
62    /// Retired id to successor id.
63    pub ids: BTreeMap<String, String>,
64    /// Every retired id whose successor is not the plain bucket-prefix rename,
65    /// mapped to why the plain rename could not be used. Required, and empty
66    /// when every successor is the plain rename.
67    pub collisions: BTreeMap<String, String>,
68}

70fn test_name<'a>(id: &'a str, bucket: &str) -> Option<&'a str> {
71    id.strip_prefix(bucket)
72        .and_then(|rest| rest.strip_prefix('/'))
73        .filter(|name| {
74            !name.is_empty()
75                && name
76                    .bytes()
77                    .all(|byte| byte.is_ascii_lowercase() || byte.is_ascii_digit() || byte == b'-')
78                && !name.starts_with('-')
79                && !name.ends_with('-')
80        })
81}

83impl RetiredIds {
84    /// Read and structurally validate the repository's retired-id map.
85    pub fn load(root: &Path) -> Result<Self, String> {
86        let path = root.join(RETIRED_IDS_FILE);
87        let text = std::fs::read_to_string(&path)
88            .map_err(|error| format!("cannot read {}: {error}", path.display()))?;
89        Self::parse(&text).map_err(|error| format!("{}: {error}", path.display()))
90    }
91
92    /// Parse and structurally validate a retired-id map.
93    ///
94    /// This refuses anything that would make a historical id resolve
95    /// ambiguously: an unknown schema, an empty retirement, a retired id outside
96    /// its retired bucket, a successor outside its successor bucket, a retired
97    /// id listed twice, two retired ids sharing a successor, or a successor that
98    /// is itself retired (a chain that a single lookup would leave half-done).
99    ///
100    /// It also refuses a successor that is not the retired id's own test: a
101    /// successor must be the plain bucket-prefix rename unless the retired id is
102    /// listed in `collisions` with a reason, and a listed collision must be a
103    /// retired id of the same retirement whose successor is not the plain
104    /// rename.
105    pub fn parse(text: &str) -> Result<Self, String> {
106        let map: Self = serde_json::from_str(text)
107            .map_err(|error| format!("invalid retired-id map: {error}"))?;
108        if map.schema != RETIRED_IDS_SCHEMA {
109            return Err(format!(
110                "retired-id map schema must be {RETIRED_IDS_SCHEMA}, found {}",
111                map.schema
112            ));
113        }
114        if map.description.trim().is_empty() {
115            return Err("retired-id map has no description".into());
116        }
117        if map.retirements.is_empty() {
118            return Err("retired-id map has no retirement".into());
119        }
120        let mut retired = BTreeSet::new();
121        let mut successors = BTreeSet::new();
122        let mut buckets = BTreeSet::new();
123        for retirement in &map.retirements {
124            let bucket = &retirement.retired_bucket;
125            if bucket.is_empty() || retirement.successor_bucket.is_empty() {
126                return Err("a retirement names an empty bucket".into());
127            }
128            if bucket == &retirement.successor_bucket {
129                return Err(format!("retirement of {bucket} names itself as successor"));
130            }
131            if !buckets.insert(bucket.clone()) {
132                return Err(format!("bucket {bucket} is retired twice"));
133            }
134            if retirement.last_live_commit.len() != 40
135                || !retirement
136                    .last_live_commit
137                    .bytes()
138                    .all(|byte| byte.is_ascii_digit() || (b'a'..=b'f').contains(&byte))
139            {
140                return Err(format!(
141                    "retirement of {bucket} needs a full lowercase last_live_commit SHA"
142                ));
143            }
144            if retirement.reason.trim().is_empty() {
145                return Err(format!("retirement of {bucket} gives no reason"));
146            }
147            if retirement.ids.is_empty() {
148                return Err(format!("retirement of {bucket} retires no id"));
149            }
150            for (old, reason) in &retirement.collisions {
151                if !retirement.ids.contains_key(old) {
152                    return Err(format!(
153                        "collision {old:?} is not a retired id of bucket {bucket}"
154                    ));
155                }
156                if reason.trim().is_empty() {
157                    return Err(format!("collision {old:?} gives no reason"));
158                }
159            }
160            for (old, new) in &retirement.ids {
161                let Some(old_name) = test_name(old, bucket) else {
162                    return Err(format!(
163                        "retired id {old:?} is not a test of bucket {bucket}"
164                    ));
165                };
166                let Some(new_name) = test_name(new, &retirement.successor_bucket) else {
167                    return Err(format!(
168                        "successor {new:?} of {old:?} is not a test of bucket {}",
169                        retirement.successor_bucket
170                    ));
171                };
172                let listed = retirement.collisions.contains_key(old);
173                if old_name == new_name && listed {
174                    return Err(format!(
175                        "collision {old:?} maps to the plain prefix rename {new:?}; list only successors that differ from it"
176                    ));
177                }
178                if old_name != new_name && !listed {
179                    return Err(format!(
180                        "successor {new:?} of {old:?} is neither the prefix rename {}/{old_name} nor a listed collision",
181                        retirement.successor_bucket
182                    ));
183                }
184                if !retired.insert(old.clone()) {
185                    return Err(format!("retired id {old:?} is listed twice"));
186                }
187                if !successors.insert(new.clone()) {
188                    return Err(format!("successor {new:?} is shared by two retired ids"));
189                }
190            }
191        }
192        if let Some(chained) = retired.intersection(&successors).next() {
193            return Err(format!(
194                "{chained:?} is both a retired id and a successor; record the final id directly"
195            ));
196        }
197        Ok(map)
198    }
199
200    /// The live id that carries `id`'s history: its successor if `id` is
201    /// retired, otherwise `id` itself.
202    pub fn resolve<'a>(&'a self, id: &'a str) -> &'a str {
203        self.successor(id).unwrap_or(id)
204    }
205
206    /// The successor of a retired id, or `None` if `id` is not retired.
207    pub fn successor(&self, id: &str) -> Option<&str> {
208        self.retirements
209            .iter()
210            .find_map(|retirement| retirement.ids.get(id).map(String::as_str))
211    }
212
213    /// The retirement of `bucket`, if that bucket was retired.
214    pub fn retirement(&self, bucket: &str) -> Option<&Retirement> {
215        self.retirements
216            .iter()
217            .find(|retirement| retirement.retired_bucket == bucket)
218    }
219
220    /// Every successor id of the retired `bucket`, or an error if the bucket
221    /// was never retired.
222    pub fn successors_of(&self, bucket: &str) -> Result<BTreeSet<String>, String> {
223        self.retirement(bucket)
224            .map(|retirement| retirement.ids.values().cloned().collect())
225            .ok_or_else(|| format!("bucket {bucket} has no retirement in {RETIRED_IDS_FILE}"))
226    }
227
228    /// Check the map against the ids and buckets the manifests declare now.
229    ///
230    /// No retired bucket may be live, no retired id may be live, and every
231    /// successor must be live in its successor bucket. With the structural
232    /// checks in [`RetiredIds::parse`], this makes the map a bijection from its
233    /// retired ids onto a set of live ids.
234    pub fn check_live(
235        &self,
236        live_ids: &BTreeSet<String>,
237        live_buckets: &BTreeSet<String>,
238    ) -> Result<(), String> {
239        for retirement in &self.retirements {
240            if live_buckets.contains(&retirement.retired_bucket) {
241                return Err(format!(
242                    "retired bucket {} still has a manifest",
243                    retirement.retired_bucket
244                ));
245            }
246            if !live_buckets.contains(&retirement.successor_bucket) {
247                return Err(format!(
248                    "successor bucket {} of {} has no manifest",
249                    retirement.successor_bucket, retirement.retired_bucket
250                ));
251            }
252            for (old, new) in &retirement.ids {
253                if live_ids.contains(old) {
254                    return Err(format!(
255                        "retired id {old:?} is still declared by a manifest"
256                    ));
257                }
258                if !live_ids.contains(new) {
259                    return Err(format!(
260                        "successor {new:?} of retired id {old:?} is not declared by any manifest"
261                    ));
262                }
263            }
264        }
265        Ok(())
266    }
267}

273    fn pairs(items: &[(&str, &str)]) -> BTreeMap<String, String> {
274        items
275            .iter()
276            .map(|(k, v)| (k.to_string(), v.to_string()))
277            .collect()
278    }
279
280    fn map_with(ids: &[(&str, &str)], collisions: &[(&str, &str)]) -> String {
281        serde_json::json!({
282            "schema": 1,
283            "description": "test map",
284            "retirements": [{
285                "retired_bucket": "old",
286                "successor_bucket": "new",
287                "last_live_commit": "a".repeat(40),
288                "reason": "folded",
289                "ids": pairs(ids),
290                "collisions": pairs(collisions),
291            }],
292        })
293        .to_string()
294    }
295
296    fn map(ids: &[(&str, &str)]) -> String {
297        map_with(ids, &[])
298    }
299
300    fn set(items: &[&str]) -> BTreeSet<String> {
301        items.iter().map(|item| item.to_string()).collect()
302    }

304    #[test]
305    fn resolves_retired_ids_and_passes_live_ids_through() {
306        let ids = RetiredIds::parse(&map_with(
307            &[("old/a", "new/a"), ("old/b", "new/b-pair")],
308            &[("old/b", "new/b names a different program")],
309        ))
310        .unwrap();
311        assert_eq!(ids.resolve("old/a"), "new/a");
312        assert_eq!(ids.resolve("old/b"), "new/b-pair");
313        assert_eq!(ids.resolve("new/a"), "new/a");
314        assert_eq!(ids.resolve("other/x"), "other/x");
315        assert_eq!(
316            ids.successors_of("old").unwrap(),
317            set(&["new/a", "new/b-pair"])
318        );
319        assert!(ids.successors_of("new").is_err());
320        ids.check_live(&set(&["new/a", "new/b-pair", "new/c"]), &set(&["new"]))
321            .unwrap();
322    }

324    #[test]
325    fn refuses_every_non_bijective_or_ambiguous_map() {
326        for (ids, collisions, needle) in [
327            (
328                vec![("old/a", "new/x"), ("old/b", "new/x")],
329                vec![("old/a", "new/a is taken"), ("old/b", "new/b is taken")],
330                "shared by two",
331            ),
332            (
333                vec![("other/a", "new/a")],
334                vec![],
335                "not a test of bucket old",
336            ),
337            (
338                vec![("old/a", "other/a")],
339                vec![],
340                "not a test of bucket new",
341            ),
342            (vec![("old/A", "new/a")], vec![], "not a test of bucket old"),
343            (vec![], vec![], "retires no id"),
344            // A successor that is neither the prefix rename nor a listed
345            // collision: a retired id pointed at an unrelated test.
346            (
347                vec![("old/a", "new/a"), ("old/b", "new/unrelated")],
348                vec![],
349                "neither the prefix rename new/b nor a listed collision",
350            ),
351            // A listed collision whose successor is the plain rename.
352            (
353                vec![("old/a", "new/a")],
354                vec![("old/a", "new/a is taken")],
355                "maps to the plain prefix rename",
356            ),
357            // A listed collision that the retirement does not retire.
358            (
359                vec![("old/a", "new/a-pair")],
360                vec![("old/a", "new/a is taken"), ("old/z", "new/z is taken")],
361                "is not a retired id of bucket old",
362            ),
363            // A listed collision with no reason.
364            (
365                vec![("old/a", "new/a-pair")],
366                vec![("old/a", "  ")],
367                "gives no reason",
368            ),
369        ] {
370            let error = RetiredIds::parse(&map_with(&ids, &collisions)).unwrap_err();
371            assert!(error.contains(needle), "{error}");
372        }
373        let mut undeclared: serde_json::Value =
374            serde_json::from_str(&map(&[("old/a", "new/a")])).unwrap();
375        undeclared["retirements"][0]
376            .as_object_mut()
377            .unwrap()
378            .remove("collisions");
379        assert!(
380            RetiredIds::parse(&undeclared.to_string())
381                .unwrap_err()
382                .contains("missing field `collisions`")
383        );
384        let mut chained: serde_json::Value =
385            serde_json::from_str(&map(&[("old/a", "new/a")])).unwrap();
386        chained["retirements"]
387            .as_array_mut()
388            .unwrap()
389            .push(serde_json::json!({
390                "retired_bucket": "new", "successor_bucket": "newer",
391                "last_live_commit": "b".repeat(40), "reason": "again",
392                "ids": {"new/a": "newer/a"}, "collisions": {},
393            }));
394        let error = RetiredIds::parse(&chained.to_string()).unwrap_err();
395        assert!(
396            error.contains("both a retired id and a successor"),
397            "{error}"
398        );
399        let mut unknown: serde_json::Value =
400            serde_json::from_str(&map(&[("old/a", "new/a")])).unwrap();
401        unknown["extra"] = serde_json::json!(true);
402        assert!(RetiredIds::parse(&unknown.to_string()).is_err());
403        let mut short: serde_json::Value =
404            serde_json::from_str(&map(&[("old/a", "new/a")])).unwrap();
405        short["retirements"][0]["last_live_commit"] = serde_json::json!("abc");
406        assert!(
407            RetiredIds::parse(&short.to_string())
408                .unwrap_err()
409                .contains("last_live_commit")
410        );
411    }
