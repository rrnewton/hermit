1//! One retry-count schema from innermost test to run ledger (owner policy,
2//! 2026-10-30): every failing test gets a bounded retry at the innermost
3//! harness; classification labels attempts but does not gate the retry.
4//!
5//! This module only defines and derives the record. It changes no retry
6//! behaviour: producers emit [`RetryRecordV1`], and the ledger carries
7//! [`RetryRollupV1`] beside `framework_retries_v1`.
8
9use super::*;
10use crate::runner::MAX_ATTEMPTS_PER_CELL;

12/// Per-test retry record emitted by every innermost harness (e2e
13/// test-harness, nextest wrapper, Buck ingest) in the same shape.
14#[derive(Clone, Debug, PartialEq, Eq, Deserialize, Serialize)]
15#[serde(deny_unknown_fields)]
16pub struct RetryRecordV1 {
17    pub test: String,
18    pub mode: String,
19    pub backend: Option<String>,
20    pub attempts: u64,
21    pub retries: u64,
22    pub attempt_outcomes: Vec<String>,
23    /// Diagnosis class per attempt (e.g. `product_failure`, `pass`);
24    /// never used to decide whether a retry happened.
25    pub attempt_classes: Vec<Option<String>>,
26    /// Error kind per attempt (e.g. `infrastructure` for an
27    /// understood-infrastructure signature, `timeout` for no_result);
28    /// the signature half of the diagnosis, recorded beside the class so
29    /// a later flakiness investigation can judge the infra classifier.
30    #[serde(default)]
31    pub attempt_error_kinds: Vec<Option<String>>,
32    /// Named cause per attempt (`skid_overshoot`, `host_input_changed`,
33    /// or the error kind), so skid-overshoot keeps its own bucket,
34    /// separate from other understood-infrastructure causes.
35    #[serde(default)]
36    pub attempt_signatures: Vec<Option<String>>,
37    pub final_outcome: String,
38    pub flaky_pass: bool,
39}

41/// Run-level rollup recorded in the ledger.
42#[derive(Clone, Debug, PartialEq, Eq, Deserialize, Serialize)]
43#[serde(deny_unknown_fields)]
44pub struct RetryRollupV1 {
45    pub schema: u32,
46    /// The generic e2e bound, from `MAX_ATTEMPTS_PER_CELL`. A skid-only
47    /// history may run to the separate skid budget
48    /// (`MAX_SKID_ATTEMPTS_PER_CELL`, the shared
49    /// `detcore_model::SKID_RETRY_MAX_ATTEMPTS`); such a record's own
50    /// attempt count says so, not this field.
51    pub max_attempts_per_test: u64,
52    pub total_retries: u64,
53    pub retried_tests: u64,
54    pub flaky_passes: u64,
55    /// Retries (failed attempts beyond none; one per failed attempt) by
56    /// that attempt's failure class, for classifier-trust reporting.
57    /// Diagnosis only: it does not change the landing signal.
58    #[serde(default)]
59    pub retries_by_class: std::collections::BTreeMap<String, u64>,
60    /// Retries grouped two levels: by failure class (the four classes,
61    /// unchanged), in `retries_by_class`, and within a class by
62    /// subcategory (`retries_by_subcategory`, class -> subcategory ->
63    /// count). Understood-infrastructure subcategories are the stable
64    /// typed signatures: `pmu_skid`, `host_input_changed`,
65    /// `sandbox_denied`.
66    #[serde(default)]
67    pub retries_by_subcategory:
68        std::collections::BTreeMap<String, std::collections::BTreeMap<String, u64>>,
69    /// Flaky passes whose every failed attempt was
70    /// `understood_infrastructure_failure`. Reported separately so the
71    /// owner can later decide whether these count as clean; today they
72    /// still make the run green-flaky like any other retry.
73    #[serde(default)]
74    pub infra_recovered: u64,
75    /// Zero retries run-wide is the clean score.
76    pub clean: bool,
77    pub tests: Vec<RetryRecordV1>,
78}

80impl RetryRollupV1 {
81    pub fn from_records(tests: Vec<RetryRecordV1>) -> Self {
82        let total_retries = tests.iter().map(|t| t.retries).sum();
83        let retried_tests = tests.iter().filter(|t| t.retries > 0).count() as u64;
84        let flaky_passes = tests.iter().filter(|t| t.flaky_pass).count() as u64;
85        let mut retries_by_class: std::collections::BTreeMap<String, u64> = Default::default();
86        let mut retries_by_subcategory: std::collections::BTreeMap<
87            String,
88            std::collections::BTreeMap<String, u64>,
89        > = Default::default();
90        let mut infra_recovered = 0u64;
91        for test in &tests {
92            for (i, (class, outcome)) in test
93                .attempt_classes
94                .iter()
95                .zip(&test.attempt_outcomes)
96                .enumerate()
97            {
98                if outcome != "PASS" {
99                    if let Some(class) = class {
100                        *retries_by_class.entry(class.clone()).or_default() += 1;
101                    }
102                    let signature = test.attempt_signatures.get(i).and_then(|s| s.as_deref());
103                    // The four classes stay as they are; skid is a
104                    // subcategory of understood infrastructure, not a
105                    // fifth class. Nextest skid records carry the legacy
106                    // class label `skid_overshoot`; normalize it here.
107                    let class = match class.as_deref() {
108                        Some("skid_overshoot") => {
109                            Some("understood_infrastructure_failure".to_string())
110                        }
111                        other => other.map(str::to_string),
112                    };
113                    let subcategory = match signature {
114                        Some("skid_overshoot") => "pmu_skid",
115                        Some("host_input_changed") => "host_input_changed",
116                        Some("sandbox-denied") | Some("sandbox_denied") => "sandbox_denied",
117                        Some(other) => other,
118                        None => "unspecified",
119                    }
120                    .to_string();
121                    let class_key = class.unwrap_or_else(|| "unclassified".to_string());
122                    *retries_by_subcategory
123                        .entry(class_key)
124                        .or_default()
125                        .entry(subcategory)
126                        .or_default() += 1;
127                }
128            }
129            let failed_classes: Vec<&String> = test
130                .attempt_classes
131                .iter()
132                .zip(&test.attempt_outcomes)
133                .filter(|(_, outcome)| *outcome != "PASS")
134                .filter_map(|(class, _)| class.as_ref())
135                .collect();
136            if test.flaky_pass
137                && !failed_classes.is_empty()
138                && failed_classes
139                    .iter()
140                    .all(|c| c.as_str() == "understood_infrastructure_failure")
141            {
142                infra_recovered += 1;
143            }
144        }
145        Self {
146            schema: 1,
147            max_attempts_per_test: MAX_ATTEMPTS_PER_CELL,
148            total_retries,
149            retried_tests,
150            flaky_passes,
151            retries_by_class,
152            retries_by_subcategory,
153            infra_recovered,
154            clean: total_retries == 0,
155            tests,
156        }
157    }

159    /// Derive the rollup from the existing framework-retries census.
160    /// Attempts beyond the first are retries; a retried test whose final
161    /// outcome is PASS is a flaky pass.
162    pub fn from_framework_retries(retries: &FrameworkRetriesV1) -> Self {
163        // Per-attempt values come from the census's retained vectors
164        // when it carries them (one entry per attempt, from the raw
165        // rows). Records written before those fields retain only the
166        // first attempt: for those, first+final is the whole history at
167        // the generic bound of 2, and a longer legacy record's
168        // intermediate attempts are marked unknown (empty outcome, no
169        // class/signature/kind) rather than reconstructed from the
170        // first attempt — copying the first attempt's class onto a
171        // later product attempt would mislabel it as infrastructure.
172        fn retained_or_legacy<T: Clone>(
173            cell: &RetriedCellV1,
174            retained: &[T],
175            first: T,
176            final_at_two: T,
177            legacy_final: T,
178            unknown: T,
179        ) -> Vec<T> {
180            if retained.len() as u64 == cell.attempts {
181                return retained.to_vec();
182            }
183            if cell.attempts <= 1 {
184                return vec![first];
185            }
186            if cell.attempts == 2 {
187                return vec![first, final_at_two];
188            }
189            let mut values = vec![unknown; cell.attempts as usize];
190            values[0] = first;
191            *values.last_mut().expect("attempts >= 3") = legacy_final;
192            values
193        }
194        let tests = retries
195            .cells
196            .iter()
197            .map(|cell| RetryRecordV1 {
198                test: cell.cell.test.clone(),
199                mode: cell.cell.mode.clone(),
200                backend: Some(cell.cell.backend.clone()),
201                attempts: cell.attempts,
202                retries: cell.attempts.saturating_sub(1),
203                attempt_outcomes: retained_or_legacy(
204                    cell,
205                    &cell.attempt_outcomes,
206                    cell.first_outcome.clone(),
207                    cell.final_outcome.clone(),
208                    cell.final_outcome.clone(),
209                    String::new(),
210                ),
211                attempt_classes: retained_or_legacy(
212                    cell,
213                    &cell.attempt_failure_classes,
214                    cell.first_failure_class.clone(),
215                    if cell.final_outcome == "PASS" {
216                        None
217                    } else {
218                        cell.first_failure_class.clone()
219                    },
220                    None,
221                    None,
222                ),
223                attempt_signatures: retained_or_legacy(
224                    cell,
225                    &cell.attempt_signatures,
226                    cell.first_signature.clone(),
227                    if cell.final_outcome == "PASS" {
228                        None
229                    } else {
230                        cell.first_signature.clone()
231                    },
232                    None,
233                    None,
234                ),
235                attempt_error_kinds: retained_or_legacy(
236                    cell,
237                    &cell.attempt_error_kinds,
238                    cell.first_error_kind.clone(),
239                    if cell.final_outcome == "PASS" {
240                        None
241                    } else {
242                        cell.first_error_kind.clone()
243                    },
244                    None,
245                    None,
246                ),
247                final_outcome: cell.final_outcome.clone(),
248                flaky_pass: cell.final_outcome == "PASS",
249            })
250            .collect();
251        Self::from_records(tests)
252    }
253
254    pub fn from_inputs(inputs: &BTreeMap<String, Vec<u8>>) -> Result<Self, String> {
255        Ok(Self::from_framework_retries(
256            &FrameworkRetriesV1::from_inputs(inputs)?,
257        ))
258    }
259}

261/// The landing signal for a validate run (Step 4): red on any product
262/// failure or an incomplete run; otherwise green-clean when no retry ran,
263/// green-flaky (still landing-eligible) with the top flaky tests named.
264pub fn landing_signal(
265    product_failures: usize,
266    validation_complete: bool,
267    run_succeeded: bool,
268    rollup: &RetryRollupV1,
269    extra_retries: u64,
270    extra_flaky_tests: Vec<String>,
271) -> (&'static str, Vec<String>) {
272    if product_failures > 0 || !validation_complete || !run_succeeded {
273        return ("red", Vec::new());
274    }
275    if rollup.total_retries == 0 && extra_retries == 0 {
276        return ("green-clean", Vec::new());
277    }
278    let mut flaky: Vec<&RetryRecordV1> = rollup
279        .tests
280        .iter()
281        .filter(|t| t.flaky_pass && t.retries > 0)
282        .collect();
283    flaky.sort_by(|a, b| b.retries.cmp(&a.retries).then_with(|| a.test.cmp(&b.test)));
284    let mut top: Vec<String> = flaky
285        .iter()
286        .map(|t| format!("{} ({} retries)", t.test, t.retries))
287        .collect();
288    top.extend(extra_flaky_tests);
289    // Global top by retries, parsed back out of the "(N retries)" suffix.
290    top.sort_by(|a, b| {
291        let n = |s: &str| {
292            s.rsplit('(')
293                .next()
294                .and_then(|r| r.split(' ').next())
295                .and_then(|n| n.parse::<u64>().ok())
296                .unwrap_or(0)
297        };
298        n(b).cmp(&n(a)).then_with(|| a.cmp(b))
299    });
300    top.truncate(10);
301    ("green-flaky", top)
302}

308    #[test]
309    fn rollup_counts_retries_and_clean() {
310        let clean = RetryRollupV1::from_records(vec![]);
311        assert!(clean.clean && clean.total_retries == 0);
312        assert_eq!(clean.max_attempts_per_test, MAX_ATTEMPTS_PER_CELL);
313        let r = RetryRecordV1 {
314            test: "fixture/x".into(),
315            mode: "verify".into(),
316            backend: Some("dbt".into()),
317            attempts: 2,
318            retries: 1,
319            attempt_outcomes: vec!["FAIL".into(), "PASS".into()],
320            attempt_classes: vec![Some("product_failure".into()), None],
321            attempt_error_kinds: vec![None, None],
322            attempt_signatures: vec![None, None],
323            final_outcome: "PASS".into(),
324            flaky_pass: true,
325        };
326        let rollup = RetryRollupV1::from_records(vec![r]);
327        assert!(!rollup.clean);
328        assert_eq!(
329            (
330                rollup.total_retries,
331                rollup.retried_tests,
332                rollup.flaky_passes
333            ),
334            (1, 1, 1)
335        );
336        let value = serde_json::to_value(&rollup).unwrap();
337        assert_eq!(
338            serde_json::from_value::<RetryRollupV1>(value).unwrap(),
339            rollup
340        );
341    }

390        assert!(serde_json::from_str::<RetryRecordV1>(legacy).is_ok());

395        // A mixed history (skid, skid, product FAIL) must count its

408            attempts: 3,
409            first_outcome: "ERROR".into(),
410            first_failure_class: Some("understood_infrastructure_failure".into()),

413            first_result: None,
414            first_divergent_scheduler_turn: None,
415            first_divergent_record: None,
416            first_divergent_syscall: None,
417            final_outcome: "FAIL".into(),
418            attempt_outcomes: vec!["ERROR".into(), "ERROR".into(), "FAIL".into()],

456    #[test]
457    fn landing_signal_distinguishes_clean_flaky_and_red() {
458        let clean = RetryRollupV1::from_records(vec![]);

460            landing_signal(0, true, true, &clean, 0, vec![]),
