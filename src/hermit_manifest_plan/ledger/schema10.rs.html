50    /// Freeze exactly the bytes observed by the producer before finalization.
51    /// Comparison semantics remain the existing cell/comparison readers' job.
52    pub fn from_inputs(
53        run_id: &str,
54        hermit_sha: &str,
55        inputs: &BTreeMap<String, Vec<u8>>,
56    ) -> Result<Self, String> {
57        if !nonblank_component(run_id) || !is_lower_hex(hermit_sha, 40) {
58            return Err("raw result census requires a run identity and full measured SHA".into());
59        }
60        let mut files = Vec::new();
61        let mut seen = BTreeSet::new();
62        for (path, bytes) in inputs {
63            if path
64                .split('/')
65                .any(|part| part.is_empty() || matches!(part, "." | "..") || part.contains('\0'))
66                || path.rsplit('/').next() != Some("results.jsonl")
67            {
68                return Err("raw result census path is not a relative results.jsonl path".into());
69            }
70            let text = std::str::from_utf8(bytes)
71                .map_err(|error| format!("raw result census input is not UTF-8: {error}"))?;
72            let mut rows = Vec::new();
73            for (number, line) in text.lines().enumerate() {
74                if line.trim().is_empty() {
75                    continue;
76                }
77                let row = read_schema10_source_result(line.as_bytes())?;
78                if row.get("schema").and_then(Value::as_u64) != Some(4)
79                    || row.get("run_id").and_then(Value::as_str) != Some(run_id)
80                    || row.get("hermit_sha").and_then(Value::as_str) != Some(hermit_sha)
81                    || row.get("source_tree_dirty").and_then(Value::as_bool) != Some(false)
82                {
83                    return Err(
84                        "raw result census input has a different run or clean source identity"
85                            .into(),
86                    );
87                }
88                let field = |name: &str| {
89                    row.get(name)
90                        .and_then(Value::as_str)
91                        .map(str::to_string)
92                        .ok_or_else(|| format!("raw result census input omitted {name}"))
93                };
94                let cell = CellIdentity {
95                    lane: field("lane")?,
96                    category: field("category")?,
97                    test: field("test")?,
98                    mode: field("mode")?,
99                    backend: field("backend")?,
100                };
101                validate_identity(&cell)?;
102                let attempt = row
103                    .get("attempt")
104                    .and_then(Value::as_u64)
105                    .filter(|value| *value > 0)
106                    .ok_or("raw result census input omitted a positive attempt")?;
107                if !seen.insert((cell.clone(), attempt)) {
108                    return Err("raw result census repeats a cell attempt".into());
109                }
110                rows.push(RawResultInputRowV1 {
111                    line: number as u64 + 1,
112                    cell,
113                    attempt,
114                });
115            }
116            files.push(RawResultInputFileV1 {
117                path: path.clone(),
118                bytes: bytes.len() as u64,
119                sha256: hex_digest(bytes),
120                rows,
121            });
122        }
123        Ok(Self {
124            schema: 1,
125            run_id: run_id.into(),
126            hermit_sha: hermit_sha.into(),
127            files,
128        })
129    }

131    pub fn validate_for_row(&self, row: &HistoryRow) -> Result<(), String> {
132        if self.schema != 1
133            || !matches!(row.schema_version, Some(5..=10))
134            || !nonblank_component(&self.run_id)
135            || !is_lower_hex(&self.hermit_sha, 40)
136            || row.run_id.as_deref() != Some(&self.run_id)
137            || row.commit.as_deref() != Some(&self.hermit_sha)
138            || row.tree_dirty != Some(false)
139        {
140            return Err("raw result census has an unsupported version or row identity".into());
141        }
142        let mut previous: Option<&str> = None;
143        let mut seen = BTreeSet::new();
144        for file in &self.files {
145            if file
146                .path
147                .split('/')
148                .any(|part| part.is_empty() || matches!(part, "." | "..") || part.contains('\0'))
149                || file.path.rsplit('/').next() != Some("results.jsonl")
150                || previous.is_some_and(|path| path >= file.path.as_str())
151                || !is_lower_hex(&file.sha256, 64)
152            {
153                return Err("raw result census has an invalid or repeated file identity".into());
154            }
155            previous = Some(&file.path);
156            let mut line = 0;
157            for input in &file.rows {
158                validate_identity(&input.cell)?;
159                if input.line <= line
160                    || input.attempt == 0
161                    || !seen.insert((&input.cell, input.attempt))
162                {
163                    return Err("raw result census has an invalid or repeated row identity".into());
164                }
165                line = input.line;
166            }
167        }
168        Ok(())
169    }

171    pub fn verify_inputs(
172        &self,
173        row: &HistoryRow,
174        inputs: &BTreeMap<String, Vec<u8>>,
175    ) -> Result<(), String> {
176        self.validate_for_row(row)?;
177        let actual = Self::from_inputs(&self.run_id, &self.hermit_sha, inputs)?;
178        if *self != actual {
179            return Err(
180                "current result population or bytes differ from the producer-finalized raw census"
181                    .into(),
182            );
183        }
184        Ok(())
185    }

188impl HistoryRow {
189    /// Verify finalized producer semantics against the complete retained raw
190    /// population and, when required, the original plan/cell/test artifacts.
191    /// Returns whether that population has zero rows, only after the full
192    /// selected-zero checks. Empty files remain separate census members.
193    ///
194    /// This does not establish canonical origin, select a ledger row, retain
195    /// file custody, or authorize a history proof. The canonical adapter must
196    /// acquire the exact row and repeat its query before durable publication.
197    pub fn verify_finalized_raw_input_bytes(
198        &self,
199        measured: &str,
200        started_at: &str,
201        inputs: &BTreeMap<String, Vec<u8>>,
202        artifact_bytes: Option<(&[u8], &[u8], &[u8])>,
203    ) -> Result<bool, String> {
204        self.run_id
205            .as_deref()
206            .filter(|value| !value.trim().is_empty())
207            .ok_or("finalized row omitted its run identity")?;
208        if self.commit.as_deref() != Some(measured)
209            || self.tree_dirty != Some(false)
210            || self.commit_anchored != Some(true)
211            || self.started_at.as_deref() != Some(started_at)
212            || self
213                .finished_at
214                .as_deref()
215                .is_none_or(|value| value.trim().is_empty())
216            || !matches!(self.result.as_deref(), Some("pass" | "fail" | "no_result"))
217        {
218            return Err(
219                "finalized row does not bind a terminal clean measured run and its stamp".into(),
220            );
221        }
222        let census = self.raw_result_input_census_v1()?.ok_or_else(|| {
223            format!(
224                "finalized run has no producer-bound raw input census: {}",
225                self.extra
226                    .get("raw_result_input_census_error")
227                    .and_then(Value::as_str)
228                    .unwrap_or("historical absence cannot authorize current writeback")
229            )
230        })?;
231        census.verify_inputs(self, inputs)?;
232        let zero_cells = census.files.iter().all(|file| file.rows.is_empty());
233        if self.schema_version == Some(10) || zero_cells {
234            self.constructed_plan_artifact()?
235                .ok_or("zero-current proof requires schema 10")?;
236            let (plan, cells, tests) = artifact_bytes
237                .ok_or("finalized proof omitted required plan/cell/test artifact bytes")?;
238            let verified = self
239                .verify_schema10_artifact_bytes(plan, cells, tests)?
240                .ok_or("zero-current proof did not establish schema 10 evidence")?;
241            let cells = &verified.cell_results;
242            let recorded = cells
243                .cells
244                .iter()
245                .map(|cell| cell.identity())
246                .collect::<BTreeSet<_>>();
247            let actual = census
248                .files
249                .iter()
250                .flat_map(|file| &file.rows)
251                .map(|row| row.cell.clone())
252                .collect::<BTreeSet<_>>();
253            if recorded != actual {
254                return Err(
255                    "current raw cells differ from the verified recorded artifact population"
256                        .into(),
257                );
258            }
259            for cell in &cells.cells {
260                if let Some(selected_attempt) = cell.selected_attempt {
261                    if !census
262                        .files
263                        .iter()
264                        .flat_map(|file| &file.rows)
265                        .any(|input| {
266                            input.cell == cell.identity() && input.attempt == selected_attempt
267                        })
268                    {
269                        return Err(
270                            "verified selected attempt is absent from the raw input census".into(),
271                        );
272                    }
273                }
274            }
275            // Nonempty partial failures retain their missing planned work.
276            // Empty raw input must satisfy every existing selected-zero check.
277            if zero_cells
278                && (cells.selected_count != 0
279                    || cells.recorded_count != 0
280                    || cells.artifact.row_count != 0
281                    || !cells.selected.is_empty()
282                    || !cells.selected_backend_parity.is_empty()
283                    || !cells.cells.is_empty()
284                    || !verified.missing_cells.is_empty()
285                    || !verified.missing_backend_parity.is_empty()
286                    || !verified.missing_test_producers.is_empty()
287                    || !verified.full_test_results)
288            {
289                return Err(
290                    "empty current results contradict the finalized selected population".into(),
291                );
292            }
293        }
294        Ok(zero_cells)
295    }

297    /// Preserve absence and the outer extension map's historical serialization.
298    /// A present unknown/malformed version refuses, rather than becoming absent.
299    pub fn raw_result_input_census_v1(&self) -> Result<Option<RawResultInputCensusV1>, String> {
300        let Some(value) = self.extra.get("raw_result_input_census_v1") else {
301            return Ok(None);
302        };
303        if self.extra.get("raw_result_input_census_error").is_some() {
304            return Err("raw result census cannot carry both proof and publication error".into());
305        }
306        let census: RawResultInputCensusV1 = serde_json::from_value(value.clone())
307            .map_err(|error| format!("invalid raw result census: {error}"))?;
308        census.validate_for_row(self)?;
309        Ok(Some(census))
310    }

348/// Read the original harness row before any map conversion can erase duplicate
349/// fields. Non-null durations must fit an exact unsigned 64-bit integer before
350/// buffering; other numbers retain serde_json's existing representation, which
351/// includes legitimate floating-point diversity measurements in ordinary rows.
352/// Historical runner and ledger decoders retain their existing behavior.
353pub fn read_schema10_source_result(bytes: &[u8]) -> Result<Value, String> {
354    struct ExactValue(Value);
355    impl<'de> Deserialize<'de> for ExactValue {
356        fn deserialize<D: Deserializer<'de>>(deserializer: D) -> Result<Self, D::Error> {
357            struct Visitor;
358            impl<'de> serde::de::Visitor<'de> for Visitor {
359                type Value = ExactValue;
360
361                fn expecting(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
362                    f.write_str("JSON with unique object fields and exact integer durations")
363                }
364
365                fn visit_unit<E: serde::de::Error>(self) -> Result<Self::Value, E> {
366                    Ok(ExactValue(Value::Null))
367                }
368
369                fn visit_bool<E: serde::de::Error>(self, value: bool) -> Result<Self::Value, E> {
370                    Ok(ExactValue(Value::Bool(value)))
371                }
372
373                fn visit_i64<E: serde::de::Error>(self, value: i64) -> Result<Self::Value, E> {
374                    Ok(ExactValue(Value::from(value)))
375                }
376
377                fn visit_u64<E: serde::de::Error>(self, value: u64) -> Result<Self::Value, E> {
378                    Ok(ExactValue(Value::from(value)))
379                }
380
381                fn visit_f64<E: serde::de::Error>(self, value: f64) -> Result<Self::Value, E> {
382                    serde_json::Number::from_f64(value)
383                        .map(|number| ExactValue(Value::Number(number)))
384                        .ok_or_else(|| E::custom("non-finite JSON number"))
385                }
386
387                fn visit_str<E: serde::de::Error>(self, value: &str) -> Result<Self::Value, E> {
388                    Ok(ExactValue(Value::String(value.into())))
389                }
390
391                fn visit_seq<A: serde::de::SeqAccess<'de>>(
392                    self,
393                    mut seq: A,
394                ) -> Result<Self::Value, A::Error> {
395                    let mut values = Vec::new();
396                    while let Some(value) = seq.next_element::<ExactValue>()? {
397                        values.push(value.0);
398                    }
399                    Ok(ExactValue(Value::Array(values)))
400                }
401
402                fn visit_map<A: serde::de::MapAccess<'de>>(
403                    self,
404                    mut map: A,
405                ) -> Result<Self::Value, A::Error> {
406                    let mut values = serde_json::Map::new();
407                    while let Some(key) = map.next_key::<String>()? {
408                        if values.contains_key(&key) {
409                            return Err(serde::de::Error::custom(format!(
410                                "duplicate field `{key}` in schema 10 source result"
411                            )));
412                        }
413                        let value = map.next_value::<ExactValue>()?.0;
414                        values.insert(key, value);
415                    }
416                    Ok(ExactValue(Value::Object(values)))
417                }
418            }
419            deserializer.deserialize_any(Visitor)
420        }
421    }
422    let value = serde_json::from_slice::<ExactValue>(bytes)
423        .map(|value| value.0)
424        .map_err(|error| format!("malformed schema 10 source result: {error}"))?;
425    let exact_duration = |object: &Value| {
426        if object
427            .get("duration_ms")
428            .is_some_and(|value| !value.is_null() && value.as_u64().is_none())
429        {
430            Err("schema 10 duration_ms must be an exact unsigned 64-bit integer".to_string())
431        } else {
432            Ok(())
433        }
434    };
435    exact_duration(&value)?;
436    if let Some(attempts) = value.get("attempts").and_then(Value::as_array) {
437        for attempt in attempts {
438            exact_duration(attempt)?;
439        }
440    }
441    crate::cpu_evidence::validate_cpu_observations_in_source_row(&value)?;
442    Ok(value)
443}

553impl ConstructedValidationPlanV10 {
554    pub fn constructed_dag(&self) -> Result<DagConfig, String> {
555        if self.schema != 1
556            || !nonblank_component(&self.run_id)
557            || !is_lower_hex(&self.hermit_sha, 40)
558        {
559            return Err("schema 10 constructed plan identity is malformed".into());
560        }
561        let cfg = dag_from_json(&self.dag_json)
562            .map_err(|error| format!("invalid schema 10 constructed DAG: {error}"))?;
563        if dag_to_json(&cfg) != self.dag_json {
564            return Err(
565                "schema 10 constructed plan is not the exact canonical selected DAG".into(),
566            );
567        }
568        Ok(cfg)
569    }
570
571    fn populations(&self) -> Result<(Vec<CellIdentity>, Vec<BackendParityRelation>), String> {
572        let cfg = self.constructed_dag()?;
573        let expected =
574            crate::validation_dag::expected_cells_from_json(&self.expected_e2e_plan_json)?
575                .iter()
576                .map(exact_identity)
577                .collect::<Result<BTreeSet<_>, _>>()?;
578        let mut selected = BTreeSet::new();
579        let mut relations = BTreeSet::new();
580        let mut tags = BTreeSet::new();
581        for step in &cfg.steps {
582            if !tags.insert(step.tag()) {
583                return Err("constructed plan repeats a step identity".into());
584            }
585            let parity = crate::backend_parity_policy::selects_ptrace_parity(step)?;
586            let mut owned = BTreeSet::new();
587            for manifest in step.effective_result_manifests().iter() {
588                let identity = exact_identity(manifest)?;
589                if !expected.contains(&identity) || !owned.insert(identity.clone()) {
590                    return Err(format!(
591                        "{} owns an unknown or repeated expected cell",
592                        step.tag()
593                    ));
594                }
595                selected.insert(identity);
596            }
597            if let Some(selector) = &step.manifest {
598                let required = expected
599                    .iter()
600                    .filter(|cell| selector_matches(selector, cell))
601                    .cloned()
602                    .collect::<BTreeSet<_>>();
603                if owned != required {
604                    return Err(format!(
605                        "{} result ownership differs from its expected manifest selection",
606                        step.tag()
607                    ));
608                }
609            }
610            if parity {
611                for cell in owned
612                    .into_iter()
613                    .filter(|cell| cell.mode == "verify" && cell.backend != "ptrace")
614                {
615                    if !relations.insert(BackendParityRelation::ptrace(cell)) {
616                        return Err(
617                            "constructed plan selects one backend parity relation more than once"
618                                .into(),
619                        );
620                    }
621                }
622            }
623        }
624        Ok((
625            selected.into_iter().collect(),
626            relations.into_iter().collect(),
627        ))
628    }

630    pub fn planned_cells(&self) -> Result<Vec<CellIdentity>, String> {
631        self.populations().map(|(cells, _)| cells)
632    }
633
634    pub fn planned_backend_parity_relations(&self) -> Result<Vec<BackendParityRelation>, String> {
635        self.populations().map(|(_, relations)| relations)
636    }

979impl CellArtifactResultV10 {
980    /// Compact this artifact row into the terminal ledger record, binding a
981    /// COMPARED verdict to the exact attempt it was computed from.
982    ///
983    /// The coordinates are parameters rather than fields so that no caller can
984    /// produce a terminal record without stating which attempt it read. That
985    /// is the whole point: an unbound compared verdict is exactly the row this
986    /// work exists to stop being written.
987    pub fn summary(&self, run_id: &str, hermit_sha: &str) -> Result<CellResultV10, String> {
988        self.summary_for_contract(CellBindingContract::SelectedAttemptV1, run_id, hermit_sha)
989    }
990
991    /// A legacy summary retains the original observations, without inventing
992    /// the attempt metadata that its producer did not record.
993    pub fn summary_for_contract(
994        &self,
995        contract: CellBindingContract,
996        run_id: &str,
997        hermit_sha: &str,
998    ) -> Result<CellResultV10, String> {
999        match (contract, self.selected_attempt) {
1000            (CellBindingContract::LegacyUnbound, None)
1001            | (CellBindingContract::SelectedAttemptV1, Some(1..)) => {}
1002            _ => {
1003                return Err("schema 10 artifact attempt differs from its binding contract".into());
1004            }
1005        }
1006        let backend_parity = match &self.backend_parity {
1007            RequiredNullable::Null => RequiredNullable::Null,
1008            RequiredNullable::Value(parity) => {
1009                if parity.candidate_verdict(&self.identity())? != self.cell_verdict {
1010                    return Err("schema 10 cell ordinary verdict differs from the selected candidate attempt".into());
1011                }
1012                // A parity row carries its whole attempt history, so the
1013                // recorded ordinal is checkable rather than merely asserted.
1014                if let Some(attempt) = self.selected_attempt {
1015                    if parity.candidate_attempt_number(&self.identity())? != attempt {
1016                        return Err(
1017                            "schema 10 parity cell recorded an attempt its own history did not select"
1018                                .into(),
1019                        );
1020                    }
1021                }
1022                RequiredNullable::Value(parity.summary(&self.identity())?)
1023            }
1024        };
1025        let cell_verdict = compact_cell_verdict(&self.cell_verdict);
1026        // Only a compared verdict read an attempt. Binding a by-design or
1027        // unavailable verdict would name an event that was never published.
1028        let evidence_binding = self.selected_attempt.and_then(|attempt| {
1029            matches!(
1030                cell_verdict,
1031                CellVerdict::ComparedAndMatched { .. } | CellVerdict::ComparedAndDiverged { .. }
1032            )
1033            .then(|| {
1034                CellEvidenceBinding::for_validate_compared(
1035                    run_id,
1036                    &self.identity(),
1037                    hermit_sha,
1038                    attempt,
1039                )
1040            })
1041        });
1042        let recorded_attempt = evidence_binding
1043            .as_ref()
1044            .map(|binding| binding.selected_attempt);
1045        Ok(CellResultV10 {
1046            lane: self.lane.clone(),
1047            category: self.category.clone(),
1048            test: self.test.clone(),
1049            mode: self.mode.clone(),
1050            backend: self.backend.clone(),
1051            cell_verdict,
1052            backend_parity,
1053            selected_attempt: recorded_attempt,
1054            evidence_binding,
1055        })
1056    }

1213    /// Decode the current attempt shape without changing historical runner
1214    /// deserialization. In particular, every nullable field must be present.
1215    pub fn from_value(value: Value) -> Result<Self, String> {

1218    }

1220    fn bind_role(&self, backend: &str, index: &str) -> Result<(), String> {
1221        let attempt = &self.0;
1222        if attempt.index != index || attempt.cwd.is_empty() || attempt.shell_command.is_empty() {
1223            return Err("schema 10 parity attempt has incomplete invocation identity".into());
1224        }
1225        let separator = attempt
1226            .argv
1227            .iter()
1228            .position(|arg| arg == "--")
1229            .ok_or("schema 10 parity invocation omitted its guest separator")?;
1230        let prefix = &attempt.argv[..separator];
1231        let mut backends = Vec::new();
1232        for (position, argument) in prefix.iter().enumerate() {
1233            if argument == "--backend" {
1234                backends.push(
1235                    prefix
1236                        .get(position + 1)
1237                        .map(String::as_str)
1238                        .ok_or("schema 10 parity invocation has an incomplete backend option")?,
1239                );
1240            } else if let Some(value) = argument.strip_prefix("--backend=") {
1241                backends.push(value);
1242            }
1243        }
1244        if backends != [backend]
1245            || attempt.argv[separator + 1..] != attempt.guest_argv
1246            || attempt.guest_argv.is_empty()
1247            || !prefix.iter().any(|arg| arg == "--verify")
1248            || !prefix.iter().any(|arg| arg == "--verify-strict")
1249            || !prefix.iter().any(|arg| arg == "--strict")
1250            // The canonical producer emits no negated policy options. Refuse
1251            // every such option independently, including future opt-outs that
1252            // are not represented in the comparison report's policy fields.
1253            || prefix.iter().any(|arg| arg.starts_with("--no-"))
1254        {

1341    /// The candidate and reference keep their actual ordinary comparison
1342    /// verdicts. A path refusal is considered before interpreting a divergence.
1343    pub fn ordinary_verdict(&self, backend: &str, index: &str) -> Result<CellVerdict, String> {
1344        self.bind_role(backend, index)?;
1345        let attempt = &self.0;
1346        let report = self.report()?;
1347        if !self.path_eligible(backend)? {
1348            return Ok(unavailable(
1349                "Backend execution path is ineligible; exact evidence is retained",
1350            ));
1351        }
1352        if attempt.status.is_some() && attempt.signal.is_some() {
1353            return Err("schema 10 attempt records both exit status and signal".into());
1354        }
1355        let reason = || {
1356            attempt
1357                .reason
1358                .clone()
1359                .unwrap_or_else(|| "Attempt produced no completed canonical comparison".into())
1360        };
1361        let Some(report) = report else {
1362            let stopped = attempt.status.is_some_and(|status| status != 0)
1363                || attempt.signal.is_some_and(|signal| signal > 0)
1364                || (attempt.timed_out && attempt.status.is_none() && attempt.signal.is_none());
1365            if attempt.outcome != "ERROR"
1366                || !stopped
1367                || attempt.error_kind.as_deref().is_none_or(str::is_empty)
1368            {
1369                return Err("schema 10 missing report lacks an explicit failed or stopped process disposition".into());
1370            }
1371            return Ok(unavailable(reason()));
1372        };
1373        if !matches!(report.verdict, Verdict::Matched | Verdict::Diverged) {
1374            if attempt.outcome == "PASS" {
1375                return Err("schema 10 PASS has no completed ordinary comparison".into());
1376            }
1377            return Ok(unavailable(reason()));
1378        }
1379        if attempt.timed_out {
1380            if attempt.outcome == "PASS" {
1381                return Err("schema 10 timed-out attempt claims PASS".into());
1382            }
1383            return Ok(unavailable(reason()));
1384        }
1385        report.require_canonical_comparison()?;
1386        let raw: Value =
1387            serde_json::from_str(attempt.verification_report.as_ref().expect("report exists"))
1388                .map_err(|error| error.to_string())?;
1389        let comparison: ComparisonSpec = serde_json::from_value(raw["comparison"].clone())
1390            .map_err(|error| format!("schema 10 ordinary comparison is malformed: {error}"))?;
1391        let counts: RequiredNullable<ComparedLogCounts> =
1392            serde_json::from_value(raw["compared_log_messages"].clone()).map_err(|error| {
1393                format!("schema 10 ordinary comparison counts are malformed: {error}")
1394            })?;
1395        if !comparison.is_canonical_bitwise_info_v1(&counts) {
1396            return Ok(unavailable(
1397                "Ordinary comparison does not satisfy BitwiseInfoV1/all_records_v1",
1398            ));
1399        }
1400        if report.verdict == Verdict::Matched {
1401            report.require_canonical_match()?;
1402            report.require_exact_output_match()?;
1403            if attempt.outcome != "PASS"
1404                || attempt.status != Some(0)
1405                || attempt.signal.is_some()
1406                || attempt.error_kind.is_some()
1407            {
1408                return Err("schema 10 ordinary match contradicts its process result".into());
1409            }
1410            Ok(CellVerdict::ComparedAndMatched {
1411                comparison_tier: ComparisonTier::CanonicalBitwise,
1412                comparison,
1413                bitwise_parity: true,
1414                compared_log_messages: counts,
1415            })
1416        } else {
1417            if attempt.outcome != "FAIL"
1418                || report.verified
1419                || report.bitwise_parity
1420                || !attempt.status.is_some_and(|status| status != 0)
1421                || attempt.signal.is_some()
1422            {
1423                return Err("schema 10 ordinary divergence contradicts its process result".into());
1424            }
1425            Ok(CellVerdict::ComparedAndDiverged {
1426                comparison_tier: ComparisonTier::CanonicalBitwise,
1427                comparison,
1428                bitwise_parity: false,
1429                compared_log_messages: counts,
1430            })
1431        }
1432    }

1829/// Validate the existing canonical ordinary-comparison contract independently
1830/// of its carrier. Public derived operands use the same mode and INFO policy.
1831pub fn validate_ordinary_verdict(
1832    identity: &CellIdentity,
1833    verdict: &CellVerdict,
1834) -> Result<(), String> {
1835    match verdict {
1836        CellVerdict::ComparedAndMatched {
1837            comparison_tier,
1838            comparison,
1839            bitwise_parity,
1840            compared_log_messages,
1841        }
1842        | CellVerdict::ComparedAndDiverged {
1843            comparison_tier,
1844            comparison,
1845            bitwise_parity,
1846            compared_log_messages,
1847        } => {
1848            let expected_time = match identity.mode.as_str() {
1849                "verify" | "chaos" => true,
1850                "replay" => false,
1851                _ => return Err("schema 10 non-comparison mode carries a compared verdict".into()),
1852            };
1853            if *comparison_tier != ComparisonTier::CanonicalBitwise
1854                || !comparison.is_canonical_bitwise_info_v1_for_time_policy(
1855                    expected_time,
1856                    compared_log_messages,
1857                )
1858                || *bitwise_parity != matches!(verdict, CellVerdict::ComparedAndMatched { .. })
1859            {
1860                return Err(
1861                    "schema 10 ordinary verdict contradicts its canonical comparison".into(),
1862                );
1863            }
1864        }
1865        CellVerdict::PerformsNoComparisonByDesign {
1866            comparison_tier,
1867            reason,
1868        } => {
1869            if !matches!(identity.mode.as_str(), "naked" | "custom")
1870                || *comparison_tier != ComparisonTier::DeclaredButUnverifiable
1871                || reason.trim().is_empty()
1872            {
1873                return Err("schema 10 no-comparison verdict contradicts its mode".into());
1874            }
1875        }
1876        CellVerdict::UnavailableWithReason {
1877            comparison_tier,

1893    /// Refuse cell evidence whose COMPARED verdicts are not each bound to a
1894    /// source attempt consistent with the row that carries them.
1895    ///
1896    /// WHAT THIS ESTABLISHES, stated exactly, because the previous version
1897    /// claimed an axis it did not check:
1898    ///
1899    /// * presence -- a compared verdict carries a binding, and a non-comparing
1900    ///   verdict does not;
1901    /// * run, tree and cell -- the binding belongs to this evidence;
1902    /// * attempt CONSISTENCY -- the binding's ordinal equals the ledger row's
1903    ///   own `selected_attempt`, which is a real second operand rather than
1904    ///   the same expression twice;
1905    /// * uniqueness -- two compared cells cannot bind the same attempt of the
1906    ///   same cell.
1907    ///
1908    /// WHAT IT DOES NOT ESTABLISH, and nothing in the ledger can: that the
1909    /// recorded ordinal is the attempt the verdict was actually computed from.
1910    /// `verify_cell_artifact_bytes` re-derives it from the digest-bound
1911    /// artifact, which protects it from later tampering but not from being
1912    /// wrong when written; only the parity path re-derives it from an attempt
1913    /// history. For an ordinary cell it remains a producer assertion, and the
1914    /// end-to-end resolution against published rows is what would refute it.
1915    pub fn require_bound_compared_cells(&self) -> Result<(), String> {
1916        if self.binding_contract != CellBindingContract::SelectedAttemptV1 {
1917            return Err("schema 10 comparison evidence is legacy-unbound".into());
1918        }
1919        let mut seen = BTreeSet::new();
1920        for cell in &self.cells {
1921            let compared = matches!(
1922                cell.cell_verdict,
1923                CellVerdict::ComparedAndMatched { .. } | CellVerdict::ComparedAndDiverged { .. }
1924            );
1925            let identity = cell.identity();
1926            let named = CellEvidenceBinding::series_cell_key(&identity);
1927            if !compared {
1928                if cell.evidence_binding.is_some() || cell.selected_attempt.is_some() {
1929                    return Err(format!(
1930                        "cell {named} states no comparison yet carries an evidence binding"
1931                    ));
1932                }
1933                continue;
1934            }
1935            let binding = cell
1936                .evidence_binding
1937                .as_ref()
1938                .ok_or_else(|| format!("compared cell {named} carries no evidence binding"))?;
1939            binding.verify_against(&self.run_id, &self.hermit_sha, &identity)?;
1940            let recorded = cell.selected_attempt.ok_or_else(|| {
1941                format!("compared cell {named} carries a binding but no selected_attempt")
1942            })?;
1943            if binding.selected_attempt != recorded {
1944                return Err(format!(
1945                    "compared cell {named} binds attempt {} while the row records {recorded}",
1946                    binding.selected_attempt
1947                ));
1948            }
1949            if !seen.insert((named.clone(), binding.selected_attempt)) {
1950                return Err(format!(
1951                    "attempt {} of cell {named} is bound by more than one compared cell",
1952                    binding.selected_attempt
1953                ));
1954            }
1955        }
1956        Ok(())
1957    }

1960    /// only from the binding.
1961    ///
1962    /// Deliberately NOT a list of published `event_id`s. Predicting one is
