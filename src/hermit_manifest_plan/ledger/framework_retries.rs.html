1//! Every cell a validation's harness ran more than once, carried into the run
2//! record (<https://github.com/rrnewton/hermit/issues/1845>).
3//!
4//! A retry that passes makes its cell pass, so without this a divergence the
5//! retry hid leaves no trace in the run record. It is an independent versioned
6//! extension computed from the same raw `results.jsonl` bytes the raw input
7//! census binds, and it revises no verdict.
8
9use super::*;
10use crate::runner::outcome_after_retries;
11
12/// The cells of one run that have more than one attempt row, whatever caused
13/// the retry.
14#[derive(Clone, Debug, PartialEq, Eq, Deserialize, Serialize)]
15#[serde(deny_unknown_fields)]
16pub struct FrameworkRetriesV1 {
17    pub schema: u32,
18    /// Cells with more than one attempt.
19    pub retried_cells: u64,
20    /// Of those, the cells whose history selected PASS: a failure the retry
21    /// turned into a pass.
22    pub retried_cells_final_pass: u64,
23    pub cells: Vec<RetriedCellV1>,
24}
25
26/// One retried cell: what its first attempt observed, where it first
27/// diverged, and the outcome its whole history selected.
28#[derive(Clone, Debug, PartialEq, Eq, Deserialize, Serialize)]
29#[serde(deny_unknown_fields)]
30pub struct RetriedCellV1 {
31    /// The `results.jsonl` path, relative to the result root, as in the census.
32    pub path: String,
33    pub cell: CellIdentity,
34    pub attempts: u64,
35    pub first_outcome: String,
36    pub first_failure_class: Option<String>,
37    pub first_error_kind: Option<String>,
38    pub first_result: Option<String>,
39    pub first_divergent_scheduler_turn: Option<u64>,
40    pub first_divergent_record: Option<u64>,
41    pub first_divergent_syscall: Option<u64>,
42    pub final_outcome: String,
43}
44
45impl FrameworkRetriesV1 {
46    /// Group each file's rows by cell, in attempt order, and keep every cell
47    /// with more than one attempt. A row without an identity, a positive
48    /// attempt or an outcome, or a history [`outcome_after_retries`] refuses,
49    /// refuses the whole extension rather than undercounting.
50    pub fn from_inputs(inputs: &BTreeMap<String, Vec<u8>>) -> Result<Self, String> {
51        let mut cells = Vec::new();
52        for (path, bytes) in inputs {
53            let text = std::str::from_utf8(bytes)
54                .map_err(|error| format!("{path}: results are not UTF-8: {error}"))?;
55            let mut histories: BTreeMap<CellIdentity, Vec<(u64, Value)>> = BTreeMap::new();
56            for line in text.lines().filter(|line| !line.trim().is_empty()) {
57                let row: Value = serde_json::from_str(line)
58                    .map_err(|error| format!("{path}: malformed result row: {error}"))?;
59                let field = |name: &str| {
60                    row.get(name)
61                        .and_then(Value::as_str)
62                        .map(str::to_string)
63                        .ok_or_else(|| format!("{path}: result row omitted {name}"))
64                };
65                let cell = CellIdentity {
66                    lane: field("lane")?,
67                    category: field("category")?,
68                    test: field("test")?,
69                    mode: field("mode")?,
70                    backend: field("backend")?,
71                };
72                let attempt = row
73                    .get("attempt")
74                    .and_then(Value::as_u64)
75                    .filter(|attempt| *attempt > 0)
76                    .ok_or_else(|| format!("{path}: result row omitted a positive attempt"))?;
77                field("outcome")?;
78                histories.entry(cell).or_default().push((attempt, row));
79            }
80            for (cell, mut history) in histories {
81                if history.len() < 2 {
82                    continue;
83                }
84                history.sort_by_key(|(attempt, _)| *attempt);
85                let outcome = |row: &Value| row["outcome"].as_str().unwrap_or_default().to_string();
86                let final_outcome = outcome_after_retries(
87                    history
88                        .iter()
89                        .map(|(attempt, row)| (*attempt, row["outcome"].as_str().unwrap_or(""))),
90                )
91                .map_err(|error| format!("{path}: {} ({}): {error}", cell.test, cell.mode))?;
92                let first = &history[0].1;
93                let text = |name: &str| first.get(name).and_then(Value::as_str).map(str::to_string);
94                let number = |name: &str| first.get(name).and_then(Value::as_u64);
95                cells.push(RetriedCellV1 {
96                    path: path.clone(),
97                    cell,
98                    attempts: history.len() as u64,
99                    first_outcome: outcome(first),
100                    first_failure_class: text("failure_class"),
101                    first_error_kind: text("error_kind"),
102                    first_result: text("result"),
103                    first_divergent_scheduler_turn: number("first_divergent_scheduler_turn"),
104                    first_divergent_record: number("first_divergent_record"),
105                    first_divergent_syscall: number("first_divergent_syscall"),
106                    final_outcome: final_outcome.to_string(),
107                });
108            }
109        }
110        Ok(Self {
111            schema: 1,
112            retried_cells: cells.len() as u64,
113            retried_cells_final_pass: cells
114                .iter()
115                .filter(|cell| cell.final_outcome == "PASS")
116                .count() as u64,
117            cells,
118        })
119    }
120}

122#[cfg(test)]
123mod tests {
124    use super::*;
125
126    fn row(test: &str, attempt: u64, outcome: &str, diverged: bool) -> String {
127        let mut row = serde_json::json!({
128            "lane": "portable", "category": "fixture", "test": test, "mode": "verify",
129            "backend": "ptrace", "attempt": attempt, "outcome": outcome,
130        });
131        if outcome == "FAIL" {
132            row["failure_class"] = "product_failure".into();
133            row["result"] = "determinism-failure".into();
134        }
135        if diverged {
136            row["first_divergent_scheduler_turn"] = 4.into();
137            row["first_divergent_record"] = 9.into();
138            row["first_divergent_syscall"] = 2.into();
139        }
140        format!("{row}\n")
141    }
142
143    #[test]
144    fn every_retried_cell_is_counted_with_its_first_attempt_and_selected_outcome() {
145        let one = [
146            row("fixture/once", 1, "PASS", false),
147            // Rows of one cell need not be adjacent, nor in attempt order.
148            row("fixture/rescued", 2, "PASS", false),
149            row("fixture/twice", 1, "FAIL", true),
150            row("fixture/rescued", 1, "FAIL", true),
151            row("fixture/twice", 2, "FAIL", true),
152        ]
153        .concat();
154        let inputs = BTreeMap::from([
155            ("one/results.jsonl".to_string(), one.into_bytes()),
156            ("empty/results.jsonl".to_string(), Vec::new()),
157        ]);
158        let retries = FrameworkRetriesV1::from_inputs(&inputs).unwrap();
159        assert_eq!(
160            (retries.retried_cells, retries.retried_cells_final_pass),
161            (2, 1)
162        );
163        let rescued = &retries.cells[0];
164        assert_eq!(rescued.cell.test, "fixture/rescued");
165        assert_eq!(
166            (
167                rescued.path.as_str(),
168                rescued.attempts,
169                rescued.first_outcome.as_str(),
170                rescued.first_failure_class.as_deref(),
171                rescued.first_error_kind.as_deref(),
172                rescued.first_result.as_deref(),
173                rescued.first_divergent_scheduler_turn,
174                rescued.first_divergent_record,
175                rescued.first_divergent_syscall,
176                rescued.final_outcome.as_str(),
177            ),
178            (
179                "one/results.jsonl",
180                2,
181                "FAIL",
182                Some("product_failure"),
183                None,
184                Some("determinism-failure"),
185                Some(4),
186                Some(9),
187                Some(2),
188                "PASS"
189            )
190        );
191        assert_eq!(retries.cells[1].cell.test, "fixture/twice");
192        assert_eq!(retries.cells[1].final_outcome, "FAIL");
193        // The record round-trips through its strict reader.
194        let value = serde_json::to_value(&retries).unwrap();
195        assert_eq!(
196            serde_json::from_value::<FrameworkRetriesV1>(value).unwrap(),
197            retries
198        );
199    }

201    #[test]
202    fn an_unreadable_history_refuses_rather_than_undercounting() {
203        for (case, rows) in [
204            ("no attempt", "{\"lane\":\"portable\",\"category\":\"fixture\",\"test\":\"fixture/x\",\"mode\":\"verify\",\"backend\":\"ptrace\",\"outcome\":\"FAIL\"}\n".to_string()),
205            ("attempt after a pass", [row("fixture/x", 1, "PASS", false), row("fixture/x", 2, "FAIL", false)].concat()),
206            ("a gap", [row("fixture/x", 1, "FAIL", false), row("fixture/x", 3, "PASS", false)].concat()),
207            ("malformed", "{".to_string()),
208        ] {
209            let inputs = BTreeMap::from([("one/results.jsonl".to_string(), rows.into_bytes())]);
210            assert!(FrameworkRetriesV1::from_inputs(&inputs).is_err(), "{case}");
211        }
212    }
