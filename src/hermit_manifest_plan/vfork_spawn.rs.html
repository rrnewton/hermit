9//! Starting a program inside an invocation cgroup without copying this
10//! process (dh-7n47nt).
11//!
12//! `std::process::Command` with a `pre_exec` closure, which the cgroup join
13//! needed, starts every child with fork(). fork() copies this process's
14//! address space while it holds the memory-map lock, and every other thread
15//! that page-faults meanwhile waits. In the library's test process (about 330
16//! MB, 80-140 threads) each fork took a median 114-135 ms and up to 3 s, and
17//! CPU readers and census walks in the same process stalled behind them.
18//!
19//! Here the child is created with clone3(CLONE_VM | CLONE_VFORK |
20//! CLONE_INTO_CGROUP), as glibc's posix_spawn creates its children
21//! (`CLONE_VM | CLONE_VFORK` on a stack of their own), with the kernel placing
22//! it in the invocation cgroup as it is created. The child shares this
23//! process's memory until it execs, and this thread waits until then. So the
24//! child runs only [`vfork_child`]: raw system calls on storage this thread
25//! prepared beforehand, no allocation, no locks and no libc call, on a stack
26//! of its own, never on this thread's.
27//!
28//! clone3 needs Linux 5.3 and CLONE_INTO_CGROUP 5.7. Where clone3 is refused
29//! as a call (ENOSYS, E2BIG or EINVAL, as on older kernels or under a seccomp
30//! filter, or EPERM from a filter, which a probe tells apart from a cgroup
31//! that refused the child), [`Started::Unusable`] tells the caller to start
32//! the program the old way, and the refusal is remembered for the rest of the
33//! process, with one line on stderr. A refusal that is about the cgroup itself
34//! is a failed join.

64/// How [`start`] ended.

66pub(crate) enum Started {
67    /// The program runs as process `pid`, in the cgroup.
68    Running { pid: u32 },
69    /// clone3 cannot start programs in a cgroup here, for `reason`; start the

72    /// The kernel would not create the child in the cgroup.

79/// The probe child's owned lifecycle state. Held under
80/// [`Clone3Support`]'s lock for every signal, reap marking
81/// and fd close.
82/// An owned file descriptor: closes when dropped, and is
83/// neither Copy nor Clone, so it has exactly one owner.
84struct OwnedFd(RawFd);

86impl OwnedFd {
87    fn raw(&self) -> RawFd {
88        self.0
89    }
90}

92impl Drop for OwnedFd {
93    fn drop(&mut self) {
94        unsafe {
95            libc::close(self.0);
96        }
97    }

100/// The probe child's owned wait handle: its pidfd when the
101/// clone returned one, which path is in use, and its pid.
102/// Moved as a whole when the background reaper takes over.
103struct OwnedProbeChild {
104    /// The pidfd, owned; closed when this handle drops,
105    /// after its reap.
106    pidfd: Option<OwnedFd>,
107    /// Whether waits and signals use the pidfd (true) or
108    /// the numeric pid (false, only when CLONE_PIDFD was
109    /// refused at clone time).
110    via_pidfd: bool,
111    /// The child's numeric pid (for the fallback path and
112    /// for test inspection).
113    pid: libc::pid_t,
114}

116/// Who holds the probe child's wait handle. Exactly one
117/// owner at a time: the slot holds it while the caller owns
118/// it; taking it out for the background reaper is a move
119/// (the slot then says HandedOff and the caller can neither
120/// wait nor signal); after the reap it says Reaped.
121enum ProbeChildSlot {
122    /// No child has been created.
123    NoChild,
124    /// The caller owns the handle.
125    Owned(OwnedProbeChild),
126    /// The background reaper owns the handle.
127    HandedOff,
128    /// Reaped (or shown to have no waitable child left).
129    Reaped,
130}
131
132/// Locks the probe child's lifecycle handle.
133fn probe_child_slot(support: &Clone3Support) -> std::sync::MutexGuard<'_, ProbeChildSlot> {
134    support
135        .probe_child
136        .lock()
137        .unwrap_or_else(|poisoned| poisoned.into_inner())
138}

551    /// Record that this process still owes a reap for the probe
552    /// child `pid`, keeping ownership of it until a retry reaps
553    /// it. Called when a wait failed or a reaper could not be
554    /// started; never called when a wait showed no waitable
555    /// child remains (ECHILD), which completes the obligation.
556    fn keep_unreaped_probe_child(&self, pid: libc::pid_t) {
557        let mut record = self
558            .unreaped_probe_child
559            .lock()
560            .unwrap_or_else(|poisoned| poisoned.into_inner());
561        if *record != Some(pid) {
562            eprintln!(
563                "manifest-plan: probe child {pid} is still unreaped; \
564                 this process keeps owing its reap and retries it at the next start"
565            );
566        }
567        *record = Some(pid);
568    }

570    /// Retry the reap owed for a probe child, without blocking:
571    /// one WNOHANG wait. Returns true when no waitable child
572    /// remains (it was reaped, or there is none), clearing the
573    /// record; false while the child is still there or its wait
574    /// still fails, keeping the record for the next retry.
575    pub(crate) fn reap_unreaped_probe_child(&self) -> bool {
576        // The whole snapshot-wait-clear transaction holds the
577        // record lock: two concurrent starters cannot snapshot
578        // the same pid, have one reap it, and then wait on a
579        // pid that may since have been reused.
580        let mut record = self
581            .unreaped_probe_child
582            .lock()
583            .unwrap_or_else(|poisoned| poisoned.into_inner());
584        let Some(pid) = *record else {
585            return true;
586        };
587        if matches!(*probe_child_slot(self), ProbeChildSlot::HandedOff) {
588            // The reaper owns the wait right now; a retry by
589            // this caller must neither clear the debt nor mark
590            // anything reaped.
591            return false;
592        }
593        match wait_probe_child(self, pid, false, false) {
594            ChildWait::Reaped(_) | ChildWait::NoWaitable => {
595                *record = None;
596                drop(record);
597                mark_probe_child_reaped(self);
598                true
599            }
600            ChildWait::Failed(errno) => {
601                if errno != Some(libc::EINTR) {
602                    eprintln!(
603                        "manifest-plan: retrying the reap of probe child {pid} failed \
604                         (errno {errno:?}); keeping the obligation"
605                    );
606                }
607                false
608            }
609            ChildWait::StillRunning => false,
610        }
611    }

614    /// this probe child's own pidfd or its duplicate, by the
615    /// inode numbers recorded when they were created. Other
616    /// tests' pidfds have other inodes and never count, and
617    /// one leaked past its reap keeps its inode and still
618    /// counts.
619    #[cfg(test)]
620    pub(crate) fn probe_child_fds_still_open(&self) -> Option<usize> {
621        if self.probe_child_fd_census_shared.load(Ordering::Relaxed) {
622            return None;
623        }
624        use std::os::unix::fs::MetadataExt as _;
625        let inodes = self
626            .probe_child_fd_inodes
627            .lock()
628            .unwrap_or_else(|poisoned| poisoned.into_inner())
629            .clone();
630        if inodes.is_empty() {
631            return Some(0);
632        }
633        let entries = match std::fs::read_dir("/proc/self/fd") {
634            Ok(entries) => entries,
635            Err(error) => {
636                panic!("cannot read /proc/self/fd for the probe child's fd census: {error}")
637            }
638        };
639        let mut count = 0;
640        for entry in entries {
641            let entry = match entry {
642                Ok(entry) => entry,
643                Err(error) => {
644                    panic!("cannot read an entry of /proc/self/fd for the fd census: {error}")
645                }
646            };
647            match std::fs::metadata(entry.path()) {
648                Ok(meta) => {

651                    }
652                }

654                    // An fd of another test that closed between
655                    // the listing and the stat: it was never one
656                    // of this child's held fds, which stay open
657                    // for the whole census.
658                    continue;
659                }
660                Err(error) => {
661                    panic!("cannot stat an fd for the probe child's fd census: {error}")
662                }
663            }
664        }
665        Some(count)

823    /// The startup probe, run once per process: create one
824    /// dedicated probe cgroup (through `create_probe_cgroup`, the
825    /// caller's one chance to make it), start one trivial program
826    /// in it, and watch how it dies. Creation, the probe, the
827    /// unusable decision, cleanup and publication all happen
828    /// inside this one initialization: concurrent callers wait
829    /// for the same completed decision and observe the same
830    /// record. Some kernels kill a CLONE_INTO_CGROUP child with
831    /// SIGKILL before it can run when this process's cgroup has
832    /// been killed before (its kill sequence differs from the
833    /// fresh cgroup's); on such a kernel every program started
834    /// here would die the same way, so clone3 is remembered
835    /// unusable for the whole process and every start takes the
836    /// fork path instead. Any other outcome (the child ran, or
837    /// clone3 itself was refused, which `start` classifies on
838    /// the real call) leaves clone3 in use.
839    ///
840    /// The initialization runs inside this one `get_or_init`,
841    /// on the calling thread, and no caller proceeds past it
842    /// without the published decision. It forks nothing: the
843    /// probe child is a direct clone3, so no fork lands in any
844    /// thread's interval (an invocation's start is measured to
845    /// fork this process zero times).
846    pub(crate) fn probe_cgroup_clone_once(
847        self: &std::sync::Arc<Self>,
848        create_probe_cgroup: impl FnOnce() -> Option<(RawFd, Box<dyn std::any::Any + Send>)>,
849    ) {
850        self.initialize_cgroup_probe(create_probe_cgroup);
851    }
852
853    /// The one initialization [`Clone3Support::probe_cgroup_clone_once`]
854    /// runs, on whichever thread first reaches it.
855    fn initialize_cgroup_probe(
856        self: &std::sync::Arc<Self>,
857        create_probe_cgroup: impl FnOnce() -> Option<(RawFd, Box<dyn std::any::Any + Send>)>,
858    ) {
859        self.cgroup_probe.get_or_init(|| {
860            let Some((cgroup, guard)) = create_probe_cgroup() else {
861                return CgroupProbeRecord {
862                    outcome: "unavailable".to_string(),
863                    kernel: kernel_release(),
864                    reason: None,
865                    child_wait: "none".to_string(),
866                };
867            };
868            let outcome = cgroup_probe_outcome(std::sync::Arc::clone(self), cgroup);
869            drop(guard);
870            let record = CgroupProbeRecord {
871                outcome: outcome.name().to_string(),
872                kernel: kernel_release(),
873                child_wait: if self.probe_child_pid.load(Ordering::Relaxed) <= 0 {
874                    "none".to_string()
875                } else if self.probe_child_waited_via_pidfd.load(Ordering::Relaxed) {
876                    "pidfd".to_string()
877                } else {
878                    "pid".to_string()
879                },
880                reason: match outcome {
881                    CgroupProbeOutcome::ChildKilledBySigkill => Some(format!(
882                        "clone3-into-cgroup refused by kernel: the probe child was killed \
883                         by SIGKILL before it could run (kernel {})",
884                        kernel_release()
885                    )),
886                    CgroupProbeOutcome::TimedOut => Some(format!(
887                        "clone3-into-cgroup probe timed out after {:?} (kernel {}); \
888                         starting with fork() instead",
889                        self.probe_bound,
890                        kernel_release()
891                    )),
892                    _ => None,
893                },
894            };
895            if let Some(reason) = &record.reason {
896                remember_unusable(self, reason);
897            }
898            record
899        });
900    }

926/// How the startup probe's child ended.
927#[derive(Clone, Copy, Debug, PartialEq, Eq)]
928enum CgroupProbeOutcome {
929    /// The child started in the cgroup and ran.
930    ChildRan,
931    /// The kernel killed the child with SIGKILL before it could run.
932    ChildKilledBySigkill,
933    /// No verdict: clone3 was refused, or the child could not start
934    /// for another reason. Not a kernel verdict either way.
935    NoVerdict,
936    /// The probe did not finish inside its bound; its child was
937    /// killed to reap it. Not the kernel bug: the bound, not a
938    /// wait status, decided.
939    TimedOut,
940    /// Waiting for the probe child failed with an error other
941    /// than ECHILD: nothing about the child is proven, and its
942    /// reap stays owed. A failure of the wait itself, given its
943    /// own label rather than folded into a verdict.
944    WaitFailed,
945}

949        match self {

959/// This kernel's release, for the probe's refusal message.
960fn kernel_release() -> String {
961    // SAFETY: `utsname` is plain old data and zero is a valid value.
962    let mut name: libc::utsname = unsafe { std::mem::zeroed() };
963    if unsafe { libc::uname(&mut name) } != 0 {
964        return "unknown".to_string();
965    }
966    let release = name
967        .release
968        .iter()
969        .take_while(|byte| **byte != 0)
970        .map(|byte| *byte as u8)
971        .collect::<Vec<u8>>();
972    String::from_utf8_lossy(&release).into_owned()
973}

1022/// and `NoWaitable` complete the cleanup obligation; the
1023/// pidfd this process holds from the child's clone names
1024/// the exact child, so no wait or signal here can reach a
1025/// different process that later reuses the numeric pid.

1029    Reaped(i32),
1030    /// The child is still running.
1031    StillRunning,
1032    /// No waitable child remains (ECHILD); no status exists.

1038/// The wait status a completed `waitid` reports, synthesized
1039/// in waitpid's encoding, or none when no child changed
1040/// state. Reading the siginfo fields is unsafe: the union
1041/// member is only meaningful for a completed wait.
1042fn reaped_status(info: &libc::siginfo_t) -> Option<i32> {
1043    unsafe {
1044        if info.si_pid() == 0 {
1045            return None;
1046        }
1047        Some(match info.si_code {
1048            libc::CLD_EXITED => info.si_status() << 8,
1049            libc::CLD_DUMPED => info.si_status() | 0x80,
1050            _ => info.si_status(),
1051        })
1052    }
1053}

1055/// Signal the probe child, through its pidfd when this
1056/// process holds one (the usual case), and by its numeric
1057/// pid only on the fallback path where no pidfd exists.
1058fn kill_probe_child(support: &Clone3Support, pid: libc::pid_t) -> i32 {
1059    // The whole check-and-signal holds the slot lock, and
1060    // only the caller's owned handle may be signalled: after
1061    // the move to the reaper (or after the reap) there is
1062    // nothing here to signal.
1063    let slot = probe_child_slot(support);
1064    let ProbeChildSlot::Owned(owned) = &*slot else {
1065        return -libc::ESRCH;
1066    };
1067    if let (true, Some(pidfd)) = (owned.via_pidfd, &owned.pidfd) {
1068        // SAFETY: signalling this exact child by its pidfd,
1069        // held open under the lock.
1070        return unsafe {
1071            syscall6(
1072                libc::SYS_pidfd_send_signal,
1073                [pidfd.raw() as usize, libc::SIGKILL as usize, 0, 0, 0, 0],
1074            )
1075        } as i32;
1076    }
1077    let _ = pid;
1078    // SAFETY: the numeric pid path, used only when the clone
1079    // refused CLONE_PIDFD (no pidfd was ever obtained); the
1080    // child is still owned and unreaped, under the same lock
1081    // as every wait and reap marking.
1082    unsafe { libc::kill(owned.pid, libc::SIGKILL) }
1083}

1148        }
1149    }
1150    let _ = (grace, witnessed_entry);
1151    let _ = pid;
1152    let mut slot = probe_child_slot(support);
1153    let ProbeChildSlot::Owned(owned) = &*slot else {
1154        return ChildWait::NoWaitable;
1155    };
1156    let waited = wait_child_once(owned, false);
1157    if matches!(waited, ChildWait::Reaped(_) | ChildWait::NoWaitable) {
1158        let old = std::mem::replace(&mut *slot, ProbeChildSlot::Reaped);
1159        drop(slot);
1160        drop(old);
1161    }
1162    waited
1163}

1165/// One wait on an owned probe child handle: through its
1166/// pidfd when it has one, and by its numeric pid only on
1167/// the clone path that refused CLONE_PIDFD. The caller (or
1168/// the reaper, after the move) is the handle's only owner.
1169fn wait_child_once(owned: &OwnedProbeChild, blocking: bool) -> ChildWait {
1170    if let (true, Some(pidfd)) = (owned.via_pidfd, &owned.pidfd) {
1171        // SAFETY: `info` is plain old data and zero is valid.
1172        let mut info: libc::siginfo_t = unsafe { std::mem::zeroed() };
1173        let flags = if blocking {
1174            libc::WEXITED
1175        } else {
1176            libc::WEXITED | libc::WNOHANG
1177        };
1178        let waited =
1179            unsafe { libc::waitid(libc::P_PIDFD, pidfd.raw() as libc::id_t, &mut info, flags) };
1180        if waited == 0 {
1181            return match reaped_status(&info) {
1182                Some(status) => ChildWait::Reaped(status),
1183                None => {
1184                    if blocking {
1185                        ChildWait::NoWaitable
1186                    } else {
1187                        ChildWait::StillRunning
1188                    }
1189                }
1190            };
1191        }
1192        return match io::Error::last_os_error().raw_os_error() {
1193            Some(libc::ECHILD) => ChildWait::NoWaitable,
1194            errno => ChildWait::Failed(errno),
1195        };
1196    }
1197    let mut status = 0;
1198    let waited = unsafe {
1199        libc::waitpid(
1200            owned.pid,
1201            &mut status,
1202            if blocking { 0 } else { libc::WNOHANG },
1203        )
1204    };
1205    if waited == owned.pid {
1206        return ChildWait::Reaped(status);
1207    }
1208    if waited == 0 {
1209        return ChildWait::StillRunning;
1210    }
1211    match io::Error::last_os_error().raw_os_error() {
1212        Some(libc::ECHILD) => ChildWait::NoWaitable,
1213        errno => ChildWait::Failed(errno),
1214    }
1215}

1217/// Read the entry byte if it is already in the pipe, without
1218/// blocking. Used after the child is known to have exited:
1219/// the child may have written and died before the parent's
1220/// last poll, and the byte, not the timing, decides.
1221fn drain_entry_pipe(pipe: RawFd, entered: bool) -> bool {
1222    if entered {
1223        return true;
1224    }
1225    let mut poll = libc::pollfd {
1226        fd: pipe,
1227        events: libc::POLLIN,
1228        revents: 0,
1229    };
1230    if unsafe { libc::poll(&mut poll, 1, 0) } > 0 {
1231        let mut byte = [0u8; 1];
1232        if unsafe { libc::read(pipe, byte.as_mut_ptr().cast(), 1) } == 1 {
1233            return true;
1234        }
1235    }
1236    false
1237}

1239fn cgroup_probe_outcome(
1240    support: std::sync::Arc<Clone3Support>,
1241    cgroup: RawFd,
1242) -> CgroupProbeOutcome {
1243    #[cfg(test)]
1244    {
1245        support
1246            .cgroup_probe_attempts
1247            .fetch_add(1, Ordering::Relaxed);
1248        if support.inject_cgroup_probe_kill {
1249            return CgroupProbeOutcome::ChildKilledBySigkill;
1250        }
1251    }
1252    // Everything the child needs is prepared before the clone:
1253    // the entry pipe, and /bin/true's path and argv/envp arrays.
1254    let mut pipe_fds = [0; 2];
1255    if unsafe { libc::pipe2(pipe_fds.as_mut_ptr(), libc::O_CLOEXEC) } != 0 {
1256        return CgroupProbeOutcome::NoVerdict;
1257    }
1258    let path = match std::ffi::CString::new("/bin/true") {
1259        Ok(path) => path,
1260        Err(_) => return CgroupProbeOutcome::NoVerdict,
1261    };
1262    let argv = [path.as_ptr(), std::ptr::null::<libc::c_char>()];
1263    let envp = [std::ptr::null::<libc::c_char>()];
1264    // Ask for a pidfd atomically at creation (CLONE_PIDFD):
1265    // only this parent uses the fd after clone3 returns, so
1266    // the kernel's put_user/fd_install ordering inside clone
1267    // cannot race its use here. On a kernel that refuses
1268    // CLONE_PIDFD, the clone below is retried without it and
1269    // the numeric pid path is used instead.
1270    let mut pidfd_slot: i32 = -1;
1271    let mut args = CloneArgs {
1272        flags: CLONE_INTO_CGROUP | CLONE_PIDFD,
1273        pidfd: &mut pidfd_slot as *mut i32 as usize as u64,
1274        child_tid: 0,
1275        parent_tid: 0,
1276        exit_signal: libc::SIGCHLD as u64,
1277        stack: 0,
1278        stack_size: 0,
1279        tls: 0,
1280        set_tid: 0,
1281        set_tid_size: 0,
1282        cgroup: cgroup as u64,
1283    };
1284    let mut result = unsafe {
1285        syscall6(
1286            libc::SYS_clone3,
1287            [
1288                &args as *const CloneArgs as usize,
1289                core::mem::size_of::<CloneArgs>(),
1290                0,
1291                0,
1292                0,
1293                0,
1294            ],
1295        )
1296    };
1297    if result < 0 && pidfd_slot < 0 {
1298        // Possibly a kernel without CLONE_PIDFD: retry once
1299        // without it. A failed clone created no child, so the
1300        // retry cannot create a second one.
1301        args.flags = CLONE_INTO_CGROUP;
1302        args.pidfd = 0;
1303        result = unsafe {
1304            syscall6(
1305                libc::SYS_clone3,
1306                [
1307                    &args as *const CloneArgs as usize,
1308                    core::mem::size_of::<CloneArgs>(),
1309                    0,
1310                    0,
1311                    0,
1312                    0,
1313                ],
1314            )
1315        };
1316    }
1317    if result == 0 {
1318        // The probe child. Raw system calls only, until execve:
1319        // one entry byte on the pipe, then execve. Everything it
1320        // touches was prepared before the clone.
1321        unsafe {
1322            syscall6(libc::SYS_close, [pipe_fds[0] as usize, 0, 0, 0, 0, 0]);
1323            #[cfg(test)]
1324            if support.inject_probe_child_hangs_before_entry {
1325                loop {
1326                    syscall6(libc::SYS_pause, [0; 6]);
1327                }
1328            }
1329            let byte = *b"E";
1330            syscall6(
1331                libc::SYS_write,
1332                [
1333                    pipe_fds[1] as usize,
1334                    &byte as *const u8 as usize,
1335                    1,
1336                    0,
1337                    0,
1338                    0,
1339                ],
1340            );
1341            #[cfg(test)]
1342            if support.inject_probe_child_kills_after_entry {
1343                // Outside the pidfd rule by the design's own
1344                // ruling: a process signalling itself (tgkill
1345                // on its own thread) cannot hit a reused pid.
1346                let pid = syscall6(libc::SYS_getpid, [0; 6]);
1347                syscall6(
1348                    libc::SYS_tgkill,
1349                    [pid as usize, pid as usize, libc::SIGKILL as usize, 0, 0, 0],
1350                );
1351            }
1352            #[cfg(test)]
1353            if support.inject_probe_child_hangs_after_entry {
1354                loop {
1355                    syscall6(libc::SYS_pause, [0; 6]);
1356                }
1357            }
1358            syscall6(
1359                libc::SYS_execve,
1360                [
1361                    path.as_ptr() as usize,
1362                    argv.as_ptr() as usize,
1363                    envp.as_ptr() as usize,
1364                    0,
1365                    0,
1366                    0,
1367                ],
1368            );
1369            loop {
1370                syscall6(libc::SYS_exit_group, [127, 0, 0, 0, 0, 0]);
1371            }
1372        }
1373    }
1374    unsafe {
1375        libc::close(pipe_fds[1]);
1376    }
1377    if result < 0 {
1378        unsafe {
1379            libc::close(pipe_fds[0]);
1380        }
1381        return CgroupProbeOutcome::NoVerdict;
1382    }
1383    let pid = result as libc::pid_t;
1384    support.probe_child_pid.store(pid, Ordering::Relaxed);
1385    {
1386        *probe_child_slot(&support) = ProbeChildSlot::Owned(OwnedProbeChild {
1387            pidfd: if pidfd_slot >= 0 {
1388                Some(OwnedFd(pidfd_slot))
1389            } else {
1390                None
1391            },
1392            via_pidfd: pidfd_slot >= 0,
1393            pid,
1394        });
1395        support
1396            .probe_child_waited_via_pidfd
1397            .store(pidfd_slot >= 0, Ordering::Relaxed);
1398        #[cfg(test)]
1399        if pidfd_slot >= 0 {
1400            fn fd_ino(fd: i32) -> (u64, u64) {
1401                let mut stat: libc::stat = unsafe { std::mem::zeroed() };
1402                if unsafe { libc::fstat(fd, &mut stat) } == 0 {
1403                    (stat.st_dev, stat.st_ino)
1404                } else {
1405                    panic!(
1406                        "cannot stat the probe child's pidfd for the fd census: {}",
1407                        io::Error::last_os_error()
1408                    );
1409                }
1410            }
1411            let dup = unsafe { libc::fcntl(pidfd_slot, libc::F_DUPFD_CLOEXEC, 0) };
1412            if dup < 0 {
1413                panic!(
1414                    "cannot duplicate the probe child's pidfd for test cleanup: {}",
1415                    io::Error::last_os_error()
1416                );
1417            }
1418            *support
1419                .probe_child_cleanup_pidfd
1420                .lock()
1421                .unwrap_or_else(|poisoned| poisoned.into_inner()) = Some(OwnedFd(dup));
1422            *support
1423                .probe_child_fd_inodes
1424                .lock()
1425                .unwrap_or_else(|poisoned| poisoned.into_inner()) =
1426                vec![fd_ino(pidfd_slot), fd_ino(dup)];
1427            // Attribution check: a pidfd for a different
1428            // process (ourselves) must have a different
1429            // (dev, inode). On pre-pidfs kernels every pidfd
1430            // shares one anon inode, and the census in the
1431            // tests refuses by name instead of miscounting.
1432            let self_pidfd = unsafe { libc::syscall(libc::SYS_pidfd_open, libc::getpid(), 0usize) };
1433            if self_pidfd < 0 {
1434                // The attribution check itself could not run.
1435                // That is a fixture failure of its own, NOT
1436                // evidence of a shared inode: never let it
1437                // stand in for pidfs absence.
1438                panic!(
1439                    "cannot open a pidfd for this process to check the fd census attribution: {}",
1440                    io::Error::last_os_error()
1441                );
1442            }
1443            let shared = fd_ino(self_pidfd as i32) == fd_ino(pidfd_slot);
1444            support
1445                .probe_child_fd_census_shared
1446                .store(shared, Ordering::Relaxed);
1447            unsafe {
1448                libc::close(self_pidfd as i32);
1449            }
1450        }
1451    }
1452    // Poll the entry pipe and the child to the bound. The child is
1453    // this process's own: nobody else can reap it, so its pid
1454    // stays valid until this loop does.
1455    let started = std::time::Instant::now();
1456    let mut entered = false;
1457    let outcome = loop {
1458        if !entered {
1459            let remaining = support.probe_bound.saturating_sub(started.elapsed());
1460            let mut poll = libc::pollfd {
1461                fd: pipe_fds[0],
1462                events: libc::POLLIN,
1463                revents: 0,
1464            };
1465            // A test seam holds the read back until after the
1466            // reap below, so the drain there is the only path by
1467            // which the byte can be seen.
1468            #[cfg(test)]
1469            let ready = if support.inject_probe_parent_read_delay {
1470                std::thread::sleep(std::time::Duration::from_millis(2));
1471                0
1472            } else {
1473                unsafe { libc::poll(&mut poll, 1, remaining.as_millis().min(5) as i32) }
1474            };
1475            #[cfg(not(test))]
1476            let ready = unsafe { libc::poll(&mut poll, 1, remaining.as_millis().min(5) as i32) };
1477            if ready > 0 {
1478                let mut byte = [0u8; 1];
1479                let read = unsafe { libc::read(pipe_fds[0], byte.as_mut_ptr().cast(), 1) };
1480                if read == 1 {
1481                    entered = true;
1482                }
1483            }
1484        } else {
1485            std::thread::sleep(std::time::Duration::from_millis(2));
1486        }
1487        match wait_probe_child(&support, pid, false, entered) {
1488            ChildWait::Reaped(status) => {
1489                // The child may have written its byte and died
1490                // before this loop read the pipe (a poll that
1491                // timed out or was interrupted): drain what is
1492                // already there before classifying, or a child
1493                // that entered reads as one the kernel killed
1494                // before it could run.
1495                entered = drain_entry_pipe(pipe_fds[0], entered);
1496                mark_probe_child_reaped(&support);
1497                break probe_verdict(entered, status);
1498            }
1499            ChildWait::NoWaitable => {
1500                // No waitable child remains (a SIGCHLD
1501                // disposition of SIG_IGN reaps children itself):
1502                // the cleanup this loop owns is complete, but
1503                // ECHILD carries no wait status, so no successful
1504                // run can be claimed from it. The witness,
1505                // drained as after any exit, is the only
1506                // evidence the child ran.
1507                entered = drain_entry_pipe(pipe_fds[0], entered);
1508                mark_probe_child_reaped(&support);
1509                break if entered {
1510                    CgroupProbeOutcome::ChildRan
1511                } else {
1512                    CgroupProbeOutcome::NoVerdict
1513                };
1514            }
1515            ChildWait::Failed(errno) => {
1516                if errno != Some(libc::EINTR) {
1517                    // A wait that failed another way proves
1518                    // nothing about the child: it is still this
1519                    // process's own, so keep owing its reap for
1520                    // the next start to retry, and take no kernel
1521                    // verdict from it.
1522                    eprintln!(
1523                        "manifest-plan: waiting for the clone3 probe child {pid} failed \
1524                         (errno {errno:?}); no verdict is taken from it"
1525                    );
1526                    support.keep_unreaped_probe_child(pid);
1527                    break CgroupProbeOutcome::WaitFailed;
1528                }
1529            }
1530            ChildWait::StillRunning => {}
1531        }
1532        if started.elapsed() >= support.probe_bound {
1533            // Kill and reap, both bounded: the kill's result is
1534            // checked (the child may already be dead), the reap
1535            // polls for a one-second grace, and a child still not
1536            // reaped then (stuck in the kernel, say) is handed to
1537            // a background reaper while the timeout is published.
1538            // It stays this process's child throughout, so nobody
1539            // else can reap it and its pid cannot be reused first.
1540            #[cfg(test)]
1541            let killed = if let Some(errno) = support.inject_probe_timeout_kill_errno {
1542                -errno as isize
1543            } else {
1544                kill_probe_child(&support, pid) as isize
1545            };
1546            #[cfg(not(test))]
1547            let killed = kill_probe_child(&support, pid) as isize;
1548            // Reaped by this loop, or no waitable child remains:
1549            // either completes the cleanup. Any other wait error
1550            // completes nothing and hands the child on, below.
1551            let mut reaped = false;
1552            let mut no_waitable = false;
1553            let grace = std::time::Instant::now();
1554            while grace.elapsed() < std::time::Duration::from_secs(1) {
1555                match wait_probe_child(&support, pid, true, entered) {
1556                    ChildWait::Reaped(_) => {
1557                        reaped = true;
1558                        break;
1559                    }
1560                    ChildWait::NoWaitable => {
1561                        no_waitable = true;
1562                        break;
1563                    }
1564                    ChildWait::Failed(Some(libc::EINTR)) => continue,
1565                    ChildWait::Failed(_) => break,
1566                    ChildWait::StillRunning => {}
1567                }
1568                std::thread::sleep(std::time::Duration::from_millis(5));
1569            }
1570            if reaped || no_waitable {
1571                mark_probe_child_reaped(&support);
1572            }
1573            if killed != 0 && !reaped && !no_waitable {
1574                // The kill failed (the child was already dead, or
1575                // a policy refuses kill): ownership passes to the
1576                // reaper or the retry record below. The timeout
1577                // stands either way.
1578                eprintln!(
1579                    "manifest-plan: clone3 probe kill of child {pid} failed; \
1580                     its cleanup is handed on"
1581                );
1582            }
1583            if !reaped && !no_waitable {
1584                // Hand the owned handle to the reaper over a
1585                // channel: a move. On a successful spawn the
1586                // caller sends it and keeps no copy; on a
1587                // failed spawn (or a failed send) the caller
1588                // still owns it and runs the fallback below.
1589                #[cfg(test)]
1590                let spawn_fails = support.inject_probe_reaper_spawn_fails;
1591                #[cfg(not(test))]
1592                let spawn_fails = false;
1593                let mut spawn_error: Option<io::Error> = if spawn_fails {
1594                    Some(io::Error::from_raw_os_error(libc::EAGAIN))
1595                } else {
1596                    None
1597                };
1598                let mut handed_off = false;
1599                if !spawn_fails {
1600                    match take_probe_child(&support) {
1601                        Some(owned) => {
1602                            let (sender, receiver) = std::sync::mpsc::channel::<OwnedProbeChild>();
1603                            let rs = std::sync::Arc::clone(&support);
1604                            match std::thread::Builder::new()
1605                                .name("clone3-probe-reaper".to_string())
1606                                .spawn(move || {
1607                                    if let Ok(owned) = receiver.recv() {
1608                                        reap_owned(owned, &rs);
1609                                    }
1610                                }) {
1611                                Ok(_handle) => match sender.send(owned) {
1612                                    Ok(()) => handed_off = true,
1613                                    Err(error) => {
1614                                        return_probe_child(&support, error.0);
1615                                        spawn_error =
1616                                            Some(io::Error::from_raw_os_error(libc::EAGAIN));
1617                                    }
1618                                },
1619                                Err(error) => {
1620                                    // The spawn failed before the
1621                                    // send: the caller still owns
1622                                    // the handle. Put it back and
1623                                    // run the fallback below.
1624                                    return_probe_child(&support, owned);
1625                                    spawn_error = Some(error);
1626                                }
1627                            }
1628                        }
1629                        None => {
1630                            spawn_error = Some(io::Error::from_raw_os_error(libc::EAGAIN));
1631                        }
1632                    }
1633                }
1634                let spawned: Result<(), io::Error> = if handed_off {
1635                    Ok(())
1636                } else {
1637                    Err(spawn_error.unwrap_or_else(|| io::Error::from_raw_os_error(libc::EAGAIN)))
1638                };
1639                if let Err(error) = spawned {
1640                    // No reaper could be started: this caller
1641                    // stays the owner, but it may not wait
1642                    // unboundedly for the child. It kills the
1643                    // child (by pidfd where held) and polls for
1644                    // one grace; a child still unreaped then is
1645                    // recorded loudly as owed, and no verdict is
1646                    // taken from its cleanup.
1647                    eprintln!(
1648                        "manifest-plan: cannot spawn the clone3 probe reaper ({error}); \
1649                         killing the probe child and polling its reap here instead"
1650                    );
1651                    #[cfg(test)]
1652                    let fallback_killed =
1653                        if let Some(errno) = support.inject_probe_fallback_kill_errno {
1654                            -errno as isize
1655                        } else {
1656                            kill_probe_child(&support, pid) as isize
1657                        };
1658                    #[cfg(not(test))]
1659                    let fallback_killed = kill_probe_child(&support, pid) as isize;
1660                    if fallback_killed != 0 {
1661                        eprintln!(
1662                            "manifest-plan: killing probe child {pid} after the failed \
1663                             reaper spawn also failed (errno {fallback_killed})"
1664                        );
1665                    }
1666                    let fallback_grace = std::time::Instant::now();
1667                    let mut fallback_done = false;
1668                    while fallback_grace.elapsed() < std::time::Duration::from_secs(1) {
1669                        match wait_probe_child(&support, pid, false, entered) {
1670                            ChildWait::Reaped(_) | ChildWait::NoWaitable => {
1671                                fallback_done = true;
1672                                break;
1673                            }
1674                            ChildWait::Failed(Some(libc::EINTR)) => continue,
1675                            ChildWait::Failed(_) => break,
1676                            ChildWait::StillRunning => {}
1677                        }
1678                        std::thread::sleep(std::time::Duration::from_millis(5));
1679                    }
1680                    if fallback_done {
1681                        mark_probe_child_reaped(&support);
1682                    } else {
1683                        support.keep_unreaped_probe_child(pid);
1684                        break CgroupProbeOutcome::NoVerdict;
1685                    }
1686                }
1687            }
1688            break CgroupProbeOutcome::TimedOut;
1689        }
1690    };
1691    unsafe {
1692        libc::close(pipe_fds[0]);
1693    }
1694    #[cfg(test)]
1695    if !support.injected_wait_consumed() {
1696        // No injected wait answered, so no fixture cleanup
1697        // needs the duplicate: take and drop it here, so a
1698        // real probe leaves no extra fd behind.
1699        let _dup = support
1700            .probe_child_cleanup_pidfd
1701            .lock()
1702            .unwrap_or_else(|poisoned| poisoned.into_inner())
1703            .take();
1704    }
1705    outcome
1706}

1708/// Reap `pid`, retrying a wait that a signal interrupts. A
1709/// wait that fails another way does not complete the cleanup:
1710/// the child is recorded as still owed, for the next start to
1711/// retry, rather than silently dropped.
1712/// Reap the probe child this reaper owns, retrying a wait
1713/// that a signal interrupts. It is the handle's only owner
1714/// after the move. A wait that fails another way returns
1715/// the owned handle to the slot as a still-owed debt, for
1716/// the next start to retry, rather than silently dropping
1717/// it.
1718fn reap_owned(owned: OwnedProbeChild, support: &Clone3Support) {
1719    loop {
1720        match wait_child_once(&owned, true) {
1721            ChildWait::Reaped(_) | ChildWait::NoWaitable => {
1722                mark_probe_child_reaped_from_reaper(support);
1723                return;
1724            }
1725            ChildWait::Failed(Some(libc::EINTR)) => continue,
1726            ChildWait::Failed(_) => {
1727                support.return_owned_as_debt(owned);
1728                return;
1729            }
1730            ChildWait::StillRunning => continue,
1731        }
1732    }
1733}

1735/// The probe's verdict from the child's entry witness and its wait
1736/// status. The kernel bug kills the child while the kernel creates
1737/// it, before it enters the child code, so only a SIGKILL with no
1738/// entry is that refusal; a child that entered and was then killed
1739/// (for example by its own program, after exec) is a child that
1740/// ran, and clone3 stays usable.
1741fn probe_verdict(entered: bool, status: i32) -> CgroupProbeOutcome {
1742    if libc::WIFEXITED(status) && libc::WEXITSTATUS(status) == 0 {
1743        CgroupProbeOutcome::ChildRan
1744    } else if !entered && libc::WIFSIGNALED(status) && libc::WTERMSIG(status) == libc::SIGKILL {
1745        CgroupProbeOutcome::ChildKilledBySigkill
1746    } else if entered && libc::WIFSIGNALED(status) {
1747        // Entered, then killed (for example by its own program):
1748        // the clone path works, so this is a child that ran.
1749        CgroupProbeOutcome::ChildRan

1759/// What an errno from clone3 means. The cgroup errnos are those of Linux
1760/// v6.12's `cgroup_css_set_fork` (kernel/cgroup/cgroup.c) and what it calls;
1761/// the others are clone3's own argument checks (kernel/fork.c).
1762#[derive(Clone, Copy, Debug, PartialEq, Eq)]
1763enum Refusal {
1764    /// clone3, or one of its flags, is not available here: an older kernel
1765    /// (ENOSYS before 5.3; E2BIG or EINVAL for arguments or a flag it does not
1766    /// know, before 5.7) or a seccomp filter (ENOSYS).
1767    Unusable,
1768    /// Either a seccomp filter refused clone3 or the cgroup's write
1769    /// permission check did (`inode_permission`, for an immutable file or a
1770    /// security module); [`probe_refuses_clone3`] tells which.
1771    FilterOrJoin,
1772    /// The cgroup refused the child: not a cgroup directory (EBADF), not
1773    /// allowed (EACCES, ENOENT), on a read-only mount (EROFS), removed
1774    /// (ENODEV), a cgroup that may not hold processes (EBUSY, EOPNOTSUPP).
1775    Join,
1776    /// Anything else, such as EAGAIN or ENOMEM: no child was created.
1777    Failed,
1778}
1779
1780fn refusal(errno: i32) -> Refusal {
1781    match errno {
1782        libc::ENOSYS | libc::E2BIG | libc::EINVAL => Refusal::Unusable,
1783        libc::EPERM => Refusal::FilterOrJoin,
1784        libc::EBADF
1785        | libc::EACCES
1786        | libc::ENOENT
1787        | libc::EROFS
1788        | libc::ENODEV
1789        | libc::EBUSY
1790        | libc::EOPNOTSUPP => Refusal::Join,
1791        _ => Refusal::Failed,
1792    }
1793}

1795/// Whether clone3 itself is refused here, asked by a call that cannot create
1796/// a child: clone3 with no arguments, which Linux refuses with EINVAL before
1797/// it reads anything (`copy_clone_args_from_user`: a size below
1798/// `CLONE_ARGS_SIZE_VER0`), unless a seccomp filter refuses the call first
1799/// (EPERM or ENOSYS).
1800fn probe_refuses_clone3(support: &Clone3Support) -> bool {
1801    matches!(probe_clone3(support), libc::EPERM | libc::ENOSYS)
1802}

1831/// Start `program` in its cgroup, unless clone3 is unusable here.
1832pub(crate) fn start(support: &Clone3Support, program: &Program) -> Started {
1833    // A probe child whose cleanup could not complete is still
1834    // this process's own; retry its reap here, where every real
1835    // start passes, without blocking.
1836    support.reap_unreaped_probe_child();
1837    if let Some(reason) = support.unusable.get() {
1838        return Started::Unusable {
1839            reason: reason.clone(),
1840        };
1841    }
1842    let prepared = match Prepared::new(support, program) {
1843        Ok(prepared) => prepared,
1844        Err(error) => return Started::Failed(error),
1845    };
1846    let result = CHILD_STACK.with_borrow_mut(|stack| {
1847        if stack.is_none() {
1848            *stack = Some(ChildStack::new()?);
1849        }
1850        let stack = stack.as_ref().expect("set above");
1851        Ok::<i64, io::Error>(clone_into_cgroup(support, &prepared, stack, program.cgroup))
1852    });
1853    let result = match result {
1854        Ok(result) => result,
1855        Err(error) => return Started::Failed(error),
1856    };
1857    if result < 0 {
1858        let errno = (-result) as i32;
1859        let error = io::Error::from_raw_os_error(errno);
1860        let unusable = |error: io::Error| {
1861            let reason = format!("clone3 with CLONE_INTO_CGROUP was refused: {error}");
1862            remember_unusable(support, &reason);
1863            Started::Unusable { reason }
1864        };
1865        return match refusal(errno) {
1866            Refusal::Unusable => unusable(error),
1867            Refusal::FilterOrJoin if probe_refuses_clone3(support) => unusable(error),
1868            Refusal::FilterOrJoin | Refusal::Join => Started::JoinFailed(error),
1869            Refusal::Failed => Started::Failed(error),
1870        };
1871    }
1872    let pid = result as u32;
1873    // CLONE_VFORK returned only after the child execed or exited, and the
1874    // kernel's completion of that wait orders the child's store before this
1875    // load. A child killed by a signal before it execed stored nothing, so it
1876    // reads as a started program that a signal killed, which its wait
1877    // reports.
1878    let failure = prepared.state.failure.load(Ordering::Relaxed);
1879    if failure != 0 {
1880        // The child stored why it could not exec and exited; reap it, retrying
1881        // a wait that a signal interrupts so no zombie is left.
1882        let mut status = 0;
1883        while unsafe { libc::waitpid(pid as libc::pid_t, &mut status, 0) } == -1
1884            && io::Error::last_os_error().raw_os_error() == Some(libc::EINTR)
1885        {}
1886        return Started::Failed(io::Error::from_raw_os_error(failure));
1887    }
1888    Started::Running { pid }
1889}

1907/// Calls clone3 for `prepared` on `stack`, with every signal blocked on this
1908/// thread so that no handler of this process can run in the child before it
1909/// resets them. Returns the child's PID or the negative errno.
1910fn clone_into_cgroup(
1911    support: &Clone3Support,
1912    prepared: &Prepared,
1913    stack: &ChildStack,
1914    cgroup: RawFd,
1915) -> i64 {
1916    #[cfg(test)]
1917    {
1918        let attempt = support.attempts.fetch_add(1, Ordering::Relaxed);
1919        if let Some(errno) = support.inject {
1920            if attempt < support.inject_limit {
1921                return -i64::from(errno);
1922            }
1923        }
1924    }
1925    #[cfg(not(test))]
1926    let _ = support;
1927    let args = CloneArgs {
1928        flags: (libc::CLONE_VM | libc::CLONE_VFORK) as u64 | CLONE_INTO_CGROUP,
1929        pidfd: 0,
1930        child_tid: 0,
1931        parent_tid: 0,
1932        exit_signal: libc::SIGCHLD as u64,
1933        stack: stack.low as u64,
1934        stack_size: CHILD_STACK_BYTES as u64,
1935        tls: 0,
1936        set_tid: 0,
1937        set_tid_size: 0,
1938        cgroup: cgroup as u64,
1939    };
1940    // Block every signal, glibc's internal ones too, which pthread_sigmask
1941    // leaves unblocked; glibc's posix_spawn blocks them the same way.
1942    let all: u64 = !0;
1943    let mut previous: u64 = 0;
1944    unsafe {
1945        syscall6(
1946            libc::SYS_rt_sigprocmask,
1947            [
1948                libc::SIG_SETMASK as usize,
1949                &all as *const u64 as usize,
1950                &mut previous as *mut u64 as usize,
1951                8,
1952                0,
1953                0,
1954            ],
1955        );
1956    }
1957    let result = unsafe { arch::clone3_then_child(&args, &prepared.state) };
1958    unsafe {
1959        syscall6(
1960            libc::SYS_rt_sigprocmask,
1961            [
1962                libc::SIG_SETMASK as usize,
1963                &previous as *const u64 as usize,
1964                0,
1965                8,
1966                0,
1967                0,
1968            ],
1969        );
1970    }
1971    result
1972}

1975const CLONE_INTO_CGROUP: u64 = 0x2_0000_0000;

1980/// The child's stack: no more than raw system calls run on it. Its frames
1981/// stay far under 1 KiB even unoptimized, and execve runs on the kernel's
1982/// stack, so 64 KiB leaves more than 64 times what it needs. A guard page
1983/// below it turns an overflow into the child's own SIGSEGV, never a write
1984/// into this process's memory; [`start`] then reports a started program that
1985/// SIGSEGV killed, as for any child a signal kills before it execs.
1986const CHILD_STACK_BYTES: usize = 64 * 1024;
1987
1988struct ChildStack {
1989    /// The mapping: the guard page, then the stack.
1990    base: *mut libc::c_void,
1991    len: usize,
1992    /// The stack's lowest address; it grows down from `low + CHILD_STACK_BYTES`.
1993    low: *mut u8,
1994}
1995
1996impl ChildStack {
1997    fn new() -> io::Result<Self> {
1998        let page = usize::try_from(unsafe { libc::sysconf(libc::_SC_PAGESIZE) })
1999            .map_err(|_| io::Error::other("cannot read the page size"))?;
2000        let len = page + CHILD_STACK_BYTES;
2001        let base = unsafe {
2002            libc::mmap(
2003                std::ptr::null_mut(),
2004                len,
2005                libc::PROT_READ | libc::PROT_WRITE,
2006                libc::MAP_PRIVATE | libc::MAP_ANONYMOUS | libc::MAP_STACK,
2007                -1,
2008                0,
2009            )
2010        };
2011        if base == libc::MAP_FAILED {
2012            return Err(io::Error::last_os_error());
2013        }
2014        if unsafe { libc::mprotect(base, page, libc::PROT_NONE) } != 0 {
2015            let error = io::Error::last_os_error();
2016            unsafe { libc::munmap(base, len) };
2017            return Err(error);
2018        }
2019        Ok(Self {
2020            base,
2021            len,
2022            low: unsafe { base.cast::<u8>().add(page) },
2023        })
2024    }
2025}
2026
2027impl Drop for ChildStack {
2028    fn drop(&mut self) {
2029        unsafe { libc::munmap(self.base, self.len) };
2030    }
2031}

2048/// What [`vfork_child`] reads through one pointer. Every pointer names
2049/// storage in [`Prepared`], which outlives the child's use of it: this thread
2050/// waits in clone3 until the child has execed or exited.
2051#[repr(C)]
2052pub(crate) struct ChildState {
2053    /// Paths to try, in order, to execute the program.
2054    candidates: *const *const libc::c_char,
2055    candidate_count: usize,
2056    /// NULL-terminated argv and envp.
2057    argv: *const *const libc::c_char,
2058    envp: *const *const libc::c_char,
2059    /// `/bin/sh`, a slot for the path that failed with ENOEXEC, then argv
2060    /// from its second entry, NULL-terminated: how execvp runs a file without
2061    /// a `#!` line. The child writes the slot.
2062    shell_argv: *mut *const libc::c_char,
2063    cwd: *const libc::c_char,
2064    stdio: [RawFd; 3],
2065    /// The errno that stopped the child, or 0 if it reached the program.
2066    failure: AtomicI32,
2067    /// For a test: the child kills itself first.
2068    #[cfg(test)]
2069    kill_before_exec: bool,
2070}
2071
2072impl Prepared {
2073    fn new(support: &Clone3Support, program: &Program) -> io::Result<Self> {
2074        #[cfg(not(test))]
2075        let _ = support;
2076        let invalid = |what: &str| io::Error::new(io::ErrorKind::InvalidInput, what.to_string());
2077        let c_string =
2078            |bytes: Vec<u8>| CString::new(bytes).map_err(|_| invalid("a NUL byte in an argument"));
2079        let mut strings = Vec::new();
2080        let mut pointers = Vec::new();
2081        let name = program
2082            .argv
2083            .first()
2084            .ok_or_else(|| invalid("an empty argv"))?;
2085        // The environment the program gets: this process's, with the
2086        // overrides set over it.
2087        let mut environment: BTreeMap<OsString, OsString> = std::env::vars_os().collect();
2088        for (key, value) in program.env_overrides {
2089            environment.insert(key.into(), value.into());
2090        }
2091        let candidates: Vec<Vec<u8>> = if name.contains('/') {
2092            vec![name.as_bytes().to_vec()]
2093        } else {
2094            // execvp's search: each element of PATH, an empty one meaning the
2095            // working directory, and glibc's default when PATH is unset.
2096            let path = environment
2097                .get(std::ffi::OsStr::new("PATH"))
2098                .map(|path| path.as_bytes().to_vec())
2099                .unwrap_or_else(|| b"/bin:/usr/bin".to_vec());
2100            path.split(|byte| *byte == b':')
2101                .map(|dir| {
2102                    if dir.is_empty() {
2103                        name.as_bytes().to_vec()
2104                    } else {
2105                        let mut full = dir.to_vec();
2106                        full.push(b'/');
2107                        full.extend_from_slice(name.as_bytes());
2108                        full
2109                    }
2110                })
2111                .collect()
2112        };
2113        let pointer_list = |items: Vec<Vec<u8>>, strings: &mut Vec<CString>| {
2114            let mut list: Vec<*const libc::c_char> = Vec::with_capacity(items.len() + 1);
2115            for item in items {
2116                let string = c_string(item)?;
2117                list.push(string.as_ptr());
2118                strings.push(string);
2119            }
2120            Ok::<_, io::Error>(list)
2121        };
2122        let candidate_list = pointer_list(candidates, &mut strings)?;
2123        let candidate_count = candidate_list.len();
2124        let mut argv = pointer_list(
2125            program
2126                .argv
2127                .iter()
2128                .map(|arg| arg.as_bytes().to_vec())
2129                .collect(),
2130            &mut strings,
2131        )?;
2132        argv.push(std::ptr::null());
2133        let mut envp = pointer_list(
2134            environment
2135                .into_iter()
2136                .map(|(key, value)| {
2137                    let mut pair = key.into_vec();
2138                    pair.push(b'=');
2139                    pair.extend_from_slice(value.as_bytes());
2140                    pair
2141                })
2142                .collect(),
2143            &mut strings,
2144        )?;
2145        envp.push(std::ptr::null());
2146        let mut shell_argv = pointer_list(vec![b"/bin/sh".to_vec()], &mut strings)?;
2147        shell_argv.push(std::ptr::null());
2148        shell_argv.extend_from_slice(&argv[1..]);
2149        let cwd = c_string(program.cwd.as_os_str().as_bytes().to_vec())?;
2150        let cwd_pointer = cwd.as_ptr();
2151        strings.push(cwd);
2152        let state = ChildState {
2153            candidates: candidate_list.as_ptr(),
2154            candidate_count,
2155            argv: argv.as_ptr(),
2156            envp: envp.as_ptr(),
2157            shell_argv: shell_argv.as_mut_ptr(),
2158            cwd: cwd_pointer,
2159            stdio: program.stdio,
2160            failure: AtomicI32::new(0),
2161            #[cfg(test)]
2162            kill_before_exec: support.kill_child_before_exec,
2163        };
2164        pointers.push(candidate_list);
2165        pointers.push(argv);
2166        pointers.push(envp);
2167        pointers.push(shell_argv);
2168        Ok(Self {
2169            state,
2170            _strings: strings,
2171            _pointers: pointers,
2172        })
2173    }
2174}

2192/// The child: set up the program's process state as `std::process::Command`
2193/// would, then execute it, or store why not and exit.
2194///
2195/// # Safety
2196/// Entered only from [`arch::clone3_then_child`], on the child's own stack,
2197/// with `state` prepared by [`Prepared::new`]. It never returns, allocates,
2198/// takes a lock, or calls libc.
2199unsafe extern "C" fn vfork_child(state: *const ChildState) -> ! {
2200    unsafe {
2201        let state = &*state;
2202        let fail = |errno: isize| -> ! {
2203            state.failure.store(errno as i32, Ordering::Relaxed);
2204            // exit_group does not return; the loop only gives the closure
2205            // its `!` type.
2206            loop {
2207                syscall6(libc::SYS_exit_group, [127, 0, 0, 0, 0, 0]);
2208            }
2209        };
2210        let check = |result: isize| {
2211            if result < 0 {
2212                fail(-result);
2213            }
2214        };
2215        #[cfg(test)]
2216        if state.kill_before_exec {
2217            // Outside the pidfd rule by the design's own
2218            // ruling: a process signalling itself (tgkill
2219            // on its own thread) cannot hit a reused pid.
2220            let pid = syscall6(libc::SYS_getpid, [0; 6]);
2221            syscall6(
2222                libc::SYS_tgkill,
2223                [pid as usize, pid as usize, libc::SIGKILL as usize, 0, 0, 0],
2224            );
2225        }
2226        // The program leads a process group of its own.
2227        check(syscall6(libc::SYS_setpgid, [0, 0, 0, 0, 0, 0]));
2228        // Its stdin, stdout and stderr. The caller passes descriptors above 2,
2229        // so no target is also a later source; dup3 leaves the copies open
2230        // across exec.
2231        let mut target = 0;
2232        while target < 3 {
2233            check(syscall6(
2234                libc::SYS_dup3,
2235                [state.stdio[target] as usize, target, 0, 0, 0, 0],
2236            ));
2237            target += 1;
2238        }
2239        check(syscall6(
2240            libc::SYS_chdir,
2241            [state.cwd as usize, 0, 0, 0, 0, 0],
2242        ));
2243        // Every signal this process handles goes back to its default action
2244        // before any is unblocked, so no handler of this process can run in
2245        // the child; SIGPIPE, which Rust programs ignore, too, as
2246        // `std::process::Command` resets it. exec would reset handled ones
2247        // anyway; ignored ones stay ignored, as with Command. clone3's
2248        // CLONE_CLEAR_SIGHAND would reset the handled ones but leaves SIG_IGN
2249        // alone (`flush_signal_handlers(tsk, 0)`), so it would not reset
2250        // SIGPIPE as Command does; hence this loop. The child has its own copy
2251        // of the handlers (no CLONE_SIGHAND), so none of this touches this
2252        // process's.
2253        let mut signal = 1;
2254        while signal < 65 {
2255            if signal != libc::SIGKILL as usize && signal != libc::SIGSTOP as usize {
2256                let mut action = KernelSigaction {
2257                    handler: 0,
2258                    flags: 0,
2259                    restorer: 0,
2260                    mask: 0,
2261                };
2262                let read = syscall6(
2263                    libc::SYS_rt_sigaction,
2264                    [
2265                        signal,
2266                        0,
2267                        &mut action as *mut KernelSigaction as usize,
2268                        8,
2269                        0,
2270                        0,
2271                    ],
2272                );
2273                let handled = action.handler != libc::SIG_DFL && action.handler != libc::SIG_IGN;
2274                if read == 0 && (handled || signal == libc::SIGPIPE as usize) {
2275                    let default = KernelSigaction {
2276                        handler: libc::SIG_DFL,
2277                        flags: 0,
2278                        restorer: 0,
2279                        mask: 0,
2280                    };
2281                    check(syscall6(
2282                        libc::SYS_rt_sigaction,
2283                        [
2284                            signal,
2285                            &default as *const KernelSigaction as usize,
2286                            0,
2287                            8,
2288                            0,
2289                            0,
2290                        ],
2291                    ));
2292                }
2293            }
2294            signal += 1;
2295        }
2296        let empty: u64 = 0;
2297        check(syscall6(
2298            libc::SYS_rt_sigprocmask,
2299            [
2300                libc::SIG_SETMASK as usize,
2301                &empty as *const u64 as usize,
2302                0,
2303                8,
2304                0,
2305                0,
2306            ],
2307        ));
2308        // execvp's search: stop at the first error other than one that means
2309        // "not here"; report EACCES if any candidate gave it.
2310        let mut denied = false;
2311        let mut last = libc::ENOENT as isize;
2312        let mut index = 0;
2313        while index < state.candidate_count {
2314            let path = *state.candidates.add(index);
2315            let mut errno = -syscall6(
2316                libc::SYS_execve,
2317                [
2318                    path as usize,
2319                    state.argv as usize,
2320                    state.envp as usize,
2321                    0,
2322                    0,
2323                    0,
2324                ],
2325            );
2326            if errno as i32 == libc::ENOEXEC {
2327                // As glibc's execvp (`maybe_script_execute`): run the file
2328                // under the shell, and judge the shell's error like any other
2329                // candidate's, so a missing shell continues the search.
2330                *state.shell_argv.add(1) = path;
2331                let shell = *state.shell_argv;
2332                errno = -syscall6(
2333                    libc::SYS_execve,
2334                    [
2335                        shell as usize,
2336                        state.shell_argv as usize,
2337                        state.envp as usize,
2338                        0,
2339                        0,
2340                        0,
2341                    ],
2342                );
2343            }
2344            match errno as i32 {
2345                libc::EACCES => denied = true,
2346                libc::ENOENT | libc::ENOTDIR | libc::ESTALE | libc::ENODEV | libc::ETIMEDOUT => {}

2361    /// A raw system call (the register convention of Reverie's
2362    /// `reverie-inguest` exec-stop bridge, `arch::syscall6`): number and
2363    /// result in rax, arguments in rdi, rsi, rdx, r10, r8, r9; the
2364    /// instruction clobbers rcx and r11.
2365    #[inline(always)]
2366    pub(super) unsafe fn syscall6(number: usize, args: [usize; 6]) -> isize {
2367        let result: isize;
2368        unsafe {
2369            core::arch::asm!(
2370                "syscall",
2371                inlateout("rax") number => result,
2372                in("rdi") args[0], in("rsi") args[1], in("rdx") args[2],
2373                in("r10") args[3], in("r8") args[4], in("r9") args[5],
2374                lateout("rcx") _, lateout("r11") _,
2375                options(nostack),
2376            );
2377        }
2378        result
2379    }

2381    /// clone3, then, in the child, [`super::vfork_child`] on the child's own
2382    /// stack. Returns, in this thread only, the child's PID or the negative
2383    /// errno.
2384    ///
2385    /// Why the child never touches this thread's stack:
2386    /// - `args.stack`/`stack_size` name the child's stack, and the kernel
2387    ///   starts the child with rsp at its top (`stack + stack_size`, 16-byte
2388    ///   aligned). Every register but rax (0) and rsp is this thread's.
2389    /// - In the child, the instructions after `syscall` up to `call` read
2390    ///   only registers (rax, r12) and write only rbp and rdi. `call` pushes
2391    ///   its return address at the new rsp, on the child's stack, and
2392    ///   `vfork_child` and everything it calls run there and never return.
2393    /// - This block is `nostack`, so nothing of this thread lives below its
2394    ///   rsp that the block could disturb, and the child's rsp never points
2395    ///   into this thread's stack. This thread waits in the kernel until the
2396    ///   child execs or exits (CLONE_VFORK), then resumes at the `test` with
2397    ///   its own rsp, rax = PID or -errno.
2398    pub(super) unsafe fn clone3_then_child(
2399        args: *const CloneArgs,
2400        state: *const ChildState,
2401    ) -> i64 {
2402        let result: i64;
2403        unsafe {
2404            core::arch::asm!(
2405                "syscall",
2406                "test rax, rax",
2407                "jnz 2f",
2408                "xor ebp, ebp",
2409                "mov rdi, r12",
2410                "call {child}",
2411                "ud2",
2412                "2:",
2413                child = sym super::vfork_child,
2414                inlateout("rax") libc::SYS_clone3 => result,
2415                inlateout("rdi") args => _,
2416                in("rsi") core::mem::size_of::<CloneArgs>(),
2417                in("r12") state,
2418                lateout("rcx") _,
2419                lateout("r11") _,
2420                options(nostack),
2421            );
2422        }
2423        result
2424    }

2456    /// - `args.stack`/`stack_size` name the child's stack, and the kernel
2457    ///   starts the child with sp at its top (`stack + stack_size`, 16-byte
2458    ///   aligned). Every register but x0 (0) and sp is this thread's.
2459    /// - In the child, the instructions after `svc` up to `bl` read only
2460    ///   registers (x0, x20) and write only x29, x30 and x0. `bl` stores its
2461    ///   return address in x30, not in memory, and `vfork_child` and
2462    ///   everything it calls run on the child's stack and never return.

2475                "svc #0",
2476                "cbnz x0, 2f",

2479                "mov x0, x20",
2480                "bl {child}",
2481                "brk #1",
2482                "2:",
2483                child = sym super::vfork_child,
2484                in("x8") libc::SYS_clone3,
2485                inlateout("x0") args => result,
2486                in("x1") core::mem::size_of::<CloneArgs>(),
