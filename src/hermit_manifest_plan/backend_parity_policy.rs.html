1//! Exact command bytes for the two backend-parity-c manifest selectors.
2//!
3//! Those selectors no longer exist in a generated plan: slice S6 of
4//! <https://github.com/rrnewton/hermit/issues/3301> folded the backend-parity-c
5//! bucket into c-programs and removed `e2e.manifest_backend_parity_c` and its
6//! `_on_host` twin. Their tags, commands and category below are the spellings
7//! of retained plans, which a reader must still match byte for byte.
8//!
9//! The generator emits only the ordinary spellings: since
10//! <https://github.com/rrnewton/hermit/issues/3301> no newly constructed plan
11//! asks the harness for a ptrace reference run, and the generator's invariants
12//! refuse a step that does. The parity spellings stay here only so that
13//! schema-10 plans retained before that change keep reading with the cells and
14//! parity relations they were published with.
15//!
16//! The generator and retained-plan reader share exact command bytes; a reader
17//! never infers execution policy from a substring of an arbitrary shell command.
18

26/// Pinned-root names forwarded since the Buck release build joined validate.
27pub(crate) const RELEASE_BUILD_ENV: &str =
28    " --env HERMIT_VALIDATE_RELEASE_BUILD_MODE --env HERMIT_VALIDATE_BUCK_DOTSLASH";
29
30/// Pinned-root names forwarded since the parity post-pass runs in every
31/// harness process (<https://github.com/rrnewton/hermit/issues/3301>): the one
32/// guest epoch pinned per run, and the post-pass's two activation variables.
33/// They were added together after the release-build names, so every older
34/// spelling lacks all three.
35pub(crate) const PARITY_POST_PASS_ENV: [&str; 3] = [
36    "HERMIT_EPOCH",
37    crate::parity::PARITY_POST_PASS_ENV,
38    crate::parity::PARITY_SELECT_ENV,
39];
40
41/// `command` without the [`PARITY_POST_PASS_ENV`] names: the spelling of a
42/// plan retained before they were forwarded. Each name must appear exactly
43/// once; the hosted command, which forwards nothing, must have none.
44pub(crate) fn without_parity_post_pass_env(tag: &str, command: &str) -> Result<String, String> {
45    let words = PARITY_POST_PASS_ENV.map(|name| format!(" --env {name} "));
46    if tag == "e2e.manifest_backend_parity_c_on_host"
47        && words.iter().all(|word| !command.contains(word.as_str()))
48    {
49        return Ok(command.to_owned());
50    }
51    let mut older = command.to_owned();
52    for (name, word) in PARITY_POST_PASS_ENV.iter().zip(&words) {
53        if older.matches(word.as_str()).count() != 1 {
54            return Err(format!(
55                "{tag} has no single {name} environment name to date"
56            ));
57        }
58        older = older.replacen(word.as_str(), " ", 1);
59    }
60    Ok(older)
61}

63pub(crate) fn selects_ptrace_parity(step: &Step) -> Result<bool, String> {
64    let expected = match step.tag().as_str() {
65        "e2e.manifest_backend_parity_c" => {
66            // Generate trusted expected wrapper bytes with the same policy as
67            // the DAG producer. Never normalize the received command: doing so
68            // would silently restore a removed environment or argv guard.
69            let expected = |command| {
70                crate::validation_dag::refresh_pinned_root_environment(&step.tag(), command)
71            };
72            Some((
73                expected(PORTABLE_PARITY_COMMAND)?,
74                expected(PORTABLE_ORDINARY_COMMAND)?,
75            ))
76        }
77        "e2e.manifest_backend_parity_c_on_host" => Some((
78            HOSTED_PARITY_COMMAND.to_owned(),
79            HOSTED_ORDINARY_COMMAND.to_owned(),
80        )),
81        _ => None,
82    };
83    let Some((parity, ordinary)) = expected else {
84        if step.cmd.contains("--parity-reference") {
85            return Err(format!(
86                "{} has unrecognized backend parity policy",
87                step.tag()
88            ));
89        }
90        return Ok(false);
91    };
92    // Plans retained before the release-build names were forwarded omit
93    // exactly that contiguous run; the refresher appends missing names in
94    // declaration order, so this is the exact older spelling, not a relaxation.
95    let pre_release_env = |command: String| match command.matches(RELEASE_BUILD_ENV).count() {
96        0 if step.tag() == "e2e.manifest_backend_parity_c_on_host" => Ok(command),
97        1 => Ok(command.replacen(RELEASE_BUILD_ENV, "", 1)),
98        _ => Err(format!(
99            "{} has no single release-build environment run to date",
100            step.tag()
101        )),
102    };
103    // Plans retained before the parity post-pass names were forwarded omit
104    // exactly those names from the current spelling, for the same reason.
105    let pre_epoch_parity = without_parity_post_pass_env(&step.tag(), &parity)?;
106    let pre_epoch_ordinary = without_parity_post_pass_env(&step.tag(), &ordinary)?;
107    let previous_parity = pre_release_env(pre_epoch_parity.clone())?;
108    let previous_ordinary = pre_release_env(pre_epoch_ordinary.clone())?;
109    let legacy_parity = previous_parity.replacen(" --results ", " --jobs 8 --results ", 1);
110    let legacy_ordinary = previous_ordinary.replacen(" --results ", " --jobs 8 --results ", 1);
111    let selects_parity = match step.cmd.as_str() {
112        command if command == parity => true,
113        command if command == ordinary => false,
114        command if command == pre_epoch_parity => true,
115        command if command == pre_epoch_ordinary => false,
116        command if command == previous_parity => true,
117        command if command == previous_ordinary => false,
118        // Schema-10 artifacts written before scheduler-owned width retain the
119        // literal worker count. They remain readable as evidence, but an
120        // unrecognized command still cannot pass this exact-command check.
121        command if command == legacy_parity => true,
122        command if command == legacy_ordinary => false,
123        _ => {
124            return Err(format!(
125                "{} differs from its declared backend parity command",
126                step.tag()
127            ));
128        }
129    };
130    let selector = step
131        .manifest
132        .as_ref()
133        .ok_or_else(|| format!("{} omitted its backend parity selector", step.tag()))?;
134    if selector.lane != "portable"
135        || selector.category != "backend-parity-c"
136        || selector.test.is_some()
137        || selector.mode.is_some()
138        || selector.backend.is_some()
139    {
140        return Err(format!(
141            "{} changed its backend parity population",
142            step.tag()
143        ));
144    }
145    Ok(selects_parity)
146}
