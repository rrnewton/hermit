13pub(crate) fn selects_ptrace_parity(step: &Step) -> Result<bool, String> {
14    let expected = match step.tag().as_str() {
15        "e2e.manifest_backend_parity_c" => {
16            // Generate trusted expected wrapper bytes with the same policy as
17            // the DAG producer. Never normalize the received command: doing so
18            // would silently restore a removed environment or argv guard.
19            let expected = |command| {
20                crate::validation_dag::refresh_pinned_root_environment(&step.tag(), command)
21            };
22            Some((
23                expected(PORTABLE_PARITY_COMMAND)?,
24                expected(PORTABLE_ORDINARY_COMMAND)?,
25            ))
26        }
27        "e2e.manifest_backend_parity_c_on_host" => Some((
28            HOSTED_PARITY_COMMAND.to_owned(),
29            HOSTED_ORDINARY_COMMAND.to_owned(),
30        )),
31        _ => None,
32    };
33    let Some((parity, ordinary)) = expected else {
34        if step.cmd.contains("--parity-reference") {
35            return Err(format!(
36                "{} has unrecognized backend parity policy",
37                step.tag()
38            ));
39        }
40        return Ok(false);
41    };
42    if step.cmd != parity && step.cmd != ordinary {
43        return Err(format!(
44            "{} differs from its declared backend parity command",
45            step.tag()
46        ));
47    }
48    let selector = step
49        .manifest
50        .as_ref()
51        .ok_or_else(|| format!("{} omitted its backend parity selector", step.tag()))?;
52    if selector.lane != "portable"
53        || selector.category != "backend-parity-c"
54        || selector.test.is_some()
55        || selector.mode.is_some()
56        || selector.backend.is_some()
57    {
58        return Err(format!(
59            "{} changed its backend parity population",
60            step.tag()
61        ));
62    }
63    Ok(step.cmd == parity)
64}
