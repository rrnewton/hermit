1//! Change-set selection for the tool self-tests (`selftest.<name>` nodes) that
2//! run only when a change touches their trigger paths.
3//!
4//! A self-test with trigger paths ([`crate::validation_dag::ToolSelfTest`]'s
5//! `run_when_changed`) runs when any path changed since the merge base with
6//! `origin/main`, committed or not, lies under one of them ([`is_under`]). A self-test that
7//! does not run prints one line containing [`NOT_RUN_MARKER`], so the node log
8//! and the scheduler's one-line node summary both say it was not run.
9//!
10//! Selection errs toward running. Main always runs, and so does any change set
11//! this module cannot resolve: a shallow clone, a missing `origin/main`, an
12//! empty change set, or any git failure.
13
14use std::collections::BTreeSet;
15use std::path::Path;

17use crate::git_environment::git_command;

19/// Set to `all` to run every self-test whatever the change set is.
20pub const SELECTION_ENV: &str = "HERMIT_SELFTEST_SELECTION";
21
22/// The text that marks a self-test the selection did not run.
23pub const NOT_RUN_MARKER: &str = " NOT RUN by file selection: ";
24
25/// Whether a self-test runs, and why.
26#[derive(Debug, PartialEq, Eq)]
27pub enum Decision {
28    Run(String),
29    Skip { base: String, changed: Vec<String> },
30}
31
32/// The one line printed for a self-test that is not run.
33pub fn not_run_line(name: &str, base: &str, changed: &[String], triggers: &[&str]) -> String {
34    const SHOWN: usize = 5;
35    let mut shown = changed
36        .iter()
37        .take(SHOWN)
38        .cloned()
39        .collect::<Vec<_>>()
40        .join(", ");
41    if changed.len() > SHOWN {
42        shown.push_str(&format!(", and {} more", changed.len() - SHOWN));
43    }
44    format!(
45        "self-test {name}{NOT_RUN_MARKER}none of the {} path(s) changed since merge base {base} \
46         ({shown}) is under {}; main runs it, and {SELECTION_ENV}=all runs it here",
47        changed.len(),
48        triggers.join(" or ")
49    )
50}
51
52/// Whether `path` is under `trigger`: inside it when the trigger is a
53/// directory (it ends in `/`), otherwise the trigger itself or, for a
54/// submodule, a path inside it. `agent-utils` is not a prefix of
55/// `agent-utils-notes.md`.
56pub fn is_under(path: &str, trigger: &str) -> bool {
57    if trigger.ends_with('/') {
58        return path.starts_with(trigger);
59    }
60    path == trigger
61        || path
62            .strip_prefix(trigger)
63            .is_some_and(|rest| rest.starts_with('/'))
64}

66/// Decide from the resolved change set. Pure, so every rule is testable
67/// without a repository.
68pub fn decide(
69    selection_env: Option<&str>,
70    change_set: Result<(String, Vec<String>), String>,
71    triggers: &[&str],
72) -> Decision {
73    match selection_env {
74        None | Some("") => {}
75        Some("all") => return Decision::Run(format!("{SELECTION_ENV}=all")),
76        Some(other) => {
77            return Decision::Run(format!(
78                "{SELECTION_ENV}={other:?} is not a known value, so it runs"
79            ));
80        }
81    }
82    let (base, changed) = match change_set {
83        Ok(change_set) => change_set,
84        Err(reason) => return Decision::Run(reason),
85    };
86    if changed.is_empty() {
87        return Decision::Run(format!(
88            "no path changed since merge base {base}, so there is nothing to select on"
89        ));
90    }
91    if let Some((path, trigger)) = changed.iter().find_map(|path| {
92        triggers
93            .iter()
94            .find(|trigger| is_under(path, trigger))
95            .map(|trigger| (path, trigger))
96    }) {
97        return Decision::Run(format!("changed path {path} is under {trigger}"));
98    }
99    Decision::Skip { base, changed }
100}

102/// The merge base with `origin/main` and every path changed since it,
103/// committed or not. `Err` names why the change set could not be resolved;
104/// the caller then runs the self-test.
105pub fn change_set(root: &Path) -> Result<(String, Vec<String>), String> {
106    let git = |args: &[&str]| -> Result<Vec<u8>, String> {
107        let output = git_command()
108            .arg("-C")
109            .arg(root)
110            .args(args)
111            .output()
112            .map_err(|error| format!("cannot run git {}: {error}", args.join(" ")))?;
113        if !output.status.success() {
114            return Err(format!(
115                "git {} failed: {}",
116                args.join(" "),
117                String::from_utf8_lossy(&output.stderr).trim()
118            ));
119        }
120        Ok(output.stdout)
121    };
122    let text = |bytes: Vec<u8>| String::from_utf8_lossy(&bytes).trim().to_string();
123    let mains = text(git(&[
124        "for-each-ref",
125        "--contains",
126        "HEAD",
127        "--format=%(refname)",
128        "refs/heads/main",
129        "refs/remotes/*/main",
130    ])?);
131    if !mains.is_empty() {
132        return Err(format!(
133            "HEAD is on main ({}), and main runs every self-test",
134            mains.split_whitespace().collect::<Vec<_>>().join(", ")
135        ));
136    }
137    if text(git(&["rev-parse", "--is-shallow-repository"])?) != "false" {
138        return Err("this clone is shallow, so its merge base cannot be trusted".into());
139    }
140    let base = text(git(&["merge-base", "HEAD", "refs/remotes/origin/main"])?);
141    if base.len() != 40 || !base.bytes().all(|b| b.is_ascii_hexdigit()) {
142        return Err(format!("git merge-base printed {base:?}, not a commit"));
143    }
144    let mut changed = BTreeSet::new();
145    for listing in [
146        // A rename lists both its old and its new path. A submodule counts
147        // as changed when its commit moves, whatever `diff.ignoreSubmodules`
148        // the clone sets.
149        git(&[
150            "diff",
151            "--name-only",
152            "--no-renames",
153            "--ignore-submodules=dirty",
154            "-z",
155            &base,
156            "--",
157        ])?,
158        git(&["ls-files", "--others", "--exclude-standard", "-z"])?,
159    ] {
160        for path in listing.split(|&b| b == 0).filter(|path| !path.is_empty()) {
161            changed.insert(
162                String::from_utf8(path.to_vec())
163                    .map_err(|_| "git listed a changed path that is not UTF-8".to_string())?,
164            );
165        }
166    }
167    Ok((base, changed.into_iter().collect()))
168}

206    fn a_change_under_a_trigger_runs() {
207        for path in [
208            "ci/compat-envelope/scorecard.rs",
209            "ci/compat-envelope/testdata/series-snapshot/README.md",

217        }

229                matches!(decide(None, change(&[path]), paths), Decision::Run(_)),

241    /// A trigger that names no tracked path never fires, so a misspelt or
242    /// moved input would silently stop selecting its self-test.
243    #[test]
244    fn every_trigger_names_a_tracked_path() {
245        let root = Path::new(env!("CARGO_MANIFEST_DIR")).join("../..");
246        let output = git_command()
247            .arg("-C")
248            .arg(&root)
249            .args(["ls-files", "-z"])
250            .output()
251            .unwrap();
252        assert!(output.status.success(), "git ls-files: {output:?}");
253        let tracked = output
254            .stdout
255            .split(|&b| b == 0)
256            .map(|path| String::from_utf8_lossy(path).into_owned())
257            .collect::<Vec<_>>();
258        let mut checked = 0;
259        for tool in crate::validation_dag::TOOL_SELF_TESTS {
260            for trigger in tool.run_when_changed.unwrap_or_default() {
261                assert!(
262                    tracked.iter().any(|path| is_under(path, trigger)),
263                    "self-test {}: trigger {trigger} names no tracked path",
264                    tool.name
265                );
266                checked += 1;
267            }
268        }
269        assert!(checked > 0, "no self-test has a trigger to check");
270    }

272    /// A scratch repository for `change_set`, removed on drop.
273    struct Scratch(std::path::PathBuf);
274
275    impl Scratch {
276        fn new(label: &str) -> Self {
277            let path = std::env::temp_dir().join(format!(
278                "hermit-self-test-selection-{label}-{}",
279                std::process::id()
280            ));
281            let _ = std::fs::remove_dir_all(&path);
282            std::fs::create_dir_all(&path).unwrap();
283            Self(path)
284        }
285
286        fn git(&self, args: &[&str]) -> String {
287            let output = git_command()
288                .env("GIT_CONFIG_GLOBAL", "/dev/null")
289                .env("GIT_CONFIG_NOSYSTEM", "1")
290                .args([
291                    "-c",
292                    "user.name=fixture",
293                    "-c",
294                    "user.email=fixture@example.invalid",
295                ])
296                .arg("-C")
297                .arg(&self.0)
298                .args(args)
299                .output()
300                .unwrap();
301            assert!(output.status.success(), "git {args:?}: {output:?}");
302            String::from_utf8(output.stdout).unwrap().trim().to_string()
303        }
304
305        fn commit(&self, path: &str) -> String {
306            let file = self.0.join(path);
307            std::fs::create_dir_all(file.parent().unwrap()).unwrap();
308            std::fs::write(&file, path).unwrap();
309            self.git(&["add", "--", path]);
310            self.git(&["commit", "-qm", path]);
311            self.git(&["rev-parse", "HEAD"])
312        }
313    }
314
315    impl Drop for Scratch {
316        fn drop(&mut self) {
317            let _ = std::fs::remove_dir_all(&self.0);
318        }
319    }

321    #[test]
322    fn change_set_reads_the_branch_diff_and_untracked_paths_and_runs_main() {
323        let repo = Scratch::new("branch");
324        repo.git(&["init", "-q", "-b", "main"]);
325        let base = repo.commit("README.md");
326        repo.git(&["update-ref", "refs/remotes/origin/main", &base]);
327        repo.git(&["switch", "-q", "-c", "topic"]);
328        repo.commit("docs/one.md");
329        std::fs::create_dir_all(repo.0.join("ci/compat-envelope")).unwrap();
330        std::fs::write(repo.0.join("ci/compat-envelope/new.rs"), "untracked").unwrap();
331        assert_eq!(
332            change_set(&repo.0),
333            Ok((
334                base.clone(),
335                vec![
336                    "ci/compat-envelope/new.rs".to_string(),
337                    "docs/one.md".to_string()
338                ]
339            ))
340        );
341        // Without the untracked file the branch touches no trigger.
342        std::fs::remove_file(repo.0.join("ci/compat-envelope/new.rs")).unwrap();
343        assert!(matches!(
344            decide(None, change_set(&repo.0), TRIGGERS),
345            Decision::Skip { base: skipped, .. } if skipped == base
346        ));
347
348        // A commit already contained in main runs, through the local branch
349        // or through any remote's main.
350        repo.git(&["switch", "-q", "main"]);
351        let on_main = change_set(&repo.0).unwrap_err();
352        assert!(
353            on_main.contains("HEAD is on main (refs/heads/main"),
354            "{on_main}"
355        );
356        repo.git(&["switch", "-q", "topic"]);
357        let topic = repo.git(&["rev-parse", "HEAD"]);
358        repo.git(&["branch", "-q", "-D", "main"]);
359        repo.git(&["update-ref", "refs/remotes/upstream/main", &topic]);
360        let remote_main = change_set(&repo.0).unwrap_err();
361        assert!(
362            remote_main.contains("HEAD is on main (refs/remotes/upstream/main)"),
363            "{remote_main}"
364        );
365        repo.git(&["update-ref", "-d", "refs/remotes/upstream/main"]);
366
367        // No origin/main to take a merge base from runs.
368        repo.git(&["update-ref", "-d", "refs/remotes/origin/main"]);
369        let no_base = change_set(&repo.0).unwrap_err();
370        assert!(no_base.contains("git merge-base"), "{no_base}");
371    }

373    #[test]
374    fn change_set_lists_deleted_renamed_and_submodule_paths() {
375        let repo = Scratch::new("moves");
376        repo.git(&["init", "-q", "-b", "main"]);
377        repo.commit("ci/compat-envelope/old.rs");
378        repo.commit("ci/compat-envelope/gone.rs");
379        // An unpopulated submodule, as a clone without `--recurse-submodules`
380        // leaves it: an empty directory.
381        std::fs::create_dir_all(repo.0.join("agent-utils")).unwrap();
382        let gitlink = |commit: &str| {
383            repo.git(&[
384                "update-index",
385                "--add",
386                "--cacheinfo",
387                &format!("160000,{commit},agent-utils"),
388            ]);
389        };
390        gitlink(&"1".repeat(40));
391        repo.git(&["commit", "-qm", "submodule"]);
392        let base = repo.git(&["rev-parse", "HEAD"]);
393        repo.git(&["update-ref", "refs/remotes/origin/main", &base]);
394        repo.git(&["switch", "-q", "-c", "topic"]);
395        repo.git(&["mv", "ci/compat-envelope/old.rs", "docs-old.rs"]);
396        repo.git(&["rm", "-q", "ci/compat-envelope/gone.rs"]);
397        repo.git(&["commit", "-qm", "move"]);
398        let (_, moved) = change_set(&repo.0).unwrap();
399        assert_eq!(
400            moved,
401            vec![
402                "ci/compat-envelope/gone.rs".to_string(),
403                "ci/compat-envelope/old.rs".to_string(),
404                "docs-old.rs".to_string(),
405            ]
406        );
407
408        // A moved submodule commit is listed even when the clone asks diff to
409        // ignore submodules.
410        repo.git(&["reset", "-q", "--hard", &base]);
411        repo.git(&["config", "diff.ignoreSubmodules", "all"]);
412        gitlink(&"2".repeat(40));
413        repo.git(&["commit", "-qm", "bump"]);
414        assert_eq!(
415            change_set(&repo.0),
416            Ok((base, vec!["agent-utils".to_string()]))
417        );
418    }
