1//! What a Buck e2e cell may take from its bundle instead of computing it again.
2//!
3//! A Buck validation runs every e2e cell as its own `test-harness run`
4//! process over one immutable bundle (ci/buck-e2e/defs.bzl): the same Hermit
5//! binary, harness binaries and source snapshot for each of the 2,248 cells of
6//! ci/expected-e2e-plan.json. Two of the harness's per-cell costs depend on
7//! the bundle alone:
8//!
9//! - the SHA-256 of the Hermit binary, taken before and after each cell (about
10//!   0.04 CPU-s each for a 62 MB release Hermit);
11//! - the canonical manifest planner's acceptance of the source snapshot, a
12//!   child process of about 0.09 CPU-s whose output the harness discards.
13//!
14//! The bundle rule computes both once into `BUNDLE_RECORD.json`
15//! (ci/buck-e2e/bundle-record), and cell.sh names that file in
16//! [`BUNDLE_RECORD_ENV`]. A cell still checks everything it checked before:
17//!
18//! - The Hermit binary must have the digest the record names, or the cell is an
19//!   infrastructure error. The digest comes from [`binary_digest`], which hashes
20//!   a file once per process and again only when its identity changes (device,
21//!   inode, size, modification and status-change times), and which may also
22//!   take it from a per-host cache keyed by that identity
23//!   ([`DIGEST_CACHE_DIR_ENV`]). Any write to or replacement of the binary
24//!   changes its status-change time or inode, so a swapped binary is hashed
25//!   afresh and refused, and a binary that changes during a cell still makes
26//!   that cell an infrastructure error.
27//! - The planner's verdict is taken from the record only when the record says
28//!   it accepted this very source SHA with this very planner binary (by
29//!   digest); in every other case the harness runs the planner itself, as
30//!   before.
31

47/// The bundle record a cell reads (set by ci/buck-e2e/cell.sh).
48pub const BUNDLE_RECORD_ENV: &str = "HERMIT_E2E_BUNDLE_RECORD";
49/// A host-local directory of verified file digests (set by cell.sh for cells
50/// that run on the host).
51pub const DIGEST_CACHE_DIR_ENV: &str = "HERMIT_E2E_DIGEST_CACHE_DIR";
52pub const BUNDLE_RECORD_SCHEMA: &str = "hermit-e2e-bundle-record/v1";
53
54/// A file younger than this is hashed every time and never remembered. File
55/// times advance in clock ticks, so a write within the tick of the identity
56/// read could leave the identity unchanged (Git's "racy clean" index entries);
57/// a bundle's binaries are minutes old by the time any cell runs.
58const SETTLED: Duration = Duration::from_secs(2);
59
60#[derive(Clone, Debug, Deserialize, PartialEq, Eq)]
61#[serde(deny_unknown_fields)]
62pub struct BundleRecord {
63    pub schema: String,
64    /// The commit the bundle's `src/` was archived from (its SOURCE_SHA).
65    pub source_sha: String,
66    /// The SHA-256 of the bundle's `hermit/hermit`.
67    pub hermit_sha256: String,
68    /// The SHA-256 of the bundle's `bin/hermit-manifest-plan`.
69    pub planner_sha256: String,
70    /// What that planner said about `src/`: `accepted` or `rejected`.
71    pub manifest_plan: String,
72}
73
74impl BundleRecord {
75    pub fn parse(bytes: &[u8]) -> Result<Self, String> {
76        let record = serde_json::from_slice::<Self>(bytes).map_err(|error| error.to_string())?;
77        if record.schema != BUNDLE_RECORD_SCHEMA {
78            return Err(format!(
79                "schema is {:?}, not {BUNDLE_RECORD_SCHEMA:?}",
80                record.schema
81            ));
82        }
83        let hex40 = |value: &str| value.len() == 40 && is_lower_hex(value);
84        let hex64 = |value: &str| value.len() == 64 && is_lower_hex(value);
85        if !hex40(&record.source_sha) {
86            return Err(format!(
87                "source_sha {:?} is not a full commit",
88                record.source_sha
89            ));
90        }
91        for (name, value) in [
92            ("hermit_sha256", &record.hermit_sha256),
93            ("planner_sha256", &record.planner_sha256),
94        ] {
95            if !hex64(value) {
96                return Err(format!("{name} {value:?} is not a SHA-256"));
97            }
98        }
99        if !matches!(record.manifest_plan.as_str(), "accepted" | "rejected") {
100            return Err(format!(
101                "manifest_plan is {:?}, not accepted or rejected",
102                record.manifest_plan
103            ));
104        }
105        Ok(record)
106    }
107
108    pub fn load(path: &Path) -> Result<Self, String> {
109        let bytes = fs::read(path).map_err(|error| {
110            format!("cannot read the bundle record {}: {error}", path.display())
111        })?;
112        Self::parse(&bytes)
113            .map_err(|error| format!("bundle record {} is invalid: {error}", path.display()))
114    }
115}
116
117fn is_lower_hex(value: &str) -> bool {
118    value
119        .bytes()
120        .all(|b| matches!(b, b'0'..=b'9' | b'a'..=b'f'))
121}
122
123/// The record [`BUNDLE_RECORD_ENV`] names, read once per process: `Ok(None)`
124/// when the variable is unset, an error when it names an unreadable or invalid
125/// record.
126pub fn record_from_env() -> Result<Option<(PathBuf, BundleRecord)>, String> {
127    static RECORD: OnceLock<Result<Option<(PathBuf, BundleRecord)>, String>> = OnceLock::new();
128    RECORD
129        .get_or_init(|| {
130            let Some(path) = std::env::var_os(BUNDLE_RECORD_ENV).filter(|v| !v.is_empty()) else {
131                return Ok(None);
132            };
133            let path = PathBuf::from(path);
134            BundleRecord::load(&path).map(|record| Some((path, record)))
135        })
136        .clone()
137}

139/// The Hermit digest the bundle records, if a record is in force.
140pub fn recorded_hermit_sha256() -> Result<Option<String>, String> {
141    Ok(record_from_env()?.map(|(_, record)| record.hermit_sha256))
142}

144/// Whether the planner's acceptance of `root` may be taken from `record` at
145/// `record_path` instead of running `planner`: only when the record sits
146/// beside `root` in one bundle, names `source_sha`, says `accepted`, and
147/// `planner` has the digest the record names. `Err` says why not, for the
148/// caller to print before it runs the planner itself.
149pub fn manifest_plan_accepted(
150    record_path: &Path,
151    record: &BundleRecord,
152    root: &Path,
153    planner: &Path,
154    source_sha: &str,
155) -> Result<(), String> {
156    let bundle_of = |path: &Path| fs::canonicalize(path).ok()?.parent().map(Path::to_path_buf);
157    if bundle_of(record_path).is_none() || bundle_of(record_path) != bundle_of(root) {
158        return Err(format!(
159            "the record {} is not in the bundle of {}",
160            record_path.display(),
161            root.display()
162        ));
163    }
164    if record.source_sha != source_sha {
165        return Err(format!(
166            "the record is for source {}, not {source_sha}",
167            record.source_sha
168        ));
169    }
170    if record.manifest_plan != "accepted" {
171        return Err(format!("the record says {}", record.manifest_plan));
172    }
173    match binary_digest(planner) {
174        Some(digest) if digest == record.planner_sha256 => Ok(()),
175        Some(digest) => Err(format!(
176            "{} has sha256 {digest}, but the record names {}",
177            planner.display(),
178            record.planner_sha256
179        )),
180        None => Err(format!("cannot read {}", planner.display())),
181    }
182}

184/// Everything about a file that a write or a replacement changes.
185#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
186struct FileIdentity {
187    dev: u64,
188    ino: u64,
189    size: u64,
190    mtime: (i64, i64),
191    ctime: (i64, i64),
192}
193
194impl FileIdentity {
195    fn of(path: &Path) -> Option<Self> {
196        let meta = fs::metadata(path).ok()?;
197        meta.is_file().then(|| Self {
198            dev: meta.dev(),
199            ino: meta.ino(),
200            size: meta.size(),
201            mtime: (meta.mtime(), meta.mtime_nsec()),
202            ctime: (meta.ctime(), meta.ctime_nsec()),
203        })
204    }
205
206    /// Whether both file times are at least [`SETTLED`] before `now`.
207    fn settled(&self, now: SystemTime) -> bool {
208        let Ok(now) = now.duration_since(UNIX_EPOCH) else {
209            return false;
210        };
211        [self.mtime, self.ctime].iter().all(|&(secs, nsecs)| {
212            u64::try_from(secs).is_ok_and(|secs| {
213                let at = Duration::new(secs, u32::try_from(nsecs).unwrap_or(0));
214                now.checked_sub(at).is_some_and(|age| age >= SETTLED)
215            })
216        })
217    }
218
219    fn cache_name(&self) -> String {
220        let key = format!(
221            "{}:{}:{}:{}.{}:{}.{}",
222            self.dev, self.ino, self.size, self.mtime.0, self.mtime.1, self.ctime.0, self.ctime.1
223        );
224        format!("{:x}", Sha256::digest(key.as_bytes()))
225    }
226}

228/// The SHA-256 of the file at `path`, or `None` when it cannot be read.
229/// It streams the file rather than reading it whole: a debug Hermit is about
230/// 450 MB.
231pub fn file_digest(path: &Path) -> Option<String> {
232    let mut digest = Sha256::new();
233    std::io::copy(&mut fs::File::open(path).ok()?, &mut digest).ok()?;
234    Some(format!("{:x}", digest.finalize()))
235}
236
237/// [`file_digest`] of `path`, hashed once per process for as long as the
238/// file's identity is unchanged, and taken from [`DIGEST_CACHE_DIR_ENV`]'s
239/// per-host cache when that names a directory.
240pub fn binary_digest(path: &Path) -> Option<String> {
241    let cache = std::env::var_os(DIGEST_CACHE_DIR_ENV)
242        .filter(|v| !v.is_empty())
243        .map(PathBuf::from);
244    binary_digest_with(path, cache.as_deref())
245}
246
247type Memo = Mutex<HashMap<PathBuf, (FileIdentity, String)>>;
248
249fn binary_digest_with(path: &Path, cache: Option<&Path>) -> Option<String> {
250    static MEMO: OnceLock<Memo> = OnceLock::new();
251    binary_digest_in(MEMO.get_or_init(Default::default), path, cache)
252}
253
254fn binary_digest_in(memo: &Memo, path: &Path, cache: Option<&Path>) -> Option<String> {
255    let Some(before) = FileIdentity::of(path) else {
256        return file_digest(path);
257    };
258    if let Some((identity, digest)) = memo.lock().ok()?.get(path) {
259        if *identity == before {
260            return Some(digest.clone());
261        }
262    }
263    let settled = before.settled(SystemTime::now());
264    let remember = |digest: &str| {
265        if let Ok(mut memo) = memo.lock() {
266            memo.insert(path.to_path_buf(), (before, digest.to_string()));
267        }
268    };
269    if let (true, Some(cache)) = (settled, cache) {
270        if let Ok(text) = fs::read_to_string(cache.join(before.cache_name())) {
271            let digest = text.trim_end_matches('\n');
272            if digest.len() == 64 && is_lower_hex(digest) {
273                remember(digest);
274                return Some(digest.to_string());
275            }
276        }
277    }
278    let digest = file_digest(path)?;
279    if settled && FileIdentity::of(path) == Some(before) {
280        remember(&digest);
281        if let Some(cache) = cache {
282            write_cache_entry(cache, &before.cache_name(), &digest);
283        }
284    }
285    Some(digest)
286}
287
288/// Best effort: a cell that cannot write the cache hashes, as it would anyway.
289fn write_cache_entry(cache: &Path, name: &str, digest: &str) {
290    if fs::create_dir_all(cache).is_err() {
291        return;
292    }
293    let partial = cache.join(format!(".{name}.{}", std::process::id()));
294    if fs::write(&partial, format!("{digest}\n")).is_ok()
295        && fs::rename(&partial, cache.join(name)).is_ok()
296    {
297        return;
298    }
299    let _ = fs::remove_file(&partial);
300}

306    fn scratch(name: &str) -> PathBuf {
307        let dir = std::env::temp_dir().join(format!(
308            "bundle-record-{name}-{}-{:?}",
309            std::process::id(),
310            std::thread::current().id()
311        ));
312        let _ = fs::remove_dir_all(&dir);
313        fs::create_dir_all(&dir).unwrap();
314        dir

317    /// A clock [`SETTLED`] ahead: ctime cannot be set back, so a test file is
318    /// settled only against a later clock.
319    fn later() -> SystemTime {
320        SystemTime::now() + SETTLED + Duration::from_secs(1)
321    }

323    fn record_json(manifest_plan: &str) -> String {
324        format!(
325            r#"{{"schema":"{BUNDLE_RECORD_SCHEMA}","source_sha":"{}","hermit_sha256":"{}","planner_sha256":"{}","manifest_plan":"{manifest_plan}"}}"#,
326            "a".repeat(40),
327            "b".repeat(64),
328            "c".repeat(64)
329        )
330    }

332    #[test]
333    fn a_record_is_parsed_only_in_its_exact_shape() {
334        let record = BundleRecord::parse(record_json("accepted").as_bytes()).unwrap();
335        assert_eq!(record.manifest_plan, "accepted");
336        assert!(BundleRecord::parse(record_json("maybe").as_bytes()).is_err());
337        let extra = record_json("accepted").replace('}', r#","more":1}"#);
338        assert!(BundleRecord::parse(extra.as_bytes()).is_err());
339        let short = record_json("accepted").replace(&"b".repeat(64), &"b".repeat(63));
340        assert!(BundleRecord::parse(short.as_bytes()).is_err());
341        let upper = record_json("accepted").replace(&"a".repeat(40), &"A".repeat(40));
342        assert!(BundleRecord::parse(upper.as_bytes()).is_err());
343        let old = record_json("accepted").replace("/v1", "/v0");
344        assert!(BundleRecord::parse(old.as_bytes()).is_err());
345    }

347    #[test]
348    fn a_digest_is_remembered_only_while_the_file_is_unchanged() {
349        let dir = scratch("memo");
350        let file = dir.join("hermit");
351        fs::write(&file, b"one").unwrap();
352        let memo = Memo::default();
353        // A file written just now is never remembered.
354        assert_eq!(
355            binary_digest_in(&memo, &file, None),
356            Some(format!("{:x}", Sha256::digest(b"one")))
357        );
358        assert!(memo.lock().unwrap().is_empty());
359        // A remembered digest is returned while the identity matches...
360        let identity = FileIdentity::of(&file).unwrap();
361        memo.lock()
362            .unwrap()
363            .insert(file.clone(), (identity, "f".repeat(64)));
364        assert_eq!(binary_digest_in(&memo, &file, None), Some("f".repeat(64)));
365        // ...and not after a write, which hashes the new content. (A new size:
366        // two writes within one clock tick may share a ctime on a kernel
367        // without fine-grained timestamps. A remembered digest is only ever
368        // of a settled file, whose next write is a tick later.)
369        fs::write(&file, b"second").unwrap();
370        assert_eq!(
371            binary_digest_in(&memo, &file, None),
372            Some(format!("{:x}", Sha256::digest(b"second")))
373        );
374        // A replacement (a new inode) is hashed too.
375        memo.lock().unwrap().insert(
376            file.clone(),
377            (FileIdentity::of(&file).unwrap(), "f".repeat(64)),
378        );
379        let other = dir.join("other");
380        fs::write(&other, b"three").unwrap();
381        fs::rename(&other, &file).unwrap();
382        assert_eq!(
383            binary_digest_in(&memo, &file, None),
384            Some(format!("{:x}", Sha256::digest(b"three")))
385        );
386        assert_eq!(binary_digest_in(&memo, &dir.join("missing"), None), None);
387        fs::remove_dir_all(&dir).unwrap();
388    }

390    #[test]
391    fn a_cache_entry_is_keyed_by_identity_and_read_only_for_a_settled_file() {
392        let dir = scratch("cache");
393        let file = dir.join("hermit");
394        fs::write(&file, b"one").unwrap();
395        let identity = FileIdentity::of(&file).unwrap();
396        assert!(!identity.settled(SystemTime::now()));
397        assert!(identity.settled(later()));
398        let cache = dir.join("cache");
399        write_cache_entry(&cache, &identity.cache_name(), &"e".repeat(64));
400        assert_eq!(
401            fs::read_to_string(cache.join(identity.cache_name())).unwrap(),
402            format!("{}\n", "e".repeat(64))
403        );
404        // The file is not settled, so the entry is ignored and the file hashed.
405        assert_eq!(
406            binary_digest_in(&Memo::default(), &file, Some(&cache)),
407            Some(format!("{:x}", Sha256::digest(b"one")))
408        );
409        // A write changes the identity, and so the entry's name.
410        fs::write(&file, b"second").unwrap();
411        assert_ne!(
412            FileIdentity::of(&file).unwrap().cache_name(),
413            identity.cache_name()
414        );
415        // A cache that cannot be written is no failure.
416        let blocked = dir.join("blocked");
417        fs::write(&blocked, b"").unwrap();
418        write_cache_entry(&blocked, "x", &"e".repeat(64));
419        fs::remove_dir_all(&dir).unwrap();
420    }
