4#[derive(Clone, Debug, Deserialize, Serialize)]
5pub struct VerificationReport {
6    pub verified: bool,
7    pub bitwise_parity: bool,
8    pub verdict: String,
9    /// `None` when the producer reached no verdict. This is a DOCUMENTED
10    /// producer state, not a malformed report: `VerificationReport::no_result()`
11    /// in hermit-cli sets it, and its doc says "`null` when no verdict was
12    /// reached". Declaring it non-optional made a legitimate no-result
13    /// deserialise as "unreadable report", which reported an infrastructure
14    /// fault where the truth was that nothing had been recorded.
15    #[serde(deserialize_with = "present_but_nullable_comparison")]
16    pub comparison: Option<ComparisonReport>,
17    pub compared_log_messages: Option<ComparedLogMessages>,
18}
19
20#[derive(Clone, Debug, Deserialize, Serialize)]
21pub struct ComparisonReport {
22    pub strictness: String,
23    pub compare_logs: bool,
24}
25
26#[derive(Clone, Debug, Deserialize, Serialize)]
27pub struct ComparedLogMessages {
28    pub left: u64,
29    pub right: u64,
30}
31
32/// Accept `null` but not a MISSING field. serde maps an absent `Option` field to
33/// `None` by default, which would let a report that never mentioned a comparison
34/// read as a legitimate no-result. Naming a `deserialize_with` makes the field
35/// required again while still admitting an explicit null.
36fn present_but_nullable_comparison<'de, D>(
37    deserializer: D,
38) -> Result<Option<ComparisonReport>, D::Error>
39where
40    D: serde::Deserializer<'de>,
41{
42    Option::<ComparisonReport>::deserialize(deserializer)
43}
44
45impl VerificationReport {
46    /// `null` is legal ONLY for a report that reached no verdict. The producer
47    /// documents it that way ("`null` when no verdict was reached"), and the
48    /// sibling consumer in ci/compat-envelope/pressure-test.rs already applies
49    /// exactly this rule. A null comparison beside a "matched" verdict is a
50    /// self-contradictory report and stays refused at parse.
51    fn require_null_comparison_matches_verdict(self) -> Result<Self, String> {
52        if self.comparison.is_none() && self.verdict != "no_result" {
53            return Err(format!(
54                "incomplete verification report: comparison is null but verdict is {}; null is legal only for no_result",
55                self.verdict
56            ));
57        }
58        Ok(self)
59    }
60
61    /// Parse the complete typed receipt. A top-level JSON object is not enough:
62    /// every nested field used by admission must deserialize successfully.
63    #[allow(dead_code)] // this file is path-included by consumers that use one parse form
64    pub fn from_json_slice(bytes: &[u8]) -> Result<Self, String> {
65        serde_json::from_slice::<Self>(bytes)
66            .map_err(|error| format!("incomplete verification report: {error}"))?
67            .require_null_comparison_matches_verdict()
68    }
69
70    #[allow(dead_code)] // this file is path-included by consumers that use one parse form
71    pub fn from_json_value(value: serde_json::Value) -> Result<Self, String> {
72        serde_json::from_value::<Self>(value)
73            .map_err(|error| format!("incomplete verification report: {error}"))?
74            .require_null_comparison_matches_verdict()
75    }
76
77    /// Prove that the invocation actually compared the canonical INFO evidence.
78    /// This is separate from whether that comparison matched: a canonical
79    /// divergence is a product result, while a stripped/output-only/empty-log
80    /// comparison is incomplete evidence and therefore an infrastructure result.
81    pub fn require_canonical_comparison(&self) -> Result<(), String> {
82        let counts = self.compared_log_messages.as_ref();
83        // No comparison at all is a distinct outcome from a comparison that was
84        // too weak, and it is still not admissible. Naming it separately is the
85        // whole point: "no verdict was recorded" is actionable against the
86        // producer, while the old "unreadable report" pointed at the reader.
87        let Some(comparison) = self.comparison.as_ref() else {
88            return Err(format!(
89                "verification recorded no comparison at all (verdict={}), so there is no canonical INFO evidence to admit",
90                self.verdict
91            ));
92        };
93        if comparison.strictness == "canonical"
94            && comparison.compare_logs
95            && counts.is_some_and(|counts| counts.left > 0 && counts.right > 0)
96        {
97            Ok(())
98        } else {
99            Err(format!(
100                "verification did not compare canonical non-vacuous INFO evidence: strictness={} compare_logs={} messages={}/{}",
101                comparison.strictness,
102                comparison.compare_logs,
103                counts.map_or(0, |counts| counts.left),
104                counts.map_or(0, |counts| counts.right),
105            ))
106        }
107    }
108
109    /// Admit a green only when both the canonical comparison and the match
110    /// claim agree. `verified=true` alone is intentionally insufficient: KVM's
111    /// output-only fallback can report it with zero compared INFO messages.
112    pub fn require_canonical_match(&self) -> Result<(), String> {
113        self.require_canonical_comparison()?;
114        if self.verified && self.verdict == "matched" && self.bitwise_parity {
115            Ok(())
116        } else {
117            Err(format!(
118                "canonical verification did not match: verified={} verdict={} bitwise_parity={}",
119                self.verified, self.verdict, self.bitwise_parity
120            ))
121        }
122    }
123}

129    fn report(strictness: &str, compare_logs: bool, left: u64, right: u64) -> VerificationReport {
130        VerificationReport {
131            verified: true,
132            bitwise_parity: true,
133            verdict: "matched".into(),
134            comparison: Some(ComparisonReport {
135                strictness: strictness.into(),
136                compare_logs,
137            }),
138            compared_log_messages: Some(ComparedLogMessages { left, right }),
139        }
140    }
141
142    #[test]
143    fn brackets_every_canonical_match_requirement() {
144        assert!(
145            report("canonical", true, 1, 1)
146                .require_canonical_match()
147                .is_ok()
148        );
149        let mut weak = report("canonical", true, 1, 1);
150        weak.verified = false;
151        assert!(weak.require_canonical_match().is_err());
152        let mut weak = report("canonical", true, 1, 1);
153        weak.verdict = "diverged".into();
154        assert!(weak.require_canonical_match().is_err());
155        let mut weak = report("canonical", true, 1, 1);
156        weak.bitwise_parity = false;
157        assert!(weak.require_canonical_match().is_err());
158        assert!(
159            report("stripped", true, 1, 1)
160                .require_canonical_match()
161                .is_err()
162        );
163        assert!(
164            report("canonical", false, 1, 1)
165                .require_canonical_match()
166                .is_err()
167        );
168        assert!(
169            report("canonical", true, 0, 1)
170                .require_canonical_match()
171                .is_err()
172        );
173    }

175    /// The exact bytes hermit writes for a run that reached no verdict, copied
176    /// from a real capture rather than reconstructed. Before this was `Option`,
177    /// parsing these failed with "invalid type: null, expected struct
178    /// ComparisonReport at line 1 column 80" and the cell was scored ERROR
179    /// "verification report is unreadable" -- an infrastructure fault where the
180    /// truth was that the producer had recorded nothing.
181    const REAL_NO_RESULT_REPORT: &str = concat!(
182        r#"{"verified":false,"bitwise_parity":false,"verdict":"no_result","#,
183        r#""comparison":null,"compared_log_messages":null,"guest_exit_code":null,"#,
184        r#""guest_signal":null,"first_divergent_scheduler_turn":null,"#,
185        r#""first_divergent_virtual_nanoseconds":null}"#,
186    );
187
188    #[test]
189    fn a_real_no_result_report_parses_instead_of_reading_as_unreadable() {
190        let parsed = VerificationReport::from_json_slice(REAL_NO_RESULT_REPORT.as_bytes())
191            .expect("a documented no-result report must parse");
192        assert_eq!(parsed.verdict, "no_result");
193        assert!(parsed.comparison.is_none());
194    }

196    #[test]
197    fn a_no_result_report_is_still_inadmissible_and_names_the_real_reason() {
198        // Parsing it is NOT admitting it. The point of the change is that the
199        // refusal now points at the producer that recorded nothing, instead of
200        // claiming the report could not be read.
201        let parsed = VerificationReport::from_json_slice(REAL_NO_RESULT_REPORT.as_bytes())
202            .expect("a documented no-result report must parse");
203        let refusal = parsed
204            .require_canonical_comparison()
205            .expect_err("a report with no comparison must never be admitted");
206        assert!(
207            refusal.contains("recorded no comparison at all"),
208            "refusal must name the missing comparison, got: {refusal}"
209        );
210        assert!(
211            refusal.contains("no_result"),
212            "refusal must carry the verdict, got: {refusal}"
213        );
214        assert!(parsed.require_canonical_match().is_err());
215    }

217    #[test]
218    fn a_missing_comparison_field_is_refused_rather_than_defaulting_to_none() {
219        // serde would otherwise map an absent Option field to None, letting a
220        // report that never mentioned a comparison read as a legitimate
221        // no-result. Optional must not mean absent.
222        let absent = r#"{"verified":false,"bitwise_parity":false,"verdict":"no_result","compared_log_messages":null}"#;
223        assert!(VerificationReport::from_json_slice(absent.as_bytes()).is_err());
224    }

226    #[test]
227    fn a_malformed_comparison_object_is_still_refused_at_parse() {
228        // Optional must not mean lenient: a comparison that is present but the
229        // wrong shape is still an unreadable report, not a silent None.
230        let malformed = REAL_NO_RESULT_REPORT.replace(r#""comparison":null"#, r#""comparison":7"#);
231        assert!(VerificationReport::from_json_slice(malformed.as_bytes()).is_err());
232        let missing_field = REAL_NO_RESULT_REPORT.replace(
233            r#""comparison":null"#,
234            r#""comparison":{"compare_logs":true}"#,
235        );
236        assert!(VerificationReport::from_json_slice(missing_field.as_bytes()).is_err());
237    }
