1//! A fresh cgroup v2 child that holds one budgeted E2E invocation.
2//!
3//! The runner reads the invocation's live CPU from the kernel's own counter
4//! (`cpu.stat` `usage_usec`) in that child, instead of scanning every host
5//! process for members of the invocation's process group. The counter needs no
6//! controller: cgroup v2 keeps it for every cgroup, and it covers every process
7//! in the cgroup, including one that left the process group or session.
8//!
9//! The helpers below follow `OwnedAttemptCgroup` in the nextest CPU wrapper
10//! (`src/bin/nextest-cpu-wrapper.rs`): the same descriptor identity checks and
11//! the same bounded control-file parsing. They are copied rather than shared
12//! because the wrapper is a separate binary; sharing them is a follow-up.
13

37/// Set to exactly `1` by a launcher that runs the E2E manifest without
38/// cgroups on purpose. Only then may the runner fall back to the agent-utils
39/// process-group scan, and only when the invocation cgroup cannot be created
40/// for lack of permission or of a cgroup v2 hierarchy. Any other value is
41/// refused.
42pub const ALLOW_PROCESS_GROUP_CPU_SCAN_ENV: &str = "HERMIT_E2E_ALLOW_PROCESS_GROUP_CPU_SCAN";
43
44/// How long the processes left in the cgroup after its leader was reaped get
45/// to exit after `cgroup.kill`. Matches the runner's SIGTERM-to-SIGKILL grace.
46const LEFTOVER_KILL_GRACE: Duration = Duration::from_secs(10);
47const LEFTOVER_POLL_INTERVAL: Duration = Duration::from_millis(50);
48/// How many levels of cgroups the command may leave below its invocation
49/// cgroup. A nested runner adds one level for each invocation it runs, so a
50/// real tree is a few levels deep; a deeper one is refused, not walked.
51const MAX_NESTED_CGROUP_DEPTH: usize = 32;
52const CGROUP2_SUPER_MAGIC: libc::c_long = 0x6367_7270;
53const CGROUP_ROOT: &str = "/sys/fs/cgroup";
54
55static NEXT_INVOCATION: AtomicU64 = AtomicU64::new(0);

57/// Read the fallback marker. Unset means the fallback is not allowed.
58pub fn process_group_scan_allowed(value: Option<&OsStr>) -> Result<bool, String> {
59    match value {
60        None => Ok(false),
61        Some(value) if value.as_bytes() == b"1" => Ok(true),
62        Some(value) => Err(format!(
63            "{ALLOW_PROCESS_GROUP_CPU_SCAN_ENV}={value:?} is not a recognized value: set it to exactly 1 to allow the process-group CPU scan when an invocation cgroup cannot be created, or leave it unset"
64        )),
65    }
66}

68/// Why an invocation cgroup could not be created.
69#[derive(Clone, Debug, Eq, PartialEq)]
70pub struct CreateError {
71    /// True only when the failure shows that this process cannot place an
72    /// invocation in a cgroup of its own here at all: there is no cgroup v2
73    /// hierarchy at the runner's cgroup, or the runner lacks permission to
74    /// create a child cgroup or to move a process into one. Every other
75    /// failure (a malformed `/proc/self/cgroup`, a parent that is not a
76    /// directory, a name collision, a failure after the child was created) is
77    /// not eligible, so the marker never hides it.
78    pub fallback_eligible: bool,
79    pub message: String,
80}
81
82impl CreateError {
83    fn eligible(message: String) -> Self {
84        Self {
85            fallback_eligible: true,
86            message,
87        }
88    }
89
90    fn ineligible(message: String) -> Self {
91        Self {
92            fallback_eligible: false,
93            message,
94        }
95    }
96
97    fn from_errno(message: String, error: &io::Error, eligible_errnos: &[i32]) -> Self {
98        let eligible = error
99            .raw_os_error()
100            .is_some_and(|errno| eligible_errnos.contains(&errno));
101        Self {
102            fallback_eligible: eligible,
103            message,
104        }
105    }
106}

114fn file_identity(file: &File, label: &str) -> Result<FileIdentity, String> {
115    let mut stat = unsafe { std::mem::zeroed::<libc::stat>() };
116    if unsafe { libc::fstat(file.as_raw_fd(), &mut stat) } != 0 {
117        return Err(format!(
118            "cannot inspect {label} identity: {}",
119            io::Error::last_os_error()
120        ));
121    }
122    Ok(FileIdentity {
123        device: stat.st_dev,
124        inode: stat.st_ino,
125    })
126}

128fn cgroup_text(file: &File, label: &str) -> Result<String, String> {
129    const LIMIT: usize = 8192;
130    let mut bytes = Vec::new();
131    let mut chunk = [0u8; 1024];
132    loop {
133        let count = file
134            .read_at(&mut chunk, bytes.len() as u64)
135            .map_err(|error| format!("cannot read {label}: {error}"))?;
136        if count == 0 {
137            break;
138        }
139        if bytes.len() + count > LIMIT {
140            return Err(format!("{label} exceeds the {LIMIT}-byte accounting bound"));
141        }
142        bytes.extend_from_slice(&chunk[..count]);
143    }
144    std::str::from_utf8(&bytes)
145        .map(str::to_owned)
146        .map_err(|error| format!("{label} is not valid UTF-8: {error}"))
147}

149fn cgroup_field(file: &File, file_label: &str, field: &str) -> Result<u64, String> {
150    let text = cgroup_text(file, file_label)?;
151    let mut found = None;
152    for line in text.lines() {
153        let mut words = line.split_whitespace();
154        let Some(name) = words.next() else {
155            continue;
156        };
157        let value = words
158            .next()
159            .ok_or_else(|| format!("{file_label} has no value for {name:?}"))?;
160        if words.next().is_some() {
161            return Err(format!("{file_label} has extra fields on line {line:?}"));
162        }
163        if name == field {
164            if found.is_some() {
165                return Err(format!("{file_label} repeats field {field:?}"));
166            }
167            found = Some(
168                value
169                    .parse::<u64>()
170                    .map_err(|error| format!("{file_label} has invalid {field}: {error}"))?,
171            );
172        }
173    }
174    found.ok_or_else(|| format!("{file_label} is missing field {field:?}"))
175}

177fn openat_file(directory: &File, name: &str, flags: i32) -> io::Result<File> {
178    let name = CString::new(name).expect("invocation cgroup control names contain no NUL");
179    let fd = unsafe { libc::openat(directory.as_raw_fd(), name.as_ptr(), flags, 0) };
180    if fd < 0 {
181        return Err(io::Error::last_os_error());
182    }
183    Ok(unsafe { File::from_raw_fd(fd) })
184}

186fn open_directory_at(directory: &File, name: &CStr) -> io::Result<File> {
187    let fd = unsafe {
188        libc::openat(
189            directory.as_raw_fd(),
190            name.as_ptr(),
191            libc::O_RDONLY | libc::O_CLOEXEC | libc::O_DIRECTORY | libc::O_NOFOLLOW,
192            0,
193        )
194    };
195    if fd < 0 {
196        return Err(io::Error::last_os_error());
197    }
198    Ok(unsafe { File::from_raw_fd(fd) })
199}

201/// The identity of the entry `name` in `directory`, without following a link,
202/// and whether it is a directory.
203fn entry_identity(directory: &File, name: &CStr) -> io::Result<(FileIdentity, bool)> {
204    let mut stat = unsafe { std::mem::zeroed::<libc::stat>() };
205    if unsafe {
206        libc::fstatat(
207            directory.as_raw_fd(),
208            name.as_ptr(),
209            &mut stat,
210            libc::AT_SYMLINK_NOFOLLOW,
211        )
212    } != 0
213    {
214        return Err(io::Error::last_os_error());
215    }
216    Ok((
217        FileIdentity {
218            device: stat.st_dev,
219            inode: stat.st_ino,
220        },
221        stat.st_mode & libc::S_IFMT == libc::S_IFDIR,
222    ))
223}

225/// The names of the cgroups directly below `directory`, which are its
226/// subdirectories; its control files are skipped.
227fn nested_cgroup_names(directory: &File) -> io::Result<Vec<CString>> {
228    // fdopendir owns the descriptor it is given and moves its offset, so it
229    // gets a fresh descriptor for the same directory.
230    let fd = unsafe {
231        libc::openat(
232            directory.as_raw_fd(),
233            c".".as_ptr(),
234            libc::O_RDONLY | libc::O_CLOEXEC | libc::O_DIRECTORY,
235            0,
236        )
237    };
238    if fd < 0 {
239        return Err(io::Error::last_os_error());
240    }
241    let stream = unsafe { libc::fdopendir(fd) };
242    if stream.is_null() {
243        let error = io::Error::last_os_error();
244        unsafe {
245            libc::close(fd);
246        }
247        return Err(error);
248    }
249    let mut names = Vec::new();
250    let outcome = loop {
251        // readdir reports an error only through errno.
252        unsafe {
253            *libc::__errno_location() = 0;
254        }
255        let entry = unsafe { libc::readdir(stream) };
256        if entry.is_null() {
257            let error = io::Error::last_os_error();
258            break match error.raw_os_error() {
259                Some(0) => Ok(()),
260                _ => Err(error),
261            };
262        }
263        // SAFETY: a non-null readdir result stays valid until the next
264        // readdir or closedir on this stream, and its name is NUL-terminated.
265        let (name, kind) = unsafe { (CStr::from_ptr((*entry).d_name.as_ptr()), (*entry).d_type) };
266        if matches!(name.to_bytes(), b"." | b"..") {
267            continue;
268        }
269        let is_directory = match kind {
270            libc::DT_DIR => true,
271            libc::DT_UNKNOWN => match entry_identity(directory, name) {
272                Ok((_, is_directory)) => is_directory,
273                Err(error) => break Err(error),
274            },
275            _ => false,
276        };
277        if is_directory {
278            names.push(name.to_owned());
279        }
280    };
281    unsafe {
282        libc::closedir(stream);
283    }
284    outcome.map(|()| names)
285}

287/// Remove every cgroup below `directory`, deepest first. `level` is how far
288/// below the invocation cgroup the entries of `directory` are. Each nested
289/// cgroup is opened through its parent's descriptor without following a link
290/// and must be on the invocation cgroup's filesystem, and its name is removed
291/// only while it still names the directory that was just emptied.
292fn remove_nested_cgroups(
293    directory: &File,
294    path: &Path,
295    device: u64,
296    level: usize,
297) -> Result<(), String> {
298    let names = nested_cgroup_names(directory)
299        .map_err(|error| format!("cannot list cgroup {}: {error}", path.display()))?;
300    for name in names {
301        let nested_path = path.join(OsStr::from_bytes(name.to_bytes()));
302        if level > MAX_NESTED_CGROUP_DEPTH {
303            return Err(format!(
304                "nested cgroup {} is more than {MAX_NESTED_CGROUP_DEPTH} levels below the invocation cgroup",
305                nested_path.display()
306            ));
307        }
308        let nested = open_directory_at(directory, &name).map_err(|error| {
309            format!(
310                "cannot open nested cgroup {}: {error}",
311                nested_path.display()
312            )
313        })?;
314        let identity = file_identity(&nested, "nested cgroup")?;
315        if identity.device != device {
316            return Err(format!(
317                "nested cgroup {} is on device {}, expected {device}",
318                nested_path.display(),
319                identity.device
320            ));
321        }
322        remove_nested_cgroups(&nested, &nested_path, device, level + 1)?;
323        let (named, is_directory) = entry_identity(directory, &name).map_err(|error| {
324            format!(
325                "cannot authenticate nested cgroup {}: {error}",
326                nested_path.display()
327            )
328        })?;
329        if named != identity || !is_directory {
330            return Err(format!(
331                "nested cgroup {} was replaced",
332                nested_path.display()
333            ));
334        }
335        if unsafe { libc::unlinkat(directory.as_raw_fd(), name.as_ptr(), libc::AT_REMOVEDIR) } != 0
336        {
337            return Err(format!(
338                "cannot remove nested cgroup {}: {}",
339                nested_path.display(),
340                io::Error::last_os_error()
341            ));
342        }
343    }
344    Ok(())
345}

347/// Map the unified (`0::`) line of `/proc/self/cgroup` to its directory under
348/// `/sys/fs/cgroup`. The path is relative to this process's cgroup namespace,
349/// as `/sys/fs/cgroup` is for a hierarchy mounted inside that namespace.
350fn unified_cgroup_directory(raw: &str) -> Result<PathBuf, CreateError> {
351    let mut unified = raw.lines().filter_map(|line| line.strip_prefix("0::"));
352    let path = unified.next().ok_or_else(|| {
353        CreateError::eligible("/proc/self/cgroup has no unified cgroup v2 entry".into())
354    })?;
355    if unified.next().is_some() {
356        return Err(CreateError::ineligible(
357            "/proc/self/cgroup has multiple unified cgroup v2 entries".into(),
358        ));
359    }
360    let relative = Path::new(path.trim_start_matches('/'));
361    if relative.components().any(|component| {
362        !matches!(
363            component,
364            std::path::Component::Normal(_) | std::path::Component::CurDir
365        )
366    }) {
367        return Err(CreateError::ineligible(format!(
368            "unified cgroup path {path:?} is not a relative kernel path"
369        )));
370    }
371    Ok(Path::new(CGROUP_ROOT).join(relative))
372}
373
374fn current_cgroup_directory() -> Result<PathBuf, CreateError> {
375    let raw = fs::read_to_string("/proc/self/cgroup").map_err(|error| {
376        CreateError::from_errno(
377            format!("cannot read /proc/self/cgroup: {error}"),
378            &error,
379            &[libc::ENOENT],
380        )
381    })?;
382    unified_cgroup_directory(&raw)
383}

385fn invocation_name() -> String {
386    // The process nonce keeps a directory left behind by an earlier runner
387    // that had the same PID from colliding with this runner's names.
388    static NONCE: OnceLock<u64> = OnceLock::new();
389    let nonce = NONCE.get_or_init(|| {
390        SystemTime::now()
391            .duration_since(UNIX_EPOCH)
392            .map_or(0, |elapsed| elapsed.as_nanos() as u64)
393    });
394    format!(
395        "hermit-e2e-invocation-{}-{nonce:x}-{}",
396        std::process::id(),
397        NEXT_INVOCATION.fetch_add(1, Ordering::Relaxed)
398    )
399}

401/// One invocation's cgroup: a fresh, empty child of the runner's own cgroup,
402/// held through descriptors whose identity is checked before every use.
403pub struct InvocationCgroup {
404    parent: File,
405    child: File,
406    cpu_stat: File,
407    events: File,
408    procs_read: File,
409    procs_write: File,
410    kill: File,
411    name: CString,
412    identity: FileIdentity,
413    path: PathBuf,
414    finished: Option<Result<(), String>>,
415}
416
417impl InvocationCgroup {
418    /// Create the cgroup under the runner's own cgroup.
419    pub fn create() -> Result<Self, CreateError> {
420        Self::create_in(&current_cgroup_directory()?)
421    }
422
423    /// Create the cgroup under `parent_path`, which must be the cgroup that
424    /// holds this process: moving a process into the child needs write access
425    /// to the `cgroup.procs` of the common ancestor of both cgroups.
426    pub fn create_in(parent_path: &Path) -> Result<Self, CreateError> {
427        let parent = fs::OpenOptions::new()
428            .read(true)
429            .custom_flags(libc::O_CLOEXEC | libc::O_DIRECTORY | libc::O_NOFOLLOW)
430            .open(parent_path)
431            .map_err(|error| {
432                CreateError::from_errno(
433                    format!("cannot open cgroup {}: {error}", parent_path.display()),
434                    &error,
435                    &[libc::EACCES, libc::EPERM, libc::ENOENT],
436                )
437            })?;
438        let mut filesystem = unsafe { std::mem::zeroed::<libc::statfs>() };
439        if unsafe { libc::fstatfs(parent.as_raw_fd(), &mut filesystem) } != 0 {
440            return Err(CreateError::ineligible(format!(
441                "cannot inspect the filesystem of cgroup {}: {}",
442                parent_path.display(),
443                io::Error::last_os_error()
444            )));
445        }
446        if filesystem.f_type != CGROUP2_SUPER_MAGIC {
447            return Err(CreateError::eligible(format!(
448                "{} is not a cgroup v2 directory",
449                parent_path.display()
450            )));
451        }
452        // The child process moves itself out of this cgroup before exec. That
453        // is permitted only with write access to this cgroup's cgroup.procs, so
454        // check it now, while a refusal can still choose the fallback, rather
455        // than at spawn time.
456        openat_file(
457            &parent,
458            "cgroup.procs",
459            libc::O_WRONLY | libc::O_CLOEXEC | libc::O_NOFOLLOW,
460        )
461        .map_err(|error| {
462            CreateError::from_errno(
463                format!(
464                    "cannot move processes out of cgroup {}: {}/cgroup.procs: {error}",
465                    parent_path.display(),
466                    parent_path.display()
467                ),
468                &error,
469                &[libc::EACCES, libc::EPERM, libc::EROFS],
470            )
471        })?;
472        let name_text = invocation_name();
473        let path = parent_path.join(&name_text);
474        let name = CString::new(name_text.as_bytes())
475            .map_err(|_| CreateError::ineligible("cgroup name contains a NUL byte".into()))?;
476        if unsafe { libc::mkdirat(parent.as_raw_fd(), name.as_ptr(), 0o755) } != 0 {
477            let error = io::Error::last_os_error();
478            return Err(CreateError::from_errno(
479                format!(
480                    "cannot create invocation cgroup {}: {error}",
481                    path.display()
482                ),
483                &error,
484                &[libc::EACCES, libc::EPERM, libc::EROFS],
485            ));
486        }
487        let result = (|| {
488            let child = openat_file(
489                &parent,
490                &name_text,
491                libc::O_RDONLY | libc::O_CLOEXEC | libc::O_DIRECTORY | libc::O_NOFOLLOW,
492            )
493            .map_err(|error| format!("cannot open invocation cgroup: {error}"))?;
494            let identity = file_identity(&child, "invocation cgroup")?;
495            let open_control = |control: &str, flags: i32| -> Result<File, String> {
496                let file = openat_file(&child, control, flags | libc::O_CLOEXEC | libc::O_NOFOLLOW)
497                    .map_err(|error| format!("cannot open invocation cgroup {control}: {error}"))?;
498                let control_identity = file_identity(&file, control)?;
499                if control_identity.device != identity.device {
500                    return Err(format!(
501                        "invocation cgroup {control} is on device {}, expected {}",
502                        control_identity.device, identity.device
503                    ));
504                }
505                Ok(file)
506            };
507            let cpu_stat = open_control("cpu.stat", libc::O_RDONLY)?;
508            let events = open_control("cgroup.events", libc::O_RDONLY)?;
509            let procs_read = open_control("cgroup.procs", libc::O_RDONLY)?;
510            let procs_write = open_control("cgroup.procs", libc::O_WRONLY)?;
511            let kill = open_control("cgroup.kill", libc::O_WRONLY)?;
512            let mut created = Self {
513                parent: parent.try_clone().map_err(|error| {
514                    format!("cannot retain the parent cgroup descriptor: {error}")
515                })?,
516                child,
517                cpu_stat,
518                events,
519                procs_read,
520                procs_write,
521                kill,
522                name: name.clone(),
523                identity,
524                path: path.clone(),
525                finished: None,
526            };
527            let fresh = created.verify_identity().and_then(|()| {
528                if created.cpu_usage_usec()? != 0 {
529                    return Err("fresh invocation cgroup has nonzero cpu.stat usage_usec".into());
530                }
531                if created.populated()? || !created.procs_empty()? {
532                    return Err("fresh invocation cgroup is not empty before enrollment".into());
533                }
534                Ok(())
535            });
536            match fresh {
537                Ok(()) => Ok(created),
538                Err(error) => {
539                    // The directory is removed below; Drop must not repeat it.
540                    created.finished = Some(Err(error.clone()));
541                    Err(error)
542                }
543            }
544        })();
545        result.map_err(|error| {
546            let message = if unsafe {
547                libc::unlinkat(parent.as_raw_fd(), name.as_ptr(), libc::AT_REMOVEDIR)
548            } == 0
549            {
550                format!("invocation cgroup {}: {error}", path.display())
551            } else {
552                format!(
553                    "invocation cgroup {}: {error}; removing the partly initialized cgroup also failed: {}",
554                    path.display(),
555                    io::Error::last_os_error()
556                )
557            };
558            CreateError::ineligible(message)
559        })
560    }

562    pub fn path(&self) -> &Path {
563        &self.path
564    }

566    /// This cgroup as the unified (`0::`) line of `/proc/<pid>/cgroup` names
567    /// it for a process inside it, or `None` when its directory is not under
568    /// `/sys/fs/cgroup`.
569    pub fn kernel_path(&self) -> Option<String> {
570        let relative = self.path.strip_prefix(CGROUP_ROOT).ok()?.to_str()?;
571        Some(format!("/{relative}"))
572    }

574    /// The `cgroup.procs` descriptor a child writes `0` to before exec. It is
575    /// close-on-exec, so the program never inherits it.
576    pub fn enrollment_fd(&self) -> RawFd {
577        self.procs_write.as_raw_fd()
578    }

580    fn verify_identity(&self) -> Result<(), String> {
581        let held = file_identity(&self.child, "held invocation cgroup")?;
582        if held != self.identity {
583            return Err("held invocation cgroup identity changed".into());
584        }
585        let mut stat = unsafe { std::mem::zeroed::<libc::stat>() };
586        if unsafe {
587            libc::fstatat(
588                self.parent.as_raw_fd(),
589                self.name.as_ptr(),
590                &mut stat,
591                libc::AT_SYMLINK_NOFOLLOW,
592            )
593        } != 0
594        {
595            return Err(format!(
596                "cannot authenticate invocation cgroup path {}: {}",
597                self.path.display(),
598                io::Error::last_os_error()
599            ));
600        }
601        let named = FileIdentity {
602            device: stat.st_dev,
603            inode: stat.st_ino,
604        };
605        if named != self.identity || stat.st_mode & libc::S_IFMT != libc::S_IFDIR {
606            return Err(format!(
607                "invocation cgroup path {} was replaced",
608                self.path.display()
609            ));
610        }
611        Ok(())
612    }

614    /// Total CPU of every process that has run in this cgroup, in
615    /// microseconds. The kernel keeps it monotonic for one cgroup.
616    pub fn cpu_usage_usec(&self) -> Result<u64, String> {
617        self.verify_identity()?;
618        cgroup_field(&self.cpu_stat, "invocation cgroup cpu.stat", "usage_usec")
619    }
620
621    /// Whether a live process remains. A zombie no longer counts.
622    pub fn populated(&self) -> Result<bool, String> {
623        self.verify_identity()?;
624        match cgroup_field(&self.events, "invocation cgroup.events", "populated")? {
625            0 => Ok(false),
626            1 => Ok(true),
627            value => Err(format!(
628                "invocation cgroup.events has invalid populated value {value}"
629            )),
630        }
631    }

633    /// Whether `cgroup.procs` lists no process. One short read decides it, so a
634    /// long list cannot exceed a parsing bound.
635    pub fn procs_empty(&self) -> Result<bool, String> {
636        self.verify_identity()?;
637        let mut chunk = [0u8; 64];
638        let count = self
639            .procs_read
640            .read_at(&mut chunk, 0)
641            .map_err(|error| format!("cannot read invocation cgroup.procs: {error}"))?;
642        Ok(count == 0)
643    }

645    /// Send SIGKILL to every process in this cgroup and its descendants.
646    pub fn kill(&self) -> Result<(), String> {
647        self.verify_identity()?;
648        loop {
649            let written = unsafe { libc::write(self.kill.as_raw_fd(), b"1\n".as_ptr().cast(), 2) };
650            if written == 2 {
651                return Ok(());
652            }
653            if written < 0 && io::Error::last_os_error().kind() == io::ErrorKind::Interrupted {
654                continue;
655            }
656            return Err(if written < 0 {
657                format!(
658                    "cannot write invocation cgroup.kill: {}",
659                    io::Error::last_os_error()
660                )
661            } else {
662                format!("short write to invocation cgroup.kill: {written} bytes")
663            });
664        }
665    }
666
667    fn empty(&self) -> Result<bool, String> {
668        Ok(!self.populated()? && self.procs_empty()?)
669    }
670
671    fn kill_leftovers(&self) -> Result<(), String> {
672        if self.empty()? {
673            return Ok(());
674        }
675        self.kill()?;
676        let deadline = Instant::now() + LEFTOVER_KILL_GRACE;
677        loop {
678            if self.empty()? {
679                return Ok(());
680            }
681            if Instant::now() >= deadline {
682                return Err(format!(
683                    "processes remain {} s after cgroup.kill",
684                    LEFTOVER_KILL_GRACE.as_secs()
685                ));
686            }
687            thread::sleep(LEFTOVER_POLL_INTERVAL);
688        }
689    }
690
691    fn remove_empty(&self) -> Result<(), String> {
692        self.verify_identity()?;
693        if !self.empty()? {
694            return Err("it is still populated".into());
695        }
696        // The command may have made cgroups of its own below this one, as a
697        // nested runner does, and rmdir refuses a cgroup that still has a
698        // child. `populated` covers the whole subtree, so nothing is alive
699        // anywhere below here and each nested cgroup can go, deepest first.
700        remove_nested_cgroups(&self.child, &self.path, self.identity.device, 1)?;
701        self.verify_identity()?;
702        if unsafe {
703            libc::unlinkat(
704                self.parent.as_raw_fd(),
705                self.name.as_ptr(),
706                libc::AT_REMOVEDIR,
707            )
708        } != 0
709        {
710            return Err(format!("rmdir: {}", io::Error::last_os_error()));
711        }
712        Ok(())
713    }
714
715    /// Kill whatever is left in the cgroup, wait up to 10 s for it to exit, and
716    /// remove the cgroup together with any cgroups the command made below it.
717    /// Called once its leader has been reaped (or could not be), and by Drop;
718    /// later calls return the first outcome.
719    pub fn finish(&mut self) -> Result<(), String> {
720        if let Some(outcome) = &self.finished {
721            return outcome.clone();
722        }
723        let outcome = self
724            .kill_leftovers()
725            .and_then(|()| self.remove_empty())
726            .map_err(|error| {
727                format!(
728                    "cannot remove invocation cgroup {}: {error}",
729                    self.path.display()
730                )
731            });
732        self.finished = Some(outcome.clone());
733        outcome
734    }
735}

737impl Drop for InvocationCgroup {
738    fn drop(&mut self) {
739        if self.finished.is_none() {
740            if let Err(error) = self.finish() {
741                eprintln!("{error}");
742            }
743        }
744    }

967    /// When dropped, removes what a failed test left of an invocation cgroup
968    /// this test process created: the cgroups nested in it, deepest first, and
969    /// then the cgroup itself. The location is the cgroup directory, or a file
970    /// holding a copy of a member's `/proc/<pid>/cgroup`. Any other directory
971    /// is left alone.
972    pub(crate) struct RemoveLeftoverCgroup(pub(crate) PathBuf);
973
974    impl Drop for RemoveLeftoverCgroup {
975        fn drop(&mut self) {
976            fn remove_tree(directory: &Path) {
977                if let Ok(entries) = fs::read_dir(directory) {
978                    for entry in entries.flatten() {
979                        if entry.file_type().is_ok_and(|kind| kind.is_dir()) {
980                            remove_tree(&entry.path());
981                        }
982                    }
983                }
984                let _ = fs::remove_dir(directory);
985            }
986            let directory = if self.0.starts_with(CGROUP_ROOT) {
987                self.0.clone()
988            } else {
989                let Some(relative) = fs::read_to_string(&self.0).ok().and_then(|text| {
990                    text.lines()
991                        .find_map(|line| line.strip_prefix("0::").map(str::to_owned))
992                }) else {
993                    return;
994                };
995                Path::new(CGROUP_ROOT).join(relative.trim_start_matches('/'))
996            };
997            let ours = format!("hermit-e2e-invocation-{}-", std::process::id());
998            if directory
999                .file_name()
1000                .is_some_and(|name| name.to_string_lossy().starts_with(&ours))
1001            {
1002                remove_tree(&directory);
1003            }
1004        }
1005    }
