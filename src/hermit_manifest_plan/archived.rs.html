1//! The archived execution paths: the backend and the preprocessing step the
2//! owner stopped developing, whose tests left the per-commit validation for
3//! the nightly pressure run.
4//!
5//! Owner direction, 2026-10-25: "mark our sabre and e9patch backends as
6//! archived for now and move their tests to nightly only and not actively
7//! develop them." Archived code still builds in every validation (the
8//! third-party-backends workspace build, Clippy and the SaBRe smoke test
9//! `test.sabre_examples`), so it cannot rot silently, but no feature work
10//! lands on it.
11//!
12//! A manifest cell on an archived backend that would belong to the full run
13//! type belongs to [`ARCHIVED_RUN_TYPE`] instead ([`crate::runner::cell_labels`]).
14//! Full validation, its portable and hosted subsets, and the Buck route (whose
15//! cells are `ci/expected-e2e-plan.json`'s) therefore do not select it, and the
16//! scorecard reports it as not selected by full: a red cell, which the nightly
17//! pressure sweep's red part runs three times a night, and a verify cell, which
18//! its parity-all part measures beside ptrace. A cell that already names a
19//! focused run type, such as `sabre-compat-only`, keeps it.
20
21/// What an archived entry is. Hermit AGENTS.md, "Backend Definition": only a
22/// path that loads Detcore as a Reverie tool is a backend; e9patch rewrites a
23/// binary that then runs on the ptrace backend.
24#[derive(Clone, Copy, Debug, Eq, PartialEq)]
25pub enum ArchivedKind {
26    /// A Reverie backend running `Detcore<XxxGuest>`, named by manifest cells.
27    Backend,
28    /// Preprocessing whose output runs on the ptrace backend. It has no
29    /// manifest cells; its corpus is `ci/compat/corpus-e9patch.json`, run by
30    /// `scripts/validate.rs --e9patch-compat-only`.
31    PreprocessingWithPtrace,
32}
33
34/// One archived execution path.
35#[derive(Clone, Copy, Debug)]
36pub struct Archived {
37    /// The `--backend` spelling.
38    pub name: &'static str,
39    pub kind: ArchivedKind,
40    /// The day of the owner's direction (US Eastern).
41    pub since: &'static str,
42    /// Where its tests run now that per-commit validation runs only its smoke
43    /// checks: "nightly only" when a scheduled run exercises them, "on demand
44    /// only" when nothing scheduled does.
45    pub schedule: &'static str,
46    /// What still runs per commit, and what runs only nightly.
47    pub coverage: &'static str,
48}
49

83/// The run type of an archived backend's cells that would otherwise be full's.
84/// No validation profile selects it; the pressure test runs these cells under
85/// it, as it runs every focused run type's.
86pub const ARCHIVED_RUN_TYPE: &str = "archived-nightly";
87
88/// Whether manifest cells on `backend` are archived.
89pub fn is_archived_backend(backend: &str) -> bool {
90    ARCHIVED
91        .iter()
92        .any(|entry| entry.kind == ArchivedKind::Backend && entry.name == backend)
93}

95#[cfg(test)]
96mod tests {
97    use super::*;
98
99    #[test]
100    fn sabre_is_an_archived_backend_and_e9patch_is_archived_preprocessing() {
101        assert!(is_archived_backend("sabre"));
102        // e9patch is not a backend, so no manifest cell is archived by its name.
103        assert!(!is_archived_backend("e9patch"));
104        assert!(
105            ARCHIVED.iter().any(|entry| entry.name == "e9patch"
106                && entry.kind == ArchivedKind::PreprocessingWithPtrace)
107        );
108        for backend in [
109            "ptrace",
110            "dbt",
111            "kvm",
112            "liteinst",
113            "in-guest-trap",
114            "native",
115        ] {
116            assert!(!is_archived_backend(backend), "{backend}");
117        }
118    }
