57fn parse_format() -> Format {
58    let mut format = Format::Text;
59    let mut args = std::env::args().skip(1);
60    while let Some(arg) = args.next() {
61        let value = if arg == "--format" {
62            args.next()
63                .unwrap_or_else(|| die("--format requires a value"))
64        } else if let Some(value) = arg.strip_prefix("--format=") {
65            value.to_string()
66        } else {
67            die(format!("unknown argument: {arg}"));
68        };
69        format = match value.as_str() {
70            "text" => Format::Text,
71            "json" => Format::Json,
72            "harness-json" => Format::HarnessJson,
73            _ => die(format!("unknown format: {value}")),
74        };
75    }
76    format
77}

79fn main() {
80    let format = parse_format();
81    let script_dir = PathBuf::from(env!("CARGO_MANIFEST_DIR")).join("../../tests/e2e/manifests");
82    let repo_root = script_dir.join("../../..");
83
84    let mut manifests: Vec<PathBuf> = std::fs::read_dir(&script_dir)
85        .unwrap_or_else(|error| die(format!("cannot read {}: {error}", script_dir.display())))
86        .filter_map(|entry| entry.ok().map(|entry| entry.path()))
87        .filter(|path| {
88            path.extension()
89                .is_some_and(|extension| extension == "toml")
90        })
91        .collect();
92    manifests.sort();
93    if manifests.is_empty() {
94        die(format!(
95            "no *.toml manifests found in {}",
96            script_dir.display()
97        ));
98    }
99
100    let mut rows = Vec::new();
101    let mut seen_ids = BTreeSet::new();
102    let mut seen_programs = BTreeSet::new();
103    let mut documents = Vec::new();
104
105    for path in &manifests {
106        let text = std::fs::read_to_string(path)
107            .unwrap_or_else(|error| die(format!("cannot read {}: {error}", path.display())));
108        let document: Value = text
109            .parse()
110            .unwrap_or_else(|error| die(format!("{}: invalid TOML: {error}", path.display())));
111        let location = path.file_name().unwrap().to_string_lossy().to_string();
112        ensure_keys(&document, &["schema", "bucket", "test"], &location);
113
114        if document.get("schema").and_then(Value::as_integer) != Some(2) {
115            die(format!("{location}: schema must be 2"));
116        }
117        let bucket = required_string(&document, "bucket", &location);
118        let stem = path.file_stem().unwrap().to_string_lossy();
119        if bucket != stem {
120            die(format!(
121                "{location}: bucket `{bucket}` must equal file stem `{stem}`"
122            ));
123        }
124        let tests = document
125            .get("test")
126            .and_then(Value::as_array)
127            .filter(|tests| !tests.is_empty())
128            .unwrap_or_else(|| die(format!("{location}: missing non-empty [[test]] array")));
129        for test in tests {
130            validate_and_expand(
131                test,
132                bucket,
133                &location,
134                &repo_root,
135                &mut seen_ids,
136                &mut seen_programs,
137                &mut rows,
138            );
139        }
140        documents.push(document);
141    }
142
143    validate_front_door(&repo_root, &documents);
144
145    rows.sort_by(|left, right| {
146        (&left.bucket, &left.id, &left.mode, &left.backend).cmp(&(
147            &right.bucket,
148            &right.id,
149            &right.mode,
150            &right.backend,
151        ))
152    });
153
154    match format {
155        Format::HarnessJson => {
156            println!(
157                "{}",
158                serde_json::to_string(&documents)
159                    .unwrap_or_else(|error| die(format!("cannot encode manifests: {error}")))
160            );
161        }
162        Format::Json => {
163            let output: Vec<_> = rows
164                .iter()
165                .map(|row| {
166                    json!({
167                        "bucket": row.bucket,
168                        "test": row.id,
169                        "lane": row.lane,
170                        "mode": row.mode,
171                        "backend": row.backend,
172                        "ci": row.ci,
173                    })
174                })
175                .collect();
176            println!(
177                "{}",
178                serde_json::to_string(&output)
179                    .unwrap_or_else(|error| die(format!("cannot encode plan: {error}")))
180            );
181        }
182        Format::Text => {
183            println!(
184                "{:<10}\t{:<38}\t{:<10}\t{:<8}\t{:<5}\tBUCKET",
185                "LANE", "TEST", "MODE", "BACKEND", "CI"
186            );
187            for row in &rows {
188                println!(
189                    "{:<10}\t{:<38}\t{:<10}\t{:<8}\t{:<5}\t{}",
190                    row.lane, row.id, row.mode, row.backend, row.ci, row.bucket
191                );
192            }
193            eprintln!(
194                "\nPASS: {} manifest(s), {} test(s), {} enabled plan cells validated",
195                manifests.len(),
196                seen_ids.len(),
197                rows.len()
198            );
199        }
200    }
201}

203fn json_string_set(value: &JsonValue, key: &str, location: &str) -> BTreeSet<String> {
204    let values = value
205        .get(key)
206        .and_then(JsonValue::as_array)
207        .unwrap_or_else(|| die(format!("{location}: `{key}` must be an array")));
208    let result: BTreeSet<String> = values
209        .iter()
210        .map(|item| {
211            item.as_str()
212                .filter(|item| !item.is_empty())
213                .map(str::to_string)
214                .unwrap_or_else(|| {
215                    die(format!(
216                        "{location}: `{key}` entries must be non-empty strings"
217                    ))
218                })
219        })
220        .collect();
221    if result.len() != values.len() {
222        die(format!("{location}: `{key}` contains duplicate entries"));
223    }
224    result
225}

227fn names_backend(value: &str) -> bool {
228    value
229        .split(|character: char| !character.is_ascii_alphanumeric())
230        .any(|token| {
231            let token = token.to_ascii_lowercase();
232            matches!(
233                token.as_str(),
234                "ptrace" | "dbt" | "dynamorio" | "kvm" | "sabre" | "e9patch"
235            ) || token.starts_with("liteinst")
236        })
237}

239fn backend_private_guest_files(inventory: &JsonValue) -> BTreeSet<String> {
240    inventory
241        .get("files")
242        .and_then(JsonValue::as_array)
243        .unwrap_or_else(|| die(format!("{TEST_INVENTORY}: `files` must be an array")))
244        .iter()
245        .filter(|entry| {
246            entry.get("disposition").and_then(JsonValue::as_str) == Some("guest-fixture")
247        })
248        .filter_map(|entry| {
249            let path = entry.get("path").and_then(JsonValue::as_str)?;
250            let runner = entry
251                .get("runner")
252                .and_then(JsonValue::as_str)
253                .unwrap_or("");
254            let parity_private = path.starts_with("tests/backend-parity/")
255                || runner.contains("tests/backend-parity/");
256            (parity_private || names_backend(path) || names_backend(runner))
257                .then(|| path.to_string())
258        })
259        .collect()
260}

262fn asymmetric_manifest_tests(documents: &[Value]) -> BTreeSet<String> {
263    let mut asymmetric = BTreeSet::new();
264    for document in documents {
265        for test in document
266            .get("test")
267            .and_then(Value::as_array)
268            .into_iter()
269            .flatten()
270        {
271            let Some(id) = test.get("id").and_then(Value::as_str) else {
272                continue;
273            };
274            let Some(modes) = test.get("modes").and_then(Value::as_table) else {
275                continue;
276            };
277            let mut has_ptrace_front_door = false;
278            let mut has_backend_without_ptrace = false;
279            for mode in MODES.into_iter().filter(|mode| *mode != "naked") {
280                let enabled = modes
281                    .get(mode)
282                    .and_then(Value::as_table)
283                    .and_then(|spec| spec.get("backends_enabled"))
284                    .and_then(Value::as_array)
285                    .into_iter()
286                    .flatten()
287                    .filter_map(Value::as_str)
288                    .collect::<Vec<_>>();
289                let has_ptrace = enabled.contains(&"ptrace");
290                has_ptrace_front_door |= has_ptrace;
291                has_backend_without_ptrace |= !enabled.is_empty() && !has_ptrace;
292            }
293            if !has_ptrace_front_door || has_backend_without_ptrace {
294                asymmetric.insert(id.to_string());
295            }
296        }
297    }
298    asymmetric
299}

301fn enforce_exact_ratchet(label: &str, actual: &BTreeSet<String>, baseline: &BTreeSet<String>) {
302    let unexpected: Vec<_> = actual.difference(baseline).cloned().collect();
303    let stale: Vec<_> = baseline.difference(actual).cloned().collect();
304    if !unexpected.is_empty() || !stale.is_empty() {
305        die(format!(
306            "matrix symmetry {label} changed; unexpected={unexpected:?}, stale_baseline={stale:?}. New compatibility coverage must enter a shared schema-v2 TOML manifest, establish ptrace first, and declare every backend/mode cell; remove migrated debt from {MATRIX_SYMMETRY_BASELINE}"
307        ));
308    }
309}

311fn validate_front_door(repo_root: &Path, documents: &[Value]) {
312    let baseline_path = repo_root.join(MATRIX_SYMMETRY_BASELINE);
313    let baseline_text = std::fs::read_to_string(&baseline_path)
314        .unwrap_or_else(|error| die(format!("cannot read {}: {error}", baseline_path.display())));
315    let baseline: JsonValue = serde_json::from_str(&baseline_text).unwrap_or_else(|error| {
316        die(format!(
317            "{}: invalid JSON: {error}",
318            baseline_path.display()
319        ))
320    });
321    let baseline_keys: BTreeSet<_> = baseline
322        .as_object()
323        .unwrap_or_else(|| die(format!("{MATRIX_SYMMETRY_BASELINE}: expected an object")))
324        .keys()
325        .map(String::as_str)
326        .collect();
327    let expected_keys: BTreeSet<_> = [
328        "schema",
329        "asymmetric_manifest_tests",
330        "backend_private_guest_files",
331    ]
332    .into_iter()
333    .collect();
334    if baseline_keys != expected_keys {
335        die(format!(
336            "{MATRIX_SYMMETRY_BASELINE}: keys must be exactly {expected_keys:?}, got {baseline_keys:?}"
337        ));
338    }
339    if baseline.get("schema").and_then(JsonValue::as_u64) != Some(1) {
340        die(format!("{MATRIX_SYMMETRY_BASELINE}: schema must be 1"));
341    }
342    let expected_asymmetric = json_string_set(
343        &baseline,
344        "asymmetric_manifest_tests",
345        MATRIX_SYMMETRY_BASELINE,
346    );
347    let expected_private = json_string_set(
348        &baseline,
349        "backend_private_guest_files",
350        MATRIX_SYMMETRY_BASELINE,
351    );
352
353    let inventory_path = repo_root.join(TEST_INVENTORY);
354    let inventory_text = std::fs::read_to_string(&inventory_path)
355        .unwrap_or_else(|error| die(format!("cannot read {}: {error}", inventory_path.display())));
356    let inventory: JsonValue = serde_json::from_str(&inventory_text).unwrap_or_else(|error| {
357        die(format!(
358            "{}: invalid JSON: {error}",
359            inventory_path.display()
360        ))
361    });
362
363    enforce_exact_ratchet(
364        "manifest ptrace-front-door debt",
365        &asymmetric_manifest_tests(documents),
366        &expected_asymmetric,
367    );
368    enforce_exact_ratchet(
369        "backend-private guest debt",
370        &backend_private_guest_files(&inventory),
371        &expected_private,
372    );
373}

375fn required_string<'a>(value: &'a Value, key: &str, location: &str) -> &'a str {
376    value
377        .get(key)
378        .and_then(Value::as_str)
379        .filter(|value| !value.is_empty())
380        .unwrap_or_else(|| die(format!("{location}: missing non-empty string `{key}`")))
381}
382
383fn string_array(value: Option<&Value>, location: &str) -> Vec<String> {
384    value
385        .and_then(Value::as_array)
386        .unwrap_or_else(|| die(format!("{location}: expected an array")))
387        .iter()
388        .map(|item| {
389            item.as_str()
390                .filter(|item| !item.is_empty())
391                .map(str::to_string)
392                .unwrap_or_else(|| die(format!("{location}: array values must be strings")))
393        })
394        .collect()
395}

397fn validate_direct(value: &Value, id: &str) {
398    match value {
399        Value::String(command) if !command.trim().is_empty() => {}
400        Value::String(_) => die(format!("{id}: direct command must not be empty")),
401        Value::Array(_) => {
402            if string_array(Some(value), &format!("{id}.direct")).is_empty() {
403                die(format!("{id}: direct argv must not be empty"));
404            }
405        }
406        _ => die(format!(
407            "{id}: direct must be a shell command string or an argv array"
408        )),
409    }
410}

412fn ensure_keys(value: &Value, allowed: &[&str], location: &str) {
413    let table = value
414        .as_table()
415        .unwrap_or_else(|| die(format!("{location}: expected a table")));
416    let allowed: BTreeSet<_> = allowed.iter().copied().collect();
417    let actual: BTreeSet<_> = table.keys().map(String::as_str).collect();
418    let unknown: Vec<_> = actual.difference(&allowed).copied().collect();
419    if !unknown.is_empty() {
420        die(format!("{location}: unknown keys: {unknown:?}"));
421    }
422}

424fn is_file_or_symlink(path: &Path) -> bool {
425    path.is_file()
426        || std::fs::symlink_metadata(path).is_ok_and(|metadata| metadata.file_type().is_symlink())
427}

429#[allow(clippy::too_many_arguments)]
430fn validate_and_expand(
431    test: &Value,
432    bucket: &str,
433    location: &str,
434    repo_root: &Path,
435    seen_ids: &mut BTreeSet<String>,
436    seen_programs: &mut BTreeSet<String>,
437    rows: &mut Vec<PlanRow>,
438) {
439    let id = required_string(test, "id", location).to_string();
440    ensure_keys(
441        test,
442        &[
443            "id",
444            "description",
445            "lane",
446            "requires",
447            "timeout_seconds",
448            "occasional",
449            "program",
450            "direct",
451            "observation",
452            "build",
453            "modes",
454            "slow_reason",
455            "preprocessors",
456        ],
457        &id,
458    );
459    if !id.starts_with(&format!("{bucket}/"))
460        || !id.chars().all(|character| {
461            character.is_ascii_lowercase() || character.is_ascii_digit() || "-/".contains(character)
462        })
463        || id
464            .strip_prefix(&format!("{bucket}/"))
465            .is_none_or(|suffix| suffix.is_empty() || suffix.starts_with('-'))
466    {
467        die(format!(
468            "{location}: id `{id}` must be lowercase and start with `{bucket}/`"
469        ));
470    }
471    if !seen_ids.insert(id.clone()) {
472        die(format!("duplicate test id across manifests: {id}"));
473    }
474    required_string(test, "description", &id);
475
476    let lane = required_string(test, "lane", &id);
477    if lane != "portable" && lane != "privileged" {
478        die(format!(
479            "{id}: lane must be portable|privileged, got `{lane}`"
480        ));
481    }
482    match test.get("timeout_seconds").and_then(Value::as_integer) {
483        Some(timeout) if (1..=1800).contains(&timeout) => {}
484        other => die(format!(
485            "{id}: timeout_seconds must be 1..=1800, got {other:?}"
486        )),
487    }
488    if test.get("occasional").and_then(Value::as_bool).is_none() {
489        die(format!("{id}: occasional must be a boolean"));
490    }
491    let _requires = string_array(test.get("requires"), &format!("{id}.requires"));
492
493    let program = test.get("program").and_then(Value::as_str);
494    let direct = test.get("direct");
495    let mut program_path = None;
496    match (program, direct) {
497        (Some(_), Some(_)) => die(format!("{id}: set only one of `program`/`direct`")),
498        (None, None) => die(format!("{id}: must set `program` or `direct`")),
499        (Some(program), None) => {
500            let extension = Path::new(program)
501                .extension()
502                .and_then(|extension| extension.to_str())
503                .unwrap_or("");
504            if !["sh", "c", "rs"].contains(&extension) {
505                die(format!("{id}: program `{program}` must end in .sh/.c/.rs"));
506            }
507            if !program.starts_with("tests/") || program.split('/').any(|part| part == "..") {
508                die(format!(
509                    "{id}: program must be a repo-relative path below tests/: {program}"
510                ));
511            }
512            let path = repo_root.join(program);
513            if !is_file_or_symlink(&path) {
514                die(format!("{id}: program path does not exist: {program}"));
515            }
516            program_path = Some(path);
517            if !seen_programs.insert(program.to_string()) {
518                die(format!(
519                    "program appears in multiple manifest tests: {program}"
520                ));
521            }
522        }
523        (None, Some(direct)) => validate_direct(direct, &id),
524    }
525
526    if let Some(build) = test.get("build") {
527        ensure_keys(build, &["cflags", "rustflags"], &format!("{id}.build"));
528        for key in ["cflags", "rustflags"] {
529            if build.get(key).is_some() {
530                let _flags = string_array(build.get(key), &format!("{id}.build.{key}"));
531            }
532        }
533    }
534    if let Some(reason) = test.get("slow_reason") {
535        if reason.as_str().is_none_or(str::is_empty) {
536            die(format!("{id}: slow_reason must be a non-empty string"));
537        }
538    }
539    if let Some(preprocessors) = test.get("preprocessors") {
540        let preprocessors = string_array(Some(preprocessors), &format!("{id}.preprocessors"));
541        if preprocessors.iter().any(|value| value != "e9patch") {
542            die(format!("{id}: the only supported preprocessor is e9patch"));
543        }
544    }
545
546    validate_observation(test, &id);
547
548    let modes = test
549        .get("modes")
550        .and_then(Value::as_table)
551        .unwrap_or_else(|| die(format!("{id}: missing [test.modes]")));
552    let actual_modes: BTreeSet<_> = modes.keys().map(String::as_str).collect();
553    let expected_modes: BTreeSet<_> = MODES.into_iter().collect();
554    if actual_modes != expected_modes {
555        die(format!(
556            "{id}: modes must be exactly {:?}, got {:?}",
557            expected_modes, actual_modes
558        ));
559    }
560
561    let row_start = rows.len();
562    for mode in MODES {
563        validate_mode(&id, bucket, lane, mode, modes.get(mode).unwrap(), rows);
564    }
565    if rows[row_start..].iter().any(|row| row.ci)
566        && program_path.as_ref().is_some_and(|path| !path.is_file())
567    {
568        die(format!(
569            "{id}: CI-enabled program symlink target is unavailable: {}",
570            program.unwrap()
571        ));
572    }
573}

575fn validate_observation(test: &Value, id: &str) {
576    let observation_value = test
577        .get("observation")
578        .unwrap_or_else(|| die(format!("{id}: observation must be a table")));
579    ensure_keys(
580        observation_value,
581        &["status", "stdout", "stderr", "artifacts"],
582        &format!("{id}.observation"),
583    );
584    let observation = observation_value.as_table().unwrap();
585    for key in ["status", "stdout", "stderr"] {
586        if observation.get(key).and_then(Value::as_bool).is_none() {
587            die(format!("{id}: observation.{key} must be a boolean"));
588        }
589    }
590    for artifact in string_array(
591        observation.get("artifacts"),
592        &format!("{id}.observation.artifacts"),
593    ) {
594        if artifact.starts_with('/') || artifact.split('/').any(|part| part == "..") {
595            die(format!(
596                "{id}: observation artifact must stay below E2E_TMPDIR: {artifact}"
597            ));
598        }
599    }
600}

602fn validate_mode(
603    id: &str,
604    bucket: &str,
605    lane: &str,
606    mode: &str,
607    spec: &Value,
608    rows: &mut Vec<PlanRow>,
609) {
610    let spec_value = spec;
611    let spec = spec_value
612        .as_table()
613        .unwrap_or_else(|| die(format!("{id}: modes.{mode} must be a table")));
614    let mut allowed = vec!["ci", "backends_enabled", "backends_disabled", "guest_args"];
615    match mode {
616        "naked" => allowed.extend(["runs", "assert"]),
617        "chaos" => allowed.extend(["seeds", "assert"]),
618        "custom" => allowed.extend(["args", "assert"]),
619        _ => {}
620    }
621    ensure_keys(spec_value, &allowed, &format!("{id}.modes.{mode}"));
622    let ci = spec
623        .get("ci")
624        .and_then(Value::as_bool)
625        .unwrap_or_else(|| die(format!("{id}: modes.{mode}.ci must be a boolean")));
626    let enabled = string_array(
627        spec.get("backends_enabled"),
628        &format!("{id}.modes.{mode}.backends_enabled"),
629    );
630    let disabled = spec
631        .get("backends_disabled")
632        .and_then(Value::as_table)
633        .unwrap_or_else(|| {
634            die(format!(
635                "{id}: modes.{mode}.backends_disabled must be a table"
636            ))
637        });
638    for (backend, reason) in disabled {
639        if reason.as_str().is_none_or(str::is_empty) {
640            die(format!(
641                "{id}: modes.{mode}.backends_disabled.{backend} needs a reason"
642            ));
643        }
644    }
645
646    let expected: BTreeSet<&str> = if mode == "naked" {
647        ["native"].into_iter().collect()
648    } else {
649        KNOWN_BACKENDS.into_iter().collect()
650    };
651    let enabled_set: BTreeSet<&str> = enabled.iter().map(String::as_str).collect();
652    let disabled_set: BTreeSet<&str> = disabled.keys().map(String::as_str).collect();
653    if enabled_set.len() != enabled.len()
654        || !enabled_set.is_disjoint(&disabled_set)
655        || enabled_set
656            .union(&disabled_set)
657            .copied()
658            .collect::<BTreeSet<_>>()
659            != expected
660    {
661        die(format!(
662            "{id}: modes.{mode} must partition {:?}; enabled={enabled_set:?}, disabled={disabled_set:?}",
663            expected
664        ));
665    }
666    if let Some(guest_args) = spec.get("guest_args") {
667        let guest_args = guest_args
668            .as_table()
669            .unwrap_or_else(|| die(format!("{id}: modes.{mode}.guest_args must be a table")));
670        for (backend, args) in guest_args {
671            if !enabled_set.contains(backend.as_str()) {
672                die(format!(
673                    "{id}: modes.{mode}.guest_args.{backend} names a backend that is not enabled"
674                ));
675            }
676            if string_array(
677                Some(args),
678                &format!("{id}.modes.{mode}.guest_args.{backend}"),
679            )
680            .is_empty()
681            {
682                die(format!(
683                    "{id}: modes.{mode}.guest_args.{backend} must contain at least one argument"
684                ));
685            }
686        }
687    }
688    if ci && enabled.is_empty() {
689        die(format!(
690            "{id}: modes.{mode} is CI-enabled but has no backend"
691        ));
692    }
693    if mode == "naked" && ci {
694        die(format!(
695            "{id}: naked is opt-in meta-CI and must set ci=false"
696        ));
697    }
698    if mode == "replay" && enabled.iter().any(|backend| backend != "ptrace") {
699        die(format!("{id}: replay is ptrace-only, got {enabled:?}"));
700    }
701
702    if mode == "naked" && !enabled.is_empty() {
703        let runs = spec
704            .get("runs")
705            .and_then(Value::as_integer)
706            .unwrap_or_else(|| die(format!("{id}: enabled naked mode requires runs")));
707        if !(3..=5).contains(&runs) {
708            die(format!("{id}: naked.runs must be 3..=5, got {runs}"));
709        }
710        let assertions = spec
711            .get("assert")
712            .and_then(Value::as_table)
713            .unwrap_or_else(|| die(format!("{id}: naked.assert must be a table")));
714        ensure_keys(
715            spec.get("assert").unwrap(),
716            &["min_distinct"],
717            &format!("{id}.modes.naked.assert"),
718        );
719        let min_distinct = assertions
720            .get("min_distinct")
721            .and_then(Value::as_integer)
722            .unwrap_or_else(|| die(format!("{id}: naked.assert.min_distinct is required")));
723        if !(2..=runs).contains(&min_distinct) {
724            die(format!(
725                "{id}: naked.assert.min_distinct must be 2..={runs}, got {min_distinct}"
726            ));
727        }
728    }
729    if mode == "chaos" && !enabled.is_empty() {
730        let seeds = spec
731            .get("seeds")
732            .and_then(Value::as_array)
733            .unwrap_or_else(|| die(format!("{id}: enabled chaos mode requires seeds")));
734        let unique: BTreeSet<_> = seeds.iter().filter_map(Value::as_integer).collect();
735        if seeds.len() < 2 || unique.len() != seeds.len() {
736            die(format!(
737                "{id}: chaos seeds must contain at least two unique integers"
738            ));
739        }
740        let assertions = spec
741            .get("assert")
742            .and_then(Value::as_table)
743            .unwrap_or_else(|| die(format!("{id}: enabled chaos mode requires assert")));
744        ensure_keys(
745            spec.get("assert").unwrap(),
746            &["min_distinct", "min_passes", "min_failures"],
747            &format!("{id}.modes.chaos.assert"),
748        );
749        for key in ["min_distinct", "min_passes", "min_failures"] {
750            match assertions.get(key).and_then(Value::as_integer) {
751                Some(value) if value >= 0 && (key != "min_distinct" || value >= 2) => {}
752                other => die(format!(
753                    "{id}: chaos.assert.{key} has invalid value {other:?}"
754                )),
755            }
756        }
757    }
758    if mode == "custom" && !enabled.is_empty() {
759        let args = string_array(spec.get("args"), &format!("{id}.modes.custom.args"));
760        if args.is_empty() {
761            die(format!("{id}: enabled custom mode requires args"));
762        }
763        let assertions = spec
764            .get("assert")
765            .and_then(Value::as_table)
766            .unwrap_or_else(|| die(format!("{id}: enabled custom mode requires assert")));
767        ensure_keys(
768            spec.get("assert").unwrap(),
769            &["runs", "repeat_identical"],
770            &format!("{id}.modes.custom.assert"),
771        );
772        let runs = assertions
773            .get("runs")
774            .and_then(Value::as_integer)
775            .unwrap_or_else(|| die(format!("{id}: custom.assert.runs is required")));
776        if !(3..=5).contains(&runs)
777            || assertions.get("repeat_identical").and_then(Value::as_bool) != Some(true)
778        {
779            die(format!(
780                "{id}: custom must require 3..=5 runs with repeat_identical=true"
781            ));
782        }
783    }
784
785    for backend in enabled {
786        rows.push(PlanRow {
787            bucket: bucket.to_string(),
788            id: id.to_string(),
789            lane: lane.to_string(),
790            mode: mode.to_string(),
791            backend,
792            ci,
793        });
794    }
795}
