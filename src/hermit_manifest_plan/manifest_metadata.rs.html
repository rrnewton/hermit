107pub fn build_export(root: &Path) -> Result<ManifestMetadata, String> {

109    build_export_from_inputs(root, inputs, || ManifestInputs::read(root))
110}

112fn build_export_from_inputs(
113    root: &Path,
114    inputs: ManifestInputs,
115    read_current: impl FnOnce() -> Result<ManifestInputs, String>,
116) -> Result<ManifestMetadata, String> {
117    let manifest_sha256_before = manifest_sha256(&inputs)?;
118    let manifests = ManifestSet::from_inputs(root, &inputs)?;
119    let selected_by_full_cells = manifests.select(&Selection {
120        population: Some(Population::Required),
121        ..Selection::default()
122    })?;
123    let selected_by_full_ids = selected_by_full_cells
124        .iter()
125        .map(|cell| cell.id.clone())
126        .collect::<std::collections::BTreeSet<_>>();
127    let tests = manifests
128        .all_tests()
129        .map(|(category, _, _, test)| test_metadata(category, test))
130        .collect();
131
132    let mut cells = Vec::new();
133    // Keep both source populations: selection by full validation is a separate
134    // fact from whether a comparable cell is present in the manifest.
135    for population in [Population::Enabled, Population::Disabled] {
136        let selected = manifests.select(&Selection {
137            population: Some(population),
138            include_occasional: true,
139            include_manual: true,
140            ..Selection::default()
141        })?;
142        for cell in selected {
143            if cell.id.mode != "custom" {
144                cells.push(cell_metadata(
145                    &cell,
146                    selected_by_full_ids.contains(&cell.id),
147                )?);
148            }
149        }
150    }
151    sort_and_require_unique_cells("comparable cells in the manifest", &mut cells)?;
152
153    let mut selected_by_full_custom_commands = selected_by_full_cells
154        .into_iter()
155        .filter(|cell| cell.id.mode == "custom")
156        .map(|cell| cell_metadata(&cell, true))
157        .collect::<Result<Vec<_>, _>>()?;
158    sort_and_require_unique_cells(
159        "custom commands selected by full validation",
160        &mut selected_by_full_custom_commands,
161    )?;
162
163    let manifest_sha256 =
164        require_stable_manifest_sha(manifest_sha256_before, manifest_sha256(&read_current()?)?)?;
165
166    Ok(ManifestMetadata {
167        schema: EXPORT_SCHEMA,
168        manifest_schema: MANIFEST_SCHEMA,
169        manifest_sha256,
170        tests,
171        cells,
172        selected_by_full_custom_commands,
173    })
174}

176fn require_stable_manifest_sha(before: String, after: String) -> Result<String, String> {
177    if before == after {
178        Ok(before)
179    } else {
180        Err(format!(
181            "manifest inputs changed while they were being read: before={before}, after={after}"
182        ))
183    }
184}

186fn test_metadata(category: &str, test: &TestRecipe) -> TestMetadata {
187    let direct = test.direct.as_ref().map(|direct| match direct {
188        DirectCommand::Shell(command) => DirectMetadata::Shell {
189            command: command.clone(),
190        },
191        DirectCommand::Argv(argv) => DirectMetadata::Argv { argv: argv.clone() },
192    });
193    let build = test.build.as_ref().map(|build| BuildMetadata {
194        cflags: build.cflags.clone(),
195        rustflags: build.rustflags.clone(),
196    });
197    TestMetadata {
198        id: test.id.clone(),
199        description: test.description.clone(),
200        category: category.to_string(),
201        lane: test.lane.clone(),
202        requires: test.requires.clone(),
203        occasional: test.occasional,
204        program: test.program.clone(),
205        direct,
206        build,
207        observation: ObservationMetadata {
208            status: test.observation.status,
209            stdout: test.observation.stdout,
210            stderr: test.observation.stderr,
211            artifacts: test.observation.artifacts.clone(),
212        },
213        preprocessors: test.preprocessors.clone(),
214    }
215}

217fn cell_metadata(cell: &SelectedCell, selected_by_full: bool) -> Result<CellMetadata, String> {
218    let backend = cell.id.backend.as_deref().unwrap_or("native").to_string();
219    let recipe = cell
220        .test
221        .modes
222        .get(&cell.id.mode)
223        .ok_or_else(|| format!("{} has no {} mode", cell.id.test, cell.id.mode))?;
224    let configured_selection = configured_selection(recipe)
225        .map_err(|error| format!("{}: {} {error}", cell.id.test, cell.id.mode))?;
226    if selected_by_full && !cell.enabled {
227        return Err(format!(
228            "{}/{}@{} is selected by full validation but is not applicable",
229            cell.id.test, cell.id.mode, backend
230        ));
231    }
232    if selected_by_full && !configured_selection.selected(&backend) {
233        return Err(format!(
234            "{}/{}@{} is selected by full validation but its manifest selection is false",
235            cell.id.test, cell.id.mode, backend
236        ));
237    }
238    if selected_by_full && cell.test.occasional {
239        return Err(format!(
240            "{}/{}@{} is selected by full validation but its test is marked occasional",
241            cell.id.test, cell.id.mode, backend
242        ));
243    }
244
245    let (not_selected_by_full_reason, not_applicable_reason) = if cell.enabled {
246        let reason = if selected_by_full {
247            None
248        } else if let Some(reason) = configured_selection.reason(&backend) {
249            Some(reason.clone())
250        } else if cell.test.occasional {
251            Some(CiDisabledReasonData {
252                result: None,
253                evidence: None,
254                reason: "This test is marked occasional, and full validation does not select occasional tests."
255                    .to_string(),
256            })
257        } else {
258            return Err(format!(
259                "{}/{}@{} is not selected by full validation without a reason",
260                cell.id.test, cell.id.mode, backend
261            ));
262        };
263        (reason, None)
264    } else {
265        (
266            None,
267            Some(
268                recipe
269                    .backends_disabled
270                    .get(&backend)
271                    .cloned()
272                    .ok_or_else(|| {
273                        format!(
274                            "{}/{}@{} is not applicable without a reason",
275                            cell.id.test, cell.id.mode, backend
276                        )
277                    })?,
278            ),
279        )
280    };
281    let (current_reproducer, current_reproducer_unavailable_reason) =
282        exact_cell_reproducer(&cell.id.test, &cell.id.mode, &backend, cell.enabled);
283
284    Ok(CellMetadata {
285        test: cell.id.test.clone(),
286        category: cell.category.clone(),
287        lane: cell.test.lane.clone(),
288        mode: cell.id.mode.clone(),
289        backend: backend.clone(),
290        selected_by_full,
291        not_selected_by_full_reason,
292        not_applicable_reason,
293        timeout_seconds: cell.timeout_seconds,
294        cpu_timeout_seconds: cell.cpu_timeout_seconds,
295        guest_args: recipe.guest_args.get(&backend).cloned().unwrap_or_default(),
296        workdir: recipe.workdir.clone(),
297        current_reproducer,
298        current_reproducer_unavailable_reason,
299    })
300}

311fn exact_cell_reproducer(
312    test: &str,
313    mode: &str,
314    backend: &str,
315    applicable: bool,
316) -> (Option<CurrentReproducer>, Option<String>) {
317    if !applicable && mode == "naked" && backend == "native" {
318        return (
319            None,
320            Some(
321                "test-harness --probe-disabled requires --backend, but its backend selector does not accept native"
322                    .to_string(),
323            ),
324        );
325    }
326
327    let mut argv = vec![
328        "target/debug/test-harness".to_string(),
329        "run".to_string(),
330        if applicable {
331            "--include-manual".to_string()
332        } else {
333            "--probe-disabled".to_string()
334        },
335        "--include-occasional".to_string(),
336        "--test".to_string(),
337        test.to_string(),
338        "--mode".to_string(),
339        mode.to_string(),
340    ];
341    if backend != "native" {
342        argv.extend(["--backend".to_string(), backend.to_string()]);
343    }
344    let shell_command = argv
345        .iter()
346        .map(|argument| shell_quote(argument))
347        .collect::<Vec<_>>()
348        .join(" ");
349    (
350        Some(CurrentReproducer {
351            argv,
352            shell_command,
353        }),
354        None,
355    )
356}

358fn shell_quote(value: &str) -> String {
359    if !value.is_empty()
360        && value
361            .bytes()
362            .all(|byte| byte.is_ascii_alphanumeric() || b"_@%+=:,./-".contains(&byte))
363    {
364        value.to_string()
365    } else {
366        format!("'{}'", value.replace('\'', "'\"'\"'"))
367    }
368}

370fn sort_and_require_unique_cells(label: &str, cells: &mut [CellMetadata]) -> Result<(), String> {
371    cells.sort_by(|left, right| cell_key(left).cmp(&cell_key(right)));
372    for pair in cells.windows(2) {
373        if cell_key(&pair[0]) == cell_key(&pair[1]) {
374            return Err(format!(
375                "{label} contains duplicate identity {}/{}/{}@{}",
376                pair[0].category, pair[0].test, pair[0].mode, pair[0].backend
377            ));
378        }
379    }
380    Ok(())
381}

393fn manifest_sha256(inputs: &ManifestInputs) -> Result<String, String> {
394    let mut digest = Sha256::new();
395    digest.update(b"hermit-manifest-metadata-v1\0");
396    for (path, source) in inputs.named_sources() {
397        let name = path
398            .file_name()
399            .and_then(|name| name.to_str())
400            .ok_or_else(|| format!("manifest name is not UTF-8: {}", path.display()))?;
401        let contents = source.as_bytes();
402        digest.update(
403            u64::try_from(name.len())
404                .map_err(|_| "manifest name length does not fit u64".to_string())?
405                .to_le_bytes(),
406        );
407        digest.update(name.as_bytes());
408        digest.update(
409            u64::try_from(contents.len())
410                .map_err(|_| "manifest length does not fit u64".to_string())?
411                .to_le_bytes(),
412        );
413        digest.update(contents);
414    }
415    Ok(format!("{:x}", digest.finalize()))
416}

435    // Inspect absolute path references within field values, not substrings of
436    // relative paths such as hermit-cli/src/bin/hermit/run.rs. Delimiters also
437    // cover quoted prose/shell paths, flag assignments, linker comma arguments
438    // and file:line citations. Recognize the finite compiler path options below
439    // only when the option itself begins a token; a relative path containing
440    // -I/src is not one. This does not parse arbitrary shell/compiler syntax.
441    fn starts_path_token(before: &str) -> bool {
442        before.chars().next_back().is_none_or(|c| {
443            c.is_whitespace() || matches!(c, '\'' | '"' | '`' | '=' | ':' | ',' | '(' | '[' | '{')
444        })
445    }
446
447    fn contains_absolute_root_reference(value: &Value, root: &str) -> bool {
448        match value {
449            Value::String(text) => text.match_indices(root).any(|(offset, _)| {
450                let before = &text[..offset];
451                let after = text[offset + root.len()..].chars().next();
452                let starts_path = starts_path_token(before)
453                    || [
454                        "-I",
455                        "-L",
456                        "-B",
457                        "-isystem",
458                        "-iquote",
459                        "-idirafter",
460                        "-include",
461                        "-imacros",
462                        "-isysroot",
463                        "-iprefix",
464                        "-o",
465                        "-MF",
466                    ]
467                    .iter()
468                    .any(|&option| before.strip_suffix(option).is_some_and(starts_path_token));
469                let ends_component = after.is_none_or(|c| {
470                    c.is_whitespace()
471                        || matches!(
472                            c,
473                            '/' | '\'' | '"' | '`' | '=' | ':' | ',' | ';' | ')' | ']' | '}'
474                        )
475                });
476                starts_path && ends_component
477            }),
478            Value::Array(values) => values
479                .iter()
480                .any(|value| contains_absolute_root_reference(value, root)),
481            Value::Object(fields) => fields
482                .values()
483                .any(|value| contains_absolute_root_reference(value, root)),
484            _ => false,
485        }
486    }

488    fn require_root_independent_exports(exports: &[String], roots: &[&Path]) -> Result<(), String> {
489        let first = exports.first().ok_or("missing exported metadata")?;
490        if exports.iter().any(|export| export != first) {
491            return Err("exports differ across repository roots".into());
492        }
493        let value: Value = serde_json::from_str(first).unwrap();
494        for root in roots {
495            assert!(root.is_absolute());
496            if contains_absolute_root_reference(&value, root.to_str().unwrap()) {
497                return Err(format!(
498                    "absolute checkout-root reference: {}",
499                    root.display()
500                ));
501            }
502        }
503        Ok(())
504    }

611    struct TemporaryManifestRoot(PathBuf);

613    impl TemporaryManifestRoot {
614        fn alias_of(root: &Path, name: &str) -> Self {
615            let nonce = std::time::SystemTime::now()
616                .duration_since(std::time::UNIX_EPOCH)
617                .unwrap()
618                .as_nanos();
619            let alias = std::env::temp_dir().join(format!(
620                "hermit-manifest-metadata-{name}-{}-{nonce}",
621                std::process::id()
622            ));
623            fs::create_dir(&alias).unwrap();
624            let alias = Self(alias);
625            std::os::unix::fs::symlink(root.join("tests"), alias.path().join("tests")).unwrap();
626            alias
627        }
628
629        fn new(name: &str, manifest: &str) -> Self {
630            let nonce = std::time::SystemTime::now()
631                .duration_since(std::time::UNIX_EPOCH)
632                .unwrap()
633                .as_nanos();
634            let root = std::env::temp_dir().join(format!(
635                "hermit-manifest-metadata-{name}-{}-{nonce}",
636                std::process::id()
637            ));
638            let directory = root.join("tests/e2e/manifests");
639            fs::create_dir_all(&directory).unwrap();
640            fs::write(
641                directory.join("defaults.yaml"),
642                "schema: 3\ntimeout_seconds: 15\ncpu_timeout_seconds: 7\nnextest: []\n",
643            )
644            .unwrap();
645            fs::write(directory.join(format!("{name}.yaml")), manifest).unwrap();
646            Self(root)
647        }
648
649        fn path(&self) -> &Path {
650            &self.0
651        }
652    }
