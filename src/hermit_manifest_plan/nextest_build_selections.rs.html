524/// Recover the exact authored payload from the one supported execution wrapper.
525/// Both the generator and its preparation audit inspect the same command bytes
526/// the container's final bash executes, including literal shell quoting.
527pub(super) fn execution_command(step: &dagrun::model::Step) -> Result<String, String> {
528    if !step.cmd.starts_with("./ci/hermetic/run-in-pinned-root.sh ") {
529        return Ok(step.cmd.clone());
530    }
531    let args = shell_words::split(&step.cmd).map_err(|error| format!("{}: {error}", step.tag()))?;
532    let boundary = args
533        .iter()
534        .position(|arg| arg == "--")
535        .ok_or_else(|| format!("{} omits its pinned-root command boundary", step.tag()))?;
536    match &args[boundary + 1..] {
537        [shell, option, guard, argv0, payload]
538            if shell == "bash"
539                && option == "-c"
540                && argv0 == "bash"
541                && (guard == crate::validation_dag::PINNED_ROOT_COMMAND_GUARD
542                    || guard == crate::validation_dag::LEGACY_PINNED_ROOT_COMMAND_GUARD) =>
543        {
544            Ok(payload.clone())
545        }
546        _ => Err(format!(
547            "{} has an unrecognized pinned-root command",
548            step.tag()
549        )),
550    }
551}

553fn command_arguments(command: &str, marker: &str) -> Result<Vec<String>, String> {
554    let (_, tail) = command
555        .split_once(marker)
556        .ok_or_else(|| format!("missing {marker}"))?;
557    let tail = tail.replace("${CI:+--profile ci}", "");
558    let mut quote = None;
559    let mut escape = false;
560    let mut end = tail.len();
561    for (offset, ch) in tail.char_indices() {
562        if escape {
563            escape = false;
564            continue;
565        }
566        if ch == '\\' && quote != Some('\'') {
567            escape = true;
568            continue;
569        }
570        if let Some(active) = quote {
571            if ch == active {
572                quote = None;
573            }
574        } else if ch == '\'' || ch == '"' {
575            quote = Some(ch);
576        } else if matches!(ch, ';' | '&' | '|' | '>' | '<') {
577            end = offset;
578            break;
579        }
580    }
581    let args = shell_words::split(&tail[..end])
582        .map_err(|e| format!("cannot parse authored Nextest arguments: {e}"))?;
583    Ok(args)
584}

586pub(super) fn assert_command_selection(step: &dagrun::model::Step) -> Result<(), String> {
587    use crate::nextest_binaries::REQUIRED_ENV;
588    use crate::nextest_binaries::SELECTION_ENV;
589    use crate::nextest_binaries::split_arguments;
590    let tag = step.tag();
591    let command = execution_command(step)?;
592    let raw = step
593        .env
594        .get(SELECTION_ENV)
595        .ok_or_else(|| format!("{tag} has no prepared build selection"))?;
596    let expected: Vec<String> = serde_json::from_str(raw).map_err(|e| format!("{tag}: {e}"))?;
597    if step.env.get(REQUIRED_ENV).map(String::as_str) != Some("1") {
598        return Err(format!(
599            "{tag} does not require prepared Nextest executables"
600        ));
601    }
602    for marker in ["run-nextest-counted.sh", "nextest-binaries.rs list"] {
603        if !command.contains(marker) {
604            continue;
605        }
606        let mut arguments = command_arguments(&command, marker)?;
607        if marker == "run-nextest-counted.sh" {
608            match arguments.first().map(String::as_str) {
609                Some("--calibration-host") => {
610                    arguments.remove(0);
611                }
612                Some("--calibration-launch-proof") => {
613                    if arguments.get(1).map(String::as_str)
614                        != Some(crate::nextest_cohort::PINNED_PROOF_PATH)
615                    {
616                        return Err(format!("{tag} has an unsupported calibration launch proof"));
617                    }
618                    arguments.drain(..2);
619                }
620                _ => {}
621            }
622        }
623        let parsed = split_arguments(&arguments).map_err(|error| format!("{tag}: {error}"))?;
624        if parsed.build != expected {
625            return Err(format!(
626                "{tag} command has Cargo selection {:?}, declared {expected:?}",
627                parsed.build
628            ));
629        }
630    }
631    let direct = "nextest-binaries.rs executable ";
632    if let Some((_, rest)) = command.split_once(direct) {
633        if command.matches(direct).count() != 1 {
634            return Err(format!("{tag} has ambiguous prepared executable lookups"));
635        }
636        let arguments = rest
637            .split_once(')')
638            .ok_or_else(|| format!("{tag} has no executable lookup boundary"))?
639            .0;
640        let arguments = shell_words::split(arguments).map_err(|e| format!("{tag}: {e}"))?;
641        if arguments.len() != 2 {
642            return Err(format!(
643                "{tag} must name one prepared package and test target"
644            ));
645        }
646        let actual = vec![
647            "-p".to_string(),
648            arguments[0].clone(),
649            "--test".to_string(),
650            arguments[1].clone(),
651        ];
652        if actual != expected {
653            return Err(format!(
654                "{tag} executable lookup selects {actual:?}, declared {expected:?}"
655            ));
656        }
657    }
658    if command.contains("cargo nextest list") || command.contains("cargo nextest run") {
659        return Err(format!("{tag} can bypass prepared metadata and compile"));
660    }
661    Ok(())
662}

664// `prepare` is one fixed command; only its fixed, profile-matched packaging
665// finalizer may follow it. Never truncate an arbitrary shell command at `&&`.
666fn preparation_profile<'a>(tag: &str, command: &'a str) -> Result<Option<&'a str>, String> {
667    let marker = "./ci/nextest-binaries.rs prepare ";
668    let Some((_, rest)) = command.split_once(marker) else {
669        return Ok(None);
670    };
671    if command.matches(marker).count() != 1 {
672        return Err(format!("{tag} has duplicate prepared commands"));
673    }
674    let profile = rest.split_whitespace().next().unwrap_or("");
675    let finalized = crate::nextest_binaries::cargo_profile_for(profile).map(|directory| {
676        format!("{profile} && target/{directory}/stage-detcore-liteinst --git target")
677    });
678    if profile.is_empty() || (rest != profile && finalized.as_deref() != Some(rest)) {
679        return Err(format!(
680            "{tag} has an ambiguous prepared profile or finalizer"
681        ));
682    }
683    Ok(Some(profile))
684}

686/// A producer of a unifying profile (`nextest_binaries::unifies`) must end
687/// with that profile's unified workspace build followed by `prepare`. Any other
688/// build -- different features, or a second `-p hermit-run --bin hermit` link --
689/// makes preparation re-resolve features, recompile, and relink the unhashed
690/// executable the tests run. That build also links the Hermit the E2E cells
691/// run, so its features must keep every third-party backend. The pinned-root
692/// producer runs its build in `validation_dag::PINNED_WORKSPACE_COMPILE_TAG`
693/// and its `prepare` in `PINNED_WORKSPACE_PREPARE_TAG`; the pair is judged as
694/// the one payload it rejoins to.
695pub(super) fn assert_producers_build_the_unified_selection(
696    cfg: &dagrun::model::DagConfig,
697) -> Result<(), String> {
698    for step in &cfg.steps {
699        let command = if step.tag() == crate::validation_dag::PINNED_WORKSPACE_PREPARE_TAG {
700            crate::validation_dag::pinned_workspace_producer_payload(cfg)?
701        } else {
702            execution_command(step)?
703        };
704        let Some(profile) = preparation_profile(&step.tag(), &command)? else {
705            continue;
706        };
707        if !crate::nextest_binaries::unifies(profile) {
708            continue;
709        }
710        let expected = crate::nextest_binaries::unified_prebuild_command(cfg, profile)?;
711        if command.matches("--workspace --all-targets").count() != 1
712            || !command.ends_with(&expected)
713            || command.contains("--bin hermit")
714        {
715            return Err(format!(
716                "{} must end with its unified workspace build `{expected}` and build nothing else in between",
717                step.tag()
718            ));
719        }
720        if !expected.contains("hermit-run/third-party-backends") {
721            return Err(format!(
722                "{} links the E2E Hermit without hermit-run/third-party-backends: `{expected}`",
723                step.tag()
724            ));
725        }
726    }
727    Ok(())
728}

730pub(super) fn assert_preparation_dependencies(
731    cfg: &dagrun::model::DagConfig,
732) -> Result<(), String> {
733    use std::collections::BTreeMap;
734    use std::collections::BTreeSet;
735
736    use crate::nextest_binaries::SELECTION_ENV;
737    use crate::nextest_binaries::config_selections;
738    use crate::nextest_binaries::selection_key;
739    let by_tag = cfg
740        .steps
741        .iter()
742        .map(|step| (step.tag(), step))
743        .collect::<BTreeMap<_, _>>();
744    let mut producers = BTreeMap::new();
745    for step in &cfg.steps {
746        let command = execution_command(step)?;
747        if let Some(profile) = preparation_profile(&step.tag(), &command)? {
748            producers.insert(
749                step.tag(),
750                (
751                    step.cmd.starts_with("./ci/hermetic/run-in-pinned-root.sh "),
752                    config_selections(cfg, profile)?,
753                ),
754            );
755        }
756    }
757    for step in &cfg.steps {
758        let Some(raw) = step.env.get(SELECTION_ENV) else {
759            continue;
760        };
761        let args: Vec<String> = serde_json::from_str(raw).map_err(|e| e.to_string())?;
762        let key = selection_key(&args);
763        let mut ancestors = BTreeSet::new();
764        let mut pending = step.deps.clone();
765        while let Some(dependency) = pending.pop() {
766            if ancestors.insert(dependency.clone()) {
767                let ancestor = by_tag
768                    .get(&dependency)
769                    .ok_or_else(|| format!("{} has missing dependency {dependency}", step.tag()))?;
770                pending.extend(ancestor.deps.iter().cloned());
771            }
772        }
773        if !ancestors.iter().any(|tag| {
774            producers.get(tag).is_some_and(|(pinned, selections)| {
775                *pinned == step.cmd.starts_with("./ci/hermetic/run-in-pinned-root.sh ")
776                    && selections.get(&key) == Some(&args)
777            })
778        }) {
779            return Err(format!(
780                "{} can run before any producer in the same filesystem root of its exact Cargo selection {args:?}",
781                step.tag()
782            ));
783        }
784    }
785    Ok(())
786}
