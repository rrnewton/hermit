6/// The verification verdict, independent of the guest's exit status.
7///
8/// A guest that deterministically exits nonzero can still match, while a guest
9/// that exits zero can still diverge. Consumers must read this value rather than
10/// infer the verdict from either the process status or human-readable stderr.
11#[derive(Clone, Copy, Debug, Deserialize, Eq, PartialEq, Serialize)]
12#[serde(rename_all = "snake_case")]
13pub enum Verdict {
14    /// The two runs matched on every compared dimension.
15    Matched,
16    /// The two runs diverged; verification failed.
17    Diverged,
18    /// Verification stopped before it could reach a verdict.
19    NoResult,
20    /// Verification completed enough work to name an understood machine or
21    /// harness failure, so the receipt is neither a product verdict nor an
22    /// unknown no-result.
23    InfrastructureError,
24}
25
26impl std::fmt::Display for Verdict {
27    fn fmt(&self, formatter: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
28        formatter.write_str(match self {
29            Self::Matched => "matched",
30            Self::Diverged => "diverged",
31            Self::NoResult => "no_result",
32            Self::InfrastructureError => "infrastructure_error",
33        })
34    }
35}

37/// Why verification did not reach a verdict.
38///
39/// This distinction is functional evidence. `FirstRunRejected` means the guest
40/// ran and its disposition was rejected by `--verify-allow`; `NotRun` is the
41/// pre-run stamp left when the invocation died before recording a more specific
42/// result. Collapsing both into `no_result` made a completed guest refusal and a
43/// container or launcher failure indistinguishable to the result reader.
44#[derive(Clone, Debug, Deserialize, Eq, PartialEq, Serialize)]
45#[serde(rename_all = "snake_case", tag = "kind")]
46pub enum NoResultReason {
47    /// The invocation did not replace the pre-run stamp with a more specific
48    /// outcome.
49    NotRun,
50    /// Run 1 completed and its exit status was rejected by `--verify-allow`.
51    FirstRunRejected {
52        exit_code: Option<i32>,
53        signal: Option<i32>,
54        stdout_bytes: u64,
55        stderr_bytes: u64,
56    },
57    /// The log comparator refused to claim either agreement or divergence.
58    ComparisonRefused { detail: String },
59}

61/// The understood infrastructure failure recorded by verification.
62#[derive(Clone, Debug, Deserialize, Eq, PartialEq, Serialize)]

65    /// A ptrace PMU timer interrupt arrived after its target. Both runs and

189/// The complete machine-readable report written by `--verify-json`.
190///
191/// This type is shared by the Hermit producer, the manifest runner, the
192/// compatibility pressure test, and the scorecard. The serialized field order
193/// is retained because existing report bytes are hashed as evidence.
194#[derive(Clone, Debug, Deserialize, PartialEq, Serialize)]
195pub struct VerificationReport {
196    pub verified: bool,
197    pub bitwise_parity: bool,
198    pub verdict: Verdict,
199    #[serde(default)]
200    pub no_result_reason: Option<NoResultReason>,
201    #[serde(default)]
202    pub infrastructure_error: Option<InfrastructureError>,
203    /// `None` when the producer reached no verdict. This is a DOCUMENTED
204    /// producer state, not a malformed report: `VerificationReport::no_result()`
205    /// in hermit-cli sets it, and its doc says "`null` when no verdict was
206    /// reached". Declaring it non-optional made a legitimate no-result
207    /// deserialise as "unreadable report", which reported an infrastructure
208    /// fault where the truth was that nothing had been recorded.
209    #[serde(deserialize_with = "present_but_nullable_comparison")]
210    pub comparison: Option<ComparisonReport>,
211    pub compared_log_messages: Option<ComparedLogMessages>,
212    #[serde(default, skip_serializing_if = "Option::is_none")]
213    pub dbt_counted_branches: Option<DbtCountedBranchComparison>,
214    /// Runtime totals for the two compared executions when the producer could
215    /// read both run summaries.
216    #[serde(default, skip_serializing_if = "Option::is_none")]
217    pub runtime: Option<VerificationRuntime>,
218    /// The guest's exit code when an exited status was available to the
219    /// producer. `None` means either that `guest_signal` records a signal
220    /// disposition, or that execution did not return a guest status at all;
221    /// current producers must not leave this null for `ExitStatus::Exited`.
222    #[serde(default)]
223    pub guest_exit_code: Option<i32>,
224    /// The guest's signal when a signaled status was available to the producer.
225    /// When both disposition fields are null, the writer had no guest status.
226    #[serde(default)]
227    pub guest_signal: Option<i32>,
228    /// WHERE the divergence began, in scheduler-turn units. `None` when the
229    /// logs matched, when no comparison ran, or when the report predates the
230    /// field.
231    ///
232    /// `#[serde(default)]` is deliberate here and is the OPPOSITE choice from
233    /// `ComparisonReport::record_envelope` above. That field is an admission
234    /// gate, so a missing value must never be supplied on the producer's
235    /// behalf. This one is a diagnostic position: absent means "not located",
236    /// which is exactly what `None` says. Requiring it would turn every
237    /// retained pre-field report into an unreadable-report ERROR and
238    /// misclassify old evidence as an infrastructure fault -- the same trap
239    /// documented on `comparison` just above.
240    #[serde(default)]
241    pub first_divergent_scheduler_turn: Option<u64>,
242    /// WHERE the divergence began, in virtual-nanosecond units. Same
243    /// tolerant-default rationale as the field above.
244    ///
245    /// NOTE both of these are the position of the PRECEDING scheduler COMMIT,
246    /// so when no COMMIT precedes the differing record they collapse to the
247    /// origin and BOUND the divergence rather than locating it. The field below
248    /// is the one that locates it.
249    #[serde(default)]
250    pub first_divergent_virtual_nanoseconds: Option<u64>,
251    /// 1-based index of the first differing compared record -- the LOCATION of
252    /// the divergence rather than a bound on it, and the only one of the three
253    /// that is a true log prefix. It shares a unit with
254    /// `compared_log_messages`, so `record / compared` is the fraction of the
255    /// log that was deterministic.
256    ///
257    /// `null` and `0` are not the same claim: null is "no divergence located",
258    /// while 0 would mean the very first record differed. Nothing writes 0 --
259    /// the index is 1-based. Same tolerant-default rationale as its siblings.
260    #[serde(default)]
261    pub first_divergent_record: Option<u64>,
262    /// How many syscalls the guest had COMPLETED when the divergence appeared,
263    /// from detcore's own `finish syscall #N` counter.
264    ///
265    /// NOT interchangeable with `first_divergent_record`: one counts guest work
266    /// and the other counts compared log records, and they move at completely
267    /// different rates. Measured on one real divergence: record 98, syscall 37.
268    /// `null` also covers a genuine state -- a run that diverged before any
269    /// syscall completed.
270    #[serde(default)]
271    pub first_divergent_syscall: Option<u64>,
272    /// First differing compared message from the left execution, with the
273    /// separately recorded syscall number, scheduler turn, and committed time
274    /// removed. Older reports omit it and remain readable.
275    #[serde(default)]
276    pub first_divergent_left_message: Option<String>,
277    /// Corresponding first differing compared message from the right execution.
278    #[serde(default)]
279    pub first_divergent_right_message: Option<String>,
280}

282/// Accept `null` but not a MISSING field. serde maps an absent `Option` field to
283/// `None` by default, which would let a report that never mentioned a comparison
284/// read as a legitimate no-result. Naming a `deserialize_with` makes the field
285/// required again while still admitting an explicit null.
286fn present_but_nullable_comparison<'de, D>(
287    deserializer: D,
288) -> Result<Option<ComparisonReport>, D::Error>
289where
290    D: serde::Deserializer<'de>,
291{
292    Option::<ComparisonReport>::deserialize(deserializer)
293}

319    /// `null` is legal only when no comparison completed: either the producer
320    /// reached no verdict, or it named an infrastructure error before it could
321    /// compare. A null comparison beside a product verdict is contradictory
322    /// and stays refused at parse.
323    fn require_consistent_outcome_fields(self) -> Result<Self, String> {
324        if self.comparison.is_none()
325            && !matches!(
326                self.verdict,
327                Verdict::NoResult | Verdict::InfrastructureError
328            )
329        {
330            return Err(format!(
331                "incomplete verification report: comparison is null but verdict is {}; null is legal only for no_result or infrastructure_error",
332                self.verdict
333            ));
334        }
335        match (&self.verdict, &self.infrastructure_error) {
336            (Verdict::InfrastructureError, Some(InfrastructureError::SkidOvershoot { count }))
337                if *count > 0 => {}
338            (Verdict::InfrastructureError, Some(InfrastructureError::SkidOvershoot { .. })) => {
339                return Err(
340                    "incomplete verification report: skid_overshoot count must be positive".into(),
341                );
342            }
343            (Verdict::InfrastructureError, None) => {
344                return Err(
345                    "incomplete verification report: infrastructure_error verdict omitted infrastructure_error"
346                        .into(),
347                );
348            }
349            (_, Some(_)) => {
350                return Err(format!(
351                    "inconsistent verification report: verdict={} carries infrastructure_error",
352                    self.verdict
353                ));
354            }
355            (_, None) => {}
356        }
357        Ok(self)
358    }

360    /// Parse the complete typed receipt. A top-level JSON object is not enough:
361    /// every nested field used by admission must deserialize successfully.
362    #[allow(dead_code)] // this file is path-included by consumers that use one parse form
363    pub fn from_json_slice(bytes: &[u8]) -> Result<Self, String> {
364        serde_json::from_slice::<Self>(bytes)
365            .map_err(|error| format!("incomplete verification report: {error}"))?
366            .require_consistent_outcome_fields()
367    }

369    #[allow(dead_code)] // this file is path-included by consumers that use one parse form
370    pub fn from_json_value(value: serde_json::Value) -> Result<Self, String> {
371        serde_json::from_value::<Self>(value)
372            .map_err(|error| format!("incomplete verification report: {error}"))?
373            .require_consistent_outcome_fields()
374    }

376    /// Parse a report written by the current producer and require every field
377    /// it promises to emit, preserving duplicate-field refusal from the original
378    /// bytes. Parsing only a Value would silently collapse duplicate object keys.
379    #[allow(dead_code)] // path-included readers use different parse forms
380    pub fn from_current_json_slice(bytes: &[u8]) -> Result<Self, String> {
381        let report = Self::from_json_slice(bytes)?;
382        let value = serde_json::from_slice(bytes)
383            .map_err(|error| format!("incomplete verification report: {error}"))?;
384        Self::from_current_json_value(value)?;
385        Ok(report)
386    }

388    /// Parse a report written by the current producer and require every field
389    /// that producer promises to emit. Retained-report readers use
390    /// [`Self::from_json_value`] instead so fields added after an old run remain
391    /// honest absence rather than making the whole report unreadable.
392    #[allow(dead_code)] // path-included readers use different parse forms
393    pub fn from_current_json_value(value: serde_json::Value) -> Result<Self, String> {
394        let object = value
395            .as_object()
396            .ok_or_else(|| "incomplete verification report: expected an object".to_string())?;
397        for field in [
398            "verified",
399            "bitwise_parity",
400            "verdict",
401            "infrastructure_error",
402            "comparison",
403            "no_result_reason",
404            "compared_log_messages",
405            "guest_exit_code",
406            "guest_signal",
407            "first_divergent_scheduler_turn",
408            "first_divergent_virtual_nanoseconds",
409            "first_divergent_record",
410            "first_divergent_syscall",
411            "first_divergent_left_message",
412            "first_divergent_right_message",
413        ] {
414            if !object.contains_key(field) {
415                return Err(format!(
416                    "incomplete verification report: missing current producer field `{field}`"
417                ));
418            }
419        }
420        if let Some(comparison) = object
421            .get("comparison")
422            .and_then(serde_json::Value::as_object)
423        {
424            for field in [
425                "strictness",
426                "display_name",
427                "compare_logs",
428                "compare_io_buffers",
429                "log_scope",
430                "record_envelope",
431                "virtualize_time",
432                "strip_lines",
433                "canonicalize_addresses",
434                "full_trace",
435                "exact_remainder",
436                "stripped_prefixes",
437                "canonicalizations",
438                "ignore_lines",
439                "skip_commit",
440                "skip_detlog",
441            ] {
442                if !comparison.contains_key(field) {
443                    return Err(format!(
444                        "incomplete verification report: missing current comparison field `{field}`"
445                    ));
446                }
447            }
448        }
449        let report = Self::from_json_value(value)?;
450        match (&report.verdict, &report.no_result_reason) {
451            (Verdict::NoResult, Some(NoResultReason::ComparisonRefused { detail }))
452                if detail.trim().is_empty() =>
453            {
454                Err("incomplete verification report: comparison_refused omitted its detail".into())
455            }
456            (Verdict::NoResult, Some(_)) => Ok(report),
457            (Verdict::NoResult, None) => Ok(report),
458            (_, Some(_)) => Err(format!(
459                "inconsistent verification report: verdict={} carries no_result_reason",
460                report.verdict
461            )),
462            (_, None) => Ok(report),
463        }
464    }
465
466    /// Prove that the invocation actually compared the canonical INFO evidence.
467    /// This is separate from whether that comparison matched: a canonical
468    /// divergence is a product result, while a stripped/output-only/empty-log
469    /// comparison is incomplete evidence and therefore an infrastructure result.
470    pub fn require_canonical_comparison(&self) -> Result<(), String> {
471        let counts = self.compared_log_messages.as_ref();
472        // No comparison at all is a distinct outcome from a comparison that was
473        // too weak, and it is still not admissible. Naming it separately is the
474        // whole point: "no verdict was recorded" is actionable against the
475        // producer, while the old "unreadable report" pointed at the reader.
476        let Some(comparison) = self.comparison.as_ref() else {
477            return Err(format!(
478                "verification recorded no comparison at all (verdict={}), so there is no canonical INFO evidence to admit",
479                self.verdict
480            ));
481        };
482        if comparison.strictness == LogCompareStrictness::Canonical
483            && comparison.compare_logs
484            && comparison.record_envelope.is_canonical()
485            && counts.is_some_and(|counts| counts.left > 0 && counts.right > 0)
486        {
487            Ok(())
488        } else {
489            Err(format!(
490                "verification did not compare canonical non-vacuous INFO evidence: strictness={} compare_logs={} record_envelope={} messages={}/{}",
491                comparison.strictness,
492                comparison.compare_logs,
493                comparison.record_envelope.as_str(),
494                counts.map_or(0, |counts| counts.left),
495                counts.map_or(0, |counts| counts.right),
496            ))
497        }
498    }
499
500    /// Admit a green only when both the canonical comparison and the match
501    /// claim agree. `verified=true` alone is intentionally insufficient: KVM's
502    /// output-only fallback can report it with zero compared INFO messages.
503    pub fn require_canonical_match(&self) -> Result<(), String> {
504        self.require_canonical_comparison()?;
505        if self.verified && self.verdict == Verdict::Matched && self.bitwise_parity {
506            Ok(())
507        } else {
508            Err(format!(
509                "canonical verification did not match: verified={} verdict={} bitwise_parity={}",
510                self.verified, self.verdict, self.bitwise_parity
511            ))
512        }
513    }

576    /// An EARLIER divergence must report a SMALLER value in both units. This is
577    /// the ordering the whole metric rests on: a fix that moves the divergence
