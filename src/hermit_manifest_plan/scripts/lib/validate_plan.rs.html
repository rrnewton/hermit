7//! Committed-plan selection and audit helpers for the validate driver.
8//!
9//! # The single rule this module exists to enforce
10//!
11//! **Nothing validate runs may execute outside `dagrun`.** Every gate --
12//! preflight submodule init, the Reverie pin check, the manifest gate, each CI
13//! node, and each compatibility probe -- is a node in the single committed
14//! `ci/dag/validate.json`. Profiles select that immutable graph by labels; this
15//! module does not compile or rewrite a runtime graph. The driver makes exactly
16//! one kind of call (`run_dag_boxed_ordered`) and never spawns work itself. The
17//! previous Phase-1 wrapper had a `run_subprocess_gate` helper that
18//! shelled out for the three preflight gates; that was a second execution path
19//! inside the driver, so those gates were unboxed, untimed by the runner, and
20//! invisible to its typed accounting. It is gone.
21//!
22//! # Every synthesized node MUST declare its caps — measured, not assumed
23//!
24//! `dagrun` applies its SMALL "forcing function" floor (1 GiB / 1 core
25//! / 10 s CPU) **only** through its own CLI, behind `--small-default-cap`. A
26//! LIBRARY consumer — which this driver is — gets `DagConfig::default()`, i.e.
27//! `default_step_mem_cap_bytes: None`, `default_step_cpu_count: None`,
28//! `default_step_cpu_timeout: 0`. That is deliberate on the runner's side (an
29//! always-on floor would wedge concurrent validates on the shared checkout), but
30//! it means **an undeclared node is boxed in name only**.
31//!
32//! Measured on this box at the time of writing, through this exact library path:
33//! a node declaring nothing allocated 2 GiB and burned 40 s of CPU and PASSED.
34//! A node declaring `hard_mem_max_bytes = 256 MiB` and allocating 4 GiB was
35//! `OOM-KILLED (hit inner MemoryMax; 3 oom_kill event(s))` at `peak≈256.0 MiB`
36//! and failed the run. Boxing works; it just has to be asked for.
37//!
38//! So every committed node declares `timeout`, `cpu_timeout`, and a memory
39//! hint, and `generate-validation-dag --check` (`validation_dag::assert_declared_caps`)
40//! is the fail-closed audit that keeps it true.
41//! The committed graph deliberately has no global CPU fallback: the generator
42//! records each measured or inherited node budget explicitly, including the
43//! hosted-privileged repair that replaced the retired graph's unusable implicit
44//! 10-second forcing default.

117        let s = |v: &str| v.to_string();
118        match self {
119            CompatMode::PortableStrict => vec![
120                s("run"),

142                        "--mount=type=bind,source={nsswitch},target=/etc/nsswitch.conf,readonly"

144                }
145                v.push(s("--strict"));
146                v.push(s("--verify"));
147                v.push(s("--"));
148                v
149            }
150        }
151    }

167/// What the compatibility summary should DO about one measured row.
168///
169/// Extracted as a pure function of (mode, outcome, table membership) so the decision can be
170/// bracketed without running a guest, and so the reporting text and the blocking verdict cannot
171/// drift apart -- they are now two readings of the same value rather than two independent
172/// branches of one `if`.
173#[derive(Debug, Clone, Copy, PartialEq, Eq)]
174pub enum CompatDisposition {
175    /// Passed, and nothing in the tables says otherwise.
176    Passed,
177    /// Passed while listed as known fail-closed. The EXPECTATION is stale, not the run.
178    PassedButListedFailClosed,
179    /// Failed while listed as known fail-closed, and STILL BLOCKING. Reporting the row's reason
180    /// is not the same as excusing it; this variant exists so the reason can be printed without
181    /// the failure being downgraded.
182    KnownFailClosedBlocking,
183    /// Failed as a bounded portable diagnostic: nonblocking by prior policy.
184    PortableDiagnostic,
185    /// Failed with nothing to say about it.
186    Blocking,
187}
188
189impl CompatDisposition {
190    /// Whether this row must fail the run.
191    ///
192    /// The ONE property that must not regress while reporting improves. `KnownFailClosedBlocking`
193    /// is deliberately blocking: a listed row under `PortableStrict` was already blocking before
194    /// the reason was printed, and printing it must not change that.
195    pub fn is_blocking(self) -> bool {
196        matches!(
197            self,
198            CompatDisposition::KnownFailClosedBlocking | CompatDisposition::Blocking
199        )
200    }
201}

203/// Classify one measured compatibility row.
204///
205/// Pure: it takes membership as booleans rather than the tables themselves, so a bracket can
206/// exercise every combination without constructing or planting a corpus, and so production
207/// keeps reading the real tables.
208///
209/// `PortableStrict` gains REPORTING ONLY. Every other mode --
210/// `E9patch` -- consults neither table and gains nothing: a failure there is
211/// blocking exactly as before.
212pub fn classify_compat_outcome(
213    mode: CompatMode,
214    ok: bool,
215    listed_failclosed: bool,
216    listed_diagnostic: bool,
217) -> CompatDisposition {
218    // Only the strict mode consults the fail-closed table at all; it describes what
219    // `--strict` refuses, which says nothing about the other backends.
220    let consults_failclosed = mode == CompatMode::PortableStrict && listed_failclosed;
221    if ok {
222        if consults_failclosed {
223            return CompatDisposition::PassedButListedFailClosed;
224        }
225        return CompatDisposition::Passed;
226    }
227    if consults_failclosed {
228        return CompatDisposition::KnownFailClosedBlocking;
229    }
230    if mode == CompatMode::PortableStrict && listed_diagnostic {
231        return CompatDisposition::PortableDiagnostic;
232    }
233    CompatDisposition::Blocking
234}

257        manifest: None,
258        integration_test_binaries: None,
259        result_manifests: None,
260        labels: Vec::new(),
261        deps,

269        engine_only: false,

272        cpu_timeout,

289/// Shell-quote one argv element for embedding in a `bash -c` command string.
290///
291/// The corpus carries argv ARRAYS (that is how it was extracted, and it is what
292/// keeps a workload containing spaces, quotes, or `$` from being re-split). The
293/// runner takes a single shell string, so each element is single-quoted here with
294/// the standard `'\''` escape. Getting this wrong would silently mutate guest
295/// commands, so it is exercised by `--self-test`.
296pub fn shell_quote(arg: &str) -> String {
297    if !arg.is_empty()
298        && arg
299            .bytes()
300            .all(|b| b.is_ascii_alphanumeric() || b"@%+=:,./-_".contains(&b))
301    {
302        return arg.to_string();
303    }
304    format!("'{}'", arg.replace('\'', r"'\''"))
305}
306
307/// Join an argv into a shell command string.
308pub fn shell_join<I: IntoIterator<Item = S>, S: AsRef<str>>(argv: I) -> String {
309    argv.into_iter()
310        .map(|a| shell_quote(a.as_ref()))
311        .collect::<Vec<_>>()
312        .join(" ")
313}
314
315/// A fixture passed as a positional argument to `bash -c` is expanded by the
316/// outer shell, before Hermit clears the guest environment. Keep it double-quoted
317/// so the generated DAG's later `$VALIDATE_RUN_STATE` substitution is expanded
318/// once without word splitting. Literal shell metacharacters in constructed paths
319/// must still survive unchanged; the placeholder is inserted after this quoting.
320fn shell_quote_fixture_operand(path: &str) -> String {
321    let mut quoted = String::from("\"");
322    for ch in path.chars() {
323        if matches!(ch, '\\' | '"' | '$' | '`') {
324            quoted.push('\\');
325        }
326        quoted.push(ch);
327    }
328    quoted.push('"');
329    quoted
330}

332fn compat_guest_command(row: &validate_corpus::CorpusRow, paths: &CorpusPaths) -> String {
333    row.argv
334        .iter()
335        .enumerate()
336        .map(|(index, arg)| {
337            // bash -c SCRIPT NAME ARG...: only explicit fixture operands get this
338            // treatment. Never interpolate the script body or alter ordinary argv.
339            if index >= 4
340                && row.argv[0] == "bash"
341                && row.argv[1] == "-c"
342                && Path::new(arg).starts_with(paths.real_compat_fixtures)
343            {
344                shell_quote_fixture_operand(arg)
345            } else {
346                shell_quote(arg)
347            }
348        })
349        .collect::<Vec<_>>()
350        .join(" ")
351}

353/// The always-on preflight gates and the manifest-audit binary producer, as DAG nodes.
354///
355/// Submodule verification is deliberately non-mutating and first. Initializing
356/// or repairing a checkout before observing it would erase the exact drift this
357/// gate exists to detect. A caller with an uninitialized checkout must run
358/// `make checkout-all` explicitly, then retry validation.
359pub fn preflight_nodes(root: &Path) -> Result<Vec<Step>, String> {
360    let committed = validation_config(root)?;
361    let tags = [
362        "pre.submodules",
363        "pre.reverie_pin",
364        "build.rust_scripts",
365        MANIFEST_PLAN_PRODUCER_TAG,
366        "gate.manifest",
367    ];
368    tags.into_iter()
369        .map(|tag| {
370            committed
371                .steps
372                .iter()
373                .find(|step| step.tag() == tag)
374                .cloned()
375                .ok_or_else(|| format!("committed validation DAG lost preflight node {tag}"))
376        })
377        .collect()
378}

399            .ancestors()

402            .to_path_buf()
403    }

511// ------------------------------------------------- host-capability requirements
512//
513// A node can require a facility the MACHINE either has or does not have. Before
514// this section such a node had exactly two outcomes — it passed, or it failed —
515// so "this host cannot run it" and "this ran and it is broken" were the same
516// record. On a host without CPUID faulting `privileged-cpuid.faulting` failed in
517// 0.11 s with exit 101 and an empty detail block, which reads like a broken
518// build, and its eager-exit cost the other twelve in-flight nodes
519// (hermit#2135, hermit#2148, hermit#2205).
520//
521// The fix is a THIRD recorded outcome, host-inapplicable, decided BEFORE the
522// node is spawned. Five properties keep it from becoming a way to excuse a node
523// that is merely broken:
524//
525//  1. The decision never reads the node. It is made from an out-of-band probe of
526//     the machine during plan construction; a node's exit code, stderr, or panic
527//     message cannot produce it. A broken node still runs, still fails, and is
528//     still refused.
529//  2. The capability vocabulary is CLOSED and shared with stress-series
530//     ([`HostCapability`]). A DAG naming an unknown capability is a
531//     plan-construction refusal, not a skip.
532//  3. The probe fails closed TOWARD RUNNING. Absence requires two independent
533//     sources to agree; a probe error, an unexpected errno, an unreadable
534//     `/proc/cpuinfo`, or disagreement between the sources all resolve to
535//     PRESENT, so the node runs and any failure is real.
536//  4. There is no override that manufactures absence.
537//     `HERMIT_VALIDATE_HOST_CAPABILITY_PRESENT` can only force a capability
538//     PRESENT, i.e. can only cause MORE to run.
539//  5. Recording it honestly is what costs the receipt. The node is written to
540//     the ledger as a typed intentional skip whose reason is `host-inapplicable`,
541//     and the parent's separately-reviewed consumer allowlist
542//     (`ci-hub/validate/gate_completeness.py::ALLOWED_INTENTIONAL_SKIP_REASONS`,
543//     `ci-hub/lib/qualifying_receipt.rs::intentional_skip_count`) admits only
544//     `empty-manifest-bucket`. A run carrying a host-inapplicable node therefore
545//     does NOT qualify as landing authority until the owner opts that reason in.
546//     The mechanism cannot buy a green; it can only stop one node's absence from
547//     destroying the other forty.
548
549/// The stable ledger reason for a node the machine provably cannot run.
550///
551/// The word is the owner's, from hermit#2205: "The missing concept is a third,
552/// recorded outcome: **host-inapplicable**, distinct from both pass and fail."
553pub const HOST_INAPPLICABLE_REASON: &str = "host-inapplicable";
554
555/// One node the machine provably cannot run, and why.
556#[derive(Clone, Debug)]
557pub struct HostInapplicableNode {
558    pub tag: String,
559    pub capability: HostCapability,
560    pub evidence: String,
561    /// `None`: the whole node is withheld and never spawns. `Some`: the node
562    /// RUNS, but `target/debug/test-harness` withholds some of its manifest
563    /// cells as host-inapplicable. Either way the run did not cover what those
564    /// cells verify, so both count against completeness alike.
565    pub cells: Option<WithheldCells>,
566}
567
568/// The cells of a RUNNING manifest bucket node that the harness withholds.
569#[derive(Clone, Copy, Debug, PartialEq, Eq)]
570pub struct WithheldCells {
571    pub withheld: usize,
572    pub selected: usize,
573}
574
575impl HostInapplicableNode {
576    /// Whether the node itself still runs, with only some cells withheld.
577    pub fn runs(&self) -> bool {
578        self.cells.is_some()
579    }
580}

582/// The `requires_host_capability` declarations in one lane, keyed by runner tag.
583///
584/// Reads the sole committed validation DAG through the same path helper.
585/// An unparseable capability name is an ERROR: refusing the whole run is the
586/// only safe response to a declaration nobody can evaluate.
587pub fn lane_host_capability_requirements(
588    root: &Path,
589    lane: &str,
590) -> Result<BTreeMap<String, HostCapability>, String> {
591    let cfg = validation_config(root)?;
592    let mut out = BTreeMap::new();
593    for step in &cfg.steps {
594        if !step.labels.iter().any(|label| label == lane) {
595            continue;
596        }
597        for capability in [HostCapability::CpuidFaulting, HostCapability::Kvm] {
598            if step.labels.iter().any(|label| label == capability.value()) {
599                out.insert(step.tag(), capability);
600            }
601        }
602    }
603    Ok(out)
604}

619/// Split a step list into what will run and what the machine cannot run.
620///
621/// PURE: it consumes an already-resolved absence map, so `--self-test` brackets
622/// both directions without touching the machine.
623///
624/// A step is withheld ONLY when it declares a capability that is in `absent`.
625/// A step with no declaration is never withheld, whatever is absent — that is
626/// what stops this from excusing a node that is merely broken.
627///
628/// Withholding a node that another RETAINED node depends on would silently
629/// orphan work, so it is a refusal rather than a cascade.
630pub fn partition_host_inapplicable(
631    steps: Vec<Step>,
632    requirements: &BTreeMap<String, HostCapability>,
633    absent: &BTreeMap<HostCapability, String>,
634) -> Result<(Vec<Step>, Vec<HostInapplicableNode>), String> {
635    let mut keep = Vec::with_capacity(steps.len());
636    let mut withheld = Vec::new();
637    for step in steps {
638        let tag = step.tag();
639        match requirements.get(&tag) {
640            Some(capability) if absent.contains_key(capability) => {
641                withheld.push(HostInapplicableNode {
642                    tag,
643                    capability: *capability,
644                    evidence: absent[capability].clone(),
645                    cells: None,
646                })
647            }
648            _ => keep.push(step),
649        }
650    }
651    let gone: std::collections::BTreeSet<&str> = withheld.iter().map(|n| n.tag.as_str()).collect();
652    let mut orphaned = Vec::new();
653    for step in &keep {
654        for dep in &step.deps {
655            if gone.contains(dep.as_str()) {
656                orphaned.push(format!("{} depends on {dep}", step.tag()));
657            }
658        }
659    }
660    if !orphaned.is_empty() {
661        return Err(format!(
662            "refusing to withhold a host-inapplicable node that other nodes depend on: {}; \
663             a missing capability must not silently cascade into unrun work",
664            orphaned.join(", ")
665        ));
666    }
667    Ok((keep, withheld))
668}

670/// Build the compatibility-corpus nodes for one mode.
671///
672/// One DAG node PER PROBE. That is a deliberate change from the bash, which ran
673/// all ~191 probes serially inside a single gate:
674///   * each probe now gets its own wall + CPU + memory box, so one runaway row
675///     cannot consume the whole gate's budget;
676///   * each probe's verdict is a TYPED `StepOutcome`, so the summary table is
677///     built from structured results instead of a scraped TSV; and
678///   * the corpus becomes parallel, which is where a large part of the wall-clock
679///     win in this profile is expected to come from.
680pub fn compat_nodes(
681    root: &Path,
682    mode: CompatMode,
683    hermit_bin: &str,
684    nsswitch: &str,
685    paths: &CorpusPaths,
686    gate_dep: Option<&str>,
687) -> Result<Vec<Step>, String> {
688    compat_nodes_for(
689        root, mode, hermit_bin, nsswitch, paths, gate_dep, None, None,
690    )
691}

693/// [`compat_nodes`] with two extra knobs used by the `super` suite's
694/// `run_portable_slow_strict_diagnostics` port (validate.sh:4603).
695///
696/// * `only` restricts the corpus to an explicit label set AND suppresses the
697///   `PORTABLE_STRICT_SUPER_ONLY` skip — because that gate exists precisely to
698///   defer those four heavy rows *to this suite*, so the suite that runs them
699///   must not also honor the deferral.
700/// * `wall_override` replaces the per-row 60s corpus budget. The bash gave the
701///   whole group of four one 600s `run_check_with_timeout`; each of these rows
702///   is a full compile-link-run or JVM startup workload, so inheriting the
703///   group's budget per node is the faithful reading. The 60s corpus default
704///   would fail all four for lack of time and report it as a compatibility loss.
705#[allow(clippy::too_many_arguments)]
706pub fn compat_nodes_for(
707    root: &Path,
708    mode: CompatMode,
709    hermit_bin: &str,
710    nsswitch: &str,
711    paths: &CorpusPaths,
712    gate_dep: Option<&str>,
713    only: Option<&std::collections::BTreeSet<String>>,
714    wall_override: Option<i64>,
715) -> Result<Vec<Step>, String> {
716    let rows = validate_corpus::load(root, mode.corpus_name(), paths)?;
717    let super_only = validate_corpus::portable_super_only();
718    let mut out = Vec::new();
719    for row in rows {
720        if let Some(keep) = only {
721            if !keep.contains(&row.label) {
722                continue;
723            }
724        }
725        // Heavy runtime workloads are deferred out of the portable profile to the
726        // scheduled super suite (validate.sh:3090) — unless this IS that suite,
727        // which names them explicitly through `only`.
728        if only.is_none()
729            && mode == CompatMode::PortableStrict
730            && super_only.contains_key(row.label.as_str())
731        {
732            continue;
733        }
734        let mut argv: Vec<String> = vec![hermit_bin.to_string()];
735        argv.extend(mode.run_args(&row.label, nsswitch));
736        let wall = wall_override.unwrap_or_else(|| mode.timeout_for(&row.label));
737        let description = match mode {
738            CompatMode::E9patch => format!(
739                "One per-program probe of the e9patch-compat-only profile, generated from a row of ci/compat/{}. It runs `target/release/hermit --backend e9patch run --strict --verify -- <program>` with stdin from /dev/null. That is e9patch binary-rewriting preprocessing with the ptrace backend, not a backend of its own: Hermit rewrites the guest's main ELF with e9tool and e9patch, then runs the rewritten program on the ptrace backend under strict mode, twice, comparing the runs with the Stripped comparator of plain --verify, so no row establishes L2. The rows range from coreutils such as echo and cat to bash pipelines and tests/compat/real_compat_workload.sh workloads; whoami, groups, pinky, logname, tar and chown also bind the files-only nsswitch.conf over /etc/nsswitch.conf. The profile counts every failure as blocking, for example a rewritten binary that crashes or two runs whose output differs.",
740                mode.corpus_name()
741            ),
742            CompatMode::PortableStrict => match super_only.get(row.label.as_str()) {
743                Some(workload) if only.is_some() => format!(
744                    "Runs the {label} row of the portable strict compatibility corpus, a {workload}, as `target/release/hermit run --strict --verify --base-env=minimal --no-virtualize-cpuid --max-timeslice=disabled` with a tmpfs /test working directory, stdin from /dev/null, on the default ptrace backend. Plain --verify runs the guest twice and compares the runs with the Stripped comparator, so the row establishes L1 with a lossy repeat check, not L2. The portable profile defers this row to the super profile because it is too heavy for the corpus's {COMPAT_TIMEOUT_S}-second budget; here it gets {wall} seconds and runs after {fixtures}, which prepares the files the real_compat_workload.sh rows read. A missing toolchain or fixture, a syscall strict mode refuses, or output that differs between the two runs turns it red, and a failure fails the super run.",
745                    label = row.label,
746                    fixtures = gate_dep.unwrap_or("its fixtures node"),
747                ),
748                _ => format!(
749                    "Runs the {label} row of the portable strict compatibility corpus as `target/release/hermit run --strict --verify --base-env=minimal --no-virtualize-cpuid --max-timeslice=disabled` with a tmpfs /test working directory, stdin from /dev/null, on the default ptrace backend, comparing the two runs with the Stripped comparator of plain --verify. A syscall strict mode refuses or output that differs between the runs turns it red.",
750                    label = row.label,
751                ),
752            },
753        };
754        let mut step = node(
755            "compat",
756            &sanitize_job(&row.label),
757            &format!("{} compatibility: {}", mode.display_name(), row.label),
758            format!(
759                "{} {} </dev/null",
760                shell_join(&argv),
761                compat_guest_command(&row, paths)
762            ),
763            gate_dep.map(|d| vec![d.to_string()]).unwrap_or_default(),
764            wall,
765            COMPAT_CPU_TIMEOUT_S.max(wall),
766            COMPAT_MEM_BYTES,
767        );
768        step.description = description;
769        out.push(step);
770    }
771    if out.is_empty() {
772        return Err(format!("compatibility mode {mode:?} selected zero probes"));
773    }
774    Ok(out)
775}

803/// Assemble a `DagConfig`, CARRYING every top-level field from `base`.
804///
805/// # Why this takes a base at all
806///
807/// It used to be `DagConfig { steps, ..Default::default() }`, which loaded a DAG
808/// file, kept its steps, and threw its configuration away. That is not a
809/// hypothetical: it hung a full validate for 14 minutes at 0% CPU.
810/// `ci/dag/validate.json` declares `resource_caps {manifest_guest: 8}`;
811/// dropping it leaves `res_free` evaluating `unwrap_or(0) >= 1` for the 13
812/// steps demanding `manifest_guest`, so none can be admitted. The scheduler's
813/// only exit is `running.is_empty() && done + skipped >= steps.len()`, so with
814/// work neither runnable nor accounted it slept at 50 ms forever -- no error, no
815/// exit, 21 of ~58 nodes done.
816///
817/// `resource_caps` failed LOUDLY (a visible hang). The quieter one matters more:
818/// `default_step_timeout` is 600 s in portable and 120 s in privileged, and
819/// reverted to `DEFAULT_STEP_TIMEOUT` (1800 s) -- every step's wall cap loosened
820/// 3x and 15x respectively, with nothing to see. `mem_cap_factor`,
821/// `mem_cap_floor_bytes` and `outer_mem_safety_factor` happen to equal their
822/// defaults today, so they would have broken the first time anyone tuned them.
823///
824/// Hence: carry the base wholesale, and let [`assert_config_carried`] prove it.
825pub fn config_from_base(base: &DagConfig, steps: Vec<Step>, description: &str) -> DagConfig {
826    let mut cfg = base.clone();
827    cfg.steps = steps;
828    cfg.description = description.to_string();
829    // Synthetic generator/self-test configs use a bounded fallback. Production
830    // selections retain the explicit per-node budgets in the committed DAG.
831    cfg.default_step_cpu_timeout = SYNTHETIC_DEFAULT_CPU_TIMEOUT_S;
832    cfg
833}

835/// Synthesised plans that have no source DAG file (compat, quick, envelope, ...).
836pub fn config_from(steps: Vec<Step>, description: &str) -> DagConfig {
837    config_from_base(&DagConfig::default(), steps, description)
838}

840/// Field-by-field proof that `derived` carried `base`'s configuration.
841///
842/// Enumerated deliberately rather than derived from a `PartialEq`: a new
843/// `DagConfig` field must force a decision here instead of silently defaulting,
844/// which is the exact failure this function exists to prevent. `steps` and
845/// `description` are expected to differ; `default_step_cpu_timeout` is the one
846/// documented divergence above.
847pub fn assert_config_carried(base: &DagConfig, derived: &DagConfig) -> Result<(), String> {
848    let mut bad: Vec<String> = Vec::new();
849    if base.resource_caps != derived.resource_caps {
850        bad.push(format!(
851            "resource_caps {:?} != {:?}",
852            base.resource_caps, derived.resource_caps
853        ));
854    }
855    if base.mem_cap_factor != derived.mem_cap_factor {
856        bad.push(format!(
857            "mem_cap_factor {} != {}",
858            base.mem_cap_factor, derived.mem_cap_factor
859        ));
860    }
861    if base.mem_cap_floor_bytes != derived.mem_cap_floor_bytes {
862        bad.push(format!(
863            "mem_cap_floor_bytes {} != {}",
864            base.mem_cap_floor_bytes, derived.mem_cap_floor_bytes
865        ));
866    }
867    if base.outer_mem_safety_factor != derived.outer_mem_safety_factor {
868        bad.push(format!(
869            "outer_mem_safety_factor {} != {}",
870            base.outer_mem_safety_factor, derived.outer_mem_safety_factor
871        ));
872    }
873    if base.default_step_timeout != derived.default_step_timeout {
874        bad.push(format!(
875            "default_step_timeout {} != {}",
876            base.default_step_timeout, derived.default_step_timeout
877        ));
878    }
879    if base.default_jobs_flag != derived.default_jobs_flag {
880        bad.push(format!(
881            "default_jobs_flag {:?} != {:?}",
882            base.default_jobs_flag, derived.default_jobs_flag
883        ));
884    }
885    if base.default_jobs_env != derived.default_jobs_env {
886        bad.push(format!(
887            "default_jobs_env {:?} != {:?}",
888            base.default_jobs_env, derived.default_jobs_env
889        ));
890    }
891    if base.default_step_mem_cap_bytes != derived.default_step_mem_cap_bytes {
892        bad.push(format!(
893            "default_step_mem_cap_bytes {:?} != {:?}",
894            base.default_step_mem_cap_bytes, derived.default_step_mem_cap_bytes
895        ));
896    }
897    if base.default_step_cpu_count != derived.default_step_cpu_count {
898        bad.push(format!(
899            "default_step_cpu_count {:?} != {:?}",
900            base.default_step_cpu_count, derived.default_step_cpu_count
901        ));
902    }
903    if bad.is_empty() {
904        Ok(())
905    } else {
906        Err(bad.join("; "))
907    }
908}

910/// FAIL CLOSED on capacity that can never be granted.
911///
912/// A step demanding a resource the config does not cap is unschedulable FOREVER,
913/// and the scheduler expresses that as an infinite 50 ms sleep rather than an
914/// error. Refusing up front converts a silent 14-minute hang into a named
915/// refusal before a single node runs.
916pub fn ungrantable_resources(cfg: &DagConfig) -> Vec<String> {
917    let mut bad = Vec::new();
918    for s in &cfg.steps {
919        for (r, n) in &s.hint.resources {
920            let cap = cfg.resource_caps.get(r).copied().unwrap_or(0);
921            if cap < *n {
922                bad.push(format!(
923                    "{} demands {r}={n} but resource_caps grants {cap}",
924                    s.tag()
925                ));
926            }
927        }
928    }
929    bad
930}
