7//! Committed-plan selection and audit helpers for the validate driver.
8//!
9//! # The single rule this module exists to enforce
10//!
11//! **Nothing validate runs may execute outside `dagrun`.** Every gate --
12//! preflight submodule init, the Reverie pin check, the manifest gate, each CI
13//! node, and each compatibility probe -- is a node in the single committed
14//! `ci/dag/validate.json`. Profiles select that immutable graph by labels; this
15//! module does not compile or rewrite a runtime graph. The driver makes exactly
16//! one kind of call (`run_dag_boxed_ordered`) and never spawns work itself. The
17//! previous Phase-1 wrapper had a `run_subprocess_gate` helper that
18//! shelled out for the three preflight gates; that was a second execution path
19//! inside the driver, so those gates were unboxed, untimed by the runner, and
20//! invisible to its typed accounting. It is gone.
21//!
22//! # Every synthesized node MUST declare its caps — measured, not assumed
23//!
24//! `dagrun` applies its SMALL "forcing function" floor (1 GiB / 1 core
25//! / 10 s CPU) **only** through its own CLI, behind `--small-default-cap`. A
26//! LIBRARY consumer — which this driver is — gets `DagConfig::default()`, i.e.
27//! `default_step_mem_cap_bytes: None`, `default_step_cpu_count: None`,
28//! `default_step_cpu_timeout: 0`. That is deliberate on the runner's side (an
29//! always-on floor would wedge concurrent validates on the shared checkout), but
30//! it means **an undeclared node is boxed in name only**.
31//!
32//! Measured on this box at the time of writing, through this exact library path:
33//! a node declaring nothing allocated 2 GiB and burned 40 s of CPU and PASSED.
34//! A node declaring `hard_mem_max_bytes = 256 MiB` and allocating 4 GiB was
35//! `OOM-KILLED (hit inner MemoryMax; 3 oom_kill event(s))` at `peak≈256.0 MiB`
36//! and failed the run. Boxing works; it just has to be asked for.
37//!
38//! So every committed node declares `timeout`, `cpu_timeout`, and a memory
39//! hint, and `generate-validation-dag --check` (`validation_dag::assert_declared_caps`)
40//! is the fail-closed audit that keeps it true.
41//! The committed graph deliberately has no global CPU fallback: the generator
42//! records each measured or inherited node budget explicitly, including the
43//! hosted-privileged repair that replaced the retired graph's unusable implicit
44//! 10-second forcing default.

168        job: job.to_string(),

173        manifest: None,
174        integration_test_binaries: None,
175        result_manifests: None,
176        labels: Vec::new(),
177        deps,

185        engine_only: false,
186        delegated_children: false,

188        cpu_timeout,

201        fail_fast_family: None,
202    }

205/// Shell-quote one argv element for embedding in a `bash -c` command string.
206///
207/// The corpus carries argv ARRAYS (that is how it was extracted, and it is what
208/// keeps a workload containing spaces, quotes, or `$` from being re-split). The
209/// runner takes a single shell string, so each element is single-quoted here with
210/// the standard `'\''` escape. Getting this wrong would silently mutate guest
211/// commands, so it is exercised by `--self-test`.
212pub fn shell_quote(arg: &str) -> String {
213    if !arg.is_empty()
214        && arg
215            .bytes()
216            .all(|b| b.is_ascii_alphanumeric() || b"@%+=:,./-_".contains(&b))
217    {
218        return arg.to_string();
219    }
220    format!("'{}'", arg.replace('\'', r"'\''"))
221}
222
223/// Join an argv into a shell command string.
224pub fn shell_join<I: IntoIterator<Item = S>, S: AsRef<str>>(argv: I) -> String {
225    argv.into_iter()
226        .map(|a| shell_quote(a.as_ref()))
227        .collect::<Vec<_>>()
228        .join(" ")
229}
230
231/// A fixture passed as a positional argument to `bash -c` is expanded by the
232/// outer shell, before Hermit clears the guest environment. Keep it double-quoted
233/// so the generated DAG's later `$VALIDATE_RUN_STATE` substitution is expanded
234/// once without word splitting. Literal shell metacharacters in constructed paths
235/// must still survive unchanged; the placeholder is inserted after this quoting.
236fn shell_quote_fixture_operand(path: &str) -> String {
237    let mut quoted = String::from("\"");
238    for ch in path.chars() {
239        if matches!(ch, '\\' | '"' | '$' | '`') {
240            quoted.push('\\');
241        }
242        quoted.push(ch);
243    }
244    quoted.push('"');
245    quoted
246}

248fn compat_guest_command(row: &validate_corpus::CorpusRow, paths: &CorpusPaths) -> String {
249    row.argv
250        .iter()
251        .enumerate()
252        .map(|(index, arg)| {
253            // bash -c SCRIPT NAME ARG...: only explicit fixture operands get this
254            // treatment. Never interpolate the script body or alter ordinary argv.
255            if index >= 4
256                && row.argv[0] == "bash"
257                && row.argv[1] == "-c"
258                && Path::new(arg).starts_with(paths.real_compat_fixtures)
259            {
260                shell_quote_fixture_operand(arg)
261            } else {
262                shell_quote(arg)
263            }
264        })
265        .collect::<Vec<_>>()
266        .join(" ")
267}

269/// The always-on preflight gates and the manifest-audit binary producer, as DAG nodes.
270///
271/// Submodule verification is deliberately non-mutating and first. Initializing
272/// or repairing a checkout before observing it would erase the exact drift this
273/// gate exists to detect. A caller with an uninitialized checkout must run
274/// `make checkout-all` explicitly, then retry validation.
275pub fn preflight_nodes(root: &Path) -> Result<Vec<Step>, String> {
276    let committed = validation_config(root)?;
277    let tags = [
278        "pre.submodules",
279        "pre.reverie_pin",
280        "build.rust_scripts",
281        MANIFEST_PLAN_PRODUCER_TAG,
282        "gate.manifest",
283    ];
284    tags.into_iter()
285        .map(|tag| {
286            committed
287                .steps
288                .iter()
289                .find(|step| step.tag() == tag)
290                .cloned()
291                .ok_or_else(|| format!("committed validation DAG lost preflight node {tag}"))
292        })
293        .collect()
294}

429// ------------------------------------------------- host-capability requirements
430//
431// A node can require a facility the MACHINE either has or does not have. Before
432// this section such a node had exactly two outcomes — it passed, or it failed —
433// so "this host cannot run it" and "this ran and it is broken" were the same
434// record. On a host without CPUID faulting `privileged-cpuid.faulting` failed in
435// 0.11 s with exit 101 and an empty detail block, which reads like a broken
436// build, and its eager-exit cost the other twelve in-flight nodes
437// (hermit#2135, hermit#2148, hermit#2205).
438//
439// The fix is a THIRD recorded outcome, host-inapplicable, decided BEFORE the
440// node is spawned. Five properties keep it from becoming a way to excuse a node
441// that is merely broken:
442//
443//  1. The decision never reads the node. It is made from an out-of-band probe of
444//     the machine during plan construction; a node's exit code, stderr, or panic
445//     message cannot produce it. A broken node still runs, still fails, and is
446//     still refused.
447//  2. The capability vocabulary is CLOSED and shared with stress-series
448//     ([`HostCapability`]). A DAG naming an unknown capability is a
449//     plan-construction refusal, not a skip.
450//  3. The probe fails closed TOWARD RUNNING. Absence requires two independent
451//     sources to agree; a probe error, an unexpected errno, an unreadable
452//     `/proc/cpuinfo`, or disagreement between the sources all resolve to
453//     PRESENT, so the node runs and any failure is real.
454//  4. There is no override that manufactures absence.
455//     `HERMIT_VALIDATE_HOST_CAPABILITY_PRESENT` can only force a capability
456//     PRESENT, i.e. can only cause MORE to run.
457//  5. Recording it honestly is what costs the receipt. The node is written to
458//     the ledger as a typed intentional skip whose reason is `host-inapplicable`,
459//     and the parent's separately-reviewed consumer allowlist
460//     (`ci-hub/validate/gate_completeness.py::ALLOWED_INTENTIONAL_SKIP_REASONS`,
461//     `ci-hub/lib/qualifying_receipt.rs::intentional_skip_count`) admits only
462//     `empty-manifest-bucket`. A run carrying a host-inapplicable node therefore
463//     does NOT qualify as landing authority until the owner opts that reason in.
464//     The mechanism cannot buy a green; it can only stop one node's absence from
465//     destroying the other forty.
466
467/// The stable ledger reason for a node the machine provably cannot run.
468///
469/// The word is the owner's, from hermit#2205: "The missing concept is a third,
470/// recorded outcome: **host-inapplicable**, distinct from both pass and fail."
471pub const HOST_INAPPLICABLE_REASON: &str = "host-inapplicable";

473/// One node the machine provably cannot run, and why.
474#[derive(Clone, Debug)]
475pub struct HostInapplicableNode {
476    pub tag: String,
477    pub capability: HostCapability,
478    pub evidence: String,
479    /// `None`: the whole node is withheld and never spawns. `Some`: the node
480    /// RUNS, but `target/debug/test-harness` withholds some of its manifest
481    /// cells as host-inapplicable. Either way the run did not cover what those
482    /// cells verify, so both count against completeness alike.
483    pub cells: Option<WithheldCells>,
484}

486/// The cells of a RUNNING manifest bucket node that the harness withholds.
487#[derive(Clone, Copy, Debug, PartialEq, Eq)]
488pub struct WithheldCells {
489    pub withheld: usize,
490    pub selected: usize,
491}

493impl HostInapplicableNode {
494    /// Whether the node itself still runs, with only some cells withheld.
495    pub fn runs(&self) -> bool {
496        self.cells.is_some()
497    }

500/// The `requires_host_capability` declarations in one lane, keyed by runner tag.
501///
502/// Reads the sole committed validation DAG through the same path helper.
503/// An unparseable capability name is an ERROR: refusing the whole run is the
504/// only safe response to a declaration nobody can evaluate.
505pub fn lane_host_capability_requirements(
506    root: &Path,
507    lane: &str,
508) -> Result<BTreeMap<String, HostCapability>, String> {
509    let cfg = validation_config(root)?;
510    let mut out = BTreeMap::new();
511    for step in &cfg.steps {
512        if !step.labels.iter().any(|label| label == lane) {
513            continue;
514        }
515        for capability in [HostCapability::CpuidFaulting, HostCapability::Kvm] {
516            if step.labels.iter().any(|label| label == capability.value()) {
517                out.insert(step.tag(), capability);
518            }
519        }
520    }
521    Ok(out)
522}

524/// Every `requires_host_capability` declaration in every shipped lane, under
525/// both the bare and the committed prefixed tag spelling, so the caller can look a plan's
526/// tags up directly however the plan was assembled.
527pub fn host_capability_requirements(
528    root: &Path,
529) -> Result<BTreeMap<String, HostCapability>, String> {
530    let mut out = BTreeMap::new();
531    for lane in ["portable", "privileged", "full", "quick", "super"] {
532        out.extend(lane_host_capability_requirements(root, lane)?);
533    }
534    Ok(out)
535}

537/// Split a step list into what will run and what the machine cannot run.
538///
539/// PURE: it consumes an already-resolved absence map, so `--self-test` brackets
540/// both directions without touching the machine.
541///
542/// A step is withheld ONLY when it declares a capability that is in `absent`.
543/// A step with no declaration is never withheld, whatever is absent — that is
544/// what stops this from excusing a node that is merely broken.
545///
546/// Withholding a node that another RETAINED node depends on would silently
547/// orphan work, so it is a refusal rather than a cascade.
548pub fn partition_host_inapplicable(
549    steps: Vec<Step>,
550    requirements: &BTreeMap<String, HostCapability>,
551    absent: &BTreeMap<HostCapability, String>,
552) -> Result<(Vec<Step>, Vec<HostInapplicableNode>), String> {
553    let mut keep = Vec::with_capacity(steps.len());
554    let mut withheld = Vec::new();
555    for step in steps {
556        let tag = step.tag();
557        match requirements.get(&tag) {
558            Some(capability) if absent.contains_key(capability) => {
559                withheld.push(HostInapplicableNode {
560                    tag,
561                    capability: *capability,
562                    evidence: absent[capability].clone(),
563                    cells: None,
564                })
565            }
566            _ => keep.push(step),
567        }
568    }
569    let gone: std::collections::BTreeSet<&str> = withheld.iter().map(|n| n.tag.as_str()).collect();
570    let mut orphaned = Vec::new();
571    for step in &keep {
572        for dep in &step.deps {
573            if gone.contains(dep.as_str()) {
574                orphaned.push(format!("{} depends on {dep}", step.tag()));
575            }
576        }
577    }
578    if !orphaned.is_empty() {
579        return Err(format!(
580            "refusing to withhold a host-inapplicable node that other nodes depend on: {}; \
581             a missing capability must not silently cascade into unrun work",
582            orphaned.join(", ")
583        ));
584    }
585    Ok((keep, withheld))
586}

588/// Build the compatibility-corpus nodes for one mode.
589///
590/// One DAG node PER PROBE. That is a deliberate change from the bash, which ran
591/// all ~191 probes serially inside a single gate:
592///   * each probe now gets its own wall + CPU + memory box, so one runaway row
593///     cannot consume the whole gate's budget;
594///   * each probe's verdict is a TYPED `StepOutcome`, so the summary table is
595///     built from structured results instead of a scraped TSV; and
596///   * the corpus becomes parallel, which is where a large part of the wall-clock
597///     win in this profile is expected to come from.
598pub fn compat_nodes(
599    root: &Path,
600    mode: CompatMode,
601    hermit_bin: &str,
602    nsswitch: &str,
603    paths: &CorpusPaths,
604    gate_dep: Option<&str>,
605) -> Result<Vec<Step>, String> {
606    let rows = validate_corpus::load(root, mode.corpus_name(), paths)?;
607    let mut out = Vec::new();
608    for row in rows {
609        let mut argv: Vec<String> = vec![hermit_bin.to_string()];
610        argv.extend(mode.run_args(&row.label, nsswitch));
611        let wall = mode.timeout_for(&row.label);
612        let description = match mode {
613            CompatMode::E9patch => format!(
614                "One per-program probe of the e9patch-compat-only profile, generated from a row of ci/compat/{}. It runs `target/release/hermit --backend e9patch run --strict --epoch=<canonical test epoch> --verify -- <program>` with stdin from /dev/null. That is e9patch binary-rewriting preprocessing with the ptrace backend, not a backend of its own: Hermit rewrites the guest's main ELF with e9tool and e9patch, then runs the rewritten program on the ptrace backend under strict mode, twice, comparing the runs with the Stripped comparator of plain --verify, so no row establishes L2. The rows range from coreutils such as echo and cat to bash pipelines and tests/compat/real_compat_workload.sh workloads; whoami, groups, pinky, logname, tar and chown also bind the files-only nsswitch.conf over /etc/nsswitch.conf. The profile counts every failure as blocking, for example a rewritten binary that crashes or two runs whose output differs.",
615                mode.corpus_name()
616            ),
617        };
618        let mut step = node(
619            "compat",
620            &sanitize_job(&row.label),
621            &format!("{} compatibility: {}", mode.display_name(), row.label),
622            format!(
623                "{} {} </dev/null",
624                shell_join(&argv),
625                compat_guest_command(&row, paths)
626            ),
627            gate_dep.map(|d| vec![d.to_string()]).unwrap_or_default(),
628            wall,
629            COMPAT_CPU_TIMEOUT_S.max(wall),
630            COMPAT_MEM_BYTES,
631        );
632        step.description = description;
633        out.push(step);
634    }
635    if out.is_empty() {
636        return Err(format!("compatibility mode {mode:?} selected zero probes"));
637    }
638    Ok(out)
639}

644pub fn sanitize_job(label: &str) -> String {
645    label.replace('.', "_")

648/// Load the complete committed DAG and select one label with its dependency
649/// closure while preserving every top-level scheduler setting.
650pub fn lane_config(root: &Path, lane: &str) -> Result<DagConfig, String> {

657}

667/// Assemble a `DagConfig`, CARRYING every top-level field from `base`.
668///
669/// # Why this takes a base at all
670///
671/// It used to be `DagConfig { steps, ..Default::default() }`, which loaded a DAG
672/// file, kept its steps, and threw its configuration away. That is not a
673/// hypothetical: it hung a full validate for 14 minutes at 0% CPU.
674/// `ci/dag/validate.json` declared `resource_caps {manifest_guest: 8}` (12 since 2026-10-17);
675/// dropping it leaves `res_free` evaluating `unwrap_or(0) >= 1` for the 13
676/// steps demanding `manifest_guest`, so none can be admitted. The scheduler's
677/// only exit is `running.is_empty() && done + skipped >= steps.len()`, so with
678/// work neither runnable nor accounted it slept at 50 ms forever -- no error, no
679/// exit, 21 of ~58 nodes done.
680///
681/// `resource_caps` failed LOUDLY (a visible hang). The quieter one matters more:
682/// `default_step_timeout` is 600 s in portable and 120 s in privileged, and
683/// reverted to `DEFAULT_STEP_TIMEOUT` (1800 s) -- every step's wall cap loosened
684/// 3x and 15x respectively, with nothing to see. `mem_cap_factor`,
685/// `mem_cap_floor_bytes` and `outer_mem_safety_factor` happen to equal their
686/// defaults today, so they would have broken the first time anyone tuned them.
687///
688/// Hence: carry the base wholesale, and let [`assert_config_carried`] prove it.
689pub fn config_from_base(base: &DagConfig, steps: Vec<Step>, description: &str) -> DagConfig {
690    let mut cfg = base.clone();
691    cfg.steps = steps;
692    cfg.description = description.to_string();
693    // Synthetic generator/self-test configs use a bounded fallback. Production
694    // selections retain the explicit per-node budgets in the committed DAG.
695    cfg.default_step_cpu_timeout = SYNTHETIC_DEFAULT_CPU_TIMEOUT_S;
696    cfg
697}

699/// Synthesised plans that have no source DAG file (compat, quick, envelope, ...).
700pub fn config_from(steps: Vec<Step>, description: &str) -> DagConfig {
701    config_from_base(&DagConfig::default(), steps, description)
702}

704/// Field-by-field proof that `derived` carried `base`'s configuration.
705///
706/// Enumerated deliberately rather than derived from a `PartialEq`: a new
707/// `DagConfig` field must force a decision here instead of silently defaulting,
708/// which is the exact failure this function exists to prevent. `steps` and
709/// `description` are expected to differ; `default_step_cpu_timeout` is the one
710/// documented divergence above.
711pub fn assert_config_carried(base: &DagConfig, derived: &DagConfig) -> Result<(), String> {
712    let mut bad: Vec<String> = Vec::new();
713    if base.resource_caps != derived.resource_caps {
714        bad.push(format!(
715            "resource_caps {:?} != {:?}",
716            base.resource_caps, derived.resource_caps
717        ));
718    }
719    if base.mem_cap_factor != derived.mem_cap_factor {
720        bad.push(format!(
721            "mem_cap_factor {} != {}",
722            base.mem_cap_factor, derived.mem_cap_factor
723        ));
724    }
725    if base.mem_cap_floor_bytes != derived.mem_cap_floor_bytes {
726        bad.push(format!(
727            "mem_cap_floor_bytes {} != {}",
728            base.mem_cap_floor_bytes, derived.mem_cap_floor_bytes
729        ));
730    }
731    if base.outer_mem_safety_factor != derived.outer_mem_safety_factor {
732        bad.push(format!(
733            "outer_mem_safety_factor {} != {}",
734            base.outer_mem_safety_factor, derived.outer_mem_safety_factor
735        ));
736    }
737    if base.default_step_timeout != derived.default_step_timeout {
738        bad.push(format!(
739            "default_step_timeout {} != {}",
740            base.default_step_timeout, derived.default_step_timeout
741        ));
742    }
743    if base.default_jobs_flag != derived.default_jobs_flag {
744        bad.push(format!(
745            "default_jobs_flag {:?} != {:?}",
746            base.default_jobs_flag, derived.default_jobs_flag
747        ));
748    }
749    if base.default_jobs_env != derived.default_jobs_env {
750        bad.push(format!(
751            "default_jobs_env {:?} != {:?}",
752            base.default_jobs_env, derived.default_jobs_env
753        ));
754    }
755    if base.default_step_mem_cap_bytes != derived.default_step_mem_cap_bytes {
756        bad.push(format!(
757            "default_step_mem_cap_bytes {:?} != {:?}",
758            base.default_step_mem_cap_bytes, derived.default_step_mem_cap_bytes
759        ));
760    }
761    if base.default_step_cpu_count != derived.default_step_cpu_count {
762        bad.push(format!(
763            "default_step_cpu_count {:?} != {:?}",
764            base.default_step_cpu_count, derived.default_step_cpu_count
765        ));
766    }
767    if bad.is_empty() {
768        Ok(())
769    } else {
770        Err(bad.join("; "))
771    }
772}

774/// FAIL CLOSED on capacity that can never be granted.
775///
776/// A step demanding a resource the config does not cap is unschedulable FOREVER,
777/// and the scheduler expresses that as an infinite 50 ms sleep rather than an
778/// error. Refusing up front converts a silent 14-minute hang into a named
779/// refusal before a single node runs.
780pub fn ungrantable_resources(cfg: &DagConfig) -> Vec<String> {
781    let mut bad = Vec::new();
782    for s in &cfg.steps {
783        for (r, n) in &s.hint.resources {
784            let cap = cfg.resource_caps.get(r).copied().unwrap_or(0);
785            if cap < *n {
786                bad.push(format!(
787                    "{} demands {r}={n} but resource_caps grants {cap}",
788                    s.tag()
789                ));
790            }
791        }
792    }
793    bad
794}
