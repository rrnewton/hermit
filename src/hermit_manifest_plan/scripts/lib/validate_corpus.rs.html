7//! Compatibility-corpus tables for the validate driver.
8//!
9//! # Where these rows came from (provenance matters here)
10//!
11//! `ci/compat/corpus-<mode>.json` was NOT hand-transcribed from `validate.sh`.
12//! Hand-copying ~750 probe invocations across four modes is exactly the kind of
13//! silent-drift transcription a reviewer cannot audit. Instead the tables were
14//! lifted MECHANICALLY: `strict_compatibility_probe` was temporarily intercepted
15//! so that, instead of executing, it recorded its exact `(label, argv)` — after
16//! every `COMPATIBILITY_MODE` gate in `run_compatibility_corpus` had already been
17//! evaluated by the real bash. The dump therefore reflects the code path, not a
18//! reading of it.
19//!
20//! The extraction is corroborated by the counts the bash itself declares:
21//! `strict` dumped **191** rows against `STRICT_COMPAT_TOTAL=191`, and `sabre`
22//! dumped **212** against `SABRE_COMPAT_TOTAL=212` — exact, independent matches.
23//! (The SaBRe rows have since moved into tests/e2e/manifests/compat.yaml as the
24//! sabre-compat-only run type's cells, and `corpus-sabre.json` is gone.)
25//! `rr` dumped 174 admitted rows which the driver then filtered to the 139
26//! `RR_COMPAT_PASSING_LABELS` (the bash filtered at the same point, inside
27//! `rr_compatibility_probe`), and `e9patch` dumps 172 admitted rows. (The rr
28//! rows have since moved into tests/e2e/manifests/compat.yaml as the
29//! rr-compat-only run type's replay cells, and `corpus-rr.json` is gone.)
30//!
31//! # Why a data file rather than Rust literals
32//!
33//! The corpus is DATA, and it changes on a different clock than the driver: rows
34//! are added when a program becomes compatible. Keeping it as JSON means a corpus
35//! change is a reviewable data diff, and it keeps the driver's control flow
36//! readable instead of burying it under 750 argv literals.
37
38use std::collections::BTreeMap;
39use std::path::Path;

41/// One corpus probe: a stable label and the exact guest argv to run under Hermit.
42#[derive(Clone, Debug)]
43pub struct CorpusRow {
44    pub label: String,
45    pub argv: Vec<String>,
46}
47
48/// Runtime values substituted into the `{{PLACEHOLDER}}` tokens the extractor left
49/// behind. Each corresponds to a per-run path the bash computed at startup, so the
50/// checked-in table stays host- and run-independent.
51pub struct CorpusPaths<'a> {
52    pub root_dir: &'a str,
53    pub real_compat_fixtures: &'a str,
54    pub validation_tmp_dir: &'a str,
55    pub shell_build_dir: &'a str,
56}
57
58impl CorpusPaths<'_> {
59    fn expand(&self, arg: &str) -> String {
60        arg.replace("{{ROOT_DIR}}", self.root_dir)
61            .replace("{{REAL_COMPAT_FIXTURES}}", self.real_compat_fixtures)
62            .replace("{{VALIDATION_TMP_DIR}}", self.validation_tmp_dir)
63            .replace("{{SHELL_BUILD_DIR}}", self.shell_build_dir)
64    }
65}

67/// Load `ci/compat/corpus-<mode>.json` and expand its placeholders.
68///
69/// Fails LOUDLY rather than returning an empty corpus: a silently-empty corpus
70/// would turn a compatibility gate into a no-op that still reports "pass", which
71/// is the zero-executed-tests defect class in a new costume.
72pub fn load(root: &Path, mode: &str, paths: &CorpusPaths) -> Result<Vec<CorpusRow>, String> {
73    let file = root
74        .join("ci")
75        .join("compat")
76        .join(format!("corpus-{mode}.json"));
77    let text = std::fs::read_to_string(&file)
78        .map_err(|e| format!("cannot read compatibility corpus {}: {e}", file.display()))?;
79    let doc: serde_json::Value = serde_json::from_str(&text)
80        .map_err(|e| format!("invalid JSON in {}: {e}", file.display()))?;
81    let rows = doc
82        .get("rows")
83        .and_then(|r| r.as_array())
84        .ok_or_else(|| format!("{} has no `rows` array", file.display()))?;
85    let mut out = Vec::with_capacity(rows.len());
86    for (i, row) in rows.iter().enumerate() {
87        let label = row
88            .get("label")
89            .and_then(|v| v.as_str())
90            .ok_or_else(|| format!("{} row {i}: missing string `label`", file.display()))?;
91        let argv_raw = row
92            .get("argv")
93            .and_then(|v| v.as_array())
94            .ok_or_else(|| format!("{} row {i}: missing array `argv`", file.display()))?;
95        let mut argv = Vec::with_capacity(argv_raw.len());
96        for a in argv_raw {
97            let s = a.as_str().ok_or_else(|| {
98                format!(
99                    "{} row {i} ({label}): non-string argv element",
100                    file.display()
101                )
102            })?;
103            argv.push(paths.expand(s));
104        }
105        if argv.is_empty() {
106            return Err(format!("{} row {i} ({label}): empty argv", file.display()));
107        }
108        out.push(CorpusRow {
109            label: label.to_string(),
110            argv,
111        });
112    }
113    if out.is_empty() {
114        return Err(format!("{} contained zero rows", file.display()));
115    }
116    Ok(out)
117}

121/// `STRICT_COMPAT_TOTAL` (validate.sh:1108). The corpus contains semantic
122/// workloads only; banner-only probes were removed when the E2E harness landed.
123///
124/// 191 at extraction. 193 since the `netlink-route` and `netlink-sock-diag`
125/// rows were added: the corpus reached netlink only incidentally through `ip`,
126/// `ss`, and every `python3` row, none of which asserted anything about a reply,
127/// so no cell would have failed if detcore stopped normalizing netlink at all.
128///
129/// The strict-compat-only run type no longer reads this file: since

137/// `COMPAT_SUMMARY_KNOWN_FAILURES` (validate.sh:1135). Tracked gaps excluded from
138/// the executable corpus but retained in the canonical denominator and table.
139/// Under `strict` these are NONBLOCKING: the row keeps running so the gap stays
140/// visible, mirroring the gcc vfork precedent.
141///
142/// Empty today. `make` was listed for a fail-closed `setresuid` refusal; that syscall has
143/// been determinized, and the exact `compat.make` command now passes (validate run 213,
144/// measured at a pre-rebase commit of this change, plus five focused reruns and a
145/// `--verify-strict` L2 match), so its row was a stale expectation that only produced a
146/// STALE warning. A `make` failure now blocks.
147pub fn known_failclosed() -> BTreeMap<&'static str, &'static str> {
148    BTreeMap::new()
149}
150
151/// `PORTABLE_STRICT_DIAGNOSTIC_FAILURES` (validate.sh:1147). Bounded diagnostics
152/// on the GitHub-managed portable runner: a failure here is nonblocking and the
153/// probe is given a shortened 20s budget.
154///
155/// Empty since 2026-10-07, when the compat corpus became strict (--verify-strict):
156/// its five rows (df, ranlib, top, zstd, zstd-roundtrip) passed strict in 4 of 4
157/// full runs at Hermit 0d25b6f7 and are ordinary blocking cells.

159    BTreeMap::new()
160}
