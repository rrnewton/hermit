23/// `GATE_TIMEOUT_SECONDS` (validate.sh:400). A generated super node without
24/// an explicit override inherits this historical default before it is committed.
25pub const DEFAULT_GATE_TIMEOUT_S: i64 = 600;
26
27/// `SUPER_REPETITIONS` (validate.sh:682).
28pub const SUPER_REPETITIONS_DEFAULT: i64 = 20;
29
30/// `STRICT_COMPAT_TIMEOUT` (validate.sh:1091) — the per-repetition wall bound
31/// the bash imposed with the `timeout` binary. All repetitions of a probe share
32/// one node, so this bound is a `timeout` around each repetition's Hermit run,
33/// and a repetition it kills is recorded as that repetition's failure.
34pub const SUPER_PROBE_TIMEOUT_S: i64 = 60;
35
36/// Grace between `timeout`'s SIGTERM and its SIGKILL for one repetition.
37const SUPER_PROBE_KILL_GRACE_S: i64 = 10;
38
39/// CPU budget for one stress repetition. These are sub-second guest runs; a CPU
40/// cap is what catches a spin that the wall bound would only catch at 60s. The
41/// node's CPU cap is this budget times the repetition count, and the node also
42/// fails any single repetition whose waited CPU time reached this budget.
43const SUPER_PROBE_CPU_TIMEOUT_S: i64 = 120;
44
45/// Environment variable through which dagrun hands a probe node its admitted
46/// width: the number of repetitions the node runs at the same time.
47const SUPER_PROBE_JOBS_ENV: &str = "HERMIT_SUPER_STRESS_JOBS";
48
49/// Wall slack for a probe node beyond its repetitions' bounds: the loop, the
50/// record probe's data-directory removal, and the result write. The node's
51/// wall cap is every repetition's bound end to end plus this slack, so it
52/// still holds if dagrun admits the node at width 1.
53const SUPER_PROBE_NODE_SLACK_S: i64 = 60;
54
55/// Memory cap for one probe node, all of its concurrent repetitions included.
56/// Measured 2026-10-02 on the host recorded in docs/TESTING_ENVIRONMENTS.md
57/// under "Named measurement hosts", with 20 repetitions at once, each Hermit
58/// call in its own safehermit cgroup: the 20 per-repetition peaks summed to
59/// 1052, 1226 and 1433 MiB for the strict, pipeline and record probes (largest
60/// single repetition 73 MiB), so 4 GiB is 2.8 times the largest sum. The KVM
61/// probe, measured the same way with the release binary, summed to 1363 MiB
62/// (largest 71 MiB). The DBT probe is unmeasured: no Hermit binary on the host
63/// was built with the dbt feature. It is nonblocking, and if it reaches this
64/// cap the node writes no rows, so the pass-rate table shows NO_RESULT.
65const SUPER_PROBE_MEM_BYTES: i64 = 4 * 1024 * 1024 * 1024;

79fn apply_innermost_timeout_runner(argv: &mut Vec<String>, root: &str) -> bool {
80    let Some(cargo) = argv.windows(2).position(|words| words == ["cargo", "test"]) else {
81        return false;
82    };
83    argv[cargo] = format!("{root}/ci/run-nextest-counted.sh");
84    argv.remove(cargo + 1);
85    let mut jobs = None;
86    let mut no_capture = false;
87    argv.retain(|argument| {
88        if let Some(value) = argument.strip_prefix("--test-threads=") {
89            jobs = Some(value.to_string());
90            false
91        } else if argument == "--nocapture" {
92            no_capture = true;
93            false
94        } else {
95            true
96        }
97    });
98    let split = argv
99        .iter()
100        .position(|argument| argument == "--")
101        .unwrap_or(argv.len());
102    if let Some(jobs) = jobs {
103        argv.splice(split..split, ["-j".to_string(), jobs]);
104    }
105    if no_capture {
106        let split = argv
107            .iter()
108            .position(|argument| argument == "--")
109            .unwrap_or(argv.len());
110        argv.insert(split, "--no-capture".to_string());
111    }
112    true
113}

115fn load_gates(root: &Path) -> Result<Vec<SuperGate>, String> {
116    let file = root.join("ci/super/gates.json");
117    let text = std::fs::read_to_string(&file)
118        .map_err(|error| format!("cannot read super gate table {}: {error}", file.display()))?;
119    let document: serde_json::Value = serde_json::from_str(&text)
120        .map_err(|error| format!("invalid JSON in {}: {error}", file.display()))?;
121    let rows = document
122        .get("rows")
123        .and_then(|rows| rows.as_array())
124        .ok_or_else(|| format!("{} has no `rows` array", file.display()))?;
125    let root = root.to_string_lossy();
126    let mut gates = Vec::with_capacity(rows.len());
127    for (index, row) in rows.iter().enumerate() {
128        let string = |field: &str| {
129            row.get(field)
130                .and_then(|value| value.as_str())
131                .map(str::to_string)
132        };
133        let job = string("job")
134            .ok_or_else(|| format!("{} row {index}: missing string `job`", file.display()))?;
135        let label = string("label")
136            .ok_or_else(|| format!("{} row {index}: missing string `label`", file.display()))?;
137        let timeout = row
138            .get("timeout")
139            .and_then(|value| value.as_i64())
140            .unwrap_or(0);
141        let raw = row
142            .get("argv")
143            .and_then(|value| value.as_array())
144            .ok_or_else(|| {
145                format!(
146                    "{} row {index} ({label}): missing array `argv`",
147                    file.display()
148                )
149            })?;
150        let mut argv = Vec::with_capacity(raw.len());
151        for argument in raw {
152            let argument = argument.as_str().ok_or_else(|| {
153                format!(
154                    "{} row {index} ({label}): non-string argv element",
155                    file.display()
156                )
157            })?;
158            argv.push(argument.replace("{{ROOT_DIR}}", &root));
159        }
160        apply_innermost_timeout_runner(&mut argv, &root);
161        let synthetic = string("synthetic");
162        if synthetic.is_none() && argv.is_empty() {
163            return Err(format!(
164                "{} row {index} ({label}): empty argv",
165                file.display()
166            ));
167        }
168        gates.push(SuperGate {
169            job,
170            label,
171            timeout,
172            argv,
173            synthetic,
174        });
175    }
176    if gates.is_empty() {
177        return Err(format!("{} contained zero rows", file.display()));
178    }
179    Ok(gates)
180}

184/// The five probes `run_super_stress_suite` names (validate.sh:2686, :2695, :2702).
185#[derive(Clone, Copy, PartialEq, Eq, Debug)]
186pub enum StressProbe {
187    PtraceStrictVerify,
188    PtracePipeline,
189    PtraceRecordReplay,
190    KvmVerify,
191    DbtVerify,
192}
193
194impl StressProbe {
195    pub fn slug(self) -> &'static str {
196        match self {
197            StressProbe::PtraceStrictVerify => "ptrace-strict-verify",
198            StressProbe::PtracePipeline => "ptrace-pipeline",
199            StressProbe::PtraceRecordReplay => "ptrace-record-replay",
200            StressProbe::KvmVerify => "kvm-verify",
201            StressProbe::DbtVerify => "dbt-verify",
202        }
203    }
204
205    pub fn job_stem(self) -> String {
206        self.slug().replace('-', "_")
207    }
208
209    /// The availability node this probe depends on, if any.
210    fn availability_job(self) -> Option<&'static str> {
211        match self {
212            StressProbe::KvmVerify => Some("kvm_available"),
213            StressProbe::DbtVerify => Some("dbt_available"),
214            StressProbe::PtraceStrictVerify
215            | StressProbe::PtracePipeline
216            | StressProbe::PtraceRecordReplay => None,
217        }
218    }
219
220    /// True when a failure of this probe must NOT turn the suite red.
221    ///
222    /// See the module doc: `backend_selector_supported` is undefined, so KVM and
223    /// DBT stress have never actually been measured by `validate.sh`. Their
224    /// first measurement is reported, not ratcheted.
225    pub fn nonblocking(self) -> bool {
226        matches!(self, StressProbe::KvmVerify | StressProbe::DbtVerify)
227    }
228
229    /// The node's inventory paragraph.
230    fn description(self, reps: i64) -> String {
231        let (what, binary) = match self {
232            StressProbe::PtraceStrictVerify => (
233                "hermit run --strict --verify of /bin/echo on the ptrace backend",
234                "the release Hermit binary",
235            ),
236            StressProbe::PtracePipeline => (
237                "hermit run --strict --verify of a bash pipeline (yes | head -n 64 | sha256sum) \
238                 on the ptrace backend",
239                "the release Hermit binary",
240            ),
241            StressProbe::PtraceRecordReplay => (
242                "hermit record start --verify of /bin/echo on the ptrace backend, recording into \
243                 a per-repetition data directory under the validation temporary directory and \
244                 removing it before and after",
245                "the release Hermit binary",
246            ),
247            StressProbe::KvmVerify => (
248                "hermit --backend kvm run --verify of /bin/echo",
249                "the debug Hermit binary, after superstress.kvm_available finds /dev/kvm readable \
250                 and writable",
251            ),
252            StressProbe::DbtVerify => (
253                "hermit --backend dbt run --verify of /bin/echo",
254                "the debug Hermit binary, after superstress.dbt_available completes one DBT run",
255            ),
256        };
257        let policy = if self.nonblocking() {
258            "Its failures are reported but do not block the super run: validate.sh never \
259             measured this probe (its backend guard was always false), so its first measurements \
260             are reported rather than ratcheted."
261        } else {
262            "Any failed repetition fails the node and blocks the super run."
263        };
264        format!(
265            "Runs {reps} repetitions of {what}, using {binary}. The repetitions run at the same \
266             time, as many at once as the node's admitted width ({SUPER_PROBE_JOBS_ENV}, declared \
267             as {reps}), so the probe still loads the host the way {reps} separate nodes did. Each \
268             repetition runs in its own background subshell under timeout \
269             {SUPER_PROBE_TIMEOUT_S} s (SIGKILL {SUPER_PROBE_KILL_GRACE_S} s later) and fails if \
270             its waited CPU time, read with the bash times builtin, reaches \
271             {SUPER_PROBE_CPU_TIMEOUT_S} s. Each repetition's output goes to its own log, which \
272             the node prints, indented and tagged with the repetition number, under that \
273             repetition's result line; the node writes each repetition as a pass or fail row of its schema-2 structured test results, which the super \
274             stress pass-rate table counts. {policy} If the node itself is killed by its wall, \
275             CPU or memory cap, it writes no rows and the table shows the probe as NO_RESULT; the \
276             node's failure still counts. Until 2026-10 each repetition was its own DAG node; \
277             folding them into one node per probe keeps the commands, the repetition count, the \
278             concurrency and the per-repetition wall and CPU bounds, and removes {} nodes from the \
279             plan.",
280            reps - 1
281        )
282    }
283
284    /// One repetition's shell command, reproducing `super_probe_command`
285    /// (validate.sh:2589). `iteration` is the shell expression naming the
286    /// repetition, and `bound` prefixes the Hermit invocation with that
287    /// repetition's wall bound: the repetitions share one node, so the node's
288    /// own wall cap can no longer say which repetition hung.
289    fn command(
290        self,
291        iteration: &str,
292        bound: &str,
293        release_bin: &str,
294        debug_bin: &str,
295        tmp: &Path,
296    ) -> String {
297        let rel = shell_quote(release_bin);
298        let dbg = shell_quote(debug_bin);
299        match self {
300            StressProbe::PtraceStrictVerify => format!(
301                "{bound}{rel} run --strict --verify -- /bin/echo hermit-super-{iteration} </dev/null"
302            ),
303            StressProbe::PtracePipeline => format!(
304                "{bound}{rel} run --strict --verify -- bash -c 'yes hermit | head -n 64 | sha256sum' </dev/null"
305            ),
306            StressProbe::PtraceRecordReplay => {
307                let dir = format!(
308                    "{}/super-record-{iteration}",
309                    shell_quote(&tmp.to_string_lossy())
310                );
311                // The bash removed the data dir before AND after, preserving the
312                // record phase's exit status across the second removal.
313                format!(
314                    "rm -rf {dir}; {bound}{rel} record start --verify --data-dir {dir} -- \
315                     /bin/echo hermit-super-record-{iteration} </dev/null; \
316                     status=$?; rm -rf {dir}; exit $status"
317                )
318            }
319            StressProbe::KvmVerify => format!(
320                "{bound}{dbg} --backend kvm run --verify -- /bin/echo hermit-super-kvm-{iteration} </dev/null"
321            ),
322            StressProbe::DbtVerify => format!(
323                "{bound}{dbg} --backend dbt run --verify -- /bin/echo hermit-super-dbt-{iteration} </dev/null"
324            ),
325        }
326    }
327
328    /// The node that runs every repetition of this probe.
329    ///
330    /// The separate nodes ran concurrently, so the repetitions do too: each is
331    /// a background subshell under `timeout` with the per-repetition wall bound
332    /// the separate nodes had, at most the admitted width at once. Each records
333    /// its exit status, its waited CPU time (`times`, children line) and its
334    /// output in its own files, and the node fails a repetition whose CPU time reached the
335    /// per-repetition budget, which the separate nodes' CPU caps enforced. Each
336    /// repetition is a row of the node's structured test results, written once
337    /// by the registered writer `ci/write-structured-test-counts.sh`, so a
338    /// failed repetition stays typed evidence with its own identity. The node
339    /// fails when any repetition fails.
340    fn node_command(self, reps: i64, release_bin: &str, debug_bin: &str, tmp: &Path) -> String {
341        let slug = self.slug();
342        // --verbose makes timeout say which signal it sent, in the repetition's
343        // log, so the result line names a timeout only when timeout sent one: an
344        // exit status of 124 or 137 alone could also be Hermit's own. The line
345        // must also agree with the exit status timeout returns after sending
346        // that signal, because the guest's output shares the log. The outer
347        // subshell writes to the log too, so bash's report of a killed job is
348        // tagged with its repetition rather than left untagged on stderr.
349        let bound = format!(
350            "timeout --verbose --kill-after={SUPER_PROBE_KILL_GRACE_S} {SUPER_PROBE_TIMEOUT_S} "
351        );
352        let repetition = self.command("$rep", &bound, release_bin, debug_bin, tmp);
353        let jobs_env = SUPER_PROBE_JOBS_ENV;
354        format!(
355            ": \"${{DAGRUN_TEST_COUNTS_PATH:?the structured result path is unset}}\"; \
356             width=${{{jobs_env}:-{reps}}}; \
357             case $width in ''|*[!0-9]*) echo \"super stress {slug}: {jobs_env}=$width is not a whole number\" >&2; exit 2;; esac; \
358             [ \"$width\" -ge 1 ] || width=1; [ \"$width\" -le {reps} ] || width={reps}; \
359             state=$(mktemp -d) || exit 2; trap 'rm -rf \"$state\"' EXIT; \
360             running=0; \
361             for rep in $(seq 1 {reps}); do \
362             if [ \"$running\" -ge \"$width\" ]; then wait -n; running=$((running - 1)); fi; \
363             ( ( {repetition} ); rc=$?; times >\"$state/$rep.times\"; echo \"$rc\" >\"$state/$rep.rc\" ) >\"$state/$rep.log\" 2>&1 & \
364             running=$((running + 1)); \
365             done; \
366             wait; \
367             rows=(); failed=0; \
368             for rep in $(seq 1 {reps}); do \
369             rc=$(cat \"$state/$rep.rc\" 2>/dev/null); rc=${{rc:-none}}; \
370             cpu=$(awk 'NR == 2 {{ gsub(/s/, \"\"); split($1, u, \"m\"); split($2, k, \"m\"); printf \"%.2f\", u[1] * 60 + u[2] + k[1] * 60 + k[2] }}' \"$state/$rep.times\" 2>/dev/null); cpu=${{cpu:-0}}; \
371             log=\"$state/$rep.log\"; \
372             if [ \"$rc\" = 137 ] && grep -q 'timeout: sending signal KILL' \"$log\" 2>/dev/null; then cause=' (timeout sent SIGKILL {SUPER_PROBE_KILL_GRACE_S} s after the {SUPER_PROBE_TIMEOUT_S} s bound)'; \
373             elif [ \"$rc\" = 124 ] && grep -q 'timeout: sending signal TERM' \"$log\" 2>/dev/null; then cause=' (timed out after {SUPER_PROBE_TIMEOUT_S} s)'; \
374             else case $rc in \
375             none) cause=' (wrote no exit status)';; \
376             137) cause=' (SIGKILL not sent by its timeout, e.g. the OOM killer)';; \
377             *) cause=;; \
378             esac; fi; \
379             result=pass; [ \"$rc\" = 0 ] || result=fail; \
380             if [ \"${{cpu%.*}}\" -ge {SUPER_PROBE_CPU_TIMEOUT_S} ]; then result=fail; cause=\"$cause (used $cpu CPU seconds; the per-repetition bound is {SUPER_PROBE_CPU_TIMEOUT_S})\"; fi; \
381             [ \"$result\" = pass ] || failed=$((failed + 1)); \
382             echo \"super stress {slug} repetition $rep/{reps}: $result, exit $rc, $cpu CPU s$cause\"; \
383             awk -v r=\"$rep\" '{{ print \"  [\" r \"] \" $0 }}' \"$log\" 2>/dev/null; \
384             rows+=(\"$(printf '{slug}/repetition-%02d' \"$rep\")\" \"$result\" 1); \
385             done; \
386             ./ci/write-structured-test-counts.sh {reps} 0 \"${{rows[@]}}\" || exit 2; \
387             echo \"super stress {slug}: $(({reps} - failed))/{reps} repetitions passed, up to $width at a time\"; \
388             [ \"$failed\" -eq 0 ]"
389        )
390    }
391}

401/// The two backend-availability nodes.
402///
403/// `kvm_backend_available` (validate.sh:2272) is a readable+writable `/dev/kvm`;
404/// `dbt_backend_available` (validate.sh:2276) is a real probe run, which is why
405/// it must be a node — at plan time the debug binary does not exist yet.
406fn availability_nodes(debug_bin: &str, build_dep: &str, reps: i64) -> Vec<Step> {
407    let dbg = shell_quote(debug_bin);
408    vec![
409        described(
410            node(
411                "superstress",
412                "kvm_available",
413                "KVM backend availability (gates the KVM stress rows)",
414                "test -r /dev/kvm && test -w /dev/kvm".to_string(),
415                vec![build_dep.to_string()],
416                30,
417                30,
418                256 * 1024 * 1024,
419            ),
420            format!(
421                "Runs `test -r /dev/kvm && test -w /dev/kvm` after {build_dep}, the port of \
422                 validate.sh's kvm_backend_available. Its only dependent is superstress.kvm_verify, \
423                 which runs {reps} repetitions of `hermit --backend kvm run --verify` of /bin/echo \
424                 with the debug binary; when this node fails, that node is skipped and the super \
425                 stress table prints \"SKIP kvm-verify backend unavailable (availability node \
426                 failed; 0/{reps} ran)\". Like the KVM stress node, it is nonblocking in the super \
427                 profile, so a failure is reported but does not turn the run red. A host or container \
428                 without /dev/kvm, or one where the validating user cannot open it read-write, makes \
429                 it fail."
430            ),
431        ),
432        described(
433            node(
434                "superstress",
435                "dbt_available",
436                "DBT backend availability (gates the DBT stress rows)",
437                format!(
438                    "{dbg} --log=info --backend dbt run --strict --verify -- \
439                 /bin/echo hermit-dbt-probe </dev/null >/dev/null 2>&1"
440                ),
441                vec![build_dep.to_string()],
442                60,
443                120,
444                SUPER_PROBE_MEM_BYTES,
445            ),
446            format!(
447                "Runs one real DBT run with the debug Hermit after {build_dep}: `hermit --log=info \
448             --backend dbt run --strict --verify -- /bin/echo hermit-dbt-probe`, the port of \
449                 validate.sh's dbt_backend_available. It is a node because the debug binary does not \
450                 exist when the plan is built. Its only dependent is superstress.dbt_verify, {reps} \
451                 repetitions of `hermit --backend dbt run --verify` of /bin/echo; when this probe \
452                 fails, that node is skipped and the super stress table prints \"SKIP dbt-verify \
453                 backend unavailable\". Both nodes are nonblocking in the super profile, so their \
454                 failures are reported, not gated. A strict-mode refusal or a divergence between the \
455                 two runs makes it exit nonzero; its output goes to /dev/null, so the node log does \
456                 not say which."
457            ),
458        ),
459    ]
460}

468/// Build every stress node: two availability probes plus one node per probe
469/// that runs its `reps` repetitions.
470pub fn stress_nodes(
471    release_bin: &str,
472    debug_bin: &str,
473    tmp: &Path,
474    reps: i64,
475    release_dep: &str,
476    debug_dep: &str,
477) -> Vec<Step> {
478    let mut out = availability_nodes(debug_bin, debug_dep, reps);
479    for probe in STRESS_PROBES {
480        let stem = probe.job_stem();
481        let base_dep = match probe {
482            StressProbe::KvmVerify | StressProbe::DbtVerify => debug_dep,
483            _ => release_dep,
484        };
485        let mut deps = vec![base_dep.to_string()];
486        if let Some(av) = probe.availability_job() {
487            deps.push(format!("superstress.{av}"));
488        }
489        let per_repetition = SUPER_PROBE_TIMEOUT_S + SUPER_PROBE_KILL_GRACE_S;
490        let mut step = node(
491            "superstress",
492            &stem,
493            &format!("super stress {} ({reps} repetitions)", probe.slug()),
494            probe.node_command(reps, release_bin, debug_bin, tmp),
495            deps,
496            reps * per_repetition + SUPER_PROBE_NODE_SLACK_S,
497            reps * SUPER_PROBE_CPU_TIMEOUT_S,
498            SUPER_PROBE_MEM_BYTES,
499        );
500        // Declare the width the repetitions run at, so dagrun reserves (and
501        // its cpu.max allows) that many cores. jobs_flag is empty: the width
502        // reaches the command through SUPER_PROBE_JOBS_ENV, never a `-j`.
503        step.hint.preferred_inner_jobs = Some(reps);
504        step.jobs_env = Some(SUPER_PROBE_JOBS_ENV.to_string());
505        step.jobs_flag = Some(String::new());
506        step.description = probe.description(reps);
507        step.result_manifests = Some(vec![ResultManifest::StructuredTestResults(
508            StructuredTestResultsManifest::current(format!("superstress.{stem}")),
509        )]);
510        out.push(step);
511    }
512    out
513}

525/// Recompute `run_super_probe`'s report from typed `StepOutcome`s.
526///
527/// The bash scraped its own tee'd text file (`$VALIDATION_TMP_DIR/super-report`);
528/// this reads the runner's structured verdicts, so the printed rate and the
529/// blocking decision cannot disagree with what actually ran.
530pub fn stress_rates(outcomes: &[StepOutcome], reps: i64) -> Vec<ProbeRate> {
531    let mut rates = Vec::new();
532    for probe in STRESS_PROBES {
533        let tag = format!("superstress.{}", probe.job_stem());
534        // A repetition ran when the probe node reported a result row for it. A
535        // node that was skipped, aborted, or exited without its result file ran
536        // none, which the verdict reports as NO_RESULT rather than as passes.
537        let rows = outcomes
538            .iter()
539            .filter(|o| o.tag == tag && !o.aborted)
540            .filter_map(|o| o.test_results.as_ref())
541            .flatten();
542        let mut passed = 0usize;
543        let mut ran = 0usize;
544        for row in rows {
545            ran += 1;
546            if row.passed {
547                passed += 1;
548            }
549        }
550        rates.push(ProbeRate {
551            probe: *probe,
552            passed,
553            ran,
554            planned: reps as usize,
555        });
556    }
557    rates
558}

560/// Print the pass-rate table and return the BLOCKING failure count.
561///
562/// A probe is blocking iff it is a ptrace probe (the three the bash actually
563/// measured) and it did not pass every planned repetition. KVM/DBT rates are
564/// printed with the reason they are nonblocking, so the number is visible
565/// without silently becoming a gate on its first appearance.
566pub fn stress_verdict(rates: &[ProbeRate], reps: i64, jobs: i64, host_cpus: usize) -> usize {
567    println!("\n== Super stress pass rates ==");
568    println!("Repetitions: {reps}; scheduler width: {jobs}; online CPUs: {host_cpus}");
569    let mut blocking = 0usize;
570    for r in rates {
571        let slug = r.probe.slug();
572        if r.ran == 0 {
573            println!(
574                "  SKIP {slug:<24} backend unavailable (availability node failed; 0/{reps} ran)"
575            );
576            continue;
577        }
578        let pct = 100 * r.passed / r.planned.max(1);
579        if r.passed == r.planned {
580            println!("  ✅ {slug:<24} {}/{} (100%)", r.passed, r.planned);
581        } else if r.probe.nonblocking() {
582            println!(
583                "  ⚠️  {slug:<24} {}/{} ({pct}%) FLAKY/FAILING — NONBLOCKING: this row was dead \
584                 code in validate.sh (`backend_selector_supported` is undefined, so the guard was \
585                 always false) and has never been measured; reporting it, not ratcheting it.",
586                r.passed, r.planned
587            );
588        } else {
589            println!(
590                "  ⚠️  {slug:<24} {}/{} ({pct}%) FLAKY/FAILING",
591                r.passed, r.planned
592            );
593            blocking += 1;
594        }
595    }
596    blocking
597}

601/// Focused controls for the reporting policy that remains live after the
602/// committed DAG became the sole source of super-plan construction.
603pub fn self_test(root: &Path) -> Result<String, String> {
604    let gates = load_gates(root)?;
605    // 31 since the liteinst_python3_verify_diagnostics row left with the
606    // ptrace-hosted LiteInst hybrid (https://github.com/rrnewton/hermit/issues/3520);
607    // it had already stopped selecting any test.
608    if gates.len() != 31 {
609        return Err(format!(
610            "super source table has {} rows; the mechanical extraction requires exactly 31",
611            gates.len()
612        ));
613    }
614    let synthetic = gates
615        .iter()
616        .filter_map(|gate| gate.synthetic.as_deref())
617        .collect::<Vec<_>>();
618    let expected_synthetic = [
619        "portable_slow_strict_diagnostics",
620        "super_stress_suite",
621        "calibrated_analyze_tests",
622    ];
623    if synthetic.len() != expected_synthetic.len()
624        || expected_synthetic
625            .iter()
626            .any(|name| !synthetic.contains(name))
627    {
628        return Err(format!(
629            "super source table synthetic rows changed: {synthetic:?}"
630        ));
631    }
632    let nextest_rows = gates
633        .iter()
634        .filter(|gate| {
635            gate.argv
636                .iter()
637                .any(|argument| argument.ends_with("/ci/run-nextest-counted.sh"))
638        })
639        .count();
640    if nextest_rows < 20
641        || gates
642            .iter()
643            .any(|gate| gate.argv.windows(2).any(|words| words == ["cargo", "test"]))
644    {
645        return Err(format!(
646            "super source table cargo-test conversion is incomplete: {nextest_rows} nextest rows"
647        ));
648    }
649    let calibrated = gates
650        .iter()
651        .find(|gate| gate.synthetic.as_deref() == Some("calibrated_analyze_tests"))
652        .ok_or_else(|| "super source table lost calibrated_analyze_tests".to_string())?;
653    let normalized_calibrated = calibrated
654        .argv
655        .iter()
656        .filter(|argument| !argument.starts_with("--test-threads="))
657        .cloned()
658        .collect::<Vec<_>>();
659    if normalized_calibrated
660        .iter()
661        .any(|argument| argument.starts_with("--test-threads="))
662        || normalized_calibrated.len() + 1 != calibrated.argv.len()
663        || calibrated.job.is_empty()
664        || calibrated.label.is_empty()
665        || calibrated.timeout < 0
666    {
667        return Err(format!(
668            "calibrated analyze source arguments were not normalized: {:?}",
669            calibrated.argv
670        ));
671    }
672    let committed = std::fs::read_to_string(root.join("ci/dag/validate.json"))
673        .map_err(|error| format!("cannot read committed super graph: {error}"))?;
674    let committed = dagrun::io::dag_from_json(&committed)
675        .map_err(|error| format!("cannot parse committed super graph: {error}"))?;
676    let emitted = committed
677        .steps
678        .iter()
679        .find(|step| step.tag() == "super.pmu_analyze_hello_race_stress_calibrated_skid")
680        .ok_or_else(|| "committed graph lost calibrated analyze node".to_string())?;
681    if !emitted.cmd.contains("./ci/run-nextest-counted.sh")
682        || emitted.cmd.contains("--test-threads=")
683    {
684        return Err(format!(
685            "committed calibrated analyze command was not normalized: {}",
686            emitted.cmd
687        ));
688    }
689
690    let bad_root = std::env::temp_dir().join(format!(
691        "validate-super-source-self-test-{}",
692        std::process::id()
693    ));
694    let bad_file = bad_root.join("ci/super/gates.json");
695    std::fs::create_dir_all(bad_file.parent().expect("fixture has a parent"))
696        .map_err(|error| format!("cannot create super negative fixture: {error}"))?;
697    let mut refused = 0;
698    for (description, body) in [
699        ("no rows array", r#"{"rows": {}}"#),
700        ("empty rows", r#"{"rows": []}"#),
701        (
702            "row without argv",
703            r#"{"rows":[{"job":"j","label":"l","timeout":0}]}"#,
704        ),
705        (
706            "non-string argv",
707            r#"{"rows":[{"job":"j","label":"l","argv":[7]}]}"#,
708        ),
709    ] {
710        std::fs::write(&bad_file, body)
711            .map_err(|error| format!("cannot write super negative fixture: {error}"))?;
712        if load_gates(&bad_root).is_ok() {
713            let _ = std::fs::remove_dir_all(&bad_root);
714            return Err(format!("super source loader accepted {description}"));
715        }
716        refused += 1;
717    }
718    let _ = std::fs::remove_dir_all(&bad_root);
719
720    let reps = 2;
721    let all_green = STRESS_PROBES
722        .iter()
723        .copied()
724        .map(|probe| ProbeRate {
725            probe,
726            passed: reps as usize,
727            ran: reps as usize,
728            planned: reps as usize,
729        })
730        .collect::<Vec<_>>();
731    if stress_verdict(&all_green, reps, 1, 1) != 0 {
732        return Err("super stress verdict: an all-passing population must be accepted".into());
733    }
734
735    let mut ptrace_miss = all_green.clone();
736    ptrace_miss[0].passed -= 1;
737    if stress_verdict(&ptrace_miss, reps, 1, 1) != 1 {
738        return Err("super stress verdict: a ptrace miss must remain blocking".into());
739    }
740
741    let mut kvm_miss = all_green;
742    let kvm = kvm_miss
743        .iter_mut()
744        .find(|rate| rate.probe == StressProbe::KvmVerify)
745        .ok_or_else(|| "super stress verdict: KVM control is absent".to_string())?;
746    kvm.passed -= 1;
747    if stress_verdict(&kvm_miss, reps, 1, 1) != 0 {
748        return Err(
749            "super stress verdict: the first KVM measurement must remain nonblocking".into(),
750        );
751    }
752
753    let standin = stress_standin_bracket(root)?;
754
755    Ok(format!(
756        "super source: 31 rows, 3 synthetic expansions, {nextest_rows} nextest rows, {refused} malformed tables refused; stress verdict bracketed; {standin}"
757    ))
758}

760/// Run a probe node's generated command against a stand-in Hermit, so the
761/// shell itself is exercised and not only inspected: each repetition's row and
762/// result line with its cause, the node's exit status, the per-repetition log,
763/// the width bound, and a malformed width. A `timeout` shim first on PATH runs
764/// the real GNU timeout with a 1 s bound and 1 s kill grace, so the causes are
765/// matched against timeout's own messages.
766fn stress_standin_bracket(root: &Path) -> Result<String, String> {
767    let dir = std::env::temp_dir().join(format!(
768        "validate-super-standin-self-test-{}",
769        std::process::id()
770    ));
771    let _ = std::fs::remove_dir_all(&dir);
772    std::fs::create_dir_all(&dir)
773        .map_err(|error| format!("cannot create super stand-in fixture: {error}"))?;
774    let result = run_stress_standin(root, &dir);
775    let _ = std::fs::remove_dir_all(&dir);
776    result
777}
778
779fn write_executable(path: &Path, script: &str) -> Result<(), String> {
780    use std::os::unix::fs::PermissionsExt;
781
782    std::fs::write(path, script)
783        .map_err(|error| format!("cannot write {}: {error}", path.display()))?;
784    std::fs::set_permissions(path, std::fs::Permissions::from_mode(0o755))
785        .map_err(|error| format!("cannot make {} executable: {error}", path.display()))
786}
787
788fn run_stress_standin(root: &Path, dir: &Path) -> Result<String, String> {
789    const REPS: i64 = 9;
790    const WIDTH: usize = 3;
791    let probe = StressProbe::PtraceStrictVerify;
792    let slug = probe.slug();
793    // The probe's last argument is hermit-super-<repetition>. Each repetition
794    // records how many repetitions were running when it started, so the width
795    // bound is measured rather than assumed. Repetition 3 ends its output
796    // without a newline; 5 outlives the bound; 6 also ignores SIGTERM; 7 prints
797    // timeout's KILL message itself; 8 and 9 exit 137 and 124 themselves.
798    let standin = dir.join("hermit");
799    write_executable(
800        &standin,
801        &format!(
802            "#!/usr/bin/env bash\n\
803             rep=${{!#}}; rep=${{rep##*-}}; d={dir}\n\
804             mkdir \"$d/running.$rep\"\n\
805             ls -d \"$d\"/running.* | wc -l >\"$d/peak.$rep\"\n\
806             sleep 0.2\n\
807             rmdir \"$d/running.$rep\"\n\
808             case $rep in\n\
809             2) exit 7;;\n\
810             3) printf 'hermit-super-3'; exit 0;;\n\
811             4) echo \"stand-in diagnostic $rep\" >&2; exit 3;;\n\
812             5) sleep 5;;\n\
813             6) trap '' TERM; sleep 5;;\n\
814             7) echo \"timeout: sending signal KILL to command 'x'\"; exit 5;;\n\
815             8) exit 137;;\n\
816             9) exit 124;;\n\
817             esac\n\
818             echo \"hermit-super-$rep\"\n",
819            dir = shell_quote(&dir.to_string_lossy())
820        ),
821    )?;
822    // The node calls `timeout --verbose --kill-after=G T <command...>`; the
823    // shim drops its own directory from PATH and keeps only the command. Only
824    // repetitions 5 and 6 are meant to outlive the bound, so only they get the
825    // 1 s one; the rest get 30 s, so a loaded host cannot time out a
826    // repetition that should exit on its own (9's own 124 then read as
827    // timeout's, at load average 300 on 2026-10-07).
828    let shim_dir = dir.join("bin");
829    std::fs::create_dir_all(&shim_dir)
830        .map_err(|error| format!("cannot create super stand-in shim dir: {error}"))?;
831    write_executable(
832        &shim_dir.join("timeout"),
833        "#!/usr/bin/env bash\n\
834         rep=${!#}; rep=${rep##*-}\n\
835         case $rep in 5|6) bound=1;; *) bound=30;; esac\n\
836         PATH=${PATH#*:} exec timeout --verbose --kill-after=1 \"$bound\" \"${@:4}\"\n",
837    )?;
838    let path = format!(
839        "{}:{}",
840        shim_dir.display(),
841        std::env::var("PATH").unwrap_or_default()
842    );
843    let command = probe.node_command(REPS, &standin.to_string_lossy(), "/nonexistent", dir);
844    let run = |width: &str, counts: &Path| {
845        std::process::Command::new("bash")
846            .arg("-c")
847            .arg(&command)
848            .current_dir(root)
849            .env("PATH", &path)
850            .env("DAGRUN_TEST_COUNTS_PATH", counts)
851            .env(SUPER_PROBE_JOBS_ENV, width)
852            .output()
853            .map_err(|error| format!("cannot run super stand-in node: {error}"))
854    };
855
856    let counts = dir.join("counts.json");
857    let output = run(&WIDTH.to_string(), &counts)?;
858    let stdout = String::from_utf8_lossy(&output.stdout);
859    let stderr = String::from_utf8_lossy(&output.stderr);
860    if output.status.code() != Some(1) {
861        return Err(format!(
862            "super stand-in node with failed repetitions exited {:?}, not 1:\n{stdout}{stderr}",
863            output.status.code()
864        ));
865    }
866    if !stderr.is_empty() {
867        return Err(format!(
868            "super stand-in node wrote untagged stderr:\n{stderr}"
869        ));
870    }
871    let killed = format!(
872        " (timeout sent SIGKILL {SUPER_PROBE_KILL_GRACE_S} s after the {SUPER_PROBE_TIMEOUT_S} s bound)"
873    );
874    let timed_out = format!(" (timed out after {SUPER_PROBE_TIMEOUT_S} s)");
875    let own_kill = " (SIGKILL not sent by its timeout, e.g. the OOM killer)";
876    for (rep, verdict, cause) in [
877        (1, "pass, exit 0,", ""),
878        (2, "fail, exit 7,", ""),
879        (3, "pass, exit 0,", ""),
880        (4, "fail, exit 3,", ""),
881        (5, "fail, exit 124,", timed_out.as_str()),
882        (6, "fail, exit 137,", killed.as_str()),
883        (7, "fail, exit 5,", ""),
884        (8, "fail, exit 137,", own_kill),
885        (9, "fail, exit 124,", ""),
886    ] {
887        let prefix = format!("super stress {slug} repetition {rep}/{REPS}: {verdict} ");
888        let suffix = format!(" CPU s{cause}");
889        if !stdout
890            .lines()
891            .any(|line| line.starts_with(&prefix) && line.ends_with(&suffix))
892        {
893            return Err(format!(
894                "super stand-in output lacks {prefix:?}...{suffix:?}:\n{stdout}"
895            ));
896        }
897    }
898    let first_after_three = format!("super stress {slug} repetition 4/{REPS}:");
899    for line in [
900        "  [1] hermit-super-1\n".to_string(),
901        format!("  [3] hermit-super-3\n{first_after_three}"),
902        "  [4] stand-in diagnostic 4\n".to_string(),
903        "  [5] timeout: sending signal TERM".to_string(),
904        "  [6] timeout: sending signal KILL".to_string(),
905    ] {
906        if !stdout.contains(&line) {
907            return Err(format!(
908                "super stand-in output lacks the repetition log line {line:?}:\n{stdout}"
909            ));
910        }
911    }
912    let text = std::fs::read_to_string(&counts)
913        .map_err(|error| format!("super stand-in wrote no structured results: {error}"))?;
914    let document: serde_json::Value = serde_json::from_str(&text)
915        .map_err(|error| format!("super stand-in results are not JSON: {error}"))?;
916    let rows = document["results"]
917        .as_array()
918        .ok_or_else(|| format!("super stand-in results have no rows: {text}"))?;
919    let failed = rows
920        .iter()
921        .filter(|row| row["result"] == "fail")
922        .filter_map(|row| row["id"].as_str())
923        .collect::<Vec<_>>();
924    let expected_failed = [2, 4, 5, 6, 7, 8, 9].map(|rep| format!("{slug}/repetition-{rep:02}"));
925    if document["executed_tests"] != REPS
926        || rows.len() != REPS as usize
927        || rows.iter().filter(|row| row["result"] == "pass").count() != 2
928        || failed != expected_failed
929    {
930        return Err(format!(
931            "super stand-in rows are not passes 01 and 03 with the rest failed: {text}"
932        ));
933    }
934    let mut peak = 0;
935    for rep in 1..=REPS {
936        let observed = std::fs::read_to_string(dir.join(format!("peak.{rep}")))
937            .map_err(|error| format!("super stand-in repetition {rep} did not run: {error}"))?;
938        let observed = observed
939            .trim()
940            .parse::<usize>()
941            .map_err(|error| format!("super stand-in repetition {rep} peak: {error}"))?;
942        peak = peak.max(observed);
943    }
944    if !(2..=WIDTH).contains(&peak) {
945        return Err(format!(
946            "super stand-in ran at most {peak} repetitions at once with width {WIDTH}"
947        ));
948    }
949
950    let refused_counts = dir.join("refused.json");
951    let refused = run("three", &refused_counts)?;
952    if refused.status.code() != Some(2) || refused_counts.exists() {
953        return Err(format!(
954            "super stand-in node accepted a malformed width: exit {:?}",
955            refused.status.code()
956        ));
957    }
958    Ok(format!(
959        "stand-in probe node: 2/{REPS} passed, the planted failures carried their causes \
960         (real timeout TERM and KILL, own 137, own 124 and a printed timeout message), \
961         at most {peak} of width {WIDTH} at once, malformed width refused"
962    ))
963}

968        .ok()
969        .and_then(|v| v.parse::<i64>().ok())
970        .filter(|n| *n > 0)
971        .unwrap_or(SUPER_REPETITIONS_DEFAULT)
