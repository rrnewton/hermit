111// A direct typed reader must not silently overwrite duplicate environment keys.
112// The schema10 raw reader separately rejects duplicates before Value buffering.
113fn deserialize_env<'de, D: Deserializer<'de>>(
114    deserializer: D,
115) -> Result<BTreeMap<String, String>, D::Error> {
116    struct Visitor;
117    impl<'de> serde::de::Visitor<'de> for Visitor {
118        type Value = BTreeMap<String, String>;
119        fn expecting(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
120            f.write_str("unique string environment overrides")
121        }
122        fn visit_map<A: serde::de::MapAccess<'de>>(
123            self,
124            mut map: A,
125        ) -> Result<Self::Value, A::Error> {
126            let mut values = BTreeMap::new();
127            while let Some((key, value)) = map.next_entry::<String, String>()? {
128                if values.insert(key.clone(), value).is_some() {
129                    return Err(serde::de::Error::custom(format!(
130                        "duplicate environment key {key}"
131                    )));
132                }
133            }
134            Ok(values)
135        }
136    }
137    deserializer.deserialize_map(Visitor)
138}

229impl WaitCpuObservation {
230    pub(crate) fn from_rusage(usage: &libc::rusage) -> Self {
231        let user = RawTimeval {
232            seconds: usage.ru_utime.tv_sec,
233            microseconds: usage.ru_utime.tv_usec,
234        };
235        let system = RawTimeval {
236            seconds: usage.ru_stime.tv_sec,
237            microseconds: usage.ru_stime.tv_usec,
238        };
239        let measured = (|| {
240            let user_usec = raw_usec(&user).ok_or("wait4 returned an invalid user CPU duration")?;
241            let system_usec =
242                raw_usec(&system).ok_or("wait4 returned an invalid system CPU duration")?;
243            let total_usec = user_usec
244                .checked_add(system_usec)
245                .ok_or("wait4 CPU usage overflowed u64")?;
246            Ok::<_, &str>((user_usec, system_usec, total_usec))
247        })();
248        match measured {
249            Ok((user_usec, system_usec, total_usec)) => Self::Measured {
250                user_usec,
251                system_usec,
252                total_usec,
253            },
254            Err(reason) => Self::Invalid {
255                user,
256                system,
257                reason: reason.into(),
258            },
259        }
260    }
261
262    pub(crate) fn total(&self) -> Result<u64, String> {
263        match self {
264            Self::Measured { total_usec, .. } => Ok(*total_usec),
265            Self::Invalid { reason, .. } => Err(reason.clone()),
266        }
267    }

352fn valid_elapsed(value: &Elapsed) -> Result<(), String> {
353    require(
354        value.nanoseconds < 1_000_000_000,

358fn raw_usec(value: &RawTimeval) -> Option<u64> {
359    let seconds = u64::try_from(value.seconds).ok()?;
360    let microseconds = u64::try_from(value.microseconds).ok()?;
361    (microseconds < 1_000_000).then_some(seconds.checked_mul(1_000_000)?.checked_add(microseconds)?)
362}

364impl LiveCpuEnabled {
365    pub(crate) fn new() -> Self {
366        Self {
367            source: LiveCpuSource::AgentUtilsPairedPidfdStatV1,
368            registration: RegistrationObservation::NotAttempted,

372            unavailable_polls: 0,
373            first: RequiredNullable::Null,
374            last: RequiredNullable::Null,
375            high_water: RequiredNullable::Null,
376            timeout_trigger: RequiredNullable::Null,
377            last_error: RequiredNullable::Null,
378        }
379    }
380
381    /// Summarize an existing monitor poll. This does not call the CPU source.
382    pub(crate) fn record_poll(
383        &mut self,
384        at: Duration,
385        source_called: bool,
386        observation: &Result<u64, CpuError>,
387        triggered: bool,
388    ) {
389        self.polls += 1;
390        self.source_sample_calls += u64::from(source_called);
391        match observation {
392            Ok(cpu_usec) => {
393                self.valid_polls += 1;
394                let point = CpuPoint {
395                    poll: self.polls,
396                    at: at.into(),
397                    cpu_usec: *cpu_usec,
398                };
399                if matches!(self.first, RequiredNullable::Null) {
400                    self.first = RequiredNullable::Value(point.clone());
401                }
402                if nullable(&self.high_water).is_none_or(|high| point.cpu_usec > high.cpu_usec) {
403                    self.high_water = RequiredNullable::Value(point.clone());
404                }
405                self.last = RequiredNullable::Value(point.clone());
406                if triggered {
407                    self.timeout_trigger = RequiredNullable::Value(point);
408                }
409            }
410            Err(error) => {
411                self.unavailable_polls += 1;
412                self.last_error = RequiredNullable::Value(error.clone());
413            }
414        }
415    }
416
417    fn validate(&self, launched: bool, final_at: Option<&Elapsed>) -> Result<(), String> {
418        require(
419            self.valid_polls.checked_add(self.unavailable_polls) == Some(self.polls),
420            "poll counts differ",
421        )?;
422        require(
423            self.source_sample_calls <= self.polls,
424            "more reader calls than polls",
425        )?;
426        match &self.registration {
427            RegistrationObservation::NotAttempted => require(
428                !launched && self.polls == 0 && self.source_sample_calls == 0,
429                "registration not attempted after launch",
430            )?,
431            RegistrationObservation::BoundOnce => require(
432                launched && self.source_sample_calls == self.polls,
433                "bound reader call count differs",
434            )?,
435            RegistrationObservation::Unavailable { reason } => {
436                require(
437                    launched
438                        && nonempty(reason)
439                        && self.source_sample_calls == 0
440                        && self.valid_polls == 0,
441                    "unavailable registration claims samples",
442                )?;
443                if let Some(error) = nullable(&self.last_error) {
444                    require(
445                        error.stage == CpuErrorStage::Registration,
446                        "registration refusal has a different error stage",
447                    )?;
448                }
449            }
450        }
451        require(
452            nullable(&self.last_error).is_some() == (self.unavailable_polls > 0),
453            "unavailable polls and error presence differ",
454        )?;
455        if let Some(error) = nullable(&self.last_error) {
456            require(nonempty(&error.reason), "empty live error")?;
457            if matches!(self.registration, RegistrationObservation::BoundOnce) {
458                require(
459                    error.stage != CpuErrorStage::Registration,
460                    "bound reader claims registration error",
461                )?;
462            }
463        }
464        let (first, last, high) = (
465            nullable(&self.first),
466            nullable(&self.last),
467            nullable(&self.high_water),
468        );
469        require(
470            [first, last, high]
471                .iter()
472                .all(|point| point.is_some() == (self.valid_polls > 0)),
473            "valid polls and point presence differ",
474        )?;
475        for point in [first, last, high, nullable(&self.timeout_trigger)]
476            .into_iter()
477            .flatten()
478        {
479            valid_elapsed(&point.at)?;
480            require(
481                point.poll > 0 && point.poll <= self.polls,
482                "point poll is out of range",
483            )?;
484            if let Some(at) = final_at {
485                require(point.at <= *at, "live point follows final wait")?;
486            }
487        }
488        if let (Some(first), Some(last), Some(high)) = (first, last, high) {
489            require(
490                first.poll <= high.poll
491                    && high.poll <= last.poll
492                    && first.at <= high.at
493                    && high.at <= last.at,
494                "point order differs",
495            )?;
496            require(
497                high.cpu_usec >= first.cpu_usec && high.cpu_usec >= last.cpu_usec,
498                "high water is below an endpoint",
499            )?;
500            require(
501                self.valid_polls <= last.poll - first.poll + 1,
502                "valid poll count exceeds point interval",
503            )?;
504            for point in [last, high] {
505                if point.poll == first.poll {
506                    require(point == first, "same poll has different observations")?;
507                }
508                if point.poll == last.poll {
509                    require(point == last, "same last poll has different observations")?;
510                }
511            }
512            if self.valid_polls == 1 {
513                require(
514                    first == last && last == high,
515                    "single valid poll differs across summaries",
516                )?;
517            }
518        }
519        if let Some(trigger) = nullable(&self.timeout_trigger) {
520            require(
521                Some(trigger) == last && trigger.poll == self.polls,
522                "timeout trigger is not the final poll and last valid observation",
523            )?;
524        }
525        Ok(())
526    }
527}

529impl InvocationCpuObservation {
530    pub(crate) fn pending(
531        ordinal: u64,
532        role: InvocationRole,
533        command: CommandIdentity,
534        cpu_enabled: bool,
535    ) -> Self {
536        Self {
537            ordinal,
538            role,
539            command,
540            launch: LaunchObservation::NotStarted {
541                reason: NotStartedReason::WallBudgetAlreadyExhausted,
542            },
543            live: if cpu_enabled {
544                LiveCpuObservation::Enabled(Box::new(LiveCpuEnabled::new()))
545            } else {
546                LiveCpuObservation::Disabled
547            },
548            final_wait: FinalWaitObservation::NotApplicable,
549            termination: TerminationPath::NotStarted,
550            returned_cpu_charge: ReturnedCpuCharge::Unavailable,
551        }
552    }
553
554    fn returned_without_timeout(&self) -> bool {
555        matches!(
556            self.termination,
557            TerminationPath::CompletedWait4 | TerminationPath::NonterminalWait4Return
558        )
559    }
560
561    fn returned_timeout(&self) -> Option<bool> {
562        match self.termination {
563            TerminationPath::CompletedWait4 | TerminationPath::NonterminalWait4Return => {
564                Some(false)
565            }
566            TerminationPath::NotStarted | TerminationPath::FinalWaitCpuBudgetReturn => Some(true),
567            TerminationPath::CpuBudgetStop | TerminationPath::WallBudgetStop
568                if matches!(self.returned_cpu_charge, ReturnedCpuCharge::Value { .. }) =>
569            {
570                Some(true)
571            }
572            // These error bridges do not return a semantic timeout flag.
573            _ => None,
574        }
575    }
576
577    fn completed_successfully(&self) -> bool {
578        self.termination == TerminationPath::CompletedWait4
579            && matches!(
580                self.final_wait,
581                FinalWaitObservation::Reaped { raw_status: 0, .. }
582            )
583    }
584
585    fn validate(&self) -> Result<(), String> {
586        require(
587            self.ordinal > 0
588                && self.command.argv.first().is_some_and(|p| nonempty(p))
589                && nonempty(&self.command.cwd),
590            "missing invocation identity/command",
591        )?;
592        let pid = match &self.launch {
593            LaunchObservation::Spawned { pid } => {
594                require(*pid > 0, "zero launched pid")?;
595                Some(*pid)
596            }
597            LaunchObservation::NotStarted { reason } => {
598                require(
599                    self.termination == TerminationPath::NotStarted,
600                    "not-started termination differs",
601                )?;
602                require(
603                    *reason != NotStartedReason::CpuBudgetAlreadyExhausted
604                        || matches!(self.live, LiveCpuObservation::Enabled(_)),
605                    "prelaunch CPU exhaustion requires enabled CPU accounting",
606                )?;
607                None
608            }
609            LaunchObservation::SpawnFailed { reason, .. } => {
610                require(
611                    nonempty(reason) && self.termination == TerminationPath::SpawnFailed,
612                    "spawn failure termination differs",
613                )?;
614                None
615            }
616        };
617        let mut final_at = None;
618        let final_cpu = match &self.final_wait {
619            FinalWaitObservation::NotApplicable => {
620                require(pid.is_none(), "launched child has no final disposition")?;
621                None
622            }
623            FinalWaitObservation::Unavailable {
624                operation, reason, ..
625            } => {
626                require(
627                    pid.is_some() && nonempty(reason),
628                    "unavailable wait without a child/reason",
629                )?;
630                require(
631                    match operation {
632                        WaitOperation::Poll => self.termination == TerminationPath::WaitError,
633                        WaitOperation::StopGrace | WaitOperation::BlockingStop => matches!(
634                            self.termination,
635                            TerminationPath::CpuBudgetStop
636                                | TerminationPath::WallBudgetStop
637                                | TerminationPath::AccountingUnavailableStop
638                        ),
639                    },
640                    "wait operation differs from initiating branch",
641                )?;
642                None
643            }
644            FinalWaitObservation::Reaped {
645                pid: waited,
646                raw_status,
647                at,
648                cpu,
649                ..
650            }
651            | FinalWaitObservation::NonterminalReturn {
652                pid: waited,
653                raw_status,
654                at,
655                cpu,
656                ..
657            } => {
658                require(pid == Some(*waited), "waited pid differs from launched pid")?;
659                require(
660                    (0..=u16::MAX as i32).contains(raw_status),
661                    "raw wait status out of range",
662                )?;
663                require(
664                    (libc::WIFEXITED(*raw_status) || libc::WIFSIGNALED(*raw_status))
665                        == matches!(self.final_wait, FinalWaitObservation::Reaped { .. }),
666                    "wait receipt terminal state differs from raw status",
667                )?;
668                valid_elapsed(at)?;
669                final_at = Some(at);
670                match cpu {
671                    WaitCpuObservation::Measured {
672                        user_usec,
673                        system_usec,
674                        total_usec,
675                    } => {
676                        require(
677                            user_usec.checked_add(*system_usec) == Some(*total_usec),
678                            "wait CPU sum differs or overflows",
679                        )?;
680                        Some(*total_usec)
681                    }
682                    WaitCpuObservation::Invalid {
683                        user,
684                        system,
685                        reason,
686                    } => {
687                        require(
688                            nonempty(reason)
689                                && raw_usec(user)
690                                    .and_then(|u| raw_usec(system).and_then(|s| u.checked_add(s)))
691                                    .is_none(),
692                            "invalid wait CPU is actually representable or lacks reason",
693                        )?;
694                        None
695                    }
696                }
697            }
698        };
699        let trigger = match &self.live {
700            LiveCpuObservation::Disabled => None,
701            LiveCpuObservation::Enabled(live) => {
702                live.validate(pid.is_some(), final_at)?;
703                nullable(&live.timeout_trigger)
704            }
705        };
706        require(
707            trigger.is_some() == (self.termination == TerminationPath::CpuBudgetStop),
708            "CPU stop and trigger presence differ",
709        )?;
710        if pid.is_none() {
711            require(
712                matches!(self.final_wait, FinalWaitObservation::NotApplicable)
713                    && matches!(self.returned_cpu_charge, ReturnedCpuCharge::Unavailable),
714                "unstarted child claims CPU",
715            )?;
716            return Ok(());
717        }
718        require(
719            !matches!(
720                self.termination,
721                TerminationPath::NotStarted | TerminationPath::SpawnFailed
722            ),
723            "started child marked unstarted",
724        )?;
725        if self.termination == TerminationPath::AccountingUnavailableStop {
726            // The first Err starts the monitor's nonzero grace at the same
727            // instant used for its comparison, so stopping requires a later
728            // consecutive Err. Counts cannot establish the grace duration.
729            let trailing_unavailable = match &self.live {
730                LiveCpuObservation::Enabled(live) => match nullable(&live.last) {
731                    Some(last) => live.polls.checked_sub(last.poll),
732                    None => Some(live.unavailable_polls),
733                },
734                LiveCpuObservation::Disabled => None,
735            };
736            require(
737                trailing_unavailable.is_some_and(|polls| polls >= 2),
738                "accounting stop lacks two trailing unavailable polls",
739            )?;
740        }
741        match self.termination {
742            TerminationPath::CompletedWait4 => require(
743                matches!(self.final_wait, FinalWaitObservation::Reaped { .. }),
744                "completed wait did not reap the leader",
745            )?,
746            TerminationPath::NonterminalWait4Return => require(
747                matches!(
748                    self.final_wait,
749                    FinalWaitObservation::NonterminalReturn { .. }
750                ),
751                "nonterminal helper return carries a different wait receipt",
752            )?,
753            TerminationPath::FinalWaitCpuBudgetReturn => require(
754                matches!(self.live, LiveCpuObservation::Enabled(_)),
755                "final wait CPU budget return requires enabled CPU accounting",
756            )?,
757            _ => {}
758        }
759        let expected = match self.termination {
760            TerminationPath::CompletedWait4
761            | TerminationPath::NonterminalWait4Return
762            | TerminationPath::FinalWaitCpuBudgetReturn => Some((
763                final_cpu.ok_or("completed wait has no valid CPU receipt")?,
764                ChargeBasis::FinalWait4,
765            )),
766            TerminationPath::WallBudgetStop => final_cpu.map(|cpu| (cpu, ChargeBasis::FinalWait4)),
767            TerminationPath::CpuBudgetStop => final_cpu.map(|cpu| {
768                (
769                    cpu.max(trigger.expect("validated trigger").cpu_usec),
770                    ChargeBasis::MaxTriggerAndFinalWait4,
771                )
772            }),
773            TerminationPath::WaitError => {
774                require(final_cpu.is_none(), "wait error carries a valid receipt")?;
775                None
776            }
777            TerminationPath::AccountingUnavailableStop
778            | TerminationPath::NotStarted
779            | TerminationPath::SpawnFailed => None,
780        };
781        require(
782            match (&self.returned_cpu_charge, expected) {
783                (ReturnedCpuCharge::Unavailable, None) => true,
784                (ReturnedCpuCharge::Value { cpu_usec, basis }, Some((cpu, source))) => {
785                    *cpu_usec == cpu && *basis == source
786                }
787                _ => false,
788            },
789            "returned charge differs from its actual branch operands",
790        )
791    }

833impl CellCpuObservationsV1 {
834    pub fn validate(&self) -> Result<(), String> {
835        require(self.version == 1, "unknown observation version")?;
836        let b = &self.binding;
837        require(
838            [
839                &b.run_id,
840                &b.hermit_sha,
841                &b.lane,
842                &b.category,
843                &b.test,
844                &b.mode,
845            ]
846            .iter()
847            .all(|s| nonempty(s)),
848            "empty cell binding",
849        )?;
850        require(
851            b.outer_attempt > 0 && b.outer_attempt <= crate::runner::MAX_ATTEMPTS_PER_CELL,
852            "outer attempt out of range",
853        )?;
854        if let Some(backend) = nullable(&b.backend) {
855            require(nonempty(backend), "empty backend")?;
856        }
857        let mut executions = BTreeSet::new();
858        let mut preparation = false;
859        let mut normalizations = BTreeSet::new();
860        let mut comparison = false;
861        let mut verify_candidate: Option<&InvocationCpuObservation> = None;
862        for (index, invocation) in self.invocations.iter().enumerate() {
863            require(
864                u64::try_from(index).ok().and_then(|n| n.checked_add(1))
865                    == Some(invocation.ordinal),
866                "invocation ordinals are not contiguous",
867            )?;
868            invocation.validate()?;
869            if let Some(previous) = index.checked_sub(1).map(|i| &self.invocations[i]) {
870                require(
871                    previous.returned_without_timeout()
872                        && !matches!(previous.role, InvocationRole::ParityComparison { .. })
873                        && (!matches!(previous.role, InvocationRole::Preparation)
874                            || previous.completed_successfully()),
875                    "invocation follows a stopped, failed preparation, or final comparison",
876                )?;
877            }
878            let execution = |ordinal: u64| -> Result<&InvocationCpuObservation, String> {
879                let item = usize::try_from(ordinal)
880                    .ok()
881                    .and_then(|n| n.checked_sub(1))
882                    .filter(|n| *n < index)
883                    .and_then(|n| self.invocations.get(n))
884                    .ok_or("CPU role reference is not an earlier invocation")?;
885                require(
886                    matches!(item.role, InvocationRole::Execution { .. }),
887                    "CPU role does not reference execution",
888                )?;
889                Ok(item)
890            };
891            match &invocation.role {
892                InvocationRole::Preparation => {
893                    require(!preparation && index == 0, "repeated or late preparation")?;
894                    preparation = true;
895                }
896                InvocationRole::Execution {
897                    attempt_index,
898                    backend,
899                } => {
900                    require(nonempty(attempt_index), "empty execution index")?;
901                    let backend = nullable(backend).cloned();
902                    require(
903                        executions.insert((attempt_index.clone(), backend.clone())),
904                        "repeated execution identity",
905                    )?;
906                    let expected = if attempt_index == "parity-reference" {
907                        Some("ptrace")
908                    } else {
909                        nullable(&b.backend).map(String::as_str)
910                    };
911                    require(
912                        backend.as_deref() == expected,
913                        "execution backend differs from its role",
914                    )?;
915                    if attempt_index == "parity-reference" {
916                        require(
917                            b.mode == "verify"
918                                && nullable(&b.backend).is_some_and(|backend| backend != "ptrace")
919                                && verify_candidate
920                                    .is_some_and(|prior| prior.completed_successfully()),
921                            "parity reference lacks an earlier successful verify candidate",
922                        )?;
923                    } else if b.mode == "verify" {
924                        require(verify_candidate.is_none(), "repeated verify candidate")?;
925                        verify_candidate = Some(invocation);
926                    }
927                }
928                InvocationRole::PtraceNormalization { execution_ordinal } => {
929                    let prior = execution(*execution_ordinal)?;
930                    require(
931                        b.mode == "verify"
932                            && prior.returned_without_timeout()
933                            && invocation.ordinal.checked_sub(1) == Some(*execution_ordinal)
934                            && matches!(&prior.role, InvocationRole::Execution { backend, .. }
935                                if nullable(backend).map(String::as_str) == Some("ptrace")),
936                        "normalization does not reference a verify ptrace return without timeout",
937                    )?;
938                    require(
939                        normalizations.insert(*execution_ordinal),
940                        "repeated normalization",
941                    )?;
942                }
943                InvocationRole::ParityComparison {
944                    candidate_execution,
945                    reference_execution,
946                } => {
947                    require(
948                        !comparison && candidate_execution != reference_execution,
949                        "repeated or self comparison",
950                    )?;
951                    let candidate = execution(*candidate_execution)?;
952                    let reference = execution(*reference_execution)?;
953                    require(
954                        b.mode == "verify"
955                            && nullable(&b.backend).is_some_and(|backend| backend != "ptrace"),
956                        "comparison is not a verify non-ptrace candidate",
957                    )?;
958                    require(
959                        matches!(&candidate.role, InvocationRole::Execution{attempt_index,..} if attempt_index != "parity-reference")
960                            && matches!(&reference.role, InvocationRole::Execution{attempt_index,..} if attempt_index == "parity-reference"),
961                        "comparison roles are reversed or foreign",
962                    )?;
963                    require(
964                        candidate.completed_successfully() && reference.completed_successfully(),
965                        "comparison operand did not complete with exit zero",
966                    )?;
967                    comparison = true;
968                }
969            }
970        }
971        Ok(())
972    }

998    /// Bind available semantic timeout flags and the roles requiring semantic
999    /// PASS. Failure records alone do not acquire a PASS requirement. Missing
1000    /// flags remain unknown, and cannot establish a no-timeout predecessor.
1001    pub fn require_passing_prerequisites<'a>(
1002        &self,
1003        attempts: impl IntoIterator<Item = (&'a str, bool, Option<bool>)>,
1004    ) -> Result<(), String> {
1005        self.validate()?;
1006        let mut passed = BTreeSet::new();
1007        let mut seen = BTreeMap::new();
1008        for (index, is_pass, timed_out) in attempts {
1009            require(
1010                seen.insert(index, timed_out).is_none(),
1011                "repeated retained semantic attempt",
1012            )?;
1013            let invocation = self.invocations.iter().find(|item| {
1014                matches!(&item.role, InvocationRole::Execution { attempt_index, .. } if attempt_index == index)
1015            }).ok_or("CPU observations omit a retained semantic attempt")?;
1016            if let (Some(actual), Some(expected)) = (timed_out, invocation.returned_timeout()) {
1017                require(
1018                    actual == expected,
1019                    "retained timed_out differs from CPU return branch",
1020                )?;
1021            }
1022            if is_pass {
1023                passed.insert(index);
1024            }
1025        }
1026        let mut required = BTreeSet::new();
1027        for invocation in &self.invocations {
1028            match &invocation.role {
1029                InvocationRole::Execution { attempt_index, .. }
1030                    if attempt_index == "parity-reference" =>
1031                {
1032                    let candidate = self
1033                        .invocations
1034                        .iter()
1035                        .find_map(|item| match &item.role {
1036                            InvocationRole::Execution { attempt_index, .. }
1037                                if attempt_index != "parity-reference" =>
1038                            {
1039                                Some(attempt_index.as_str())
1040                            }
1041                            _ => None,
1042                        })
1043                        .ok_or("reference has no candidate semantic identity")?;
1044                    required.insert(candidate);
1045                }
1046                InvocationRole::ParityComparison {
1047                    candidate_execution,
1048                    reference_execution,
1049                } => {
1050                    for ordinal in [candidate_execution, reference_execution] {
1051                        let operand = self
1052                            .invocations
1053                            .iter()
1054                            .find(|item| item.ordinal == *ordinal)
1055                            .ok_or("comparison has no operand semantic identity")?;
1056                        let InvocationRole::Execution { attempt_index, .. } = &operand.role else {
1057                            return Err("comparison semantic prerequisite is not execution".into());
1058                        };
1059                        required.insert(attempt_index.as_str());
1060                    }
1061                }
1062                _ => {}
1063            }
1064        }
1065        require(
1066            required.is_subset(&passed),
1067            "CPU role lacks a retained passing semantic prerequisite",
1068        )?;
1069        for invocation in self
1070            .invocations
1071            .iter()
1072            .take(self.invocations.len().saturating_sub(1))
1073        {
1074            if let InvocationRole::Execution { attempt_index, .. } = &invocation.role {
1075                require(
1076                    seen.get(attempt_index.as_str()) == Some(&Some(false)),
1077                    "execution successor lacks a retained no-timeout prerequisite",
1078                )?;
1079            }
1080        }
1081        Ok(())
1082    }
1083}

1085pub fn validate_cpu_observations_in_source_row(
1086    row: &Value,
1087) -> Result<Option<CellCpuObservationsV1>, String> {
1088    let Some(raw) = row.get("cpu_observations") else {
1089        return Ok(None);
1090    };
1091    let observations: CellCpuObservationsV1 = serde_json::from_value(raw.clone())
1092        .map_err(|e| format!("malformed CPU observations: {e}"))?;
1093    observations.validate_binding(&CellCpuBinding::from_source_row(row)?)?;
1094    let mut prerequisites = Vec::new();
1095    for attempt in row
1096        .get("attempts")
1097        .and_then(Value::as_array)
1098        .ok_or("CPU source attempts is absent or not an array")?
1099    {
1100        let index = attempt
1101            .get("index")
1102            .and_then(Value::as_str)
1103            .ok_or("CPU source attempt lacks index")?;
1104        let argv: Vec<String> = serde_json::from_value(
1105            attempt
1106                .get("argv")
1107                .cloned()
1108                .ok_or("CPU source attempt lacks argv")?,
1109        )
1110        .map_err(|e| e.to_string())?;
1111        let cwd = attempt
1112            .get("cwd")
1113            .and_then(Value::as_str)
1114            .ok_or("CPU source attempt lacks cwd")?;
1115        let env: BTreeMap<String, String> = serde_json::from_value(
1116            attempt
1117                .get("env")
1118                .cloned()
1119                .ok_or("CPU source attempt lacks env")?,
1120        )
1121        .map_err(|e| e.to_string())?;
1122        observations.validate_attempt(index, &argv, cwd, &env)?;
1123        let timed_out = attempt
1124            .get("timed_out")
1125            .map(|value| {
1126                value
1127                    .as_bool()
1128                    .ok_or("CPU source attempt timed_out is not a boolean")
1129            })
1130            .transpose()?;
1131        prerequisites.push((
1132            index,
1133            attempt.get("outcome").and_then(Value::as_str) == Some("PASS")
1134                && attempt.get("status").and_then(Value::as_i64) == Some(0)
1135                && matches!(attempt.get("signal"), Some(Value::Null))
1136                && timed_out == Some(false),
1137            timed_out,
1138        ));
1139    }
1140    observations.require_passing_prerequisites(prerequisites)?;
1141    Ok(Some(observations))
1142}
