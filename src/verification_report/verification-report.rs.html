1//! Read a current `--verify-json` file through Hermit's producer-owned type.
2//!
3//! Shell and Python tests call this instead of maintaining their own accepted
4//! verdict strings. A new [`Verdict`] variant therefore makes this exhaustive
5//! match fail to compile; an unavailable, incomplete, or malformed report is a
6//! named refusal and never falls back to a human-readable banner.
7
8use std::fs;
9use std::path::Path;
10use std::process::ExitCode;
11
12use hermit::canonical_verdict::Verdict;
13use hermit::canonical_verdict::VerificationReport;
14
15fn refuse(message: impl std::fmt::Display) -> ExitCode {
16    eprintln!("verification-report: REFUSED: {message}");
17    ExitCode::from(2)
18}
19
20fn read_current_report(path: &Path) -> Result<VerificationReport, String> {
21    let bytes =
22        fs::read(path).map_err(|error| format!("cannot read {}: {error}", path.display()))?;
23    let value = serde_json::from_slice(&bytes)
24        .map_err(|error| format!("{} is not valid JSON: {error}", path.display()))?;
25    VerificationReport::from_current_json_value(value)
26        .map_err(|error| format!("{}: {error}", path.display()))
27}
28
29fn require_match(report: &VerificationReport) -> Result<(), String> {
30    match report.verdict {
31        Verdict::Matched if report.verified => Ok(()),
32        Verdict::Matched => {
33            Err("verdict is matched but verified is false; the typed fields disagree".into())
34        }
35        Verdict::Diverged if !report.verified => {
36            Err("verification verdict is diverged, not matched".into())
37        }
38        Verdict::Diverged => {
39            Err("verdict is diverged but verified is true; the typed fields disagree".into())
40        }
41        Verdict::NoResult if !report.verified => {
42            Err("verification verdict is no_result, not matched".into())
43        }
44        Verdict::NoResult => {
45            Err("verdict is no_result but verified is true; the typed fields disagree".into())
46        }
47        Verdict::InfrastructureError if !report.verified => {
48            Err("verification verdict is infrastructure_error, not matched".into())
49        }
50        Verdict::InfrastructureError => Err(
51            "verdict is infrastructure_error but verified is true; the typed fields disagree"
52                .into(),
53        ),
54    }
55}
56
57fn main() -> ExitCode {
58    let mut args = std::env::args().skip(1);
59    let Some(requirement) = args.next() else {
60        return refuse("usage: verification-report matched|canonical-match PATH");
61    };
62    let Some(path) = args.next() else {
63        return refuse("usage: verification-report matched|canonical-match PATH");
64    };
65    if args.next().is_some() {
66        return refuse("usage: verification-report matched|canonical-match PATH");
67    }
68
69    let report = match read_current_report(Path::new(&path)) {
70        Ok(report) => report,
71        Err(error) => return refuse(error),
72    };
73    if let Err(error) = require_match(&report) {
74        eprintln!("verification-report: {error}");
75        return ExitCode::FAILURE;
76    }
77
78    match requirement.as_str() {
79        "matched" => ExitCode::SUCCESS,
80        "canonical-match" => match report.require_canonical_match() {
81            Ok(()) => ExitCode::SUCCESS,
82            Err(error) => {
83                eprintln!("verification-report: {error}");
84                ExitCode::FAILURE
85            }
86        },
87        other => refuse(format!(
88            "unknown requirement {other:?}; expected matched or canonical-match"
89        )),
90    }
91}
