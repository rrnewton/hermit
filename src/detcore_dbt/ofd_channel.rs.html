1//! Callback-scoped access to Detcore's canonical open-file store.
2
3use std::cell::Cell;
4use std::sync::Once;
5
6use detcore::ChunkedOpenFileChannel;
7use detcore::GlobalState;
8use detcore::OpenFileControl;
9use detcore::OpenFileControlReply;
10use detcore::OpenFileControlTransport;
11use detcore::SharedOpenFileError;
12use reverie::Tid;
13use reverie_dbt::SyscallInvoker;
14
15#[derive(Clone, Copy)]
16struct Callback {
17    context: usize,
18    invoke: SyscallInvoker,
19    tid: Tid,
20    physical_pid: i32,
21    physical_tid: i32,
22}
23
24// Constant, destructor-free TLS: no allocator, thread park handle, socket,
25// mutable Tool borrow, or callback pointer survives its explicit guard.
26thread_local! {
27    static CALLBACK: Cell<Option<Callback>> = const { Cell::new(None) };
28}
29
30pub(super) struct Scope(Option<Callback>);
31
32impl Scope {
33    pub(super) fn enter(context: usize, invoke: SyscallInvoker, tid: Tid) -> Self {
34        static INSTALL: Once = Once::new();
35        INSTALL.call_once(|| {
36            assert!(
37                detcore::install_shared_open_file_channel(Box::new(ChunkedOpenFileChannel::new(
38                    Transport
39                ),))
40                .is_ok(),
41                "DBT open file channel already installed"
42            );
43        });
44        let callback = Callback {
45            context,
46            invoke,
47            tid,
48            // These are runtime identity facts, never guest-visible values.
49            physical_pid: unsafe { libc::syscall(libc::SYS_getpid) as i32 },
50            physical_tid: unsafe { libc::syscall(libc::SYS_gettid) as i32 },
51        };
52        Self(CALLBACK.with(|slot| slot.replace(Some(callback))))
53    }
54}
55
56impl Drop for Scope {
57    fn drop(&mut self) {
58        CALLBACK.with(|slot| slot.set(self.0));
59    }
60}
61
62struct Transport;
63
64impl OpenFileControlTransport for Transport {
65    fn call(&self, request: OpenFileControl) -> Result<OpenFileControlReply, SharedOpenFileError> {
66        let callback = CALLBACK.with(Cell::get).ok_or_else(|| {
67            SharedOpenFileError("shared OFD access outside a DBT callback".into())
68        })?;
69        let pid = unsafe { libc::syscall(libc::SYS_getpid) as i32 };
70        let tid = unsafe { libc::syscall(libc::SYS_gettid) as i32 };
71        if (pid, tid) != (callback.physical_pid, callback.physical_tid) {
72            return Err(SharedOpenFileError(
73                "inherited or foreign OFD callback scope".into(),
74            ));
75        }
76        if !reverie_dbt::sync_rpc::is_active() {
77            return Err(SharedOpenFileError(
78                "shared OFDs require the external coordinator".into(),
79            ));
80        }
81        let runtime = super::current_runtime();
82        let response = reverie_dbt::sync_rpc::send_rpc_from_guest::<GlobalState>(
83            callback.context,
84            callback.invoke,
85            callback.tid,
86            &runtime.config,
87            detcore::shared_open_file_request(request),
88        );
89        detcore::shared_open_file_reply(response)
90    }
91}
