38pub(crate) unsafe fn preload() {
39    // Retain the constructor's dispatch lifetime. The preload's allocator
40    // owns Rust allocations unconditionally; this guard does not choose
41    // between Tool storage and the guest's malloc.
42    let _private_allocations = reverie_inguest::guest::alloc::enter_dispatch();
43    // Hermit sends the coordinator and its settings in the sealed bootstrap,
44    // so none of them is on the guest's initial stack or in
45    // /proc/self/environ (detcore::in_guest_launch). A launcher that sets the
46    // environment variables instead still works, below.
47    // SAFETY: this is the preload constructor the launcher started.
48    match unsafe { reverie_liteinst::take_preload_bootstrap() } {
49        Ok(Some(bootstrap)) => {
50            let launch = detcore::in_guest_launch::InGuestLaunch::decode(&bootstrap.tool_data)
51                .unwrap_or_else(|error| {
52                    fail(&format!("the launch bootstrap is unreadable: {error}"))
53                });
54            // SAFETY: as below.
55            unsafe {
56                initialize_detcore(
57                    Mechanism::PatchedSites,
58                    InitializationInputs {
59                        coordinator_socket: bootstrap.coordinator.into_os_string(),
60                        forward_request: launch.forward_request.map(OsString::from),
61                        forward_policy: launch.forward_policy.map(OsString::from),
62                        coordinator_fingerprint: launch.coordinator_fingerprint.map(OsString::from),
63                    },
64                );
65            }
66            return;
67        }
68        Ok(None) => {}
69        Err(error) => fail(&format!(
70            "the launch bootstrap is unreadable: {}",
71            describe_io_error(&error)
72        )),
73    }
74    let Some(socket) = std::env::var_os(reverie_liteinst::COORDINATOR_ENV) else {
75        fail("the coordinator socket environment variable is missing");
76    };
77    let forward_request = std::env::var_os(DETLOG_FORWARD_ENV);
78    let forward_policy = std::env::var_os(DETLOG_FORWARD_POLICY_ENV);
79    // An empty value counts as absent: it is what the removal below leaves in
80    // main's initial environment, which a shell the guest runs passes on.
81    let coordinator_fingerprint = std::env::var_os(detcore::LITEINST_CONFIG_FINGERPRINT_ENV)
82        .filter(|value| !value.is_empty());
83    // SAFETY: the loader runs constructors while the process is still
84    // single-threaded, so nothing reads the environment concurrently.
85    unsafe {
86        // Hermit's fingerprint is for this check only, so it is removed as
87        // the DETLOG variables are: the value's bytes are zeroed, and a
88        // program reading main's initial environment (bash does) sees the
89        // name with an empty value. Later process images load this same file
90        // through LD_PRELOAD, so the first image's check covers them.
91        scrub_env(detcore::LITEINST_CONFIG_FINGERPRINT_ENV);
92        std::env::remove_var(detcore::LITEINST_CONFIG_FINGERPRINT_ENV);
93        // The value names a host socket identity: zero its bytes in the
94        // environment block too, which /proc/self/environ shows.
95        scrub_env(DETLOG_FORWARD_ENV);
96        std::env::remove_var(DETLOG_FORWARD_ENV);
97        std::env::remove_var(DETLOG_FORWARD_POLICY_ENV);
98        std::env::remove_var(reverie_liteinst::COORDINATOR_ENV);
99    }
100    // SAFETY: this is the loader's single-threaded constructor window, and
101    // its private allocation/dispatch scope remains live through installation.
102    unsafe {
103        initialize_detcore(
104            Mechanism::PatchedSites,
105            InitializationInputs {
106                coordinator_socket: socket,
107                forward_request,
108                forward_policy,
109                coordinator_fingerprint,
110            },
111        );
112    }
113}

115// The real saved-frame loader post_load runs before libc's environment initialization.
116// Copy bounded loader-owned NAME=VALUE strings; never consult std::env here.
117pub(crate) unsafe fn loader(argc: usize, argv: *const *const c_char) {
118    let _private_allocations = reverie_inguest::guest::alloc::enter_dispatch();
119    if !(1..=64).contains(&argc) || argv.is_null() {
120        fail("invalid saved-frame loader plugin argument population");
121    }
122    let mut socket = None;
123    let mut forward_request = None;
124    let mut forward_policy = None;
125    let mut fingerprint = None;
126    for index in 1..argc {
127        // SAFETY: the loader owns argc valid strings; the bounded reader
128        // refuses missing termination rather than scanning indefinitely.
129        let entry = unsafe { bounded_setting(*argv.add(index)) };
130        let Some(at) = entry.iter().position(|b| *b == b'=') else {
131            fail("saved-frame loader plugin setting is not NAME=VALUE");
132        };
133        let (name, value) = (&entry[..at], &entry[at + 1..]);
134        let slot = match name {
135            b"HERMIT_ENGINE_COORDINATOR" => &mut socket,
136            b"HERMIT_ENGINE_FORWARD_DETLOG" => &mut forward_request,
137            b"HERMIT_ENGINE_FORWARD_DETLOG_POLICY" => &mut forward_policy,
138            b"HERMIT_ENGINE_CONFIG_FINGERPRINT" => &mut fingerprint,
139            _ => fail("unknown saved-frame loader common-engine setting"),
140        };
141        if slot.replace(OsString::from_vec(value.to_vec())).is_some() {
142            fail("duplicate saved-frame loader common-engine setting");
143        }
144    }
145    let socket = socket
146        .filter(|s| !s.is_empty())
147        .unwrap_or_else(|| fail("saved-frame loader common-engine coordinator is missing"));
148    let fingerprint = fingerprint
149        .filter(|s| !s.is_empty())
150        .unwrap_or_else(|| fail("saved-frame loader common-engine fingerprint is missing"));
151    // SAFETY: genuine post_load, with fixed storage and allocation scope live.
152    unsafe {
153        initialize_detcore(
154            Mechanism::SavedFrames,
155            InitializationInputs {
156                coordinator_socket: socket,
157                forward_request,
158                forward_policy,
159                coordinator_fingerprint: Some(fingerprint),
160            },
161        )
162    };
163}

165unsafe fn bounded_setting(pointer: *const c_char) -> Vec<u8> {
166    if pointer.is_null() {
167        fail("null saved-frame loader setting");
168    }
169    for length in 0..16_384 {
170        // SAFETY: the loader promised a valid NUL-terminated string. The cap
171        // only bounds work; it cannot authenticate an arbitrary pointer.
172        if unsafe { *pointer.add(length) } == 0 {
173            return unsafe { core::slice::from_raw_parts(pointer.cast::<u8>(), length) }.to_vec();
174        }
175    }
176    fail("saved-frame loader setting exceeds the loader ABI's 16-KiB cap");
177}

179unsafe fn initialize_detcore(mechanism: Mechanism, inputs: InitializationInputs) {
180    #[cfg(feature = "dispatch-observer")]
181    crate::observation::install();
182    let InitializationInputs {
183        coordinator_socket: socket,
184        forward_request,
185        forward_policy,
186        coordinator_fingerprint,
187    } = inputs;
188    // Checked before anything is decoded: the coordinator sends Detcore's
189    // configuration as positional bincode, which a runtime built from another
190    // tree decodes as garbage and reports only as a decoding error
191    // (https://github.com/rrnewton/hermit/issues/3986). Without the variable
192    // (an image the first one executed, or a Hermit from before this check)
193    // nothing is refused: a check that rejects matched pairs is worse than
194    // none, as for the saved-frame loader plugin.
195    if let Some(mismatch) = coordinator_fingerprint
196        .as_deref()
197        .and_then(detcore::config_fingerprint_mismatch)
198    {
199        fail(&format!(
200            "the runtime at {} was built from a different tree than the hermit running it: \
201             {mismatch}. Rebuild the common engine and coordinator from matching source",
202            runtime_path()
203        ));
204    }
205    if let Some(value) = forward_request {
206        let Some((fd, identity)) = value
207            .to_str()
208            .and_then(detcore::detlog::parse_tool_output_env_value)
209        else {
210            fail("the DETLOG forwarding descriptor value is unreadable");
211        };
212        // Hermit sends the policy with the descriptor; without a readable one
213        // this process cannot forward the records ptrace would log.
214        let policy = match loader_settings::forwarding_policy(forward_policy.as_deref()) {
215            Ok(Some(policy)) => policy,
216            Err(error) => fail(&format!(
217                "the DETLOG forwarding policy is unreadable: {error}"
218            )),
219            Ok(None) => fail("the DETLOG forwarding policy is missing"),
220        };
221        // A signal handler installed by guest code that ran before this
222        // constructor could otherwise replace the number between the identity
223        // check and the reservation; and the reserved copy, which is what the
224        // Tool keeps, must itself be the socket Hermit passed.
225        let blocked = block_signals();
226        let reserved = if detcore::detlog::socket_identity(fd) != Some(identity) {
227            Err(
228                "it is not the socket Hermit passed (guest code that ran before this \
229                 constructor closed or replaced it)"
230                    .to_owned(),
231            )
232        } else {
233            // SAFETY: the process is still single-threaded and Detcore is not
234            // yet installed, which is when the reservation must be made.
235            match unsafe {
236                reverie_liteinst::reserve_tool_output_fd(
237                    fd,
238                    detcore::detlog::FORWARDING_RETIRED_NOTICE,
239                )
240            } {
241                Ok(reserved) if detcore::detlog::socket_identity(reserved) == Some(identity) => {
242                    Ok(reserved)
243                }
244                Ok(_) => Err("the reserved copy is not the socket Hermit passed".to_owned()),
245                Err(error) => Err(describe_io_error(&error)),
246            }
247        };
248        restore_signals(&blocked);
249        match reserved {
250            Ok(_) => {
251                let _ = TOOL_OUTPUT_IDENTITY.set(identity);
252                let _ = detcore::detlog::set_forwarder(forward_detlog, policy);
253            }
254            Err(error) => fail(&format!(
255                "cannot reserve the DETLOG forwarding descriptor {fd}: {error}"
256            )),
257        }
258    }
259    // Detcore's exit-dependency refusals name only the capability; this
260    // backend adds its name and the alternative.
261    detcore::exit_dependencies::set_backend_advice(
262        "the ordinary common in-guest engine cannot run this \
263         program; run it with --backend=ptrace.",
264    );
265    // Coord rulings A and D.1: Detcore checks every descriptor that arrives
266    // later; this checks the ones the process image starts with. A forked
267    // child holds only what its parent held or received.
268    match detcore::exit_dependencies::held_exit_dependency(Path::new("/proc/self/fd")) {
269        Ok(None) => {}
270        Ok(Some((fd, what))) => fail(&format!(
271            "the common in-guest engine refuses a guest that starts holding {what} (descriptor {fd}): \
272             Hermit holds every guest's turn until an exit completes, and a guest serving that \
273             descriptor could make another guest's exit wait forever; run this program with \
274             --backend=ptrace"
275        )),
276        Err(error) => fail(&format!(
277            "cannot list the descriptors this process started with: {}",
278            describe_io_error(&error)
279        )),
280    }
281    // SAFETY: the loader runs constructors before any application thread
282    // exists and before the application can install a seccomp filter, which is
283    // the window `install_tool` requires.
284    // Both paths publish exactly ToolHost<Detcore, EngineRuntime>. The
285    // concrete backend services stay boxed within the same engine DSO.
286    let installed = match mechanism {
287        Mechanism::PatchedSites => unsafe {
288            reverie_liteinst::install_tool_from_bootstrap_with_runtime::<
289                detcore::Detcore,
290                EngineRuntime,
291            >(Path::new(&socket), EngineRuntime::new)
292        },
293        Mechanism::SavedFrames => unsafe {
294            reverie_inguest::guest::tool_install::install_tool_with_runtime_and_root_initializer::<
295                detcore::Detcore,
296                _,
297                EngineRuntime,
298                _,
299            >(
300                Path::new(&socket),
301                false,
302                FrameHookHost::new(),
303                |services| EngineRuntime::new(Box::new(services)),
304                crate::loader_state::initialize_root,
305            )
306        },
307    };
308    if let Err(error) = installed {
309        fail(&describe_io_error(&error));
310    }
311    // `--backend=in-guest-trap`: the Tool's constructor recorded the host's
312    // request from the coordinator's configuration; compare it with the
313    // settings the runtime actually captured while installing, which guest
314    // code cannot change afterwards, and refuse to run the program when guest
315    // code that ran before this point changed the environment they were read
316    // from. The program's own code has not run.
317    if let Some(violation) = detcore::in_guest_site_patching::violation(
318        detcore::in_guest_site_patching::site_patching_off_required(),
319        match mechanism {
320            Mechanism::PatchedSites => reverie_liteinst::site_patching_enabled(),
321            Mechanism::SavedFrames => false,
322        },
323        reverie_liteinst::guest_stats_enabled(),
324    ) {
325        fail(&violation);
326    }
327    // A description a fork shares is reached through the coordinator, over
328    // this process's one existing connection. The first installation in a
329    // process image is the only one, and a forked child inherits it.
330    let _ = detcore::install_shared_open_file_channel(Box::new(
331        detcore::ChunkedOpenFileChannel::new(CoordinatorOpenFiles),
332    ));
333    // Detcore's warnings reach the run's log through the coordinator, over the
334    // same connection. A forked child inherits the subscriber.
335    if !detcore::guest_log::install(guest_log_call, None) {
336        fail("Detcore's in-guest subscriber was not installed: the process already has one");
337    }
338    // Detcore reports the guest signal controls this runtime refuses before
339    // Detcore sees them, once per run (`detcore::refused_signal_controls`). A
340    // forked child inherits the source.
341    let _ = detcore::refused_signal_controls::install(refused_signal_controls);
342}

344fn coordinator_call(
345    request: <detcore::GlobalState as reverie::GlobalTool>::Request,
346) -> std::io::Result<<detcore::GlobalState as reverie::GlobalTool>::Response> {
347    reverie_liteinst::blocking_global_rpc::<detcore::GlobalState>(request)
348}

350/// The guest signal controls this runtime refused since the last call
351/// (`reverie_inguest::guest::protect::take_refused_signal_controls`), as
352/// Detcore takes them.
353fn refused_signal_controls() -> detcore::refused_signal_controls::RefusedSignalControls {
354    let refused = reverie_inguest::guest::protect::take_refused_signal_controls();
355    (refused.actions, refused.alternate_stack)
356}

358/// [`coordinator_call`] for Detcore's warnings, which can be raised anywhere:
359/// it sends only where the runtime marks a Tool callback as running
360/// (`tool_callback_active`), since outside one the runtime would refuse the
361/// connection's syscalls as the guest's and end the guest. Elsewhere the
362/// warning is refused, and `detcore::guest_log` counts it.
363fn guest_log_call(
364    request: <detcore::GlobalState as reverie::GlobalTool>::Request,
365) -> std::io::Result<<detcore::GlobalState as reverie::GlobalTool>::Response> {
366    if !reverie_inguest::guest::support::tool_callback_active() {
367        return Err(std::io::Error::other(
368            "a Detcore warning from outside a Tool callback",
369        ));
370    }
371    coordinator_call(request)
372}

374/// Carries shared open file control messages to Detcore's global state from
375/// inside a Tool callback, through [`coordinator_call`].
376/// A `DetFd` method is synchronous code inside a callback's poll, so no other
377/// request of this process is in flight when it runs; if one is, the call is
378/// refused rather than interleaved.
379struct CoordinatorOpenFiles;
380
381impl detcore::OpenFileControlTransport for CoordinatorOpenFiles {
382    fn call(
383        &self,
384        control: detcore::OpenFileControl,
385    ) -> Result<detcore::OpenFileControlReply, detcore::SharedOpenFileError> {
386        let response =
387            coordinator_call(detcore::shared_open_file_request(control)).map_err(|error| {
388                detcore::SharedOpenFileError(format!(
389                    "the coordinator connection refused a shared open file message: {}",
390                    describe_io_error(&error)
391                ))
392            })?;
393        detcore::shared_open_file_reply(response)
394    }
395}

397/// Sends one Detcore record on the reserved socket. Its number can move when the
398/// guest `dup2`s onto it, so it is read for each record.
399fn forward_detlog(target: &str, record_suffix: &str, index: u64, message: std::fmt::Arguments<'_>) {
400    if let Some(socket) = authenticated_output_socket() {
401        detcore::detlog::send_forwarded_record(socket, target, record_suffix, index, message);
402    }
403}
404
405fn authenticated_output_socket() -> Option<libc::c_int> {
406    let socket = reverie_liteinst::tool_output_fd()?;
407    (TOOL_OUTPUT_IDENTITY
408        .get()
409        .is_some_and(|identity| detcore::detlog::socket_identity(socket) == Some(*identity)))
410    .then_some(socket)
411}

413// Same ordinary constructor signal masks; private InitialImage rules are
414// deliberately left in their unchanged maintained owner.
415fn block_signals() -> u64 {
416    let all = u64::MAX;
417    let mut previous = 0_u64;
418    // SAFETY: both masks are initialized and live for the synchronous syscall.
419    let _ = unsafe {
420        reverie_inguest::signal::raw_sigprocmask(libc::SIG_BLOCK, Some(&all), Some(&mut previous))
421    };
422    previous
423}
424
425fn restore_signals(previous: &u64) {
426    // SAFETY: previous is the saved initialized mask.
427    let _ = unsafe {
428        reverie_inguest::signal::raw_sigprocmask(libc::SIG_SETMASK, Some(previous), None)
429    };
430}

432/// Overwrites the bytes of `key`'s entry in the environment block with zeros.
433///
434/// # Safety
435///
436/// The process must be single-threaded.
437unsafe fn scrub_env(key: &str) {
438    unsafe extern "C" {
439        static environ: *const *mut libc::c_char;
440    }
441    let mut prefix = key.as_bytes().to_vec();
442    prefix.push(b'=');
443    // SAFETY: `environ` is a NULL-terminated array of NUL-terminated strings.
444    unsafe {
445        let mut slot = environ;
446        while !slot.is_null() && !(*slot).is_null() {
447            let entry = *slot;
448            let bytes = std::ffi::CStr::from_ptr(entry).to_bytes();
449            if bytes.starts_with(&prefix) {
450                std::ptr::write_bytes(
451                    entry.cast::<u8>().add(prefix.len()),
452                    0,
453                    bytes.len() - prefix.len(),
454                );
455            }
456            slot = slot.add(1);
457        }
458    }
459}

461/// The file this runtime was loaded from, as the dynamic loader reports it.
462fn runtime_path() -> String {
463    let mut info = std::mem::MaybeUninit::<libc::Dl_info>::zeroed();
464    // SAFETY: `dladdr` writes `info` and reads nothing else; the address is a
465    // function of this library.
466    let found = unsafe {
467        libc::dladdr(
468            crate::hermit_engine_preload_initialize_v1 as *const libc::c_void,
469            info.as_mut_ptr(),
470        )
471    };
472    // SAFETY: `dladdr` filled `info` when it returned nonzero.
473    let name = (found != 0).then(|| unsafe { info.assume_init() }.dli_fname);
474    match name.filter(|name| !name.is_null()) {
475        // SAFETY: the loader's file name is a NUL-terminated string it keeps
476        // for as long as the library stays loaded.
477        Some(name) => unsafe { std::ffi::CStr::from_ptr(name) }
478            .to_string_lossy()
479            .into_owned(),
480        None => "an unknown path".to_owned(),
481    }
482}

484pub(crate) fn fail(message: &str) -> ! {
485    crate::poison();
486    write_stderr(b"hermit common engine: initialization failed: ");
487    write_stderr(message.as_bytes());
488    write_stderr(b"\n");
489    // Preserve the ordinary leaf's libc _exit route so an installed runtime
490    // submits exit statistics. This is not an InitialImage failure path.
491    unsafe { libc::_exit(127) }
492}

494pub(crate) fn refuse(message: &[u8]) -> ! {
495    crate::poison();
496    write_stderr(message);
497    // The same trusted gate used by the existing rewritten-entry refusal.
498    unsafe { reverie_inguest::guest::support::exit_now(122) }
499}
500
501fn write_stderr(message: &[u8]) {
502    unsafe {
503        reverie_inguest::trap::raw_syscall6(
504            libc::SYS_write,
505            [
506                libc::STDERR_FILENO as u64,
507                message.as_ptr() as u64,
508                message.len() as u64,
509                0,
510                0,
511                0,
512            ],
513        );
514    }
