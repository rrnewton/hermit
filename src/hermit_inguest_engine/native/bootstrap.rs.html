74struct Bootstrap {
75    config: Arc<Config>,
76    global: Arc<GlobalState>,
77    coordinator: Option<PathBuf>,
78    physical_owner: i32,
79    // Preserve the maintained background's libc getpid observation for its
80    // exec pause predicate. The helper's raw PID and application owner are
81    // distinct launch facts and must not replace this observation.
82    pause_owner: i32,
83    image: u64,
84    // The original protected duplicate belongs to this image, including COW.
85    // It is never a forged application RPC or a replaced report descriptor.
86    _copied_report: Option<OwnedFd>,
87}

89static BOOTSTRAP: OnceLock<Bootstrap> = OnceLock::new();
90static ROOT_INSTALLATION: AtomicU8 = AtomicU8::new(0);

92unsafe fn borrowed<'a>(
93    pointer: *const u8,
94    length: usize,
95    limit: usize,
96) -> io::Result<Option<&'a [u8]>> {
97    if length > limit || (pointer.is_null() && length != 0) {
98        return Err(io::Error::from_raw_os_error(libc::EINVAL));
99    }
100    if pointer.is_null() {
101        return Ok(None);
102    }
103    Ok(Some(unsafe { std::slice::from_raw_parts(pointer, length) }))
104}

106fn current_physical_pid() -> i32 {
107    unsafe { libc::syscall(libc::SYS_getpid) as i32 }
108}

110/// Borrow only immutable launch policy and its original protected report FD.
111/// A shared-VM child must not enter the inherited host/Tool/RPC locks or clone
112/// an owning handle. BOOTSTRAP was completely published before native READY.
113pub(super) fn copied_policy(image: u64, physical_pid: i32) -> io::Result<(bool, i32)> {
114    let boot = BOOTSTRAP
115        .get()
116        .ok_or_else(|| io::Error::from_raw_os_error(libc::EAGAIN))?;
117    if crate::INSTALLATION.load(Ordering::Acquire) != 5
118        || physical_pid <= 0
119        || physical_pid != current_physical_pid()
120        || physical_pid == boot.physical_owner
121        || image == 0
122        || image != boot.image
123        || image != super::image::generation()
124    {
125        return Err(io::Error::from_raw_os_error(libc::ESTALE));
126    }
127    Ok((
128        boot.config.panic_on_unsupported_syscalls,
129        boot._copied_report.as_ref().map_or(-1, AsRawFd::as_raw_fd),
130    ))
131}

133/// # Safety
134/// Called on the genuine native background thread after the real image's
135/// fixed allocator backing, ordinary FS/TLS and native stack are established.
136/// Borrowed startup bytes remain live until copied below; callbacks remain
137/// in the exact held/hash-bound native client for this process image.
138#[unsafe(no_mangle)]
139pub unsafe extern "C" fn hermit_engine_native_background_v1(startup: *const Startup) -> i32 {
140    let result = unsafe { start(startup) };
141    match result {
142        Ok(()) => 0,
143        Err(error) => {
144            diagnostics::emit(format!("native engine: startup refused: {error}\n").as_bytes());
145            -error.raw_os_error().unwrap_or(libc::EPROTO)
146        }
147    }
148}

150unsafe fn start(startup: *const Startup) -> io::Result<()> {
151    if startup.is_null() {
152        return Err(io::Error::from_raw_os_error(libc::EINVAL));
153    }
154    let startup = unsafe { &*startup };
155    if startup.version != VERSION
156        || startup.size as usize != size_of::<Startup>()
157        || crate::INSTALLATION.load(Ordering::Acquire) != 5
158        || startup.image == 0
159        || startup.image != image::generation()
160        || startup.physical_owner <= 0
161        || startup.background_pid != current_physical_pid()
162        || startup.reserved != 0
163        || !matches!(startup.panic_on_unsupported, 0 | 1)
164    {
165        return Err(io::Error::from_raw_os_error(libc::ESTALE));
166    }
167    let mode = match startup.mode {
168        1 => Mode::Standalone,
169        2 => Mode::Coordinated,
170        _ => return Err(io::Error::from_raw_os_error(libc::EINVAL)),
171    };
172    let diagnostic = startup
173        .diagnostic
174        .ok_or_else(|| io::Error::from_raw_os_error(libc::EINVAL))?;
175    let evidence = startup
176        .evidence
177        .ok_or_else(|| io::Error::from_raw_os_error(libc::EINVAL))?;
178    let idle = startup
179        .idle
180        .ok_or_else(|| io::Error::from_raw_os_error(libc::EINVAL))?;
181    let control = unsafe { borrowed(startup.control, startup.control_len, 107)? };
182    let fingerprint = unsafe { borrowed(startup.fingerprint, startup.fingerprint_len, 256)? };
183    let json = unsafe {
184        borrowed(
185            startup.standalone_json,
186            startup.standalone_json_len,
187            CONFIG_LIMIT,
188        )?
189    };
190    let tracing_active = diagnostics::install(
191        diagnostic,
192        evidence,
193        startup.evidence_level,
194        control.is_some(),
195    )?;
196    let selected = config::select(
197        mode,
198        control.map(OsStr::from_bytes),
199        fingerprint.map(OsStr::from_bytes),
200        json.and_then(|bytes| std::str::from_utf8(bytes).ok()),
201    )?;
202    let mut config = selected.config;
203    config.panic_on_unsupported_syscalls = startup.panic_on_unsupported != 0;
204    // Preserve the native runtime's existing exit+protected-report path. A
205    // ptrace unrecoverable shutdown would suppress that report and is disabled.
206    config.exit_on_unsupported_syscall = true;
207    config.shutdown_on_unsupported_syscall = false;
208    config.unsupported_syscall_report_fd =
209        (unsafe { libc::fcntl(UNSUPPORTED_REPORT_FD, libc::F_GETFD) } >= 0)
210            .then_some(UNSUPPORTED_REPORT_FD);
211    let duplicate = unsafe {
212        libc::fcntl(
213            UNSUPPORTED_REPORT_FD,
214            libc::F_DUPFD_CLOEXEC,
215            UNSUPPORTED_REPORT_FD + 1,
216        )
217    };
218    let copied_report = (duplicate >= 0).then(|| unsafe { OwnedFd::from_raw_fd(duplicate) });
219    config.validate();
220    if diagnostics::info_enabled() {
221        let source: &[u8] = match selected.source {
222            Source::Coordinator => b"native engine: using coordinator-provided Detcore Config\n",
223            Source::StandaloneJson => b"native engine: using standalone Detcore Config\n",
224            Source::StandaloneParseFallback => {
225                b"native engine: using strict default after standalone parse failure\n"
226            }
227            Source::StandaloneDefault => b"native engine: using strict default Detcore Config\n",
228        };
229        diagnostics::emit(source);
230    }
231    if matches!(selected.source, Source::StandaloneParseFallback) {
232        diagnostics::emit(b"native engine: WARNING could not parse standalone Detcore Config; using strict default\n");
233    }
234    let config = Arc::new(config);
235    let global = Arc::new(GlobalState::init_for_external_scheduler(&config));
236    BOOTSTRAP
237        .set(Bootstrap {
238            config,
239            global: Arc::clone(&global),
240            coordinator: selected.coordinator,
241            physical_owner: startup.physical_owner,
242            pause_owner: unsafe { libc::getpid() },
243            image: startup.image,
244            _copied_report: copied_report,
245        })
246        .map_err(|_| io::Error::from_raw_os_error(libc::EALREADY))?;
247    detcore::install_shared_open_file_channel(Box::new(detcore::ChunkedOpenFileChannel::new(
248        OpenFiles,
249    )))
250    .map_err(|_| io::Error::other("native open-file channel already installed"))?;
251    let log_scheduler = diagnostics::info_enabled() && !tracing_active;
252    let observer = Arc::new(move |event: &'static str| {
253        if log_scheduler {
254            diagnostics::emit(format!("INFO detcore::scheduler: {event}\n").as_bytes());
255        }
256    });
257    // This call creates the original future and emits its startup INFO before
258    // READY releases the application's real root initialization.
259    let scheduler = global.run_external_scheduler(observer);
260    image::publish_ready(startup.image);
261    cooperative(scheduler, idle);
262    Ok(())
263}

265fn cooperative<F: Future<Output = ()>>(future: F, idle: Idler) {
266    let mut future = pin!(future);
267    let mut context = Context::from_waker(Waker::noop());
268    loop {
269        if image::SHUTDOWN.load(Ordering::Acquire) {
270            return;
271        }
272        if image::PAUSE_REQUESTED.load(Ordering::Acquire) {
273            image::PAUSED.store(true, Ordering::Release);
274            while image::PAUSE_REQUESTED.load(Ordering::Acquire)
275                && !image::SHUTDOWN.load(Ordering::Acquire)
276            {
277                unsafe {
278                    idle();
279                }
280            }
281            image::PAUSED.store(false, Ordering::Release);
282            continue;
283        }
284        match future.as_mut().poll(&mut context) {
285            Poll::Ready(()) => return,
286            Poll::Pending => unsafe {
287                idle();
288            },
289        }
290    }
291}

293/// The actual application's first root event constructs the one concrete
294/// Detcore. Background initialization never fabricates a Tool or root TID.
295pub(super) fn install_root(callback: &Callback) -> io::Result<&'static owner::Engine> {
296    callback.validate(callback.identity)?;
297    let identity = callback.identity;
298    // A genuine copied process has already installed its fresh owner through
299    // the real child post-event. It reuses the original authenticated config,
300    // never another background/GlobalState or a second Detcore::new call.
301    if let Ok(engine) = owner::installed() {
302        return Ok(engine);
303    }
304    let boot = BOOTSTRAP
305        .get()
306        .ok_or_else(|| io::Error::from_raw_os_error(libc::EAGAIN))?;
307    if boot.physical_owner != current_physical_pid()
308        || boot.physical_owner != identity.physical_pid
309        || boot.image != identity.image
310        || boot.image != image::generation()
311    {
312        return Err(io::Error::from_raw_os_error(libc::ESTALE));
313    }
314    if identity.physical_tid != identity.physical_pid {
315        return Err(io::Error::from_raw_os_error(libc::EAGAIN));
316    }
317    ROOT_INSTALLATION
318        .compare_exchange(0, 1, Ordering::Acquire, Ordering::Relaxed)
319        .map_err(|_| io::Error::from_raw_os_error(libc::EBUSY))?;
320    // No process lock is held while constructing the original Tool. Failure
321    // poisons this image's one-shot installation rather than creating another.
322    let result = (|| {
323        let config = Arc::clone(&boot.config);
324        let (cached, rpc) = match &boot.coordinator {
325            Some(path) => (
326                CoordinatorRpc::from_native_configuration(path, Arc::clone(&config))?,
327                NativeRpc::new(path.clone(), Arc::clone(&config))?,
328            ),
329            None => (
330                CoordinatorRpc::from_native_local_configuration(Arc::clone(&config)),
331                NativeRpc::local(Arc::clone(&boot.global), Arc::clone(&config)),
332            ),
333        };
334        let root = Pid::from_raw(identity.tool_pid);
335        let subscriptions = <Detcore as Tool>::subscriptions(&config);
336        let tool = Detcore::new(root, &config);
337        let host = unsafe {
338            owner::Host::new_native(
339                tool,
340                cached,
341                root,
342                subscriptions.iter_syscalls().collect(),
343                subscriptions.has_cpuid(),
344                fresh_runtime(),
345            )
346        };
347        owner::publish(owner::Engine {
348            host,
349            rpc,
350            config,
351            root,
352            physical_owner: boot.physical_owner,
353            image: boot.image,
354            pending_parents: Mutex::new(std::collections::HashMap::new()),
355            next_child_ordinal: AtomicU64::new(1),
356        })
357    })();
358    ROOT_INSTALLATION.store(if result.is_ok() { 2 } else { 3 }, Ordering::Release);
359    result
360}

362/// Immutable real launch choice. The admitted Engine owns all mutable work.
363pub(super) fn coordinated() -> bool {
364    BOOTSTRAP
365        .get()
366        .is_some_and(|boot| boot.coordinator.is_some())
367}

369/// Preserve the maintained background's observed libc PID pause predicate.
370/// The application owner is separate; a COW child's Engine cannot replace
371/// this observation or fabricate an additional scheduler background.
372pub(super) fn owns_background(physical_pid: i32) -> bool {
373    BOOTSTRAP.get().is_some_and(|boot| {
374        boot.pause_owner == physical_pid
375            && physical_pid == current_physical_pid()
376            && boot.image == image::generation()
377    })
378}

380struct OpenFiles;
381impl detcore::OpenFileControlTransport for OpenFiles {
382    fn call(
383        &self,
384        request: detcore::OpenFileControl,
385    ) -> Result<detcore::OpenFileControlReply, detcore::SharedOpenFileError> {
386        let boot = BOOTSTRAP.get().ok_or_else(|| {
387            detcore::SharedOpenFileError("native configuration is not installed".into())
388        })?;
389        if boot.coordinator.is_none() {
390            return Err(detcore::SharedOpenFileError(
391                "shared OFDs require the external coordinator".into(),
392            ));
393        }
394        let response = diagnostics::global_call(detcore::shared_open_file_request(request))
395            .map_err(|error| detcore::SharedOpenFileError(error.to_string()))?;
396        detcore::shared_open_file_reply(response)
397    }
398}

400/// Native Guest operations use the selected per-callback service table. A
401/// mistaken local dispatcher entry has no permissive HostRuntime substitute.
402struct NativeRuntime;
403pub(super) fn fresh_runtime() -> EngineRuntime {
404    EngineRuntime::new(Box::new(NativeRuntime))
405}
406impl HostRuntime for NativeRuntime {
407    fn fork_child_rebind(&self) {
408        panic!("native fork requires its original native lifecycle owner");
409    }
410    fn emit_stage(&self, stage: &[u8]) {
411        diagnostics::emit(stage);
412    }
413    #[cfg(feature = "native-first-instruction-observer")]
414    fn observe_native_entry(
415        &self,
416        stage: reverie_inguest::guest::native_services::NativeEntryStage,
417        observation: reverie_inguest::guest::native_services::NativeEntryObservation<'_, '_>,
418    ) {
419        super::first_instruction_observer::observe(stage, observation);
420    }
421    fn fork_child_reset(&self, _event: &reverie_inguest::guest::event::SyscallEvent) {
422        panic!("native fork cannot use the local runtime reset");
423    }
424    fn cpuid_interception_enabled(&self) -> bool {
425        true
426    }
427    fn exit_process_stats(&self, _tid: Pid) -> io::Result<()> {
428        Err(io::Error::other(
429            "native statistics retain their native finalization owner",
430        ))
431    }
432    fn read_clock(&self) -> io::Result<u64> {
433        Err(io::Error::other(
434            "native clock requires the actual callback branch fact",
435        ))
436    }
437    fn signal_action_supported(&self, _number: i64, _args: [u64; 6]) -> bool {
438        false
439    }
440    fn reserved_signal_mask(&self) -> u64 {
441        0
442    }
