1//! The application's actual initial auxiliary vector at native entry.
2//!
3//! The loader can rewrite the application stack while the kernel's saved
4//! vector still describes the loader. Preserve the maintained bounded walk,
5//! fault-safe memory mechanism, then the same kernel-vector fallback.
6
7use std::io;
8
9use reverie_inguest::guest::native_services::MemoryReader;
10
11type Entries = Vec<(libc::c_ulong, libc::c_ulong)>;
12
13pub(super) fn read(reader: MemoryReader) -> io::Result<Entries> {
14    application(reader)
15        .or_else(|| kernel(&std::fs::read("/proc/self/auxv").ok()?))
16        .ok_or_else(|| io::Error::other("failed to read the application's auxiliary vector"))
17}
18
19fn application(reader: MemoryReader) -> Option<Entries> {
20    let stat = std::fs::read("/proc/self/stat").ok()?;
21    let fields = &stat[stat.iter().rposition(|&byte| byte == b')')? + 2..];
22    let start = std::str::from_utf8(fields.split(|&byte| byte == b' ').nth(28 - 3)?)
23        .ok()?
24        .parse::<u64>()
25        .ok()?;
26    after_stack(start, |address| {
27        let mut word = 0_u64;
28        let status = unsafe {
29            reader(
30                address as usize,
31                (&mut word as *mut u64).cast(),
32                size_of::<u64>(),
33            )
34        };
35        (status != 0).then_some(word)
36    })
37}
38
39fn kernel(bytes: &[u8]) -> Option<Entries> {
40    let mut entries = Vec::new();
41    for pair in bytes.chunks(16) {
42        let pair: &[u8; 16] = pair.try_into().ok()?;
43        let (key, value) = pair.split_at(8);
44        let key = u64::from_ne_bytes(key.try_into().unwrap());
45        let value = u64::from_ne_bytes(value.try_into().unwrap());
46        if key == 0 && value == 0 {
47            return Some(entries);
48        }
49        entries.push((key, value));
50    }
51    None
52}
53
54fn after_stack(start: u64, read_word: impl Fn(u64) -> Option<u64>) -> Option<Entries> {
55    // Same corruption bounds as the maintained native Guest. These are not
56    // new limits on Linux's accepted argument/environment contents.
57    const MOST_POINTERS: u64 = 1 << 22;
58    const MOST_ENTRIES: u64 = 256;
59    const WORD: u64 = size_of::<u64>() as u64;
60    if start == 0 {
61        return None;
62    }
63    let argc = read_word(start)?;
64    if argc >= MOST_POINTERS {
65        return None;
66    }
67    let argv_end = start.checked_add((1 + argc) * WORD)?;
68    if read_word(argv_end)? != 0 {
69        return None;
70    }
71    let mut cursor = argv_end + WORD;
72    let mut pointers = 0;
73    while read_word(cursor)? != 0 {
74        pointers += 1;
75        if pointers == MOST_POINTERS {
76            return None;
77        }
78        cursor = cursor.checked_add(WORD)?;
79    }
80    cursor = cursor.checked_add(WORD)?;
81    let mut entries = Vec::new();
82    loop {
83        let key = read_word(cursor)?;
84        if key == libc::AT_NULL {
85            return Some(entries);
86        }
87        entries.push((key, read_word(cursor.checked_add(WORD)?)?));
88        if entries.len() as u64 == MOST_ENTRIES {
89            return None;
90        }
91        cursor = cursor.checked_add(2 * WORD)?;
92    }
93}
