6fn copied_child_flock_action(operation: Option<i32>) -> i32 {
7    let Some(operation) = operation else {
8        return -libc::ENOLCK;
9    };
10    let known_flags = libc::LOCK_SH | libc::LOCK_EX | libc::LOCK_UN | libc::LOCK_NB;
11    let base = operation & !libc::LOCK_NB;
12    let valid = operation & !known_flags == 0
13        && matches!(base, libc::LOCK_SH | libc::LOCK_EX | libc::LOCK_UN);
14    if !valid || base == libc::LOCK_UN || operation & libc::LOCK_NB != 0 {
15        0
16    } else {
17        -libc::ENOLCK
18    }
19}

21/// Whether a copied child's `fcntl` with these arguments sets or clears an OFD
22/// lock (`F_OFD_SETLK`, `F_OFD_SETLKW`, whatever the lock type). Missing
23/// arguments count as a change, so the strict refusal fails closed.
24fn copied_child_fcntl_changes_ofd_lock(args: *const u64) -> bool {
25    if args.is_null() {
26        return true;
27    }
28    // SAFETY: Reverie's DYNAMORIO client passes its live six-element syscall
29    // argument array for the duration of this callback.
30    let command = unsafe { args.add(1).read() } as i32;
31    matches!(command, libc::F_OFD_SETLK | libc::F_OFD_SETLKW)
32}
33
34/// Whether `sysno` reads the clock or a resource limit, which the DYNAMORIO client no
35/// longer answers from a table of its own (Reverie A8 step 2) and which a copied
36/// child cannot take to Detcore. A relative `clock_nanosleep` on a wall or
37/// monotonic clock only waits and reads nothing; the client's table passed it
38/// to the kernel too, so it is not counted here.
39fn copied_child_reads_clock_or_limit(sysno: Sysno, args: *const u64) -> bool {
40    match sysno {
41        Sysno::clock_gettime
42        | Sysno::clock_getres
43        | Sysno::clock_settime
44        | Sysno::gettimeofday
45        | Sysno::time
46        | Sysno::getrlimit
47        | Sysno::setrlimit
48        | Sysno::prlimit64 => true,
49        Sysno::clock_nanosleep => {
50            if args.is_null() {
51                return true;
52            }
53            // SAFETY: Reverie's DYNAMORIO client passes its live six-element syscall
54            // argument array for the duration of this callback.
55            let (clock, flags) = unsafe { (args.read() as i32, args.add(1).read() as i32) };
56            let cpu_clock = clock < 0
57                || clock == libc::CLOCK_PROCESS_CPUTIME_ID
58                || clock == libc::CLOCK_THREAD_CPUTIME_ID;
59            flags != 0 || cpu_clock
60        }
61        _ => false,
62    }
63}

65pub(super) unsafe fn copied_action(
66    sysnum: i64,
67    args: *const u64,
68    strict: bool,
69    report_fd: i32,
70    emit: fn(&[u8]),
71) -> i32 {
72    let sysno = Sysno::from(sysnum as i32);
73    // Bash probes the foreground process group in a copied child before
74    // running a background shell function. Hermit's captured stderr is not a
75    // terminal, so the instrumented root observes ENOTTY for the same request.
76    // Emulate that result instead of either exposing a host terminal's process
77    // group or aborting an otherwise deterministic child.
78    //
79    // Every other ioctl remains fail-closed: copied children still cannot enter
80    // the Rust Detcore Tool, and the ABI has no guest-memory channel for safely
81    // handling socket timestamps or arbitrary device operations.
82    if sysno == Sysno::ioctl && strict {
83        let request = if args.is_null() {
84            None
85        } else {
86            // SAFETY: Reverie's DYNAMORIO client passes its live six-element syscall
87            // argument array for the duration of this callback.
88            Some(unsafe { args.add(1).read() })
89        };
90        if request == Some(libc::TIOCGPGRP) {
91            return -libc::ENOTTY;
92        }
93        return 1;
94    }
95    // TODO-HUMAN-REVIEW(PR-981): Copied DYNAMORIO children cannot enter the Rust
96    // Detcore Tool. Strict mode therefore fails closed for receive syscalls that
97    // can expose native socket timestamps. Non-strict mode retained native
98    // behavior until the descriptor-import arm below, which now answers
99    // recvmsg and recvmmsg with EPERM there.
100    // TODO-HUMAN-REVIEW(PR-972): readlink identity canonicalization also requires
101    // Detcore mediation. This ABI has neither syscall arguments nor a memory
102    // writer, so a copied child must fail closed rather than expose native
103    // pipe/socket inode identities.
104    // A copied pre-exec child has no Detcore Tool. Preserve kernel handling
105    // for malformed operations, nonblocking locks, and unlock; only blocking
106    // acquisitions can strand the child behind a parent that is waiting for it.
107    if sysno == Sysno::flock {
108        let operation = (!args.is_null()).then(|| unsafe { args.add(1).read() as i32 });
109        return copied_child_flock_action(operation);
110    }
111    // pidfd_getfd NEVER RUNS NATIVELY IN A COPIED CHILD because it is the one native
112    // route by which a copied child can reach an open file description that
113    // ANOTHER guest process holds a Detcore claim about. A sibling that took
114    // LOCK_SH through Detcore holds a known mode for its description; a
115    // copied child that duplicates that description with pidfd_getfd and
116    // runs LOCK_UN natively (the flock arm above lets unlock through) makes
117    // the sibling's claim stale, and its next refused blocking upgrade would
118    // "restore" a lock it no longer holds. The instrumented root mediates
119    // pidfd_getfd (self only); a copied child cannot enter that handler.
120    // Evidence capture refuses this call too, but only in verify runs. The
121    // protection does not depend on the unsupported-syscall policy: a stale
122    // claim corrupts flock semantics (a "restored" lock the guest released),
123    // not only determinism, and before copied children could be admitted
124    // after a fork the up-front vfork guard, which is not policy-gated,
125    // refused this shape under --allow-unsupported-syscalls too. Only the
126    // form of the refusal follows the policy: the default fail-closed policy
127    // ends the run by name; the --allow-unsupported-syscalls opt-out
128    // answers EPERM without executing the call (the errno Linux gives when
129    // ptrace access to the target is denied), as the root answers an
130    // unserviceable call with an errno under that opt-out. Ending the run
131    // there is not an option: a copied child's fail-closed exit under the
132    // opt-out left the run hung (measured: the vfork parent stayed in
133    // kernel_clone and the tree never ended). SCM_RIGHTS reception, the
134    // other import route, is refused by the recvmsg arm below the same way.
135    if sysno == Sysno::pidfd_getfd {
136        emit(b"detcore-dynamorio: refusing pidfd_getfd in copied child: a native duplicate of another process's open file description could change a lock that process's Detcore state holds a claim about\n");
137        return if strict { 1 } else { -libc::EPERM };
138    }
139    // An OFD lock change NEVER RUNS NATIVELY IN A COPIED CHILD because the
140    // `/proc/locks` ledger in Detcore's global state records which OFD locks
141    // each description holds, and a copied child's native F_OFD_SETLK or
142    // F_OFD_SETLKW (an unlock included) changes the kernel's lock without
143    // updating that record. An OFD row prints owner -1 in every PID
144    // namespace, and the ledger keeps such a row by field match, so a stale
145    // entry admits a host process's identical lock into the guest's
146    // /proc/locks. The parent's up-front OFD vfork guard cannot see a lock a
147    // sibling took through its own split copy of the description's model, so
148    // the refusal belongs at the call. F_OFD_GETLK changes nothing and stays
149    // native, as in the root, where it also sees host locks. POSIX
150    // F_SETLK/F_SETLKW stay native: POSIX rows are not in the ledger (it
151    // keeps OFD and flock rows only) and are kept by their owner pid, and
152    // Detcore caches no POSIX lock state. The protection does not depend on
153    // the unsupported-syscall policy, for the reason the pidfd_getfd arm
154    // gives: the up-front OFD and flock vfork guards it stands in for are not
155    // policy-gated. Under the --allow-unsupported-syscalls opt-out the call
156    // is answered ENOLCK without executing, as a copied child's blocking
157    // flock is (`copied_child_flock_action`), so the kernel's lock and the
158    // ledger's record stay in agreement.
159    if sysno == Sysno::fcntl && copied_child_fcntl_changes_ofd_lock(args) {
160        emit(b"detcore-dynamorio: refusing fcntl F_OFD_SETLK/F_OFD_SETLKW in copied child: a native OFD lock change would leave the /proc/locks ledger's record of that description stale\n");
161        return if strict { 1 } else { -libc::ENOLCK };
162    }
163    // recvmsg and recvmmsg NEVER RUN NATIVELY IN A COPIED CHILD, under any
164    // policy, because an SCM_RIGHTS message is the other native route by
165    // which a copied child can import an open file description another guest
166    // process holds a Detcore claim about. A sibling's promoted description
167    // (OpenFileState::Shared) keeps its canonical known mode across the
168    // sender's sendmsg; a copied child that receives it and runs LOCK_UN
169    // natively makes that claim stale, as with pidfd_getfd above. Whether a
170    // call carries a control buffer cannot be told here: the msghdr is guest
171    // memory and this ABI has no reader (a fault in the runtime would take
172    // the place of the guest's EFAULT), so every call is refused, including
173    // one without a control buffer that would import nothing. Strict
174    // execution already refused both for native socket timestamps. Under the
175    // default fail-closed policy the run ends by name; under the
176    // --allow-unsupported-syscalls opt-out the call answers EPERM without
177    // executing (a security module can refuse a receive with it), for the
178    // reason the pidfd_getfd arm gives.
179    if matches!(sysno, Sysno::recvmsg | Sysno::recvmmsg) {
180        emit(b"detcore-dynamorio: refusing recvmsg/recvmmsg in copied child: an SCM_RIGHTS message could import an open file description another process's Detcore state holds a claim about\n");
181        return if strict { 1 } else { -libc::EPERM };
182    }
183    if matches!(sysno, Sysno::readlink | Sysno::readlinkat) && strict {
184        return 1;
185    }
186    // setpgid FAILS CLOSED IN STRICT MODE BECAUSE THE COPIED CHILD CAN PERFORM
187    // IT BUT CANNOT RECORD IT.
188    //
189    // `handle_setpgid` in the instrumented root does two things: it injects the
190    // call, so the KERNEL really moves the process group, and then it mirrors
191    // that move into Detcore's model with `set_process_group`, "so
192    // group-selecting waits do not consult host /proc state". A copied child has
193    // no Detcore Tool, so it performs the first half and silently skips the
194    // second: the kernel's process groups and Detcore's model diverge, and the
195    // next `wait` with a negative pgid selects against a model that is now
196    // wrong. That is the machinery hermit#bc9d178 reclassified setpgid to
197    // determinize in the first place.
198    //
199    // ⚠️ AND A FIXED ERRNO IS THE WRONG SHAPE HERE, WHICH IS WHY THIS IS NOT
200    // `-EPERM`. The ioctl arm above emulates ENOTTY because the instrumented
201    // root observes ENOTTY too -- that is faithful. The root observes setpgid
202    // SUCCEEDING, so no errno we return is faithful, and the choice is between
203    // two divergences. EPERM is the quiet one: setpgid's return is widely
204    // ignored, so a child would carry on believing it leads a process group it
205    // does not lead. Failing closed is the loud one, and a wrong answer about
206    // process-group identity is worse than no answer.
207    //
208    // Strict-only, matching the receive/readlink arms directly above: non-strict
209    // copied children keep native behaviour, and only strict execution -- which
210    // is where determinism is actually claimed -- refuses.
211    if sysno == Sysno::setpgid && strict {
212        return 1;
213    }
214    // getpgrp, getpgid and getsid FAIL CLOSED IN STRICT MODE BECAUSE THE
215    // KERNEL'S ANSWER IS NOT DETCORE'S.
216    //
217    // The instrumented root is answered from Detcore's model (slice B3a-4c of
218    // https://github.com/rrnewton/hermit/issues/4266): a group or session
219    // inherited from outside the container's PID namespace reads 0, as Linux
220    // shows it. In a copied child the call would run natively, and the kernel
221    // answers for the process group the DYNAMORIO launcher made for the run, whose
222    // id is the root's pid. A child would read a different group from its
223    // parent's. As with setpgid above, no fixed errno is faithful (the root's
224    // calls succeed), and a wrong answer about process-group identity is
225    // worse than no answer.
226    if matches!(sysno, Sysno::getpgrp | Sysno::getpgid | Sysno::getsid) && strict {
227        return 1;
228    }
229    // Detcore answers ordinary scheduling-policy queries with SCHED_OTHER.
230    // A copied child cannot enter the Tool and may inherit a different native
231    // policy. Refuse that strict escape without inventing a second answer.
232    if sysno == Sysno::sched_getscheduler && strict {
233        // The emitter is installed before READY and inherited by copied
234        // children. Use its raw write path, without an allocating/locking
235        // stderr fallback on the copied runtime stack.
236        emit(b"detcore-dynamorio: unsupported syscall sched_getscheduler (145) in copied child\n");
237        return 1;
238    }
239    // Ordinary scheduler parameters are written by Detcore's shared Tool.
240    // A copied child cannot enter that Tool or write its modeled output through
241    // this ABI. Refuse the strict escape without a second parameter policy.
242    if sysno == Sysno::sched_getparam && strict {
243        emit(b"detcore-dynamorio: unsupported syscall sched_getparam (143) in copied child\n");
244        return 1;
245    }
246    // Detcore's shared Tool owns affinity masks. A copied child cannot
247    // enter that Tool to write its modeled CPU set; refuse the strict native
248    // escape without installing another affinity policy here.
249    if sysno == Sysno::sched_getaffinity && strict {
250        emit(b"detcore-dynamorio: unsupported syscall sched_getaffinity (204) in copied child\n");
251        return 1;
252    }
253    // Ordinary priority queries belong to Detcore's shared policy. A copied
254    // child cannot enter that Tool, so strict execution must refuse the native
255    // answer rather than install a second priority model here.
256    if sysno == Sysno::getpriority && strict {
257        emit(b"detcore-dynamorio: unsupported syscall getpriority (140) in copied child\n");
258        return 1;
259    }
260    // A copied child's clock and resource-limit reads have no deterministic
261    // answer here: the DYNAMORIO client keeps no table of its own any more, and
262    // Detcore, which answers them in the root and in forked children, cannot
263    // be entered from a copied child. Strict mode refuses them by name rather
264    // than let a vfork child read host time or host limits; non-strict keeps
265    // native behaviour, as the arms above do.
266    if strict && copied_child_reads_clock_or_limit(sysno, args) {
267        return 1;
268    }
269    if detcore::is_deterministically_refused_syscall(sysno)
270        && (strict || !detcore::is_strict_only_deterministic_refusal_syscall(sysno))
271    {
272        return 1;
273    }
274    if !detcore::is_unsupported_syscall(sysno) {
275        return 0;
276    }
277    if strict {
278        1
279    } else {
280        append_copied_syscall_record(sysnum, report_fd);
281        0
282    }
283}

285fn append_copied_syscall_record(sysnum: i64, report_fd: i32) {
286    if report_fd == -1 {
287        return;
288    }
289    let mut buffer = [0_u8; 24];
290    let mut index = buffer.len() - 1;
291    buffer[index] = b'\n';
292    let mut value = sysnum as u64;
293    loop {
294        index -= 1;
295        buffer[index] = b'0' + (value % 10) as u8;
296        value /= 10;
297        if value == 0 {
298            break;
299        }
300    }
301    index -= 1;
302    buffer[index] = b'@';
303    let _ = unsafe {
304        libc::write(
305            report_fd,
306            buffer[index..].as_ptr().cast(),
307            buffer.len() - index,
308        )
309    };
310}

525        let _guard = COPIED_CHILD_POLICY_TEST_LOCK
526            .lock()
527            .unwrap_or_else(|poisoned| poisoned.into_inner());
528        let saved = COPIED_PANIC_ON_UNSUPPORTED.load(Ordering::Acquire);
529        COPIED_PANIC_ON_UNSUPPORTED.store(true, Ordering::Release);
530
531        // Refused before reaching the host.
532        for sysno in [Sysno::perf_event_open, Sysno::remap_file_pages] {
533            assert!(detcore::is_determinized_syscall(sysno));
534            assert_eq!(
535                copied_child_action(sysno.id() as i64),
536                1,
537                "{sysno} must not reach the host from a strict copied child"
538            );
539        }
540
541        // Still diverging: emulated on the ptrace path, native here.
542        for sysno in [Sysno::ioprio_get, Sysno::ioprio_set] {
543            assert!(detcore::is_determinized_syscall(sysno));
544            assert!(!detcore::is_deterministically_refused_syscall(sysno));
545            assert_eq!(
546                copied_child_action(sysno.id() as i64),
547                0,
548                "{sysno} disposition changed; update this test and issue #1793"
549            );
550        }
551
552        COPIED_PANIC_ON_UNSUPPORTED.store(saved, Ordering::Release);
553    }

577        let _guard = COPIED_CHILD_POLICY_TEST_LOCK
578            .lock()
579            .unwrap_or_else(|poisoned| poisoned.into_inner());
580        let saved = COPIED_PANIC_ON_UNSUPPORTED.load(Ordering::Acquire);

719    fn copied_child_refuses_keyring_syscalls_under_strict() {
720        let _guard = COPIED_CHILD_POLICY_TEST_LOCK
721            .lock()
722            .unwrap_or_else(|poisoned| poisoned.into_inner());
723        let saved = COPIED_PANIC_ON_UNSUPPORTED.load(Ordering::Acquire);
724
725        // Strict (panic-on-unsupported): keyring syscalls are refused so the
726        // copied child cannot mutate host keyrings or trigger request-key
727        // upcalls. `1` tells the native client to exit the isolated runtime
728        // tree (fail closed), matching the pre-848 Unsupported behavior.
729        COPIED_PANIC_ON_UNSUPPORTED.store(true, Ordering::Release);
730        assert_eq!(copied_child_action(libc::SYS_keyctl), 1);
731        assert_eq!(copied_child_action(libc::SYS_add_key), 1);
732        assert_eq!(copied_child_action(libc::SYS_request_key), 1);
733        // A supported syscall still runs natively even under strict mode.
734        assert_eq!(copied_child_action(libc::SYS_getpid), 0);
735
736        // Non-strict: keyring syscalls fall through to native pass-through,
737        // matching the root process's non-strict keyring behavior.
738        COPIED_PANIC_ON_UNSUPPORTED.store(false, Ordering::Release);
739        assert_eq!(copied_child_action(libc::SYS_keyctl), 0);
740        assert_eq!(copied_child_action(libc::SYS_add_key), 0);
741        assert_eq!(copied_child_action(libc::SYS_request_key), 0);
742
743        COPIED_PANIC_ON_UNSUPPORTED.store(saved, Ordering::Release);
744    }

746    #[test]
747    fn copied_child_gate_refuses_deterministic_refusal_families_in_strict() {
748        let _guard = COPIED_CHILD_POLICY_TEST_LOCK
749            .lock()
750            .unwrap_or_else(|poisoned| poisoned.into_inner());
751        // The copied pre-exec child runs natively with no Detcore tool. Under
752        // strict mode the gate must fail-close (return 1) not only for the
753        // classic Unsupported set but for the full deterministic-refusal
754        // boundary (splice/tee/vmsplice, perf_event_open, the keyring family),
755        // otherwise strict guests execute those syscalls natively against the
756        // host. Report fd is left at its -1 default so `append_copied_syscall_record`
757        // is a no-op and the non-strict branch has no observable side effect.
758        let previous = COPIED_PANIC_ON_UNSUPPORTED.load(Ordering::Acquire);
759
760        // Every family member must be recognized by the shared predicate.
761        for sysno in [
762            Sysno::splice,
763            Sysno::tee,
764            Sysno::vmsplice,
765            Sysno::perf_event_open,
766            Sysno::keyctl,
767            Sysno::add_key,
768            Sysno::request_key,
769        ] {
770            assert!(
771                detcore::is_deterministically_refused_syscall(sysno),
772                "{sysno:?} should be in Detcore's deterministic-refusal boundary"
773            );
774        }
775
776        // Strict: refused families and Unsupported syscalls fail-close (1);
777        // ordinary passthrough syscalls continue natively (0).
778        COPIED_PANIC_ON_UNSUPPORTED.store(true, Ordering::Release);
779        for sysnum in [
780            libc::SYS_splice,
781            libc::SYS_tee,
782            libc::SYS_vmsplice,
783            libc::SYS_perf_event_open,
784            libc::SYS_keyctl,
785            libc::SYS_add_key,
786            libc::SYS_request_key,
787            libc::SYS_ioctl,
788            libc::SYS_recvmsg,
789            libc::SYS_recvmmsg,
790            libc::SYS_readlink,
791            libc::SYS_readlinkat,
792        ] {
793            assert_eq!(
794                copied_child_action(sysnum),
795                1,
796                "strict copied child must refuse syscall {sysnum}"
797            );
798        }
799        let mut ioctl_args = [0; 6];
800        ioctl_args[1] = libc::TIOCGPGRP;
801        assert_eq!(
802            copied_child_action_with_args(libc::SYS_ioctl, ioctl_args),
803            -libc::ENOTTY,
804            "TIOCGPGRP must receive the deterministic non-terminal result"
805        );
806        ioctl_args[1] = 0x8906; // SIOCGSTAMP_OLD
807        assert_eq!(
808            copied_child_action_with_args(libc::SYS_ioctl, ioctl_args),
809            1,
810            "socket timestamp ioctls must remain fail-closed"
811        );
812        // SAFETY: A null argument vector is an explicit fail-closed ABI test;
813        // the callback checks it before dereferencing.
814        assert_eq!(
815            unsafe { copied_gate_for_test(libc::SYS_ioctl, std::ptr::null()) },
816            1,
817            "missing ioctl arguments must fail closed"
818        );
819        for sysnum in [libc::SYS_read, libc::SYS_write, libc::SYS_getpid] {
820            assert_eq!(
821                copied_child_action(sysnum),
822                0,
823                "strict copied child must allow ordinary syscall {sysnum}"
824            );
825        }
826
827        // Non-strict: strict-only compatibility families continue natively,
828        // while unconditional fixed-error families still fail closed because
829        // the copied-child ABI cannot inject their deterministic errno.
830        COPIED_PANIC_ON_UNSUPPORTED.store(false, Ordering::Release);
831        for sysnum in [
832            libc::SYS_splice,
833            libc::SYS_keyctl,
834            libc::SYS_ioctl,
835            libc::SYS_read,
836            libc::SYS_readlink,
837            libc::SYS_readlinkat,
838        ] {
839            assert_eq!(
840                copied_child_action(sysnum),
841                0,
842                "non-strict copied child must allow syscall {sysnum}"
843            );
844        }
845        // recvmsg and recvmmsg never run natively in a copied child: an
846        // SCM_RIGHTS message could import another process's description. Under
847        // the non-strict policy the call does not run and answers EPERM.
848        for sysnum in [libc::SYS_recvmsg, libc::SYS_recvmmsg] {
849            assert_eq!(
850                copied_child_action(sysnum),
851                -libc::EPERM,
852                "non-strict copied child must answer EPERM for syscall {sysnum}"
853            );
854        }
855        for sysnum in [
856            libc::SYS_perf_event_open,
857            libc::SYS_openat2,
858            libc::SYS_io_uring_setup,
859        ] {
860            assert_eq!(
861                copied_child_action(sysnum),
862                1,
863                "unconditional refusal must fail closed for syscall {sysnum}"
864            );
865        }
866
867        for operation in [libc::LOCK_SH, libc::LOCK_EX] {
868            let mut args = [0; 6];
869            args[1] = operation as u64;
870            assert_eq!(
871                copied_child_action_with_args(libc::SYS_flock, args),
872                -libc::ENOLCK,
873                "a copied child must not enter a blocking flock operation",
874            );
875        }
876        for operation in [
877            0,
878            libc::LOCK_SH | libc::LOCK_EX,
879            libc::LOCK_SH | libc::LOCK_NB,
880            libc::LOCK_EX | libc::LOCK_NB,
881            libc::LOCK_UN,
882            libc::LOCK_UN | libc::LOCK_NB,
883        ] {
884            let mut args = [0; 6];
885            args[1] = operation as u64;
886            assert_eq!(
887                copied_child_action_with_args(libc::SYS_flock, args),
888                0,
889                "operation {operation} must retain kernel validation or safe nonblocking behavior",
890            );
891        }
892        assert_eq!(
893            unsafe { copied_gate_for_test(libc::SYS_flock, std::ptr::null()) },
894            -libc::ENOLCK,
895            "missing copied-child flock arguments must fail closed",
896        );
897
898        COPIED_PANIC_ON_UNSUPPORTED.store(previous, Ordering::Release);
899    }
