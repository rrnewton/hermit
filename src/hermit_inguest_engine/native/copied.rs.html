1//! Real independent-COW child post, followed by its later native initializer.
2//!
3//! The C owner supplies the original operation/result before its complete
4//! post body clears the birth transaction, then samples the actual child
5//! context after that body and register canonicalization. A shared-VM/vfork
6//! child never enters this transfer. No inherited process lock is acquired.

30/// Actual native birth transaction, borrowed across its unchanged post body.
31/// These scalar facts are not the affine Rust transfer authority.
32#[derive(Clone, Copy)]
33#[repr(C)]
34pub struct CopyPost {
35    pub number: i64,
36    pub result: i64,
37    pub flags: u64,
38}
39
40// On a fatal error after the original continuation has moved, retaining this
41// allocation is deliberate. Dropping an inherited parent Record outside its
42// real later initialization/start scopes could run an inherited destructor or
43// lock. The native fatal path terminates that process; it cannot retry it.
44struct Transfer {
45    host: Host,
46    rpc: NativeRpc<GlobalState>,
47    parent: Option<CopiedNativeParent<Detcore>>,
48}

50/// # Safety
51/// This is the real child's returning post for fork/separate-VM clone/clone3.
52/// `facts` was sampled from that original birth transaction before its native
53/// post body; the supplied callback is sampled after ALL of that body and
54/// canonical register writes. The child has an independent COW address space,
55/// actual leader identity, original live image, allocator/TLS/stack entry and
56/// no remaining concurrent native use of the original scratch. This must not
57/// be called at a synthetic child post or for any shared-VM/vfork child.
58/// `output` is a unique null replacement-locator slot in the real C owner.
59#[unsafe(no_mangle)]
60pub unsafe extern "C" fn hermit_engine_native_copied_post_v1(
61    opaque: *mut c_void,
62    callback: *const Callback,
63    facts: *const CopyPost,
64    output: *mut *mut c_void,
65) -> i32 {
66    status((|| {
67        if opaque.is_null()
68            || callback.is_null()
69            || facts.is_null()
70            || output.is_null()
71            || unsafe { !output.read().is_null() }
72        {
73            return Err(io::Error::from_raw_os_error(libc::EINVAL));
74        }
75        let callback = unsafe { &*callback };
76        let facts = unsafe { facts.read() };
77        let flags = CloneFlags::from_bits_truncate(facts.flags);
78        let prohibited = CloneFlags::CLONE_VM
79            | CloneFlags::CLONE_VFORK
80            | CloneFlags::CLONE_THREAD
81            | CloneFlags::CLONE_FILES;
82        if facts.result != 0
83            || flags.intersects(prohibited)
84            || !matches!(
85                facts.number,
86                libc::SYS_fork | libc::SYS_clone | libc::SYS_clone3
87            )
88        {
89            return Err(io::Error::from_raw_os_error(libc::ESTALE));
90        }
91        let thread = unsafe { &*opaque.cast::<Thread>() };
92        let mut entry = unsafe { thread.enter_copied(callback)? };
93        let inherited = unsafe { owner::inherited(callback, thread.scalar_identity())? };
94        let original = entry
95            .state()
96            .pending
97            .as_ref()
98            .ok_or_else(|| io::Error::from_raw_os_error(libc::ESTALE))?;
99        let (number, arguments) = original.operation();
100        if number != facts.number || (number == libc::SYS_clone && arguments[0] != facts.flags) {
101            return Err(io::Error::from_raw_os_error(libc::ESTALE));
102        }
103        let ordinal = inherited.next_child_ordinal.load(Ordering::SeqCst);
104        if ordinal == 0 {
105            return Err(io::Error::from_raw_os_error(libc::ESTALE));
106        }
107        let identity = callback.identity;
108        let native_identity = unsafe { identity.admitted()? };
109        let table = callback.validate(identity)?.selected()?;
110        let auxv = unsafe { callback.copy_auxv()? };
111        let services = unsafe { callback.select(&table, &auxv)? };
112        let rebound = unsafe {
113            inherited.host.rebind_copied_native_process(
114                &mut entry.state_mut().pending,
115                &services,
116                &inherited.rpc,
117                super::bootstrap::fresh_runtime(),
118            )?
119        };
120        // All pre-transfer refusals left the original Deferred present. From
121        // this point a fatal failure must retain, not drop, its real Record.
122        let mut retained = ManuallyDrop::new(Transfer {
123            host: rebound.host,
124            rpc: rebound.rpc,
125            parent: Some(rebound.parent),
126        });
127        let transfer = &mut *retained;
128        let parent = unsafe {
129            transfer.host.complete_copied_native_post(
130                &services,
131                &transfer.rpc,
132                &mut transfer.parent,
133            )?
134        };
135        let transfer = ManuallyDrop::into_inner(retained);
136        let created = Box::into_raw(Box::new(Thread::copied(
137            identity,
138            native_identity,
139            parent,
140            flags,
141        )));
142        // Preserve the Runtime's actual inherited ordinal value. The later
143        // actual initializer alone increments it for the unchanged RNG pack;
144        // a copied process never resets it to a fresh-image value of one.
145        let engine = owner::Engine {
146            host: transfer.host,
147            rpc: transfer.rpc,
148            config: Arc::clone(&inherited.config),
149            root: Pid::from_raw(identity.tool_pid),
150            physical_owner: identity.physical_pid,
151            image: identity.image,
152            pending_parents: Mutex::new(HashMap::new()),
153            next_child_ordinal: AtomicU64::new(ordinal),
154        };
155        // Failure retains both new allocations and the original Record for
156        // the fatal process exit. It never destroys an inherited Engine.
157        unsafe {
158            owner::publish_copied(inherited, engine)?;
159        }
160        entry.state_mut().retired = true;
161        drop(entry);
162        // rebind moved the original Deferred/Record; this old locator now
163        // holds no state, scratch or copied-parent destructor to release.
164        unsafe {
165            drop(Box::from_raw(opaque.cast::<Thread>()));
166        }
167        unsafe {
168            output.write(created.cast());
169        }
170        Ok(())
171    })())
172}

174/// # Safety
175/// The actual native first-subsequent-event initializer selected by the
176/// existing copied-process owner, after the real child post above. It may not
177/// be replaced by an arbitrary syscall callback or a request to create a Tool.
178#[unsafe(no_mangle)]
179pub unsafe extern "C" fn hermit_engine_native_initialize_copied_thread_v1(
180    opaque: *mut c_void,
181    callback: *const Callback,
182) -> i32 {
183    status((|| {
184        if opaque.is_null() || callback.is_null() {
185            return Err(io::Error::from_raw_os_error(libc::EINVAL));
186        }
187        let callback = unsafe { &*callback };
188        let thread = unsafe { &*opaque.cast::<Thread>() };
189        let mut entry = thread.enter(callback)?;
190        let engine = owner::installed()?;
191        if entry.state().handle.is_some()
192            || entry.state().pending.is_some()
193            || engine.physical_owner != callback.identity.physical_pid
194            || engine.root.as_raw() != callback.identity.tool_pid
195            || callback.identity.physical_pid != callback.identity.physical_tid
196        {
197            return Err(io::Error::from_raw_os_error(libc::ESTALE));
198        }
199        let table = callback.validate(callback.identity)?.selected()?;
200        let auxv = unsafe { callback.copy_auxv()? };
201        let services = unsafe { callback.select(&table, &auxv)? };
202        let parent = entry
203            .state_mut()
204            .copied
205            .take()
206            .ok_or_else(|| io::Error::from_raw_os_error(libc::ESTALE))?;
207        let result = unsafe {
208            engine.host.initialize_postcompleted_copied_native_thread(
209                &services,
210                &engine.rpc,
211                parent.parent,
212                |original| {
213                    original.clone_flags = Some(parent.flags);
214                    Ok(())
215                },
216                |state, original| {
217                    super::lifecycle::project(state, callback.identity);
218                    state.reset_copied_open_file_locks();
219                    // The returning copied child cannot touch inherited Tool
220                    // locks in post. Before ANY child Tool callback, its fresh
221                    // state receives the maintained successful-copy invalidation.
222                    state.forget_flock_modes();
223                    let entropy = super::lifecycle::entropy(engine, callback.identity.tool_pid)?;
224                    state.reseed_child_rngs(original, entropy);
225                    Ok(())
226                },
227            )
228        };
229        let handle = match result {
230            Ok(handle) => handle,
231            Err(error) => {
232                entry.fail();
233                return Err(error);
234            }
235        };
236        entry.state_mut().handle = Some(handle);
237        drop(entry);
238        // Common lifecycle invokes actual thread_start once, no post_exec,
239        // then drops the whole inherited parent under its original scopes.
240        unsafe { super::lifecycle::finish_first_entry(thread, callback) }
241    })())
242}
