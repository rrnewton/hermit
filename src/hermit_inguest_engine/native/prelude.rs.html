45/// Private one-use witness created only after the real PrepareExec RPC. It
46/// retains mechanism identity and scalar observations, not a callback frame,
47/// borrowed auxv pointer, Rust Guest, synthetic flag or a replacement lease.
48pub(super) struct PreparedExec {
49    identity: Identity,
50    context: usize,
51    mechanisms: usize,
52    branches: u64,
53}
54
55impl PreparedExec {
56    fn new(callback: &Callback) -> Self {
57        Self {
58            identity: callback.identity,
59            context: callback.context,
60            mechanisms: callback.mechanisms as usize,
61            branches: callback.branches,
62        }
63    }
64    fn validate(&self, callback: &Callback) -> io::Result<()> {
65        callback.validate(self.identity)?;
66        if callback.context != self.context || callback.mechanisms as usize != self.mechanisms {
67            return Err(io::Error::from_raw_os_error(libc::ESTALE));
68        }
69        Ok(())
70    }
71}

90pub(super) fn prepare(
91    callback: &Callback,
92    raw: &Mechanisms,
93    number: i64,
94    arguments: [u64; 6],
95) -> Result<PreparedCall, Errno> {
96    let first = TOTAL_SYSCALLS.fetch_add(1, Ordering::Relaxed) == 0;
97    if first {
98        super::diagnostics::emit(b"detcore-dynamorio: entered Rust syscall callback\n");
99    }
100    let read = raw.read_memory.ok_or(Errno::EIO)?;
101    let write = raw.write_memory.ok_or(Errno::EIO)?;
102    let mut selected = arguments;
103    // Exact Current-ABI self-prlimit lowering. Queued signal targets remain
104    // in the Tool namespace; the native injector lowers them after handling.
105    if number == libc::SYS_prlimit64 && selected[0] as i32 == callback.identity.tool_pid {
106        selected[0] = callback.identity.physical_pid as u32 as u64;
107    }
108    if !prlimit_new_limit_is_readable(number, &arguments, |address, bytes| unsafe {
109        read(address, bytes.as_mut_ptr(), bytes.len()) != 0
110    }) || !multiplexed_io_inputs_are_readable(number, &arguments, |address, bytes| unsafe {
111        read(address, bytes.as_mut_ptr(), bytes.len()) != 0
112    }) {
113        return Err(Errno::EFAULT);
114    }
115    let probe = if number == libc::SYS_getrandom {
116        match getrandom_writable_prefix(&arguments, |address, bytes| unsafe {
117            Ok(write(address, bytes.as_ptr(), bytes.len()))
118        }) {
119            Some(Ok(probe)) => {
120                selected[1] = probe.writable as u64;
121                Some(probe)
122            }
123            Some(Err(error)) => return Err(error),
124            None => None,
125        }
126    } else {
127        None
128    };
129    branches(callback.branches);
130    let clones = process_clone_properties(number, &arguments, |address, bytes| unsafe {
131        read(address, bytes.as_mut_ptr(), bytes.len()) != 0
132    });
133    Ok(PreparedCall {
134        arguments: selected,
135        probe,
136        clones,
137        first,
138    })
139}

155/// Concrete work inside the already admitted common Guest. The outer entry
156/// alone retains/completes its native lease; this function cannot reenter it.
157/// # Safety
158/// All borrowed arguments belong to the same with_native_guest callback.
159pub(super) unsafe fn dispatch(
160    call: DispatchCall<'_>,
161    pending_exec: &mut Option<PreparedExec>,
162    tool: &Detcore,
163    guest: &mut ConcreteGuest<'_>,
164    tail: &TailResult,
165    services: &NativeServices<'_>,
166) -> NativeCallbackOutcome<Result<i64, Error>> {
167    let DispatchCall {
168        engine,
169        callback,
170        number,
171        original,
172        prepared,
173    } = call;
174    if number == libc::SYS_execveat {
175        return NativeCallbackOutcome::Ready(Ok(-i64::from(Errno::ENOSYS.into_raw())));
176    }
177    if prepared.clones.shares_files_without_thread {
178        return failure(b"detcore-dynamorio: refusing process clone with CLONE_FILES without CLONE_THREAD; copied child cannot share descriptor-table provenance\n");
179    }
180    if prepared.clones.shares_memory_without_thread {
181        return failure(b"detcore-dynamorio: refusing process clone with CLONE_VM without CLONE_THREAD or CLONE_VFORK; copied child cannot own inherited Detcore state\n");
182    }
183    if prepared.clones.blocks_parent && guest.thread_state().has_unsafe_vfork_flock_state() {
184        return failure(b"detcore-dynamorio: refusing vfork/CLONE_VFORK while an open file description may hold a flock\n");
185    }
186    if prepared.clones.blocks_parent && guest.thread_state().has_unsafe_vfork_ofd_state() {
187        return failure(b"detcore-dynamorio: refusing vfork/CLONE_VFORK while an open file description may hold an OFD lock\n");
188    }
189    if requires_native_lifecycle(number) {
190        if engine.config.sequentialize_threads
191            && super::bootstrap::coordinated()
192            && (number == libc::SYS_execve || prepared.clones.copies_process)
193        {
194            guest
195                .thread_state()
196                .promote_shared_open_files()
197                .expect("could not publish shared OFDs before native lifecycle");
198        }
199        if number == libc::SYS_execve {
200            if pending_exec.is_some() {
201                return NativeCallbackOutcome::Ready(Err(io::Error::from_raw_os_error(
202                    libc::EBUSY,
203                )
204                .into()));
205            }
206            let mm = guest.thread_state().mm_id;
207            let blocking = guest.thread_state().exec_blocking_overrides();
208            let robust = drive_native_ready(tool.exec_robust_list_wakes(guest));
209            let read = callback
210                .validate(callback.identity)
211                .expect("admitted native callback")
212                .read_memory
213                .expect("admitted native reader");
214            let image = read_guest_path(original[0], read)
215                .and_then(|path| tool.exec_image_at(guest, libc::AT_FDCWD, &path));
216            drive_native_ready(detcore::prepare_exec(guest, mm, blocking, robust, image));
217            *pending_exec = Some(PreparedExec::new(callback));
218            if super::bootstrap::owns_background(callback.identity.physical_pid) {
219                super::image::request_pause();
220                assert_eq!(
221                    super::image::generation(),
222                    callback.identity.image,
223                    "native image generation changed while pausing for exec"
224                );
225            }
226        }
227        // This affine continuation is still active through the entire real C
228        // post body. Scalar action3 preserves the old execute-original route,
229        // rather than selecting action2's rewritten-operation path.
230        return NativeCallbackOutcome::Deferred {
231            number,
232            args: original,
233        };
234    }
235    let read = callback
236        .validate(callback.identity)
237        .expect("admitted native callback")
238        .read_memory
239        .expect("admitted native reader");
240    update_memory_hash(number, &original, read);
241    if prepared.first {
242        super::diagnostics::emit(b"detcore-dynamorio: dispatching first syscall through Detcore\n");
243    }
244    let saved = prepared
245        .probe
246        .filter(|probe| probe.writable < probe.requested)
247        .map(|probe| (probe, guest.thread_state().prng.clone()));
248    unsafe {
249        engine.host.dispatch_native_syscall_in_guest_with(
250            tool,
251            guest,
252            tail,
253            services,
254            &engine.rpc,
255            move |guest, mut outcome| {
256                if let Some((probe, original_prng)) = saved {
257                    guest.thread_state_mut().prng = original_prng;
258                    advance_getrandom_prng(&mut guest.thread_state_mut().prng, probe.consumed());
259                    // Observe the REAL Tool success here, before the common
260                    // classifier converts Errno. A concurrent later write
261                    // fault must not be relabelled a successful short fill.
262                    if matches!(outcome, NativeCallbackOutcome::Ready(Ok(_))) {
263                        let value = if probe.writable == 0 {
264                            -i64::from(Errno::EFAULT.into_raw())
265                        } else {
266                            probe.writable as i64
267                        };
268                        outcome = NativeCallbackOutcome::Ready(Ok(value));
269                    }
270                }
271                outcome
272            },
273        )
274    }
275}

290/// # Safety
291/// Real returning PARENT/native same-process post work only. A copied/vfork
292/// child must not enter its inherited Tool/registry here. C retains its birth
293/// owner and completes all post bookkeeping before final Deferred completion.
294#[unsafe(no_mangle)]
295pub unsafe extern "C" fn hermit_engine_native_process_clone_result_v1(
296    opaque: *mut c_void,
297    callback: *const Callback,
298    facts: *const ClonePost,
299) -> i32 {
300    status((|| {
301        if opaque.is_null() || callback.is_null() || facts.is_null() {
302            return Err(io::Error::from_raw_os_error(libc::EINVAL));
303        }
304        let callback = unsafe { &*callback };
305        let facts = unsafe { facts.read() };
306        let thread = unsafe { &*opaque.cast::<Thread>() };
307        let mut entry = thread.enter(callback)?;
308        let engine = owner::installed()?;
309        let table = callback.validate(callback.identity)?.selected()?;
310        let auxv = unsafe { callback.copy_auxv()? };
311        let services = unsafe { callback.select(&table, &auxv)? };
312        let pending = entry
313            .state_mut()
314            .pending
315            .as_mut()
316            .ok_or_else(|| io::Error::from_raw_os_error(libc::EPROTO))?;
317        let mut post = event(callback, facts.number, [0; 6], 0);
318        unsafe {
319            engine.host.with_native_post_step(
320                pending,
321                &services,
322                &engine.rpc,
323                &mut post,
324                |tool, guest, _tail| {
325                    if successful_process_clone_result(facts.number, facts.result) {
326                        guest.thread_state().forget_flock_modes();
327                    }
328                    if facts.result > 0
329                        && facts.tool_child_tid > 0
330                        && matches!(
331                            facts.number,
332                            libc::SYS_fork | libc::SYS_clone | libc::SYS_clone3
333                        )
334                        && facts.flags & (libc::CLONE_THREAD | libc::CLONE_VFORK) as u64 == 0
335                        && facts.exit_signal < 0
336                    {
337                        return NativeCallbackOutcome::Ready(Err(io::Error::other(
338                            "invalid native child exit signal",
339                        )
340                        .into()));
341                    }
342                    if let Some((tid, flags, signal)) = process_child_registration(
343                        facts.number,
344                        facts.result,
345                        facts.tool_child_tid,
346                        facts.flags,
347                        facts.exit_signal,
348                    ) {
349                        drive_native_ready(tool.register_external_child(
350                            guest,
351                            tid,
352                            facts.child_tid_address as usize,
353                            flags,
354                            signal,
355                            Some((facts.result as i32, facts.result as i32)),
356                        ));
357                    }
358                    NativeCallbackOutcome::Ready(Ok(()))
359                },
360            )
361        }
362        .map_err(io::Error::other)
363    })())
364}

366/// # Safety
367/// Actual rejected native exec on the same original task/image. The private
368/// PreparedExec and original Deferred are consumed/borrowed exactly once.
369#[unsafe(no_mangle)]
370pub unsafe extern "C" fn hermit_engine_native_exec_failed_v1(
371    opaque: *mut c_void,
372    callback: *const Callback,
373) -> i32 {
374    // Same maintained ordering: scheduler progress is restored BEFORE the
375    // CancelExec RPC, including the uninitialized/no-preparation path.
376    super::image::resume();
377    if opaque.is_null() {
378        return 0;
379    }
380    status((|| {
381        if callback.is_null() {
382            return Err(io::Error::from_raw_os_error(libc::EINVAL));
383        }
384        let actual = unsafe { &*callback };
385        let thread = unsafe { &*opaque.cast::<Thread>() };
386        let mut entry = thread.enter(actual)?;
387        let Some(prepared) = entry.state().prepared_exec.as_ref() else {
388            return Ok(());
389        };
390        prepared.validate(actual)?;
391        let branches = prepared.branches;
392        // Only the scalar branch observation is retained from the original
393        // pre-event. All pointed-to current facts are borrowed afresh now.
394        let callback = Callback {
395            branches,
396            ..*actual
397        };
398        let engine = owner::installed()?;
399        let table = callback.validate(callback.identity)?.selected()?;
400        let auxv = unsafe { callback.copy_auxv()? };
401        let services = unsafe { callback.select(&table, &auxv)? };
402        let _prepared = entry
403            .state_mut()
404            .prepared_exec
405            .take()
406            .expect("checked prepared exec");
407        let pending = entry
408            .state_mut()
409            .pending
410            .as_mut()
411            .ok_or_else(|| io::Error::from_raw_os_error(libc::EPROTO))?;
412        let mut post = event(&callback, libc::SYS_execve, [0; 6], 0);
413        unsafe {
414            engine.host.with_native_post_step(
415                pending,
416                &services,
417                &engine.rpc,
418                &mut post,
419                |_tool, guest, _tail| {
420                    drive_native_ready(detcore::cancel_exec(guest));
421                    NativeCallbackOutcome::Ready(Ok(()))
422                },
423            )
424        }
425        .map_err(io::Error::other)
426    })())
427}

429/// Read the actual native Tool counters; C retains its original evidence wire.
430/// # Safety
431/// Each output is aligned/writable for one u64 and owned by this native call.
432#[unsafe(no_mangle)]
433pub unsafe extern "C" fn hermit_engine_native_totals_v1(
434    branches: *mut u64,
435    syscalls: *mut u64,
436    rewritten: *mut u64,
437    memory_hash: *mut u64,
438) -> i32 {
439    if branches.is_null() || syscalls.is_null() || rewritten.is_null() || memory_hash.is_null() {
440        return -libc::EINVAL;
441    }
442    unsafe {
443        branches.write(TOTAL_BRANCHES.load(Ordering::Relaxed));
444        syscalls.write(TOTAL_SYSCALLS.load(Ordering::Relaxed));
445        rewritten.write(TOTAL_REWRITTEN.load(Ordering::Relaxed));
446        memory_hash.write(MEMORY_HASH.load(Ordering::SeqCst));
447    }
448    0
449}

453fn read_guest_path(address: u64, read_memory: MemoryReader) -> Option<PathBuf> {
454    use std::os::unix::ffi::OsStringExt as _;
455    const PAGE: u64 = 4096;
456    let mut path = Vec::new();
457    let mut at = address;
458    while path.len() < libc::PATH_MAX as usize {
459        let chunk = (PAGE - at % PAGE) as usize;
460        let mut bytes = vec![0_u8; chunk];
461        if unsafe { read_memory(at as usize, bytes.as_mut_ptr(), chunk) } == 0 {
462            return None;
463        }
464        if let Some(end) = bytes.iter().position(|&byte| byte == 0) {
465            path.extend_from_slice(&bytes[..end]);
466            return (path.len() < libc::PATH_MAX as usize)
467                .then(|| PathBuf::from(std::ffi::OsString::from_vec(path)));
468        }
469        path.extend_from_slice(&bytes);
470        at += chunk as u64;
471    }
472    None
473}

475fn advance_getrandom_prng(prng: &mut impl rand::Rng, bytes: usize) {
476    let mut words = [0_u64; RANDOM_FILL_CHUNK_BYTES / std::mem::size_of::<u64>()];
477    let mut advanced = 0;
478    while advanced < bytes {
479        let chunk_len = (bytes - advanced).min(RANDOM_FILL_CHUNK_BYTES);
480        let chunk =
481            unsafe { std::slice::from_raw_parts_mut(words.as_mut_ptr().cast::<u8>(), chunk_len) };
482        prng.fill(chunk);
483        advanced += chunk_len;
484    }
485}

487fn update_memory_hash(sysnum: i64, args: &[u64], read_memory: MemoryReader) {
488    if sysnum != libc::SYS_write {
489        return;
490    }
491    let address = args[1] as usize;
492    let length = args[2] as usize;
493    if address == 0 || length > MAX_OBSERVED_BUFFER {
494        return;
495    }
496
497    let mut bytes = vec![0; length];
498    if unsafe { read_memory(address, bytes.as_mut_ptr(), length) } == 0 {
499        return;
500    }
501
502    let mut hash = FNV_OFFSET;
503    for byte in sysnum
504        .to_le_bytes()
505        .into_iter()
506        .chain(args[0].to_le_bytes())
507        .chain((length as u64).to_le_bytes())
508        .chain(bytes)
509    {
510        hash ^= u64::from(byte);
511        hash = hash.wrapping_mul(FNV_PRIME);
512    }
513    MEMORY_HASH.fetch_add(hash, Ordering::SeqCst);
