9// A schedule replay of a --chaos recording follows the recorded slice ends
10// rather than drawing chaos timeslices of its own: two threads spin through
11// branch runs longer than a typical chaos slice and print the virtual clock
12// after each, so a replay that preempts at a point of its own leaves the
13// recording (--die-on-desync) or prints other times.
14//
15// Both runs are --strict: a replay follows its record exactly only where the
16// retired-branch counter passes Reverie's validation, and where it does not
17// (an AMD Zen host without the SpecLockMap fix) Hermit refuses the run, as for
18// the other strict fixtures, rather than let skid make this fixture pass or
19// fail by chance. The record and the outputs go in the fresh working
20// directory the lit runner makes for the fixture.
21//
22// RUN: %hermit run --strict --chaos --seed=3 --record-preemptions-to=r.json -- %me > rec && %hermit run --strict --chaos --seed=3 --die-on-desync --replay-schedule-from=r.json -- %me > rep && diff rec rep

24use std::hint::black_box;
25use std::time::SystemTime;
26use std::time::UNIX_EPOCH;
27
28fn work(id: u64) {
29    for round in 0..6 {
30        let mut sink = 0u64;
31        for i in 0..3_000_000u64 {
32            sink = sink.wrapping_add(black_box(i ^ id));
33        }
34        black_box(sink);
35        let now = SystemTime::now().duration_since(UNIX_EPOCH).unwrap();
36        println!("t{id} r{round} {}.{:09}", now.as_secs(), now.subsec_nanos());
37    }
38}
39
40fn main() {
41    let other = std::thread::spawn(|| work(1));
42    work(2);
43    other.join().unwrap();
44}
