34/// Environment variable containing the coordinator's Unix-domain socket path.
35// TODO-HUMAN-REVIEW(PR-745): Review the private SaBRe exec environment contract.
36pub const RPC_SOCKET_ENV: &str = "REVERIE_SABRE_HERMIT_RPC_SOCKET";
37
38/// Private opt-in for forwarding injected-process Detcore INFO events.
39pub const DETLOG_FORWARD_ENV: &str = "REVERIE_SABRE_HERMIT_FORWARD_DETLOG";

41// AUTONOMOUS-BOT-IMPLEMENTED
42// TODO-HUMAN-REVIEW(PR-771): Review fork-inherited SaBRe coordinator discovery.
43static RPC_SOCKET: OnceLock<PathBuf> = OnceLock::new();
44
45thread_local! {
46    // Function detours are registered before the process-local tool is built.
47    // Adapter construction may itself call libc getrandom, so recursive calls
48    // must use the captured original while the first call initializes the
49    // process-local Plugin.
50    static LIBC_GETRANDOM_ACTIVE: Cell<bool> = const { Cell::new(false) };
51}
52
53struct DetourGuard<'a>(&'a Cell<bool>);
54
55impl Drop for DetourGuard<'_> {
56    fn drop(&mut self) {
57        self.0.set(false);
58    }
59}

61fn coordinator_socket() -> Option<PathBuf> {
62    if let Some(socket) = RPC_SOCKET.get() {
63        return Some(socket.clone());
64    }
65
66    // SAFETY: Plugin construction runs before SaBRe starts guest callbacks.
67    let requested = unsafe { sabre::take_private_env(RPC_SOCKET_ENV) };
68    remember_coordinator_socket(&RPC_SOCKET, requested.as_deref())
69}

71struct RawStderr;
72
73impl Write for RawStderr {
74    fn write(&mut self, bytes: &[u8]) -> io::Result<usize> {
75        loop {
76            let written = unsafe {
77                libc::write(
78                    libc::STDERR_FILENO,
79                    bytes.as_ptr().cast::<libc::c_void>(),
80                    bytes.len(),
81                )
82            };
83            if written >= 0 {
84                return Ok(written as usize);
85            }
86            let error = io::Error::last_os_error();
87            if error.kind() != io::ErrorKind::Interrupted {
88                return Err(error);
89            }
90        }
91    }
92
93    fn flush(&mut self) -> io::Result<()> {
94        Ok(())
95    }
96}

98fn forward_detlog(message: std::fmt::Arguments<'_>) {
99    let mut stderr = RawStderr;
100    let _ = stderr.write_all(b"INFO detcore: DETLOG ");
101    let _ = stderr.write_fmt(message);
102    let _ = stderr.write_all(b"\n");
103}
104
105fn init_detlog_forwarder() {
106    // SAFETY: Plugin construction runs before SaBRe starts guest callbacks.
107    let requested = unsafe { sabre::take_private_env(DETLOG_FORWARD_ENV) };
108    if requested.as_deref() != Some(OsStr::new("1")) {
109        return;
110    }
111
112    // Stderr is protected by reverie-sabre and is captured separately during
113    // verification. A direct sink avoids tracing's thread-local dispatcher:
114    // libc may issue its final exit_group after Rust TLS destruction begins.
115    let _ = detcore::detlog::set_forwarder(forward_detlog);
116}

118fn remember_coordinator_socket(
119    slot: &OnceLock<PathBuf>,
120    requested: Option<&OsStr>,
121) -> Option<PathBuf> {
122    slot.get().cloned().or_else(|| {
123        let requested = requested.map(PathBuf::from)?;
124        Some(slot.get_or_init(|| requested).clone())
125    })
126}

128// AUTONOMOUS-BOT-IMPLEMENTED
129// TODO-HUMAN-REVIEW(PR-845): Review SaBRe guest comm-name restoration.
130fn guest_comm_from_args(args: impl IntoIterator<Item = OsString>) -> Option<CString> {
131    let program = args.into_iter().next()?;
132    let name = Path::new(&program).file_name()?.as_bytes();
133    CString::new(&name[..name.len().min(15)]).ok()
134}
135
136fn restore_guest_comm_name(thread_id: u32) {
137    if thread_id != unsafe { libc::getpid() as u32 } {
138        return;
139    }
140    let Some(name) = guest_comm_from_args(std::env::args_os()) else {
141        return;
142    };
143    unsafe {
144        libc::prctl(libc::PR_SET_NAME, name.as_ptr() as usize, 0, 0, 0);
145    }
146}

154fn is_post_load_bootstrap_random(syscall: &Syscall) -> bool {
155    matches!(
156        syscall,
157        Syscall::Getrandom(call)
158            if call.buflen() == 32 && call.flags() == libc::GRND_NONBLOCK as usize
159    )
160}

162/// Returns the Detcore SaBRe plugin built beside the running Hermit binary.
163// AUTONOMOUS-BOT-IMPLEMENTED
164// TODO-HUMAN-REVIEW(PR-738): Review the Hermit-to-SaBRe plugin artifact boundary.
165pub fn runtime_library_path() -> io::Result<PathBuf> {
166    let executable = std::env::current_exe()?;
167    let directory = executable.parent().ok_or_else(|| {
168        io::Error::new(
169            io::ErrorKind::NotFound,
170            "Hermit executable has no parent directory",
171        )
172    })?;
173    [
174        directory.join("libdetcore_sabre.so"),
175        directory.join("deps/libdetcore_sabre.so"),
176    ]
177    .into_iter()
178    .find(|path| path.is_file())
179    .ok_or_else(|| {
180        io::Error::new(
181            io::ErrorKind::NotFound,
182            format!(
183                "libdetcore_sabre.so was not built beside {}",
184                executable.display()
185            ),
186        )
187    })
188}

190struct Plugin {
191    adapter: RemoteReverieAdapter<Detcore>,
192    // The SaBRe-injected runtime requests its hash seed on the first rewritten
193    // syscall after post-load. Keep that tool-private draw out of Detcore's
194    // guest-visible random stream.
195    post_load_syscall_pending: AtomicBool,
196}
197
198impl Plugin {
199    fn connect() -> Self {
200        init_detlog_forwarder();
201        let socket = coordinator_socket().unwrap_or_else(|| panic!("{RPC_SOCKET_ENV} is not set"));
202
203        let adapter = RemoteReverieAdapter::connect(socket)
204            .expect("failed to connect Detcore SaBRe plugin to coordinator");
205
206        Self {
207            adapter,
208            post_load_syscall_pending: AtomicBool::new(false),
209        }
210    }
211
212    // AUTONOMOUS-BOT-IMPLEMENTED
213    // TODO-HUMAN-REVIEW(PR-1117): Review SaBRe bootstrap-random isolation.
214    fn handle_post_load_syscall(&self, syscall: &Syscall) -> Option<Result<usize, Errno>> {
215        if !self.post_load_syscall_pending.swap(false, Ordering::AcqRel) {
216            return None;
217        }
218
219        if !is_post_load_bootstrap_random(syscall) {
220            return None;
221        }
222
223        let Syscall::Getrandom(call) = syscall else {
224            unreachable!("bootstrap-random classifier accepted a non-getrandom syscall")
225        };
226        let buffer = call.buf().ok_or(Errno::EFAULT);
227        Some(buffer.and_then(|buffer| {
228            let mut memory = LocalMemory::new();
229            memory.write_exact(buffer, &[0; 32]).map(|()| 32)
230        }))
231    }
232
233    fn handle_vdso(&self, sysno: Sysno, args: SyscallArgs) -> i32 {
234        self.adapter
235            .handle_syscall(Syscall::from_raw(sysno, args))
236            .map_or_else(|errno| -errno.into_raw(), |result| result as i32)
237    }
238}

240#[sabre::tool]
241impl reverie_sabre::Tool for Plugin {
242    type Client = ();
243
244    // AUTONOMOUS-BOT-IMPLEMENTED
245    // TODO-HUMAN-REVIEW(PR-1214): Review libc getrandom function interception.
246    // GNU patch calls glibc's getrandom wrapper from libc, whose raw syscall
247    // site SaBRe deliberately does not rewrite. Route that wrapper through the
248    // same Detcore syscall handler used by rewritten guest call sites.
249    #[detour(lib = "libc", func = "getrandom")]
250    fn libc_getrandom(
251        buffer: *mut libc::c_void,
252        length: libc::size_t,
253        flags: libc::c_uint,
254    ) -> libc::ssize_t {
255        LIBC_GETRANDOM_ACTIVE.with(|active| {
256            if active.replace(true) {
257                return Self::libc_getrandom_undetoured(buffer, length, flags);
258            }
259            let _guard = DetourGuard(active);
260
261            let syscall = Syscall::from_raw(
262                Sysno::getrandom,
263                SyscallArgs::new(buffer as usize, length, flags as usize, 0, 0, 0),
264            );
265            match <Self as reverie_sabre::ToolGlobal>::global()
266                .adapter
267                .handle_syscall(syscall)
268            {
269                Ok(result) => result as libc::ssize_t,
270                Err(error) => {
271                    unsafe { *libc::__errno_location() = error.into_raw() };
272                    -1
273                }
274            }
275        })
276    }
277
278    fn new(_client: Self::Client) -> Self {
279        Self::connect()
280    }
281
282    fn new_without_legacy_rpc() -> Option<Self> {
283        Some(Self::connect())
284    }
285
286    fn syscall(&self, syscall: Syscall, _memory: &LocalMemory) -> Result<usize, Errno> {
287        if let Some(result) = self.handle_post_load_syscall(&syscall) {
288            return result;
289        }
290        self.adapter.handle_syscall(syscall)
291    }
292
293    fn syscall_with_inject<F>(
294        &self,
295        syscall: Syscall,
296        _memory: &LocalMemory,
297        inject: F,
298    ) -> Result<usize, Errno>
299    where
300        F: FnMut() -> usize + Send + Sync,
301    {
302        self.adapter.handle_syscall_with_inject(syscall, inject)
303    }
304
305    // AUTONOMOUS-BOT-IMPLEMENTED
306    // TODO-HUMAN-REVIEW(PR-755): Review SaBRe RDTSC virtualization.
307    fn rdtsc(&self) -> u64 {
308        require_virtual_rdtsc(self.adapter.handle_rdtsc())
309    }
310
311    // AUTONOMOUS-BOT-IMPLEMENTED
312    // TODO-HUMAN-REVIEW(PR-755): Review SaBRe clock_gettime VDSO virtualization.
313    fn vdso_clock_gettime(&self, clockid: libc::clockid_t, tp: *mut libc::timespec) -> i32 {
314        self.handle_vdso(
315            Sysno::clock_gettime,
316            SyscallArgs::new(clockid as usize, tp as usize, 0, 0, 0, 0),
317        )
318    }
319
320    // AUTONOMOUS-BOT-IMPLEMENTED
321    // TODO-HUMAN-REVIEW(PR-755): Review SaBRe getcpu VDSO virtualization.
322    fn vdso_getcpu(&self, cpu: *mut u32, node: *mut u32, unused: usize) -> i32 {
323        self.handle_vdso(
324            Sysno::getcpu,
325            SyscallArgs::new(cpu as usize, node as usize, unused, 0, 0, 0),
326        )
327    }
328
329    // AUTONOMOUS-BOT-IMPLEMENTED
330    // TODO-HUMAN-REVIEW(PR-755): Review SaBRe gettimeofday VDSO virtualization.
331    fn vdso_gettimeofday(&self, tv: *mut libc::timeval, tz: *mut libc::timezone) -> i32 {
332        self.handle_vdso(
333            Sysno::gettimeofday,
334            SyscallArgs::new(tv as usize, tz as usize, 0, 0, 0, 0),
335        )
336    }
337
338    // AUTONOMOUS-BOT-IMPLEMENTED
339    // TODO-HUMAN-REVIEW(PR-755): Review SaBRe time VDSO virtualization.
340    fn vdso_time(&self, tloc: *mut libc::time_t) -> i32 {
341        self.handle_vdso(Sysno::time, SyscallArgs::new(tloc as usize, 0, 0, 0, 0, 0))
342    }
343
344    fn on_thread_start(&self, thread_id: u32) {
345        restore_guest_comm_name(thread_id);
346        self.adapter.handle_thread_start(thread_id);
347    }
348
349    fn on_post_load(&self) {
350        self.adapter.handle_post_exec();
351        self.post_load_syscall_pending
352            .store(true, Ordering::Release);
353    }
354
355    fn on_thread_exit(&self, thread_id: u32) {
356        self.adapter.handle_thread_exit(thread_id);
357    }
358}
