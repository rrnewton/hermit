55fn coordinator_socket() -> Option<PathBuf> {
56    if let Some(socket) = RPC_SOCKET.get() {
57        return Some(socket.clone());
58    }
59
60    // SAFETY: Plugin construction runs before SaBRe starts guest callbacks.
61    let requested = unsafe { sabre::take_private_env(RPC_SOCKET_ENV) };
62    remember_coordinator_socket(&RPC_SOCKET, requested.as_deref())
63}

65// The coordinator sends its per-target INFO answer (detcore::detlog::ForwardPolicy), so a
66// target-scoped RUST_LOG keeps the same records here as in-process. A value this plugin
67// cannot read is a coordinator/plugin mismatch: say so rather than drop records silently.
68fn detlog_forward_policy(requested: Option<&OsStr>) -> Result<Option<ForwardPolicy>, String> {
69    let Some(requested) = requested else {
70        return Ok(None);
71    };
72    let requested = requested
73        .to_str()
74        .ok_or_else(|| format!("{DETLOG_FORWARD_ENV} is not UTF-8"))?;
75    ForwardPolicy::decode(requested).map(Some)
76}

78// Says why records are not forwarded, on the standard error the forwarder would use,
79// with one raw write as detcore::detlog::forward_to_stderr does.
80fn report_unforwarded_detlogs(error: &str) {
81    let line = format!("hermit SaBRe plugin: no DETLOG forwarding: {error}\n");
82    // SAFETY: `line` is a live, initialized byte slice of the given length.
83    let _ = unsafe {
84        libc::write(
85            libc::STDERR_FILENO,
86            line.as_ptr().cast::<libc::c_void>(),
87            line.len(),
88        )
89    };
90}

92fn init_detlog_forwarder() {
93    // SAFETY: Plugin construction runs before SaBRe starts guest callbacks.
94    let requested = unsafe { sabre::take_private_env(DETLOG_FORWARD_ENV) };
95    let policy = match detlog_forward_policy(requested.as_deref()) {
96        Ok(Some(policy)) => policy,
97        Ok(None) => return,
98        Err(error) => {
99            report_unforwarded_detlogs(&error);
100            return;
101        }
102    };
103
104    // With the socket `hermit run --verify` passes, each record goes to it from
105    // the descriptor reverie-sabre protects from the guest, tagged with the
106    // sending thread, and the coordinator writes it into the log at that
107    // thread's scheduler turn, as for in-guest LiteInst. Without one, records
108    // go to stderr, which reverie-sabre keeps open and verification captures
109    // separately.
110    // A socket that was requested but could not be adopted (closed or replaced
111    // before adoption, or given up) never falls back to stderr, which the
112    // guest may have pointed at a file of its own: its records are counted and
113    // dropped, and the coordinator refuses the run (detlog::check_forwarded_count).
114    let forwarder: detcore::detlog::DetlogForwarder =
115        if sabre::tool_output_fd().is_some() || sabre::tool_output_requested() {
116            // If reverie-sabre ever has to give the socket up (a guest dup onto
117            // its number with no free descriptor to move it to), the coordinator
118            // reads this and refuses the run's records as incomplete.
119            sabre::set_retirement_message(detcore::detlog::FORWARDING_RETIRED_NOTICE);
120            forward_detlog_to_socket
121        } else {
122            detcore::detlog::forward_to_stderr
123        };
124    let _ = detcore::detlog::set_forwarder(forwarder, policy);
125}

127/// Sends one Detcore record on the coordinator's DETLOG socket.
128fn forward_detlog_to_socket(
129    target: &str,
130    record_suffix: &str,
131    index: u64,
132    message: std::fmt::Arguments<'_>,
133) {
134    sabre::with_tool_output(|socket| {
135        detcore::detlog::send_forwarded_record(socket, target, record_suffix, index, message)
136    });
137}

139fn remember_coordinator_socket(
140    slot: &OnceLock<PathBuf>,
141    requested: Option<&OsStr>,
142) -> Option<PathBuf> {
143    slot.get().cloned().or_else(|| {
144        let requested = requested.map(PathBuf::from)?;
145        Some(slot.get_or_init(|| requested).clone())
146    })
147}

149// The coordinator publishes its fingerprint to the image it starts, as a
150// launch-specification setting, and take_private_env consumes it. A forked child
151// constructs the plugin again from the parent's memory, where it is already gone, so
152// without this memo the child took the "published no fingerprint" branch and
153// wrote that warning, with this build's fingerprint, to the guest's own fd 2.
154// Fork inherits this slot as it inherits RPC_SOCKET; execve starts a new image,
155// where reverie-sabre passes the REVERIE_SABRE_* settings again in a launch
156// specification.
157static COORDINATOR_FINGERPRINT: OnceLock<Option<String>> = OnceLock::new();
158
159/// What the coordinator published to this process image (`None`: nothing),
160/// read once. The flag is true only for the call that read it; later calls,
161/// including a forked child's construction, reuse that reading.
162fn remember_coordinator_fingerprint(
163    slot: &OnceLock<Option<String>>,
164    take: impl FnOnce() -> Option<String>,
165) -> (Option<String>, bool) {
166    let mut first = false;
167    let published = slot.get_or_init(|| {
168        first = true;
169        take()
170    });
171    (published.clone(), first)
172}

176fn guest_comm_from_args(args: impl IntoIterator<Item = OsString>) -> Option<CString> {
177    let program = args.into_iter().next()?;
178    let name = Path::new(&program).file_name()?.as_bytes();
179    CString::new(&name[..name.len().min(15)]).ok()
180}

182fn restore_guest_comm_name(thread_id: u32) {
183    if thread_id != unsafe { libc::getpid() as u32 } {
184        return;
185    }
186    let Some(name) = guest_comm_from_args(std::env::args_os()) else {
187        return;
188    };
189    unsafe {
190        libc::prctl(libc::PR_SET_NAME, name.as_ptr() as usize, 0, 0, 0);
191    }
192}

200fn is_post_load_bootstrap_random(syscall: &Syscall) -> bool {
201    matches!(
202        syscall,
203        Syscall::Getrandom(call)
204            if call.buflen() == 32 && call.flags() == libc::GRND_NONBLOCK as usize
205    )
206}

208/// Returns the Detcore SaBRe plugin built beside the running Hermit binary.
209// AUTONOMOUS-BOT-IMPLEMENTED
210// TODO-HUMAN-REVIEW(PR-738): Review the Hermit-to-SaBRe plugin artifact boundary.
211pub fn runtime_library_path() -> io::Result<PathBuf> {
212    let executable = std::env::current_exe()?;
213    let directory = executable.parent().ok_or_else(|| {
214        io::Error::new(
215            io::ErrorKind::NotFound,
216            "Hermit executable has no parent directory",
217        )
218    })?;
219    [
220        directory.join("libdetcore_sabre.so"),
221        directory.join("deps/libdetcore_sabre.so"),
222    ]
223    .into_iter()
224    .find(|path| path.is_file())
225    .ok_or_else(|| {
226        io::Error::new(
227            io::ErrorKind::NotFound,
228            format!(
229                "libdetcore_sabre.so was not built beside {}",
230                executable.display()
231            ),
232        )
233    })
234}

260fn initial_image_generation(pid: i32, bytes: &[u8]) -> io::Result<u64> {
261    // The kernel's comm field is raw bytes, including possible ") " bytes.
262    // Decode only the PID and start-time fields, outside the final delimiter.
263    let end = bytes
264        .windows(2)
265        .rposition(|pair| pair == b") ")
266        .ok_or_else(|| io::Error::other("invalid process stat"))?;
267    let prefix = &bytes[..end];
268    let owner_end = prefix
269        .windows(2)
270        .position(|pair| pair == b" (")
271        .ok_or_else(|| io::Error::other("invalid stat owner"))?;
272    let owner = std::str::from_utf8(&prefix[..owner_end]).map_err(io::Error::other)?;
273    if owner.parse::<i32>().map_err(io::Error::other)? != pid {
274        return Err(io::Error::other("initial random handoff owner mismatch"));
275    }
276    let start = bytes[end + 2..]
277        .split(u8::is_ascii_whitespace)
278        .filter(|field| !field.is_empty())
279        .nth(19)
280        .ok_or_else(|| io::Error::other("missing process generation"))?;
281    std::str::from_utf8(start)
282        .map_err(io::Error::other)?
283        .parse()
284        .map_err(io::Error::other)
285}

287fn own_initial_image(pid: i32) -> io::Result<detcore::random::InitialImage> {
288    fn bounded(path: &str) -> io::Result<Vec<u8>> {
289        let mut bytes = Vec::new();
290        std::fs::File::open(path)?
291            .take(4097)
292            .read_to_end(&mut bytes)?;
293        if bytes.len() > 4096 {
294            return Err(io::Error::other("initial image identity exceeds bound"));
295        }
296        Ok(bytes)
297    }
298    let before = bounded("/proc/self/stat")?;
299    let start_time_ticks = initial_image_generation(pid, &before)?;
300    let auxv = bounded("/proc/self/auxv")?;
301    if auxv.len() % 16 != 0 {
302        return Err(io::Error::other("incomplete initial auxv"));
303    }
304    let mut at_random = None;
305    let mut terminated = false;
306    for row in auxv.as_chunks::<16>().0 {
307        let key = u64::from_ne_bytes(row[..8].try_into().unwrap());
308        let value = u64::from_ne_bytes(row[8..].try_into().unwrap());
309        if key == libc::AT_NULL {
310            terminated = value == 0;
311            break;
312        }
313        if key == libc::AT_RANDOM && at_random.replace(value as usize).is_some() {
314            return Err(io::Error::other("duplicate initial AT_RANDOM"));
315        }
316    }
317    if !terminated {
318        return Err(io::Error::other("unterminated initial auxv"));
319    }
320    Ok(detcore::random::InitialImage {
321        pid,
322        start_time_ticks,
323        at_random: at_random
324            .filter(|p| *p != 0)
325            .ok_or_else(|| io::Error::other("missing initial AT_RANDOM"))?,
326    })
327}

329impl Plugin {
330    /// Refuse to connect when this plugin and the coordinator were built from
331    /// different configuration or clock RPC definitions.
332    ///
333    /// This plugin is a separate Cargo artifact that lands in the same target
334    /// directory as `hermit`, so changing `Config` or `DetTime` -- or merely
335    /// switching branches -- leaves it stale while everything still looks built.
336    /// `Config` crosses the wire during the handshake, and `DetTime` is the first
337    /// field in each Detcore request. A stale plugin can decode either against
338    /// the wrong layout. The damage was never the staleness; it was the
339    /// diagnosis cost: one added `bool` field surfaced as
340    /// `Decode(InvalidBooleanValue(20))` at connect, which names no version and
341    /// points nowhere near the plugin, and it blocked every SaBRe measurement
342    /// until someone guessed.
343    ///
344    /// Checked BEFORE the RPC connect so the mismatch is reported instead of
345    /// being re-encountered as a codec error a few frames later.
346    fn check_coordinator_compatibility() {
347        let (expected, first_reading) =
348            remember_coordinator_fingerprint(&COORDINATOR_FINGERPRINT, || {
349                // SAFETY: plugin construction runs before SaBRe starts guest callbacks.
350                unsafe { sabre::take_private_env(CONFIG_FINGERPRINT_ENV) }
351                    .map(|v| v.to_string_lossy().into_owned())
352            });
353        let ours = config_wire_fingerprint();
354        match expected {
355            Some(expected) if expected == ours => {}
356            Some(expected) => panic!(
357                "Detcore SaBRe plugin/coordinator MISMATCH: this plugin's Config and clock RPC \
358                 definitions have fingerprint {ours}, the coordinator expects {expected}. \
359                 The plugin is a separate artifact in the same target directory and is stale -- rebuild \
360                 it against this coordinator: cargo build -p detcore-sabre"
361            ),
362            // An older coordinator does not publish the fingerprint. Say so and
363            // continue: refusing here would break pairs that are actually fine,
364            // and a guard that rejects matched pairs is worse than no guard.
365            None if first_reading => eprintln!(
366                "detcore-sabre: coordinator published no configuration and clock RPC fingerprint \
367                 ({CONFIG_FINGERPRINT_ENV} unset); proceeding unguarded. This plugin's fingerprint \
368                 is {ours}."
369            ),
370            // A forked child of an image that already said so.
371            None => {}
372        }
373    }
374
375    fn connect() -> Self {
376        init_detlog_forwarder();
377        Self::check_coordinator_compatibility();
378        let socket = coordinator_socket().unwrap_or_else(|| panic!("{RPC_SOCKET_ENV} is not set"));
379
380        let adapter = RemoteReverieAdapter::<Detcore>::connect_with_root_initializer(
381            socket,
382            |config, pid, state| {
383                // This runs only after the existing inherited-fork decision, on
384                // the normally constructed guest root. No clocks/metadata are
385                // imported and no coordinator readiness event is manufactured.
386                let image = own_initial_image(pid.as_raw())?;
387                let mut bytes = [0; detcore::random::MAX_INITIAL_STATE_BYTES];
388                let count = sabre::bootstrap::take_state(&mut bytes)
389                    .map_err(io::Error::other)?
390                    .ok_or_else(|| {
391                        io::Error::other("required initial random handoff was not negotiated")
392                    })?;
393                match detcore::random::decode_loader_state(&bytes[..count], config, image)
394                    .map_err(io::Error::other)?
395                {
396                    detcore::random::LoaderState::InitialRandom { .. } => {
397                        state
398                            .apply_initial_random_state(&bytes[..count], config, image)
399                            .map_err(io::Error::other)?;
400                    }
401                    detcore::random::LoaderState::ObservedExecContinuation
402                    | detcore::random::LoaderState::InitialStaticLegacy => {
403                        // The held loader and supervisor proved this legacy
404                        // image. Preserve every normal constructor field and
405                        // let the existing post-exec callback run unchanged.
406                    }
407                }
408                Ok(())
409            },
410        )
411        .expect("failed to connect Detcore SaBRe plugin to coordinator");
412
413        Self {
414            adapter,
415            post_load_syscall_pending: AtomicBool::new(false),
416        }
417    }
418
419    // AUTONOMOUS-BOT-IMPLEMENTED
420    // TODO-HUMAN-REVIEW(PR-1117): Review SaBRe bootstrap-random isolation.
421    fn handle_post_load_syscall(&self, syscall: &Syscall) -> Option<Result<usize, Errno>> {
422        if !self.post_load_syscall_pending.swap(false, Ordering::AcqRel) {
423            return None;
424        }
425
426        if !is_post_load_bootstrap_random(syscall) {
427            return None;
428        }
429
430        let Syscall::Getrandom(call) = syscall else {
431            unreachable!("bootstrap-random classifier accepted a non-getrandom syscall")
432        };
433        let buffer = call.buf().ok_or(Errno::EFAULT);
434        Some(buffer.and_then(|buffer| {
435            let mut memory = LocalMemory::new();
436            memory.write_exact(buffer, &[0; 32]).map(|()| 32)
437        }))
438    }
439
440    fn handle_vdso(&self, sysno: Sysno, args: SyscallArgs) -> i32 {
441        self.adapter
442            .handle_syscall(Syscall::from_raw(sysno, args))
443            .map_or_else(|errno| -errno.into_raw(), |result| result as i32)
444    }
445}

451    // AUTONOMOUS-BOT-IMPLEMENTED
452    // TODO-HUMAN-REVIEW(PR-1214): Review libc getrandom function interception.
453    // Guest calls retain libc's algorithm, return/errno and domain without
454    // constructing the tool. Plugin calls use Linux directly so their native
455    // entropy cannot seed libc/vDSO opaque state later shared with the guest.
456    // Rewritten libc and initial dynamic-bootstrap vDSO syscall sites still
457    // intercept guest entropy; static/later-exec vDSO coverage is unchanged.
458    // Serving a guest public request directly from Detcore would bypass libc's
459    // key-refill/ChaCha path and change ptrace-parity bytes.
460    #[detour(lib = "libc", func = "getrandom")]
461    fn libc_getrandom(
462        buffer: *mut libc::c_void,
463        length: libc::size_t,
464        flags: libc::c_uint,
465    ) -> libc::ssize_t {
466        if unsafe { sabre::ffi::calling_from_plugin() } {
467            // libc::syscall preserves the C return/errno convention and leaves
468            // the caller's domain intact. The kernel validates the buffer.
469            return unsafe {
470                libc::syscall(libc::SYS_getrandom, buffer, length, flags) as libc::ssize_t
471            };
472        }
473        Self::libc_getrandom_undetoured(buffer, length, flags)
474    }

477        true

546    /// Tell Detcore a signal arrived, before the guest's handler runs.
547    ///
548    /// Reverie's central handler delivers the signal, but nothing forwarded
549    /// that fact to the tool: this method was the default empty one, and the
550    /// remote adapter carried syscalls, RDTSC and lifecycle events but no
551    /// signals. So Detcore never recorded a `SignalReceived` event and never
552    /// made the `ResourceID::InboundSignal` request its scheduler needs
553    /// (`detcore/src/lib.rs:1228`), and under this backend its model of the
554    /// guest simply had no signals in it.
555    ///
556    /// Detcore answers with the signal to deliver, or `None` to suppress it.
557    /// Suppression is not actionable from here -- Reverie's central handler has
558    /// already committed to running the guest action by the time it calls this
559    /// -- so a suppressed or failed forward is reported rather than dropped.
560    fn handle_signal_event(&self, signal: i32) {
561        let Ok(signal) = Signal::try_from(signal) else {
562            return;
563        };
564        match self.adapter.handle_signal(signal) {
565            Ok(Some(_)) => {}
566            Ok(None) => sabre::eprintln!(
567                "detcore-sabre: Detcore suppressed signal {}, but Reverie's central handler has already committed to the guest action",
568                signal
569            ),
570            Err(error) => sabre::eprintln!(
571                "detcore-sabre: forwarding signal {} to Detcore failed: {}",
572                signal,
573                error
574            ),
575        }
576    }

578    fn on_thread_start(&self, thread_id: u32) {
579        restore_guest_comm_name(thread_id);
580        self.adapter.handle_thread_start(thread_id);
581    }

583    fn on_post_load(&self) {
584        self.adapter.handle_post_exec();
585        self.post_load_syscall_pending
586            .store(true, Ordering::Release);

601    fn initial_generation_accepts_raw_worker_comm_and_rejects_invalid_identity() {
602        struct RestoreComm {
603            tid: i32,
604            name: [u8; 16],
605        }
606        impl Drop for RestoreComm {
607            fn drop(&mut self) {
608                assert_eq!(unsafe { libc::syscall(libc::SYS_gettid) } as i32, self.tid);
609                assert_eq!(
610                    unsafe { libc::prctl(libc::PR_SET_NAME, self.name.as_ptr()) },
611                    0
612                );
613            }
614        }
615        // /proc/self/stat names the leader. Use this owned libtest worker's
616        // actual stat to exercise the same production parser without renaming
617        // the leader or claiming that an injected SaBRe guest was executed.
618        let tid = unsafe { libc::syscall(libc::SYS_gettid) } as i32;
619        let path = format!("/proc/{tid}/stat");
620        let expected = initial_image_generation(tid, &std::fs::read(&path).unwrap()).unwrap();
621        let mut restore = RestoreComm { tid, name: [0; 16] };
622        assert_eq!(
623            unsafe { libc::prctl(libc::PR_GET_NAME, restore.name.as_mut_ptr()) },
624            0
625        );
626        assert_eq!(
627            unsafe { libc::prctl(libc::PR_SET_NAME, c"raw) \xff) task".as_ptr()) },
628            0
629        );
630        let bytes = std::fs::read(path).unwrap();
631        assert!(bytes.len() <= 4096);
632        assert!(std::str::from_utf8(&bytes).is_err());
633        assert!(bytes.windows(12).any(|row| row == b"raw) \xff) task"));
634        assert_eq!(initial_image_generation(tid, &bytes).unwrap(), expected);
635        assert!(
636            initial_image_generation(tid + 1, &bytes)
637                .unwrap_err()
638                .to_string()
639                .contains("owner mismatch")
640        );
641        assert!(initial_image_generation(tid, b"missing delimiter").is_err());
642        let end = bytes.windows(2).rposition(|row| row == b") ").unwrap();
643        let mut fields: Vec<&[u8]> = bytes[end + 2..]
644            .split(u8::is_ascii_whitespace)
645            .filter(|field| !field.is_empty())
646            .collect();
647        let mut malformed = bytes[..end + 2].to_vec();
648        fields[19] = b"not-a-number";
649        malformed.extend(fields.join(&b' '));
650        assert!(initial_image_generation(tid, &malformed).is_err());
651        assert!(initial_image_generation(tid, &bytes[..end + 2]).is_err());
652        let owner_end = bytes.windows(2).position(|row| row == b" (").unwrap();
653        let mut malformed_owner = b"not-a-pid".to_vec();
654        malformed_owner.extend_from_slice(&bytes[owner_end..]);
655        assert!(initial_image_generation(tid, &malformed_owner).is_err());
656        drop(restore);
657    }

702    #[test]
703    fn registers_libc_getrandom_detour() {
704        let detour = <Plugin as reverie_sabre::Tool>::detours()
705            .iter()
706            .find(|detour| unsafe { CStr::from_ptr(detour.fn_name) }.to_bytes() == b"getrandom")
707            .expect("libc getrandom detour should be registered");
708
709        assert_eq!(
710            unsafe { CStr::from_ptr(detour.lib_name) }.to_bytes(),
711            b"libc"
712        );
713
714        use std::cell::Cell;
715
716        type Original = fn(*mut libc::c_void, libc::size_t, libc::c_uint) -> libc::ssize_t;
717        type Detour =
718            unsafe extern "C" fn(*mut libc::c_void, libc::size_t, libc::c_uint) -> libc::ssize_t;
719
720        #[derive(Clone, Copy, Debug, PartialEq, Eq)]
721        struct Observation {
722            buffer: usize,
723            length: usize,
724            flags: libc::c_uint,
725            from_plugin: bool,
726            errno: libc::c_int,
727        }
728
729        thread_local! {
730            static EXPECTED_BUFFER: Cell<*mut libc::c_void> = const { Cell::new(std::ptr::null_mut()) };
731            static OBSERVED: Cell<Option<Observation>> = const { Cell::new(None) };
732            static CALLS: Cell<usize> = const { Cell::new(0) };
733        }
734
735        fn controlled_original(
736            buffer: *mut libc::c_void,
737            length: libc::size_t,
738            flags: libc::c_uint,
739        ) -> libc::ssize_t {
740            OBSERVED.set(Some(Observation {
741                buffer: buffer as usize,
742                length,
743                flags,
744                from_plugin: unsafe { sabre::ffi::calling_from_plugin() },
745                errno: unsafe { *libc::__errno_location() },
746            }));
747            CALLS.set(CALLS.get() + 1);
748            if flags == 0x8000_0001 {
749                unsafe { *libc::__errno_location() = libc::EINVAL };
750                return -1;
751            }
752            if length == 0 {
753                return 0;
754            }
755            // Do not dereference a pointer or extent corrupted by the wrapper;
756            // the caller's exact observation and output assertions report it.
757            if !buffer.is_null() && buffer == EXPECTED_BUFFER.get() && length == 16 && flags == 0 {
758                unsafe {
759                    std::ptr::copy_nonoverlapping(b"libc-vd".as_ptr(), buffer.cast::<u8>(), 7);
760                }
761            }
762            7
763        }
764
765        struct RestoreThreadState {
766            from_plugin: bool,
767            errno: libc::c_int,
768        }
769
770        impl Drop for RestoreThreadState {
771            fn drop(&mut self) {
772                EXPECTED_BUFFER.set(std::ptr::null_mut());
773                unsafe {
774                    if self.from_plugin {
775                        sabre::ffi::enter_plugin();
776                    } else {
777                        sabre::ffi::exit_plugin();
778                    }
779                    *libc::__errno_location() = self.errno;
780                }
781            }
782        }
783
784        let _restore = RestoreThreadState {
785            from_plugin: unsafe { sabre::ffi::calling_from_plugin() },
786            errno: unsafe { *libc::__errno_location() },
787        };
788        // The macro stores Original's Rust signature and returns its C-ABI
789        // stub through SaBRe's erased function-pointer ABI. Exercise those
790        // generated functions, not a direct call to the detour body.
791        // This test alone installs the original; all probe state is per-thread.
792        let original = unsafe {
793            std::mem::transmute::<Original, sabre::ffi::void_void_fn>(
794                controlled_original as Original,
795            )
796        };
797        let stub = unsafe {
798            std::mem::transmute::<sabre::ffi::void_void_fn, Detour>((detour.icept_callback)(
799                original,
800            ))
801        };
802        unsafe { sabre::ffi::exit_plugin() };
803        for (length, flags, expected_result, expected_errno) in [
804            (16, 0, 7, libc::E2BIG),
805            (16, 0x8000_0001, -1, libc::EINVAL),
806            (0, 0, 0, libc::E2BIG),
807        ] {
808            let mut bytes = [0xa5u8; 16];
809            let buffer = if length == 0 {
810                std::ptr::null_mut()
811            } else {
812                bytes.as_mut_ptr().cast::<libc::c_void>()
813            };
814            EXPECTED_BUFFER.set(buffer);
815            OBSERVED.set(None);
816            CALLS.set(0);
817            unsafe { *libc::__errno_location() = libc::E2BIG };
818            let result = unsafe { stub(buffer, length, flags) };
819            let errno = unsafe { *libc::__errno_location() };
820            let domain_after = unsafe { sabre::ffi::calling_from_plugin() };
821            assert_eq!(CALLS.get(), 1);
822            assert_eq!(
823                OBSERVED.get(),
824                Some(Observation {
825                    buffer: buffer as usize,
826                    length,
827                    flags,
828                    from_plugin: false,
829                    errno: libc::E2BIG,
830                })
831            );
832            assert_eq!(result, expected_result);
833            assert_eq!(errno, expected_errno);
834            assert!(!domain_after);
835            let mut expected_bytes = [0xa5; 16];
836            if expected_result == 7 {
837                expected_bytes[..7].copy_from_slice(b"libc-vd");
838            }
839            assert_eq!(bytes, expected_bytes);
840            EXPECTED_BUFFER.set(std::ptr::null_mut());
841        }
842
843        unsafe { sabre::ffi::enter_plugin() };
844        for (length, flags, null_buffer, expected_result, expected_errno) in [
845            (16, 0, false, 16, libc::E2BIG),
846            (16, 0x8000_0001, false, -1, libc::EINVAL),
847            (0, 0, true, 0, libc::E2BIG),
848            (1, 0, true, -1, libc::EFAULT),
849        ] {
850            let mut bytes = [0xa5u8; 24];
851            let buffer = if null_buffer {
852                std::ptr::null_mut()
853            } else {
854                bytes[4..20].as_mut_ptr().cast::<libc::c_void>()
855            };
856            EXPECTED_BUFFER.set(buffer);
857            OBSERVED.set(None);
858            CALLS.set(0);
859            unsafe { *libc::__errno_location() = libc::E2BIG };
860            let result = unsafe { stub(buffer, length, flags) };
861            let errno = unsafe { *libc::__errno_location() };
862            let domain_after = unsafe { sabre::ffi::calling_from_plugin() };
863            assert_eq!(CALLS.get(), 0, "plugin entropy must bypass libc state");
864            assert_eq!(OBSERVED.get(), None);
865            assert_eq!(result, expected_result);
866            assert_eq!(errno, expected_errno);
867            assert!(domain_after);
868            assert_eq!(&bytes[..4], &[0xa5; 4]);
869            assert_eq!(&bytes[20..], &[0xa5; 4]);
870            if expected_result <= 0 {
871                assert_eq!(bytes, [0xa5; 24]);
872            }
873            EXPECTED_BUFFER.set(std::ptr::null_mut());
874        }
875    }
