9//! Plugin-private definitions of two glibc symbols newer than the oldest
10//! supported host's glibc (2.34), so the guest's dynamic loader never has to
11//! find them (<https://github.com/rrnewton/hermit/issues/3652>).
12//!
13//! SHARED FILE: detcore-liteinst/src/lib.rs compiles this same file into the
14//! in-guest LiteInst runtime with `#[path]`
15//! (<https://github.com/rrnewton/hermit/issues/3967>), so an edit here changes
16//! both guest preloads. Each preload compiles its own copy because the hidden
17//! definitions must sit in that cdylib's own object for its static link to
18//! bind them.
19//!
20//! Statically linked C code would otherwise import them from the guest's libc:
21//!
22//! - `__isoc23_strtol` (GLIBC_2.38): glibc 2.38+ headers rename `strtol` to it
23//!   when C is compiled as C23, GCC 15's default; mimalloc's options.c calls
24//!   `strtol`.
25//! - `_dl_find_object` (GLIBC_2.35): libgcc_eh.a (see each crate's
26//!   build.rs) looks up a program counter's unwind tables with it when it was
27//!   built against glibc 2.35+. The private definition forwards to the
28//!   guest's real `_dl_find_object` when its glibc has one (2.35 and later, or
29//!   a backport). Only without it does it search with `dl_iterate_phdr`, which
30//!   sees the caller's loader namespace only. The real lookup sees every
31//!   namespace, so a panic that unwinds through a frame of a library loaded
32//!   with `dlmopen(LM_ID_NEWLM)` is still caught; with `dl_iterate_phdr` alone
33//!   it aborts ("failed to initiate panic, error 5",
34//!   <https://github.com/rrnewton/hermit/issues/3980>).
35//!
36//!   The real function is found with Reverie's `reverie::glibc_symbol`,
37//!   which reads libc.so.6's symbol and version tables itself through
38//!   `dl_iterate_phdr` and calls no `dl*` function: `dlsym`/`dlvsym` take the
39//!   loader's `dl_load_lock`, which `dlopen` holds while it runs
40//!   constructors, and change `dlerror` state. An `.init_array` entry looks it
41//!   up while the preload is initialized. A lookup before that (a panic in an
42//!   earlier constructor) does it then. After the first lookup the unwinder
43//!   calls glibc's own `_dl_find_object`: lock-free and async-signal-safe, it
44//!   sees every namespace. The lookup itself takes only `dl_iterate_phdr`'s
45//!   recursive `dl_load_write_lock`, under which glibc runs no constructor or
46//!   destructor; its remaining exposure, before initialization, is the one
47//!   libgcc's unwinder had on every unwind before glibc 2.35 (see that
48//!   module). Reverie's reverie-liteinst/src/glibc_compat.rs does the same for
49//!   Reverie's own preload.
50//!
51//! The assembly below defines both names with hidden visibility. The static
52//! link binds those references here, and the names stay out of the plugin's
53//! dynamic symbol table, so the guest's own references still bind to its libc.
54

73/// C23 `strtol`. It differs from C17 `strtol` only in accepting a `0b` or `0B`
74/// prefix before binary digits when `base` is 0 or 2; C17 parses that as the
75/// number 0 followed by text.
76unsafe extern "C" fn isoc23_strtol(
77    nptr: *const c_char,
78    endptr: *mut *mut c_char,
79    base: c_int,
80) -> c_long {
81    if base == 0 || base == 2 {
82        // SAFETY: the caller passes a NUL-terminated string, as strtol requires.
83        if let Some(value) = unsafe { binary_prefixed(nptr, endptr) } {
84            return value;
85        }
86    }
87    // SAFETY: the caller's arguments are exactly strtol's.
88    unsafe { libc::strtol(nptr, endptr, base) }
89}

91/// Parses `[space][sign]0b<binary digits>` as C23 does, or returns `None` if
92/// `nptr` does not start that way.
93///
94/// # Safety
95///
96/// `nptr` must be NUL-terminated, and `endptr` null or writable.
97unsafe fn binary_prefixed(nptr: *const c_char, endptr: *mut *mut c_char) -> Option<c_long> {
98    let byte = |offset: usize| unsafe { *nptr.add(offset) as u8 };
99    let mut at = 0;
100    while unsafe { libc::isspace(c_int::from(byte(at))) } != 0 {
101        at += 1;
102    }
103    let negative = byte(at) == b'-';
104    if negative || byte(at) == b'+' {
105        at += 1;
106    }
107    if byte(at) != b'0'
108        || !matches!(byte(at + 1), b'b' | b'B')
109        || !matches!(byte(at + 2), b'0' | b'1')
110    {
111        return None;
112    }
113    at += 2;
114    let mut magnitude: u64 = 0;
115    let mut overflow = false;
116    while let digit @ (b'0' | b'1') = byte(at) {
117        match magnitude.checked_mul(2) {
118            Some(doubled) => magnitude = doubled | u64::from(digit - b'0'),
119            None => overflow = true,
120        }
121        at += 1;
122    }
123    if !endptr.is_null() {
124        // SAFETY: the caller allows writing `endptr`; strtol stores a pointer
125        // into its (const) input the same way.
126        unsafe { *endptr = nptr.add(at).cast_mut() };
127    }
128    let value = if negative {
129        if overflow || magnitude > c_long::MIN.unsigned_abs() {
130            None
131        } else {
132            Some(0_i64.wrapping_sub_unsigned(magnitude))
133        }
134    } else {
135        c_long::try_from(magnitude).ok().filter(|_| !overflow)
136    };
137    Some(value.unwrap_or_else(|| {
138        // SAFETY: errno is this thread's.
139        unsafe { *libc::__errno_location() = libc::ERANGE };
140        if negative { c_long::MIN } else { c_long::MAX }
141    }))
142}

144/// glibc's `struct dl_find_object` on x86_64.
145#[repr(C)]
146struct DlFindObject {
147    dlfo_flags: u64,
148    dlfo_map_start: *mut c_void,
149    dlfo_map_end: *mut c_void,
150    dlfo_link_map: *mut c_void,
151    dlfo_eh_frame: *mut c_void,
152    dlfo_reserved: [u64; 7],
153}

155/// The loaded object containing a program counter, as `_dl_find_object` reports it.
156#[derive(Debug, PartialEq, Eq)]
157struct LoadedObject {
158    map_start: usize,
159    map_end: usize,
160    eh_frame: usize,
161}
162
163type DlFindObjectFn = unsafe extern "C" fn(*mut c_void, *mut DlFindObject) -> c_int;
164
165/// The guest's real `_dl_find_object`, as an address: [`UNRESOLVED`] until
166/// [`resolve_real_dl_find_object`] has run, [`ABSENT`] when its libc has none.
167static REAL_DL_FIND_OBJECT: AtomicUsize = AtomicUsize::new(UNRESOLVED);
168const UNRESOLVED: usize = 0;
169const ABSENT: usize = 1;
170
171#[used]
172#[unsafe(link_section = ".init_array")]
173static RESOLVE_REAL_DL_FIND_OBJECT: extern "C" fn() = resolve_real_dl_find_object;
174
175/// Finds the guest's real `_dl_find_object@GLIBC_2.35` and caches it. It
176/// runs as an `.init_array` entry, and on a lookup before that entry has run.
177/// Two threads racing to run it store the same answer.
178extern "C" fn resolve_real_dl_find_object() {
179    let address =
180        reverie::glibc_symbol::glibc_versioned_function(b"_dl_find_object", b"GLIBC_2.35")
181            .unwrap_or(ABSENT);
182    REAL_DL_FIND_OBJECT.store(address, Ordering::Release);
183}
184
185/// The guest's real `_dl_find_object`, if its libc has one.
186fn real_dl_find_object() -> Option<DlFindObjectFn> {
187    if REAL_DL_FIND_OBJECT.load(Ordering::Acquire) == UNRESOLVED {
188        resolve_real_dl_find_object();
189    }
190    match REAL_DL_FIND_OBJECT.load(Ordering::Acquire) {
191        UNRESOLVED | ABSENT => None,
192        // SAFETY: any other value is glibc's `_dl_find_object`, whose
193        // signature this is.
194        address => Some(unsafe { std::mem::transmute::<usize, DlFindObjectFn>(address) }),
195    }
196}

198/// The guest's real `_dl_find_object` when it has one; otherwise a lookup over
199/// `dl_iterate_phdr`, the one libgcc used before glibc 2.35. Either returns 0
200/// and fills `result` if a loaded object maps `pc`, else -1. The fallback
201/// leaves `dlfo_link_map` null: libgcc reads only `dlfo_eh_frame`.
202unsafe extern "C" fn dl_find_object(pc: *mut c_void, result: *mut DlFindObject) -> c_int {
203    if let Some(real) = real_dl_find_object() {
204        // SAFETY: the caller's arguments are exactly `_dl_find_object`'s.
205        return unsafe { real(pc, result) };
206    }
207    // SAFETY: as above.
208    unsafe { phdr_find_object(pc, result) }
209}
210
211/// The `dl_iterate_phdr` fallback of [`dl_find_object`].
212///
213/// # Safety
214///
215/// `result` must be writable storage for one `struct dl_find_object`.
216unsafe fn phdr_find_object(pc: *mut c_void, result: *mut DlFindObject) -> c_int {
217    let Some(object) = loaded_object_containing(pc as usize) else {
218        return -1;
219    };
220    // SAFETY: the caller passes a writable `struct dl_find_object`.
221    unsafe {
222        result.write(DlFindObject {
223            dlfo_flags: 0,
224            dlfo_map_start: object.map_start as *mut c_void,
225            dlfo_map_end: object.map_end as *mut c_void,
226            dlfo_link_map: std::ptr::null_mut(),
227            dlfo_eh_frame: object.eh_frame as *mut c_void,
228            dlfo_reserved: [0; 7],
229        })
230    };
231    0
232}

234fn loaded_object_containing(pc: usize) -> Option<LoadedObject> {
235    struct Search {
236        pc: usize,
237        found: Option<LoadedObject>,
238    }
239
240    unsafe extern "C" fn visit(
241        info: *mut libc::dl_phdr_info,
242        _size: libc::size_t,
243        data: *mut c_void,
244    ) -> c_int {
245        // SAFETY: `data` is the `Search` passed below, and glibc passes a valid
246        // `info` whose program headers stay mapped during the callback.
247        let (search, info) = unsafe { (&mut *data.cast::<Search>(), &*info) };
248        let headers = unsafe { std::slice::from_raw_parts(info.dlpi_phdr, info.dlpi_phnum.into()) };
249        let base = info.dlpi_addr as usize;
250        let mut object = LoadedObject {
251            map_start: usize::MAX,
252            map_end: 0,
253            eh_frame: 0,
254        };
255        let mut contains_pc = false;
256        for header in headers {
257            let start = base.wrapping_add(header.p_vaddr as usize);
258            match header.p_type {
259                libc::PT_LOAD => {
260                    let end = start.wrapping_add(header.p_memsz as usize);
261                    object.map_start = object.map_start.min(start);
262                    object.map_end = object.map_end.max(end);
263                    contains_pc |= (start..end).contains(&search.pc);
264                }
265                libc::PT_GNU_EH_FRAME => object.eh_frame = start,
266                _ => {}
267            }
268        }
269        if !contains_pc {
270            return 0;
271        }
272        search.found = Some(object);
273        1
274    }
275
276    let mut search = Search { pc, found: None };
277    // SAFETY: `visit` matches dl_iterate_phdr's callback contract and `search`
278    // outlives the call.
279    unsafe { libc::dl_iterate_phdr(Some(visit), (&raw mut search).cast()) };
280    search.found
281}

297    /// A Rust panic unwinds through a C frame of a library loaded into
298    /// NAMESPACE (`new`: `dlmopen(LM_ID_NEWLM)`; `same`: `dlopen`) and is
299    /// caught by `catch_unwind` beneath it
300    /// (https://github.com/rrnewton/hermit/issues/3980). This test binary
301    /// links the same static unwinder and `_dl_find_object` as the preloads,
302    /// so the unwinder must find the bridge's unwind tables in either
303    /// namespace. The scenario runs in a child, because a failed unwind aborts
304    /// the process.
305    fn panic_through_bridge(namespace: &str, test: &str) {
306        if std::env::var(NAMESPACE_CHILD).as_deref() == Ok(namespace) {
307            panic_through_bridge_in_this_process(namespace);
308            return;
309        }
310        let directory = std::env::temp_dir().join(format!(
311            "hermit-glibc-compat-unwind-{}-{namespace}",
312            std::process::id()
313        ));
314        std::fs::create_dir_all(&directory).unwrap();
315        let source = directory.join("bridge.c");
316        let bridge = directory.join("bridge.so");
317        std::fs::write(
318            &source,
319            "void bridge_call(void (*callback)(void)) { callback(); }\n",
320        )
321        .unwrap();
322        let built = Command::new("cc")
323            .args(["-shared", "-fPIC", "-fexceptions", "-O0", "-o"])
324            .arg(&bridge)
325            .arg(&source)
326            .status()
327            .expect("the namespace unwind tests need a C compiler (cc) on PATH");
328        assert!(built.success(), "cc failed: {built}");
329        let output = Command::new(std::env::current_exe().unwrap())
330            .args(["--exact", test, "--nocapture", "--test-threads=1"])
331            .env(NAMESPACE_CHILD, namespace)
332            .env(BRIDGE_PATH, &bridge)
333            .output()
334            .unwrap();
335        std::fs::remove_dir_all(&directory).unwrap();
336        let stdout = String::from_utf8_lossy(&output.stdout);
337        let stderr = String::from_utf8_lossy(&output.stderr);
338        assert!(
339            output.status.success() && stdout.contains(&format!("namespace={namespace} caught=1")),
340            "the child exited with {}\nstdout:\n{stdout}\nstderr:\n{stderr}",
341            output.status
342        );

351    /// Loads the bridge at `path` into NAMESPACE and reports whether a panic
352    /// through it was caught.
353    fn catch_a_panic_through_the_bridge(namespace: &str, path: &CStr) -> bool {
354        // SAFETY: `path` names a shared object built by the parent.
355        let handle = unsafe {
356            match namespace {
357                "new" => libc::dlmopen(libc::LM_ID_NEWLM, path.as_ptr(), libc::RTLD_NOW),
358                _ => libc::dlopen(path.as_ptr(), libc::RTLD_NOW),
359            }
360        };
361        assert!(
362            !handle.is_null(),
363            "cannot load the bridge into namespace {namespace}"
364        );
365        // SAFETY: `handle` is a loaded library.
366        let symbol = unsafe { libc::dlsym(handle, c"bridge_call".as_ptr()) };
367        assert!(!symbol.is_null());
368        // SAFETY: bridge.c defines bridge_call with this signature, and its
369        // frame has unwind tables (-fexceptions).
370        let bridge_call = unsafe {
371            std::mem::transmute::<*mut c_void, unsafe extern "C-unwind" fn(extern "C-unwind" fn())>(
372                symbol,
373            )
374        };
375        extern "C-unwind" fn callback() {
376            panic!("unwinding through the bridge");
377        }
378        // SAFETY: as above.
379        std::panic::catch_unwind(|| unsafe { bridge_call(callback) }).is_err()
380    }

622        let mut end = std::ptr::null_mut();

624        unsafe { *libc::__errno_location() = 0 };
625        let value = unsafe { isoc23_strtol(text.as_ptr(), &mut end, base) };
626        let errno = unsafe { *libc::__errno_location() };
627        (value, end as usize - text.as_ptr() as usize, errno)

656    /// Set in the child process a cold-unwind test starts.
657    const COLD_CHILD: &str = "HERMIT_GLIBC_COMPAT_UNWIND_COLD_CHILD";
658    /// The constructor library: its constructor runs the function whose
659    /// address HERMIT_GLIBC_COMPAT_UNWIND_WORKER holds (set by the child itself)
660    /// on a new thread and waits for it, for at most one second, while
661    /// `dlopen` holds the loader lock. It records whether it timed out.
662    const CONSTRUCTOR_LIBRARY: &str = r#"
663#define _GNU_SOURCE
664#include <pthread.h>
665#include <stdlib.h>
666#include <time.h>
667int constructor_timed_out = -1;
668static void *worker(void *function) {
669    ((void (*)(void))function)();
670    return 0;
671}
672__attribute__((constructor)) static void wait_for_worker(void) {
673    const char *address = getenv("HERMIT_GLIBC_COMPAT_UNWIND_WORKER");
674    pthread_t thread;
675    struct timespec deadline;
676    if (!address) return;
677    pthread_create(&thread, 0, worker, (void *)strtoull(address, 0, 16));
678    clock_gettime(CLOCK_REALTIME, &deadline);
679    deadline.tv_sec += 1;
680    constructor_timed_out = pthread_timedjoin_np(thread, 0, &deadline) != 0;
681    if (constructor_timed_out) pthread_detach(thread);
682}
683"#;

685    /// Runs `test` again in a child with COLD_CHILD set to `case`, so its
686    /// panic is the first unwind in the process, and returns its stdout once
687    /// it exits successfully within ten seconds.
688    fn cold_child(case: &str, test: &str, extra: &[(&str, &std::ffi::OsStr)]) -> String {
689        let mut command = Command::new(std::env::current_exe().unwrap());
690        command
691            .args(["--exact", test, "--nocapture", "--test-threads=1"])
692            .env(COLD_CHILD, case)
693            .stdout(std::process::Stdio::piped())
694            .stderr(std::process::Stdio::piped());
695        for (name, value) in extra {
696            command.env(name, value);
697        }
698        let mut child = command.spawn().unwrap();
699        let deadline = std::time::Instant::now() + std::time::Duration::from_secs(10);
700        while child.try_wait().unwrap().is_none() {
701            if std::time::Instant::now() > deadline {
702                child.kill().unwrap();
703                break;
704            }
705            std::thread::sleep(std::time::Duration::from_millis(10));
706        }
707        let output = child.wait_with_output().unwrap();
708        let stdout = String::from_utf8_lossy(&output.stdout).into_owned();
709        assert!(
710            output.status.success(),
711            "the {case} child exited with {}\nstdout:\n{stdout}\nstderr:\n{}",
712            output.status,
713            String::from_utf8_lossy(&output.stderr)
714        );
715        stdout
716    }

720    extern "C" fn worker_catches_a_panic() {

725    /// A library constructor waits for a worker thread whose panic is the
726    /// process's first unwind. The unwind must not need the loader lock that
727    /// `dlopen` holds while the constructor waits, so the constructor must not
728    /// time out.
729    #[test]
730    fn a_constructor_waiting_for_a_workers_first_unwind_does_not_time_out() {
731        const TEST: &str = "glibc_compat::tests::a_constructor_waiting_for_a_workers_first_unwind_does_not_time_out";
732        if std::env::var(COLD_CHILD).as_deref() == Ok("constructor") {
733            // This process's own address of the worker (the parent's differs:
734            // the binary is position-independent).
735            let worker = CString::new(format!(
736                "{:x}",
737                worker_catches_a_panic as extern "C" fn() as usize
738            ))
739            .unwrap();
740            // SAFETY: both strings are NUL-terminated, and no other thread of
741            // this child reads the environment.
742            assert_eq!(
743                unsafe {
744                    libc::setenv(
745                        c"HERMIT_GLIBC_COMPAT_UNWIND_WORKER".as_ptr(),
746                        worker.as_ptr(),
747                        1,
748                    )
749                },
750                0
751            );
752            let path = CString::new(std::env::var(BRIDGE_PATH).unwrap()).unwrap();
753            // SAFETY: `path` names the constructor library the parent built.
754            let handle = unsafe { libc::dlopen(path.as_ptr(), libc::RTLD_NOW) };
755            assert!(!handle.is_null());
756            // SAFETY: `handle` is a loaded library defining this int.
757            let timed_out =
758                unsafe { *libc::dlsym(handle, c"constructor_timed_out".as_ptr()).cast::<c_int>() };
759            while !WORKER_CAUGHT.load(Ordering::SeqCst) {
760                std::thread::sleep(std::time::Duration::from_millis(1));
761            }
762            println!("constructor_timed_out={timed_out} worker_caught=1");
763            return;
764        }
765        let directory = std::env::temp_dir().join(format!(
766            "hermit-glibc-compat-constructor-{}",
767            std::process::id()
768        ));
769        std::fs::create_dir_all(&directory).unwrap();
770        let source = directory.join("constructor.c");
771        let library = directory.join("constructor.so");
772        std::fs::write(&source, CONSTRUCTOR_LIBRARY).unwrap();
773        let built = Command::new("cc")
774            .args(["-shared", "-fPIC", "-O0", "-pthread", "-o"])
775            .arg(&library)
776            .arg(&source)
777            .status()
778            .expect("the cold-unwind tests need a C compiler (cc) on PATH");
779        assert!(built.success(), "cc failed: {built}");
780        let stdout = cold_child("constructor", TEST, &[(BRIDGE_PATH, library.as_os_str())]);
781        std::fs::remove_dir_all(&directory).unwrap();
782        assert!(
783            stdout.contains("constructor_timed_out=0 worker_caught=1"),
784            "{stdout}"
785        );
786    }
