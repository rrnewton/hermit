9//! Host-side per-physical-run filesystem and network isolation.
10//!
11//! Callers must create their tracer runtime inside the callback and finish its
12//! cleanup before returning. A pre-existing executor could spawn the guest in
13//! its original namespace. This helper does not change the calling thread's
14//! namespace, cwd, or filesystem sharing. It leaves /tmp transport visible
15//! unless the run binds paths into a private /tmp ([`Isolation::binds`]); then
16//! only the paths in [`Isolation::preserve`] stay reachable there.
17

31/// The pinned-root runner requests a fresh /test for each physical run.
32pub const REQUEST_ENV: &str = "HERMIT_E2E_EMPTY_WORKDIR";
33/// The pre-existing mountpoint supplied by the pinned-root container.
34pub const WORKDIR: &str = "/test";
35/// The directory a run with binds gets as a fresh tmpfs.
36pub const TMP_DIR: &str = "/tmp";
37
38/// One host path bound into the run's private /tmp, as `hermit run --bind`
39/// binds it for the other backends: `source` must exist, and `target` is an
40/// absolute path below /tmp, created as a file or a directory to match.
41#[derive(Clone, Debug, Eq, PartialEq)]
42pub struct BindMount {
43    pub source: PathBuf,
44    pub target: PathBuf,
45}
46
47/// The filesystem and network isolation one physical run gets.
48#[derive(Clone, Debug, Default, Eq, PartialEq)]
49pub struct Isolation {
50    /// Create a fresh network namespace with loopback up and a fresh /sys.
51    /// False leaves the caller's network namespace in place.
52    pub local_networking: bool,
53    /// Mount a fresh tmpfs on the pre-existing [`WORKDIR`].
54    pub test_workdir: bool,
55    /// Mount a fresh tmpfs on /tmp and bind each source at its target there,
56    /// in order. Empty leaves /tmp as it is.
57    pub binds: Vec<BindMount>,
58    /// Paths below the original /tmp, made before the run, that the run must
59    /// still reach at the same path once /tmp is replaced: bound at that path
60    /// before the binds. Ignored when there are no binds.
61    pub preserve: Vec<PathBuf>,
62}
63
64impl Isolation {
65    /// Whether the run needs neither filesystem nor network isolation.
66    pub fn is_empty(&self) -> bool {
67        !self.local_networking && !self.test_workdir && self.binds.is_empty()
68    }
69}

71/// Refuse malformed requests instead of silently running without isolation.
72pub fn requested_workdir(value: Option<&OsStr>) -> io::Result<Option<&'static Path>> {
73    match value {
74        None => Ok(None),
75        Some(value) if value == OsStr::new(WORKDIR) => Ok(Some(Path::new(WORKDIR))),
76        Some(value) => Err(io::Error::new(
77            io::ErrorKind::InvalidInput,
78            format!("{REQUEST_ENV} must be {WORKDIR}, got {value:?}"),
79        )),
80    }
81}

83/// Run on a new host thread with a fresh tmpfs mounted at /test.
84///
85/// This requires CAP_SYS_ADMIN in the owning user namespace (the pinned root
86/// grants it; [`enter_root_user_namespace`] gives it to an unprivileged
87/// process) and an existing /test directory. No user-namespace fallback or
88/// shared-directory substitute is used here. Setup errors return before the callback can launch a guest. The
89/// scoped thread is joined on success, error and panic; a callback panic keeps
90/// its original payload. Each invocation creates its own mount namespace.
91pub fn with_isolated_workdir<F, T>(run: F) -> io::Result<T>
92where
93    F: FnOnce() -> T + Send,
94    T: Send,
95{
96    with_isolation(
97        &Isolation {
98            test_workdir: true,
99            ..Isolation::default()
100        },
101        run,
102    )
103}

105/// Run on a new host thread in a new mount namespace with the requested mounts
106/// and, for local networking, a new network namespace with only loopback.
107///
108/// The same contract as [`with_isolated_workdir`]: CAP_SYS_ADMIN is required,
109/// every setup error returns before the callback can launch a guest, and the
110/// mounts disappear with the namespace. Local networking also needs
111/// CAP_NET_ADMIN to bring loopback up; failure never falls back to host networking.
112/// A bind target must be a normal absolute path strictly below /tmp; a bind
113/// source and a preserved path are resolved before /tmp is replaced, so they
114/// may lie below the original /tmp.
115pub fn with_isolation<F, T>(isolation: &Isolation, run: F) -> io::Result<T>
116where
117    F: FnOnce() -> T + Send,
118    T: Send,
119{
120    for bind in &isolation.binds {
121        check_bind_target(&bind.target)?;
122    }
123    std::thread::scope(|scope| {
124        let thread = std::thread::Builder::new()
125            .name("hermit-test-workdir".into())
126            .spawn_scoped(scope, move || {
127                enter_namespace(isolation)?;
128                Ok(run())
129            })?;
130        match thread.join() {
131            Ok(result) => result,
132            Err(panic) => std::panic::resume_unwind(panic),
133        }
134    })
135}

137/// Whether the calling process holds CAP_SYS_ADMIN (in its effective set), which
138/// [`with_isolation`]'s mount namespace needs.
139pub fn has_cap_sys_admin() -> io::Result<bool> {
140    const CAP_SYS_ADMIN: u32 = 21;
141    let status = std::fs::read_to_string("/proc/self/status")?;
142    let effective = status
143        .lines()
144        .find_map(|line| line.strip_prefix("CapEff:"))
145        .and_then(|hex| u64::from_str_radix(hex.trim(), 16).ok())
146        .ok_or_else(|| io::Error::other("/proc/self/status has no readable CapEff line"))?;
147    Ok(effective & (1 << CAP_SYS_ADMIN) != 0)
148}

150/// Make the calling process root in a new user namespace that maps root to the
151/// caller's own user and group, exactly as `reverie::process::Container::map_root`
152/// maps the guest of every other backend. The process then holds CAP_SYS_ADMIN
153/// over the mount namespaces it creates, so [`with_isolation`] works on a host
154/// that grants no privilege, and a guest sees the same identity as under the
155/// other backends.
156///
157/// Linux refuses to move a multithreaded process into a new user namespace, so
158/// this must run before the process starts its first thread; it fails, with
159/// the thread count, instead of leaving the process where it was.
160pub fn enter_root_user_namespace() -> io::Result<()> {
161    let threads = std::fs::read_dir("/proc/self/task")?.count();
162    if threads != 1 {
163        return Err(io::Error::other(format!(
164            "a new user namespace needs a single-threaded process; this one has {threads} threads"
165        )));
166    }
167    let uid = unsafe { libc::geteuid() };
168    let gid = unsafe { libc::getegid() };
169    syscall_result(
170        unsafe { libc::unshare(libc::CLONE_NEWUSER) },
171        "unshare user namespace",
172    )?;
173    let write = |path: &str, contents: String| {
174        std::fs::write(path, contents)
175            .map_err(|error| io::Error::new(error.kind(), format!("write {path}: {error}")))
176    };
177    write("/proc/self/uid_map", format!("0 {uid} 1"))?;
178    write("/proc/self/setgroups", "deny".to_string())?;
179    write("/proc/self/gid_map", format!("0 {gid} 1"))
180}

182fn syscall_result(result: libc::c_int, operation: &str) -> io::Result<()> {
183    if result == 0 {
184        Ok(())
185    } else {
186        let error = io::Error::last_os_error();
187        Err(io::Error::new(
188            error.kind(),
189            format!("{operation}: {error}"),
190        ))
191    }
192}

194/// Refuse a bind target the private /tmp cannot hold: relative, outside /tmp,
195/// /tmp itself, or with `.`/`..` components that could leave it.
196pub fn check_bind_target(target: &Path) -> io::Result<()> {
197    let mut components = target.components();
198    let below_tmp = components.next() == Some(Component::RootDir)
199        && components.next() == Some(Component::Normal(OsStr::new("tmp")))
200        && target.components().count() > 2
201        && components.all(|component| matches!(component, Component::Normal(_)));
202    if below_tmp {
203        Ok(())
204    } else {
205        Err(io::Error::new(
206            io::ErrorKind::InvalidInput,
207            format!(
208                "bind target {} must be a normal absolute path below {TMP_DIR}",
209                target.display()
210            ),
211        ))
212    }
213}

215fn path_cstring(path: &Path) -> io::Result<CString> {
216    CString::new(path.as_os_str().as_bytes()).map_err(|_| {
217        io::Error::new(
218            io::ErrorKind::InvalidInput,
219            format!("path {} contains a NUL byte", path.display()),
220        )
221    })
222}

224/// A path opened before /tmp is replaced: `O_PATH` keeps the original file
225/// reachable through `/proc/thread-self/fd` after its path is covered.
226struct OpenedSource {
227    file: File,
228    is_dir: bool,
229    target: PathBuf,
230}
231
232fn open_source(source: &Path, target: &Path) -> io::Result<OpenedSource> {
233    let file = std::fs::OpenOptions::new()
234        .read(true)
235        .custom_flags(libc::O_PATH)
236        .open(source)
237        .map_err(|error| {
238            io::Error::new(
239                error.kind(),
240                format!("open bind source {}: {error}", source.display()),
241            )
242        })?;
243    let is_dir = file.metadata()?.is_dir();
244    Ok(OpenedSource {
245        file,
246        is_dir,
247        target: target.to_path_buf(),
248    })
249}

251fn mount_tmpfs(target: &std::ffi::CStr, what: &str) -> io::Result<()> {
252    syscall_result(
253        unsafe {
254            libc::mount(
255                c"tmpfs".as_ptr(),
256                target.as_ptr(),
257                c"tmpfs".as_ptr(),
258                libc::MS_NOSUID | libc::MS_NODEV,
259                c"mode=1777".as_ptr().cast(),
260            )
261        },
262        what,
263    )
264}
265
266fn bind_opened(source: &OpenedSource) -> io::Result<()> {
267    let target = &source.target;
268    if let Some(parent) = target.parent() {
269        std::fs::create_dir_all(parent)?;
270    }
271    let created = if source.is_dir {
272        std::fs::create_dir(target)
273    } else {
274        std::fs::OpenOptions::new()
275            .write(true)
276            .create_new(true)
277            .open(target)
278            .map(drop)
279    };
280    match created {
281        Err(error) if error.kind() != io::ErrorKind::AlreadyExists => {
282            return Err(io::Error::new(
283                error.kind(),
284                format!("create bind target {}: {error}", target.display()),
285            ));
286        }
287        _ => {}
288    }
289    let from = CString::new(format!("/proc/thread-self/fd/{}", source.file.as_raw_fd()))
290        .expect("a descriptor path has no NUL");
291    let to = path_cstring(target)?;
292    syscall_result(
293        unsafe {
294            libc::mount(
295                from.as_ptr(),
296                to.as_ptr(),
297                std::ptr::null(),
298                libc::MS_BIND | libc::MS_REC,
299                std::ptr::null(),
300            )
301        },
302        &format!("bind {}", target.display()),
303    )
304}

306/// Open the setup socket only after unsharing, so its ioctls address the new
307/// network namespace. CLOEXEC and OwnedFd keep it out of the launched guest.
308fn bring_loopback_up() -> io::Result<()> {
309    let socket = unsafe {
310        libc::socket(
311            libc::AF_INET,
312            libc::SOCK_DGRAM | libc::SOCK_CLOEXEC,
313            libc::IPPROTO_IP,
314        )
315    };
316    if socket < 0 {
317        let error = io::Error::last_os_error();
318        return Err(io::Error::new(
319            error.kind(),
320            format!("open loopback setup socket: {error}"),
321        ));
322    }
323    // SAFETY: socket is a newly opened descriptor owned by this function.
324    let socket = unsafe { OwnedFd::from_raw_fd(socket) };
325    // SAFETY: an all-zero ifreq is valid; the short interface name stays NUL-terminated.
326    let mut interface: libc::ifreq = unsafe { std::mem::zeroed() };
327    interface.ifr_name[0] = b'l' as libc::c_char;
328    interface.ifr_name[1] = b'o' as libc::c_char;
329    syscall_result(
330        unsafe { libc::ioctl(socket.as_raw_fd(), libc::SIOCGIFFLAGS, &mut interface) },
331        "read loopback interface flags",
332    )?;
333    // SAFETY: SIOCGIFFLAGS initialized this union member on success.
334    interface.ifr_ifru.ifru_flags =
335        unsafe { interface.ifr_ifru.ifru_flags } | libc::IFF_UP as libc::c_short;
336    syscall_result(
337        unsafe { libc::ioctl(socket.as_raw_fd(), libc::SIOCSIFFLAGS, &interface) },
338        "bring loopback interface up",
339    )
340}

342fn enter_namespace(isolation: &Isolation) -> io::Result<()> {
343    // A new thread is essential: unshare also separates its fs_struct, which
344    // setns could not restore to the caller's original CLONE_FS sharing.
345    syscall_result(
346        unsafe { libc::unshare(libc::CLONE_NEWNS) },
347        "unshare mount namespace",
348    )?;
349    if isolation.local_networking {
350        syscall_result(
351            unsafe { libc::unshare(libc::CLONE_NEWNET) },
352            "unshare network namespace",
353        )?;
354    }
355    syscall_result(
356        unsafe {
357            libc::mount(
358                std::ptr::null(),
359                c"/".as_ptr(),
360                std::ptr::null(),
361                libc::MS_REC | libc::MS_PRIVATE,
362                std::ptr::null(),
363            )
364        },
365        "make mount propagation private",
366    )?;
367    if isolation.local_networking {
368        syscall_result(
369            unsafe {
370                libc::mount(
371                    std::ptr::null(),
372                    c"/sys".as_ptr(),
373                    c"sysfs".as_ptr(),
374                    0,
375                    std::ptr::null(),
376                )
377            },
378            "mount per-run network sysfs",
379        )?;
380        bring_loopback_up()?;
381    }
382    if isolation.test_workdir {
383        mount_tmpfs(c"/test", "mount per-run /test tmpfs")?;
384    }
385    if isolation.binds.is_empty() {
386        return Ok(());
387    }
388    // Everything is opened before the tmpfs covers /tmp, so a source or a
389    // preserved path below the original /tmp is still reachable.
390    let mut opened = Vec::new();
391    for path in &isolation.preserve {
392        if path.starts_with(TMP_DIR) {
393            opened.push(open_source(path, path)?);
394        }
395    }
396    for bind in &isolation.binds {
397        opened.push(open_source(&bind.source, &bind.target)?);
398    }
399    mount_tmpfs(c"/tmp", "mount per-run /tmp tmpfs")?;
400    opened.iter().try_for_each(bind_opened)
401}

407    #[test]
408    fn request_is_exact_and_fail_closed() {
409        assert_eq!(requested_workdir(None).unwrap(), None);
410        assert_eq!(
411            requested_workdir(Some(OsStr::new("/test"))).unwrap(),
412            Some(Path::new("/test"))
413        );
414        for value in [b"".as_slice(), b"/tmp", b"/test/", b"/test\0", b"/test\xff"] {
415            let error = requested_workdir(Some(OsStr::from_bytes(value))).unwrap_err();
416            assert_eq!(error.kind(), io::ErrorKind::InvalidInput);
417            assert!(
418                error
419                    .to_string()
420                    .contains("HERMIT_E2E_EMPTY_WORKDIR must be /test")
421            );
422        }
423    }

425    #[test]
426    fn bind_targets_must_stay_strictly_below_tmp() {
427        // `Path::components` drops an interior `.`, so `/tmp/./test` is `/tmp/test`.
428        for target in ["/tmp/test", "/tmp/e2e/home", "/tmp/a/b/c", "/tmp/./test"] {
429            check_bind_target(Path::new(target)).unwrap();
430        }
431        for target in [
432            "",
433            "tmp/test",
434            "/tmp",
435            "/tmp/",
436            "/test",
437            "/tmpx/test",
438            "/tmp/../etc",
439            "/tmp/a/../../etc",
440            "/var/tmp/test",
441        ] {
442            let error = check_bind_target(Path::new(target)).unwrap_err();
443            assert_eq!(error.kind(), io::ErrorKind::InvalidInput, "{target}");
444            assert!(
445                error.to_string().contains("below /tmp"),
446                "{target}: {error}"
447            );
448        }
449    }
