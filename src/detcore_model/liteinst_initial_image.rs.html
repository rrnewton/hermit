7//! Bounded InitialImage launch packet shared by the host and private runtime.
8//!
9//! Packet bytes, nonce and file seals are not authorization. The coordinator
10//! separately redeems a one-use launch instance against its actual process watch.
11//! Decoding validates the complete byte shape without allocating. Only the host
12//! encodes packets; the private initializer consumes borrowed fields.
13
14use std::fmt;
15
16use crate::config::Config;
17
18/// Maximum complete packet size, including its fixed header.
19pub const MAX_PACKET_LEN: usize = 4096;
20/// Magic naming this explicit InitialImage protocol.
21pub const MAGIC: [u8; 8] = *b"HLIINIT1";
22/// Only this wire version is accepted.
23pub const VERSION: u16 = 1;
24/// Bytes before the three variable length fields.
25pub const HEADER_LEN: usize = 98;
26
27/// Borrowed fields of one complete, bounded launch packet.
28#[derive(Clone, Copy, Debug, Eq, PartialEq)]
29pub struct Packet<'a> {
30    /// Coordinator-held one-use challenge to redeem; not authority by itself.
31    pub nonce: &'a [u8; 32],
32    /// Independently supplied SHA-256 of the exact wire Config bytes.
33    pub config_digest: &'a [u8; 32],
34    /// Sixteen ASCII hex characters naming the actual Config/RPC schema.
35    pub schema_fingerprint: &'a [u8; 16],
36    /// Nonempty raw Unix coordinator path bytes (UTF-8 is not required).
37    pub coordinator: &'a [u8],
38    /// Nonempty UTF-8 inherited DETLOG fd and physical socket identity.
39    pub detlog_identity: &'a str,
40    /// Nonempty UTF-8 encoded genuine DETLOG forwarding policy.
41    pub forward_policy: &'a str,
42}
43
44/// A packet shape refusal, with no heap-backed error payload.
45#[derive(Clone, Copy, Debug, Eq, PartialEq)]
46pub enum PacketError {
47    /// Complete size is outside the bound or differs from the advertised size.
48    Size,
49    /// Magic does not name this protocol.
50    Magic,
51    /// Unsupported version.
52    Version,
53    /// Field lengths do not consume the complete packet exactly.
54    Length,
55    /// A mandatory text/path field is empty.
56    Empty,
57    /// A text/path field contains a NUL byte.
58    Nul,
59    /// DETLOG identity or policy is not UTF-8.
60    Utf8,
61    /// Schema fingerprint is not exactly sixteen ASCII hex characters.
62    Fingerprint,
63}
64impl fmt::Display for PacketError {
65    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
66        f.write_str(match self {
67            Self::Size => "InitialImage packet size mismatch",
68            Self::Magic => "InitialImage packet magic mismatch",
69            Self::Version => "InitialImage packet version mismatch",
70            Self::Length => "InitialImage packet field length mismatch",
71            Self::Empty => "InitialImage packet mandatory field is empty",
72            Self::Nul => "InitialImage packet field contains NUL",
73            Self::Utf8 => "InitialImage packet DETLOG field is not UTF-8",
74            Self::Fingerprint => "InitialImage packet schema fingerprint is not ASCII hex",
75        })
76    }
77}
78impl std::error::Error for PacketError {}
79
80/// Decode and validate all fields and exact full consumption before allocation.
81pub fn decode(bytes: &[u8]) -> Result<Packet<'_>, PacketError> {
82    if !(HEADER_LEN..=MAX_PACKET_LEN).contains(&bytes.len()) {
83        return Err(PacketError::Size);
84    }
85    if bytes[..8] != MAGIC {
86        return Err(PacketError::Magic);
87    }
88    if u16::from_be_bytes([bytes[8], bytes[9]]) != VERSION {
89        return Err(PacketError::Version);
90    }
91    if usize::from(u16::from_be_bytes([bytes[10], bytes[11]])) != bytes.len() {
92        return Err(PacketError::Size);
93    }
94    let lengths = [92, 94, 96].map(|i| usize::from(u16::from_be_bytes([bytes[i], bytes[i + 1]])));
95    let path_end = HEADER_LEN + lengths[0];
96    let identity_end = path_end + lengths[1];
97    let policy_end = identity_end + lengths[2];
98    if policy_end != bytes.len() {
99        return Err(PacketError::Length);
100    }
101    // Positive lengths and the exact end prove all three bounds before slicing.
102    let coordinator = &bytes[HEADER_LEN..path_end];
103    let identity = &bytes[path_end..identity_end];
104    let policy = &bytes[identity_end..policy_end];
105    check_text(coordinator)?;
106    check_text(identity)?;
107    check_text(policy)?;
108    let packet = Packet {
109        nonce: bytes[12..44].try_into().map_err(|_| PacketError::Length)?,
110        config_digest: bytes[44..76].try_into().map_err(|_| PacketError::Length)?,
111        schema_fingerprint: bytes[76..92]
112            .try_into()
113            .map_err(|_| PacketError::Fingerprint)?,
114        coordinator,
115        detlog_identity: std::str::from_utf8(identity).map_err(|_| PacketError::Utf8)?,
116        forward_policy: std::str::from_utf8(policy).map_err(|_| PacketError::Utf8)?,
117    };
118    check_fingerprint(packet.schema_fingerprint)?;
119    Ok(packet)
120}

122/// Host-side encoding with the same shape checks as the borrowed decoder.
123pub fn encode(packet: Packet<'_>) -> Result<Vec<u8>, PacketError> {
124    let fields = [
125        packet.coordinator,
126        packet.detlog_identity.as_bytes(),
127        packet.forward_policy.as_bytes(),
128    ];
129    for field in fields {
130        if field.len() > MAX_PACKET_LEN {
131            return Err(PacketError::Size);
132        }
133        check_text(field)?;
134    }
135    check_fingerprint(packet.schema_fingerprint)?;
136    let total = HEADER_LEN
137        .checked_add(fields.iter().map(|field| field.len()).sum::<usize>())
138        .filter(|total| *total <= MAX_PACKET_LEN)
139        .ok_or(PacketError::Size)?;
140    let mut bytes = Vec::with_capacity(total);
141    bytes.extend_from_slice(&MAGIC);
142    bytes.extend_from_slice(&VERSION.to_be_bytes());
143    bytes.extend_from_slice(&(total as u16).to_be_bytes());
144    bytes.extend_from_slice(packet.nonce);
145    bytes.extend_from_slice(packet.config_digest);
146    bytes.extend_from_slice(packet.schema_fingerprint);
147    for field in fields {
148        bytes.extend_from_slice(&(field.len() as u16).to_be_bytes());
149    }
150    for field in fields {
151        bytes.extend_from_slice(field);
152    }
153    Ok(bytes)
154}
155
156fn check_text(bytes: &[u8]) -> Result<(), PacketError> {
157    if bytes.is_empty() {
158        return Err(PacketError::Empty);
159    }
160    if bytes.contains(&0) {
161        return Err(PacketError::Nul);
162    }
163    Ok(())
164}
165fn check_fingerprint(bytes: &[u8; 16]) -> Result<(), PacketError> {
166    if bytes.iter().all(u8::is_ascii_hexdigit) {
167        Ok(())
168    } else {
169        Err(PacketError::Fingerprint)
170    }
171}

173/// Refuse unsupported InitialImage configuration without changing it.
174///
175/// This first slice excludes PMU/RCB time, maximum/target timeslices, imprecise
176/// timers, chaos and slowdown, preemption recording/replay and stack traces,
177/// schedule replay, and record/replay execution modes. `record_host_inputs`
178/// remains supported for genuine strict verification. No startup-accounting,
179/// successor exec, fork or thread capability is implied by this predicate.
180pub fn config_violation(config: &Config) -> Option<&'static str> {
181    if config.use_rcb_time() || config.max_timeslice.is_some() || config.target_timeslice.is_some()
182    {
183        return Some("InitialImage does not support RCB time or preemption timeslices");
184    }
185    if config.imprecise_timers
186        || config.chaos
187        || config.chaos_target_races
188        || config.chaos_per_thread_slowdown
189    {
190        return Some("InitialImage does not support chaos or imprecise timers");
191    }
192    if config.record_preemptions
193        || config.record_preemptions_to.is_some()
194        || config.replay_preemptions_from.is_some()
195        || config.replay_schedule_from.is_some()
196        || config.preemption_stacktrace
197        || config.preemption_stacktrace_log_file.is_some()
198        || !config.stacktrace_event.is_empty()
199        || config.stacktrace_signal.is_some()
200    {
201        return Some(
202            "InitialImage does not support preemption/schedule recording, replay or stack traces",
203        );
204    }
205    if config.recordreplay_modes || config.replaying || config.replay_data.is_some() {
206        return Some("InitialImage does not support record/replay execution");
207    }
208    None
209}

226    #[test]
227    fn exact_wire_shape_and_borrowed_roundtrip_accept_raw_path() {
228        let packet = packet();
229        let encoded = encode(packet).unwrap();
230        assert_eq!(
231            &encoded[..12],
232            &[b'H', b'L', b'I', b'I', b'N', b'I', b'T', b'1', 0, 1, 0, 134]
233        );
234        assert_eq!(HEADER_LEN, 98);
235        let decoded = decode(&encoded).unwrap();
236        assert_eq!(decoded, packet);
237        assert_eq!(decoded.coordinator.as_ptr(), encoded[HEADER_LEN..].as_ptr());
238        assert_eq!(decoded.nonce.as_ptr(), encoded[12..].as_ptr());
239    }

241    #[test]
242    fn malformed_shapes_and_every_truncated_prefix_refuse() {
243        let valid = encode(packet()).unwrap();
244        for length in 0..valid.len() {
245            assert!(decode(&valid[..length]).is_err(), "prefix {length}");
246        }
247        for (index, byte, expected) in [
248            (0, 0, PacketError::Magic),
249            (9, 2, PacketError::Version),
250            (11, 0, PacketError::Size),
251            (76, b'g', PacketError::Fingerprint),
252            (93, 0, PacketError::Length),
253            (98, 0, PacketError::Nul),
254        ] {
255            let mut bytes = valid.clone();
256            bytes[index] = byte;
257            assert_eq!(decode(&bytes), Err(expected), "offset {index}");
258        }
259        for field in 0..3 {
260            let mut bytes = valid.clone();
261            let index = 92 + field * 2;
262            let length = u16::from_be_bytes([bytes[index], bytes[index + 1]]) as usize;
263            let start = 98
264                + [
265                    packet().coordinator.len(),
266                    packet().detlog_identity.len(),
267                    packet().forward_policy.len(),
268                ][..field]
269                    .iter()
270                    .sum::<usize>();
271            bytes.drain(start..start + length);
272            bytes[index..index + 2].copy_from_slice(&0_u16.to_be_bytes());
273            let total = bytes.len() as u16;
274            bytes[10..12].copy_from_slice(&total.to_be_bytes());
275            assert_eq!(
276                decode(&bytes),
277                Err(PacketError::Empty),
278                "empty field {field}"
279            );
280        }
281        let mut trailing = valid.clone();
282        trailing.push(1);
283        assert_eq!(decode(&trailing), Err(PacketError::Size));
284        let total = trailing.len() as u16;
285        trailing[10..12].copy_from_slice(&total.to_be_bytes());
286        assert_eq!(decode(&trailing), Err(PacketError::Length));
287        let identity_start = 98 + packet().coordinator.len();
288        for (byte, error) in [(0, PacketError::Nul), (0xff, PacketError::Utf8)] {
289            let mut bytes = valid.clone();
290            bytes[identity_start] = byte;
291            assert_eq!(decode(&bytes), Err(error));
292        }
293        let policy_start = identity_start + packet().detlog_identity.len();
294        let mut bytes = valid;
295        bytes[policy_start] = 0xff;
296        assert_eq!(decode(&bytes), Err(PacketError::Utf8));
297    }

299    #[test]
300    fn packet_bound_and_host_validation_have_no_fallback() {
301        let mut packet = packet();
302        let path = vec![
303            b'x';
304            MAX_PACKET_LEN
305                - HEADER_LEN
306                - packet.detlog_identity.len()
307                - packet.forward_policy.len()
308        ];
309        packet.coordinator = &path;
310        let bytes = encode(packet).unwrap();
311        assert_eq!(bytes.len(), 4096);
312        assert!(decode(&bytes).is_ok());
313        let oversized = vec![b'x'; path.len() + 1];
314        packet.coordinator = &oversized;
315        assert_eq!(encode(packet), Err(PacketError::Size));
316        packet.coordinator = b"";
317        assert_eq!(encode(packet), Err(PacketError::Empty));
318        packet.coordinator = b"x\0";
319        assert_eq!(encode(packet), Err(PacketError::Nul));
320        packet.coordinator = b"x";
321        packet.detlog_identity = "";
322        assert_eq!(encode(packet), Err(PacketError::Empty));
323        packet.detlog_identity = "1";
324        packet.forward_policy = "\0";
325        assert_eq!(encode(packet), Err(PacketError::Nul));
326        assert_eq!(decode(&vec![1; 4097]), Err(PacketError::Size));
327    }

339        let mut allowed = bounded_config();

349            |c| c.target_timeslice = Some(std::num::NonZeroU64::new(1).unwrap()),

371            assert_eq!(
372                bincode::serde::encode_to_vec(&config, bincode::config::legacy()).unwrap(),
373                before
374            );
375        }
