9//! The host identity of a file a guest opened.
10//!
11//! Hermit maps each host inode it sees to a deterministic one in the order it
12//! first sees them, so a host file that is REPLACED while a run is going (a
13//! package manager rewriting `/etc/ld.so.cache` between the opens of two
14//! guest processes, for example) gives that run one more inode than the other
15//! run, and every later inode number differs. `hermit run --verify` then
16//! reports a divergence that the guest did not cause. Each verification run
17//! therefore records one [`HostInputRecord`] per host file the guest opens.
18//! Detcore reports each open to its global state
19//! ([`crate::config::Config::record_host_inputs`]), which writes the run's
20//! records to a side file ([`crate::config::Config::host_input_log`]) when the
21//! run ends, never part of the compared log. On a divergence the two runs'
22//! records are compared to name such a change.
23
24use serde::Deserialize;
25use serde::Serialize;
26
27/// The host's identity for a file at the moment the guest opened it.
28#[derive(Clone, Copy, Debug, Deserialize, Eq, Hash, PartialEq, Serialize)]
29pub struct HostFileIdentity {
30    /// The host `st_dev`.
31    pub dev: u64,
32    /// The host `st_ino`.
33    pub ino: u64,
34    /// The host `st_size`.
35    pub size: i64,
36    /// The host `st_mtime`, seconds.
37    pub mtime_sec: i64,
38    /// The host `st_mtime`, nanoseconds.
39    pub mtime_nsec: i64,
40}
41
42impl HostFileIdentity {
43    /// The identity in a host `struct stat`.
44    pub fn from_stat(stat: &libc::stat) -> Self {
45        Self {
46            dev: stat.st_dev,
47            ino: stat.st_ino,
48            size: stat.st_size,
49            mtime_sec: stat.st_mtime,
50            mtime_nsec: stat.st_mtime_nsec,
51        }
52    }
53}
54
55impl std::fmt::Display for HostFileIdentity {
56    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
57        write!(
58            f,
59            "dev:ino {}:{}:{} size {} mtime {}.{:09}",
60            libc::major(self.dev),
61            libc::minor(self.dev),
62            self.ino,
63            self.size,
64            self.mtime_sec,
65            self.mtime_nsec
66        )
67    }
68}

70/// One host file a guest opened: the path it named, made absolute against its
71/// working directory or directory descriptor, the host identity of the file
72/// the open reached, and the open itself, as the thread (`dtid`) and that
73/// thread's syscall number (`syscall`, the `finish syscall #N` of the run's
74/// log) that made it. A path that is not UTF-8 is recorded lossily.
75#[derive(Clone, Debug, Deserialize, Eq, PartialEq, Serialize)]
76pub struct HostInputRecord {
77    pub path: String,
78    pub dtid: u64,
79    pub syscall: u64,
80    #[serde(flatten)]
81    pub identity: HostFileIdentity,
82}
83
84/// A path the guest itself rebound in its file namespace during the run:
85/// renamed to or from, linked, unlinked, made as a symlink, node or directory,
86/// removed as a directory, or opened to create (see
87/// [`crate::config::Config::record_host_inputs`]). Made absolute as an open's
88/// path is. A host file whose path, or a directory above it, the guest
89/// rebound may have been replaced by the guest, so such a change is never
90/// named as a host input change.
91#[derive(Clone, Debug, Deserialize, Eq, PartialEq, Serialize)]
92#[serde(deny_unknown_fields)]
93pub struct HostMutationRecord {
94    pub mutated: String,
95}
96
97/// The last line of a complete host-input log: how many records, of both
98/// kinds, precede it. A log without it, or with another count, is incomplete.
99#[derive(Clone, Copy, Debug, Deserialize, Eq, PartialEq, Serialize)]
100#[serde(deny_unknown_fields)]
101pub struct HostInputLogEnd {
102    pub records: u64,
103}
