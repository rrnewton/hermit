7//! Separate, inactive native Initial/Successor bootstrap input data.
8//!
9//! No decoder fallback to HLIINIT1 exists. Packet bytes, socket identities,
10//! nonces, tickets and digests are not admission or kernel-exec authority. A
11//! future trusted native consumer must compare independently held expectations,
12//! verify distinct protected actual sockets, and redeem its one-use instance.
13//! The runtime/Tool composite is delivered separately on the bounded state path.
14
15use std::fmt;
16
17/// Entire control packet bound, unchanged from the existing private input ABI.
18pub const MAX_PACKET_LEN: usize = 4096;
19/// A protocol distinct from the unchanged HLIINIT1 packet.
20pub const MAGIC: [u8; 8] = *b"HLNEXEC1";
21/// Only this wire version is admitted by the shape decoder.
22pub const VERSION: u16 = 1;
23/// Initial header, including all four variable-field lengths.
24pub const INITIAL_HEADER_LEN: usize = 200;
25/// Successor header additionally carries the composite state digest.
26pub const SUCCESSOR_HEADER_LEN: usize = 232;
27const TOKEN_START: usize = 160;
28const TOKEN_END: usize = 192;
29const LENGTH_BYTES: usize = 8;
30
31/// The selected input shape; not a native lifecycle authorization.
32#[derive(Clone, Copy, Debug, Eq, PartialEq)]
33#[repr(u16)]
34pub enum Kind {
35    /// Fresh one-use native launch instance.
36    Initial = 1,
37    /// One-use native exec successor transaction.
38    Successor = 2,
39}
40
41/// Distinct fixed fields for the two native input kinds.
42#[derive(Clone, Copy, Eq, PartialEq)]
43pub enum Image<'a> {
44    /// Independently issued nonce, to be consumed by actual launch admission.
45    Initial {
46        /// Thirty-two opaque bytes, with no replay or authority check here.
47        nonce: &'a [u8; 32],
48    },
49    /// Independently issued ticket and expected exact composite-state digest.
50    Successor {
51        /// Thirty-two opaque bytes, consumed before Config/state delivery.
52        ticket: &'a [u8; 32],
53        /// SHA256 of the entire exact Tool/runtime composite, not a prefix.
54        composite_digest: &'a [u8; 32],
55    },
56}
57impl fmt::Debug for Image<'_> {
58    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
59        // Shape diagnostics must not print a protected nonce or exec ticket.
60        match self {
61            Self::Initial { .. } => f.write_str("Initial(<private nonce>)"),
62            Self::Successor {
63                composite_digest, ..
64            } => f
65                .debug_struct("Successor")
66                .field("ticket", &"<private>")
67                .field("composite_digest", composite_digest)
68                .finish(),
69        }
70    }
71}
72impl Image<'_> {
73    /// Exact wire kind corresponding to these fields.
74    pub fn kind(&self) -> Kind {
75        match self {
76            Self::Initial { .. } => Kind::Initial,
77            Self::Successor { .. } => Kind::Successor,
78        }
79    }
80    fn header_len(&self) -> usize {
81        match self {
82            Self::Initial { .. } => INITIAL_HEADER_LEN,
83            Self::Successor { .. } => SUCCESSOR_HEADER_LEN,
84        }
85    }
86}

88/// Borrowed native control fields. Decoding establishes byte shape only.
89#[derive(Clone, Copy, Debug, Eq, PartialEq)]
90pub struct Packet<'a> {
91    /// Canonical legacy wire Config SHA256 supplied independently by admission.
92    pub config_digest: &'a [u8; 32],
93    /// Existing exactly sixteen ASCII-hex Config schema characters.
94    pub config_schema: &'a [u8; 16],
95    /// Full declared Rust native source SHA256, independently computed.
96    pub native_source_digest: &'a [u8; 32],
97    /// Held immutable target identity; bytes here do not prove its file.
98    pub target_digest: &'a [u8; 32],
99    /// Held native bootstrap/interpreter identity; not exec attestation.
100    pub bootstrap_digest: &'a [u8; 32],
101    /// One of the distinct nonce or ticket/composite input shapes.
102    pub image: Image<'a>,
103    /// Nonempty NUL-free raw Unix coordinator path; UTF-8 is not required.
104    pub coordinator: &'a [u8],
105    /// Unchanged UTF-8 DETLOG fd/device/inode identity string contract.
106    pub detlog_identity: &'a str,
107    /// Unchanged UTF-8 genuine forwarding policy, compared by the consumer.
108    pub forward_policy: &'a str,
109    /// Separate UTF-8 inherited observer fd/device/inode identity data. The
110    /// consumer must verify its actual socket is distinct and protected; string
111    /// inequality is insufficient and this codec does not query descriptors.
112    pub observer_identity: &'a str,
113}

115/// Independently held expected data, not values learned from packet decoding.
116#[derive(Clone, Copy, Debug, Eq, PartialEq)]
117pub struct ExpectedBindings<'a> {
118    /// Canonical full wire Config SHA256.
119    pub config_digest: &'a [u8; 32],
120    /// Actual compiled Config schema, compared byte-for-byte, never by prefix.
121    pub config_schema: &'a [u8; 16],
122    /// Full actual compiled native source digest.
123    pub native_source_digest: &'a [u8; 32],
124    /// Independently held immutable target digest.
125    pub target_digest: &'a [u8; 32],
126    /// Independently held bootstrap/interpreter digest.
127    pub bootstrap_digest: &'a [u8; 32],
128}

130/// Exact data disagreement. No variant is a kernel errno or authorization.
131#[derive(Clone, Copy, Debug, Eq, PartialEq)]
132pub enum BindingMismatch {
133    ConfigDigest,
134    ConfigSchema,
135    NativeSourceDigest,
136    TargetDigest,
137    BootstrapDigest,
138}
139impl Packet<'_> {
140    /// Compare complete fixed bindings against independent expected inputs.
141    /// This neither verifies sockets nor redeems the nonce/ticket, checks the
142    /// composite state bytes, or establishes a watched birth/exec transition.
143    pub fn check_bindings(&self, expected: ExpectedBindings<'_>) -> Result<(), BindingMismatch> {
144        for (actual, expected, error) in [
145            (
146                self.config_digest,
147                expected.config_digest,
148                BindingMismatch::ConfigDigest,
149            ),
150            (
151                self.native_source_digest,
152                expected.native_source_digest,
153                BindingMismatch::NativeSourceDigest,
154            ),
155            (
156                self.target_digest,
157                expected.target_digest,
158                BindingMismatch::TargetDigest,
159            ),
160            (
161                self.bootstrap_digest,
162                expected.bootstrap_digest,
163                BindingMismatch::BootstrapDigest,
164            ),
165        ] {
166            if actual != expected {
167                return Err(error);
168            }
169        }
170        if self.config_schema != expected.config_schema {
171            return Err(BindingMismatch::ConfigSchema);
172        }
173        Ok(())
174    }
175    fn variable_fields(&self) -> [&[u8]; 4] {
176        [
177            self.coordinator,
178            self.detlog_identity.as_bytes(),
179            self.forward_policy.as_bytes(),
180            self.observer_identity.as_bytes(),
181        ]
182    }
183}

185/// Bounded shape error with no allocated error message.
186#[derive(Clone, Copy, Debug, Eq, PartialEq)]
187pub enum PacketError {
188    Size,
189    Magic,
190    Version,
191    Kind,
192    Reserved,
193    Length,
194    Empty,
195    Nul,
196    Utf8,
197    Schema,
198}
199impl fmt::Display for PacketError {
200    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
201        f.write_str(match self {
202            Self::Size => "native input complete size mismatch",
203            Self::Magic => "native input magic mismatch",
204            Self::Version => "native input version mismatch",
205            Self::Kind => "native input kind mismatch",
206            Self::Reserved => "native input reserved bits are nonzero",
207            Self::Length => "native input field lengths do not consume exactly",
208            Self::Empty => "native input mandatory field is empty",
209            Self::Nul => "native input field contains NUL",
210            Self::Utf8 => "native input identity or policy is not UTF-8",
211            Self::Schema => "native input Config schema is not sixteen ASCII hex characters",
212        })
213    }
214}
215impl std::error::Error for PacketError {}

217fn word(bytes: &[u8], start: usize) -> usize {
218    usize::from(u16::from_be_bytes([bytes[start], bytes[start + 1]]))
219}
220fn check_field(bytes: &[u8]) -> Result<(), PacketError> {
221    if bytes.is_empty() {
222        return Err(PacketError::Empty);
223    }
224    if bytes.contains(&0) {
225        return Err(PacketError::Nul);
226    }
227    Ok(())
228}
229fn check_schema(bytes: &[u8; 16]) -> Result<(), PacketError> {
230    if bytes.iter().all(u8::is_ascii_hexdigit) {
231        Ok(())
232    } else {
233        Err(PacketError::Schema)
234    }
235}

237/// Validate the complete new packet shape without allocation or old fallback.
238pub fn decode(bytes: &[u8]) -> Result<Packet<'_>, PacketError> {
239    if !(INITIAL_HEADER_LEN..=MAX_PACKET_LEN).contains(&bytes.len()) {
240        return Err(PacketError::Size);
241    }
242    if bytes[..8] != MAGIC {
243        return Err(PacketError::Magic);
244    }
245    if word(bytes, 8) != usize::from(VERSION) {
246        return Err(PacketError::Version);
247    }
248    if word(bytes, 10) != bytes.len() {
249        return Err(PacketError::Size);
250    }
251    let kind = match word(bytes, 12) {
252        1 => Kind::Initial,
253        2 => Kind::Successor,
254        _ => return Err(PacketError::Kind),
255    };
256    if word(bytes, 14) != 0 {
257        return Err(PacketError::Reserved);
258    }
259    let header = match kind {
260        Kind::Initial => INITIAL_HEADER_LEN,
261        Kind::Successor => SUCCESSOR_HEADER_LEN,
262    };
263    if bytes.len() < header {
264        return Err(PacketError::Size);
265    }
266    let lengths_start = header - LENGTH_BYTES;
267    let mut ends = [header; 5];
268    for index in 0..4 {
269        ends[index + 1] = ends[index]
270            .checked_add(word(bytes, lengths_start + index * 2))
271            .filter(|end| *end <= bytes.len())
272            .ok_or(PacketError::Length)?;
273    }
274    if ends[4] != bytes.len() {
275        return Err(PacketError::Length);
276    }
277    let fields = std::array::from_fn::<_, 4, _>(|index| &bytes[ends[index]..ends[index + 1]]);
278    for field in fields {
279        check_field(field)?;
280    }
281    let token = bytes[TOKEN_START..TOKEN_END]
282        .try_into()
283        .map_err(|_| PacketError::Length)?;
284    let image = match kind {
285        Kind::Initial => Image::Initial { nonce: token },
286        Kind::Successor => Image::Successor {
287            ticket: token,
288            composite_digest: bytes[TOKEN_END..TOKEN_END + 32]
289                .try_into()
290                .map_err(|_| PacketError::Length)?,
291        },
292    };
293    let packet = Packet {
294        config_digest: bytes[16..48].try_into().map_err(|_| PacketError::Length)?,
295        config_schema: bytes[48..64].try_into().map_err(|_| PacketError::Schema)?,
296        native_source_digest: bytes[64..96].try_into().map_err(|_| PacketError::Length)?,
297        target_digest: bytes[96..128].try_into().map_err(|_| PacketError::Length)?,
298        bootstrap_digest: bytes[128..160]
299            .try_into()
300            .map_err(|_| PacketError::Length)?,
301        image,
302        coordinator: fields[0],
303        detlog_identity: std::str::from_utf8(fields[1]).map_err(|_| PacketError::Utf8)?,
304        forward_policy: std::str::from_utf8(fields[2]).map_err(|_| PacketError::Utf8)?,
305        observer_identity: std::str::from_utf8(fields[3]).map_err(|_| PacketError::Utf8)?,
306    };
307    check_schema(packet.config_schema)?;
308    Ok(packet)
309}

311/// Host-side encoding of the same shape. State bytes do not enter this packet.
312pub fn encode(packet: Packet<'_>) -> Result<Vec<u8>, PacketError> {
313    let fields = packet.variable_fields();
314    let mut total = packet.image.header_len();
315    for field in fields {
316        if field.len() > MAX_PACKET_LEN {
317            return Err(PacketError::Size);
318        }
319        check_field(field)?;
320        total = total
321            .checked_add(field.len())
322            .filter(|total| *total <= MAX_PACKET_LEN)
323            .ok_or(PacketError::Size)?;
324    }
325    check_schema(packet.config_schema)?;
326    let mut bytes = Vec::with_capacity(total);
327    bytes.extend_from_slice(&MAGIC);
328    bytes.extend_from_slice(&VERSION.to_be_bytes());
329    bytes.extend_from_slice(&(total as u16).to_be_bytes());
330    bytes.extend_from_slice(&(packet.image.kind() as u16).to_be_bytes());
331    bytes.extend_from_slice(&0_u16.to_be_bytes());
332    bytes.extend_from_slice(packet.config_digest);
333    bytes.extend_from_slice(packet.config_schema);
334    bytes.extend_from_slice(packet.native_source_digest);
335    bytes.extend_from_slice(packet.target_digest);
336    bytes.extend_from_slice(packet.bootstrap_digest);
337    match packet.image {
338        Image::Initial { nonce } => bytes.extend_from_slice(nonce),
339        Image::Successor {
340            ticket,
341            composite_digest,
342        } => {
343            bytes.extend_from_slice(ticket);
344            bytes.extend_from_slice(composite_digest);
345        }
346    }
347    for field in fields {
348        bytes.extend_from_slice(&(field.len() as u16).to_be_bytes());
349    }
350    for field in fields {
351        bytes.extend_from_slice(field);
352    }
353    Ok(bytes)
354}

356/// Encode into caller-owned storage with exactly the same wire bytes as
357/// `encode`. Bounds and mandatory fields are checked before the first write.
358/// No allocation or packet-supplied authentication is performed.
359pub fn encode_into(packet: Packet<'_>, output: &mut [u8]) -> Result<usize, PacketError> {
360    let fields = packet.variable_fields();
361    let mut total = packet.image.header_len();
362    for field in fields {
363        if field.len() > MAX_PACKET_LEN {
364            return Err(PacketError::Size);
365        }
366        check_field(field)?;
367        total = total
368            .checked_add(field.len())
369            .filter(|total| *total <= MAX_PACKET_LEN)
370            .ok_or(PacketError::Size)?;
371    }
372    check_schema(packet.config_schema)?;
373    if output.len() < total {
374        return Err(PacketError::Size);
375    }
376    let mut offset = 0;
377    let mut append = |bytes: &[u8]| {
378        output[offset..offset + bytes.len()].copy_from_slice(bytes);
379        offset += bytes.len();
380    };
381    append(&MAGIC);
382    append(&VERSION.to_be_bytes());
383    append(&(total as u16).to_be_bytes());
384    append(&(packet.image.kind() as u16).to_be_bytes());
385    append(&0_u16.to_be_bytes());
386    append(packet.config_digest);
387    append(packet.config_schema);
388    append(packet.native_source_digest);
389    append(packet.target_digest);
390    append(packet.bootstrap_digest);
391    match packet.image {
392        Image::Initial { nonce } => append(nonce),
393        Image::Successor {
394            ticket,
395            composite_digest,
396        } => {
397            append(ticket);
398            append(composite_digest);
399        }
400    }
401    for field in fields {
402        append(&(field.len() as u16).to_be_bytes());
403    }
404    for field in fields {
405        append(field);
406    }
407    if offset != total {
408        return Err(PacketError::Length);
409    }
410    Ok(total)
411}
