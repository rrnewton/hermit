36/// One mount row whose kernel-private root must be replaced before it becomes
37/// guest-visible.
38///
39/// The CLI populates this only after proving, with held file descriptors in the
40/// completed mount namespace, that the mount is one Hermit created.  The raw
41/// mount ID is namespace-local, so these entries are valid only for the one
42/// container run whose configuration carries them.
43#[derive(Debug, Serialize, Deserialize, Clone, Eq, PartialEq)]
44pub struct MountInfoRootRewrite {
45    /// Mount ID read from the held target descriptor's `/proc/self/fdinfo`.
46    pub raw_mount_id: u64,
47    /// Stable guest-visible replacement for the row's root field.
48    pub deterministic_root: Vec<u8>,
49    /// Exact encoded kernel root prefix used for descendant mount rows.
50    ///
51    /// This is present only for a proven private `/tmp`. Mounts installed below
52    /// that directory before it is bound over guest `/tmp` otherwise expose the
53    /// randomly named backing directory in mountinfo field 5.
54    #[serde(default)]
55    pub raw_root_prefix: Option<Vec<u8>>,
56    /// Guest-visible prefix replacing `raw_root_prefix`.
57    #[serde(default)]
58    pub deterministic_root_prefix: Option<Vec<u8>>,
59    /// Exact encoded host path prefix used for descendant mountpoints.
60    #[serde(default)]
61    pub raw_mountpoint_prefix: Option<Vec<u8>>,
62    /// Guest-visible prefix replacing `raw_mountpoint_prefix`.
63    #[serde(default)]
64    pub deterministic_mountpoint_prefix: Option<Vec<u8>>,
65}

640    ///
641    /// WHAT IT SEES THAT THE MAPPING HASHES DO NOT. `--detlog-heap` and `--detlog-stack` hash a
642    /// whole named mapping, so their coverage is decided by where the guest happened to ALLOCATE
643    /// a buffer. Measured, three runs per cell, same netlink exchange with only the receive
644    /// buffer's home changed: a `[stack]` buffer is missed by `--detlog-heap`, a `[heap]` buffer
645    /// is missed by `--detlog-stack`, and a BSS/static or anonymous-`mmap` buffer is missed by
646    /// BOTH even with both enabled. Anonymous `mmap` is where glibc puts any `malloc` above the

665    /// that must be requested is not a standard. The opt-out exists for the
666    /// deliberate case (bulk I/O where the cost matters and content parity is
667    /// not the question), not as the ordinary setting.
668    ///
669    /// COST OF THE DEFAULT, measured 2026-08-24 on a 316-core x86_64 Linux
670    /// build host: a typical small test guest pays about ONE MILLISECOND

674    /// DEBUG build costs +3.4s, roughly 50x more, because the hash loop is
675    /// unoptimized -- so a debug-built node moving tens of megabytes is the one

722fn try_parse_numbers_with_colon(from_str: &str) -> anyhow::Result<(DetTid, u64)> {
723    if let Some((thread_id_str, time_str)) = from_str.split_once(':') {
724        Ok((
725            thread_id_str
726                .parse::<DetTid>()
727                .map_err(anyhow::Error::msg)?,
728            time_str.parse::<u64>().map_err(anyhow::Error::msg)?,
729        ))
730    } else {
731        anyhow::bail!(
732            "unable to parse <thread_id>:<logical_time> from '{}'",
733            from_str
734        )
735    }
736}

739    <bytesize::ByteSize as FromStr>::from_str(from_str)

741        .map_err(anyhow::Error::msg)
742}

759    pub fn minimum_max_timeslice_nanos(&self) -> u64 {

1346/// A fingerprint of this build's [`Config`] payload and the configuration and
1347/// clock RPC definitions shared by a plugin and its coordinator.
1348///
1349/// # Why this exists
1350///
1351/// An out-of-process plugin such as `libdetcore_sabre.so` is a separate Cargo
1352/// artifact that lands in the same target directory as `hermit`. Changing
1353/// `Config` or `DetTime` -- or merely switching branches -- leaves the plugin
1354/// stale while everything still *looks* built. `Config` is transferred during
1355/// the RPC handshake, and `DetTime` is the first field in every Detcore request.
1356/// A stale plugin decodes either against the wrong layout and the failure
1357/// surfaces as an opaque codec error: measured, one added `bool` field
1358/// produced `Decode(InvalidBooleanValue(20))` at connect, which points nowhere
1359/// near "your plugin is from a different build" and cost a long diagnosis while
1360/// blocking every SaBRe measurement.
1361///
1362/// # What it measures
1363///
1364/// Two encodings of `Config::default()` and the source definitions for the
1365/// configuration and clock RPC fields are fingerprinted with separate domains:
1366///
1367/// - the exact legacy-bincode bytes used by Reverie RPC, which detect changes
1368///   such as `u32` to `u64` even when both default to JSON number zero; and
1369/// - the JSON encoding, which carries every field name and makes a pure rename
1370///   visible even though bincode is positional; and
1371/// - the source files defining `Config`, its local serialized field types, and
1372///   `DetTime`, which catch wire-incompatible changes hidden by a default such
1373///   as `Option<u64>::None` to `Option<u32>::None`, or an added clock field that
1374///   leaves both encodings of `Config` unchanged.
1375///
1376/// The source and JSON domains are deliberately stricter than the wire format
1377/// strictly requires. A documentation-only edit in one of these files can
1378/// require rebuilding the plugin; missing a wire-incompatible hidden variant
1379/// can make it decode the handshake or a subsequent request at the wrong offsets.
1380pub fn config_wire_fingerprint() -> String {
1381    let config = Config::default();
1382    let wire = bincode::serde::encode_to_vec(&config, bincode::config::legacy())
1383        .expect("Config::default() must encode with the Reverie RPC bincode configuration");
1384    let named_shape = serde_json::to_string(&config)
1385        .expect("Config::default() must encode as JSON for field-name checking");
1386    fingerprint_of_config_material(&wire, &named_shape, CONFIG_DEFINITION_SOURCES)
1387}
1388
1389/// Domain-separated FNV-1a over wire bytes, named JSON, and defining source.
1390/// This is a mismatch detector, not a security boundary. Length-prefixing each
1391/// domain prevents two different source-file boundaries from hashing the same
1392/// concatenation.
1393fn fingerprint_of_config_material(
1394    wire: &[u8],
1395    named_shape: &str,
1396    definition_sources: &[&[u8]],
1397) -> String {
1398    let mut hash: u64 = 0xcbf2_9ce4_8422_2325;
1399    let mut update = |domain: u8, bytes: &[u8]| {
1400        for byte in std::iter::once(&domain)
1401            .chain((bytes.len() as u64).to_le_bytes().iter())
1402            .chain(bytes.iter())
1403        {
1404            hash ^= u64::from(*byte);
1405            hash = hash.wrapping_mul(0x1000_0000_01b3);
1406        }
1407    };
1408    update(0, wire);
1409    update(1, named_shape.as_bytes());
1410    for source in definition_sources {
1411        update(2, source);
1412    }
1413    format!("{hash:016x}")
1414}

1715    fn config_fingerprint_is_stable_and_shape_sensitive() {
1716        // STABLE: a build must agree with itself, or the guard would reject a
1717        // MATCHED pair -- which would be worse than having no guard at all.

1724        assert_eq!(
1725            fingerprint_of_config_material(&wire, &base, CONFIG_DEFINITION_SOURCES),
1726            config_wire_fingerprint()
1727        );
1728
1729        // SHAPE-SENSITIVE, checked on the same mechanism the real function uses.
1730        // One added field is exactly the change that caused the outage.
1731        let with_extra_field = format!("{},\"a_new_flag\":false}}", &base[..base.len() - 1]);
1732        assert_ne!(
1733            fingerprint_of_config_material(&wire, &with_extra_field, CONFIG_DEFINITION_SOURCES),
1734            config_wire_fingerprint()
1735        );
1736        // A removed field.
1737        let removed = base.replacen("\"virtualize_time\":true,", "", 1);
1738        assert_ne!(
1739            fingerprint_of_config_material(&wire, &removed, CONFIG_DEFINITION_SOURCES),
1740            config_wire_fingerprint()
1741        );
1742        // A pure rename, which bincode would tolerate but which we still refuse.
1743        let renamed = base.replacen("\"virtualize_time\"", "\"virtualise_time\"", 1);
1744        assert_ne!(
1745            fingerprint_of_config_material(&wire, &renamed, CONFIG_DEFINITION_SOURCES),
1746            config_wire_fingerprint()
1747        );
1748
1749        // The counterexample the JSON-only fingerprint missed: serde_json emits
1750        // the same text for integer zero regardless of width, but legacy bincode
1751        // changes the payload width. A stale peer would decode every following
1752        // field at the wrong offset.
1753        #[derive(Serialize)]
1754        struct U32Field {
1755            field: u32,
1756        }
1757        #[derive(Serialize)]
1758        struct U64Field {
1759            field: u64,
1760        }

1762        let u64_value = U64Field { field: 0 };

1764        let u64_json = serde_json::to_string(&u64_value).unwrap();

1766            u32_json, u64_json,
1767            "the planted JSON collision must be real"
1768        );

1770            bincode::serde::encode_to_vec(&u32_value, bincode::config::legacy()).unwrap();
1771        let u64_wire =
1772            bincode::serde::encode_to_vec(&u64_value, bincode::config::legacy()).unwrap();
1773        assert_ne!(u32_wire, u64_wire, "the planted wire retype must be real");
1774        assert_ne!(
1775            fingerprint_of_config_material(&u32_wire, &u32_json, &[b"struct S { field: u32 }"]),
1776            fingerprint_of_config_material(&u64_wire, &u64_json, &[b"struct S { field: u64 }"]),
1777            "a wire-incompatible integer retype must change the fingerprint"
1778        );

1784            field: Option<u32>,

1804            fingerprint_of_config_material(
1805                &optional_u32_wire,
1806                &optional_u32_json,
1807                &[b"struct S { field: Option<u32> }"]
1808            ),
1809            fingerprint_of_config_material(
1810                &optional_u64_wire,
1811                &optional_u64_json,
