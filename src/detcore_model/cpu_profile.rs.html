9//! The virtual CPU a guest sees (<https://github.com/rrnewton/hermit/issues/4243>).
10
11use serde::Deserialize;
12use serde::Serialize;
13
14/// Which x86-64 microarchitecture level the virtual CPUID table describes.
15#[derive(
16    Debug,
17    Clone,
18    Copy,
19    Default,
20    PartialEq,
21    Eq,
22    Hash,
23    Serialize,
24    Deserialize,
25    clap::ValueEnum
26)]
27pub enum CpuProfile {
28    /// x86-64-v2, the default: the table every earlier Hermit has served.
29    #[default]
30    #[value(name = "x86-64-v2")]
31    #[serde(rename = "x86-64-v2")]
32    X86_64V2,
33    /// x86-64-v3: v2 plus AVX2, BMI1/2, FMA, F16C, LZCNT, MOVBE and XSAVE.
34    /// Offered only on the ptrace backend, on hosts that pass its startup
35    /// qualification.
36    #[value(name = "x86-64-v3")]
37    #[serde(rename = "x86-64-v3")]
38    X86_64V3,
39}
40
41impl CpuProfile {
42    /// Whether this is the default profile, which needs no qualification and
43    /// binds no host facts.
44    pub fn is_default(&self) -> bool {
45        *self == CpuProfile::X86_64V2
46    }
47
48    /// The name used on the command line.
49    pub fn name(&self) -> &'static str {
50        match self {
51            CpuProfile::X86_64V2 => "x86-64-v2",
52            CpuProfile::X86_64V3 => "x86-64-v3",
53        }
54    }
55}
56
57impl std::fmt::Display for CpuProfile {
58    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
59        f.write_str(self.name())
60    }
61}

63/// The host facts a v3 guest's CPUID and xstate answers are built from,
64/// captured once at startup on a host that passed the qualification, and bound
65/// into a recording so that replay refuses a host where any of them differs.
66///
67/// On the ptrace backend a guest's XGETBV and XSAVE run on the host, so leaf
68/// 0xD must describe the host's XCR0 and save layout; everything else in the v3
69/// table is fixed.
70#[derive(Debug, Clone, PartialEq, Eq, Hash, Serialize, Deserialize)]
71pub struct HostXstate {
72    /// CPUID leaf 0 EBX, EDX, ECX: the vendor string.
73    pub vendor: [u32; 3],
74    /// CPUID leaf 1 EAX: family, model and stepping.
75    pub signature: u32,
76    /// XCR0, as XGETBV(0) reads it.
77    pub xcr0: u64,
78    /// The host's CPUID leaf 0xD rows the guest table is built from, as
79    /// `[subleaf, eax, ebx, ecx, edx]`: subleaf 0 and each enabled component
80    /// from 2 up, in order.
81    pub xsave_leaves: Vec<[u32; 5]>,
82    /// `arch_prctl` ARCH_GET_XCOMP_SUPP, ARCH_GET_XCOMP_PERM and
83    /// ARCH_GET_XCOMP_GUEST_PERM.
84    pub xcomp_supp: u64,
85    pub xcomp_perm: u64,
86    pub xcomp_guest_perm: u64,
87    /// The first component index ARCH_REQ_XCOMP_PERM answers EINVAL for
88    /// (Linux's XFEATURE_MAX); below it the answer is EOPNOTSUPP.
89    pub xfeature_max: u32,
90    /// The qualification this host passed: its case count and whether the
91    /// forced-exit arm ran.
92    pub probe_cases: u32,
93    pub probe_forced_exit: bool,
94}
95
96impl HostXstate {
97    /// The host's leaf 0xD row for `subleaf`, if it was captured.
98    pub fn xsave_leaf(&self, subleaf: u32) -> Option<[u32; 4]> {
99        self.xsave_leaves
100            .iter()
101            .find(|row| row[0] == subleaf)
102            .map(|row| [row[1], row[2], row[3], row[4]])
103    }
104}

106/// The residual as a literal, so that the command-line help can include it
107/// with `concat!` while [`V3_RESIDUAL`] stays the one source of its words.
108#[macro_export]
109macro_rules! v3_residual {
110    () => {
111        "With --cpu-profile x86-64-v3, Hermit checks at startup that this host does not \
112         change the XSTATE_BV bits a guest reads (with XSAVE, XSAVEC or XSAVEOPT, or in a \
113         signal frame) when the host saves and restores the guest's registers. The check \
114         samples a fixed set of cases and forces only CPUID exits, so a CPU or hypervisor \
115         that behaves differently only on other instruction paths, or on VM exits the guest \
116         cannot see, is not excluded. Firmware (SMM) entries cannot be checked at all. \
117         Default runs use x86-64-v2, which this does not affect."
118    };
119}

121/// What the host's XSAVE results can depend on, stated wherever v3 is offered:
122/// the `--cpu-profile` help, `--verify` output, the recording and the docs.
123pub const V3_RESIDUAL: &str = v3_residual!();
124
125/// The long help of `--cpu-profile`, for `hermit run` and `hermit record start`.
126pub const CPU_PROFILE_LONG_HELP: &str = concat!(
127    "The CPU the guest sees: x86-64-v2 (the default) or x86-64-v3. v3 is offered only on \
128     the ptrace backend, on hosts that pass a startup qualification \
129     (`hermit host-capabilities --cpu-profile x86-64-v3` runs it alone). A recording binds \
130     the host facts its guest's answers come from, and replay refuses a host where any of \
131     them differs (<https://github.com/rrnewton/hermit/issues/4243>).\n\n",
132    v3_residual!()
133);

135#[cfg(test)]
136mod tests {
137    use clap::ValueEnum;
138
139    use super::*;
140
141    #[test]
142    fn names_round_trip_on_the_command_line_and_in_serde() {
143        for profile in CpuProfile::value_variants() {
144            assert_eq!(CpuProfile::from_str(profile.name(), false), Ok(*profile));
145            let json = serde_json::to_string(profile).unwrap();
146            assert_eq!(json, format!("\"{}\"", profile.name()));
147            assert_eq!(serde_json::from_str::<CpuProfile>(&json).unwrap(), *profile);
148        }
149        assert_eq!(CpuProfile::default(), CpuProfile::X86_64V2);
150        assert!(CpuProfile::X86_64V2.is_default());
151        assert!(!CpuProfile::X86_64V3.is_default());
152    }
