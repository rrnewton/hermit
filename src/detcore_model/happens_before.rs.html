9//! Happens-before edges: a sparse, authored partial order over dynamic events.
10//!
11//! Where `--replay-schedule-from` replays a *complete* total order captured from
12//! a prior run, a happens-before specification pins down only the *few* events
13//! that matter for a race and lets the deterministic scheduler fill in the rest.
14//! An agent (or human) that already knows a target race can therefore construct
15//! it deterministically instead of blind seed-search.
16//!
17//! # Model
18//!
19//! An [`Anchor`] names a precise, deterministic per-thread stop point. Following
20//! the owner's refinement of RFC #1146, the *primary* addressing is a
21//! [`Position`] — "after N syscalls" or "after M retired conditional branches
22//! (RCBs)" on a specific thread — optionally decorated with a [`CodeLocation`]
23//! (function and/or source line, resolved from debug info) for readability. The
24//! RFC's richer addressing modes (Nth occurrence of a named syscall, RIP hit, or
25//! function entry) remain expressible but are deliberately *not* the lead
26//! addressing scheme.
27//!
28//! A [`HappensBeforeEdge`] states that one anchor must be observed before another
29//! thread is allowed to proceed past its anchor. A [`HappensBeforeSpec`] is the
30//! whole authored partial order: a table of named threads, a table of named
31//! events (anchors), and the edge list connecting them.
32//!
33//! This module owns the *model* only: parsing (JSON and a terse DSL),
34//! normalization, and static validation (name resolution, exactly-one-position,
35//! and cycle detection). Resolving a [`CodeLocation`] to a concrete address via
36//! debug info, and enforcing edges in the scheduler, live in higher layers.
37

69/// The declarative happens-before specification, as read from a JSON file or
70/// desugared from the terse DSL. This mirrors the reviewed RFC #1146 file format
71/// verbatim so an authored file round-trips.
72#[derive(PartialEq, Eq, Debug, Clone, Serialize, Deserialize)]
73pub struct HappensBeforeSpec {
74    /// Schema version: [`HAPPENS_BEFORE_VERSION_1`] or
75    /// [`HAPPENS_BEFORE_VERSION`].
76    pub version: u32,
77
78    /// Version 2 only: how long a thread may stay held at a gate, in committed
79    /// virtual nanoseconds, before the run is refused
80    /// (`HERMIT_HB_HOLD_BUDGET_EXCEEDED`). Defaults to [`DEFAULT_MAX_HOLD_NS`].
81    #[serde(default, skip_serializing_if = "Option::is_none")]
82    pub max_hold_ns: Option<u64>,
83
84    /// Symbolic thread labels mapped to a resolution rule, so authors need not
85    /// hard-code raw `DetTid`s.
86    #[serde(default)]
87    pub threads: BTreeMap<String, ThreadSpec>,
88
89    /// Named events (anchors). Naming events separately from edges lets one event
90    /// participate in several edges and keeps the edge list readable.
91    #[serde(default)]
92    pub events: BTreeMap<String, EventSpec>,
93
94    /// The partial order itself.
95    #[serde(default)]
96    pub edges: Vec<EdgeSpec>,
97}

118/// A single named event (anchor) in the declarative format.
119///
120/// The addressing fields are flat and optional to match the RFC JSON; exactly one
121/// *position* selector must be set (see [`HappensBeforeSpec::normalize`]). The
122/// two owner-preferred primaries are [`syscalls`](Self::syscalls) ("after N
123/// syscalls") and [`rcbs`](Self::rcbs) ("after M RBCs").
124#[derive(PartialEq, Eq, Debug, Clone, Default, Serialize, Deserialize)]
125pub struct EventSpec {
126    /// The thread this event is on: a key into [`HappensBeforeSpec::threads`], or
127    /// a raw integer `DetTid`.
128    pub thread: String,
129
130    // ---- primary positions (owner's refinement) ----
131    /// After the thread has executed this many syscalls in total (any syscall).
132    #[serde(default, skip_serializing_if = "Option::is_none")]
133    pub syscalls: Option<u64>,
134
135    /// After the thread has retired this many conditional branches (its RCB
136    /// clock reaches this absolute value). Accepted as `rcbs` or `rcb`.
137    #[serde(default, alias = "rcb", skip_serializing_if = "Option::is_none")]
138    pub rcbs: Option<u64>,
139
140    // ---- code location (readability; resolves to a RIP via debug info) ----
141    /// A function name; with `line` this is "function+line", the preferred
142    /// human-legible location.
143    #[serde(default, skip_serializing_if = "Option::is_none")]
144    pub func: Option<String>,
145
146    /// A source file name (optional companion to `line`).
147    #[serde(default, skip_serializing_if = "Option::is_none")]
148    pub file: Option<String>,
149
150    /// A source line number.
151    #[serde(default, skip_serializing_if = "Option::is_none")]
152    pub line: Option<u32>,
153
154    // ---- RFC richer addressing (expressible, not led with) ----
155    /// A specific syscall by name (e.g. `"futex"`), the Nth occurrence of which
156    /// is the anchor. Combine with `phase` and `nth`.
157    #[serde(default, skip_serializing_if = "Option::is_none")]
158    pub syscall: Option<String>,
159
160    /// Which phase of the named `syscall` to anchor on.
161    #[serde(default, skip_serializing_if = "Option::is_none")]
162    pub phase: Option<PhaseSpec>,
163
164    /// Restrict the named `syscall` to calls whose first argument is this file
165    /// descriptor, for example `{"syscall": "writev", "fd": 9, "nth": 3}`: the
166    /// thread's third `writev` to fd 9. The descriptor is matched by number at
167    /// the time of the call: a write to a `dup` of fd 9 under another number
168    /// does not count, and after fd 9 is closed and the number reused, calls
169    /// on the new fd 9 do. Only valid with `syscall`, and only for
170    /// a syscall whose first argument is a file descriptor
171    /// ([`syscall_takes_fd_first`]).
172    #[serde(default, skip_serializing_if = "Option::is_none")]
173    pub fd: Option<i32>,
174
175    /// Version 2 only: restrict `"syscall": "futex"` to calls whose operation
176    /// is this value. The kernel takes `int futex_op`, so the low 32 bits of
177    /// the second argument are compared, read as a signed `int`; the upper bits
178    /// of the register are ignored. The value must fit an `i32`.
179    #[serde(default, skip_serializing_if = "Option::is_none")]
180    pub futex_op: Option<i64>,
181
182    /// Version 2 only: count this event's syscall occurrences strictly after
183    /// the entry at which the named event is reached on the same thread, so
184    /// `nth` is 1-based from there. The base entry itself never counts, even
185    /// when it matches. The base must be a count or syscall-occurrence anchor
186    /// on the same thread and may not itself use `from`.
187    #[serde(default, skip_serializing_if = "Option::is_none")]
188    pub from: Option<String>,
189
190    /// A raw instruction pointer, as a hex string like `"0x401f3c"` or decimal.
191    #[serde(default, skip_serializing_if = "Option::is_none")]
192    pub rip: Option<String>,
193
194    /// A cooperative marker name (guest-emitted). Reserved for a future backend.
195    #[serde(default, skip_serializing_if = "Option::is_none")]
196    pub mark: Option<String>,
197
198    /// Which occurrence of the addressed point (1-based). Defaults to 1. Only
199    /// meaningful for the occurrence-counted modes (`syscall`, `func`, `rip`,
200    /// `mark`); the absolute-count primaries (`syscalls`, `rcbs`) ignore it.
201    #[serde(default, skip_serializing_if = "Option::is_none")]
202    pub nth: Option<u64>,
203}

300/// The deterministic per-thread stop point that an anchor addresses.
301///
302/// All variants reduce to "a predicate over this thread's deterministic event
303/// stream plus an occurrence count," but the two leading variants
304/// ([`Position::SyscallCount`] and [`Position::Rcb`]) are absolute counts that
305/// need no per-anchor occurrence tracking.
306#[derive(PartialEq, Eq, Debug, Clone)]
307pub enum Position {
308    /// After the thread has executed exactly this many syscalls (any syscall).
309    SyscallCount(u64),
310
311    /// When the thread's RCB clock reaches this absolute value.
312    Rcb(u64),
313
314    /// The `nth` occurrence of a specific syscall on the thread, optionally
315    /// phase-qualified and restricted to one file descriptor. The occurrence is
316    /// counted per thread, over that thread's calls that match `sysno` (and
317    /// `fd`), so an anchor needs no calibration run to find a syscall count.
318    ///
319    /// Two properties to know when writing one:
320    /// - What is counted is syscall *entries*, the same unit as
321    ///   [`Position::SyscallCount`]. A matching call that a signal interrupts
322    ///   and the kernel restarts is entered, and counted, again.
323    /// - The counters belong to the thread, like its syscall count: a new
324    ///   thread starts at zero, an exec keeps them, and after a non-leader exec
325    ///   the calling thread's counters continue under the leader's thread id.
326    /// - By default it fires at the prehook: an edge then orders the BEFORE
327    ///   thread *reaching* its syscall, not that syscall's effect. The held
328    ///   thread is re-admitted at the next drain, so a BEFORE syscall that
329    ///   takes more scheduler turns than the AFTER thread needs (a `write` to
330    ///   stdout takes a checkpoint turn and a resource turn) can let the AFTER
331    ///   thread's effect land first, even under the default scheduler.
332    /// - With `phase` posthook it fires when the syscall's result is final:
333    ///   a value or an errno, `EINTR` included, decided before the guest
334    ///   resumes (<https://github.com/rrnewton/hermit/issues/3929>). An entry
335    ///   that returns an internal restart code is refused
336    ///   (`HERMIT_HB_POSTHOOK_INTERRUPTED`), because the kernel decides after
337    ///   that point whether it returns at all. A call that only a signal
338    ///   handler ends (`pause`, `rt_sigsuspend`) always returns one, so a
339    ///   posthook anchor on it always refuses. Occurrences are counted at entry
340    ///   either way, and the five syscalls that do not return on success are
341    ///   refused at load ([`syscall_never_returns_on_success`]).
342    /// - What a hard edge orders depends on both ends. A posthook BEFORE and a
343    ///   prehook AFTER order the BEFORE syscall's completed effect before the
344    ///   AFTER syscall runs. A posthook AFTER holds its thread only after its
345    ///   own call, so that call's effect may come before the BEFORE one's.
346    ///   With a prehook BEFORE, effects may land in either order.
347    /// - A posthook BEFORE on a call that only a held thread can complete
348    ///   never fires. A child wait or futex wait on the held thread ends in
349    ///   `HERMIT_HB_ANCHOR_NEVER_FIRED`; a read whose only writer is held is
350    ///   external input to the scheduler, and the run ends at `--timeout`.
351    Syscall {
352        /// The syscall number.
353        sysno: Sysno,
354        /// Restrict to a phase, or match any phase when `None`.
355        phase: Option<SyscallPhase>,
356        /// 1-based occurrence.
357        nth: u64,
358        /// Restrict to calls whose first argument is this file descriptor.
359        fd: Option<i32>,
360        /// Restrict `futex` to calls whose operation (the low 32 bits of the
361        /// second argument, as an `int`) is this value.
362        futex_op: Option<i32>,
363    },
364
365    /// The `nth` execution of the instruction at an absolute address. The address
366    /// is resolved later when it comes from a [`CodeLocation`].
367    Rip {
368        /// Absolute instruction pointer, or `None` until resolved from a
369        /// [`CodeLocation`].
370        addr: Option<u64>,
371        /// 1-based occurrence.
372        nth: u64,
373    },
374
375    /// A cooperative guest marker. Reserved for a future backend.
376    Marker {
377        /// Marker name.
378        name: String,
379        /// 1-based occurrence.
380        nth: u64,
381    },
382}
383
384impl fmt::Display for Position {
385    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
386        match self {
387            Position::SyscallCount(n) => write!(f, "after {} syscalls", n),
388            Position::Rcb(m) => write!(f, "at RCB {}", m),
389            Position::Syscall {
390                sysno,
391                phase,
392                nth,
393                fd,
394                futex_op,
395            } => {
396                write!(f, "{}", sysno.name())?;
397                if let Some(fd) = fd {
398                    write!(f, "(fd={})", fd)?;
399                }
400                if let Some(op) = futex_op {
401                    write!(f, "(op={})", op)?;
402                }
403                if let Some(p) = phase {
404                    write!(f, "@{:?}", p)?;
405                }
406                write!(f, "#{}", nth)
407            }
408            Position::Rip { addr, nth } => match addr {
409                Some(a) => write!(f, "@{:#x}#{}", a, nth),
410                None => write!(f, "@<unresolved>#{}", nth),
411            },
412            Position::Marker { name, nth } => write!(f, "mark:{}#{}", name, nth),
413        }
414    }
415}

417/// A fully normalized anchor: a named, deterministic per-thread stop point.
418#[derive(PartialEq, Eq, Debug, Clone)]
419pub struct Anchor {
420    /// The event name (map key), for diagnostics and edge references.
421    pub name: String,
422    /// The thread this anchor is on.
423    pub thread: ThreadRef,
424    /// The deterministic position selector.
425    pub position: Position,
426    /// Optional human-legible / debug-info-resolved code location.
427    pub location: CodeLocation,
428    /// For a relative anchor, the event on the same thread after which its
429    /// occurrences are counted (`from`).
430    pub from: Option<String>,
431}
432
433impl fmt::Display for Anchor {
434    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
435        write!(f, "{}[{}: {}", self.name, self.thread.label, self.position)?;
436        if let Some(base) = &self.from {
437            write!(f, " after {}", base)?;
438        }
439        if !self.location.is_empty() {
440            write!(f, " ({})", self.location)?;
441        }
442        write!(f, "]")
443    }
444}

446impl Anchor {
447    /// True for an enforced syscall-occurrence anchor: a [`Position::Syscall`]
448    /// with no phase, the prehook phase or the posthook phase. Its
449    /// occurrences are counted at the prehook of each matching syscall entry.
450    /// It fires at the prehook of the thread's `nth` matching entry, or, for
451    /// the posthook phase ([`Anchor::fires_at_posthook`]), when that entry
452    /// completes. The polling phase is not enforced.
453    pub fn is_syscall_occurrence(&self) -> bool {
454        matches!(
455            self.position,
456            Position::Syscall {
457                phase: None | Some(SyscallPhase::Prehook) | Some(SyscallPhase::Posthook),
458                ..
459            }
460        )
461    }
462
463    /// True for a syscall-occurrence anchor that fires when its syscall
464    /// completes: after the call's effect and result, before the guest
465    /// resumes (<https://github.com/rrnewton/hermit/issues/3929>). An edge
466    /// whose BEFORE anchor fires at the posthook orders that syscall's
467    /// completed effect, not just its entry.
468    pub fn fires_at_posthook(&self) -> bool {
469        matches!(
470            self.position,
471            Position::Syscall {
472                phase: Some(SyscallPhase::Posthook),
473                ..
474            }
475        )
476    }
477
478    /// The per-thread counter this anchor's occurrences are counted in: its
479    /// syscall number, fd and futex operation, or `None` for any other kind of
480    /// anchor.
481    pub fn occurrence_key(&self) -> Option<OccurrenceKey> {
482        match &self.position {
483            Position::Syscall {
484                sysno,
485                fd,
486                futex_op,
487                ..
488            } if self.is_syscall_occurrence() => Some((*sysno as usize, *fd, *futex_op)),
489            _ => None,
490        }
491    }
492
493    /// True when this is an enforced syscall-occurrence anchor and a call to
494    /// `sysno` with first two arguments `args` is one of the calls it counts.
495    pub fn syscall_occurrence_matches(&self, sysno: Sysno, args: [usize; 2]) -> bool {
496        match &self.position {
497            Position::Syscall {
498                sysno: want,
499                fd,
500                futex_op,
501                ..
502            } if self.is_syscall_occurrence() && *want == sysno => {
503                // A file descriptor and a futex operation are each an `int`:
504                // the kernel reads the low 32 bits of the register.
505                fd.is_none_or(|fd| args[0] as u32 as i32 == fd)
506                    && futex_op.is_none_or(|op| args[1] as u32 as i32 == op)
507            }
508            _ => false,
509        }
510    }
511}

513/// True for a syscall that does not return to its caller when it succeeds: the
514/// exec family replaces the image, and the exit family ends the thread. A
515/// posthook anchor on one is refused at load.
516pub fn syscall_never_returns_on_success(sysno: Sysno) -> bool {
517    matches!(
518        sysno,
519        Sysno::execve | Sysno::execveat | Sysno::exit | Sysno::exit_group | Sysno::rt_sigreturn
520    )

613/// A validated, normalized happens-before program: anchors indexed by name plus

629impl HappensBeforeProgram {
630    /// Anchors that still require debug-info resolution (an unresolved RIP from a
631    /// code location, i.e. a `func`/`line` that has not been turned into an
632    /// address yet).
633    pub fn unresolved_locations(&self) -> impl Iterator<Item = &Anchor> {
634        self.anchors.values().filter(|a| {
635            matches!(a.position, Position::Rip { addr: None, .. }) && !a.location.is_empty()
636        })
637    }
638
639    /// True when thread `dettid` must check in with the scheduler at its
640    /// syscall number `count`: some anchor sits at exactly
641    /// [`Position::SyscallCount`]`(count)` and names `dettid`, or names its
642    /// thread by spawn ordinal (which only the scheduler can resolve, so the
643    /// guest side must ask). For any other (thread, count) the scheduler's
644    /// checkpoint finds no anchor and does nothing, so skipping it changes no
645    /// gate. It costs one scheduler round trip per anchor its thread reaches,
646    /// not one per syscall (<https://github.com/rrnewton/hermit/issues/3877>),
647    /// and threads a spec does not name keep their schedule: a count another
648    /// thread passes early (QEMU's main loop makes about a million syscalls a
649    /// run) no longer adds a scheduler turn there.
650    pub fn may_have_syscall_count_anchor_at(&self, dettid: DetTid, count: u64) -> bool {
651        self.anchors.values().any(|a| {
652            matches!(a.position, Position::SyscallCount(n) if n == count)
653                && a.thread.dettid.is_none_or(|d| d == dettid)
654        })
655    }
656
657    /// True when the program has any enforced syscall-occurrence anchor
658    /// ([`Anchor::is_syscall_occurrence`]), so the guest must count its calls.
659    pub fn has_syscall_occurrence_anchors(&self) -> bool {
660        self.anchors.values().any(Anchor::is_syscall_occurrence)
661    }
662
663    /// Count one syscall entry of thread `dettid` (`sysno`, first two
664    /// arguments `args`, the thread's `entry_count`th syscall) in that
665    /// thread's `counters`, and return the names of the syscall-occurrence
666    /// anchors this entry fires.
667    ///
668    /// `counters` belongs to one thread and is kept with its other per-thread
669    /// counts (the syscall count that count anchors use), so it starts empty
670    /// for every new thread. Per entry, in this order:
671    /// 1. Every distinct occurrence key (syscall, fd, futex operation) an
672    ///    anchor names that the entry matches is incremented once; an entry
673    ///    that matches both an fd-specific and an fd-agnostic anchor counts
674    ///    toward each.
675    /// 2. Each relative anchor whose base this entry reaches gets its origin:
676    ///    its own key's count after step 1, so the base entry is never counted
677    ///    in its window. A count base is reached when `entry_count` equals its
678    ///    count, whether or not the entry matches any key.
679    /// 3. An absolute anchor fires when its key's count equals `nth`; a
680    ///    relative anchor fires when its count minus its origin equals `nth`.
681    ///
682    /// An entry is counted exactly once, when the thread enters the syscall,
683    /// so a thread held at an anchor and re-admitted does not count again. The
684    /// result names anchors on `dettid` or on a spawn ordinal; the scheduler
685    /// resolves spawn ordinals and fires only the anchors on this thread.
686    pub fn count_syscall_occurrences(
687        &self,
688        counters: &mut HbThreadCounters,
689        dettid: DetTid,
690        sysno: Sysno,
691        args: [usize; 2],
692        entry_count: u64,
693    ) -> Vec<String> {
694        let on_thread = |a: &&Anchor| a.thread.dettid.is_none_or(|d| d == dettid);
695        let keys: BTreeSet<OccurrenceKey> = self
696            .anchors
697            .values()
698            .filter(|a| a.syscall_occurrence_matches(sysno, args))
699            .filter_map(Anchor::occurrence_key)
700            .collect();
701        for key in &keys {
702            *counters.counts.entry(*key).or_insert(0) += 1;
703        }
704        for relative in self.anchors.values().filter(on_thread) {
705            let Some(base) = relative.from.as_ref().and_then(|b| self.anchors.get(b)) else {
706                continue;
707            };
708            if counters.origins.contains_key(&relative.name) {
709                continue;
710            }
711            let reached = match &base.position {
712                Position::SyscallCount(n) => *n == entry_count,
713                Position::Syscall { nth, .. } => {
714                    base.syscall_occurrence_matches(sysno, args)
715                        && base
716                            .occurrence_key()
717                            .is_some_and(|key| counters.counts.get(&key) == Some(nth))
718                }
719                _ => false,
720            };
721            if reached {
722                let origin = relative
723                    .occurrence_key()
724                    .and_then(|key| counters.counts.get(&key).copied())
725                    .unwrap_or(0);
726                counters.origins.insert(relative.name.clone(), origin);
727            }
728        }
729        if keys.is_empty() {
730            return Vec::new();
731        }
732        self.anchors
733            .values()
734            .filter(on_thread)
735            .filter(|a| a.syscall_occurrence_matches(sysno, args))
736            .filter(|a| match (&a.position, a.occurrence_key()) {
737                (Position::Syscall { nth, .. }, Some(key)) => {
738                    let count = counters.counts.get(&key).copied().unwrap_or(0);
739                    match &a.from {
740                        None => count == *nth,
741                        Some(_) => counters
742                            .origins
743                            .get(&a.name)
744                            .is_some_and(|origin| count - origin == *nth),
745                    }
746                }
747                _ => false,
748            })
749            .map(|a| a.name.clone())
750            .collect()
751    }
752
753    /// True for a version-2 program, which carries a hold budget and needs an
754    /// effective preemption timer to charge it.
755    pub fn is_version_2(&self) -> bool {
756        self.version == HAPPENS_BEFORE_VERSION
757    }
758
759    /// The relative anchors and their bases, in name order.
760    pub fn relative_anchors(&self) -> impl Iterator<Item = (&Anchor, &Anchor)> {
761        self.anchors.values().filter_map(|a| {
762            a.from
763                .as_ref()
764                .and_then(|b| self.anchors.get(b))
765                .map(|base| (a, base))
766        })
767    }
768
769    /// Anchors whose position kind the current scheduler does not yet enforce:
770    /// everything other than [`Position::SyscallCount`] and the syscall
771    /// occurrence anchors ([`Anchor::is_syscall_occurrence`]). Reported so a run
772    /// does not silently ignore an authored ordering constraint it cannot honor.
773    pub fn unenforced_positions(&self) -> impl Iterator<Item = &Anchor> {
774        self.anchors.values().filter(|a| {
775            !matches!(a.position, Position::SyscallCount(_)) && !a.is_syscall_occurrence()
776        })
777    }
778}

1065    /// Parse a JSON specification. A version-2 spec is also checked for
1066    /// fields this build does not know ([`HappensBeforeError::UnknownField`]),
1067    /// so a misspelled field is refused instead of silently ignored; version 1
1068    /// keeps its old, lenient handling of unknown fields.
1069    ///
1070    /// The spec itself is deserialized from the original text, not from the
1071    /// `Value` used for that check: a `Value` keeps only the last of two equal
1072    /// keys, which would accept `"nth": 1, "nth": 2` as `nth` 2. Deserializing
1073    /// the text refuses a duplicated field, in either version.
1074    pub fn from_json(s: &str) -> anyhow::Result<HappensBeforeSpec> {
1075        let value: serde_json::Value = serde_json::from_str(s)?;
1076        if value.get("version").and_then(serde_json::Value::as_u64)
1077            == Some(u64::from(HAPPENS_BEFORE_VERSION))
1078        {
1079            check_known_fields(&value)?;
1080        }
1081        Ok(serde_json::from_str(s)?)
1082    }
1083
1084    /// Serialize to pretty JSON.
1085    pub fn to_json(&self) -> anyhow::Result<String> {
1086        Ok(serde_json::to_string_pretty(self)?)
1087    }
1088
1089    /// Normalize and statically validate: check the version, resolve thread and
1090    /// event references, require exactly one position per event, parse syscalls
1091    /// and RIPs, and confirm the edge graph is acyclic.
1092    pub fn normalize(&self) -> Result<HappensBeforeProgram, HappensBeforeError> {
1093        if self.version != HAPPENS_BEFORE_VERSION && self.version != HAPPENS_BEFORE_VERSION_1 {
1094            return Err(HappensBeforeError::UnsupportedVersion(self.version));
1095        }
1096        if self.version == HAPPENS_BEFORE_VERSION_1 {
1097            self.refuse_version_2_fields()?;
1098        }
1099
1100        let mut anchors = BTreeMap::new();
1101        for (name, ev) in &self.events {
1102            anchors.insert(name.clone(), self.normalize_event(name, ev)?);
1103        }
1104        check_relative_bases(&anchors)?;
1105
1106        // Resolve edges against the anchor table.
1107        let mut edges = Vec::with_capacity(self.edges.len());
1108        for e in &self.edges {
1109            if !anchors.contains_key(&e.before) {
1110                return Err(HappensBeforeError::UnknownEvent {
1111                    which: "before".to_string(),
1112                    name: e.before.clone(),
1113                });
1114            }
1115            if !anchors.contains_key(&e.after) {
1116                return Err(HappensBeforeError::UnknownEvent {
1117                    which: "after".to_string(),
1118                    name: e.after.clone(),
1119                });
1120            }
1121            edges.push(HappensBeforeEdge {
1122                before: e.before.clone(),
1123                after: e.after.clone(),
1124                strength: e.strength,
1125            });
1126        }
1127
1128        detect_cycle(&anchors, &edges)?;
1129        // A relative event is reached only after its base, on the same thread:
1130        // an activation edge base -> relative. With the hard edges, a cycle
1131        // means some event could only be reached after itself.
1132        let mut with_activation = edges.clone();
1133        with_activation.extend(
1134            anchors
1135                .values()
1136                .filter_map(|a| a.from.as_ref().map(|base| (base, a)))
1137                .map(|(base, a)| HappensBeforeEdge {
1138                    before: base.clone(),
1139                    after: a.name.clone(),
1140                    strength: Strength::Hard,
1141                }),
1142        );
1143        with_activation.retain(|e| e.strength == Strength::Hard);
1144        if let Err(HappensBeforeError::Cycle(names)) = detect_cycle(&anchors, &with_activation) {
1145            return Err(HappensBeforeError::FromCycle(names));
1146        }
1147
1148        let max_hold_ns = (self.version == HAPPENS_BEFORE_VERSION)
1149            .then(|| self.max_hold_ns.unwrap_or(DEFAULT_MAX_HOLD_NS));
1150        Ok(HappensBeforeProgram {
1151            version: self.version,
1152            anchors,
1153            edges,
1154            max_hold_ns,
1155        })
1156    }

1158    /// Refuse, in a version-1 spec, every field that version 2 added.
1159    fn refuse_version_2_fields(&self) -> Result<(), HappensBeforeError> {
1160        if self.max_hold_ns.is_some() {
1161            return Err(HappensBeforeError::NewFieldInVersion1 {
1162                event: None,
1163                field: "max_hold_ns".to_owned(),
1164            });
1165        }
1166        for (name, ev) in &self.events {
1167            for (field, set) in [
1168                ("from", ev.from.is_some()),
1169                ("futex_op", ev.futex_op.is_some()),
1170            ] {
1171                if set {
1172                    return Err(HappensBeforeError::NewFieldInVersion1 {
1173                        event: Some(name.clone()),
1174                        field: field.to_owned(),
1175                    });
1176                }
1177            }
1178        }
1179        Ok(())
1180    }

1182    /// Resolve one event into a normalized [`Anchor`].
1183    fn normalize_event(&self, name: &str, ev: &EventSpec) -> Result<Anchor, HappensBeforeError> {
1184        let thread = self.resolve_thread(name, &ev.thread)?;
1185
1186        // A code location can accompany any position; it also *supplies* a RIP
1187        // position when no explicit position selector is present.
1188        let location = CodeLocation {
1189            function: ev.func.clone(),
1190            file: ev.file.clone(),
1191            line: ev.line,
1192        };
1193
1194        // Determine which *explicit* position selectors are present. A code
1195        // location (`func`/`file`/`line`) is descriptive and may accompany any
1196        // one of these — the owner's primary anchor is "function foo on thread T
1197        // after N syscalls / M RBCs", i.e. a code location *and* a count. The
1198        // code location only *becomes* the (deferred RIP) position when no
1199        // explicit selector is present at all.
1200        let mut found: Vec<&str> = Vec::new();
1201        if ev.syscalls.is_some() {
1202            found.push("syscalls");
1203        }
1204        if ev.rcbs.is_some() {
1205            found.push("rcbs");
1206        }
1207        if ev.syscall.is_some() {
1208            found.push("syscall");
1209        }
1210        if ev.rip.is_some() {
1211            found.push("rip");
1212        }
1213        if ev.mark.is_some() {
1214            found.push("mark");
1215        }
1216        let has_code_location = !location.is_empty();
1217
1218        // Reject multiple explicit selectors outright. A single explicit
1219        // selector wins as the position (code location stays descriptive). Zero
1220        // explicit selectors is only valid when a code location supplies a RIP.
1221        if found.len() > 1 {
1222            return Err(HappensBeforeError::AmbiguousPosition {
1223                event: name.to_string(),
1224                found: found.iter().map(|s| s.to_string()).collect(),
1225            });
1226        }
1227        if found.is_empty() && !has_code_location {
1228            return Err(HappensBeforeError::AmbiguousPosition {
1229                event: name.to_string(),
1230                found: Vec::new(),
1231            });
1232        }
1233
1234        if ev.fd.is_some() && ev.syscall.is_none() {
1235            return Err(HappensBeforeError::FdWithoutSyscall {
1236                event: name.to_string(),
1237            });
1238        }
1239        if ev.futex_op.is_some() && ev.syscall.is_none() {
1240            return Err(HappensBeforeError::FutexOpWithoutFutex {
1241                event: name.to_string(),
1242            });
1243        }
1244
1245        if ev.nth == Some(0) && ev.syscalls.is_none() && ev.rcbs.is_none() {
1246            return Err(HappensBeforeError::NthZero {
1247                event: name.to_string(),
1248            });
1249        }
1250
1251        let nth = ev.nth.unwrap_or(1);
1252        let position = if let Some(n) = ev.syscalls {
1253            Position::SyscallCount(n)
1254        } else if let Some(m) = ev.rcbs {
1255            Position::Rcb(m)
1256        } else if let Some(sc) = &ev.syscall {
1257            let sysno = Sysno::from_str(sc).map_err(|_| HappensBeforeError::UnknownSyscall {
1258                event: name.to_string(),
1259                name: sc.clone(),
1260            })?;
1261            if ev.fd.is_some() && !syscall_takes_fd_first(sysno) {
1262                return Err(HappensBeforeError::FdNotFirstArgument {
1263                    event: name.to_string(),
1264                    syscall: sc.clone(),
1265                });
1266            }
1267            let futex_op = match ev.futex_op {
1268                None => None,
1269                Some(_) if sysno != Sysno::futex => {
1270                    return Err(HappensBeforeError::FutexOpWithoutFutex {
1271                        event: name.to_string(),
1272                    });
1273                }
1274                Some(value) => Some(i32::try_from(value).map_err(|_| {
1275                    HappensBeforeError::FutexOpOutOfRange {
1276                        event: name.to_string(),
1277                        value,
1278                    }
1279                })?),
1280            };
1281            if ev.phase == Some(PhaseSpec::Posthook) && syscall_never_returns_on_success(sysno) {
1282                return Err(HappensBeforeError::PosthookOnNoReturnSyscall {
1283                    event: name.to_string(),
1284                    syscall: sc.clone(),
1285                });
1286            }
1287            Position::Syscall {
1288                sysno,
1289                phase: ev.phase.map(Into::into),
1290                nth,
1291                fd: ev.fd,
1292                futex_op,
1293            }
1294        } else if let Some(rip) = &ev.rip {
1295            let addr = parse_rip(rip).ok_or_else(|| HappensBeforeError::BadRip {
1296                event: name.to_string(),
1297                text: rip.clone(),
1298            })?;
1299            Position::Rip {
1300                addr: Some(addr),
1301                nth,
1302            }
1303        } else if let Some(mark) = &ev.mark {
1304            Position::Marker {
1305                name: mark.clone(),
1306                nth,
1307            }
1308        } else {
1309            // Code-location-only: a RIP to be resolved later from debug info.
1310            debug_assert!(has_code_location);
1311            Position::Rip { addr: None, nth }
1312        };
1313
1314        Ok(Anchor {
1315            name: name.to_string(),
1316            thread,
1317            position,
1318            location,
1319            from: ev.from.clone(),
1320        })
1321    }
1322
1323    /// Resolve an event's `thread` field to a [`ThreadRef`], consulting the
1324    /// `threads` table and falling back to a raw integer id.
1325    fn resolve_thread(&self, event: &str, thread: &str) -> Result<ThreadRef, HappensBeforeError> {
1326        if let Some(spec) = self.threads.get(thread) {
1327            Ok(ThreadRef {
1328                label: spec.label.clone().unwrap_or_else(|| thread.to_string()),
1329                dettid: spec.dettid.map(DetTid::from_raw),
1330                spawn_ordinal: spec.spawn_ordinal,
1331            })
1332        } else if let Ok(raw) = thread.parse::<i32>() {
1333            Ok(ThreadRef {
1334                label: thread.to_string(),
1335                dettid: Some(DetTid::from_raw(raw)),
1336                spawn_ordinal: None,
1337            })
1338        } else {
1339            Err(HappensBeforeError::UnknownThread {
1340                event: event.to_string(),
1341                thread: thread.to_string(),
1342            })
1343        }
1344    }
1345}

1347/// Check every relative anchor's `from`: the base exists, is on the same
1348/// thread, is a count or syscall-occurrence anchor and is not itself relative;
1349/// the relative anchor is a syscall-occurrence anchor.
1350fn check_relative_bases(anchors: &BTreeMap<String, Anchor>) -> Result<(), HappensBeforeError> {
1351    for anchor in anchors.values() {
1352        let Some(base_name) = &anchor.from else {
1353            continue;
1354        };
1355        let event = anchor.name.clone();
1356        let base = base_name.clone();
1357        let Some(base_anchor) = anchors.get(base_name) else {
1358            return Err(HappensBeforeError::FromUnknownEvent { event, base });
1359        };
1360        if !anchor.is_syscall_occurrence() {
1361            return Err(HappensBeforeError::FromOnNonOccurrence { event });
1362        }
1363        if !same_thread(&base_anchor.thread, &anchor.thread) {
1364            return Err(HappensBeforeError::FromOtherThread { event, base });
1365        }
1366        if base_anchor.from.is_some() {
1367            return Err(HappensBeforeError::FromRelativeBase { event, base });
1368        }
1369        if !matches!(base_anchor.position, Position::SyscallCount(_))
1370            && !base_anchor.is_syscall_occurrence()
1371        {
1372            return Err(HappensBeforeError::FromUnenforcedPosition { event, base });
1373        }
1374    }
1375    Ok(())
1376}

1378/// True when two thread references resolve to the same thread whatever they
1379/// are called: the same `dettid`, or, with no `dettid` on either, the same
1380/// `spawn_ordinal`. A `dettid` and a `spawn_ordinal` cannot be shown equal
1381/// before the run, so they are not.
1382fn same_thread(a: &ThreadRef, b: &ThreadRef) -> bool {
1383    match (a.dettid, b.dettid) {
1384        (Some(x), Some(y)) => x == y,
1385        (None, None) => a.spawn_ordinal.is_some() && a.spawn_ordinal == b.spawn_ordinal,
1386        _ => false,
1387    }

1390/// Refuse a field a version-2 spec does not define. The field lists mirror
1391/// the serde structs above, including the `rcb` alias.
1392fn check_known_fields(value: &serde_json::Value) -> Result<(), HappensBeforeError> {
1393    const SPEC: &[&str] = &["version", "max_hold_ns", "threads", "events", "edges"];
1394    const THREAD: &[&str] = &["label", "dettid", "spawn_ordinal"];
1395    const EVENT: &[&str] = &[
1396        "thread", "syscalls", "rcbs", "rcb", "func", "file", "line", "syscall", "phase", "fd",
1397        "futex_op", "from", "rip", "mark", "nth",
1398    ];
1399    const EDGE: &[&str] = &["before", "after", "strength"];
1400    fn check(
1401        object: Option<&serde_json::Map<String, serde_json::Value>>,
1402        known: &[&str],
1403        place: impl Fn() -> String,
1404    ) -> Result<(), HappensBeforeError> {
1405        for field in object.into_iter().flat_map(|o| o.keys()) {
1406            if !known.contains(&field.as_str()) {
1407                return Err(HappensBeforeError::UnknownField {
1408                    place: place(),
1409                    field: field.clone(),
1410                });
1411            }
1412        }
1413        Ok(())
1414    }
1415    check(value.as_object(), SPEC, || "the spec".to_owned())?;
1416    for (name, thread) in value
1417        .get("threads")
1418        .and_then(serde_json::Value::as_object)
1419        .into_iter()
1420        .flatten()
1421    {
1422        check(thread.as_object(), THREAD, || format!("thread '{name}'"))?;
1423    }
1424    for (name, event) in value
1425        .get("events")
1426        .and_then(serde_json::Value::as_object)
1427        .into_iter()
1428        .flatten()
1429    {
1430        check(event.as_object(), EVENT, || format!("event '{name}'"))?;
1431    }
1432    for (index, edge) in value
1433        .get("edges")
1434        .and_then(serde_json::Value::as_array)
1435        .into_iter()
1436        .flatten()
1437        .enumerate()
1438    {
1439        check(edge.as_object(), EDGE, || format!("edge {index}"))?;
1440    }
1441    Ok(())
1442}

1454/// Detect a cycle in the edge graph via depth-first search, returning the cycle
1455/// path if one exists. Anchors are visited in name order for determinism.
1456fn detect_cycle(
1457    anchors: &BTreeMap<String, Anchor>,
1458    edges: &[HappensBeforeEdge],
1459) -> Result<(), HappensBeforeError> {
1460    // Adjacency: before -> [after...]
1461    let mut adj: BTreeMap<&str, Vec<&str>> = BTreeMap::new();
1462    for name in anchors.keys() {
1463        adj.entry(name.as_str()).or_default();
1464    }
1465    for e in edges {
1466        adj.entry(e.before.as_str())
1467            .or_default()
1468            .push(e.after.as_str());
1469    }
1470
1471    #[derive(Clone, Copy, PartialEq)]
1472    enum Mark {
1473        Visiting,
1474        Done,
1475    }
1476    let mut state: BTreeMap<&str, Mark> = BTreeMap::new();
1477
1478    // Iterative DFS to avoid stack overflow on deep chains, tracking the current
1479    // path so we can report a concrete cycle.
1480    for root in adj.keys().copied() {
1481        if state.contains_key(root) {
1482            continue;
1483        }
1484        // Stack of (node, index of next neighbor to visit).
1485        let mut stack: Vec<(&str, usize)> = vec![(root, 0)];
1486        let mut path: Vec<&str> = vec![root];
1487        state.insert(root, Mark::Visiting);
1488
1489        while let Some(&mut (node, ref mut idx)) = stack.last_mut() {
1490            let neighbors = &adj[node];
1491            if *idx < neighbors.len() {
1492                let next = neighbors[*idx];
1493                *idx += 1;
1494                match state.get(next) {
1495                    Some(Mark::Visiting) => {
1496                        // Found a back-edge: assemble the cycle from `path`.
1497                        let start = path.iter().position(|&n| n == next).unwrap_or(0);
1498                        let mut cycle: Vec<String> =
1499                            path[start..].iter().map(|s| s.to_string()).collect();
1500                        cycle.push(next.to_string());
1501                        return Err(HappensBeforeError::Cycle(cycle));
1502                    }
1503                    Some(Mark::Done) => {}
1504                    None => {
1505                        state.insert(next, Mark::Visiting);
1506                        path.push(next);
1507                        stack.push((next, 0));
1508                    }
1509                }
1510            } else {
1511                state.insert(node, Mark::Done);
1512                stack.pop();
1513                path.pop();
1514            }
1515        }
1516    }
1517    Ok(())
1518}

1520// ================================================================================
1521// Terse DSL
1522// ================================================================================
1523//
1524// One edge per non-empty, non-comment line:
1525//
1526//     writer:free_buffer#342  <  reader:read_buffer#97
1527//     writer:futex@post#5     <  reader:@0x401f3c#1
1528//     A:rcb=123456            <  B:sc=97
1529//
1530// Each side is `thread:anchor[#ordinal]`. The anchor token is one of:
1531//   * `name`            -> function name (code location)
1532//   * `@0xADDR`         -> raw RIP
1533//   * `syscall@phase`   -> a named syscall, optional `@pre`/`@post`/`@polling`
1534//   * `rcb=M`           -> after M RBCs (owner primary)
1535//   * `sc=N`            -> after N syscalls (owner primary)
1536// A trailing `#N` sets the occurrence ordinal (ignored by `rcb=`/`sc=`).
1537// A `!soft` suffix on the line marks the edge soft; default is hard.
1538
1539impl HappensBeforeSpec {
1540    /// Parse the terse line-oriented DSL into a specification. Symbolic threads
1541    /// mentioned by name become entries in the `threads` table.
1542    pub fn from_dsl(input: &str) -> Result<HappensBeforeSpec, HappensBeforeError> {
1543        // The DSL has no version-2 syntax, so it builds a version-1 spec.
1544        let mut spec = HappensBeforeSpec {
1545            version: HAPPENS_BEFORE_VERSION_1,
1546            max_hold_ns: None,
1547            threads: BTreeMap::new(),
1548            events: BTreeMap::new(),
1549            edges: Vec::new(),
1550        };
1551        let mut seen_names: BTreeSet<String> = BTreeSet::new();
1552
1553        for (i, raw_line) in input.lines().enumerate() {
1554            let lineno = i + 1;
1555            let line = strip_comment(raw_line).trim();
1556            if line.is_empty() {
1557                continue;
1558            }
1559
1560            // Optional trailing "!soft" / "!hard".
1561            let (body, strength) = if let Some(b) = line.strip_suffix("!soft") {
1562                (b.trim(), Strength::Soft)
1563            } else if let Some(b) = line.strip_suffix("!hard") {
1564                (b.trim(), Strength::Hard)
1565            } else {
1566                (line, Strength::Hard)
1567            };
1568
1569            let (lhs, rhs) = body
1570                .split_once('<')
1571                .ok_or_else(|| HappensBeforeError::DslSyntax {
1572                    line: lineno,
1573                    message: "expected '<' separating two events".to_string(),
1574                })?;
1575
1576            let before = parse_dsl_side(lhs.trim(), lineno, &mut spec, &mut seen_names)?;
1577            let after = parse_dsl_side(rhs.trim(), lineno, &mut spec, &mut seen_names)?;
1578            spec.edges.push(EdgeSpec {
1579                before,
1580                after,
1581                strength,
1582            });
1583        }
1584        Ok(spec)
1585    }
1586}

1588/// Strip a `#`-or-`//` comment, but not a `#ordinal` that is part of a token.
1589/// We treat `//` as the only comment marker to avoid clashing with `#ordinal`.
1590fn strip_comment(line: &str) -> &str {
1591    match line.find("//") {
1592        Some(idx) => &line[..idx],
1593        None => line,
1594    }
1595}

1597/// Parse one side of a DSL edge, registering the event and thread in `spec`, and
1598/// returning the generated event name.
1599fn parse_dsl_side(
1600    token: &str,
1601    lineno: usize,
1602    spec: &mut HappensBeforeSpec,
1603    seen_names: &mut BTreeSet<String>,
1604) -> Result<String, HappensBeforeError> {
1605    let (thread, anchor) = token
1606        .split_once(':')
1607        .ok_or_else(|| HappensBeforeError::DslSyntax {
1608            line: lineno,
1609            message: format!("expected 'thread:anchor' in '{}'", token),
1610        })?;
1611    let thread = thread.trim();
1612    let anchor = anchor.trim();
1613    if thread.is_empty() || anchor.is_empty() {
1614        return Err(HappensBeforeError::DslSyntax {
1615            line: lineno,
1616            message: format!("empty thread or anchor in '{}'", token),
1617        });
1618    }
1619
1620    // Split a trailing "#ordinal".
1621    let (anchor_body, nth) = match anchor.split_once('#') {
1622        Some((a, n)) => {
1623            let parsed = n
1624                .parse::<u64>()
1625                .map_err(|_| HappensBeforeError::DslSyntax {
1626                    line: lineno,
1627                    message: format!("bad ordinal '#{}'", n),
1628                })?;
1629            (a.trim(), Some(parsed))
1630        }
1631        None => (anchor, None),
1632    };
1633
1634    let mut ev = EventSpec {
1635        thread: thread.to_string(),
1636        nth,
1637        ..Default::default()
1638    };
1639
1640    if let Some(rest) = anchor_body.strip_prefix('@') {
1641        // raw rip: @0x...
1642        ev.rip = Some(rest.to_string());
1643    } else if let Some(m) = anchor_body.strip_prefix("rcb=") {
1644        ev.rcbs = Some(
1645            m.parse::<u64>()
1646                .map_err(|_| HappensBeforeError::DslSyntax {
1647                    line: lineno,
1648                    message: format!("bad rcb count '{}'", m),
1649                })?,
1650        );
1651        ev.nth = None;
1652    } else if let Some(n) = anchor_body.strip_prefix("sc=") {
1653        ev.syscalls = Some(
1654            n.parse::<u64>()
1655                .map_err(|_| HappensBeforeError::DslSyntax {
1656                    line: lineno,
1657                    message: format!("bad syscall count '{}'", n),
1658                })?,
1659        );
1660        ev.nth = None;
1661    } else if let Some((sc, phase)) = anchor_body.split_once('@') {
1662        // syscall@phase
1663        ev.syscall = Some(sc.to_string());
1664        ev.phase = Some(parse_dsl_phase(phase, lineno)?);
1665    } else if is_syscall_name(anchor_body) {
1666        // bare syscall name
1667        ev.syscall = Some(anchor_body.to_string());
1668    } else {
1669        // function name (code location)
1670        ev.func = Some(anchor_body.to_string());
1671    }
1672
1673    // Generate a stable, unique event name from the token.
1674    let base = sanitize_name(token);
1675    let mut ev_name = base.clone();
1676    let mut suffix = 1;
1677    while seen_names.contains(&ev_name) && spec.events.get(&ev_name) != Some(&ev) {
1678        suffix += 1;
1679        ev_name = format!("{}_{}", base, suffix);
1680    }
1681    seen_names.insert(ev_name.clone());
1682    spec.events.entry(ev_name.clone()).or_insert(ev);
1683
1684    // Register the thread label if not already present and not a raw id.
1685    if thread.parse::<i32>().is_err() {
1686        spec.threads
1687            .entry(thread.to_string())
1688            .or_insert(ThreadSpec {
1689                label: Some(thread.to_string()),
1690                dettid: None,
1691                spawn_ordinal: None,
1692            });
1693    }
1694
1695    Ok(ev_name)
1696}

1698fn parse_dsl_phase(phase: &str, lineno: usize) -> Result<PhaseSpec, HappensBeforeError> {
1699    match phase.trim().to_ascii_lowercase().as_str() {
1700        "pre" | "prehook" => Ok(PhaseSpec::Prehook),
1701        "post" | "posthook" => Ok(PhaseSpec::Posthook),
1702        "poll" | "polling" => Ok(PhaseSpec::Polling),
1703        other => Err(HappensBeforeError::DslSyntax {
1704            line: lineno,
1705            message: format!("unknown syscall phase '{}'", other),
1706        }),
1707    }
1708}

1715/// Turn a DSL token into a valid, readable event-name slug.
1716fn sanitize_name(token: &str) -> String {
1717    let mut out = String::with_capacity(token.len());
1718    for ch in token.chars() {
1719        if ch.is_ascii_alphanumeric() || ch == '_' {
1720            out.push(ch);
1721        } else {
1722            out.push('_');
1723        }
1724    }
1725    if out.is_empty() {
1726        out.push('e');
1727    }
1728    out
1729}

2544    /// The design's per-entry order: increment, then record origins, then
2545    /// fire. Absolute and relative anchors on one matcher share its counter;

2555                "after_m2": {"thread": "7", "syscall": "getppid", "from": "m2", "nth": 1},
2556                "c4": {"thread": "7", "syscalls": 4},
2557                "after_c4": {"thread": "7", "syscall": "getppid", "from": "c4", "nth": 2},

2559                "after_gp1": {"thread": "7", "syscall": "getppid", "from": "gp1", "nth": 1}}}"#,
2560        )

2571            relative,
2572            [("after_c4", "c4"), ("after_gp1", "gp1"), ("after_m2", "m2")]
