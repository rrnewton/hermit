9//! Regression guest for the Landlock sandbox syscalls (KVM ratchet round 14).
10//!
11//! `landlock_create_ruleset`, `landlock_add_rule`, and `landlock_restrict_self`
12//! were classified Unsupported, which fail-closes under `--strict` and aborts
13//! any program that probes or installs a Landlock sandbox. Landlock is an LSM
14//! whose availability and ABI version depend on the host kernel build
15//! (`CONFIG_SECURITY_LANDLOCK`) and runtime LSM stacking, so forwarding these to
16//! the host is nondeterministic. Detcore now determinizes them to a fixed
17//! `ENOSYS` -- the errno a kernel built without Landlock returns -- so a guest
18//! sees a consistent "sandbox unavailable" answer independent of the host.
19//!
20//! glibc does not wrap these syscalls, so this guest issues them raw via
21//! `libc::syscall`. It must observe `ENOSYS` for each. Runs deterministically
22//! under `hermit run --strict --verify`.
23
24use std::io::Error;
25
26/// Issue a raw syscall and return the errno it failed with (asserting it did
27/// fail with -1, which is how `libc::syscall` reports a negative kernel return).
28fn expect_enosys(name: &str, ret: libc::c_long) {
29    assert_eq!(ret, -1, "{name} should fail (-1) under hermit, got {ret}");
30    let errno = Error::last_os_error().raw_os_error().unwrap();
31    assert_eq!(
32        errno,
33        libc::ENOSYS,
34        "{name} should be determinized to ENOSYS, got errno {errno}"
35    );
36}
37
38fn main() {
39    // 1. The canonical Landlock ABI-version probe:
40    //    landlock_create_ruleset(NULL, 0, LANDLOCK_CREATE_RULESET_VERSION).
41    //    On a Landlock kernel this returns the ABI version (>0); Detcore
42    //    determinizes it to ENOSYS.
43    const LANDLOCK_CREATE_RULESET_VERSION: u32 = 1;
44    let ver = unsafe {
45        libc::syscall(
46            libc::SYS_landlock_create_ruleset,
47            std::ptr::null::<libc::c_void>(),
48            0usize,
49            LANDLOCK_CREATE_RULESET_VERSION,
50        )
51    };
52    expect_enosys("landlock_create_ruleset(version probe)", ver);
53
54    // 2. landlock_add_rule against a bogus ruleset fd: must also refuse ENOSYS
55    //    before any argument validation (the syscall itself is unavailable).
56    let add = unsafe {
57        libc::syscall(
58            libc::SYS_landlock_add_rule,
59            -1i32,
60            0u32,
61            std::ptr::null::<libc::c_void>(),
62            0u32,
63        )
64    };
65    expect_enosys("landlock_add_rule", add);
66
67    // 3. landlock_restrict_self against a bogus ruleset fd: same deterministic
68    //    ENOSYS.
69    let restrict = unsafe { libc::syscall(libc::SYS_landlock_restrict_self, -1i32, 0u32) };
70    expect_enosys("landlock_restrict_self", restrict);
71
72    println!("Landlock syscalls determinized to ENOSYS. Test complete.");
73}
