9//! Regression guest for shutdown(2) (KVM ratchet round 12).
10//!
11//! shutdown is the lone remaining member of the socket family that was
12//! classified Unsupported (fail-closing under --strict) while all of its
13//! siblings were Determinized. This guest exercises SHUT_WR followed by a read
14//! that must observe EOF, proving Detcore forwards shutdown and its half-close
15//! effect is visible to the peer. Runs deterministically under
16//! `hermit run --strict --verify`.
17
18use std::os::fd::AsRawFd;
19
20use nix::sys::socket::AddressFamily;
21use nix::sys::socket::SockFlag;
22use nix::sys::socket::SockType;
23use nix::sys::socket::socketpair;
24
25fn main() {
26    if matches!(std::env::var("HERMIT_MODE"), Ok(mode) if mode == "record") {
27        // Record mode currently hangs for socketpair-based guests; mirror the
28        // sibling socketpair test's skip.
29        eprintln!("Skipping test in record mode.");
30        return;
31    }
32
33    let (sock1, sock2) = socketpair(
34        AddressFamily::Unix,
35        SockType::Stream,
36        None,
37        SockFlag::empty(),
38    )
39    .unwrap();
40
41    // Normal write/read across the pair works before shutdown.
42    nix::unistd::write(&sock1, b"Hello").unwrap();
43    let mut buf = [0u8; 5];
44    let n = nix::unistd::read(&sock2, &mut buf).unwrap();
45    assert_eq!(n, 5);
46    assert_eq!(&buf[..], b"Hello");
47
48    // Half-close the write direction of sock1. This is the syscall under test
49    // (glibc's shutdown wrapper issues SYS_shutdown directly).
50    let rc = unsafe { libc::shutdown(sock1.as_raw_fd(), libc::SHUT_WR) };
51    assert_eq!(rc, 0, "shutdown(SHUT_WR) should succeed, got errno");
52
53    // After SHUT_WR on sock1, the peer sock2 must observe EOF (0 bytes).
54    let mut buf2 = [0u8; 8];
55    let eof = nix::unistd::read(&sock2, &mut buf2).unwrap();
56    assert_eq!(eof, 0, "peer read after SHUT_WR must return EOF");
57
58    // A full shutdown of the remaining socket is also accepted deterministically.
59    let rc2 = unsafe { libc::shutdown(sock2.as_raw_fd(), libc::SHUT_RDWR) };
60    assert_eq!(rc2, 0, "shutdown(SHUT_RDWR) should succeed, got errno");
61
62    println!("shutdown(2) half-close observed EOF. Test complete.");
63}
