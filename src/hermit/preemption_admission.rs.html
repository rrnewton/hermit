9//! Refuse timer requests that the selected runtime cannot implement.
10//!
11//! This is launch policy, not a timer implementation or a replacement for
12//! Detcore's scheduling policy. Cooperative scheduling at existing guest
13//! events remains available without a preemption timer.
14
15use detcore::Config;
16
17use crate::Backend;
18
19/// A named, pre-guest refusal of an unsupported preemption setting.
20#[derive(Debug, Clone, PartialEq, Eq)]
21pub struct UnsupportedPreemption {
22    backend: Backend,
23    setting: &'static str,
24}
25
26impl std::fmt::Display for UnsupportedPreemption {
27    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
28        write!(
29            f,
30            "unsupported_preemption: --backend={} refuses {}: this runtime cannot deliver \
31             Detcore preemption timer events; the guest has not started",
32            self.backend.as_str(),
33            self.setting,
34        )?;
35        if self.setting == "--max-timeslice" {
36            write!(
37                f,
38                "; to request execution without that bound, pass --max-timeslice=disabled"
39            )?;
40        }
41        Ok(())
42    }
43}
44
45impl std::error::Error for UnsupportedPreemption {}
46
47/// Check command-line requests before launching a guest or consuming input.
48/// `maximum_requested` records parser provenance, including config-file and
49/// `--from` inputs. The inherited maximum is not a timer request: it also
50/// controls shared clock accounting and cooperative scheduling, and stays
51/// unchanged. `skid_margin_requested` represents the CLI-only PMU setting.
52///
53/// This check never rewrites the configuration. Library APIs without command-
54/// line provenance retain their existing admission behavior.
55/// A schedule-event recording or cooperative target alone does not request a
56/// timer. Preemption-point replay does, and is refused without inspecting or
57/// silently reducing the requested recording to a cooperative subset.
58pub fn refuse_unsupported_preemption(
59    config: &Config,
60    backend: Backend,
61    maximum_requested: bool,
62    skid_margin_requested: bool,
63) -> Result<(), UnsupportedPreemption> {
64    if crate::backend_delivers_preemption_timer(backend) {
65        return Ok(());
66    }
67    let setting = if maximum_requested && config.max_timeslice.is_some() {
68        "--max-timeslice"
69    } else if config.imprecise_timers {
70        "--imprecise-timers"
71    } else if !config.interrupt_at.is_empty() {
72        "--interrupt-at"
73    } else if config.preemption_stacktrace {
74        "--preemption-stacktrace"
75    } else if config.preemption_stacktrace_log_file.is_some() {
76        "--preemption-stacktrace-log-file"
77    } else if config.panic_on_rcb_overshoot {
78        "--panic-on-rbc-overshoot (--panic-on-rcb-overshoot)"
79    } else if skid_margin_requested {
80        "--skid-margin"
81    } else if config.replay_preemptions_from.is_some() {
82        "--replay-preemptions-from"
83    } else {
84        return Ok(());
85    };
86    Err(UnsupportedPreemption { backend, setting })
87}

89#[cfg(test)]
90mod tests {
91    use clap::Parser;
92
93    use super::*;
94
95    const TIMERLESS: [Backend; 5] = [
96        Backend::Dbt,
97        Backend::Sabre,
98        Backend::Liteinst,
99        Backend::InGuestTrap,
100        Backend::Kvm,
101    ];
102
103    #[test]
104    fn explicit_active_maximum_is_refused_without_normalizing_the_request() {
105        for args in [
106            vec!["detcore"],
107            vec!["detcore", "--max-timeslice=1000"],
108            vec!["detcore", "--preemption-timeout=1000"],
109            vec!["detcore", "--no-rcb-time"],
110        ] {
111            let config = Config::parse_from(args);
112            let before = serde_json::to_value(&config).unwrap();
113            for backend in TIMERLESS {
114                let error =
115                    refuse_unsupported_preemption(&config, backend, true, false).unwrap_err();
116                assert_eq!(error.setting, "--max-timeslice");
117                let message = error.to_string();
118                assert!(message.contains(backend.as_str()), "{message}");
119                assert!(message.contains("unsupported_preemption"), "{message}");
120                assert!(
121                    message.contains("pass --max-timeslice=disabled"),
122                    "{message}"
123                );
124                assert_eq!(serde_json::to_value(&config).unwrap(), before);
125            }
126            refuse_unsupported_preemption(&config, Backend::Ptrace, true, false).unwrap();
127        }
128    }
129
130    #[test]
131    fn explicit_timer_controls_still_refuse_after_disabling_the_maximum() {
132        for (args, setting) in [
133            (vec!["--imprecise-timers"], "--imprecise-timers"),
134            (vec!["--interrupt-at=3:1000"], "--interrupt-at"),
135            (vec!["--preemption-stacktrace"], "--preemption-stacktrace"),
136            (
137                vec!["--preemption-stacktrace-log-file=unused"],
138                "--preemption-stacktrace-log-file",
139            ),
140            (
141                vec!["--panic-on-rbc-overshoot"],
142                "--panic-on-rbc-overshoot (--panic-on-rcb-overshoot)",
143            ),
144            (
145                vec!["--panic-on-rcb-overshoot"],
146                "--panic-on-rbc-overshoot (--panic-on-rcb-overshoot)",
147            ),
148            (
149                vec!["--replay-preemptions-from=unused"],
150                "--replay-preemptions-from",
151            ),
152        ] {
153            let config = Config::parse_from(
154                ["detcore", "--max-timeslice=disabled"]
155                    .into_iter()
156                    .chain(args),
157            );
158            for backend in TIMERLESS {
159                let error =
160                    refuse_unsupported_preemption(&config, backend, true, false).unwrap_err();
161                assert_eq!(error.setting, setting);
162            }
163            refuse_unsupported_preemption(&config, Backend::Ptrace, true, false).unwrap();
164        }
165        let config = Config::parse_from(["detcore", "--max-timeslice=disabled"]);
166        for backend in TIMERLESS {
167            assert_eq!(
168                refuse_unsupported_preemption(&config, backend, true, true)
169                    .unwrap_err()
170                    .setting,
171                "--skid-margin",
172            );
173        }
174        refuse_unsupported_preemption(&config, Backend::Ptrace, true, true).unwrap();
175    }

177    #[test]
178    fn cooperative_and_boundary_event_settings_do_not_require_a_timer() {
179        for disabled in [
180            "--max-timeslice=disabled",
181            "--max-timeslice=0",
182            "--preemption-timeout=0",
183        ] {
184            let config = Config::parse_from([
185                "detcore",
186                disabled,
187                "--chaos",
188                "--chaos-per-thread-slowdown",
189                "--target-timeslice=1000",
190                "--target-timeslice-syscalls-only",
191                "--record-preemptions-to=unused",
192                "--replay-schedule-from=unused-schedule",
193                "--stacktrace-event=1",
194                "--chaos-slowdown-max-factor=4.5",
195                "--chaos-epoch-length-ns=1000",
196            ]);
197            for backend in TIMERLESS {
198                refuse_unsupported_preemption(&config, backend, true, false).unwrap();
199            }
200        }
201    }
