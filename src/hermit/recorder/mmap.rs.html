31const PAGE_SIZE: usize = 4096;
32// Added in Linux 6.13 and not yet exposed by the pinned libc crate.
33const MADV_GUARD_REMOVE: i32 = 103;

35/// One line of `/proc/<pid>/maps`.
36#[derive(Debug, Clone, Copy, PartialEq, Eq)]
37struct Mapping {
38    start: usize,
39    end: usize,
40    /// `PROT_*` bits from the permission column.
41    prot: i32,
42    private: bool,
43    /// Whether the mapping has a backing inode. Replay represents recorded
44    /// file mappings as anonymous memory, so these are the ranges whose
45    /// contents can differ after advice that drops pages.
46    file_backed: bool,
47    /// Whether this is shared anonymous memory (`MAP_SHARED|MAP_ANONYMOUS`,
48    /// which Linux backs with a deleted `/dev/zero` inode, or System V
49    /// shared memory). Replay maps these live, so they need no refill.
50    anonymous_shmem: bool,
51}
52
53/// `(major, minor)` of a device as `/proc/<pid>/maps` prints it.
54type Device = (u32, u32);
55
56/// The device of the kernel's internal shmem mount, found from a memfd,
57/// which Linux creates on that mount.
58///
59/// Shared anonymous memory and System V segments live on the same mount, and
60/// their names there (`/dev/zero`, `/SYSV<key>`) are chosen by the kernel:
61/// nothing can create a named file on it. A file elsewhere with one of those
62/// names is on another device. `None` if no memfd could be made, in which
63/// case every file-backed mapping is refilled.
64fn shmem_device() -> Option<Device> {
65    static DEVICE: OnceLock<Option<Device>> = OnceLock::new();
66    *DEVICE.get_or_init(|| {
67        // SAFETY: the name is a valid C string; the result is checked.
68        let fd = unsafe { libc::memfd_create(c"hermit-shmem-device".as_ptr(), libc::MFD_CLOEXEC) };
69        if fd < 0 {
70            return None;
71        }
72        // SAFETY: memfd_create returned a new descriptor that nothing else owns.
73        let fd = unsafe { OwnedFd::from_raw_fd(fd) };
74        let dev = std::fs::metadata(format!("/proc/self/fd/{}", fd.as_raw_fd()))
75            .ok()?
76            .dev();
77        Some((libc::major(dev), libc::minor(dev)))
78    })
79}

81fn parse_device(field: &str) -> Option<Device> {
82    let (major, minor) = field.split_once(':')?;
83    Some((
84        u32::from_str_radix(major, 16).ok()?,
85        u32::from_str_radix(minor, 16).ok()?,
86    ))
87}

89fn parse_maps(text: &str, shmem_device: Option<Device>) -> Vec<Mapping> {
90    text.lines()
91        .filter_map(|line| {
92            let mut fields = line.split_ascii_whitespace();
93            let (start, end) = fields.next()?.split_once('-')?;
94            let perms = fields.next()?.as_bytes();
95            let _offset = fields.next()?;
96            let device = parse_device(fields.next()?);
97            let inode = fields.next()?;
98            let path = fields.collect::<Vec<_>>().join(" ");
99            if perms.len() < 4 {
100                return None;
101            }
102            let mut prot = libc::PROT_NONE;
103            for (flag, bit) in [
104                (b'r', libc::PROT_READ),
105                (b'w', libc::PROT_WRITE),
106                (b'x', libc::PROT_EXEC),
107            ] {
108                if perms[..3].contains(&flag) {
109                    prot |= bit;
110                }
111            }
112            Some(Mapping {
113                start: usize::from_str_radix(start, 16).ok()?,
114                end: usize::from_str_radix(end, 16).ok()?,
115                prot,
116                private: perms[3] == b'p',
117                file_backed: inode != "0",
118                anonymous_shmem: shmem_device.is_some()
119                    && device == shmem_device
120                    && (path == "/dev/zero (deleted)" || path.starts_with("/SYSV")),
121            })
122        })
123        .collect()
124}

126/// Reads the guest's mappings. Recording cannot continue without them: an
127/// error returned to the guest would leave no event for replay to consume.
128fn read_mappings(path: &str) -> Vec<Mapping> {
129    let maps = std::fs::read(path)
130        .unwrap_or_else(|error| panic!("Cannot record madvise: {path}: {error}"));
131    parse_maps(&String::from_utf8_lossy(&maps), shmem_device())
132}

134/// The parts of `[start, end)` covered by mappings that replay represents
135/// with an anonymous stand-in, each with the protection of its mapping.
136///
137/// After their pages are dropped these can read back differently in replay:
138/// a private file mapping refaults from the file here but reads zeros there,
139/// and a shared file mapping refaults from the page cache here, including
140/// writes made through a descriptor since it was mapped, which replay's
141/// stand-in never saw. Shared anonymous memory is mapped live in replay and
142/// keeps its contents in both.
143fn refill_ranges(mappings: &[Mapping], start: usize, end: usize) -> Vec<(usize, usize, i32)> {
144    mappings
145        .iter()
146        .filter(|mapping| mapping.file_backed && !mapping.anonymous_shmem)
147        .filter_map(|mapping| {
148            let lo = mapping.start.max(start);
149            let hi = mapping.end.min(end);
150            (lo < hi).then_some((lo, hi, mapping.prot))
151        })
152        .collect()
153}
154
155/// Offset of the first private file-backed mapping in `[start, end)`.
156///
157/// Linux refuses `MADV_WIPEONFORK` with EINVAL at the first VMA that has a
158/// file, after applying the advice to the VMAs before it. Replay's anonymous
159/// replacement would accept the advice, so replay applies it only to this
160/// prefix. Shared mappings stay shared in replay and fail there too.
161fn wipeonfork_live_len(mappings: &[Mapping], start: usize, end: usize) -> Option<usize> {
162    mappings
163        .iter()
164        .find(|mapping| {
165            mapping.file_backed && mapping.private && mapping.start < end && start < mapping.end
166        })
167        .map(|mapping| mapping.start.max(start) - start)
168}
169
170/// Reads `[start, end)` of the guest through `/proc/<tid>/mem`, which also
171/// reads mappings without `PROT_READ`. Pages that cannot be read (guard pages,
172/// pages beyond the end of the file) are skipped; the readable runs are
173/// returned with their addresses.
174fn read_readable_runs(mem: &std::fs::File, start: usize, end: usize) -> Vec<(usize, Vec<u8>)> {
175    let mut runs: Vec<(usize, Vec<u8>)> = Vec::new();
176    let mut pos = start;
177    let mut run_open = false;
178    while pos < end {
179        let mut buf = vec![0u8; end - pos];
180        match mem.read_at(&mut buf, pos as u64) {
181            Ok(n) if n > 0 => {
182                buf.truncate(n);
183                if run_open {
184                    runs.last_mut().unwrap().1.extend_from_slice(&buf);
185                } else {
186                    runs.push((pos, buf));
187                    run_open = true;
188                }
189                pos += n;
190            }
191            _ => {
192                run_open = false;
193                pos = (pos & !(PAGE_SIZE - 1)) + PAGE_SIZE;
194            }
195        }
196    }
197    runs
198}

200impl Recorder {
201    pub(super) async fn handle_mmap<G: Guest<Self>>(
202        &self,
203        guest: &mut G,
204        syscall: Mmap,
205    ) -> Result<i64, Errno> {
206        // Let anonymous mappings through. This should already be deterministic
207        // because ASLR is disabled.
208        if syscall.flags().contains(MapFlags::MAP_ANONYMOUS) {
209            return guest.inject(syscall).await;
210        }
211
212        let len = syscall.len();
213
214        // Do the injection. We need to record the pointer the mapping is at and
215        // all of the bytes contained in the mapping. This is very inefficient
216        // for large mappings and should be replaced by just letting the mmap
217        // through to the real file.
218        let result = guest.inject(syscall).await;
219
220        self.record_event(
221            guest,
222            match result {
223                Ok(addr) => {
224                    let addr = Addr::from_raw(addr as usize).ok_or(Errno::EINVAL)?;
225
226                    // NOTE: We can't use `read_exact` here to slurp up the
227                    // memory map bytes. The memory size may be larger than the
228                    // physical file size for dynamic libraries. The following
229                    // `read` will read up to the physical file length, not the
230                    // length specified on the memory map. The left over bytes
231                    // that extend past the end of the physical file should be
232                    // set to zeros when we replay this `mmap`.
233                    let mut buf = vec![0u8; len];
234                    let physical_length = guest.memory().read(addr, &mut buf)?;
235
236                    // Don't store more bytes than we need to. When we create
237                    // the anonymous map later, the extra bytes will be
238                    // initialized to zero automatically.
239                    buf.truncate(physical_length);
240
241                    Ok(SyscallEvent::Mmap(MmapEvent {
242                        addr: addr.as_raw(),
243                        buf,
244                    }))
245                }
246                Err(errno) => Err(errno),
247            },
248        );
249
250        result
251    }
252
253    /// Records a guest-semantic `madvise`.
254    ///
255    /// The advice runs live in both recording and replay. Replay's anonymous
256    /// stand-ins for file mappings differ in two guest-visible ways, which the
257    /// event carries: pages dropped from a file mapping read back as file
258    /// contents here but as zeros in replay, and `MADV_WIPEONFORK` fails at a
259    /// private file mapping here but would succeed in replay.
260    // AUTONOMOUS-BOT-IMPLEMENTED
261    // TODO-HUMAN-REVIEW(https://github.com/rrnewton/hermit/pull/3589): Audit the madvise refill and WIPEONFORK prefix.
262    pub(super) async fn handle_madvise<G: Guest<Self>>(
263        &self,
264        guest: &mut G,
265        syscall: Madvise,
266    ) -> Result<i64, Errno> {
267        let advice = syscall.advice();
268        let start = syscall.addr().map(AddrMut::as_raw).unwrap_or(0);
269        let len = syscall.len();
270        // Detcore has already rejected unaligned or overflowing ranges.
271        let end = start.saturating_add(len).next_multiple_of(PAGE_SIZE);
272
273        let drops_pages = matches!(
274            advice,
275            libc::MADV_DONTNEED | libc::MADV_DONTNEED_LOCKED | MADV_GUARD_REMOVE
276        );
277        // The thread's own maps: the thread-group leader's is empty once it has
278        // exited.
279        let maps_path = format!("/proc/{}/maps", guest.tid().as_raw());
280
281        let wipeonfork_prefix = if advice == libc::MADV_WIPEONFORK {
282            let mappings = read_mappings(&maps_path);
283            wipeonfork_live_len(&mappings, start, end)
284        } else {
285            None
286        };
287
288        let result = guest.inject(syscall).await;
289
290        // Linux reports EINVAL at the first private file mapping and returns
291        // without visiting the rest of the range. Replay reproduces that by
292        // applying the advice only up to that mapping. Any other outcome
293        // means Linux accepted or failed independently of the file mappings,
294        // so replay applies the whole range and must reproduce the result.
295        let live_len = match (wipeonfork_prefix, result) {
296            (Some(prefix), Err(Errno::EINVAL)) => prefix,
297            _ => len,
298        };
299
300        let mut refills = Vec::new();
301        if drops_pages {
302            let mappings = read_mappings(&maps_path);
303            let ranges = refill_ranges(&mappings, start, end);
304            if !ranges.is_empty() {
305                let mem_path = format!("/proc/{}/mem", guest.tid().as_raw());
306                // The advice has already taken effect, so failing the guest
307                // call now would leave the recording without its event.
308                let mem = std::fs::File::open(&mem_path)
309                    .unwrap_or_else(|error| panic!("Cannot record madvise: {mem_path}: {error}"));
310                for (lo, hi, prot) in ranges {
311                    for (addr, bytes) in read_readable_runs(&mem, lo, hi) {
312                        refills.push(MadviseRefill { addr, bytes, prot });
313                    }
314                }
315            }
316        }
317
318        self.record_event(
319            guest,
320            Ok(SyscallEvent::Madvise(MadviseEvent {
321                result,
322                live_len,
323                refills,
324            })),
325        );
326
327        result
328    }
329}

350    #[test]
351    fn parses_maps_lines() {
352        let mappings = parse_maps(MAPS, SHMEM);
353        assert_eq!(mappings.len(), 9);
354        assert_eq!(
355            mappings[0],
356            Mapping {
357                start: 0x400000,
358                end: 0x401000,
359                prot: libc::PROT_READ | libc::PROT_EXEC,
360                private: true,
361                file_backed: true,
362                anonymous_shmem: false,
363            }
364        );
365        assert!(!mappings[1].file_backed);
366        assert_eq!(mappings[3].prot, libc::PROT_NONE);
367        assert!(!mappings[3].private);
368        assert!(mappings[3].file_backed);
369        assert!(mappings[3].anonymous_shmem);
370        assert!(!mappings[4].anonymous_shmem);
371        assert!(mappings[5].anonymous_shmem);
372    }

374    #[test]
375    fn refill_ranges_clip_to_the_advised_range() {
376        let mappings = parse_maps(MAPS, SHMEM);
377        assert_eq!(
378            refill_ranges(&mappings, 0x7f0000001000, 0x7f0000004000),
379            vec![(
380                0x7f0000002000,
381                0x7f0000003000,
382                libc::PROT_READ | libc::PROT_WRITE
383            )]
384        );
385        assert!(refill_ranges(&mappings, 0x7f0000000000, 0x7f0000002000).is_empty());
386    }

388    #[test]
389    fn refill_ranges_cover_shared_files_but_not_shared_anonymous_memory() {
390        let rw = libc::PROT_READ | libc::PROT_WRITE;
391        // The memfd and the two files merely named like shared anonymous
392        // memory on other devices are refilled.
393        let mappings = parse_maps(MAPS, SHMEM);
394        assert_eq!(
395            refill_ranges(&mappings, 0x7f0000003000, 0x7f0000008000),
396            vec![
397                (0x7f0000004000, 0x7f0000005000, rw),
398                (0x7f0000006000, 0x7f0000007000, rw),
399                (0x7f0000007000, 0x7f0000008000, rw),
400            ]
401        );
402        // Without a known shmem device, everything file-backed is refilled.
403        let mappings = parse_maps(MAPS, None);
404        assert!(mappings.iter().all(|mapping| !mapping.anonymous_shmem));
405        assert_eq!(
406            refill_ranges(&mappings, 0x7f0000003000, 0x7f0000008000).len(),
407            5
408        );
409    }

411    #[test]
412    fn this_hosts_shared_anonymous_memory_is_on_the_shmem_device() {
413        // SAFETY: a fresh anonymous mapping, unmapped below.
414        let addr = unsafe {
415            libc::mmap(
416                std::ptr::null_mut(),
417                PAGE_SIZE,
418                libc::PROT_READ | libc::PROT_WRITE,
419                libc::MAP_SHARED | libc::MAP_ANONYMOUS,
420                -1,
421                0,
422            )
423        };
424        assert_ne!(addr, libc::MAP_FAILED);
425        let mappings = read_mappings("/proc/self/maps");
426        let mapping = mappings
427            .iter()
428            .find(|mapping| mapping.start == addr as usize)
429            .expect("the new mapping is listed");
430        assert!(mapping.file_backed && mapping.anonymous_shmem);
431        assert!(refill_ranges(&mappings, mapping.start, mapping.end).is_empty());
432        // SAFETY: mapped above and no longer referenced.
433        assert_eq!(unsafe { libc::munmap(addr, PAGE_SIZE) }, 0);
434    }
