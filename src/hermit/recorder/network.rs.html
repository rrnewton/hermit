47fn read_bytes<M: MemoryAccess>(
48    memory: &M,
49    pointer: *mut libc::c_void,
50    length: usize,
51) -> Result<Vec<u8>, Errno> {
52    if length == 0 {
53        return Ok(Vec::new());
54    }
55    let address = Addr::<u8>::from_raw(pointer as usize).ok_or(Errno::EFAULT)?;
56    let mut bytes = vec![0; length];
57    memory.read_exact(address.cast(), &mut bytes)?;
58    Ok(bytes)
59}

76/// Copies an accepted peer address out to the guest the way Linux 7.1's
77/// `move_addr_to_user` does once the connection has been taken: read the
78/// capacity with `read_user` and clamp it to the address length; unless the
79/// result is negative, write the full length back first; reject a negative
80/// result with `EINVAL`; then copy the address prefix. Every guest access obeys
81/// the guest's page permissions. A length that cannot be written fails with
82/// `EFAULT` before any address byte is copied, and an address that faults part
83/// way fails with `EFAULT` after the length and whatever prefix was already
84/// written. Linux 7.0 and earlier copy the address before writing the length,
85/// so on those kernels the two faults leave different bytes behind; record
86/// deliberately follows the 7.1 order on every host.
87fn copy_peer_address_out<M: MemoryAccess>(
88    memory: &mut M,
89    read_user: impl FnOnce(&M, AddrMut<u8>, &mut [u8]) -> bool,
90    sockaddr: AddrMut<u8>,
91    addrlen: Option<AddrMut<u8>>,
92    address: &[u8],
93    kernel_len: libc::socklen_t,
94) -> PeerAddressCopyOut {
95    let failed = |error, addr, addr_len| PeerAddressCopyOut {
96        error: Some(error),
97        addr,
98        addr_len,
99    };
100    let Some(addrlen) = addrlen else {
101        return failed(Errno::EFAULT, Vec::new(), None);
102    };
103    let mut capacity = [0; std::mem::size_of::<libc::c_int>()];
104    if !read_user(memory, addrlen, &mut capacity) {
105        return failed(Errno::EFAULT, Vec::new(), None);
106    }
107    let len = libc::c_int::from_ne_bytes(capacity).min(kernel_len as libc::c_int);
108    let Ok(len) = usize::try_from(len) else {
109        return failed(Errno::EINVAL, Vec::new(), None);
110    };
111    if !write_user_word(memory, addrlen, &capacity, &kernel_len.to_ne_bytes()) {
112        return failed(Errno::EFAULT, Vec::new(), None);
113    }
114    let wanted = &address[..len.min(address.len())];
115    let written = write_user_prefix(memory, sockaddr, wanted);
116    if written < wanted.len() {
117        return failed(Errno::EFAULT, wanted[..written].to_vec(), Some(kernel_len));
118    }
119    PeerAddressCopyOut {
120        error: None,
121        addr: wanted.to_vec(),
122        addr_len: Some(kernel_len),
123    }
124}

126/// Reads guest memory with the access the kernel's `get_user` has on x86, where
127/// every writable user page is also readable. `process_vm_readv` needs a
128/// readable mapping, so a range it refuses is read through `/proc/<pid>/mem`,
129/// exactly, if every page of it lies in a readable or writable mapping.
130/// Execute-only mappings count as unreadable, as they are on hardware with
131/// protection keys, and protection keys the guest assigns itself are not
132/// consulted. Like the kernel's read, this races with another thread changing
133/// the mappings, and either order is a result Linux can give.
134fn read_user_bytes<M: MemoryAccess>(
135    memory: &M,
136    pid: Pid,
137    address: AddrMut<u8>,
138    buf: &mut [u8],
139) -> bool {
140    if memory.read_exact_with_user_access(address, buf).is_ok() {
141        return true;
142    }
143    let start = address.as_raw();
144    if start.checked_add(buf.len()).is_none() {
145        return false;
146    }
147    let Ok(maps) = std::fs::read_to_string(format!("/proc/{pid}/maps")) else {
148        return false;
149    };
150    user_page_chunks(start, buf.len())
151        .iter()
152        .all(|&(offset, _)| mapping_is_user_readable(&maps, start + offset))
153        && std::fs::File::open(format!("/proc/{pid}/mem"))
154            .and_then(|mem| mem.read_exact_at(buf, start as u64))
155            .is_ok()
156}

158/// Whether `address` lies in a mapping of a `/proc/<pid>/maps` listing that
159/// the kernel can read from user space on x86: one that is readable or
160/// writable.
161fn mapping_is_user_readable(maps: &str, address: usize) -> bool {
162    maps.lines().any(|line| {
163        let mut fields = line.split_whitespace();
164        let (Some(range), Some(perms)) = (fields.next(), fields.next()) else {
165            return false;
166        };
167        let Some((start, end)) = range.split_once('-') else {
168            return false;
169        };
170        let (Ok(start), Ok(end)) = (
171            usize::from_str_radix(start, 16),
172            usize::from_str_radix(end, 16),
173        ) else {
174            return false;
175        };
176        (start..end).contains(&address) && matches!(perms.as_bytes(), [b'r', ..] | [_, b'w', ..])
177    })
178}

180/// The peer address Linux's `accept` copies out for the new socket `socket`,
181/// and its full length, or `None` if Linux could not name the peer, which
182/// makes `accept` fail with `ECONNABORTED`. Linux asks the socket with
183/// `getname(.., 2)`. `getpeername` asks with 1, which on IPv4 and IPv6 also
184/// refuses a socket the peer has already reset, so for those `SO_PEERNAME`,
185/// which asks with 2, answers instead. It wants exactly the family's address
186/// size.
187fn accepted_peer_address(
188    socket: BorrowedFd,
189) -> std::io::Result<Option<(Vec<u8>, libc::socklen_t)>> {
190    let mut address = [0u8; SOCKADDR_STORAGE_LEN];
191    let mut len = SOCKADDR_STORAGE_LEN as libc::socklen_t;
192    // SAFETY: `address` holds `len` bytes.
193    if unsafe { libc::getpeername(socket.as_raw_fd(), address.as_mut_ptr().cast(), &mut len) } == 0
194    {
195        return Ok(Some((
196            address[..(len as usize).min(SOCKADDR_STORAGE_LEN)].to_vec(),
197            len,
198        )));
199    }
200    let error = std::io::Error::last_os_error();
201    if error.raw_os_error() != Some(libc::ENOTCONN) {
202        return Err(error);
203    }
204    let mut domain = [0; std::mem::size_of::<libc::c_int>()];
205    socket_option(socket, libc::SO_DOMAIN, &mut domain)?;
206    let len = match libc::c_int::from_ne_bytes(domain) {
207        libc::AF_INET => std::mem::size_of::<libc::sockaddr_in>(),
208        libc::AF_INET6 => std::mem::size_of::<libc::sockaddr_in6>(),
209        _ => return Ok(None),
210    };
211    let mut peer = vec![0; len];
212    match socket_option(socket, libc::SO_PEERNAME, &mut peer) {
213        Ok(()) => Ok(Some((peer, len as libc::socklen_t))),
214        Err(error) if error.raw_os_error() == Some(libc::ENOTCONN) => Ok(None),
215        Err(error) => Err(error),
216    }
217}

219/// Reads the `SOL_SOCKET` option `name` into all of `value`.
220fn socket_option(socket: BorrowedFd, name: libc::c_int, value: &mut [u8]) -> std::io::Result<()> {
221    let mut len = value.len() as libc::socklen_t;
222    // SAFETY: `value` holds `len` bytes.
223    let result = unsafe {
224        libc::getsockopt(
225            socket.as_raw_fd(),
226            libc::SOL_SOCKET,
227            name,
228            value.as_mut_ptr().cast(),
229            &mut len,
230        )
231    };
232    if result == 0 {
233        Ok(())
234    } else {
235        Err(std::io::Error::last_os_error())
236    }
237}

239/// The page-sized pieces of `[address, address + len)`, in address order.
240fn user_page_chunks(address: usize, len: usize) -> Vec<(usize, usize)> {
241    let mut chunks = Vec::new();
242    let mut offset = 0;
243    while offset < len {
244        let at = address.wrapping_add(offset);
245        let size = (USER_PAGE_SIZE - at % USER_PAGE_SIZE).min(len - offset);
246        chunks.push((offset, size));
247        offset += size;
248    }
249    chunks
250}

252fn write_user_chunk<M: MemoryAccess>(memory: &mut M, address: usize, bytes: &[u8]) -> usize {
253    let Some(address) = AddrMut::from_raw(address) else {
254        return 0;
255    };
256    match memory.write_with_user_access(address, bytes) {
257        Ok(written) => written,
258        Err(Errno::EFAULT) => 0,
259        Err(error) => panic!("backend cannot write guest memory with user access: {error}"),
260    }
261}

263/// Writes the longest prefix of `bytes` that the guest's page permissions
264/// allow, as `copy_to_user` does, and returns its length.
265fn write_user_prefix<M: MemoryAccess>(memory: &mut M, address: AddrMut<u8>, bytes: &[u8]) -> usize {
266    let mut written = 0;
267    for (offset, size) in user_page_chunks(address.as_raw(), bytes.len()) {
268        let chunk = &bytes[offset..offset + size];
269        let copied = write_user_chunk(memory, address.as_raw() + offset, chunk);
270        written += copied;
271        if copied < size {
272            break;
273        }
274    }
275    written
276}

278/// Stores a 4-byte word all or nothing, as `put_user` does: a word that
279/// straddles a page boundary is written high page first, and restored to
280/// `original` if the low page then refuses the write.
281fn write_user_word<M: MemoryAccess>(
282    memory: &mut M,
283    address: AddrMut<u8>,
284    original: &[u8; 4],
285    value: &[u8; 4],
286) -> bool {
287    let chunks = user_page_chunks(address.as_raw(), value.len());
288    for (index, &(offset, size)) in chunks.iter().enumerate().rev() {
289        let at = address.as_raw() + offset;
290        if write_user_chunk(memory, at, &value[offset..offset + size]) < size {
291            for &(offset, size) in &chunks[index + 1..] {
292                let at = address.as_raw() + offset;
293                write_user_chunk(memory, at, &original[offset..offset + size]);
294            }
295            return false;
296        }
297    }
298    true
299}

316/// Read as much of an EFAULT result as remains readable without allocating from
317/// an untrusted `nfds` up front. Linux may have written earlier `revents`
318/// entries before faulting on a later copy-out.
319fn read_pollfd_prefix<M: MemoryAccess>(
320    memory: &M,
321    address: AddrMut<'_, PollFd>,
322    nfds: usize,
323) -> Vec<PollFd> {
324    const ENTRIES_PER_READ: usize = 256;
325
326    let mut fds = Vec::new();
327    let mut index = 0;
328    while index < nfds {
329        let Some(chunk_address) = pollfd_address(address, index) else {
330            break;
331        };
332        let chunk_len = (nfds - index).min(ENTRIES_PER_READ);
333        let mut chunk = vec![PollFd::default(); chunk_len];
334        let bytes = unsafe {
335            std::slice::from_raw_parts_mut(
336                chunk.as_mut_ptr().cast::<u8>(),
337                std::mem::size_of_val(chunk.as_slice()),
338            )
339        };
340        let bytes_read = match memory.read(chunk_address.cast::<u8>(), bytes) {
341            Ok(bytes_read) => bytes_read,
342            Err(_) => break,
343        };
344        let complete_entries = bytes_read / std::mem::size_of::<PollFd>();
345        if complete_entries == 0 {
346            break;
347        }
348        fds.extend(chunk.into_iter().take(complete_entries));
349        index += complete_entries;
350        if bytes_read % std::mem::size_of::<PollFd>() != 0 {
351            break;
352        }
353    }
354    fds
355}

357fn capture_pollfds<M: MemoryAccess>(
358    memory: &M,
359    address: Option<AddrMut<'_, PollFd>>,
360    nfds: usize,
361    result: Result<i64, Errno>,
362) -> Result<Option<Vec<PollFd>>, Errno> {
363    match result {
364        Ok(_) | Err(Errno::EINTR) => address
365            .map(|address| read_pollfds(memory, address, nfds))
366            .transpose(),
367        // Best effort is load-bearing: a wholly bad pointer also returns
368        // EFAULT. A failed diagnostic read must not replace the recorded errno.
369        Err(Errno::EFAULT) => Ok(address.map(|address| read_pollfd_prefix(memory, address, nfds))),
370        // In particular, EINVAL for nfds > RLIMIT_NOFILE must not trigger a
371        // read or an allocation based on the invalid count.
372        _ => Ok(None),
373    }
374}

389fn capture_ppoll_event<M: MemoryAccess>(
390    memory: &M,
391    fds_address: Option<AddrMut<'_, libc::pollfd>>,
392    timeout_address: Option<AddrMut<'_, Timespec>>,
393    nfds: usize,
394    result: Result<i64, Errno>,
395) -> Result<PpollEvent, Errno> {
396    let timeout = match result {
397        // The timeout can also be updated when pollfd copy-out later faults.
398        // If EFAULT instead came from an unreadable timeout pointer, retain the
399        // syscall result and simply record no readable timeout value.
400        Err(Errno::EFAULT) => timeout_address.and_then(|address| memory.read_value(address).ok()),
401        // Raw ppoll can update its remaining-time argument on success and on
402        // error returns such as EINTR, EBADF, or EINVAL. Preserve future errno
403        // behavior too: if Linux accepted a readable timeout input before
404        // producing the result, its post-call bytes are guest-visible.
405        _ => timeout_address
406            .map(|address| memory.read_value(address))
407            .transpose()?,
408    };
409    let fds =
410        if matches!(result, Err(Errno::EFAULT)) && timeout_address.is_some() && timeout.is_none() {
411            // Linux validates the timeout input before polling. If that pointer is
412            // unreadable, this EFAULT occurred before any pollfd output copy-out;
413            // do not scan a potentially huge valid array looking for outputs that
414            // the kernel never attempted to write.
415            None
416        } else {
417            capture_pollfds(
418                memory,
419                fds_address.map(|address| address.cast::<PollFd>()),
420                nfds,
421                result,
422            )?
423        };
424
425    Ok(PpollEvent {
426        result,
427        fds_pointer_present: fds_address.is_some(),
428        fds,
429        timeout_pointer_present: timeout_address.is_some(),
430        timeout,
431    })
432}

434/// Read up to `length` bytes, stopping at the first unreadable byte. Linux
435/// clamps `nfds` to the descriptor table size, so a successful call writes a
436/// prefix of the guest's range; any readable bytes past it are unchanged guest
437/// memory, which replay restores to the same value.
438fn read_byte_prefix<M: MemoryAccess>(
439    memory: &M,
440    address: AddrMut<'_, u8>,
441    length: usize,
442) -> Vec<u8> {
443    let mut bytes = vec![0; length];
444    let mut filled = 0;
445    while filled < length {
446        let Some(chunk) = address
447            .as_raw()
448            .checked_add(filled)
449            .and_then(AddrMut::<u8>::from_raw)
450        else {
451            break;
452        };
453        match memory.read(chunk, &mut bytes[filled..]) {
454            Ok(0) | Err(_) => break,
455            Ok(count) => filled += count,
456        }
457    }
458    bytes.truncate(filled);
459    bytes
460}

462/// The kernel's descriptor-table size for `tid`, from the `FDSize:` line of
463/// `/proc/<tid>/status`. `core_sys_select` clamps `nfds` to this before it
464/// reads or writes a set, so no byte past it belongs to the call.
465fn guest_max_fds(tid: Pid) -> Option<i32> {
466    let status = std::fs::read_to_string(format!("/proc/{}/status", tid.as_raw())).ok()?;
467    parse_fd_size(&status)
468}
469
470/// A task's table holds at least `NR_OPEN_DEFAULT` (`BITS_PER_LONG`) entries.
471/// A smaller value means a task without a table, or the wrong task, and would
472/// silently record empty sets, so it is treated as unknown.
473fn parse_fd_size(status: &str) -> Option<i32> {
474    status
475        .lines()
476        .find_map(|line| line.strip_prefix("FDSize:"))
477        .and_then(|value| value.trim().parse().ok())
478        .filter(|&size: &i32| size >= 64)
479}
480
481/// The `nfds` the kernel acted on. Without a table size, fall back to the
482/// default `fs.nr_open` limit of 1048576 descriptors (128 KiB per set), so a
483/// guest passing `INT_MAX` cannot make the recorder allocate gigabytes.
484fn select_capture_nfds(nfds: i32, max_fds: Option<i32>) -> i32 {
485    nfds.min(max_fds.unwrap_or(1 << 20))
486}

488fn capture_select_event<M: MemoryAccess>(
489    memory: &M,
490    nfds: i32,
491    fd_sets: [Option<AddrMut<'_, libc::fd_set>>; 3],
492    timeout: Option<(AddrMut<'_, u8>, usize)>,
493    result: Result<i64, Errno>,
494) -> SelectEvent {
495    // fs/select.c copies the sets out after a successful wait and can fault
496    // partway through; every other result leaves them as the guest wrote them.
497    // `nfds` must already be clamped to the descriptor table.
498    let copied_out = matches!(result, Ok(_) | Err(Errno::EFAULT));
499    let length = fd_set_bytes(nfds);
500    let fd_sets = fd_sets.map(|address| {
501        address
502            .filter(|_| copied_out)
503            .map(|address| read_byte_prefix(memory, address.cast(), length))
504    });
505    // Linux may write the remaining time back on any result that reached the
506    // wait, and leaves it alone otherwise; capturing the post-call bytes is
507    // right either way, because replaying unchanged bytes changes nothing. An
508    // unreadable pointer could not have been written.
509    let timeout = timeout
510        .map(|(address, length)| read_byte_prefix(memory, address, length))
511        .filter(|bytes| !bytes.is_empty());
512    SelectEvent {
513        result,
514        fd_sets,
515        timeout,
516    }
517}

519impl Recorder {
520    pub(super) async fn handle_epoll_wait<G: Guest<Self>>(
521        &self,
522        guest: &mut G,
523        syscall: EpollWait,
524    ) -> Result<i64, Errno> {
525        let result = guest.inject(syscall).await;
526
527        let event = result.and_then(|ret| {
528            let updated = ret as usize;
529            let mut events = vec![0; updated * std::mem::size_of::<libc::epoll_event>()];
530            if !events.is_empty() {
531                guest
532                    .memory()
533                    .read_exact(syscall.events().ok_or(Errno::EFAULT)?.cast(), &mut events)?;
534            }
535            Ok(SyscallEvent::EpollWait(EpollWaitEvent {
536                events,
537                updated,
538                replay_kernel_side_effect: self
539                    .epoll_requires_replay_kernel_side_effect(guest.pid(), syscall.epfd()),
540            }))
541        });
542
543        self.record_event(guest, event);
544        result
545    }

549        guest: &mut G,
550        syscall: Poll,

555        let event =
556            capture_poll_event(&guest.memory(), syscall.fds(), len, result).map(SyscallEvent::Poll);

558        self.record_event(guest, event);

563    pub(super) async fn handle_ppoll<G: Guest<Self>>(
564        &self,
565        guest: &mut G,
566        syscall: Ppoll,
567    ) -> Result<i64, Errno> {
568        let len = syscall.nfds() as usize;
569        let timeout_is_zero = syscall
570            .timeout()
571            .and_then(|address| {
572                let timeout: Timespec = guest.memory().read_value(address).ok()?;
573                Some(timeout)
574            })
575            .is_some_and(|timeout| timeout.tv_sec == 0 && timeout.tv_nsec == 0);
576        tracing::trace!(
577            has_signal_mask = syscall.sigmask().is_some(),
578            timeout_is_zero,
579            "Recorder observed ppoll input"
580        );
581        let result = guest.inject(syscall).await;
582
583        let event = capture_ppoll_event(
584            &guest.memory(),
585            syscall.fds(),
586            syscall.timeout(),
587            len,
588            result,
589        )
590        .map(SyscallEvent::Ppoll);
591
592        self.record_event(guest, event);
593
594        result
595    }

597    pub(super) async fn handle_select<G: Guest<Self>>(
598        &self,
599        guest: &mut G,
600        syscall: Select,
601    ) -> Result<i64, Errno> {
602        let result = guest.inject(syscall).await;
603        let event = capture_select_event(
604            &guest.memory(),

637    pub(super) async fn handle_accept4<G: Guest<Self>>(
638        &self,
639        guest: &mut G,
640        syscall: Accept4,
641    ) -> Result<i64, Errno> {
642        // Without an address buffer Linux writes nothing but the descriptor,
643        // and ignores `addrlen`.
644        let Some(sockaddr) = syscall.sockaddr() else {
645            let result = guest.inject(syscall).await;
646            let event = result.map(|fd| {
647                SyscallEvent::Accept(AcceptEvent {
648                    result: Ok(accepted_fd(fd)),
649                    addr: Vec::new(),
650                    addr_len: None,
651                })
652            });
653            self.record_event(guest, event);
654            return result;
655        };
656
657        // Only after Linux has taken the connection off the queue does it read
658        // the capacity in `*addrlen`, write the full length back and copy the
659        // address out. Another thread can change or unmap both buffers while
660        // accept4 waits, and a fault in the copy-out leaves the earlier writes
661        // in place while the call fails. So accept without an address, ask the
662        // new socket for its peer, and perform that copy-out here, recording
663        // exactly the memory the guest saw change. No guest memory is borrowed
664        // for this, so the guest's stack and buffers are never touched beyond
665        // what Linux writes.
666        let fd = match guest
667            .inject(syscall.with_sockaddr(None).with_addrlen(None))
668            .await
669        {
670            Ok(fd) => fd,
671            Err(error) => {
672                self.record_event(guest, Err(error));
673                return Err(error);
674            }
675        };
676
677        // Only a guest thread closing or replacing a descriptor number it was
678        // never given could take the new socket away before this, so failing
679        // to inspect it is treated as a Hermit failure, not a guest error.
680        // The socket is in the accepting thread's table, which is not the
681        // leader's when the thread was cloned without CLONE_FILES.
682        let pid = guest.pid();
683        let peer = crate::fd::duplicate_guest_thread_fd(pid, guest.tid(), accepted_fd(fd))
684            .and_then(|socket| accepted_peer_address(socket.as_fd()))
685            .unwrap_or_else(|error| {
686                panic!("accept4 returned fd {fd}, but its peer address could not be read: {error}")
687            });
688        let copy_out = match peer {
689            Some((address, kernel_len)) => copy_peer_address_out(
690                &mut guest.memory(),
691                |memory, address, buf| read_user_bytes(memory, pid, address, buf),
692                sockaddr.cast(),
693                syscall.addrlen().map(|address| address.cast()),
694                &address,
695                kernel_len,
696            ),
697            None => PeerAddressCopyOut {
698                error: Some(Errno::ECONNABORTED),
699                addr: Vec::new(),
700                addr_len: None,
701            },
702        };
703        let result = match copy_out.error {
704            None => Ok(accepted_fd(fd)),
705            Some(error) => {
706                // Linux releases the new file when naming the peer or the
707                // copy-out fails, so the descriptor is never installed for the
708                // guest. Detcore has not seen it either, because the call
709                // returns an error. A pending signal, such as the client's
710                // SIGCHLD, can interrupt this extra syscall before it runs, so
711                // retry it: Linux's close(2) itself never reports ERESTARTSYS.
712                let close = Close::new().with_fd(accepted_fd(fd));
713                if let Err(close_error) = guest.inject_with_retry(close).await {
714                    panic!(
715                        "could not close fd {fd} after its accept4 copy-out failed: {close_error}"
716                    );
717                }
718                Err(error)
719            }
720        };
721
722        self.record_event(
723            guest,
724            Ok(SyscallEvent::Accept(AcceptEvent {
725                result,
726                addr: copy_out.addr,
727                addr_len: copy_out.addr_len,
728            })),
729        );
730
731        result.map(i64::from)
732    }

734    pub(super) async fn handle_sockopt_family<G: Guest<Self>>(
735        &self,
736        guest: &mut G,
737        syscall: SockOptFamily,
738    ) -> Result<i64, Errno> {
739        // The buffer length is both an input and output. If optlen is smaller
740        // than the real value, then the value will be truncated.
741
742        let buflen_addr = syscall.value_len().ok_or(Errno::EFAULT)?;
743
744        // `optlen` will be updated after the syscall has been injected.
745        let buflen: libc::socklen_t = guest.memory().read_value(buflen_addr)?;
746
747        let result = guest.inject(Syscall::from(syscall)).await;
748
749        let event = result.and_then(|ret| {
750            debug_assert_eq!(ret, 0);
751
752            // Linux permits a NULL value buffer when its input length is zero.
753            let value = if let Some(address) = syscall.value() {
754                let mut value = vec![0u8; buflen as usize];
755                guest
756                    .memory()
757                    .read_exact(address.cast::<u8>(), &mut value)?;
758                value
759            } else {
760                Vec::new()
761            };
762
763            // Need to read the (new) length. This might not have been updated,
764            // but we don't know until we check it.
765            let length: libc::socklen_t = guest.memory().read_value(buflen_addr)?;
766
767            Ok(SyscallEvent::SockOpt(SockOptEvent { value, length }))
768        });
769
770        self.record_event(guest, event);
771
772        result
773    }

775    pub(super) async fn handle_recvmsg<G: Guest<Self>>(
776        &self,
777        guest: &mut G,
778        syscall: Recvmsg,
779    ) -> Result<i64, Errno> {
780        let input = syscall
781            .msg()
782            .ok_or(Errno::EFAULT)
783            .and_then(|address| guest.memory().read_value(address))
784            .map(|message: libc::msghdr| (message.msg_namelen as usize, message.msg_controllen));
785        let result = guest.inject(syscall).await;
786
787        self.record_event(
788            guest,
789            result.and_then(|result| {
790                let (name_capacity, control_capacity) = input?;
791                let message_address = syscall.msg().ok_or(Errno::EFAULT)?;
792                let output: libc::msghdr = guest.memory().read_value(message_address)?;
793                let iovecs = crate::read_iovecs(&guest.memory(), &output)?;
794                let mut remaining = usize::try_from(result).map_err(|_| Errno::EINVAL)?;
795                let mut buffers = Vec::with_capacity(iovecs.len());
796                for iovec in iovecs {
797                    let length = remaining.min(iovec.iov_len);
798                    buffers.push(read_bytes(&guest.memory(), iovec.iov_base, length)?);
799                    remaining -= length;
800                }
801
802                let name_length = name_capacity.min(output.msg_namelen as usize);
803                let control_length = control_capacity.min(output.msg_controllen);
804
805                Ok(SyscallEvent::Recvmsg(RecvmsgEvent {
806                    result,
807                    iovs: buffers,
808                    name: read_bytes(&guest.memory(), output.msg_name, name_length)?,
809                    name_len: output.msg_namelen,
810                    control: read_bytes(&guest.memory(), output.msg_control, control_length)?,
811                    control_len: output.msg_controllen,
812                    flags: output.msg_flags,
813                }))
814            }),
815        );
816
817        result
818    }
