110/// Saves stdin captured before Rust's process startup can reuse a closed fd 0.
111pub fn reserve_kvm_stdin(stdin: Option<fs::File>) -> io::Result<()> {
112    let mut reservation = KVM_STDIN_RESERVATION
113        .lock()
114        .map_err(|_| io::Error::other("KVM stdin reservation lock is poisoned"))?;
115    if reservation.is_some() {
116        return Err(io::Error::new(
117            io::ErrorKind::AlreadyExists,
118            "KVM stdin is already reserved",
119        ));
120    }
121    *reservation = Some(match stdin {
122        Some(file) => KvmStdinReservation::Open(file),
123        None => KvmStdinReservation::Closed,
124    });
125    Ok(())
126}

128fn duplicate_current_stdin() -> io::Result<Option<fs::File>> {
129    // SAFETY: F_DUPFD_CLOEXEC duplicates fd 0 without taking ownership of it.
130    let duplicate = unsafe { libc::fcntl(libc::STDIN_FILENO, libc::F_DUPFD_CLOEXEC, 3) };
131    if duplicate >= 0 {
132        // SAFETY: F_DUPFD_CLOEXEC returned a new owned descriptor.
133        return Ok(Some(unsafe { fs::File::from_raw_fd(duplicate) }));
134    }
135    let error = io::Error::last_os_error();
136    if error.raw_os_error() == Some(libc::EBADF) {
137        Ok(None)
138    } else {
139        Err(error)
140    }
141}

143fn ensure_kvm_stdin_reserved() -> io::Result<()> {
144    let mut reservation = KVM_STDIN_RESERVATION
145        .lock()
146        .map_err(|_| io::Error::other("KVM stdin reservation lock is poisoned"))?;

156fn reserved_kvm_stdin() -> Result<Option<fs::File>, Error> {
157    ensure_kvm_stdin_reserved()?;
158    let reservation = KVM_STDIN_RESERVATION
159        .lock()
160        .map_err(|_| io::Error::other("KVM stdin reservation lock is poisoned"))?;
161    match reservation.as_ref() {
162        Some(KvmStdinReservation::Open(file)) => Ok(Some(file.try_clone()?)),
163        Some(KvmStdinReservation::Closed) => Ok(None),
164        None => unreachable!("stdin reservation was initialized above"),
165    }

168/// A replayable snapshot of the guest's stdin for the output-capturing backends.
169///
170/// `hermit run --verify` executes the guest twice and compares the two runs. The
171/// output-capturing backends historically fed the guest `Stdio::null()`, so any
172/// data piped into hermit (`echo prog | hermit run --strict --verify -- ...`)
173/// was silently dropped. Both runs then saw identical *empty* input and hermit
174/// reported a false "deterministic" success even though the guest never received
175/// its input. `Seekable` holds a rewindable file so both runs receive the exact
176/// same bytes; `None` means stdin should be `/dev/null` (nothing to replay, or a
177/// terminal that cannot be replayed identically to two runs).
178enum StdinSnapshot {
179    Seekable(fs::File),
180    None,
181}
182
183static OUTPUT_STDIN_SNAPSHOT: Mutex<Option<StdinSnapshot>> = Mutex::new(None);
184
185/// Records a rewindable snapshot of the process stdin so the output-capturing
186/// backends can replay identical input to each run of `hermit run --verify`.
187///
188/// `stdin` is the descriptor captured before Rust startup could reuse a closed
189/// fd 0 (see the binary's `startup_stdin`). A regular-file redirect is already
190/// seekable and is kept as-is; a pipe/fifo/socket is drained once into a
191/// seekable temporary file so it can be re-read; a terminal (or absent stdin)
192/// is treated as `/dev/null` because a live terminal cannot be replayed
193/// identically to two runs.
194pub fn reserve_output_stdin_snapshot(stdin: Option<fs::File>) -> io::Result<()> {
195    let snapshot = match stdin {
196        None => StdinSnapshot::None,
197        Some(mut file) => {
198            // SAFETY: as_raw_fd borrows the descriptor without taking ownership.
199            let is_tty = unsafe { libc::isatty(file.as_raw_fd()) } == 1;
200            if is_tty {
201                StdinSnapshot::None
202            } else if file.stream_position().is_ok() {
203                // Already seekable (e.g. `< file`): rewind before each run.
204                StdinSnapshot::Seekable(file)
205            } else {
206                // Non-seekable (pipe/fifo/socket): buffer once into a seekable
207                // temporary file so both --verify runs receive identical input.
208                let mut buffered = tempfile::tempfile()?;
209                io::copy(&mut file, &mut buffered)?;
210                buffered.seek(SeekFrom::Start(0))?;
211                StdinSnapshot::Seekable(buffered)
212            }
213        }
214    };
215    let mut reservation = OUTPUT_STDIN_SNAPSHOT
216        .lock()
217        .map_err(|_| io::Error::other("output stdin snapshot lock is poisoned"))?;
218    if reservation.is_some() {
219        return Err(io::Error::new(
220            io::ErrorKind::AlreadyExists,
221            "output stdin snapshot is already reserved",
222        ));
223    }
224    *reservation = Some(snapshot);
225    Ok(())
226}

228/// Returns a rewound descriptor for the reserved stdin snapshot, if any.
229///
230/// Each call rewinds the snapshot and hands back a fresh descriptor, so
231/// repeated `--verify` runs read identical input from the start. Returns `None`
232/// when no snapshot was reserved or the reserved stdin is a terminal/absent
233/// (nothing to replay). Runs are sequential (each executes in its own forked
234/// container child), so resetting the shared open file description's offset here
235/// is safe.
236fn output_backend_stdin_file() -> Result<Option<fs::File>, Error> {
237    let mut reservation = OUTPUT_STDIN_SNAPSHOT
238        .lock()
239        .map_err(|_| io::Error::other("output stdin snapshot lock is poisoned"))?;
240    match reservation.as_mut() {
241        Some(StdinSnapshot::Seekable(file)) => {
242            file.seek(SeekFrom::Start(0))?;
243            Ok(Some(file.try_clone()?))
244        }
245        Some(StdinSnapshot::None) | None => Ok(None),
246    }
247}
248
249/// Returns the stdin to hand a guest run in an output-capturing backend.
250///
251/// When [`reserve_output_stdin_snapshot`] has stored a replayable snapshot the
252/// file is rewound and a fresh descriptor handed to the guest, so each
253/// `--verify` run reads identical input. Otherwise (no snapshot reserved, or a
254/// terminal/absent stdin) the guest gets `/dev/null`, matching the previous
255/// behavior for callers that do not reserve a snapshot.
256fn output_backend_stdin() -> Result<Stdio, Error> {
257    match output_backend_stdin_file()? {
258        Some(file) => Ok(Stdio::from(file)),
259        None => Ok(Stdio::null()),
260    }
261}

274enum CapabilityProbe {
275    Namespaces,
276    Ptrace,
277    Seccomp,
278}
279
280fn run_capability_probe(probe: CapabilityProbe) -> Result<bool, Error> {
281    // SAFETY: The child calls only async-signal-safe syscalls and exits immediately.
282    let pid = unsafe { libc::fork() };
283    if pid == -1 {
284        return Err(std::io::Error::last_os_error()).context("Failed to fork capability probe");
285    }
286    if pid == 0 {
287        let supported = match probe {
288            CapabilityProbe::Namespaces => unsafe {
289                libc::unshare(libc::CLONE_NEWUSER | libc::CLONE_NEWPID) == 0
290            },
291            CapabilityProbe::Ptrace => {
292                // SAFETY: PTRACE_TRACEME ignores the pid and address arguments.
293                unsafe {
294                    libc::ptrace(
295                        libc::PTRACE_TRACEME,
296                        0,
297                        std::ptr::null_mut::<libc::c_void>(),
298                        std::ptr::null_mut::<libc::c_void>(),
299                    ) != -1
300                }
301            }
302            CapabilityProbe::Seccomp => {
303                let mut filter = libc::sock_filter {
304                    code: 0x06, // BPF_RET | BPF_K
305                    jt: 0,
306                    jf: 0,
307                    k: 0x7fff0000, // SECCOMP_RET_ALLOW
308                };
309                let program = libc::sock_fprog {
310                    len: 1,
311                    filter: &mut filter,
312                };
313                // SAFETY: The filter is an allow-all program with a valid one-element lifetime.
314                unsafe {
315                    libc::prctl(libc::PR_SET_NO_NEW_PRIVS, 1, 0, 0, 0) == 0
316                        && libc::syscall(
317                            libc::SYS_seccomp,
318                            1, // SECCOMP_SET_MODE_FILTER
319                            0,
320                            &program,
321                        ) == 0
322                }
323            }
324        };
325        // SAFETY: Avoid running Rust destructors after fork.
326        unsafe { libc::_exit(i32::from(!supported)) }
327    }
328
329    let mut status = 0;
330    loop {
331        // SAFETY: pid is the child created above and status points to valid storage.
332        let result = unsafe { libc::waitpid(pid, &mut status, 0) };
333        if result == pid {
334            return Ok(libc::WIFEXITED(status) && libc::WEXITSTATUS(status) == 0);
335        }
336        if result == -1 {
337            let error = std::io::Error::last_os_error();
338            if error.kind() == std::io::ErrorKind::Interrupted {
339                continue;
340            }
341            return Err(error).context("Failed to wait for capability probe");
342        }
343    }
344}

346fn validate_tracing_environment() -> Result<(), Error> {
347    if !run_capability_probe(CapabilityProbe::Namespaces)? {
348        anyhow::bail!(
349            "Hermit cannot create its required user and PID namespaces: \
350             unshare(CLONE_NEWUSER | CLONE_NEWPID) was denied. Allow unprivileged user namespaces \
351             and the unshare syscall in the host/container policy."
352        );
353    }
354    if !run_capability_probe(CapabilityProbe::Ptrace)? {
355        anyhow::bail!(
356            "Hermit cannot use ptrace in this environment: a child PTRACE_TRACEME probe was \
357             denied. Allow same-UID parent-child ptrace in the container seccomp and host \
358             Yama/LSM policy; CAP_SYS_PTRACE is normally not required. Use --namespace-only for \
359             a sandbox smoke test without syscall interception."
360        );
361    }
362    if !run_capability_probe(CapabilityProbe::Seccomp)? {
363        anyhow::bail!(
364            "Hermit cannot install its tracee seccomp filter: \
365             seccomp(SECCOMP_SET_MODE_FILTER) was denied. Allow seccomp and \
366             prctl(PR_SET_NO_NEW_PRIVS) in the container policy, or use --namespace-only for a \
367             sandbox smoke test without syscall interception."
368        );
369    }
370    Ok(())
371}

416fn validate_liteinst_runtime_library(path: &Path) -> io::Result<PathBuf> {
417    let bytes = fs::read(path).map_err(|error| {
418        io::Error::new(
419            error.kind(),
420            format!(
421                "failed to read LiteInst runtime {}: {error}",
422                path.display()
423            ),
424        )
425    })?;
426    let elf = Elf::parse(&bytes).map_err(|error| {
427        io::Error::new(
428            io::ErrorKind::InvalidData,
429            format!(
430                "LiteInst runtime {} is not an ELF DSO: {error}",
431                path.display()
432            ),
433        )
434    })?;
435    if elf.header.e_type != header::ET_DYN || elf.header.e_machine != header::EM_X86_64 {
436        return Err(io::Error::new(
437            io::ErrorKind::InvalidData,
438            format!(
439                "LiteInst runtime {} is not an x86-64 shared object",
440                path.display()
441            ),
442        ));
443    }
444
445    let required = [
446        "reverie_liteinst_initialize",
447        "reverie_liteinst_site_trap_count",
448        "reverie_liteinst_site_hook_count",
449    ];
450    for name in required {
451        if !elf
452            .dynsyms
453            .iter()
454            .any(|symbol| elf.dynstrtab.get_at(symbol.st_name) == Some(name))
455        {
456            return Err(io::Error::new(
457                io::ErrorKind::InvalidData,
458                format!(
459                    "LiteInst runtime {} is missing required export {name}",
460                    path.display()
461                ),
462            ));
463        }
464    }
465    let (initializer_index, initializer) = elf
466        .dynsyms
467        .iter()
468        .enumerate()
469        .find(|(_, symbol)| {
470            elf.dynstrtab.get_at(symbol.st_name) == Some("reverie_liteinst_initialize")
471        })
472        .ok_or_else(|| io::Error::other("checked LiteInst initializer disappeared"))?;
473    let init_array = elf
474        .section_headers
475        .iter()
476        .find(|section| section.sh_type == section_header::SHT_INIT_ARRAY)
477        .ok_or_else(|| {
478            io::Error::new(
479                io::ErrorKind::InvalidData,
480                format!(
481                    "LiteInst runtime {} has no constructor array",
482                    path.display()
483                ),
484            )
485        })?;
486    let init_start = init_array.sh_addr;
487    let init_end = init_start.saturating_add(init_array.sh_size);
488    let relocated_initializer = elf
489        .dynrelas
490        .iter()
491        .chain(elf.dynrels.iter())
492        .any(|relocation| {
493            (init_start..init_end).contains(&relocation.r_offset)
494                && relocation.r_sym == initializer_index
495        });
496    let init_bytes = usize::try_from(init_array.sh_offset)
497        .ok()
498        .and_then(|start| {
499            usize::try_from(init_array.sh_size)
500                .ok()
501                .and_then(|size| bytes.get(start..start.checked_add(size)?))
502        })
503        .unwrap_or_default();
504    let direct_initializer = init_bytes
505        .as_chunks::<8>()
506        .0
507        .iter()
508        .any(|entry| u64::from_le_bytes(*entry) == initializer.st_value);
509    if !relocated_initializer && !direct_initializer {
510        return Err(io::Error::new(
511            io::ErrorKind::InvalidData,
512            format!(
513                "LiteInst runtime {} does not register reverie_liteinst_initialize as a preload constructor",
514                path.display()
515            ),
516        ));
517    }
518    path.canonicalize()
519}

521/// Returns the LiteInst preload cdylib produced beside the Hermit binary.
522#[doc(hidden)]
523pub fn liteinst_runtime_library_path() -> io::Result<PathBuf> {
524    if let Some(path) = std::env::var_os("HERMIT_LITEINST_RUNTIME") {
525        let path = PathBuf::from(path);
526        if !path.is_file() {
527            return Err(io::Error::new(
528                io::ErrorKind::NotFound,
529                "HERMIT_LITEINST_RUNTIME does not name a regular file",
530            ));
531        }
532        return validate_liteinst_runtime_library(&path).map_err(|error| {
533            io::Error::new(
534                error.kind(),
535                format!("HERMIT_LITEINST_RUNTIME is invalid: {error}"),
536            )
537        });
538    }
539
540    let executable = std::env::current_exe()?;
541    let directory = executable.parent().ok_or_else(|| {
542        io::Error::new(
543            io::ErrorKind::NotFound,
544            "Hermit executable has no parent directory",
545        )
546    })?;
547    if let Some(path) = [
548        directory.join("libreverie_liteinst.so"),
549        directory.join("deps/libreverie_liteinst.so"),
550    ]
551    .into_iter()
552    .find(|path| path.is_file())
553    {
554        return validate_liteinst_runtime_library(&path);
555    }
556    if let Some(path) = hermit_resources::resource("libreverie_liteinst.so")?
557        && path.is_file()
558    {
559        return validate_liteinst_runtime_library(&path);
560    }
561    Err(io::Error::new(
562        io::ErrorKind::NotFound,
563        format!(
564            "libreverie_liteinst.so was not built beside {} or staged as an installed resource",
565            executable.display()
566        ),
567    ))
568}

730// TODO-HUMAN-REVIEW(PR-739): Review SaBRe loader discovery and executable validation.
731fn resolve_sabre_binary_from(
732    override_path: Option<&OsStr>,
733    packaged_path: Option<&Path>,
734    executable: &Path,
735    path_env: &OsStr,
736) -> Result<PathBuf, Error> {
737    if let Some(requested) = override_path {
738        if requested.is_empty() {
739            return Err(anyhow!("{SABRE_BINARY_ENV} is empty"));
740        }
741        let path = PathBuf::from(requested);
742        return is_executable_file(&path)
743            .then_some(path.clone())
744            .ok_or_else(|| {
745                anyhow!(
746                    "{SABRE_BINARY_ENV}={} is not an executable file",
747                    path.display()
748                )
749            });
750    }
751
752    if let Some(path) = packaged_path
753        && is_executable_file(path)
754    {
755        return Ok(path.to_path_buf());
756    }
757
758    let directory = executable
759        .parent()
760        .ok_or_else(|| anyhow!("Hermit executable has no parent directory"))?;
761    let sibling = directory.join("sabre");
762    let target_build = directory.parent().map(|target| target.join("sabre/sabre"));
763
764    if is_executable_file(&sibling) {
765        return Ok(sibling);
766    }
767    if let Some(candidate) = &target_build
768        && is_executable_file(candidate)
769    {
770        return Ok(candidate.clone());
771    }
772    if !path_env.is_empty()
773        && let Some(candidate) = std::env::split_paths(path_env)
774            .map(|directory| directory.join("sabre"))
775            .find(|candidate| is_executable_file(candidate))
776    {
777        return Ok(candidate);
778    }
779
780    Err(anyhow!(
781        "SaBRe executable was not found in the Hermit installation, beside {}, or in PATH; set {} or {SABRE_BINARY_ENV}",
782        executable.display(),
783        hermit_resources::INSTALL_DIR_ENV
784    ))
785}

791    let packaged_path = hermit_resources::resource("sabre")?;

822// TODO-HUMAN-REVIEW(PR-845): Review the neutral-name workaround for SaBRe's
823// dynamic-loader prefix collision.
824fn stage_sabre_program_in(
825    program: &Path,
826    staging_directory: &Path,
827) -> Result<Option<StagedSabreProgram>, Error> {
828    if !sabre_program_needs_neutral_name(program) {
829        return Ok(None);
830    }
831
832    let path = staging_directory.join(format!("hermit-sabre-program-{}", std::process::id()));
833    let mut source = fs::File::open(program).map_err(|error| {
834        anyhow!(
835            "failed to open SaBRe guest executable {} for neutral-name staging: {error}",
836            program.display()
837        )
838    })?;
839    let mut staged = fs::OpenOptions::new()
840        .write(true)
841        .create_new(true)
842        .mode(0o700)
843        .open(&path)
844        .map_err(|error| {
845            anyhow!(
846                "failed to stage SaBRe guest executable {} as {}: {error}",
847                program.display(),
848                path.display()
849            )
850        })?;
851    if let Err(error) = io::copy(&mut source, &mut staged) {
852        let _ = fs::remove_file(&path);
853        return Err(anyhow!(
854            "failed to stage SaBRe guest executable {} as {}: {error}",
855            program.display(),
856            path.display()
857        ));
858    }
859    drop(staged);
860
861    Ok(Some(StagedSabreProgram { path }))
862}

910const SABRE_RPC_DISCONNECT_TIMEOUT: Duration = Duration::from_secs(1);
911
912async fn wait_for_sabre_rpc_disconnects<T>(
913    global: &Arc<T>,
914    timeout: Duration,
915) -> Result<(), usize> {
916    let disconnected = tokio::time::timeout(timeout, async {
917        while Arc::strong_count(global) > 1 {
918            tokio::time::sleep(Duration::from_millis(1)).await;
919        }
920    })
921    .await;
922
923    if disconnected.is_ok() {
924        return Ok(());
925    }
926
927    let live_references = Arc::strong_count(global).saturating_sub(1);
928    if live_references == 0 {
929        Ok(())
930    } else {
931        Err(live_references)
932    }
933}

936// TODO-HUMAN-REVIEW(PR-782): Review SaBRe RPC server shutdown errors.
937async fn stop_sabre_rpc_server<E>(
938    server_task: tokio::task::JoinHandle<Result<(), E>>,
939) -> Result<(), Error>
940where
941    E: std::fmt::Display,
942{
943    server_task.abort();
944    match server_task.await {
945        Err(error) if error.is_cancelled() => Ok(()),
946        Err(error) => Err(anyhow!("SaBRe coordinator task failed: {error}")),
947        Ok(Err(error)) => Err(anyhow!("SaBRe coordinator server failed: {error}")),
948        Ok(Ok(())) => Err(anyhow!("SaBRe coordinator server stopped unexpectedly")),
949    }
950}

953// TODO-HUMAN-REVIEW(PR-789): Review complete SaBRe RPC shutdown ordering.
954async fn shutdown_sabre_rpc<T, E>(
955    server_task: tokio::task::JoinHandle<Result<(), E>>,
956    global: &Arc<T>,
957    timeout: Duration,
958) -> Result<(), Error>
959where

968    })

992/// Run one command with the Detcore tool executing inside a SaBRe plugin and
993/// the single GlobalState held by this Hermit coordinator process.
994// AUTONOMOUS-BOT-IMPLEMENTED
995// TODO-HUMAN-REVIEW(PR-738): Review SaBRe coordinator lifetime and artifact loading.
996async fn run_sabre(
997    mut command: Command,
998    config: DetConfig,
999    print_summary: bool,
1000    print_summary_to_json_file: &Option<PathBuf>,
1001    capture_output: bool,
1002) -> Result<Output, Error> {
1003    let path_evidence_file = std::env::var_os(SABRE_PATH_EVIDENCE_ENV).map(PathBuf::from);
1004    let sabre = resolve_sabre_binary()?;
1005    let plugin = sabre_runtime_library_path()
1006        .map_err(|error| anyhow!("failed to locate the Detcore SaBRe plugin: {error}"))?;
1007    let program = command.find_program().map_err(|error| {
1008        anyhow!(
1009            "failed to resolve SaBRe guest executable {:?}: {error}",
1010            command.get_program()
1011        )
1012    })?;
1013    let staged_program = stage_sabre_program_in(&program, Path::new(SABRE_STAGING_DIRECTORY))?;
1014    let launch_program = staged_program
1015        .as_ref()
1016        .map_or(program.as_path(), |staged| staged.path.as_path());
1017
1018    let socket_dir = tempfile::Builder::new()
1019        .prefix("hermit-sabre-rpc-")
1020        .tempdir()?;
1021    let socket_path = socket_dir.path().join("coordinator.sock");
1022    let fallback_ready = Arc::new(AtomicBool::new(false));
1023    let global = Arc::new(detcore::GlobalState::init_global_state(&config).await);
1024    let server = reverie_rpc_transport::RpcServer::bind_with_readiness(
1025        &socket_path,
1026        global.clone(),
1027        config.clone(),
1028        fallback_ready.clone(),
1029    )
1030    .map_err(|error| anyhow!("failed to start SaBRe coordinator RPC: {error}"))?;
1031    let server_task = tokio::spawn(async move { server.serve().await });
1032
1033    command.prepend_args([
1034        plugin.as_os_str(),
1035        OsStr::new("--"),
1036        launch_program.as_os_str(),
1037    ]);
1038    command.program(&sabre);
1039    command.env(SABRE_RPC_SOCKET_ENV, &socket_path);
1040    // Publish the shape of the Config this coordinator will send. The plugin is a
1041    // separate artifact in the same target directory, so it can be stale without
1042    // looking it; comparing here turns an opaque decode failure at connect into a
1043    // message that names the mismatch. See detcore_model::config_wire_fingerprint.
1044    command.env(
1045        detcore::CONFIG_FINGERPRINT_ENV,
1046        detcore::config_wire_fingerprint(),
1047    );
1048    command.env_remove(SABRE_PATH_EVIDENCE_ENV);
1049    command.env_remove(SABRE_DETLOG_FORWARD_ENV);
1050    if tracing::enabled!(target: "detcore", tracing::Level::INFO) {
1051        command.env(SABRE_DETLOG_FORWARD_ENV, "1");
1052    }
1053    command.env_remove("SABRE_BINARY");
1054    command.env_remove("SABRE_PLUGIN");
1055
1056    // THE SOCKET PATH IS DELIBERATELY RANDOM AND MUST NOT BE COMPARED.
1057    //
1058    // `tempfile` appends a random 6-character suffix to the directory above, so
1059    // the absolute path differs on every launch by design -- that is what keeps
1060    // concurrent hermit invocations on a shared host from binding the same
1061    // coordinator socket. Emitting it at INFO put per-launch randomness into the
1062    // INFO stream, which `--verify-strict` compares:
1063    //
1064    //   Mismatch at log messages 2 (run 1) and 2 (run 2):
1065    //   < socket=/tmp/hermit-sabre-rpc-Gw6tUi/coordinator.sock
1066    //   > socket=/tmp/hermit-sabre-rpc-7SZ94p/coordinator.sock
1067    //
1068    // That fails EVERY SaBRe verify cell before the guest executes an
1069    // instruction, for a host-side value with no guest-visible content. The
1070    // defect was in the oracle, not the backend.
1071    //
1072    // WHY THE VALUE IS DROPPED FROM INFO RATHER THAN MADE DETERMINISTIC: the two
1073    // `--verify` runs execute in SEPARATE CHILD PROCESSES, so any path stable
1074    // across both would have to be derived from state shared between them. That
1075    // trades `tempfile`'s guaranteed-fresh directory for a reused one, and a
1076    // run1 that dies holding the socket would then make run2 fail to bind -- a
1077    // real new failure mode in exchange for a cosmetic field. The path keeps its
1078    // randomness; only the compared stream stops carrying it.
1079    //
1080    // This is NOT a comparator relaxation. No normalization envelope is widened,
1081    // so any future randomness leaking into this banner still fails the
1082    // comparison. The full path remains available one level down.
1083    tracing::debug!(
1084        target: "hermit::sabre",
1085        socket = %socket_path.display(),
1086        "SaBRe coordinator RPC socket (per-launch temp path, deliberately not at INFO)",
1087    );
1088    tracing::info!(
1089        target: "hermit::sabre",
1090        guest = %program.display(),
1091        plugin = %plugin.display(),
1092        "launching Detcore guest through SaBRe with coordinator RPC",
1093    );
1094
1095    let supervised = match sabre_ptrace::run(
1096        command.into_std_lossy(),
1097        PathBuf::from(&sabre),
1098        plugin.clone(),
1099        fallback_ready,
1100        global.clone(),
1101        capture_output,
1102    )
1103    .await
1104    {
1105        Ok(supervised) => supervised,
1106        Err(error) => {
1107            global.release_all_physical_process_exits();
1108            global.force_shutdown_with_error();
1109            let _ = shutdown_sabre_rpc(server_task, &global, SABRE_RPC_DISCONNECT_TIMEOUT).await;
1110            return Err(error);
1111        }
1112    };
1113    // The supervisor returns only after every tracee reached a final kernel wait status. Release
1114    // the root process's barrier (and any intentionally unreaped child barriers) before scheduler
1115    // shutdown; no guest thread remains that could race timer fast-forward here.
1116    global.release_all_physical_process_exits();
1117    // A SaBRe execution in which no guest thread ever reached the coordinator
1118    // is an execution in which Detcore was never loaded as a Reverie tool: the
1119    // guest ran on bare Linux. Treat it exactly like a failed run so the
1120    // scheduler is torn down here instead of blocking forever in `clean_up`
1121    // below, waiting for a guest thread that will never register.
1122    let detcore_never_engaged = !supervised.path_evidence.guest_rpc_observed;
1123    let requires_forced_shutdown = !supervised.status.success() || detcore_never_engaged;
1124    if requires_forced_shutdown {
1125        global.force_shutdown_with_error();
1126    }
1127    // Emit one unconditional, machine-readable per-run fact instead of a
1128    // free-form warning. SaBRe loads the guest interpreter before the Detcore
1129    // plugin, so pre-plugin loader syscalls are structurally absent from its
1130    // observation envelope. Versioning that fact lets scorecard consumers
1131    // reject old or incomplete records rather than infer coverage from silence.
1132    //
1133    // Keep this in the controller-diagnostic stream rather than writing
1134    // directly to the process stderr that also carries captured guest stderr.
1135    // WARN is Hermit's default tracing level, and `--log-file` can therefore
1136    // separate the unconditional controller fact from guest output without a
1137    // comparator exception. A missing record remains distinct from a record
1138    // whose counters are zero.
1139    let backend_evidence = sabre_backend_evidence_line(&supervised.path_evidence);
1140    tracing::warn!(target: "hermit::sabre", "{backend_evidence}");
1141    if let Some(path) = path_evidence_file {
1142        let mut file = fs::OpenOptions::new()
1143            .create(true)
1144            .append(true)
1145            .mode(0o600)
1146            .open(&path)
1147            .map_err(|error| {
1148                anyhow!(
1149                    "failed to open SaBRe path evidence {}: {error}",
1150                    path.display()
1151                )
1152            })?;
1153        serde_json::to_writer(&mut file, &supervised.path_evidence)?;
1154        writeln!(file)?;
1155    }
1156    let output = Output {
1157        status: supervised.status.into(),
1158        stdout: supervised.stdout,
1159        stderr: supervised.stderr,
1160    };
1161
1162    shutdown_sabre_rpc(server_task, &global, SABRE_RPC_DISCONNECT_TIMEOUT).await?;
1163    let mut global = Arc::try_unwrap(global).map_err(|global| {
1164        anyhow!(
1165            "SaBRe coordinator stopped with {} live RPC reference(s)",
1166            Arc::strong_count(&global) - 1
1167        )
1168    })?;
1169    if requires_forced_shutdown {
1170        global.cancel_internal_scheduler().await;
1171    }
1172    global
1173        .clean_up(print_summary, print_summary_to_json_file)
1174        .await;
1175    tracing::info!(
1176        target: "hermit::sabre::fallback",
1177        ptrace_fallback_sites = supervised.path_evidence.ptrace_fallback_sites,
1178        trusted_shared_object_sites = supervised.path_evidence.trusted_shared_object_sites,
1179        guest_rpc_observed = supervised.path_evidence.guest_rpc_observed,
1180        "SaBRe ptrace fallback completed",
1181    );
1182    if detcore_never_engaged {
1183        return Err(anyhow!(
1184            "{}",
1185            sabre_uninstrumented_guest_message(&output.status)
1186        ));
1187    }
1188    Ok(output)
1189}

1230/// Detcore intercepting anything, so the run carries no determinism guarantee
1231/// whatsoever -- its timing, scheduling, PIDs, and clock reads all came from the
1232/// host. Reporting that as a successful `hermit run` would be a fail-open
1233/// determinism hole, so the caller turns it into a hard error and this function
1234/// supplies the diagnosis.
1235///
1236/// The dominant cause is a guest whose syscall sites SaBRe never rewrote. A
1237/// statically linked ELF is the sharp edge: it has no dynamic loader and no
1238/// shared library through which SaBRe could regain control, so an unrewritten
1239/// static client runs entirely on bare Linux with no second chance.
1240fn sabre_uninstrumented_guest_message(status: &ExitStatus) -> String {
1241    format!(
1242        "the SaBRe backend finished ({status:?}) without the guest ever reaching the Detcore \
1243         coordinator: no syscall was intercepted, so this run applied no determinization at all \
1244         and its result is not a Hermit guarantee. This means SaBRe rewrote no syscall site in \
1245         the guest -- most often a statically linked ELF, whose syscall sites SaBRe must patch \
1246         in the client image itself because there is no dynamic loader to intercept."
1247    )
1248}

1250/// Guest-physical memory available to the single-process KVM personality.
1251// The KVM personality is a sparse MAP_NORESERVE address space. QEMU needs room
1252// for its own ELF mappings in addition to the nested machine's RAM mapping.
1253const KVM_GUEST_MEMORY_BYTES: usize = 1024 * 1024 * 1024;
1254
1255/// Maximum `#!` interpreter indirection levels, matching the Linux kernel's
1256/// `BINPRM_MAX_RECURSION` limit for chained script interpreters.
1257const MAX_SHEBANG_DEPTH: usize = 4;
1258
1259/// Resolve `#!` interpreter scripts before the reverie-kvm ELF loader runs.
1260///
1261/// The KVM ELF loader can only map ELF images, so a guest program that is
1262/// actually a `#!`-script (for example `/usr/local/bin/file` -> `#!/bin/bash`,
1263/// or `/usr/bin/pkg-config` -> `#!/usr/bin/sh`) must be rewritten to launch its
1264/// interpreter, exactly as the kernel's `execve(2)` `binfmt_script` handler
1265/// does. On success the returned image is an ELF and `argv` has the interpreter,
1266/// its shebang arguments, and the script path prepended in kernel order:
1267/// `[interp, shebang_args.., script_path, <original argv[1..]>]`.
1268///
1269/// The interpreter line is parsed with hermit's shared [`Shebang`] so the KVM
1270/// backend matches how the ptrace backend and recorder treat `#!`-scripts.
1271fn resolve_kvm_shebang(
1272    resolved_program: &Path,
1273    mut argv: Vec<String>,
1274) -> Result<(PathBuf, Vec<String>, Vec<u8>), Error> {
1275    let mut load_path = resolved_program.to_path_buf();
1276    let mut image = fs::read(&load_path)
1277        .map_err(|error| anyhow!("failed to read KVM guest executable {load_path:?}: {error}"))?;
1278
1279    let mut depth = 0;
1280    while image.starts_with(b"#!") {
1281        depth += 1;
1282        if depth > MAX_SHEBANG_DEPTH {
1283            return Err(anyhow!(
1284                "too many levels of `#!` interpreter indirection loading {resolved_program:?}"
1285            ));
1286        }
1287        let (interpreter, shebang_args) = Shebang::from_buf(&image)
1288            .ok_or_else(|| anyhow!("malformed `#!` interpreter line in {load_path:?}"))?
1289            .into_parts();
1290        let interpreter_str = interpreter
1291            .to_str()
1292            .ok_or_else(|| anyhow!("non-UTF-8 `#!` interpreter path in {load_path:?}"))?
1293            .to_owned();
1294
1295        // Rewrite argv in kernel order. The prior argv[0] (the script's own
1296        // name) is dropped on the first level; on deeper levels the previous
1297        // interpreter path is preserved as a positional argument, matching
1298        // `binfmt_script`.
1299        let mut rewritten = Vec::with_capacity(argv.len() + shebang_args.len() + 2);
1300        rewritten.push(interpreter_str);
1301        for arg in &shebang_args {
1302            rewritten.push(
1303                arg.to_str()
1304                    .ok_or_else(|| anyhow!("non-UTF-8 `#!` interpreter argument in {load_path:?}"))?
1305                    .to_owned(),
1306            );
1307        }
1308        rewritten.push(load_path.to_string_lossy().into_owned());
1309        rewritten.extend_from_slice(&argv[1..]);
1310        argv = rewritten;
1311
1312        load_path = interpreter;
1313        image = fs::read(&load_path).map_err(|error| {
1314            anyhow!(
1315                "failed to read `#!` interpreter {load_path:?} for {resolved_program:?}: {error}"
1316            )
1317        })?;
1318    }
1319
1320    Ok((load_path, argv, image))
1321}

1323/// Dispatch a command onto the real reverie-kvm Tool runtime.
1324async fn run_kvm(
1325    command: &Command,
1326    mut config: DetConfig,
1327    print_summary: bool,
1328    print_summary_to_json_file: &Option<PathBuf>,
1329    capture_output: bool,
1330) -> Result<Output, Error> {
1331    let dispatch_started = Instant::now();
1332    let stdin = reserved_kvm_stdin()?;
1333    let requested_cwd = command
1334        .get_current_dir()
1335        .map(Path::to_owned)
1336        .unwrap_or(std::env::current_dir()?);
1337    let cwd = fs::canonicalize(&requested_cwd).map_err(|error| {
1338        anyhow!(
1339            "failed to resolve KVM guest working directory {:?}: {error}",
1340            requested_cwd
1341        )
1342    })?;
1343    let program = command
1344        .get_program()
1345        .to_str()
1346        .ok_or_else(|| anyhow!("KVM guest executable path is not valid UTF-8"))?
1347        .to_owned();
1348    if !cwd.is_dir() {
1349        return Err(anyhow!(
1350            "KVM guest working directory is not a directory: {:?}",
1351            cwd
1352        ));
1353    }
1354    let resolved_program = command.find_program().map_err(|error| {
1355        anyhow!("failed to resolve KVM guest executable {program:?} in the guest PATH: {error}")
1356    })?;
1357    let mut argv = Vec::with_capacity(1 + command.get_args().count());
1358    argv.push(program.clone());
1359    for argument in command.get_args() {
1360        argv.push(
1361            argument
1362                .to_str()
1363                .ok_or_else(|| anyhow!("KVM guest argument is not valid UTF-8"))?
1364                .to_owned(),
1365        );
1366    }
1367
1368    // Rewrite `#!`-scripts to their interpreter before the ELF loader sees them.
1369    let (_interpreter_path, argv, image) = resolve_kvm_shebang(&resolved_program, argv)?;
1370    // After shebang resolution the executable is the interpreter (argv[0]).
1371    let program = argv.first().cloned().unwrap_or(program);
1372    let envp = command
1373        .get_captured_envs()
1374        .into_iter()
1375        .map(|(key, value)| {
1376            let key = key
1377                .to_str()
1378                .ok_or_else(|| anyhow!("KVM guest environment key is not valid UTF-8"))?;
1379            let value = value
1380                .to_str()
1381                .ok_or_else(|| anyhow!("KVM guest environment value is not valid UTF-8"))?;
1382            Ok(format!("{key}={value}"))
1383        })
1384        .collect::<Result<Vec<_>, Error>>()?;
1385    tracing::info!(
1386        target: "hermit::kvm",
1387        program = %program,
1388        argv = ?argv,
1389        cwd = %cwd.display(),
1390        env_count = envp.len(),
1391        "launching guest through reverie-kvm",
1392    );
1393    let argv = argv.iter().map(String::as_str).collect::<Vec<_>>();
1394    let envp = envp.iter().map(String::as_str).collect::<Vec<_>>();
1395
1396    let setup_started = Instant::now();
1397    config.cpuid_virtualized_by_backend = true;
1398    config.backend_supports_madvise = false;
1399    // KVM does not enter Hermit's UTS namespace, so Detcore must provide the
1400    // same synthetic identity that the namespace-backed ptrace path exposes.
1401    // TODO-HUMAN-REVIEW(PR-998): Review KVM UTS namespace parity.
1402    config.has_uts_namespace = false;
1403    let random_seed = config.rng_seed();
1404    let mut backend = reverie_kvm::KvmBackend::new_with_stdin(KVM_GUEST_MEMORY_BYTES, stdin)
1405        .map_err(|error| anyhow!("failed to initialize reverie-kvm: {error}"))?;
1406    // AUTONOMOUS-BOT-IMPLEMENTED
1407    // TODO-HUMAN-REVIEW(PR-1120): Review KVM's canonical Detcore root identity.
1408    backend
1409        .set_root_pid(detcore::ROOT_DETPID.as_raw())
1410        .map_err(|error| anyhow!("failed to configure KVM root PID: {error}"))?;
1411    backend
1412        .install_static_elf_with_context(&image, &argv, &envp, &cwd)
1413        .map_err(|error| anyhow!("failed to load KVM guest executable {program:?}: {error}"))?;
1414    backend
1415        .set_random_seed(random_seed)
1416        .map_err(|error| anyhow!("failed to configure KVM guest random seed: {error}"))?;
1417    // The KVM backend now defaults to Tool-owned guest threads: CLONE_THREAD
1418    // workers are driven through the Detcore tool loop and their futex/CLEARTID
1419    // synchronization routes to Detcore, matching the golden ptrace model
1420    // ("follow children"). Detcore's own clone logic treats a CLONE_THREAD as
1421    // backend-uninstrumented iff `backend_dispatches_thread_tools` is false (see
1422    // detcore/src/syscalls/threads.rs), so in exactly that case opt the backend
1423    // out to host-owned threads to keep worker execution and futex ownership in
1424    // one synchronization domain (mixing them deadlocks pthread_join). In the
1425    // default (true) case the backend already follows children, so no call is
1426    // needed.
1427    if !config.backend_dispatches_thread_tools {
1428        backend.unmonitored_threads();
1429    }
1430
1431    let execution_started = Instant::now();
1432    let (global_state, code, stdout, stderr) = backend
1433        .run_static_elf_with_tool::<Detcore>(config, capture_output)
1434        .await
1435        .map_err(|error| anyhow!("KVM guest execution failed: {error}"))?;
1436    let cleanup_started = Instant::now();
1437    global_state
1438        .clean_up(print_summary, print_summary_to_json_file)
1439        .await;
1440    let teardown_started = Instant::now();
1441    // Drop explicitly so the host's KVM VM teardown cost remains observable.
1442    drop(backend);
1443    let teardown_finished = Instant::now();
1444
1445    tracing::info!(
1446        target: "hermit::kvm",
1447        prepare_us = setup_started.duration_since(dispatch_started).as_micros() as u64,
1448        setup_us = execution_started.duration_since(setup_started).as_micros() as u64,
1449        execution_us = cleanup_started.duration_since(execution_started).as_micros() as u64,
1450        cleanup_us = teardown_started.duration_since(cleanup_started).as_micros() as u64,
1451        teardown_us = teardown_finished.duration_since(teardown_started).as_micros() as u64,
1452        lifecycle_us = teardown_finished.duration_since(dispatch_started).as_micros() as u64,
1453        "reverie-kvm lifecycle phase timings",
1454    );
1455
1456    if !capture_output {
1457        std::io::stdout().write_all(&stdout)?;
1458        std::io::stderr().write_all(&stderr)?;
1459    }
1460
1461    Ok(Output {
1462        status: ExitStatus::Exited(code),
1463        stdout,
1464        stderr,
1465    })
1466}

1474// AUTONOMOUS-BOT-IMPLEMENTED
1475// TODO-HUMAN-REVIEW(PR-737): Review public DBT dispatch and child environment ownership.
1476/// Dispatch a command onto the Detcore-linked reverie-dbt runtime.
1477#[cfg(feature = "dbt")]
1478async fn run_dbt(
1479    command: Command,
1480    config: DetConfig,
1481    print_summary: bool,
1482    capture_output: bool,
1483) -> Result<Output, Error> {
1484    if !config.sequentialize_threads {
1485        return Err(anyhow!(
1486            "the dbt backend requires sequentialized threads; remove \
1487             --no-sequentialize-threads (or --strace-only) to run under --backend dbt"
1488        ));
1489    }
1490
1491    let config_json = serde_json::to_string(&config)
1492        .map_err(|error| anyhow!("failed to serialize the Detcore config for DBT: {error}"))?;
1493    let panic_on_unsupported_syscalls = config.panic_on_unsupported_syscalls;
1494    let (drrun, client) = detcore_dbt::prepare_native_client()
1495        .map_err(|error| anyhow!("failed to prepare the Detcore DynamoRIO client: {error}"))?;
1496    let mut runner = reverie_dbt::DbtRunner::new(&drrun, &client)
1497        .map_err(|error| {
1498            anyhow!(
1499                "failed to configure the DynamoRIO DBT runner (drrun={}, client={}): {error}",
1500                drrun.display(),
1501                client.display()
1502            )
1503        })?
1504        .summary(print_summary)
1505        .isolated_process_group(panic_on_unsupported_syscalls);
1506    if panic_on_unsupported_syscalls {
1507        runner = runner.client_argument("-panic-on-unsupported-syscalls");
1508    }
1509
1510    let program = command.get_program().to_owned();
1511    let mut environment = command.get_captured_envs();
1512    environment.insert(detcore_dbt::DETCONFIG_ENV.into(), config_json.into());
1513    let guest = command.into_std_lossy();
1514    tracing::info!(
1515        target: "hermit::dbt",
1516        program = ?program,
1517        drrun = %drrun.display(),
1518        client = %client.display(),
1519        "launching guest through reverie-dbt with Detcore<DbtGuest>",
1520    );
1521
1522    if capture_output {
1523        let launch = || {
1524            runner
1525                .output_with_environment(&guest, &environment)
1526                .map_err(|error| anyhow!("failed to launch drrun ({}): {error}", drrun.display()))
1527        };
1528        let mut output = launch()?;
1529        if dbt_client_thread_start_failed(&output.status) {
1530            tracing::warn!(
1531                target: "hermit::dbt",
1532                "DynamoRIO client thread failed before guest start; retrying once",
1533            );
1534            output = launch()?;
1535        }
1536        return Ok(Output {
1537            status: output.status.into(),
1538            stdout: output.stdout,
1539            stderr: output.stderr,
1540        });
1541    }
1542
1543    let launch = || {
1544        runner
1545            .status_with_environment(&guest, &environment)
1546            .map_err(|error| anyhow!("failed to launch drrun ({}): {error}", drrun.display()))
1547    };
1548    let mut status = launch()?;
1549    if dbt_client_thread_start_failed(&status) {
1550        tracing::warn!(
1551            target: "hermit::dbt",
1552            "DynamoRIO client thread failed before guest start; retrying once",
1553        );
1554        status = launch()?;
1555    }
1556    Ok(Output {
1557        status: status.into(),
1558        stdout: Vec::new(),
1559        stderr: Vec::new(),
1560    })
1561}

1637async fn run_with_backend_inner(
1638    command: Command,
1639    config: DetConfig,
1640    print_summary: bool,
1641    print_summary_to_json_file: &Option<PathBuf>,
1642    backend: Backend,

1654    }

1657        {
1658            return Ok(run_dbt(command, config, print_summary, false).await?.status);
1659        }
1660        #[cfg(not(feature = "dbt"))]
1661        {
1662            backend.ensure_available()?;
1663            unreachable!("DBT availability must fail when the feature is disabled");
1664        }
1665    }
1666    if backend == Backend::Sabre {
1667        return Ok(run_sabre(
1668            command,
1669            config,
1670            print_summary,
1671            print_summary_to_json_file,
1672            false,
1673        )
1674        .await?
1675        .status);
1676    }
1677    if backend == Backend::Liteinst {
1678        let preload = liteinst_runtime_library_path()?;
1679        let (exit_status, mut global_state) =
1680            reverie_liteinst::LiteinstBackend::run_host_with_preload::<Detcore>(
1681                command, config, preload,
1682            )
1683            .await?;
1684        if liteinst_requires_forced_shutdown(exit_status) {
1685            global_state.force_shutdown_with_error();
1686            global_state.cancel_internal_scheduler().await;
1687        }
1688        global_state
1689            .clean_up(print_summary, print_summary_to_json_file)
1690            .await;
1691        return Ok(exit_status);
1692    }
1693    ensure_backend_dispatch(backend)?;
1694
1695    let stats_request = backend_stats::request();
1696    let mut builder = reverie_ptrace::TracerBuilder::<Detcore>::new(command).config(config.clone());
1697    if config.gdbserver {
1698        builder = builder.gdbserver(config.gdbserver_port);
1699    }
1700    let (exit_status, global_state) = builder.spawn().await?.wait().await?;
1701    global_state
1702        .clean_up(print_summary, print_summary_to_json_file)
1703        .await; // Before it's dropped by this function.
1704    backend_stats::report(backend, stats_request, &backend_stats::PtraceStatsSource);
1705    Ok(exit_status)
1706}

1830    // expect this to exist in any of the functions that are called.

1834fn collect_recording_ids(
1835    entries: impl IntoIterator<Item = io::Result<fs::DirEntry>>,
1836    data_dir: &Path,
1837    mut file_type: impl FnMut(&fs::DirEntry) -> io::Result<fs::FileType>,
1838) -> Result<Vec<Id>, Error> {
1839    let mut recordings = Vec::new();
1840    for entry in entries {
1841        let entry = entry.with_context(|| {
1842            format!(
1843                "Failed to read an entry in recording inventory {}",
1844                data_dir.display()
1845            )
1846        })?;
1847        let path = entry.path();
1848        if file_type(&entry)
1849            .with_context(|| {
1850                format!(
1851                    "Failed to inspect recording inventory entry {}",
1852                    path.display()
1853                )
1854            })?
1855            .is_dir()
1856            && let Some(id) = entry
1857                .file_name()
1858                .to_str()
1859                .and_then(|name| name.parse::<Id>().ok())
1860        {
1861            recordings.push(id);
1862        }
1863    }
1864    Ok(recordings)
1865}

2008    /// Deletes a recording.
2009    pub fn remove(&self, id: Id) -> Result<(), Error> {
2010        let path = self.data_dir.join(id.to_string());
2011
2012        // Before deleting anything, make sure this file exists. This may not be a
2013        // recording if this file does not exist.
2014        let metadata_path = path.join(METADATA_NAME);
2015        let metadata = fs::metadata(&metadata_path)
2016            .with_context(|| format!("Failed to find {:?}", metadata_path))?;

2021
2022        // Do a recursive delete on the directory. Note that this does not follow
2023        // symlinks.
2024        fs::remove_dir_all(path)?;
2025
2026        Ok(())
2027    }
