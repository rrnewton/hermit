16/// A serializable error. This is useful for sending an error to the parent
17/// process. This works by converting an error into a string via its `Display`
18/// implementation. Although we lose type information in the process of
19/// converting to a string, this preserves the error message and its error chain.
20#[derive(Debug, Clone, Copy, Default, Serialize, Deserialize, Eq, PartialEq)]
21pub enum FailureKind {
22    /// The child reported a failure it decided to report.
23    #[default]
24    Error,
25    /// The child PANICKED and the panic was caught and reported.
26    ///
27    /// ⚠️ THIS IS THE ONE BIT THAT PROSE CANNOT CARRY. A caught panic and a
28    /// reported error arrive at the parent as the same shape -- a message and a
29    /// chain of causes -- so without a discriminant the parent must either
30    /// match on English or call them the same thing. It called them the same
31    /// thing, which is why a tracer panic and a bad flag were indistinguishable.
32    Panic,
33    /// The child stopped because a FAIL-CLOSED POLICY refused the run.
34    ///
35    /// ⚠️ A REFUSAL AND A REPORTED ERROR ARE THE SAME SHAPE HERE TOO, which is
36    /// the identical argument that justified `Panic`. Record mode sets
37    /// `exit_on_unsupported_syscall` and `shutdown_on_unsupported_syscall:
38    /// false`, so it returns a typed `UnsupportedSyscallError` through Reverie
39    /// instead of calling `unrecoverable_shutdown`. That path never produces an
40    /// exit status of its own, so the status channel -- which is what carries
41    /// `HERMIT_POLICY_REFUSAL_EXIT` on the run path -- has nothing to say, and
42    /// the refusal arrived indistinguishable from "hermit broke": exit 125,
43    /// `class=container-child-exit`. Two configurations of one policy reported
44    /// opposite things about the same decision.
45    PolicyRefusal,
46    /// A ptrace PMU timer interrupt arrived after its target. The count is
47    /// carried across the container boundary so the parent can publish the
48    /// named infrastructure error instead of leaving the pending no-result.
49    SkidOvershoot { count: u64 },
50    /// The child stopped because HERMIT'S OWN `--timeout` BOUND was reached.
51    ///
52    /// ⚠️ THE SAME ARGUMENT A THIRD TIME, and it is not hypothetical: measured
53    /// on 2026-08-26, `hermit run --timeout 3 -- ./spin` fired at exactly 3s
54    /// with no residue and still reported `exit 125` /
55    /// `class=cli-error` -- "hermit broke" -- because `GuestTimedOut` is
56    /// produced INSIDE the container child and every type is a string by the
57    /// time the parent sees it. Unlike the refusal path there is no status
58    /// channel to fall back on either: the timeout unwinds and RETURNS rather
59    /// than calling `unrecoverable_shutdown`, precisely because the unwind is
60    /// the point.
61    RunTimeout,
62}

64/// ⚠️ THE `kind` FIELD IS THE THIRD AND LAST OF ONE CHAIN OF FLATTENINGS.
65/// The same property -- the failure's CLASS -- was destroyed at three different
66/// boundaries, in series on one path:
67///
68///   1. `main`'s `unwrap_or_else` mapped EVERY error to one exit code;
69///   2. `with_container` flattened reverie's typed `RunError::ExitStatus` into
70///      opaque prose with `.context(..)?`;
71///   3. THIS type crosses a process boundary carrying only strings.
72///
73/// They are three distinct boundaries with three mechanisms, but they answer one
74/// question, so one discriminant serves all three. Preserving it here is what
75/// lets the parent classify without parsing English.
76///
77/// Extending a serialized type is safe HERE specifically because the writer and
78/// the reader are the SAME BINARY IMAGE -- hermit's container child and its own
79/// parent, over a fork-local pipe -- so there is no cross-version compatibility
80/// surface.
81///
82/// ⚠️ `serde(default)` BELOW IS NOT A COMPATIBILITY GUARANTEE, and must not be
83/// read as one. reverie serializes this with `bincode::config::legacy()`
84/// (`reverie-process/src/container.rs:861`), which is NOT self-describing: a
85/// record written by an older binary with two fields would fail to decode
86/// outright rather than defaulting the third. The attribute is harmless and
87/// keeps the field optional at the Rust level; the property that actually makes
88/// this safe is the same-image one above.
89#[derive(Debug, Clone, Serialize, Deserialize, Eq, PartialEq)]
90pub struct SerializableError {
91    /// The main error.
92    error: String,
93    /// The chain of causes. This is empty if there are no associated causes.
94    context: Vec<String>,
95    /// What CLASS of failure this was. See [`FailureKind`].
96    #[serde(default)]
97    kind: FailureKind,
98}
99
100impl SerializableError {
101    /// The class this failure was reported as.
102    pub fn kind(&self) -> FailureKind {
103        self.kind
104    }
105
106    /// Re-tag an error as a caught panic. Called at the catch site, which is the
107    /// only place that still knows a panic is what happened.
108    pub fn into_panic(mut self) -> Self {
109        self.kind = FailureKind::Panic;
110        self
111    }
112}

114/// Is this error a fail-closed POLICY REFUSAL rather than something that broke?
115///
116/// ⚠️ THE CHAIN WALK ALONE DOES NOT FIND IT, AND I SHIPPED A FIX THAT ONLY DID THAT.
117/// `reverie::Error::Tool` is declared `#[error(transparent)]`, which forwards
118/// `source()` to the inner `anyhow::Error`'s source -- PAST that error's own value.
119/// `UnsupportedSyscallError` IS that value, so it never appears in
120/// `err.chain()` and `.is::<_>()` is false at every link. Reproduced in isolation:
121/// a one-entry chain whose Display is right and whose `is::<T>()` is false.
122///
123/// The consequence was that record mode still reported exit 125
124/// `class=cli-error` -- measured against `tests/c/dbt_unsupported_syscall.c`, the
125/// path this whole change exists to fix, unchanged. The in-process test passed
126/// because it constructs the error directly and never crosses the wrapper.
127/// agent(hermit-001)'s codex lane asked for the end-to-end cell that would have
128/// caught it; the cell was absent because the fix did not work.
129///
130/// ⚠️ IT IS STILL A TYPE TEST, NOT A STRING MATCH. Matching the Display text would
131/// be the thing `kind` exists to avoid, and it would break the moment the message
132/// is reworded. This reaches through the one wrapper that hides the type.
133fn is_policy_refusal(err: &Error) -> bool {
134    if err
135        .chain()
136        .any(|cause| cause.is::<detcore::UnsupportedSyscallError>())
137    {
138        return true;
139    }
140    // The wrapper case: reverie hands the tool's error back inside `Tool`, whose
141    // transparent forwarding hides the value from the chain above.
142    err.chain().any(|cause| {
143        cause
144            .downcast_ref::<reverie::Error>()
145            .and_then(|reverie_error| match reverie_error {
146                reverie::Error::Tool(inner) => {
147                    inner.downcast_ref::<detcore::UnsupportedSyscallError>()
148                }
149                _ => None,
150            })
151            .is_some()
152    })
153}

155impl From<Error> for SerializableError {
156    fn from(err: Error) -> Self {
157        let error = err.to_string();
158        let context = err.chain().skip(1).map(ToString::to_string).collect();
159        // ⚠️ CLASSIFIED HERE BECAUSE THIS IS THE LAST PLACE THE TYPE EXISTS.
160        // Everything below this line is strings: `From<SerializableError> for
161        // Error` rebuilds the chain with `Error::msg`, so a downcast on the far
162        // side can only ever fail. Detecting the refusal after the boundary
163        // would mean matching on English, which is the thing `kind` exists to
164        // avoid.
165        let kind = if let Some(error) = err.downcast_ref::<crate::SkidOvershootError>() {
166            FailureKind::SkidOvershoot {
167                count: error.count(),
168            }
169        } else if is_policy_refusal(&err) {
170            FailureKind::PolicyRefusal
171        } else if err.downcast_ref::<crate::GuestTimedOut>().is_some() {
172            FailureKind::RunTimeout
173        } else {
174            FailureKind::Error
175        };
176        Self {
177            error,
178            context,
179            kind,
180        }
181    }
182}
183
184impl From<SerializableError> for Error {
185    fn from(mut err: SerializableError) -> Self {
186        if let Some(root_cause) = err.context.pop() {
187            let mut error = Self::msg(root_cause);
188
189            while let Some(context) = err.context.pop() {
190                error = error.context(context);
191            }
192
193            error.context(err.error)
194        } else {
195            Self::msg(err.error)
196        }
197    }
198}

200#[cfg(test)]
201mod tests {
202    use super::*;
203
204    #[test]
205    fn into_serializable_error() {
206        let error = Error::msg("root cause")
207            .context("a")
208            .context("b")
209            .context("c");
210
211        assert_eq!(
212            SerializableError::from(error),
213            SerializableError {
214                error: "c".into(),
215                context: vec!["b".into(), "a".into(), "root cause".into(),],
216                kind: FailureKind::Error,
217            }
218        );
219    }
220
221    #[test]
222    fn from_serializable_error() {
223        let error = Error::from(SerializableError {
224            error: "c".into(),
225            context: vec!["b".into(), "a".into(), "root cause".into()],
226            kind: FailureKind::Error,
227        });
228
229        assert_eq!(
230            error
231                .chain()
232                .map(ToString::to_string)
233                .collect::<Vec<String>>(),
234            ["c", "b", "a", "root cause"]
235                .iter()
236                .map(ToString::to_string)
237                .collect::<Vec<String>>(),
238        )
239    }
240
241    #[test]
242    fn skid_overshoot_is_serialized_as_a_policy_refusal() {
243        let refusal = SerializableError::from(Error::new(crate::SkidOvershootError::new(2)));
244        assert_eq!(refusal.kind(), FailureKind::SkidOvershoot { count: 2 });
245        assert!(refusal.error.contains("observed 2 HERMIT_SKID_OVERSHOOT"));
246
247        let contextual = SerializableError::from(
248            Error::msg("guest execution also failed").context(crate::SkidOvershootError::new(1)),
249        );
250        assert_eq!(contextual.kind(), FailureKind::SkidOvershoot { count: 1 });
251        assert!(
252            contextual
253                .error
254                .contains("observed 1 HERMIT_SKID_OVERSHOOT")
255        );
256
257        let ordinary = SerializableError::from(Error::msg("ordinary failure"));
258        assert_eq!(ordinary.kind(), FailureKind::Error);
259    }
