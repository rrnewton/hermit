16/// A serializable error. This is useful for sending an error to the parent
17/// process. This works by converting an error into a string via its `Display`
18/// implementation. Although we lose type information in the process of
19/// converting to a string, this preserves the error message and its error chain.
20#[derive(Debug, Clone, Copy, Default, Serialize, Deserialize, Eq, PartialEq)]
21pub enum FailureKind {
22    /// The child reported a failure it decided to report.
23    #[default]
24    Error,
25    /// The child PANICKED and the panic was caught and reported.
26    ///
27    /// ⚠️ THIS IS THE ONE BIT THAT PROSE CANNOT CARRY. A caught panic and a
28    /// reported error arrive at the parent as the same shape -- a message and a
29    /// chain of causes -- so without a discriminant the parent must either
30    /// match on English or call them the same thing. It called them the same
31    /// thing, which is why a tracer panic and a bad flag were indistinguishable.
32    Panic,
33    /// The child stopped because a FAIL-CLOSED POLICY refused the run.
34    ///
35    /// ⚠️ A REFUSAL AND A REPORTED ERROR ARE THE SAME SHAPE HERE TOO, which is
36    /// the identical argument that justified `Panic`. Record mode sets
37    /// `exit_on_unsupported_syscall` and `shutdown_on_unsupported_syscall:
38    /// false`, so it returns a typed `UnsupportedSyscallError` through Reverie
39    /// instead of calling `unrecoverable_shutdown`. That path never produces an
40    /// exit status of its own, so the status channel -- which is what carries
41    /// `HERMIT_POLICY_REFUSAL_EXIT` on the run path -- has nothing to say, and
42    /// the refusal arrived indistinguishable from "hermit broke": exit 125,
43    /// `class=container-child-exit`. Two configurations of one policy reported
44    /// opposite things about the same decision.
45    PolicyRefusal,
46    /// A ptrace PMU timer interrupt arrived after its target. The count is
47    /// carried across the container boundary so the parent can publish the
48    /// named infrastructure error instead of leaving the pending no-result.
49    SkidOvershoot { count: u64 },
50    /// The child stopped because HERMIT'S OWN `--timeout` BOUND was reached.
51    ///
52    /// ⚠️ THE SAME ARGUMENT A THIRD TIME, and it is not hypothetical: measured
53    /// on 2026-08-26, `hermit run --timeout 3 -- ./spin` fired at exactly 3s
54    /// with no residue and still reported `exit 125` /
55    /// `class=cli-error` -- "hermit broke" -- because `GuestTimedOut` is
56    /// produced INSIDE the container child and every type is a string by the
57    /// time the parent sees it. Unlike the refusal path there is no status
58    /// channel to fall back on either: the timeout unwinds and RETURNS rather
59    /// than calling `unrecoverable_shutdown`, precisely because the unwind is
60    /// the point.
61    RunTimeout,
62    /// The backend failed while running the guest ([`crate::BackendRunFailure`]).
63    /// Carried so the parent can record the verification run as failed by its
64    /// backend instead of leaving the pending no-result stamp.
65    BackendRunFailure,
66    /// The run ended with a scheduler terminal outcome
67    /// ([`crate::SchedulerEndedRun`]): a deadlock under Hermit's model, or its
68    /// virtual clock passed `--virtual-timeout`. Its own outcome, neither a
69    /// guest result nor an internal failure. Carried for the reason
70    /// `RunTimeout` is: the child reports it through this channel, after the
71    /// run was unwound and finished, rather than through the outcome's status,
72    /// which it exits with when its backend cannot unwind the guest.
73    TerminalOutcome(detcore::TerminalOutcome),
74}

76/// ⚠️ THE `kind` FIELD IS THE THIRD AND LAST OF ONE CHAIN OF FLATTENINGS.
77/// The same property -- the failure's CLASS -- was destroyed at three different
78/// boundaries, in series on one path:
79///
80///   1. `main`'s `unwrap_or_else` mapped EVERY error to one exit code;
81///   2. `with_container` flattened reverie's typed `RunError::ExitStatus` into
82///      opaque prose with `.context(..)?`;
83///   3. THIS type crosses a process boundary carrying only strings.
84///
85/// They are three distinct boundaries with three mechanisms, but they answer one
86/// question, so one discriminant serves all three. Preserving it here is what
87/// lets the parent classify without parsing English.
88///
89/// Extending a serialized type is safe HERE specifically because the writer and
90/// the reader are the SAME BINARY IMAGE -- hermit's container child and its own
91/// parent, over a fork-local pipe -- so there is no cross-version compatibility
92/// surface.
93///
94/// ⚠️ `serde(default)` BELOW IS NOT A COMPATIBILITY GUARANTEE, and must not be
95/// read as one. reverie serializes this with `bincode::config::legacy()`
96/// (`reverie-process/src/container.rs:861`), which is NOT self-describing: a
97/// record written by an older binary with two fields would fail to decode
98/// outright rather than defaulting the third. The attribute is harmless and
99/// keeps the field optional at the Rust level; the property that actually makes
100/// this safe is the same-image one above.
101#[derive(Debug, Clone, Serialize, Deserialize, Eq, PartialEq)]
102pub struct SerializableError {
103    /// The main error.
104    error: String,
105    /// The chain of causes. This is empty if there are no associated causes.
106    context: Vec<String>,
107    /// What CLASS of failure this was. See [`FailureKind`].
108    #[serde(default)]
109    kind: FailureKind,
110    /// An unresolved owner is separate from the primary failure class. This
111    /// same-image container field is not a cross-version wire compatibility promise.
112    cleanup_stage: Option<crate::HermitCleanupStage>,
113}
114
115impl SerializableError {
116    /// The class this failure was reported as.
117    pub fn kind(&self) -> FailureKind {
118        self.kind
119    }
120
121    /// The unresolved owner phase, if cleanup has not actually completed.
122    pub fn cleanup_stage(&self) -> Option<crate::HermitCleanupStage> {
123        self.cleanup_stage
124    }
125
126    /// Re-tag an error as a caught panic. Called at the catch site, which is the
127    /// only place that still knows a panic is what happened.
128    pub fn into_panic(mut self) -> Self {
129        self.kind = FailureKind::Panic;
130        self
131    }

134/// Is this error a fail-closed POLICY REFUSAL rather than something that broke?
135///
136/// ⚠️ THE CHAIN WALK ALONE DOES NOT FIND IT, AND I SHIPPED A FIX THAT ONLY DID THAT.
137/// `reverie::Error::Tool` is declared `#[error(transparent)]`, which forwards
138/// `source()` to the inner `anyhow::Error`'s source -- PAST that error's own value.
139/// `UnsupportedSyscallError` IS that value, so it never appears in
140/// `err.chain()` and `.is::<_>()` is false at every link. Reproduced in isolation:
141/// a one-entry chain whose Display is right and whose `is::<T>()` is false.
142///
143/// The consequence was that record mode still reported exit 125
144/// `class=cli-error` -- measured against `tests/c/dbt_unsupported_syscall.c`, the
145/// path this whole change exists to fix, unchanged. The in-process test passed
146/// because it constructs the error directly and never crosses the wrapper.
147/// agent(hermit-001)'s codex lane asked for the end-to-end cell that would have
148/// caught it; the cell was absent because the fix did not work.
149///
150/// ⚠️ IT IS STILL A TYPE TEST, NOT A STRING MATCH. Matching the Display text would
151/// be the thing `kind` exists to avoid, and it would break the moment the message
152/// is reworded. This reaches through the one wrapper that hides the type.
153fn is_policy_refusal(err: &Error) -> bool {
154    // Detcore's refusal of an unsupported syscall, and reverie-ptrace's end of
155    // a run in which the guest made a syscall through Reverie's untraced
156    // private-page stub, which no tool saw
157    // (https://github.com/rrnewton/reverie/issues/1064).
158    fn refuses(cause: &(dyn std::error::Error + 'static)) -> bool {
159        cause.is::<detcore::UnsupportedSyscallError>()
160            || cause.is::<detcore::TerminalReplayRefusal>()
161            || cause.is::<detcore::PriorityHistoryRefusal>()
162            || cause.is::<reverie_ptrace::UntracedStubSyscall>()
163            || is_chosen_root_id_taken(cause)
164    }
165    // The guest's first process could not have its chosen ID
166    // (`--no-namespace`, `Command::pid`): another task took it after Hermit
167    // checked, such as a run in another network namespace, which the window
168    // reservation does not cover. Linux's clone never fails with EEXIST but
169    // for a requested ID, so this is that refusal.
170    fn is_chosen_root_id_taken(cause: &(dyn std::error::Error + 'static)) -> bool {
171        cause
172            .downcast_ref::<reverie::process::Error>()
173            .is_some_and(|error| {
174                error.errno() == reverie::syscalls::Errno::EEXIST
175                    && error.context() == reverie::process::Context::Clone
176            })
177    }
178    if err.chain().any(refuses) {
179        return true;
180    }
181    // The wrapper case: reverie hands the tool's error back inside `Tool`, whose
182    // transparent forwarding hides the value from the chain above.
183    err.chain().any(|cause| {
184        cause
185            .downcast_ref::<reverie::Error>()
186            .is_some_and(|reverie_error| match reverie_error {
187                reverie::Error::Tool(inner) => inner.chain().any(refuses),
188                _ => false,
189            })
190    })
191}

193/// Classify only the backend's primary cause. Its secondary errors and failed
194/// cleanup observations are deliberately unreachable from this function.
195pub(crate) fn classify_ptrace_primary(error: &reverie::Error) -> FailureKind {
196    match error {
197        // A terminal outcome completes as its own outcome only when its
198        // teardown was clean (`ptrace_completion::clean_terminal_outcome`).
199        // Left as the primary of a failed run, the teardown failed or never
200        // confirmed, which is Hermit's failure.
201        reverie::Error::Tool(inner) if inner.is::<crate::SchedulerEndedRun>() => FailureKind::Error,
202        reverie::Error::Tool(inner) => classify_unwrapped(inner),
203        _ => FailureKind::Error,
204    }
205}
206
207// Follow only the primary of the finalization pair, including Tool's
208// transparent wrapper. Ordinary anyhow downcasts do not traverse its source().
209fn finalization_primary(error: &Error) -> Option<&Error> {
210    if let Some(pair) = error.downcast_ref::<detcore::FinalizationFailure>() {
211        return Some(pair.primary());
212    }
213    error
214        .chain()
215        .find_map(|cause| match cause.downcast_ref::<reverie::Error>() {
216            Some(reverie::Error::Tool(inner)) => finalization_primary(inner),
217            _ => None,
218        })
219}
220
221fn classify_preserving_cleanup(error: &Error) -> (FailureKind, Option<crate::HermitCleanupStage>) {
222    if let Some(primary) = finalization_primary(error) {
223        classify_preserving_cleanup(primary)
224    } else if let Some(error) = error.downcast_ref::<crate::HermitCleanupUnconfirmed>() {
225        (error.primary_kind(), Some(error.stage()))
226    } else if let Some(error) = error.downcast_ref::<crate::HermitPtraceFailure>() {
227        (error.kind(), None)
228    } else {
229        (classify_unwrapped(error), None)
230    }
231}
232
233fn classify_unwrapped(error: &Error) -> FailureKind {
234    if let Some(primary) = finalization_primary(error) {
235        return classify_preserving_cleanup(primary).0;
236    }
237    if let Some(error) = error.downcast_ref::<crate::SkidOvershootError>() {
238        FailureKind::SkidOvershoot {
239            count: error.count(),
240        }
241    } else if error
242        .chain()
243        .any(|cause| cause.is::<detcore_model::virtual_timeout::VirtualTimeoutWithDebugger>())
244        || is_policy_refusal(error)
245        || error
246            .downcast_ref::<crate::LiteinstInGuestRefusal>()
247            .is_some()
248        || error
249            .chain()
250            .any(|cause| cause.is::<crate::cpu_profile_admission::CpuProfileRefusal>())
251        || error
252            .chain()
253            .any(|cause| cause.is::<crate::preemption_admission::UnsupportedPreemption>())
254        || error
255            .chain()
256            .any(|cause| cause.is::<crate::preemption_admission::UnqualifiedAnnotatedReplay>())
257    {
258        FailureKind::PolicyRefusal
259    } else if error.downcast_ref::<crate::GuestTimedOut>().is_some() {
260        FailureKind::RunTimeout
261    } else if let Some(ended) = error.downcast_ref::<crate::SchedulerEndedRun>() {
262        FailureKind::TerminalOutcome(ended.outcome)
263    } else if error.downcast_ref::<crate::BackendRunFailure>().is_some() {
264        FailureKind::BackendRunFailure
265    } else {
266        FailureKind::Error
267    }
268}

270impl From<Error> for SerializableError {
271    fn from(err: Error) -> Self {
272        let error = err.to_string();
273        let context = err.chain().skip(1).map(ToString::to_string).collect();
274        // ⚠️ CLASSIFIED HERE BECAUSE THIS IS THE LAST PLACE THE TYPE EXISTS.
275        // Everything below this line is strings: `From<SerializableError> for
276        // Error` rebuilds the chain with `Error::msg`, so a downcast on the far
277        // side can only ever fail. Detecting the refusal after the boundary
278        // would mean matching on English, which is the thing `kind` exists to
279        // avoid.
280        let (kind, cleanup_stage) = classify_preserving_cleanup(&err);
281        Self {
282            error,
283            context,
284            kind,
285            cleanup_stage,
286        }
287    }
288}
289
290impl From<SerializableError> for Error {
291    fn from(mut err: SerializableError) -> Self {
292        if let Some(root_cause) = err.context.pop() {
293            let mut error = Self::msg(root_cause);
294
295            while let Some(context) = err.context.pop() {
296                error = error.context(context);
297            }
298
299            error.context(err.error)
300        } else {
301            Self::msg(err.error)
302        }
303    }
304}
