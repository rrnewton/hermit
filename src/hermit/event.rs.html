23/// Returns the stable error used for legacy ioctls whose nested output cannot
24/// be represented by the currently pinned Reverie decoder.
25pub(crate) fn deterministic_ioctl_error(request: &ioctl::Request<'_>) -> Option<Errno> {
26    match request {
27        // SIOCETHTOOL stores its output behind the data pointer nested in an
28        // ifreq. Treating it as an opaque request would lose guest-visible
29        // memory updates, so reject it identically during record and replay.
30        ioctl::Request::SIOCETHTOOL(_) | ioctl::Request::Other(SIOCETHTOOL, _) => {
31            Some(Errno::ENODEV)
32        }
33        _ => None,
34    }
35}

37/// An event. This contains everything needed to verify and reproduce the
38/// execution of a syscall.
39#[derive(Debug, Serialize, Deserialize)]
40pub struct Event {
41    /// The event that we use to reconstruct the outputs of the original syscall.
42    /// This is `Some` if need to record this syscall. If the syscall is already
43    /// deterministic, then this is `None`.
44    ///
45    /// If a recorded syscall failed, then this is `Some(Err(Errno))`. That is,
46    /// the failure should be reproduced during replay.
47    pub event: Result<SyscallEvent, Errno>,
48}
49
50/// A `SyscallEvent` contains all the necessary information to replay a system
51/// call.
52///
53/// Note that we only need a small amount of information to replay a syscall. The
54/// only side effects observable by the user are:
55///  1. Mutable pointers
56///  2. Return values.
57///
58/// No registers are modified by the kernel except for `rax` (the return value).
59/// Therefore, registers themselves do not need to be recorded since they are
60/// strictly inputs. However, any arguments that are pointers that point to
61/// mutable data expected to be modified by the kernel need to be recorded. If
62/// this rule is applied uniformly for all syscalls, then we should be able to
63/// implement full record and replay.
64#[derive(Debug, Serialize, Deserialize)]
65pub enum SyscallEvent {
66    Bytes(Vec<u8>),
67    /// The flattened output bytes of a vectored read (`readv`/`preadv`/
68    /// `preadv2`). The bytes are stored contiguously in read order; on replay
69    /// they are scattered back across the guest's `iovec` buffers. The length of
70    /// the vector is exactly the return value of the syscall.
71    Readv(Vec<u8>),
72    Write(i64),
73    Mmap(MmapEvent),
74    Recvmsg(RecvmsgEvent),
75    /// A syscall whose only value we care about is the return value. For many
76    /// syscalls, this is often the only output of the syscall and thus it is the
77    /// only piece of information that needs to be recorded.
78    Return(i64),
79    Stat(StatEvent),
80    Statfs(Vec<u8>),
81    Statx(StatxBuf),
82    Rdtsc(RdtscResult),
83    Ioctl(ioctl::Output),
84    Timespec(TimespecEvent),
85    Timeofday((Timeval, Timezone)),
86    Poll(PollEvent),
87    SockOpt(SockOptEvent),
88    EpollWait(EpollWaitEvent),
89    // AUTONOMOUS-BOT-IMPLEMENTED
90    // TODO-HUMAN-REVIEW(#662): Audit the replay filesystem event schema.
91    Open(OpenEvent),
92    /// Result of `mkdir`/`mkdirat` plus the physical side effect replay must
93    /// reconstruct when the recorded call found an existing directory.
94    Mkdir(MkdirEvent),
95    // TODO-HUMAN-REVIEW(#557): Audit the V2 record/replay event schema.
96    WriteV2(WriteEvent),
97    ReadV2(ReadEvent),
98    ReadvV2(ReadEvent),
99    FtruncateV2(FtruncateEvent),
100    /// Destination length after a successful clone ioctl.
101    FileClone(FileCloneEvent),
102    /// A successful exec and the executable images required immediately before
103    /// replaying it. Failed execs use the enclosing [`Event::event`] error.
104    Exec(ExecEvent),
105    /// The result and mutable output fields of a raw `ppoll` call.
106    Ppoll(PpollEvent),
107}

333/// Records every guest-visible output of a raw `poll` call.
334#[derive(Serialize, Deserialize, Debug)]
335pub struct PollEvent {
336    /// The exact return value or errno observed while recording.
337    pub result: Result<i64, Errno>,
338
339    /// Whether the guest supplied a non-null pollfd pointer.
340    pub fds_pointer_present: bool,
341
342    /// The post-kernel pollfd output. Successful calls contain the complete
343    /// array. An EFAULT may contain only the readable prefix because Linux can
344    /// fault after writing earlier `revents` entries.
345    pub fds: Option<Vec<PollFd>>,
346}
347
348/// Records every guest-visible output of a raw `ppoll` call.
349///
350/// Unlike `poll`, Linux treats the timeout as an in-out parameter. The pollfd
351/// and timeout copyouts are independent of the return value. Linux writes
352/// `revents` before attempting the remaining-time copyout and preserves the
353/// syscall result when that later write faults. Keep both output snapshots
354/// alongside the nested result so replay can restore the writes in kernel order
355/// before returning either success or the error.
356#[derive(Serialize, Deserialize, Debug)]
357pub struct PpollEvent {
358    /// The exact return value or errno observed while recording.
359    pub result: Result<i64, Errno>,
360
361    /// Whether the guest supplied a non-null pollfd pointer.
362    pub fds_pointer_present: bool,
363
364    /// Complete output on success or the readable output prefix on EFAULT.
365    pub fds: Option<Vec<PollFd>>,
366
367    /// Whether the guest supplied a non-null raw timeout pointer.
368    pub timeout_pointer_present: bool,
369
370    /// The exact remaining timeout after the kernel call. Raw `ppoll` mutates
371    /// this continuously; replay must restore the captured value without
372    /// rounding, freezing, or synthesizing it.
373    pub timeout: Option<Timespec>,
374}

420    #[test]
421    fn mkdir_event_schema_preserves_directory_classification() {
422        let events = vec![
423            Event {
424                event: Ok(SyscallEvent::Mkdir(MkdirEvent {
425                    result: Err(Errno::EEXIST),
426                    existing_directory: true,
427                })),
428            },
429            Event {
430                event: Ok(SyscallEvent::Mkdir(MkdirEvent {
431                    result: Err(Errno::ENOENT),
432                    existing_directory: false,
433                })),
434            },
435        ];
436        let encoded = bincode::serde::encode_to_vec(&events, bincode::config::legacy()).unwrap();
437        let (decoded, consumed): (Vec<Event>, usize) =
438            bincode::serde::decode_from_slice(&encoded, bincode::config::legacy()).unwrap();
439
440        assert_eq!(consumed, encoded.len());
441        let [first, second] = decoded.as_slice() else {
442            panic!("unexpected mkdir event count")
443        };
444        assert!(matches!(
445            &first.event,
446            Ok(SyscallEvent::Mkdir(MkdirEvent {
447                result: Err(Errno::EEXIST),
448                existing_directory: true,
449            }))
450        ));
451        assert!(matches!(
452            &second.event,
453            Ok(SyscallEvent::Mkdir(MkdirEvent {
454                result: Err(Errno::ENOENT),
455                existing_directory: false,
456            }))
457        ));
458    }

460    #[test]
461    fn exec_event_schema_preserves_non_utf8_paths_and_topology() {
462        let event = Event {
463            event: Ok(SyscallEvent::Exec(ExecEvent {
464                request: ExecRequest {
465                    dirfd: libc::AT_FDCWD,
466                    path: b"relative/\xff-program".to_vec(),
467                    flags: 0,
468                },
469                executable: ExecImage {
470                    digest: detcore::Digest::new(b"program"),
471                    mode: 0o751,
472                },
473                target: ExecTarget::Materialize(ExecMaterialization {
474                    base: ExecMaterializationBase::Cwd,
475                    path: b"relative/\xff-program".to_vec(),
476                    symlinks: vec![crate::record_replay_path::ResolvedSymlink {
477                        lookup_index: 1,
478                        target: b"target-\xfe".to_vec(),
479                    }],
480                }),
481                dependencies: Vec::new(),
482            })),
483        };
484        let encoded = bincode::serde::encode_to_vec(&event, bincode::config::legacy()).unwrap();
485        let (decoded, consumed): (Event, usize) =
486            bincode::serde::decode_from_slice(&encoded, bincode::config::legacy()).unwrap();
487
488        assert_eq!(consumed, encoded.len());
489        let Ok(SyscallEvent::Exec(decoded)) = decoded.event else {
490            panic!("unexpected exec event")
491        };
492        assert_eq!(decoded.request.path, b"relative/\xff-program");
493        let ExecTarget::Materialize(materialization) = decoded.target else {
494            panic!("unexpected exec target policy")
495        };
496        assert_eq!(materialization.path, b"relative/\xff-program");
497        assert_eq!(materialization.symlinks[0].target, b"target-\xfe");
498    }
