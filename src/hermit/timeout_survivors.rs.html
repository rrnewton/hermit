9//! Which guest processes were still running when `--timeout` fired, for the
10//! timeout report.
11//!
12//! A run lasts until every guest process has exited, not only the first one.
13//! A guest whose root process crashed or exited while a child it never reaped
14//! sat blocked (in a binder read, a FIFO open, a wait on a peer outside the
15//! container) therefore runs to the bound, and the report used to name only
16//! the bound. This names the processes that were left: their ID, command name,
17//! parent, and the system call each was in.
18//!
19//! The processes are found by tracing relation rather than by ancestry: an
20//! orphan is re-parented (to the host's init or a subreaper, or to Hermit
21//! itself when Hermit is the container's init), but its tracer is still this
22//! process. Host-side and read once, after the bound fired, so nothing here
23//! can reach the guest. A process shown "waiting in ptrace_stop" or "stopped
24//! outside a system call" is usually held by Hermit itself, at a stop Hermit
25//! has not resumed, rather than by the kernel.
26
27use std::collections::BTreeSet;
28use std::fs;
29
30/// The most processes one report names.
31const LISTED: usize = 8;
32
33/// One traced guest process, as `/proc` showed it.
34#[derive(Debug, Clone, PartialEq, Eq)]
35pub(crate) struct Survivor {
36    pub pid: i32,
37    pub comm: String,
38    pub ppid: i32,
39    /// The first field of `/proc/<pid>/syscall`: a system call number,
40    /// `running`, or `-1` (stopped, but not in a system call).
41    pub syscall: String,
42    /// `/proc/<pid>/wchan`, or empty when the kernel shows none.
43    pub wchan: String,
44}
45
46/// The guest processes this process traces now, described in one line for a
47/// timeout report, or `None` when there are none or `/proc` cannot be read.
48/// `root` is the run's root guest process, when known.
49pub(crate) fn traced_processes_now(root: Option<i32>) -> Option<String> {
50    let tracers = task_ids("/proc/self/task");
51    if tracers.is_empty() {
52        return None;
53    }
54    let mut survivors: Vec<Survivor> = task_ids("/proc")
55        .into_iter()
56        .filter_map(|pid| read_survivor(pid, &tracers))
57        .collect();
58    survivors.sort_by_key(|survivor| survivor.pid);
59    describe(&survivors, std::process::id() as i32, root)
60}
61
62fn task_ids(dir: &str) -> BTreeSet<i32> {
63    fs::read_dir(dir)
64        .into_iter()
65        .flatten()
66        .flatten()
67        .filter_map(|entry| entry.file_name().to_str()?.parse::<i32>().ok())
68        .collect()
69}
70
71/// `status` field `key` (`TracerPid`, `PPid`) as a number.
72fn status_field(status: &str, key: &str) -> Option<i32> {
73    status
74        .lines()
75        .find_map(|line| line.strip_prefix(key)?.strip_prefix(':'))
76        .and_then(|value| value.trim().parse().ok())
77}
78
79fn read_survivor(pid: i32, tracers: &BTreeSet<i32>) -> Option<Survivor> {
80    let status = fs::read_to_string(format!("/proc/{pid}/status")).ok()?;
81    let tracer = status_field(&status, "TracerPid")?;
82    if !tracers.contains(&tracer) {
83        return None;
84    }
85    let read = |name: &str| {
86        fs::read_to_string(format!("/proc/{pid}/{name}"))
87            .unwrap_or_default()
88            .trim()
89            .to_owned()
90    };
91    Some(Survivor {
92        pid,
93        comm: read("comm"),
94        ppid: status_field(&status, "PPid").unwrap_or(0),
95        syscall: read("syscall")
96            .split_whitespace()
97            .next()
98            .unwrap_or_default()
99            .to_owned(),
100        wchan: read("wchan"),
101    })
102}
103
104/// What a survivor is doing, from its `syscall` and `wchan`.
105fn activity(survivor: &Survivor) -> String {
106    let call = match survivor.syscall.as_str() {
107        "" => "in an unknown state".to_owned(),
108        "running" => "running".to_owned(),
109        "-1" => "stopped outside a system call".to_owned(),
110        number => match number.parse::<i32>() {
111            Ok(number) => format!("in {}", reverie::syscalls::Sysno::from(number)),
112            Err(_) => format!("in system call {number}"),
113        },
114    };
115    match survivor.wchan.as_str() {
116        "" | "0" => call,
117        wchan => format!("{call} (waiting in {wchan})"),
118    }
119}

121/// The report line for `survivors`. A survivor was orphaned (its parent exited,
122/// which is how a run outlives its root) when its parent is neither another
123/// survivor nor `this_process`, or when its parent is `this_process` but it is
124/// not the run's `root`: Hermit started only the root, so any other child of
125/// Hermit was re-parented to it.
126pub(crate) fn describe(
127    survivors: &[Survivor],
128    this_process: i32,
129    root: Option<i32>,
130) -> Option<String> {
131    if survivors.is_empty() {
132        return None;
133    }
134    let guests: BTreeSet<i32> = survivors.iter().map(|survivor| survivor.pid).collect();
135    let mut parts: Vec<String> = survivors
136        .iter()
137        .take(LISTED)
138        .map(|survivor| {
139            let orphaned = if survivor.ppid == this_process {
140                root.is_some_and(|root| root != survivor.pid)
141            } else {
142                !guests.contains(&survivor.ppid)
143            };
144            format!(
145                "pid {} \"{}\" (parent {}{}) {}",
146                survivor.pid,
147                survivor.comm,
148                survivor.ppid,
149                if orphaned {
150                    ", orphaned: its parent exited"
151                } else {
152                    ""
153                },
154                activity(survivor)
155            )
156        })
157        .collect();
158    if survivors.len() > LISTED {
159        parts.push(format!("and {} more", survivors.len() - LISTED));
160    }
161    Some(format!(
162        "{} guest process{} still running: {}",
163        survivors.len(),
164        if survivors.len() == 1 {
165            " was"
166        } else {
167            "es were"
168        },
169        parts.join("; ")
170    ))
171}

173#[cfg(test)]
174mod tests {
175    use super::*;
176
177    fn survivor(pid: i32, ppid: i32, syscall: &str, wchan: &str) -> Survivor {
178        Survivor {
179            pid,
180            comm: "guest".to_owned(),
181            ppid,
182            syscall: syscall.to_owned(),
183            wchan: wchan.to_owned(),
184        }
185    }
186
187    /// An orphan blocked in a system call is named with the call and where the
188    /// kernel holds it; a child of another survivor, or of this process, is not
189    /// called orphaned.
190    #[test]
191    fn a_timeout_report_names_orphans_and_the_call_each_is_in() {
192        let ioctl = libc::SYS_ioctl.to_string();
193        let line = describe(
194            &[
195                survivor(20485, 1, &ioctl, "binder_thread_read"),
196                survivor(20490, 20485, "running", "0"),
197                survivor(20491, 77, "-1", ""),
198            ],
199            77,
200            Some(20491),
201        )
202        .expect("three survivors");
203        assert_eq!(
204            line,
205            "3 guest processes were still running: \
206             pid 20485 \"guest\" (parent 1, orphaned: its parent exited) in \
207             ioctl (waiting in binder_thread_read); \
208             pid 20490 \"guest\" (parent 20485) running; \
209             pid 20491 \"guest\" (parent 77) stopped outside a system call"
210        );
211        assert_eq!(describe(&[], 77, None), None);
212    }
213
214    /// Where Hermit is the container's init, an orphan is re-parented to Hermit
215    /// itself: a child of Hermit that is not the root is still called orphaned.
216    #[test]
217    fn a_child_of_hermit_that_is_not_the_root_is_orphaned() {
218        let line = describe(
219            &[survivor(3, 1, "-1", ""), survivor(7, 1, "-1", "")],
220            1,
221            Some(3),
222        )
223        .expect("two survivors");
224        assert!(
225            line.contains("pid 3 \"guest\" (parent 1) stopped"),
226            "{line}"
227        );
228        assert!(
229            line.contains("pid 7 \"guest\" (parent 1, orphaned: its parent exited)"),
230            "{line}"
231        );
232    }
233
234    #[test]
235    fn a_timeout_report_names_at_most_eight_processes() {
236        let survivors: Vec<_> = (1..=10).map(|pid| survivor(pid, 0, "", "")).collect();
237        let line = describe(&survivors, 99, None).expect("ten survivors");
238        assert!(
239            line.starts_with("10 guest processes were still running: "),
240            "{line}"
241        );
242        assert!(line.ends_with("; and 2 more"), "{line}");
243        assert_eq!(line.matches("pid ").count(), 8, "{line}");
244    }
245
246    #[test]
247    fn status_fields_parse_as_proc_prints_them() {
248        let status = "Name:\tcat\nPPid:\t1\nTracerPid:\t4242\n";
249        assert_eq!(status_field(status, "TracerPid"), Some(4242));
250        assert_eq!(status_field(status, "PPid"), Some(1));
251        assert_eq!(status_field(status, "Uid"), None);
252    }
