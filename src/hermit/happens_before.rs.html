9//! Debug-info resolution for happens-before anchors.
10//!
11//! The anchor/edge *model* lives in [`detcore_model::happens_before`]; it leaves
12//! `func`/`file:line` code locations unresolved (a [`Position::Rip`] with
13//! `addr: None`). This module turns those human-legible locations into concrete
14//! instruction pointers by reading the target binary's symbol table (for
15//! function entry addresses) and DWARF line program (for source lines), and
16//! provides the reverse mapping (address → function/file/line) used by
17//! introspection / `--list-events`.
18//!
19//! It builds *owned* indices in a single pass and drops the mmap/DWARF context,
20//! so the resolver has no self-referential lifetimes and is cheap to keep around.
21

38/// The virtual-address extent of a function symbol.
39#[derive(Clone, Copy, Debug)]
40struct FuncExtent {
41    /// Entry virtual address.
42    addr: u64,
43    /// Size in bytes (0 when the symbol table does not record one).
44    size: u64,
45}
46
47impl FuncExtent {
48    /// True when `addr` falls within `[self.addr, self.addr + self.size)`. A
49    /// zero-sized symbol only matches its exact entry address.
50    fn contains(&self, addr: u64) -> bool {
51        if self.size == 0 {
52            addr == self.addr
53        } else {
54            addr >= self.addr && addr < self.addr + self.size
55        }
56    }
57}

59/// One resolved source-line row: an address and the file/line it maps to.
60#[derive(Clone, Debug)]
61struct LineRow {
62    addr: u64,
63    file: Option<String>,
64    line: u32,
65}

67/// A resolved source location for introspection output.
68#[derive(Clone, Debug, Default, PartialEq, Eq)]
69pub struct ResolvedLocation {
70    /// Function name containing the address, if known.
71    pub function: Option<String>,
72    /// Source file basename, if known.
73    pub file: Option<String>,
74    /// Source line, if known.
75    pub line: Option<u32>,
76}
77
78/// Owned debug-info indices for one binary.
79pub struct DebugInfoResolver {
80    /// Function name → extent (from the ELF symbol table).
81    functions: BTreeMap<String, FuncExtent>,
82    /// Line rows sorted ascending by address.
83    lines: Vec<LineRow>,
84}
85
86impl DebugInfoResolver {
87    /// Read `path` and build the symbol and line-number indices.
88    pub fn open(path: &Path) -> anyhow::Result<Self> {
89        let data = std::fs::read(path)
90            .with_context(|| format!("reading binary for debug info: {}", path.display()))?;
91        Self::from_bytes(&data)
92            .with_context(|| format!("parsing debug info from {}", path.display()))
93    }
94
95    /// Build the indices from in-memory object bytes.
96    pub fn from_bytes(data: &[u8]) -> anyhow::Result<Self> {
97        let object = object::File::parse(data)?;
98
99        // --- function symbols -------------------------------------------------
100        let mut functions: BTreeMap<String, FuncExtent> = BTreeMap::new();
101        for sym in object.symbols() {
102            if sym.kind() != SymbolKind::Text {
103                continue;
104            }
105            let addr = sym.address();
106            if addr == 0 {
107                continue;
108            }
109            if let Ok(name) = sym.name() {
110                if name.is_empty() {
111                    continue;
112                }
113                // Prefer the first definition (typically the primary symbol) but
114                // upgrade a zero-sized entry if we later learn a real size.
115                functions
116                    .entry(name.to_string())
117                    .and_modify(|e| {
118                        if e.size == 0 && sym.size() > 0 {
119                            *e = FuncExtent {
120                                addr,
121                                size: sym.size(),
122                            };
123                        }
124                    })
125                    .or_insert(FuncExtent {
126                        addr,
127                        size: sym.size(),
128                    });
129            }
130        }
131
132        // --- DWARF line rows --------------------------------------------------
133        let lines = Self::load_line_rows(&object).unwrap_or_default();
134
135        Ok(Self { functions, lines })
136    }
137
138    /// Parse the DWARF line programs into an address-sorted list of rows.
139    fn load_line_rows(object: &object::File<'_>) -> anyhow::Result<Vec<LineRow>> {
140        let endian = if object.is_little_endian() {
141            gimli::RunTimeEndian::Little
142        } else {
143            gimli::RunTimeEndian::Big
144        };
145
146        let load_section = |id: gimli::SectionId| -> Result<Cow<'_, [u8]>, gimli::Error> {
147            match object.section_by_name(id.name()) {
148                Some(section) => Ok(section
149                    .uncompressed_data()
150                    .unwrap_or(Cow::Borrowed(&[][..]))),
151                None => Ok(Cow::Borrowed(&[][..])),
152            }
153        };
154
155        let dwarf_sections = gimli::DwarfSections::load(load_section)?;
156        let dwarf = dwarf_sections.borrow(|section| gimli::EndianSlice::new(section, endian));
157
158        let mut rows: Vec<LineRow> = Vec::new();
159        let mut units = dwarf.units();
160        while let Some(header) = units.next()? {
161            let unit = dwarf.unit(header)?;
162            let Some(program) = unit.line_program.clone() else {
163                continue;
164            };
165            let mut state = program.rows();
166            while let Some((header, row)) = state.next_row()? {
167                if row.end_sequence() {
168                    continue;
169                }
170                let Some(line) = row.line() else {
171                    continue;
172                };
173                let file = row.file(header).and_then(|file_entry| {
174                    dwarf
175                        .attr_string(&unit, file_entry.path_name())
176                        .ok()
177                        .and_then(|r| r.to_string().ok().map(basename))
178                });
179                rows.push(LineRow {
180                    addr: row.address(),
181                    file,
182                    line: line.get() as u32,
183                });
184            }
185        }
186        rows.sort_by_key(|r| r.addr);
187        Ok(rows)
188    }
189
190    /// True when no debug/symbol information was found at all.
191    pub fn is_empty(&self) -> bool {
192        self.functions.is_empty() && self.lines.is_empty()
193    }
194
195    /// The entry address of a named function, if present in the symbol table.
196    pub fn function_addr(&self, name: &str) -> Option<u64> {
197        self.functions.get(name).map(|e| e.addr)
198    }
199
200    /// The lowest address whose line row matches `line` (and `file`, when given),
201    /// optionally constrained to a `(addr, size)` extent (a zero size matches only
202    /// the exact entry address).
203    pub fn line_addr(
204        &self,
205        file: Option<&str>,
206        line: u32,
207        within: Option<(u64, u64)>,
208    ) -> Option<u64> {
209        let file_base = file.map(basename);
210        let extent = within.map(|(addr, size)| FuncExtent { addr, size });
211        self.lines
212            .iter()
213            .filter(|r| r.line == line)
214            .filter(|r| match (&file_base, &r.file) {
215                (Some(want), Some(have)) => want == have,
216                (Some(_), None) => false,
217                (None, _) => true,
218            })
219            .filter(|r| extent.map(|f| f.contains(r.addr)).unwrap_or(true))
220            .map(|r| r.addr)
221            .min()
222    }
223
224    /// Resolve a [`CodeLocation`] to a concrete address, preferring
225    /// function+line (the address of `line` inside `function`), then bare
226    /// function entry, then file:line.
227    pub fn resolve_location(&self, loc: &CodeLocation) -> Option<u64> {
228        match (&loc.function, loc.line) {
229            (Some(func), Some(line)) => {
230                let extent = self.functions.get(func).copied();
231                self.line_addr(loc.file.as_deref(), line, extent.map(|e| (e.addr, e.size)))
232                    // Fall back to the function entry if the exact line row is
233                    // absent (e.g. no DWARF, or inlined line).
234                    .or(extent.map(|e| e.addr))
235            }
236            (Some(func), None) => self.function_addr(func),
237            (None, Some(line)) => self.line_addr(loc.file.as_deref(), line, None),
238            (None, None) => None,
239        }
240    }
241
242    /// Describe an address in source terms (nearest preceding line row and the
243    /// enclosing function), for introspection output.
244    pub fn describe(&self, addr: u64) -> ResolvedLocation {
245        // Enclosing function: the entry whose extent contains `addr`, else the
246        // closest preceding entry.
247        let function = self
248            .functions
249            .iter()
250            .filter(|(_, e)| e.contains(addr) || e.addr <= addr)
251            .max_by_key(|(_, e)| e.addr)
252            .map(|(name, _)| name.clone());
253
254        // Nearest preceding line row.
255        let idx = self.lines.partition_point(|r| r.addr <= addr);
256        let line_row = if idx > 0 {
257            self.lines.get(idx - 1)
258        } else {
259            None
260        };
261
262        ResolvedLocation {
263            function,
264            file: line_row.and_then(|r| r.file.clone()),
265            line: line_row.map(|r| r.line),
266        }
267    }
268}

270/// Resolve every anchor in `program` whose position is an unresolved RIP carrying
271/// a code location. Returns the names of anchors that could not be resolved.
272pub fn resolve_program(
273    program: &mut HappensBeforeProgram,
274    resolver: &DebugInfoResolver,
275) -> Vec<String> {
276    let mut unresolved = Vec::new();
277    for (name, anchor) in program.anchors.iter_mut() {
278        if let Position::Rip { addr: None, nth } = &anchor.position {
279            let nth = *nth;
280            if let Some(resolved) = resolver.resolve_location(&anchor.location) {
281                anchor.position = Position::Rip {
282                    addr: Some(resolved),
283                    nth,
284                };
285            } else {
286                unresolved.push(name.clone());
287            }
288        }
289    }
290    unresolved
291}

293/// Load a happens-before specification from a JSON file and normalize it into a
294/// [`HappensBeforeProgram`]. Code locations remain unresolved until
295/// [`resolve_program`] runs against a [`DebugInfoResolver`].
296pub fn load_program(path: &Path) -> anyhow::Result<HappensBeforeProgram> {
297    let json = std::fs::read_to_string(path)
298        .with_context(|| format!("reading happens-before spec: {}", path.display()))?;
299    let spec = HappensBeforeSpec::from_json(&json)
300        .with_context(|| format!("parsing happens-before spec: {}", path.display()))?;
301    let program = spec
302        .normalize()
303        .with_context(|| format!("normalizing happens-before spec: {}", path.display()))?;
304    Ok(program)
305}

307/// Render one anchor for `--list-events`-style introspection, showing the
308/// resolved address and source description when available.
309pub fn describe_anchor(anchor: &Anchor, resolver: Option<&DebugInfoResolver>) -> String {
310    let mut out = format!("{}", anchor);
311    if let (Position::Rip { addr: Some(a), .. }, Some(r)) = (&anchor.position, resolver) {
312        let d = r.describe(*a);
313        let where_ = match (&d.function, &d.file, d.line) {
314            (Some(f), _, Some(l)) => format!(" -> {:#x} ({}:{})", a, f, l),
315            (Some(f), _, None) => format!(" -> {:#x} ({})", a, f),
316            _ => format!(" -> {:#x}", a),
317        };
318        out.push_str(&where_);
319    }
320    out
321}

323/// The final path component of a possibly-slashed path string.
324fn basename(s: &str) -> String {
325    match s.rsplit_once('/') {
326        Some((_, base)) => base.to_string(),
327        None => s.to_string(),
328    }

339    /// Compile a tiny -g fixture to a temp path; returns None if no C compiler.
340    /// Each call gets a unique directory so parallel tests never collide.
341    fn compile_fixture() -> Option<std::path::PathBuf> {
342        use std::sync::atomic::AtomicUsize;
343        use std::sync::atomic::Ordering;
344        static COUNTER: AtomicUsize = AtomicUsize::new(0);
345
346        let cc = ["cc", "gcc", "clang"]
347            .into_iter()
348            .find(|c| Command::new(c).arg("--version").output().is_ok())?;
349
350        let uniq = COUNTER.fetch_add(1, Ordering::Relaxed);
351        let dir = std::env::temp_dir().join(format!("hb_fixture_{}_{}", std::process::id(), uniq));
352        std::fs::create_dir_all(&dir).ok()?;
353        let src = dir.join("fixture.c");
354        let bin = dir.join("fixture");
355        std::fs::write(
356            &src,
357            "int free_buffer(int x) { return x + 1; }\n\
358             int read_buffer(int x) { return x * 2; }\n\
359             int main(void) { return free_buffer(read_buffer(3)); }\n",
360        )
361        .ok()?;
362        let status = Command::new(cc)
363            .args(["-g", "-O0", "-o"])
364            .arg(&bin)
365            .arg(&src)
366            .status()
367            .ok()?;
368        if status.success() { Some(bin) } else { None }
369    }

371    #[test]
372    fn resolves_function_and_line_from_dwarf() {
373        let Some(bin) = compile_fixture() else {
374            eprintln!("skipping: no C compiler available");
375            return;
376        };
377        let resolver = DebugInfoResolver::open(&bin).unwrap();
378        assert!(!resolver.is_empty(), "expected symbols/lines in -g binary");
379
380        // Function entry resolves to a nonzero address.
381        let free_addr = resolver.function_addr("free_buffer");
382        assert!(free_addr.is_some(), "free_buffer should resolve");
383        assert_ne!(free_addr.unwrap(), 0);
384
385        // free_buffer is defined on line 1; its line row should fall within the
386        // function extent.
387        let loc = CodeLocation {
388            function: Some("free_buffer".to_string()),
389            file: None,
390            line: Some(1),
391        };
392        let addr = resolver.resolve_location(&loc);
393        assert!(addr.is_some(), "func+line should resolve");
394
395        // Reverse description of the entry address names the function.
396        let d = resolver.describe(free_addr.unwrap());
397        assert_eq!(d.function.as_deref(), Some("free_buffer"));
398
399        let _ = std::fs::remove_dir_all(bin.parent().unwrap());
400    }

406            return;

417          "edges": [ {"before": "A", "after": "B"} ]

436                other => panic!("expected resolved RIP, got {:?}", other),
