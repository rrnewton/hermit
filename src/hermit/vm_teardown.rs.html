9//! Handing KVM VM destruction to a longer-lived helper process.
10//!
11//! Closing a VM fd waits out a kernel RCU grace period in
12//! `kvm_destroy_vm` (facebookexperimental/reverie#132), on whoever
13//! closes last. A short-lived CLI gains nothing by moving that close
14//! to another of its own threads: process exit closes every descriptor
15//! the threads have not reached, in the exiting process, and inside
16//! one CPU quota the deferred waits stop hiding under guest execution
17//! (measured; see the commit that added this module). So the binary
18//! instead spawns a small helper process (its own image re-executed in
19//! a private role, `__hermit-vm-teardown-reap` in the bin's
20//! vm_teardown_helper module) before the container clone, and this
21//! sink passes each dropped backend's duplicated vCPU and VM
22//! descriptors to it over a Unix socket with SCM_RIGHTS. The kernel
23//! takes the file references at `sendmsg`, so the backend's own closes
24//! right afterwards only drop references and the helper's closes are
25//! the destroys.
26//!
27//! Ownership and lifetime, exactly:
28//!
29//! - The bin installs one sink per process, before the first container
30//!   clone, through [`install_vm_teardown_sink`]. The container child
31//!   inherits the installed sink and its socket at the clone; only the
32//!   child adopts into it. The sink starts no threads in either
33//!   process — a container clone copies none, and its callback return
34//!   exits only the calling thread — so adopters count the helper's
35//!   acknowledgements themselves, as `Inner::receive` describes.
36//! - The helper serves until the socket reaches EOF — every sender
37//!   copy closed, which happens when both the parent process and its
38//!   container children have exited or been killed — then drains its
39//!   queue and exits. Its bound past the run's own exit is the drain:
40//!   one bundle's destroy per worker thread, every prior bundle having
41//!   been destroyed or concurrently in flight since its adoption. If
42//!   the helper dies at any point, the kernel closes the descriptors
43//!   it holds in its own exit path, so every VM is still destroyed;
44//!   later adoptions see the dead socket and fall back to the
45//!   backend's synchronous destroy (the sink's duplicates close
46//!   locally, then the backend's own field drops are the last ones).
47//! - Adoption is bounded per adopting process: at most
48//!   `MAX_IN_FLIGHT_BUNDLES` unacknowledged bundles are outstanding
49//!   from it, and adoption blocks past that. A bundle keeps one guest
50//!   memory's pinned pages alive, so an unbounded backlog would be
51//!   unbounded host memory; blocking degenerates to synchronous
52//!   teardown under overload instead. Sequential container children
53//!   sharing one helper can transiently sum to a few such bounds while
54//!   the helper drains the previous child's tail.
55//!
56//! Nothing here is guest-visible: adoption happens from a backend's
57//! drop, after every guest-visible effect of that exit was published
58//! on the dropping thread (see reverie-kvm's `teardown` module). With
59//! no sink installed, `run_kvm` leaves every backend to destroy its
60//! VM synchronously, exactly as it did before this module.

75/// The data byte accompanying one forwarded descriptor pair.
76pub const BUNDLE_BYTE: u8 = b'V';
77/// The helper's per-bundle acknowledgement, sent after both closes.
78pub const ACK_BYTE: u8 = b'D';
79/// Sent once by the helper when its receive loop is running. Senders do
80/// not wait for it (the socket buffers bundles sent before the helper
81/// starts); it exists so tests and diagnostics can tell a live helper
82/// from a dead socket.
83pub const READY_BYTE: u8 = b'R';
84
85/// Bundles adopted but not yet acknowledged before adoption blocks.
86/// The helper's worker pool destroys several concurrently on top of
87/// these, so the outstanding pinned guest memory is this plus the
88/// pool size, in guest memories. Sized with the pool: a bound the
89/// pool cannot drain at fork rate serializes destroys back onto
90/// adopting threads (measured fork-slope regression at 16 with a
91/// 4-worker pool), while an unbounded one would pin unbounded guest
92/// memory. Denominated in VMs, the byte cost is the touched pages of
93/// at most this many guest windows (plus the helper's 16 workers and
94/// its channel): the nominal ceiling is ~80 windows' worth of touched
95/// pages — real guests touch tens of MiB of their 3 GiB window, so
96/// expect hundreds of MiB, not the 240 GiB a fully touched set of
97/// windows would pin. Saturation needs fork/exec churn far faster
98/// than the destroy drain.
99const MAX_IN_FLIGHT_BUNDLES: usize = 64;
100
101/// Sends one descriptor pair (`vcpu` first, mirroring the close order)
102/// with a [`BUNDLE_BYTE`] over `stream`. The kernel references both
103/// files for the receiver before this returns, so the caller may close
104/// its own copies immediately afterwards. `MSG_NOSIGNAL`: a dead
105/// helper must report `EPIPE` to the sender (which falls back to a
106/// synchronous destroy), never kill a running guest's supervisor with
107/// SIGPIPE.
108pub fn send_vm_bundle(stream: &UnixStream, vcpu: RawFd, vm: RawFd) -> io::Result<()> {
109    let mut byte = BUNDLE_BYTE;
110    let mut iov = libc::iovec {
111        iov_base: (&mut byte as *mut u8).cast(),
112        iov_len: 1,
113    };
114    let fds = [vcpu, vm];
115    let mut control = [0usize; 8];
116    let mut msg: libc::msghdr = unsafe { std::mem::zeroed() };
117    msg.msg_iov = &mut iov;
118    msg.msg_iovlen = 1;
119    msg.msg_control = control.as_mut_ptr().cast();
120    msg.msg_controllen = unsafe { libc::CMSG_SPACE(2 * size_of::<RawFd>() as u32) } as usize;
121    loop {
122        // SAFETY: `msg` names `iov` (one live byte) and `control`
123        // (sized for two descriptors); both outlive the call. The
124        // CMSG macros only offset inside `control`.
125        let sent = unsafe {
126            let cmsg = libc::CMSG_FIRSTHDR(&msg);
127            (*cmsg).cmsg_level = libc::SOL_SOCKET;
128            (*cmsg).cmsg_type = libc::SCM_RIGHTS;
129            (*cmsg).cmsg_len = libc::CMSG_LEN(2 * size_of::<RawFd>() as u32) as usize;
130            std::ptr::copy_nonoverlapping(fds.as_ptr(), libc::CMSG_DATA(cmsg).cast::<RawFd>(), 2);
131            libc::sendmsg(stream.as_raw_fd(), &msg, libc::MSG_NOSIGNAL)
132        };
133        if sent == 1 {
134            return Ok(());
135        }
136        let error = io::Error::last_os_error();
137        if error.kind() != io::ErrorKind::Interrupted {
138            return Err(error);
139        }
140    }
141}

143/// Receives one bundle sent by [`send_vm_bundle`] from `fd`.
144/// `Ok(None)` is a clean EOF before a bundle byte. A malformed message
145/// (wrong byte, truncated control data, a descriptor count other than
146/// two) is an error, and any descriptors it carried are closed before
147/// returning it: the helper treats that as fatal and exits, after
148/// which senders fall back to synchronous destroys.
149pub fn recv_vm_bundle(fd: RawFd) -> io::Result<Option<(OwnedFd, OwnedFd)>> {
150    let mut byte = 0u8;
151    let mut iov = libc::iovec {
152        iov_base: (&mut byte as *mut u8).cast(),
153        iov_len: 1,
154    };
155    let mut control = [0usize; 8];
156    let mut msg: libc::msghdr = unsafe { std::mem::zeroed() };
157    msg.msg_iov = &mut iov;
158    msg.msg_iovlen = 1;
159    msg.msg_control = control.as_mut_ptr().cast();
160    msg.msg_controllen = unsafe { libc::CMSG_SPACE(2 * size_of::<RawFd>() as u32) } as usize;
161    let received = loop {
162        // SAFETY: as in send; `msg`, `iov` and `control` outlive the call.
163        let received = unsafe { libc::recvmsg(fd, &mut msg, libc::MSG_CMSG_CLOEXEC) };
164        if received >= 0 {
165            break received;
166        }
167        let error = io::Error::last_os_error();
168        if error.kind() != io::ErrorKind::Interrupted {
169            return Err(error);
170        }
171        // A retry must not inherit a half-described control buffer.
172        msg.msg_controllen = unsafe { libc::CMSG_SPACE(2 * size_of::<RawFd>() as u32) } as usize;
173        msg.msg_flags = 0;
174    };
175    if received == 0 {
176        return Ok(None);
177    }
178    // Take ownership of whatever descriptors arrived before judging the
179    // message, so every error path below closes them.
180    let mut taken: Vec<OwnedFd> = Vec::new();
181    // SAFETY: CMSG_FIRSTHDR offsets inside the received control buffer;
182    // a null header (no ancillary data) is the malformed case.
183    let cmsg = unsafe { libc::CMSG_FIRSTHDR(&msg) };
184    if !cmsg.is_null() {
185        // SAFETY: `cmsg` describes the len-validated control buffer.
186        let (level, ty, len) = unsafe { ((*cmsg).cmsg_level, (*cmsg).cmsg_type, (*cmsg).cmsg_len) };
187        if level == libc::SOL_SOCKET && ty == libc::SCM_RIGHTS {
188            let header = unsafe { libc::CMSG_LEN(0) } as usize;
189            if len >= header && (len - header).is_multiple_of(size_of::<RawFd>()) {
190                let count = (len - header) / size_of::<RawFd>();
191                // The buffer holds at most two; a sender offering more
192                // sets MSG_CTRUNC and the kernel closes the excess.
193                for i in 0..count {
194                    // SAFETY: entry i is inside the validated data area.
195                    let raw =
196                        unsafe { std::ptr::read(libc::CMSG_DATA(cmsg).cast::<RawFd>().add(i)) };
197                    // SAFETY: the kernel installed this descriptor for
198                    // this receive; nothing else owns it.
199                    taken.push(unsafe { OwnedFd::from_raw_fd(raw) });
200                }
201            }
202        }
203    }
204    let truncated = msg.msg_flags & libc::MSG_CTRUNC != 0;
205    if received != 1 || byte != BUNDLE_BYTE || truncated || taken.len() != 2 {
206        return Err(io::Error::new(
207            io::ErrorKind::InvalidData,
208            "malformed VM teardown bundle",
209        ));
210    }
211    let mut taken = taken.into_iter();
212    Ok(Some((
213        taken.next().expect("two descriptors checked"),
214        taken.next().expect("two descriptors checked"),
215    )))
216}

218#[derive(Default)]
219struct State {
220    in_flight: usize,
221    adopted: u64,
222    destroyed: u64,
223    dead: bool,
224    /// Adoptions that fell back to closing locally because the helper
225    /// was already dead or the send failed. The silent sync fallback
226    /// this sink exists around is otherwise unobservable to an audit.
227    fallbacks: u64,
228    /// Holds of in-flight bundles, queued in send order as an
229    /// ordering approximation: each acknowledgement drops the oldest.
230    /// Pairing is NOT exact — the helper's workers close concurrently
231    /// and their acks are anonymous count bytes, and sequential
232    /// container children share this one socket, so an ack counted
233    /// here can complete a predecessor's destroy and release this
234    /// queue's head while its own VM is still being destroyed. That
235    /// early release only lets the unmap race a live VM and pay the
236    /// synchronous zap on the dropping thread (the pre-hold
237    /// behavior); it can never make the unmap incorrect, because the
238    /// descriptors the sink holds are what refcount the VM's
239    /// lifetime, not the guest-memory clone. Exact pairing would
240    /// need acknowledgement sequence numbers on the wire.
241    holds: std::collections::VecDeque<Box<dyn std::any::Any + Send>>,
242}
243
244struct Inner {
245    sender: Mutex<UnixStream>,
246    /// Serializes acknowledgement receives: there is no reader thread
247    /// (a container clone copies no threads, and a normal raw-clone
248    /// callback return exits only its calling thread, so any thread
249    /// started in the container child would keep it alive past its
250    /// result). Adopters drain pending acknowledgements themselves,
251    /// nonblocking, under this lock; an adopter that must wait for a
252    /// free in-flight slot polls the socket, then drains again.
253    receive: Mutex<()>,
254    state: Mutex<State>,
255}
256
257impl Inner {
258    fn lock(&self) -> MutexGuard<'_, State> {
259        self.state
260            .lock()
261            .unwrap_or_else(|poisoned| poisoned.into_inner())
262    }
263
264    fn fd(&self) -> RawFd {
265        self.sender
266            .lock()
267            .unwrap_or_else(|poisoned| poisoned.into_inner())
268            .as_raw_fd()
269    }
270
271    /// Counts every acknowledgement already waiting on the socket.
272    /// Marks the helper dead at EOF or on a protocol error: any byte
273    /// other than an ack or the one ready byte is taken as fatal,
274    /// like a malformed bundle on the receiving side.
275    fn drain_acks(&self) {
276        let _receive = self
277            .receive
278            .lock()
279            .unwrap_or_else(|poisoned| poisoned.into_inner());
280        let fd = self.fd();
281        let mut buf = [0u8; 64];
282        loop {
283            // SAFETY: `buf` is a live 64-byte array for the call; the
284            // descriptor is the sink's socket. MSG_DONTWAIT keeps the
285            // socket itself blocking for sends.
286            let received = unsafe {
287                libc::recv(
288                    fd,
289                    buf.as_mut_ptr().cast(),
290                    buf.len(),
291                    libc::MSG_DONTWAIT | libc::MSG_NOSIGNAL,
292                )
293            };
294            if received > 0 {
295                let mut drop_holds = Vec::new();
296                let dead = {
297                    let mut state = self.lock();
298                    for &byte in &buf[..received as usize] {
299                        match byte {
300                            ACK_BYTE => {
301                                state.destroyed = state.destroyed.saturating_add(1);
302                                state.in_flight = state.in_flight.saturating_sub(1);
303                                if let Some(hold) = state.holds.pop_front() {
304                                    drop_holds.push(hold);
305                                }
306                            }
307                            // The helper announces readiness once; no
308                            // state depends on it (the socket buffers).
309                            READY_BYTE => {}
310                            _ => state.dead = true,
311                        }
312                    }
313                    if state.dead {
314                        drop_holds.extend(state.holds.drain(..));
315                    }
316                    state.dead
317                };
318                // Holds drop outside the state lock (an unmap can be
319                // slow), still under `receive`, so their order against
320                // later acknowledgements is preserved.
321                drop(drop_holds);
322                if dead {
323                    return;
324                }
325            } else if received == 0 {
326                self.drop_all_holds();
327                return;
328            } else {
329                let error = io::Error::last_os_error();
330                match error.kind() {
331                    io::ErrorKind::WouldBlock => return,
332                    io::ErrorKind::Interrupted => {}
333                    _ => {
334                        self.drop_all_holds();
335                        return;
336                    }
337                }
338            }
339        }
340    }
341
342    /// Marks the helper dead and drops every queued hold: their VMs
343    /// are being destroyed by the kernel as the helper's descriptors
344    /// close (or were never adopted), so nothing needs them held past
345    /// the destroy any longer. Called with `receive` held, from
346    /// `drain_acks`'s terminal paths.
347    fn drop_all_holds(&self) {
348        let holds = {
349            let mut state = self.lock();
350            state.dead = true;
351            std::mem::take(&mut state.holds)
352        };
353        drop(holds);
354    }
355
356    /// Polls until the socket is readable (an acknowledgement or EOF)
357    /// or `timeout` passes (unbounded if `None`).
358    fn wait_readable(&self, timeout: Option<Duration>) {
359        let mut poll = libc::pollfd {
360            fd: self.fd(),
361            events: libc::POLLIN,
362            revents: 0,
363        };
364        let millis = timeout.map_or(-1, |t| {
365            i32::try_from(t.as_millis().max(1)).unwrap_or(i32::MAX)
366        });
367        // SAFETY: `poll` is one valid pollfd for the call. Any result
368        // (timeout, error, NVAL) just returns to the caller's recheck
369        // of state, which decides what happens next.
370        let _ = unsafe { libc::poll(&mut poll, 1, millis) };
371    }
372}

374/// The process's [`reverie_kvm::VmTeardownSink`], forwarding dropped
375/// backends' VM descriptors to the teardown helper process.
376pub struct HelperTeardownSink {
377    inner: Arc<Inner>,
378}
379
380impl HelperTeardownSink {
381    pub fn new(stream: UnixStream) -> Self {
382        Self {
383            inner: Arc::new(Inner {
384                sender: Mutex::new(stream),
385                receive: Mutex::new(()),
386                state: Mutex::new(State::default()),
387            }),
388        }
389    }
390
391    /// Bundles handed to the helper and acknowledged destroyed.
392    pub fn counts(&self) -> (u64, u64) {
393        let state = self.inner.lock();
394        (state.adopted, state.destroyed)
395    }
396
397    /// Whether the helper is known gone; adoptions then close their
398    /// duplicates locally and the backend's own drop destroys the VM.
399    pub fn is_dead(&self) -> bool {
400        self.inner.lock().dead
401    }
402
403    /// Adoptions that fell back to closing locally — the helper was
404    /// already dead, or the send failed — instead of reaching it.
405    pub fn fallbacks(&self) -> u64 {
406        self.inner.lock().fallbacks
407    }
408
409    /// Waits until every adopted bundle is acknowledged destroyed, the
410    /// helper dies, or `timeout` passes; reports whether it drained.
411    /// Nothing on the run path waits — the helper draining after the
412    /// run is the point — but tests and audits use this to prove no VM
413    /// outlives its bound.
414    pub fn wait_drained(&self, timeout: Duration) -> bool {
415        let deadline = Instant::now() + timeout;
416        loop {
417            self.inner.drain_acks();
418            let remaining = {
419                let state = self.inner.lock();
420                if state.in_flight == 0 {
421                    return true;
422                }
423                if state.dead {
424                    return false;
425                }
426                match deadline.checked_duration_since(Instant::now()) {
427                    Some(remaining) => remaining,
428                    None => return false,
429                }
430            };
431            self.inner.wait_readable(Some(remaining));
432        }
433    }
434}

436impl HelperTeardownSink {
437    fn adopt(&self, vcpu: OwnedFd, vm: OwnedFd, hold: Option<Box<dyn std::any::Any + Send>>) {
438        // Claim an in-flight slot, draining acknowledgements (and
439        // polling for the next one, if the bound is reached) until a
440        // slot frees or the helper turns out dead. No lock is held
441        // across the poll, and no thread is started: see `receive`.
442        loop {
443            self.inner.drain_acks();
444            {
445                let mut state = self.inner.lock();
446                if state.dead {
447                    state.fallbacks = state.fallbacks.saturating_add(1);
448                    return;
449                }
450                if state.in_flight < MAX_IN_FLIGHT_BUNDLES {
451                    state.in_flight += 1;
452                    state.adopted = state.adopted.saturating_add(1);
453                    break;
454                }
455            }
456            self.inner.wait_readable(None);
457        }
458        // The hold is queued under the sender lock, right after its
459        // send succeeds, so the hold queue stays in the helper's
460        // receive order even with concurrent adopters. Matching acks
461        // to holds from there is approximate, not exact (see
462        // `State::holds`): the queue order is the best ordering the
463        // count-only ack protocol offers without sequence numbers.
464        let mut hold = hold;
465        let sent = {
466            let sender = self
467                .inner
468                .sender
469                .lock()
470                .unwrap_or_else(|poisoned| poisoned.into_inner());
471            let sent = send_vm_bundle(&sender, vcpu.as_raw_fd(), vm.as_raw_fd());
472            if sent.is_ok()
473                && let Some(hold) = hold.take()
474            {
475                self.inner.lock().holds.push_back(hold);
476            }
477            sent
478        };
479        // On success the helper holds its own references and these
480        // close the sink's duplicates; on failure they were never
481        // transferred, and closing them here leaves the backend's own
482        // field drops (which run right after adoption) as the destroy.
483        drop(vcpu);
484        drop(vm);
485        if sent.is_err() {
486            let mut state = self.inner.lock();
487            state.dead = true;
488            state.adopted = state.adopted.saturating_sub(1);
489            state.in_flight = state.in_flight.saturating_sub(1);
490            state.fallbacks = state.fallbacks.saturating_add(1);
491        }
492    }
493}

505static SINK: OnceLock<Arc<HelperTeardownSink>> = OnceLock::new();

507/// Installs the process's teardown sink on `stream`, the sender end of
508/// the socket to the helper process. The binary calls this once,
509/// before its first container clone, so the container child inherits
510/// the installed sink and its socket. Returns whether this call
511/// installed it; a later call drops its stream and changes nothing.
512pub fn install_vm_teardown_sink(stream: UnixStream) -> bool {
513    SINK.set(Arc::new(HelperTeardownSink::new(stream))).is_ok()
514}
515
516/// Whether a teardown sink is installed in this process.
517pub fn vm_teardown_sink_installed() -> bool {
518    SINK.get().is_some()
519}
520
521/// The installed teardown sink, for `run_kvm` to hand to each backend.
522pub fn vm_teardown_sink() -> Option<Arc<dyn reverie_kvm::VmTeardownSink>> {
523    match SINK.get() {
524        Some(sink) => Some(sink.clone()),
525        None => None,
526    }
527}

538    fn pipe() -> (OwnedFd, OwnedFd) {
539        let mut fds = [0; 2];
540        // SAFETY: `fds` is a valid two-element array for pipe2 to fill.
541        assert_eq!(unsafe { libc::pipe2(fds.as_mut_ptr(), libc::O_CLOEXEC) }, 0);
542        // SAFETY: pipe2 just created both descriptors for this call.
543        unsafe { (OwnedFd::from_raw_fd(fds[0]), OwnedFd::from_raw_fd(fds[1])) }
544    }

546    fn is_eof(fd: &OwnedFd) -> bool {
547        // Deadline poll, not an instant check: a parallel test between
548        // fork and exec transiently holds a duplicate of every pipe
549        // write end in this process, so EOF can lag the closes this
550        // test owns by that exec window. A genuinely leaked duplicate
551        // still fails, at the deadline.
552        let deadline = Instant::now() + Duration::from_secs(5);
553        loop {
554            let mut poll = libc::pollfd {
555                fd: fd.as_raw_fd(),
556                events: libc::POLLIN,
557                revents: 0,
558            };
559            let timeout = deadline
560                .saturating_duration_since(Instant::now())
561                .as_millis();
562            // SAFETY: `poll` is one valid pollfd for the duration of the call.
563            let ready =
564                unsafe { libc::poll(&mut poll, 1, i32::try_from(timeout).unwrap_or(i32::MAX)) };
565            if ready == 1 && poll.revents & libc::POLLHUP != 0 {
566                return true;
567            }
568            if Instant::now() >= deadline {
569                return false;
570            }
571        }
572    }

608    #[test]
609    fn adoption_blocks_at_the_in_flight_bound_until_an_ack() {
610        let (sender, helper) = UnixStream::pair().unwrap();
611        let (release_tx, release_rx) = mpsc::channel::<()>();
612        let thread = std::thread::spawn(move || fake_helper(helper, Some(release_rx)));
613        let sink = Arc::new(HelperTeardownSink::new(sender));
614        let mut readers = Vec::new();
615        for _ in 0..MAX_IN_FLIGHT_BUNDLES {
616            let (read_a, write_a) = pipe();
617            let (read_b, write_b) = pipe();
618            readers.push((read_a, read_b));
619            sink.adopt_vm(write_a, write_b);
620        }
621        // The next adoption waits for an acknowledgement that the fake
622        // helper withholds until released.
623        let (done_tx, done_rx) = mpsc::channel::<()>();
624        let blocked = {
625            let sink = Arc::clone(&sink);
626            std::thread::spawn(move || {
627                let (read_a, write_a) = pipe();
628                let (read_b, write_b) = pipe();
629                sink.adopt_vm(write_a, write_b);
630                let _ = done_tx.send(());
631                (read_a, read_b)
632            })
633        };
634        assert!(
635            done_rx.recv_timeout(Duration::from_millis(200)).is_err(),
636            "adoption past the in-flight bound did not block"
637        );
638        release_tx.send(()).unwrap();
639        let (last_a, last_b) = blocked.join().unwrap();
640        readers.push((last_a, last_b));
641        // Release every remaining withheld acknowledgement.
642        drop(release_tx);
643        assert!(sink.wait_drained(Duration::from_secs(10)));
644        assert_eq!(
645            sink.counts(),
646            (
647                (MAX_IN_FLIGHT_BUNDLES + 1) as u64,
648                (MAX_IN_FLIGHT_BUNDLES + 1) as u64
649            )
650        );
651        for (read_a, read_b) in &readers {
652            assert!(is_eof(read_a) && is_eof(read_b));
653        }
654        drop(sink);
655        // The Arc in the blocked thread is gone; closing the last sink
656        // copy above ended the socket, so the helper exits.
657        thread.join().unwrap();
658    }

660    /// A hold survives until its bundle's acknowledgement is drained:
661    /// while the helper withholds the ack, the hold is not dropped,
662    /// so a guest-memory clone in it cannot unmap against a live VM.
663    #[test]
664    fn hold_drops_only_after_acknowledgement() {
665        struct Flag(Arc<std::sync::atomic::AtomicBool>);
666        impl Drop for Flag {
667            fn drop(&mut self) {
668                self.0.store(true, std::sync::atomic::Ordering::SeqCst);
669            }
670        }
671        let (sender, helper) = UnixStream::pair().unwrap();
672        let (release_tx, release_rx) = mpsc::channel::<()>();
673        let thread = std::thread::spawn(move || fake_helper(helper, Some(release_rx)));
674        let sink = HelperTeardownSink::new(sender);
675        let (_read_a, write_a) = pipe();
676        let (_read_b, write_b) = pipe();
677        let dropped = Arc::new(std::sync::atomic::AtomicBool::new(false));
678        sink.adopt_vm_with_hold(write_a, write_b, Box::new(Flag(Arc::clone(&dropped))));
679        assert!(
680            !dropped.load(std::sync::atomic::Ordering::SeqCst),
681            "hold dropped before its bundle was acknowledged"
682        );
683        release_tx.send(()).unwrap();
684        assert!(sink.wait_drained(Duration::from_secs(10)));
685        assert!(dropped.load(std::sync::atomic::Ordering::SeqCst));
686        drop(release_tx);
687        drop(sink);
688        thread.join().unwrap();
689    }

692    fn dead_helper_falls_back_to_local_close() {
693        let (sender, helper) = UnixStream::pair().unwrap();
694        drop(helper);
695        let sink = HelperTeardownSink::new(sender);
696        let (read_a, write_a) = pipe();
697        let (read_b, write_b) = pipe();
698        // The send fails (no receiver); the duplicates close locally
699        // and nothing is counted as adopted.
700        sink.adopt_vm(write_a, write_b);
701        assert!(sink.is_dead());
702        assert_eq!(sink.counts(), (0, 0));
703        assert!(is_eof(&read_a) && is_eof(&read_b));
704        // Later adoptions are local closes too, without blocking.
705        let (read_c, write_c) = pipe();
706        let (read_d, write_d) = pipe();
707        sink.adopt_vm(write_c, write_d);
708        assert!(is_eof(&read_c) && is_eof(&read_d));
709        assert_eq!(sink.fallbacks(), 2);
710    }

712    #[test]
713    fn malformed_bundle_is_refused_and_its_fds_closed() {
714        let (sender, helper) = UnixStream::pair().unwrap();
715        let (read_a, write_a) = pipe();
716        // One descriptor where a bundle promises two.
717        let sent = {
718            let mut byte = BUNDLE_BYTE;
719            let mut iov = libc::iovec {
720                iov_base: (&mut byte as *mut u8).cast(),
721                iov_len: 1,
722            };
723            let fds = [write_a.as_raw_fd()];
724            let mut control = [0usize; 8];
725            let mut msg: libc::msghdr = unsafe { std::mem::zeroed() };
726            msg.msg_iov = &mut iov;
727            msg.msg_iovlen = 1;
728            msg.msg_control = control.as_mut_ptr().cast();
729            msg.msg_controllen = unsafe { libc::CMSG_SPACE(size_of::<RawFd>() as u32) } as usize;
730            // SAFETY: as in send_vm_bundle, with a one-fd payload.
731            unsafe {
732                let cmsg = libc::CMSG_FIRSTHDR(&msg);
733                (*cmsg).cmsg_level = libc::SOL_SOCKET;
734                (*cmsg).cmsg_type = libc::SCM_RIGHTS;
735                (*cmsg).cmsg_len = libc::CMSG_LEN(size_of::<RawFd>() as u32) as usize;
736                std::ptr::write(libc::CMSG_DATA(cmsg).cast::<RawFd>(), fds[0]);
737                libc::sendmsg(sender.as_raw_fd(), &msg, libc::MSG_NOSIGNAL)
738            }
739        };
740        assert_eq!(sent, 1);
741        drop(write_a);
742        assert!(recv_vm_bundle(helper.as_raw_fd()).is_err());
743        // The received duplicate was closed with the refusal, so the
744        // pipe reaches EOF once the sender stream is gone too.
745        drop(sender);
746        assert!(is_eof(&read_a));
747    }
