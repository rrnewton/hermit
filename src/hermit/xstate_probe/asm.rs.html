9//! The probe's measured windows, written in assembly
10//! (<https://github.com/rrnewton/hermit/issues/4243>).
11//!
12//! Each arm runs from "every register of every enabled component holds a known
13//! value" through "the case's initializing step" to "XSAVE of every enabled
14//! component" inside one assembly function. No compiler-generated code runs in
15//! that window, so nothing Rust emits (a `memcpy` using SSE, say) can change the
16//! state being measured. The Rust driver prepares the buffers before a call and
17//! reads the certificates (interrupt counter, `getrusage`, schedstat) outside it.
18
19use std::arch::global_asm;
20
21/// What an arm does after its initializing step and before its XSAVE.
22#[repr(u32)]
23#[derive(Clone, Copy, Debug, PartialEq, Eq)]
24pub enum Mode {
25    /// Nothing: the reference.
26    Reference = 0,
27    /// Hand off to the peer on the same CPU and wait for its reply, so the
28    /// kernel saves this thread's FPU state, runs the peer's, and reloads ours.
29    Reload = 1,
30    /// Execute CPUID, which exits to a hypervisor when there is one.
31    ForcedExit = 2,
32}
33
34/// The initializing step of a probe case. The numbering is shared with the
35/// assembly dispatch below.
36#[repr(u32)]
37#[derive(Clone, Copy, Debug, PartialEq, Eq)]
38pub enum Kind {
39    /// Standard-form XRSTOR of one component with its header bit clear.
40    XrstorInit = 0,
41    /// Standard-form XRSTOR of one component with its header bit set and
42    /// initial values in the image.
43    XrstorHeaderSet = 1,
44    /// WRPKRU 0.
45    WrpkruZero = 2,
46    /// WRPKRU 0, then a signal to this thread, whose return restores the frame.
47    Sigreturn = 3,
48    /// VZEROUPPER with XMM live.
49    Vzeroupper = 4,
50    /// Legacy PXOR of XMM0-15 with the YMM upper halves live.
51    LegacyPxor = 5,
52    /// VEX-128 VPXOR of XMM0-15 with Hi16_ZMM live.
53    Vex128Zero = 6,
54    /// VPXORD of ZMM16-31 alone.
55    Hi16Zero = 7,
56    /// PXOR of XMM0-15, then VZEROUPPER, then ZMM16-31 zeroed.
57    PxorVzeroupperHi16 = 8,
58    /// FNINIT, FLDZ ×8, FNINIT: every ST register zero.
59    X87FullInit = 9,
60    /// Two more pushes first, then the same full x87 initialization.
61    X87DirtyFullInit = 10,
62    /// Compacted-form XRSTOR of one component with its header bit clear
63    /// (glibc's lazy-binding `_dl_runtime_resolve_xsavec` path).
64    CompactedXrstorInit = 11,
65    /// FXRSTOR of initial x87 and SSE values.
66    FxrstorInit = 12,
67    /// KXORQ of k0-7 with the vectors live.
68    OpmaskZero = 13,
69    /// VZEROALL, plus ZMM16-31 zeroed where AVX-512 is enabled.
70    VzeroallAll = 14,
71}

73/// The state of one arm call, shared with the assembly. Field offsets are fixed
74/// by the assembly below; the test `arm_ctx_layout_matches_the_assembly` checks them.
75#[repr(C)]
76#[derive(Debug)]
77pub struct ArmCtx {
78    pub kind: u32,          // 0
79    pub component: u32,     // 4
80    pub mxcsr: u32,         // 8
81    pub mode: u32,          // 12
82    pub xcr0: u64,          // 16
83    pub save_area: *mut u8, // 24: XSAVE destination, 64-byte aligned
84    pub image: *const u8,   // 32: XRSTOR/FXRSTOR source, 64-byte aligned
85    pub fd_to_peer: i32,    // 40
86    pub fd_from_peer: i32,  // 44
87    pub tgid: i32,          // 48
88    pub tid: i32,           // 52
89    pub ones: *const u8,    // 56: 64 nonzero bytes, 64-byte aligned
90    pub pkru_out: u32,      // 64
91    pub avx512: u32,        // 68: whether ZMM16-31 and k0-7 exist
92    pub cpuid_cycles: u64,  // 72: the forced exit's CPUID round trip
93    pub seq: u8,            // 80: handshake byte sent
94    pub reply: u8,          // 81: handshake byte received
95    pub pkru: u8,           // 82: nonzero when RDPKRU and WRPKRU may run
96    pub _pad: u8,           // 83
97    pub status: u32,        // 84: an `ArmStatus`
98    pub saved_pkru: u32,    // 88: scratch
99    pub saved_mxcsr: u32,   // 92: scratch
100}

102/// How an arm call ended. Anything but `Ok` voids the trial.
103#[repr(u32)]
104#[derive(Clone, Copy, Debug, PartialEq, Eq)]
105pub enum ArmStatus {
106    Ok = 0,
107    WriteFailed = 1,
108    ReadFailed = 2,
109    WrongReply = 3,
110    TgkillFailed = 4,
111}
112
113impl ArmStatus {
114    pub fn from_raw(raw: u32) -> Option<Self> {
115        Some(match raw {
116            0 => ArmStatus::Ok,
117            1 => ArmStatus::WriteFailed,
118            2 => ArmStatus::ReadFailed,
119            3 => ArmStatus::WrongReply,
120            4 => ArmStatus::TgkillFailed,
121            _ => return None,
122        })
123    }
124}

126/// The state the peer leaves on the shared CPU before it replies.
127#[repr(u32)]
128#[derive(Clone, Copy, Debug, PartialEq, Eq)]
129pub enum PeerState {
130    /// Every register dirty, PKRU `0x55555550` (differs from the probe's).
131    DirtyPkruDiffers = 0,
132    /// Every register dirty, PKRU 0 by WRPKRU.
133    DirtyWrpkruZero = 1,
134    /// Every component initialized by a zero-header XRSTOR (PKRU 0 included).
135    XrstorInitAll = 2,
136    /// Every register dirty, PKRU 0 by a header-loaded XRSTOR (bit 9 set).
137    DirtyPkruHeaderLoaded = 3,
138    /// Every register dirty, PKRU 0, then a signal and its return.
139    DirtyPkruSigreturn = 4,
140    /// Initial values with every header bit set.
141    InitValuesAllHeaderBits = 5,
142}

144/// The peer thread's state, shared with the assembly; offsets as noted.
145#[repr(C)]
146#[derive(Debug)]
147pub struct PeerCtx {
148    pub state: u32,        // 0: a `PeerState`, set before each handshake
149    pub fd_in: i32,        // 4
150    pub fd_out: i32,       // 8
151    pub avx512: u32,       // 12
152    pub image: *const u8,  // 16: the XRSTOR source for states 2, 3 and 5
153    pub xcr0: u64,         // 24
154    pub ones: *const u8,   // 32
155    pub tgid: i32,         // 40
156    pub tid: i32,          // 44
157    pub expected_cpu: u32, // 48
158    pub wrong_cpu: u32,    // 52: rounds the peer ran on another CPU
159    pub status: u32,       // 56: why the loop ended early, as `ArmStatus`
160    pub rounds: u32,       // 60
161    pub byte: u8,          // 64: received
162    pub reply: u8,         // 65: sent
163    pub pkru: u8,          // 66: nonzero when RDPKRU and WRPKRU may run
164    pub _pad: u8,          // 67
165    pub cpu_out: u32,      // 68: getcpu result
166    pub saved_pkru: u32,   // 72
167    pub saved_mxcsr: u32,  // 76
168}

170/// The handshake byte that ends the peer loop.
171pub const PEER_EXIT: u8 = 0xff;

216// WRPKRU and RDPKRU raise #UD unless the kernel enables protection keys
217// (CR4.PKE), so both run only when the context's PKRU byte is set; without it
218// the PKRU value read is 0.
219.macro XINUSE_WRPKRU value, ctxreg, flagoff
220    cmp byte ptr [\ctxreg + \flagoff], 0
221    je 91f
222    mov eax, \value
223    xor ecx, ecx
224    xor edx, edx
225    wrpkru
22691:
227.endm
228
229.macro XINUSE_RDPKRU ctxreg, flagoff
230    xor eax, eax
231    cmp byte ptr [\ctxreg + \flagoff], 0
232    je 92f
233    xor ecx, ecx
234    rdpkru
23592:
236.endm
237
238.macro XINUSE_RESTORE_PKRU ctxreg, flagoff, savedoff
239    cmp byte ptr [\ctxreg + \flagoff], 0
240    je 93f
241    mov eax, [\ctxreg + \savedoff]
242    xor ecx, ecx
243    xor edx, edx
244    wrpkru
24593:
246.endm
247
248.macro XINUSE_XRSTOR_MASK ctxreg, imgoff, maskreg
249    mov rax, \maskreg
250    mov rdx, rax
251    shr rdx, 32
252    mov r8, [\ctxreg + \imgoff]
253    xrstor [r8]
254.endm
