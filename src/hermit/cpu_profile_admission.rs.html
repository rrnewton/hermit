9//! Whether a run may use `--cpu-profile x86-64-v3`, and the host facts it
10//! binds when it may (<https://github.com/rrnewton/hermit/issues/4243>).
11//!
12//! Every class the startup qualification cannot cover is refused by name; the
13//! run never falls back to v2.

20/// Why a v3 run is refused, as one sentence after "--cpu-profile x86-64-v3
21/// refused: ".
22pub type AdmissionRefusal = String;
23
24/// A refused x86-64-v3 recording or replay, carried as an error so that it
25/// crosses the container boundary as a policy refusal (exit 122).
26#[derive(Debug, Clone, PartialEq, Eq)]
27pub struct CpuProfileRefusal(pub AdmissionRefusal);
28
29impl std::fmt::Display for CpuProfileRefusal {
30    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
31        write!(f, "--cpu-profile x86-64-v3 refused: {}", self.0)
32    }
33}
34
35impl std::error::Error for CpuProfileRefusal {}

37/// The admission request of a recording or replay, which always runs on the
38/// ptrace backend with virtual CPUID.
39pub fn record_or_replay_request(gdbserver: bool) -> AdmissionRequest<'static> {
40    AdmissionRequest {
41        backend: "ptrace",
42        ptrace: true,
43        virtualize_cpuid: true,
44        strace_only: false,
45        gdbserver,
46        namespace_only: false,
47    }
48}

50/// The first host fact in which `replay` differs from `recorded`, by name, or
51/// `None` when they are equal.
52pub fn first_difference(recorded: &HostXstate, replay: &HostXstate) -> Option<&'static str> {
53    if recorded.vendor != replay.vendor {
54        Some("the CPU vendor")
55    } else if recorded.signature != replay.signature {
56        Some("the CPU family, model or stepping")
57    } else if recorded.xcr0 != replay.xcr0 {
58        Some("XCR0")
59    } else if recorded.xsave_leaves != replay.xsave_leaves {
60        Some("the XSAVE layout (CPUID leaf 0xD)")
61    } else if (
62        recorded.xcomp_supp,
63        recorded.xcomp_perm,
64        recorded.xcomp_guest_perm,
65    ) != (
66        replay.xcomp_supp,
67        replay.xcomp_perm,
68        replay.xcomp_guest_perm,
69    ) {
70        Some("the ARCH_GET_XCOMP masks")
71    } else if recorded.xfeature_max != replay.xfeature_max {
72        Some("XFEATURE_MAX")
73    } else if (recorded.probe_cases, recorded.probe_forced_exit)
74        != (replay.probe_cases, replay.probe_forced_exit)
75    {
76        Some("the qualification it passed")
77    } else {
78        None
79    }
80}

82/// What the run asks for, as far as admission is concerned.
83#[derive(Clone, Copy, Debug)]
84pub struct AdmissionRequest<'a> {
85    /// The backend's command-line name, and whether it is the ptrace backend.
86    pub backend: &'a str,
87    pub ptrace: bool,
88    pub virtualize_cpuid: bool,
89    pub strace_only: bool,
90    pub gdbserver: bool,
91    /// `--namespace-only`, which runs the guest with no Detcore at all.
92    pub namespace_only: bool,
93}

95/// The refusals that need no host probe, in the order they are checked.
96pub fn check_request(request: &AdmissionRequest<'_>) -> Result<(), AdmissionRefusal> {
97    if !request.ptrace {
98        let why = match request.backend {
99            "kvm" => {
100                "the KVM backend switches XCR0 between the guest's and the host's on every VM \
101                 exit, a transition the startup qualification does not cover"
102            }
103            "dbt" => {
104                "Detcore answers DBT's CPUID, but the DBT runtime's configuration carries no \
105                 CPU profile, so it answers only the x86-64-v2 table"
106            }
107            "sabre" | "liteinst" | "in-guest-trap" => {
108                "this backend does not trap CPUID, so the guest would see the host's"
109            }
110            _ => "only the ptrace backend is qualified",
111        };
112        return Err(format!("--backend={}: {why}", request.backend));
113    }
114    if request.namespace_only {
115        return Err(
116            "--namespace-only runs the guest without Detcore, so nothing answers its CPUID".into(),
117        );
118    }
119    if !request.virtualize_cpuid || request.strace_only {
120        return Err(
121            "the profile is a virtual CPUID table, and this run does not virtualize CPUID".into(),
122        );
123    }
124    if request.gdbserver {
125        return Err(
126            "--gdbserver writes the guest's FP registers, and the kernel's regset write forces \
127             XSAVE header bits the qualification binds"
128                .into(),
129        );
130    }
131    Ok(())
132}

134/// Linux's version as (major, minor), from a `uname -r` release string.
135pub fn kernel_version(release: &str) -> Option<(u32, u32)> {
136    let mut parts = release.split(['.', '-']);
137    let major = parts.next()?.parse().ok()?;
138    let minor = parts.next()?.parse().ok()?;
139    Some((major, minor))
140}

142/// Linux 5.16 is the first with the `ARCH_*_XCOMP_*` queries the profile
143/// answers from the host; before it they return EINVAL.
144pub fn check_kernel(release: &str) -> Result<(), AdmissionRefusal> {
145    match kernel_version(release) {
146        Some(version) if version >= (5, 16) => Ok(()),
147        Some((major, minor)) => Err(format!(
148            "Linux {major}.{minor} predates the 5.16 xstate queries the profile answers from"
149        )),
150        None => Err(format!("cannot read the kernel version from {release:?}")),
151    }
152}

154/// A host CPUID reader, so tests can supply a host.
155pub trait HostCpuid {
156    fn cpuid(&self, leaf: u32, subleaf: u32) -> [u32; 4];
157}
158
159/// The CPUID instruction of the host this process runs on.
160pub struct NativeCpuid;
161
162impl HostCpuid for NativeCpuid {
163    fn cpuid(&self, leaf: u32, subleaf: u32) -> [u32; 4] {
164        let r = std::arch::x86_64::__cpuid_count(leaf, subleaf);
165        [r.eax, r.ebx, r.ecx, r.edx]
166    }
167}
168
169const REGISTER_NAMES: [&str; 4] = ["EAX", "EBX", "ECX", "EDX"];
170
171/// The host checks that need only CPUID, in the order they are checked:
172/// every advertised v3 feature bit natively, OSXSAVE, not a hybrid part, and
173/// not an AMD part without XSAVEERPTR (where Linux runs `fnclex; emms; fildl`
174/// before every FPU restore, a path no host here can exercise).
175pub fn check_host_cpuid(
176    host: &dyn HostCpuid,
177    advertised: &[(u32, u32, usize, u32)],
178) -> Result<(), AdmissionRefusal> {
179    let max_basic = host.cpuid(0, 0)[0];
180    let max_extended = host.cpuid(0x8000_0000, 0)[0];
181    let mut missing = Vec::new();
182    for &(leaf, subleaf, register, bits) in advertised {
183        let supported = if leaf >= 0x8000_0000 {
184            leaf <= max_extended
185        } else {
186            leaf <= max_basic
187        };
188        let native = if supported {
189            host.cpuid(leaf, subleaf)[register]
190        } else {
191            0
192        };
193        let absent = bits & !native;
194        for bit in 0..32 {
195            if absent & (1 << bit) != 0 {
196                missing.push(format!(
197                    "leaf {leaf:#x}.{subleaf} {} bit {bit}",
198                    REGISTER_NAMES[register]
199                ));
200            }
201        }
202    }
203    if !missing.is_empty() {
204        return Err(format!(
205            "the host lacks feature bits the profile advertises: {}",
206            missing.join(", ")
207        ));
208    }
209    if host.cpuid(1, 0)[2] & (1 << 27) == 0 {
210        return Err("the host kernel has not enabled XSAVE (OSXSAVE is clear)".into());
211    }
212    if max_basic >= 7 && host.cpuid(7, 0)[3] & (1 << 15) != 0 {
213        return Err(
214            "this is a hybrid part (leaf 7 EDX bit 15), whose core types the qualification \
215             would each have to sample"
216                .into(),
217        );
218    }
219    let vendor = host.cpuid(0, 0);
220    let amd = vendor[1] == 0x6874_7541;
221    let xsaveerptr = max_extended >= 0x8000_0008 && host.cpuid(0x8000_0008, 0)[1] & (1 << 2) != 0;
222    if amd && !xsaveerptr {
223        return Err(
224            "this AMD part lacks XSAVEERPTR, so Linux runs `fnclex; emms; fildl` before every \
225             FPU restore, which the qualification cannot exercise"
226                .into(),
227        );
228    }
229    Ok(())
230}

232/// The user-dynamic xstate component (AMX TILEDATA): Linux enables it in XCR0
233/// while withholding process permission, so a permission request changes
234/// process state and the three XCOMP answers differ.
235const XFEATURE_TILEDATA: u64 = 1 << 18;
236
237const ARCH_GET_XCOMP_SUPP: libc::c_int = 0x1021;
238const ARCH_GET_XCOMP_PERM: libc::c_int = 0x1022;
239const ARCH_REQ_XCOMP_PERM: libc::c_int = 0x1023;
240const ARCH_GET_XCOMP_GUEST_PERM: libc::c_int = 0x1024;
241
242fn arch_prctl_mask(cmd: libc::c_int) -> Result<u64, AdmissionRefusal> {
243    let mut mask = 0u64;
244    // SAFETY: the GET commands write one u64 through the pointer.
245    let rc = unsafe { libc::syscall(libc::SYS_arch_prctl, cmd, &mut mask as *mut u64) };
246    if rc != 0 {
247        return Err(format!(
248            "arch_prctl({cmd:#x}) failed: {}",
249            std::io::Error::last_os_error()
250        ));
251    }
252    Ok(mask)
253}
254
255/// Linux's XFEATURE_MAX as ARCH_REQ_XCOMP_PERM shows it: the first component
256/// index it answers EINVAL for. Every index below must answer EOPNOTSUPP; an
257/// index that is granted would mean a dynamic feature, which is refused
258/// before this runs.
259fn xfeature_max() -> Result<u32, AdmissionRefusal> {
260    for index in 0..64u64 {
261        // SAFETY: the request takes the component index by value.
262        let rc = unsafe { libc::syscall(libc::SYS_arch_prctl, ARCH_REQ_XCOMP_PERM, index) };
263        let errno = std::io::Error::last_os_error().raw_os_error();
264        match (rc, errno) {
265            (-1, Some(libc::EINVAL)) => return Ok(index as u32),
266            (-1, Some(libc::EOPNOTSUPP)) => {}
267            _ => {
268                return Err(format!(
269                    "ARCH_REQ_XCOMP_PERM for component {index} answered {rc} ({errno:?}), \
270                     not EOPNOTSUPP or EINVAL"
271                ));
272            }
273        }
274    }
275    Err("ARCH_REQ_XCOMP_PERM never answered EINVAL for indices 0-63".into())
276}

278/// The host facts a qualified v3 run binds.
279fn capture(probe: &xstate_probe::ProbeReport) -> Result<HostXstate, AdmissionRefusal> {
280    let host = host_facts(probe.cases as u32, probe.forced_exit)?;
281    if host.xcr0 != probe.xcr0 {
282        return Err(format!(
283            "XCR0 changed during admission ({:#x}, then {:#x})",
284            probe.xcr0, host.xcr0
285        ));
286    }
287    Ok(host)
288}

290/// This host's facts, with the qualification fields given rather than probed.
291/// The caller has checked OSXSAVE.
292fn host_facts(probe_cases: u32, probe_forced_exit: bool) -> Result<HostXstate, AdmissionRefusal> {
293    let host = NativeCpuid;
294    let leaf0 = host.cpuid(0, 0);
295    // SAFETY: XGETBV with ECX 0 is available because OSXSAVE was checked.
296    let xcr0 = unsafe { std::arch::x86_64::_xgetbv(0) };
297    let mut xsave_leaves = Vec::new();
298    for subleaf in std::iter::once(0).chain(2..64) {
299        if subleaf == 0 || xcr0 & (1 << subleaf) != 0 {
300            let [eax, ebx, ecx, edx] = host.cpuid(0xd, subleaf);
301            xsave_leaves.push([subleaf, eax, ebx, ecx, edx]);
302        }
303    }
304    let xcomp_supp = arch_prctl_mask(ARCH_GET_XCOMP_SUPP)?;
305    refuse_dynamic_features(xcr0, xcomp_supp)?;
306    Ok(HostXstate {
307        vendor: [leaf0[1], leaf0[3], leaf0[2]],
308        signature: host.cpuid(1, 0)[0],
309        xcr0,
310        xsave_leaves,
311        xcomp_supp,
312        xcomp_perm: arch_prctl_mask(ARCH_GET_XCOMP_PERM)?,
313        xcomp_guest_perm: arch_prctl_mask(ARCH_GET_XCOMP_GUEST_PERM)?,
314        xfeature_max: xfeature_max()?,
315        probe_cases,
316        probe_forced_exit,
317    })
318}

320/// Refuses a v3 replay before the qualification probe runs when the recording
321/// carries no host facts, when this host differs in one of the facts it bound
322/// (the qualification itself aside), or when its CPUID table digest is not the
323/// one this replayer computes from them. The probe costs seconds of CPU, and a
324/// replay on a different CPU, or of an altered recording, needs none of it to
325/// be refused. Returns the recorded facts.
326pub fn precheck_replay(
327    profile: CpuProfile,
328    recorded: Option<&HostXstate>,
329    recorded_digest: Option<&str>,
330) -> Result<HostXstate, CpuProfileRefusal> {
331    let refuse = |why: String| CpuProfileRefusal(why);
332    let recorded = recorded.cloned().ok_or_else(|| {
333        refuse("the recording names the profile but carries no host facts".into())
334    })?;
335    check_host_cpuid(&NativeCpuid, &detcore::v3_feature_words()).map_err(refuse)?;
336    let host = host_facts(recorded.probe_cases, recorded.probe_forced_exit).map_err(refuse)?;
337    if let Some(fact) = first_difference(&recorded, &host) {
338        return Err(refuse(format!(
339            "this host differs from the recording's in {fact}, so the guest would see a \
340             different CPU"
341        )));
342    }
343    let config = detcore::Config {
344        cpu_profile: profile,
345        cpu_profile_host: Some(recorded.clone()),
346        ..Default::default()
347    };
348    let digest = crate::record::hex_digest(&detcore::virtual_cpuid_digest(&config));
349    if recorded_digest != Some(digest.as_str()) {
350        return Err(refuse(format!(
351            "the recording's CPUID table digest {recorded_digest:?} is not this replayer's \
352             {digest}"
353        )));
354    }
355    Ok(recorded)
356}

358/// This process's successful qualification. A recording or replay qualifies
359/// in the CLI process, before its container child is forked, so the child
360/// reuses this result instead of probing again from inside the container,
361/// whose CPU affinity can be one busy CPU.
362static ADMITTED: std::sync::OnceLock<(xstate_probe::ProbeReport, HostXstate)> =
363    std::sync::OnceLock::new();
364
365/// Admits a v3 run on this host, or names why not. On success the probe report
366/// and the host facts to bind are returned, and this process's ptrace
367/// floating-point register writes are refused from then on. The request's own
368/// refusals are checked every time; the host qualification runs once per
369/// process.
370pub fn admit(
371    request: &AdmissionRequest<'_>,
372) -> Result<(xstate_probe::ProbeReport, HostXstate), AdmissionRefusal> {
373    admit_with(request, qualify_host)
374}
375
376/// [`admit`] with its host qualification as a parameter, so that a test can
377/// admit without probing.
378fn admit_with(
379    request: &AdmissionRequest<'_>,
380    qualify: impl FnOnce() -> Result<(xstate_probe::ProbeReport, HostXstate), AdmissionRefusal>,
381) -> Result<(xstate_probe::ProbeReport, HostXstate), AdmissionRefusal> {
382    check_request(request)?;
383    if let Some(admitted) = ADMITTED.get() {
384        return Ok(admitted.clone());
385    }
386    let qualified = qualify()?;
387    // The qualification covers the XSTATE_BV a guest's own instructions
388    // produce. A tracer write of its floating-point registers (PTRACE_SETFPREGS
389    // or PTRACE_SETREGSET NT_X86_XSTATE) sets that header from the tracer's
390    // buffer instead, which nothing checked, so it fails with EPERM. The flag
391    // is process-wide and set before the container child is forked, so the
392    // tracer inherits it. It is set before the admission becomes visible in
393    // [`ADMITTED`], so an admitted launch always runs with it.
394    safeptrace::forbid_fpu_register_writes();
395    Ok(ADMITTED.get_or_init(|| qualified).clone())
396}

398/// Refuses a launch of an x86-64-v3 configuration that this process's
399/// admission did not produce. Hermit's launch functions call this before any
400/// backend starts, so a library caller cannot run a v3 guest on host facts of
401/// its own, on another backend, or without the floating-point register write
402/// ban: the facts must be exactly the ones [`admit`] bound after the host
403/// qualified, and [`admit`] is the only way to bind them.
404pub fn refuse_unadmitted_launch(
405    config: &detcore::Config,
406    ptrace: bool,
407) -> Result<(), CpuProfileRefusal> {
408    if config.cpu_profile.is_default() {
409        return Ok(());
410    }
411    let refuse = |why: &str| Err(CpuProfileRefusal(why.to_owned()));
412    if !ptrace {
413        return refuse("only the ptrace backend is qualified");
414    }
415    if !config.virtualize_cpuid || config.gdbserver {
416        return refuse("the launch does not virtualize CPUID, or serves a gdbserver");
417    }
418    match (&config.cpu_profile_host, ADMITTED.get()) {
419        (None, _) => return refuse("the configuration carries no qualified host facts"),
420        (Some(_), None) => return refuse("this process did not qualify the host"),
421        (Some(facts), Some((_, admitted))) if facts != admitted => {
422            return refuse(
423                "the configuration's host facts are not the ones this process qualified",
424            );
425        }
426        _ => {}
427    }
428    if !safeptrace::fpu_register_writes_forbidden() {
429        return refuse("ptrace floating-point register writes are not forbidden");
430    }
431    Ok(())
432}

434/// The host half of [`admit`], run once per process: the kernel, the
435/// every-bit CPUID check, the dynamic-feature refusal, the probe and the host
436/// facts. `hermit host-capabilities --cpu-profile x86-64-v3` runs it alone.
437pub fn qualify_host() -> Result<(xstate_probe::ProbeReport, HostXstate), AdmissionRefusal> {
438    let release = nix::sys::utsname::uname()
439        .map_err(|e| format!("uname failed: {e}"))?
440        .release()
441        .to_string_lossy()
442        .into_owned();
443    check_kernel(&release)?;
444    check_host_cpuid(&NativeCpuid, &detcore::v3_feature_words())?;
445    // SAFETY: OSXSAVE was checked above.
446    let xcr0 = unsafe { std::arch::x86_64::_xgetbv(0) };
447    refuse_dynamic_features(xcr0, 0)?;
448    let probe = xstate_probe::qualify(xstate_probe::ProbeOptions::default())?;
449    let host = capture(&probe)?;
450    Ok((probe, host))
451}

453/// Refuses a host with a user-dynamic xstate feature (AMX TILEDATA, bit 18) in
454/// XCR0 or in ARCH_GET_XCOMP_SUPP: a guest's permission request for it changes
455/// process state, which the profile cannot bind.
456fn refuse_dynamic_features(xcr0: u64, xcomp_supp: u64) -> Result<(), AdmissionRefusal> {
457    if (xcr0 | xcomp_supp) & XFEATURE_TILEDATA != 0 {
458        return Err(
459            "the host has a user-dynamic xstate feature (AMX TILEDATA), whose permission \
460             requests change process state"
461                .into(),
462        );
463    }
464    Ok(())
465}

589        };
590        let admitted = admit_with(&request("ptrace"), || {
591            // The ban is not yet in force while the host qualifies.
592            assert!(!safeptrace::fpu_register_writes_forbidden());
593            Ok((report.clone(), host.clone()))
594        });
595        assert_eq!(admitted, Ok((report, host.clone())));
596        assert!(safeptrace::fpu_register_writes_forbidden());
597        assert_eq!(refuse_unadmitted_launch(&config, true), Ok(()));
598        assert!(refuse_unadmitted_launch(&config, false).is_err());
599        let mut other = host;
600        other.probe_cases = 0;
601        config.cpu_profile_host = Some(other);
602        let forged = refuse_unadmitted_launch(&config, true).unwrap_err();
603        assert!(
604            forged.0.contains("not the ones this process qualified"),
605            "{forged}"
606        );
607        config.cpu_profile_host = None;
608        assert!(refuse_unadmitted_launch(&config, true).is_err());
609    }

666    struct FakeHost(BTreeMap<(u32, u32), [u32; 4]>);
667
668    impl HostCpuid for FakeHost {
669        fn cpuid(&self, leaf: u32, subleaf: u32) -> [u32; 4] {
670            self.0.get(&(leaf, subleaf)).copied().unwrap_or([0; 4])
671        }
672    }
673
674    /// A bare-metal-shaped host with every v3 bit, AMD with XSAVEERPTR.
675    fn qualifying_host() -> FakeHost {
676        let mut regs = BTreeMap::new();
677        regs.insert((0, 0), [0xd, 0x6874_7541, 0x444d_4163, 0x6974_6e65]);
678        regs.insert((0x8000_0000, 0), [0x8000_0008, 0, 0, 0]);
679        regs.insert((0x8000_0008, 0), [0, 1 << 2, 0, 0]);
680        for &(leaf, subleaf, register, bits) in &detcore::v3_feature_words() {
681            regs.entry((leaf, subleaf)).or_insert([0; 4])[register] |= bits;
682        }
683        regs.get_mut(&(1, 0)).unwrap()[2] |= 1 << 27;
684        FakeHost(regs)
685    }

687    #[test]
688    fn a_host_with_every_v3_bit_qualifies_and_removing_any_one_is_named() {
689        let words = detcore::v3_feature_words();
690        let host = qualifying_host();
691        assert_eq!(check_host_cpuid(&host, &words), Ok(()));
692        // The hypervisor bit is not required: v3 clears it.
693        assert!(words.iter().all(|&(leaf, _, register, bits)| !(leaf == 1
694            && register == 2
695            && bits & (1 << 31) != 0)));
696        let mut checked = 0;
697        for &(leaf, subleaf, register, bits) in &words {
698            for bit in (0..32).filter(|b| bits & (1 << b) != 0) {
699                let mut host = qualifying_host();
700                host.0.get_mut(&(leaf, subleaf)).unwrap()[register] &= !(1 << bit);
701                if leaf == 1 && register == 2 && bit == 27 {
702                    continue; // OSXSAVE: refused by its own check, below
703                }
704                let refusal = check_host_cpuid(&host, &words).unwrap_err();
705                assert!(
706                    refusal.contains(&format!(
707                        "leaf {leaf:#x}.{subleaf} {} bit {bit}",
708                        REGISTER_NAMES[register]
709                    )),
710                    "{refusal}"
711                );
712                checked += 1;
713            }
714        }
715        let advertised: u32 = words.iter().map(|&(_, _, _, bits)| bits.count_ones()).sum();
716        // Every advertised bit but OSXSAVE, which has its own check.
717        assert_eq!(checked, advertised - 1);
718    }
