9//! Rewriting aarch64 load/store-exclusive retry loops to FEAT_LSE atomics.
10//!
11//! On aarch64 an `LDXR`/`STXR` pair retries whenever the local exclusive
12//! monitor is cleared between the two, which any exception does: a timer tick,
13//! an IPI, a page fault, a PMU overflow, a ptrace stop. Each retry retires the
14//! loop's branches again, so the retired-branch count of identical guest work
15//! depends on host interrupt timing, and Hermit's branch-counted schedule
16//! cannot be replayed exactly. A single LSE atomic cannot retry, so a program
17//! whose exclusive loops have been rewritten retires a fixed number of
18//! branches for the same work.
19//!
20//! This is binary rewriting used as *preprocessing* for the ptrace backend, in
21//! the same sense as e9patch (see `docs/E9PATCH_COMPATIBILITY.md`): it never
22//! loads Detcore as a Reverie tool and is not a backend.
23//!
24//! The decoder, the idioms and the rewrite of an ELF file's bytes are
25//! Detcore's ([`detcore::aarch64_lse`]), re-exported here; this module adds
26//! what only the overlay needs: the host check, the cache and the file
27//! writes.
28

31pub use detcore::aarch64_lse::insn;
32pub use detcore::aarch64_lse::mapping;

35/// Whether this host executes the FEAT_LSE atomics the rewrite emits.
36///
37/// The rewritten image is a program for the machine that is about to run it:
38/// without FEAT_LSE every rewritten site would raise SIGILL.
39pub fn host_has_lse_atomics() -> bool {
40    #[cfg(target_arch = "aarch64")]
41    {
42        // HWCAP_ATOMICS, Linux Documentation/arch/arm64/elf_hwcaps.rst.
43        const HWCAP_ATOMICS: libc::c_ulong = 1 << 8;
44        // SAFETY: getauxval reads the process's own auxiliary vector.
45        (unsafe { libc::getauxval(libc::AT_HWCAP) } & HWCAP_ATOMICS) != 0
46    }
47    #[cfg(not(target_arch = "aarch64"))]
48    {
49        false
50    }
51}

60/// Where rewritten images are cached, without creating or checking
61/// anything: the directory [`cache_dir`] uses. It follows this process's
62/// environment (`XDG_CACHE_HOME`, `HOME`), so a caller that remembers what
63/// it prepared must remember this too.
64///
65/// Never under `/tmp`: Hermit replaces the guest's `/tmp` with an isolated
66/// directory, so an artifact there would not be visible to the guest that
67/// has to map it.
68pub fn cache_path() -> PathBuf {
69    let base = dirs::cache_dir().map_or_else(
70        || PathBuf::from("/var/tmp").join(format!("hermit-{}", nix::unistd::geteuid().as_raw())),
71        |dir| dir.join("hermit"),
72    );
73    // An XDG cache under /tmp is no use: Hermit replaces the guest's /tmp
74    // with an isolated directory, so an artifact there is not visible to the
75    // guest that has to map it.
76    let base = if base.starts_with("/tmp") {
77        PathBuf::from("/var/tmp").join(format!("hermit-{}", nix::unistd::geteuid().as_raw()))
78    } else {
79        base
80    };
81    base.join("exclusive-rewrites")
82}

84/// The cache entry holding `rewritten`, the rewrite of `source`, which the
85/// overlay bind-mounts over the path the guest loads `source` from; written
86/// if need be.
87///
88/// The entry is named by everything that decides the bytes the guest maps:
89/// the rewriter's format, a digest of the rewritten bytes themselves, and the
90/// source's permissions, which the copy carries ([`elf::write_rewritten`]);
91/// `name`, the source's file name, is there for a reader. So two runs that
92/// would map different bytes -- two builds of the rewriter, say -- never
93/// share an entry, and what one run checked cannot be replaced by another
94/// run's bytes before it mounts it. Nor is an entry ever written in place: a
95/// missing one, or one that does not hold the bytes and the mode its name
96/// promises, is written whole under a temporary name and renamed over the
97/// name, so a run that already mounted the file it replaces keeps that file.
98pub fn cached_rewrite(
99    cache: &Path,
100    rewritten: &[u8],
101    source: &Path,
102    name: &str,
103) -> anyhow::Result<PathBuf> {
104    let mode = std::fs::metadata(source)
105        .with_context(|| format!("reading the permissions of {}", source.display()))?
106        .permissions()
107        .mode();
108    let entry = cache.join(format!(
109        "v{REWRITE_FORMAT_VERSION}-{}-{:o}-{name}",
110        detcore::Digest::new(rewritten),
111        mode & 0o7777
112    ));
113    let holds_it = std::fs::metadata(&entry)
114        .is_ok_and(|cached| cached.permissions().mode() == mode)
115        && std::fs::read(&entry).is_ok_and(|cached| cached == rewritten);
116    if !holds_it {
117        elf::write_rewritten(&entry, rewritten, source)?;
118    }
119    Ok(entry)
120}

122/// Where rewritten images are cached ([`cache_path`]), created private to
123/// this user.
124pub fn cache_dir() -> anyhow::Result<PathBuf> {
125    let dir = cache_path();
126    let base = dir
127        .parent()
128        .map(Path::to_path_buf)
129        .unwrap_or_else(|| dir.clone());
130    std::fs::create_dir_all(&dir)
131        .with_context(|| format!("failed to create the rewrite cache {}", dir.display()))?;
132    // The cache holds images the guest will execute, so every directory this
133    // code creates must be ours and private: the `/var/tmp/hermit-<uid>`
134    // fallback lives in a sticky directory anyone can write, so another user
135    // could pre-create it and choose what the cache offers. Only the
136    // directories this function creates are checked; the standard locations
137    // above them (a home directory, `/var/tmp`) are the system's business.
138    let uid = nix::unistd::geteuid().as_raw();
139    for path in [base.as_path(), dir.as_path()] {
140        let metadata = std::fs::symlink_metadata(path)
141            .with_context(|| format!("failed to inspect the rewrite cache {}", path.display()))?;
142        anyhow::ensure!(
143            metadata.is_dir() && !metadata.file_type().is_symlink() && metadata.uid() == uid,
144            "the rewrite cache path {} must be a real directory owned by uid {uid}",
145            path.display()
146        );
147        if metadata.permissions().mode() & 0o077 != 0 {
148            let mut permissions = metadata.permissions();
149            permissions.set_mode(0o700);
150            std::fs::set_permissions(path, permissions)?;
151        }
152    }
153    Ok(dir)
154}

156#[cfg(test)]
157mod tests {
158    use super::*;
159
160    /// Two rewrites of one source share one cache directory, and each gets
161    /// only its own bytes: entries are named by
162    /// the bytes they hold, written in either order, and a run that holds
163    /// one entry open still reads its bytes after the other run's write.
164    /// An entry that does not hold what its name promises is replaced by
165    /// a new file, not rewritten in place.
166    #[test]
167    fn each_rewrite_maps_only_its_own_bytes_from_one_cache() {
168        let directory = tempfile::tempdir().unwrap();
169        let cache = directory.path().join("cache");
170        std::fs::create_dir(&cache).unwrap();
171        let source = directory.path().join("libstripped.so");
172        let original = b"\x7fELF stripped, as the default leaves it".to_vec();
173        let other = b"\x7fELF stripped, another rewrite".to_vec();
174        std::fs::write(&source, &original).unwrap();
175        let entry =
176            |bytes: &[u8]| cached_rewrite(&cache, bytes, &source, "libstripped.so").unwrap();
177        let mapped = entry(&other);
178        let mut held = std::fs::File::open(&mapped).unwrap();
179        let default = entry(&original);
180        assert_ne!(default, mapped);
181        assert_eq!(std::fs::read(&mapped).unwrap(), other);
182        assert_eq!(std::fs::read(&default).unwrap(), original);
183        // An entry that lost its bytes is replaced by a new file; the one a
184        // run holds keeps what it held.
185        std::fs::remove_file(&mapped).unwrap();
186        std::fs::write(&mapped, b"someone else's bytes").unwrap();
187        let mut stale = std::fs::File::open(&mapped).unwrap();
188        assert_eq!(entry(&other), mapped);
189        assert_eq!(std::fs::read(&mapped).unwrap(), other);
190        let mut read = Vec::new();
191        std::io::Read::read_to_end(&mut stale, &mut read).unwrap();
192        assert_eq!(read, b"someone else's bytes");
193        read.clear();
194        std::io::Read::read_to_end(&mut held, &mut read).unwrap();
195        assert_eq!(read, other);
196    }
197}
