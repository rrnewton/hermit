6/// The verification verdict, independent of the guest's exit status.
7///
8/// A guest that deterministically exits nonzero can still match, while a guest
9/// that exits zero can still diverge. Consumers must read this value rather than
10/// infer the verdict from either the process status or human-readable stderr.
11#[derive(Clone, Copy, Debug, Deserialize, Eq, PartialEq, Serialize)]
12#[serde(rename_all = "snake_case")]
13pub enum Verdict {
14    /// The two runs matched on every compared dimension.
15    Matched,
16    /// The two runs diverged; verification failed.
17    Diverged,
18    /// Verification stopped before it could reach a verdict.
19    NoResult,
20    /// Verification completed enough work to name an understood machine or
21    /// harness failure, so the receipt is neither a product verdict nor an
22    /// unknown no-result.
23    InfrastructureError,
24}
25
26impl std::fmt::Display for Verdict {
27    fn fmt(&self, formatter: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
28        formatter.write_str(match self {
29            Self::Matched => "matched",
30            Self::Diverged => "diverged",
31            Self::NoResult => "no_result",
32            Self::InfrastructureError => "infrastructure_error",
33        })
34    }
35}

37/// Why verification did not reach a verdict.
38///
39/// This distinction is functional evidence. `FirstRunRejected` means the guest
40/// ran and its disposition was rejected by `--verify-allow`; `NotRun` is the
41/// pre-run stamp left when the invocation died before recording a more specific
42/// result. Collapsing both into `no_result` made a completed guest refusal and a
43/// container or launcher failure indistinguishable to the result reader.
44#[derive(Clone, Debug, Deserialize, Eq, PartialEq, Serialize)]
45#[serde(rename_all = "snake_case", tag = "kind")]
46pub enum NoResultReason {
47    /// The invocation did not replace the pre-run stamp with a more specific
48    /// outcome.
49    NotRun,
50    /// Run 1 completed and its exit status was rejected by `--verify-allow`.
51    FirstRunRejected {
52        exit_code: Option<i32>,
53        signal: Option<i32>,
54        stdout_bytes: u64,
55        stderr_bytes: u64,
56    },
57}

187/// The complete machine-readable report written by `--verify-json`.
188///
189/// This type is shared by the Hermit producer, the manifest runner, the
190/// compatibility pressure test, and the scorecard. The serialized field order
191/// is retained because existing report bytes are hashed as evidence.
192#[derive(Clone, Debug, Deserialize, PartialEq, Serialize)]
193pub struct VerificationReport {
194    pub verified: bool,
195    pub bitwise_parity: bool,
196    pub verdict: Verdict,
197    #[serde(default, skip_serializing_if = "Option::is_none")]
198    pub no_result_reason: Option<NoResultReason>,
199    #[serde(default)]
200    pub infrastructure_error: Option<InfrastructureError>,
201    /// `None` when the producer reached no verdict. This is a DOCUMENTED
202    /// producer state, not a malformed report: `VerificationReport::no_result()`
203    /// in hermit-cli sets it, and its doc says "`null` when no verdict was
204    /// reached". Declaring it non-optional made a legitimate no-result
205    /// deserialise as "unreadable report", which reported an infrastructure
206    /// fault where the truth was that nothing had been recorded.
207    #[serde(deserialize_with = "present_but_nullable_comparison")]
208    pub comparison: Option<ComparisonReport>,
209    pub compared_log_messages: Option<ComparedLogMessages>,
210    #[serde(default, skip_serializing_if = "Option::is_none")]
211    pub dbt_counted_branches: Option<DbtCountedBranchComparison>,
212    /// Runtime totals for the two compared executions when the producer could
213    /// read both run summaries.
214    #[serde(default, skip_serializing_if = "Option::is_none")]
215    pub runtime: Option<VerificationRuntime>,
216    /// The guest's exit code when an exited status was available to the
217    /// producer. `None` means either that `guest_signal` records a signal
218    /// disposition, or that execution did not return a guest status at all;
219    /// current producers must not leave this null for `ExitStatus::Exited`.
220    #[serde(default)]
221    pub guest_exit_code: Option<i32>,
222    /// The guest's signal when a signaled status was available to the producer.
223    /// When both disposition fields are null, the writer had no guest status.
224    #[serde(default)]
225    pub guest_signal: Option<i32>,
226    /// WHERE the divergence began, in scheduler-turn units. `None` when the
227    /// logs matched, when no comparison ran, or when the report predates the
228    /// field.
229    ///
230    /// `#[serde(default)]` is deliberate here and is the OPPOSITE choice from
231    /// `ComparisonReport::record_envelope` above. That field is an admission
232    /// gate, so a missing value must never be supplied on the producer's
233    /// behalf. This one is a diagnostic position: absent means "not located",
234    /// which is exactly what `None` says. Requiring it would turn every
235    /// retained pre-field report into an unreadable-report ERROR and
236    /// misclassify old evidence as an infrastructure fault -- the same trap
237    /// documented on `comparison` just above.
238    #[serde(default)]
239    pub first_divergent_scheduler_turn: Option<u64>,
240    /// WHERE the divergence began, in virtual-nanosecond units. Same
241    /// tolerant-default rationale as the field above.
242    ///
243    /// NOTE both of these are the position of the PRECEDING scheduler COMMIT,
244    /// so when no COMMIT precedes the differing record they collapse to the
245    /// origin and BOUND the divergence rather than locating it. The field below
246    /// is the one that locates it.
247    #[serde(default)]
248    pub first_divergent_virtual_nanoseconds: Option<u64>,
249    /// 1-based index of the first differing compared record -- the LOCATION of
250    /// the divergence rather than a bound on it, and the only one of the three
251    /// that is a true log prefix. It shares a unit with
252    /// `compared_log_messages`, so `record / compared` is the fraction of the
253    /// log that was deterministic.
254    ///
255    /// `null` and `0` are not the same claim: null is "no divergence located",
256    /// while 0 would mean the very first record differed. Nothing writes 0 --
257    /// the index is 1-based. Same tolerant-default rationale as its siblings.
258    #[serde(default)]
259    pub first_divergent_record: Option<u64>,
260    /// How many syscalls the guest had COMPLETED when the divergence appeared,
261    /// from detcore's own `finish syscall #N` counter.
262    ///
263    /// NOT interchangeable with `first_divergent_record`: one counts guest work
264    /// and the other counts compared log records, and they move at completely
265    /// different rates. Measured on one real divergence: record 98, syscall 37.
266    /// `null` also covers a genuine state -- a run that diverged before any
267    /// syscall completed.
268    #[serde(default)]
269    pub first_divergent_syscall: Option<u64>,
270    /// First differing compared message from the left execution, with the
271    /// separately recorded syscall number, scheduler turn, and committed time
272    /// removed. Older reports omit it and remain readable.
273    #[serde(default)]
274    pub first_divergent_left_message: Option<String>,
275    /// Corresponding first differing compared message from the right execution.
276    #[serde(default)]
277    pub first_divergent_right_message: Option<String>,
278}

280/// Accept `null` but not a MISSING field. serde maps an absent `Option` field to
281/// `None` by default, which would let a report that never mentioned a comparison
282/// read as a legitimate no-result. Naming a `deserialize_with` makes the field
283/// required again while still admitting an explicit null.
284fn present_but_nullable_comparison<'de, D>(
285    deserializer: D,
286) -> Result<Option<ComparisonReport>, D::Error>
287where
288    D: serde::Deserializer<'de>,
289{
290    Option::<ComparisonReport>::deserialize(deserializer)
291}

317    /// `null` is legal only when no comparison completed: either the producer
318    /// reached no verdict, or it named an infrastructure error before it could
319    /// compare. A null comparison beside a product verdict is contradictory
320    /// and stays refused at parse.
321    fn require_consistent_outcome_fields(self) -> Result<Self, String> {
322        if self.comparison.is_none()
323            && !matches!(
324                self.verdict,
325                Verdict::NoResult | Verdict::InfrastructureError
326            )
327        {
328            return Err(format!(
329                "incomplete verification report: comparison is null but verdict is {}; null is legal only for no_result or infrastructure_error",
330                self.verdict
331            ));
332        }
333        match (&self.verdict, &self.infrastructure_error) {
334            (Verdict::InfrastructureError, Some(InfrastructureError::SkidOvershoot { count }))
335                if *count > 0 => {}
336            (Verdict::InfrastructureError, Some(InfrastructureError::SkidOvershoot { .. })) => {
337                return Err(
338                    "incomplete verification report: skid_overshoot count must be positive".into(),
339                );
340            }
341            (Verdict::InfrastructureError, None) => {
342                return Err(
343                    "incomplete verification report: infrastructure_error verdict omitted infrastructure_error"
344                        .into(),
345                );
346            }
347            (_, Some(_)) => {
348                return Err(format!(
349                    "inconsistent verification report: verdict={} carries infrastructure_error",
350                    self.verdict
351                ));
352            }
353            (_, None) => {}
354        }
355        Ok(self)
356    }

358    /// Parse the complete typed receipt. A top-level JSON object is not enough:
359    /// every nested field used by admission must deserialize successfully.
360    #[allow(dead_code)] // this file is path-included by consumers that use one parse form
361    pub fn from_json_slice(bytes: &[u8]) -> Result<Self, String> {
362        serde_json::from_slice::<Self>(bytes)
363            .map_err(|error| format!("incomplete verification report: {error}"))?
364            .require_consistent_outcome_fields()
365    }

367    #[allow(dead_code)] // this file is path-included by consumers that use one parse form
368    pub fn from_json_value(value: serde_json::Value) -> Result<Self, String> {
369        serde_json::from_value::<Self>(value)
370            .map_err(|error| format!("incomplete verification report: {error}"))?
371            .require_consistent_outcome_fields()
372    }

374    /// Parse a report written by the current producer and require every field
375    /// that producer promises to emit. Retained-report readers use
376    /// [`Self::from_json_value`] instead so fields added after an old run remain
377    /// honest absence rather than making the whole report unreadable.
378    #[allow(dead_code)] // path-included readers use different parse forms
379    pub fn from_current_json_value(value: serde_json::Value) -> Result<Self, String> {
380        let object = value
381            .as_object()
382            .ok_or_else(|| "incomplete verification report: expected an object".to_string())?;
383        for field in [
384            "verified",
385            "bitwise_parity",
386            "verdict",
387            "infrastructure_error",
388            "comparison",
389            "compared_log_messages",
390            "guest_exit_code",
391            "guest_signal",
392            "first_divergent_scheduler_turn",
393            "first_divergent_virtual_nanoseconds",
394            "first_divergent_record",
395            "first_divergent_syscall",
396            "first_divergent_left_message",
397            "first_divergent_right_message",
398        ] {
399            if !object.contains_key(field) {
400                return Err(format!(
401                    "incomplete verification report: missing current producer field `{field}`"
402                ));
403            }
404        }
405        if let Some(comparison) = object
406            .get("comparison")
407            .and_then(serde_json::Value::as_object)
408        {
409            for field in [
410                "strictness",
411                "display_name",
412                "compare_logs",
413                "compare_io_buffers",
414                "log_scope",
415                "record_envelope",
416                "virtualize_time",
417                "strip_lines",
418                "canonicalize_addresses",
419                "full_trace",
420                "exact_remainder",
421                "stripped_prefixes",
422                "canonicalizations",
423                "ignore_lines",
424                "skip_commit",
425                "skip_detlog",
426            ] {
427                if !comparison.contains_key(field) {
428                    return Err(format!(
429                        "incomplete verification report: missing current comparison field `{field}`"
430                    ));
431                }
432            }
433        }
434        Self::from_json_value(value)
435    }

437    /// Prove that the invocation actually compared the canonical INFO evidence.
438    /// This is separate from whether that comparison matched: a canonical
439    /// divergence is a product result, while a stripped/output-only/empty-log
440    /// comparison is incomplete evidence and therefore an infrastructure result.
441    pub fn require_canonical_comparison(&self) -> Result<(), String> {
442        let counts = self.compared_log_messages.as_ref();
443        // No comparison at all is a distinct outcome from a comparison that was
444        // too weak, and it is still not admissible. Naming it separately is the
445        // whole point: "no verdict was recorded" is actionable against the
446        // producer, while the old "unreadable report" pointed at the reader.
447        let Some(comparison) = self.comparison.as_ref() else {
448            return Err(format!(
449                "verification recorded no comparison at all (verdict={}), so there is no canonical INFO evidence to admit",
450                self.verdict
451            ));
452        };
453        if comparison.strictness == LogCompareStrictness::Canonical
454            && comparison.compare_logs
455            && comparison.record_envelope.is_canonical()
456            && counts.is_some_and(|counts| counts.left > 0 && counts.right > 0)
457        {
458            Ok(())
459        } else {
460            Err(format!(
461                "verification did not compare canonical non-vacuous INFO evidence: strictness={} compare_logs={} record_envelope={} messages={}/{}",
462                comparison.strictness,
463                comparison.compare_logs,
464                comparison.record_envelope.as_str(),
465                counts.map_or(0, |counts| counts.left),
466                counts.map_or(0, |counts| counts.right),
467            ))
468        }
469    }
470
471    /// Admit a green only when both the canonical comparison and the match
472    /// claim agree. `verified=true` alone is intentionally insufficient: KVM's
473    /// output-only fallback can report it with zero compared INFO messages.
474    pub fn require_canonical_match(&self) -> Result<(), String> {
475        self.require_canonical_comparison()?;
476        if self.verified && self.verdict == Verdict::Matched && self.bitwise_parity {
477            Ok(())
478        } else {
479            Err(format!(
480                "canonical verification did not match: verified={} verdict={} bitwise_parity={}",
481                self.verified, self.verdict, self.bitwise_parity
482            ))
483        }
484    }

528    /// The record index is the only coordinate that LOCATES the divergence;
529    /// the other two are the preceding COMMIT and merely bound it. So a report
530    /// can carry a record with both bounds collapsed to the origin, and that
531    /// is a MORE informative report rather than a malformed one.
532    #[test]
533    fn a_located_record_does_not_require_its_bounding_siblings() {
534        let json = br#"{"verified":false,"bitwise_parity":false,"verdict":"diverged",
535            "comparison":{"strictness":"canonical","compare_logs":true,"record_envelope":"all_records_v1"},
536            "compared_log_messages":{"left":180,"right":180},
537            "first_divergent_record":3}"#;
538        let report = VerificationReport::from_json_slice(json).expect("report parses");
539        assert_eq!(report.first_divergent_record, Some(3));
540        assert_eq!(report.first_divergent_scheduler_turn, None);
541        assert_eq!(report.first_divergent_virtual_nanoseconds, None);
542        assert_eq!(report.first_divergent_left_message, None);
543        assert_eq!(report.first_divergent_right_message, None);
544        assert_eq!(report.first_divergent_syscall, None);
545    }

547    /// An EARLIER divergence must report a SMALLER value in both units. This is
548    /// the ordering the whole metric rests on: a fix that moves the divergence
549    /// later has to be visible as a larger number, and a regression as a
550    /// smaller one. Both figures below were measured with `hermit log-diff
551    /// --json` against one real 131-line INFO log, diverged at two different
552    /// depths -- after COMMIT turn 4 and after COMMIT turn 1.
553    #[test]
554    fn an_earlier_divergence_reports_a_smaller_position() {
555        let parse = |turn: u64, nanos: u64| {
556            let json = format!(
557                r#"{{"verified":false,"bitwise_parity":false,"verdict":"diverged",
558                    "comparison":{{"strictness":"canonical","compare_logs":true,"record_envelope":"all_records_v1"}},"compared_log_messages":{{"left":180,"right":180}},
559                    "first_divergent_scheduler_turn":{turn},
560                    "first_divergent_virtual_nanoseconds":{nanos}}}"#
561            );
562            VerificationReport::from_json_slice(json.as_bytes()).expect("report parses")
563        };
564        let late = parse(4, 1_767_225_600_002_825_515);
565        let early = parse(1, 1_767_225_600_000_500_000);
566        assert!(
567            early.first_divergent_scheduler_turn < late.first_divergent_scheduler_turn,
568            "an earlier divergence must order before a later one in scheduler turns"
569        );
570        assert!(
571            early.first_divergent_virtual_nanoseconds < late.first_divergent_virtual_nanoseconds,
572            "and in virtual nanoseconds"
573        );
