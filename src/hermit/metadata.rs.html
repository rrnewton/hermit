21/// Hermit record version. Recorded as part of hermit-record, hermit-replay
22/// will check this version and will fail if hermit-record version is newer.
23#[derive(
24    Debug,
25    Clone,
26    Copy,
27    Eq,
28    PartialEq,
29    Ord,
30    PartialOrd,
31    Deserialize,
32    Serialize
33)]
34#[repr(transparent)]
35pub struct RecordVersion(u32);
36impl RecordVersion {
37    /// Check if the recorder/replayer version is compatible with a given
38    /// recording (trace).
39    pub fn compatible_with(&self, other: &RecordVersion) -> bool {
40        self == other
41    }
42}
43
44/// hermit record/replay version.
45// NB: Increase the version number when there are breaking changes, i.e.:
46// when new syscalls or event schemas are added.
47pub(crate) const RECORD_VERSION: RecordVersion = RecordVersion(0x10a);

49/// Metadata associated with the recording. This is serialized as a JSON file.
50#[derive(Debug, Serialize, Deserialize)]
51pub struct Metadata {
52    /// The real path to the program.
53    pub exe: PathBuf,
54    /// The name of the program.
55    pub program: String,
56    /// The first argument passed to the program.
57    pub arg0: String,
58    /// Program arguments (not including arg0).
59    pub args: Vec<String>,
60    /// The working directory of the program.
61    pub current_dir: PathBuf,
62    /// The hostname in the UTS namespace used by the program.
63    pub hostname: Option<String>,
64    /// The domainname in the UTS namespace used by the program.
65    pub domainname: Option<String>,
66    /// The environment variables used by the program.
67    pub envs: BTreeMap<String, String>,
68    /// Hermit record/replay version.
69    pub version: RecordVersion,
70}
71
72impl Metadata {
73    /// Creates a new metadata object, populating it with information about a
74    /// command.
75    pub fn new(command: &Command) -> Result<Self, Error> {
76        let exe = command.find_program()?;
77
78        let program = command.get_program().to_string_lossy().into_owned();
79        let arg0 = command.get_arg0().to_string_lossy().into_owned();
80
81        let args = command
82            .get_args()
83            .map(|arg| arg.to_string_lossy().into_owned())
84            .collect();
85
86        let envs = command
87            .get_captured_envs()
88            .into_iter()
89            .map(|(k, v)| {
90                (
91                    k.to_string_lossy().into_owned(),
92                    v.to_string_lossy().into_owned(),
93                )
94            })
95            .collect();
96
97        let current_dir = command
98            .get_current_dir()
99            .map_or_else(|| env::current_dir().unwrap(), ToOwned::to_owned);
100
101        let hostname = command
102            .get_hostname()
103            .map(|s| s.to_string_lossy().into_owned());
104        let domainname = command
105            .get_domainname()
106            .map(|s| s.to_string_lossy().into_owned());
107
108        Ok(Self {
109            exe,
110            program,
111            arg0,
112            args,
113            current_dir,
114            hostname,
115            domainname,
116            envs,
117            version: RECORD_VERSION,
118        })
119    }
120
121    /// Constructs a command from the metadata.
122    pub fn command(&self) -> Command {
123        // NOTE: We bypass the normal $PATH search here by passing in the
124        // absolute path to the program directly.
125        let mut command = Command::new(&self.exe);
126        command.arg0(&self.arg0);
127        command.args(&self.args);
128        command.env_clear();
129        command.envs(&self.envs);
130        command.current_dir(&self.current_dir);
131
132        if let Some(hostname) = &self.hostname {
133            command.hostname(hostname);
134        }
135
136        if let Some(domainname) = &self.domainname {
137            command.domainname(domainname);
138        }
139
140        command
141    }
142}

144// TODO: Record this in the metadata instead of hardcoding this.
145pub fn record_or_replay_config(data: &Path) -> detcore::Config {
146    // NOTE: Record and replay should use the exact same detcore
147    // configuration. Otherwise, the behavior of the program could diverge
148    // during replay.
149    //
150    // WHY THIS IS NOT `hermit run --strict`, WRITTEN HERE ON PURPOSE.
151    //
152    // `virtualize_time: false` below is DELIBERATE, and the rationale used to live only
153    // in the separate dev-hermit workspace -- not in this repository, beside the code it
154    // governs. The cost of that was measured: three agents in one night read this line,
155    // searched this repo's source, git log, docs/ and issues, found nothing, and could
156    // not tell a design decision from a determinism bug; two coordinators then spent
157    // hours treating it as a candidate defect. A decision recorded in a different
158    // repository from the code it governs is, in practice, undocumented. Hence this
159    // comment. See rrnewton/hermit#2295.
160    //
161    // WHAT RECORD/REPLAY ACTUALLY GUARANTEES. Replay re-executes a recording against the
162    // recorded syscall data, so what must be reproducible is THIS recording's replay --
163    // not agreement between two independent recordings. Time is therefore left real: the
164    // recording captures what the guest actually observed, and replay returns those
165    // recorded values. `hermit record start --verify` records once, replays that
166    // recording, and compares the two; its success message says exactly that ("replay
167    // matched recording") and does not claim more.
168    //
169    // WHAT IT DOES NOT GUARANTEE, which is the part that misleads readers. Because time
170    // is not virtualized, two INDEPENDENT recordings of the same program observe
171    // different clock values. Demonstrated: `hermit run -- date` twice returns the
172    // identical virtual epoch, while `hermit record start -- date` twice returns real
173    // wall-clock times seconds apart. Replay fidelity says nothing about that, and
174    // nothing in either test suite currently compares two independent recordings.
175    // So do not read a green `--verify` as cross-recording determinism.
176    //
177    // This configuration differs from `hermit run --strict` on four of the five
178    // properties that define it (see run.rs: only `sequentialize_threads` matches).
179    // Any claim that recording is "strict" in that sense is wrong; the `--strict` flag
180    // on `hermit record` is accepted and ignored purely for command-line compatibility.
181    let default_config: detcore::Config = Default::default();
182    let mut config = detcore::Config {
183        panic_on_unsupported_syscalls: false,
184        exit_on_unsupported_syscall: false,
185        shutdown_on_unsupported_syscall: false,
186        unsupported_syscall_report_fd: None,
187        panic_on_rcb_overshoot: false,
188        sequentialize_threads: true,
189        runs_post_fork: default_config.runs_post_fork,
190        // Record/replay keeps a partial Detcore subscription. Complete coverage
191        // of the Determinized classification begins in v0x10a; madvise policy
192        // semantics begin in v0x102.
193        passthru_opt: true,
194        deterministic_io: false,
195        virtualize_time: false,
196        virtualize_metadata: false,
197        virtualize_cpuid: true,
198        cpuid_virtualized_by_backend: false,
199        backend_supports_madvise: true,
200        discover_live_file_metadata: false,
201        use_thread_local_clock_reads: false,
202        detect_host_clock_futex_timeouts: false,
203        syscall_clobbers_virtualized_by_backend: false,
204        cancel_killed_thread_rpcs: false,
205        backend_reports_physical_process_exits: false,
206        backend_serializes_fork_children: false,
207        backend_dispatches_thread_tools: true,
208        backend_requires_thread_directed_process_signals: false,
209        backend_virtualizes_capability_prctls: false,
210        backend_defers_vfork_child_registration: false,
211        has_uts_namespace: true,
212        // The path to the directory where syscalls will be recorded.
213        replay_data: Some(data.to_path_buf()),
214        clock_multiplier: None,
215        epoch: default_config.epoch,
216        gdbserver: false,
217        gdbserver_port: default_config.gdbserver_port,
218        kill_daemons: default_config.kill_daemons,
219        max_timeslice: default_config.max_timeslice,
220        target_timeslice: default_config.target_timeslice,
221        seed: default_config.seed,
222        rng_seed: default_config.rng_seed,
223        imprecise_timers: false,
224        chaos: false,
225        sigint_instakill: false,
226        warn_non_zero_binds: false,
227        sched_heuristic: Default::default(),
228        sched_seed: default_config.sched_seed,
229        recordreplay_modes: true,
230        record_preemptions: false,
231        record_preemptions_to: None,
232        replay_preemptions_from: None,
233        replay_schedule_from: None,
234        replay_exhausted_panic: false,
235        die_on_desync: true,
236        stacktrace_event: Vec::new(),
237        stacktrace_signal: None,
238        preemption_stacktrace: false,
239        preemption_stacktrace_log_file: None,
240        stop_after_turn: None,
241        stop_after_iter: None,
242        debug_externalize_sockets: false,
243        debug_futex_mode: BlockingMode::Precise,
244        sched_sticky_random_param: 0.0,
245        no_rcb_time: false,
246        detlog_heap: false,
247        detlog_stack: false,
248        detlog_regs: false,
249        detlog_io_buffers: false,
250        detlog_regs_cadence: 1,
251        sysinfo_uptime_offset: 120,
252        memory: 1024 * 1024 * 1024,
253        interrupt_at: vec![],
254        happens_before: None,
255        fuzz_futexes: false,
256        chaos_target_races: false,
257        chaos_per_thread_slowdown: false,
258        chaos_slowdown_max_factor: 10.0,
259        chaos_epoch_length_ns: 0,
260        fuzz_seed: None,
261    };
262    if config.max_timeslice.is_some() && !reverie_ptrace::is_perf_supported() {
263        tracing::warn!(
264            "Hardware perf counters are not supported on this machine. Records/Replays may randomly fail!"
265        );
266        config.max_timeslice = None;
267    }
268    config
269}

271#[cfg(test)]
272mod tests {
273    use reverie::Tool;
274    use reverie::syscalls::Sysno;
275
276    use super::*;
277
278    #[test]
279    fn record_version_requires_an_exact_match() {
280        assert!(RECORD_VERSION.compatible_with(&RECORD_VERSION));
281        assert!(!RECORD_VERSION.compatible_with(&RecordVersion(0x105)));
282        assert!(!RECORD_VERSION.compatible_with(&RecordVersion(0x110)));
283    }
284
285    #[test]
286    fn record_and_replay_preserve_partial_subscriptions() {
287        assert!(record_or_replay_config(Path::new("replay-data")).passthru_opt);
288    }
289
290    #[test]
291    fn record_and_replay_subscribe_every_determinized_syscall() {
292        let config = record_or_replay_config(Path::new("replay-data"));
293        let record = <detcore::Detcore<crate::recorder::Recorder> as Tool>::subscriptions(&config);
294        let replay = <detcore::Detcore<crate::replayer::Replayer> as Tool>::subscriptions(&config);
295
296        for (phase, subscriptions) in [("record", record), ("replay", replay)] {
297            let delivered = subscriptions.iter_syscalls().collect::<Vec<_>>();
298            let missing = detcore::all_pinned_syscalls()
299                .filter(|sysno| detcore::is_determinized_syscall(*sysno))
300                .filter(|sysno| !delivered.contains(sysno))
301                .collect::<Vec<_>>();
302            assert!(
303                missing.is_empty(),
304                "{phase} lets Determinized syscalls bypass Detcore: {}",
305                missing
306                    .iter()
307                    .map(|sysno| sysno.to_string())
308                    .collect::<Vec<_>>()
309                    .join(" ")
310            );
311            assert!(
312                delivered.contains(&Sysno::syslog),
313                "{phase} must deliver syslog to its deterministic Detcore handler"
314            );
315            assert!(
316                !delivered.contains(&Sysno::chdir),
317                "{phase} must leave unlisted PassThrough chdir unsubscribed"
318            );
319        }
320    }
321
322    #[test]
323    fn record_version_rejects_pre_complete_determinized_subscription_streams() {
324        assert!(!RECORD_VERSION.compatible_with(&RecordVersion(0x109)));
325        assert!(!RECORD_VERSION.compatible_with(&RecordVersion(0x104)));
326        assert!(!RECORD_VERSION.compatible_with(&RecordVersion(0x102)));
327    }
