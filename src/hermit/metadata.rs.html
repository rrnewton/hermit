22/// Hermit record version. Recorded as part of hermit-record, hermit-replay
23/// will check this version and will fail if hermit-record version is newer.
24#[derive(
25    Debug,
26    Clone,
27    Copy,
28    Eq,
29    PartialEq,
30    Ord,
31    PartialOrd,

33    Serialize

35#[repr(transparent)]
36pub struct RecordVersion(u32);
37impl RecordVersion {
38    /// Check if the recorder/replayer version is compatible with a given
39    /// recording (trace).
40    pub fn compatible_with(&self, other: &RecordVersion) -> bool {
41        self == other
42    }
43}

45/// hermit record/replay version.
46// NB: Increase the version number when there are breaking changes, i.e.:
47// when new syscalls or event schemas are added.
48//
49// AUTONOMOUS-BOT-IMPLEMENTED
50// TODO-HUMAN-REVIEW(#2373)
51// 0x10b -> 0x10c: flock(2) stopped being a Detcore no-op and now reaches
52// `record_or_replay`, so a post-fix run emits a `Return` event per flock call.
53// Replay re-issues calls for materialized descriptors and fails closed when it
54// cannot reproduce the lock side effect. A 0x10b recording contains NO flock
55// event at all -- the old handler returned Ok(0) before ever reaching the
56// recorder, even though flock was already classified Determinized -- so
57// replaying one under this build would read the next thread event for every
58// flock and desynchronize the stream. The version gate must refuse it.
59//
60// TODO-HUMAN-REVIEW(#2370)
61// 0x10d -> 0x10e: record/replay now carries successful exec image/path state and
62// an explicit ppoll timeout-pointer shape. Older readers cannot interpret those
63// events without desynchronizing the stream, so the format advances once.
64//
65// TODO-HUMAN-REVIEW(#2272)
66// 0x10c -> 0x10d: `Ppoll` becomes its OWN event rather than sharing `Poll`'s,
67// because ppoll additionally copies out a timeout and must preserve it on
68// EINTR and on a partial EFAULT copy-out; and `Poll` itself now records the
69// partial `revents` copy-out an error return leaves behind. The recorded
70// stream therefore carries a different event shape for both syscalls.
71//
72// ONE increment covers BOTH halves deliberately. This branch bumped twice --
73// once per format change -- but a reader either understands the new stream or
74// it does not, so what matters is that the version differs from every stream
75// a different shape was written under. Landing two increments would imply a
76// 0x10d recording exists that this build can read and it cannot: no build
77// ever wrote one.
78//
79// ⚠️ THIS BUMP IS WHY THE REBASE COULD NOT SIMPLY TAKE EITHER SIDE. This
80// change was authored against 0x10a and bumped to 0x10b; main has since gone
81// to 0x10c for the unrelated flock work. Keeping the branch's 0x10b would
82// move the version BACKWARDS and let a 0x10c flock recording be replayed by a
83// build whose ppoll events have a different shape. Keeping main's 0x10c
84// unchanged would be worse: the format would change with no bump at all, so a
85// recording made here would claim 0x10c while containing a `Ppoll` event the
86// 0x10c reader does not know -- exactly the desynchronization the paragraph
87// above exists to prevent. The version must go FORWARD once more.
88//
89// AUTONOMOUS-BOT-IMPLEMENTED
90// TODO-HUMAN-REVIEW(#2407)
91// 0x10e -> 0x10f: pidfd_getfd(2) is now explicitly subscribed and recorded.
92// Earlier builds let the syscall bypass the recorder, so a successful call
93// changed the guest descriptor table without a corresponding stream event.
94// The replayer now consumes one exact pidfd_getfd event and reissues the
95// syscall to reproduce the kernel side effect; accepting a 0x10e stream would
96// therefore consume the next event at every pidfd_getfd call.
97//
98// 0x10f -> 0x110: metadata now carries the recording namespace's proven
99// Hermit-owned mount IDs. Detcore applies that map after both the recorder's
100// raw read and the replayer's ReadV2 copyout. An older recording has no map and
101// would replay different guest-visible mountinfo bytes under this sanitizer,
102// so accepting it would violate replay fidelity even though the event enum did
103// not change.
104//
105// 0x110 -> 0x111: metadata now also carries the recording namespace's full
106// raw mount-ID order. That makes fdinfo's mnt_id use the same canonical identity
107// as mountinfo during replay instead of consulting the unrelated replay
108// namespace or collapsing all descriptors to one mount.
109// 0x111 -> 0x113: fdinfo mount identities are now keyed by each observed raw
110// mount ID rather than broad descriptor classes. Metadata carries only the
111// completed namespace's mountinfo row/parent order; pseudo-filesystem IDs that
112// are absent from mountinfo are allocated afterward in deterministic guest
113// observation order. Replay must rebuild that mapping from recording-time raw
114// bytes, or pidfs/nsfs/anon_inodefs can be collapsed or rejected. The draft
115// 0x112 format was never shipped.
116//
117// 0x113 -> 0x114: recordings persist the completed guest's mountinfo order and
118// unlisted fdinfo mount-ID order, and event streams use stable recording-local
119// thread IDs rather than host PID allocation. An older reader cannot rebuild
120// either mapping reliably for a plain public record API recording.
121//
122// 0x114 -> 0x115: event stream filenames now encode deterministic process-tree
123// pedigrees instead of process-local allocation order. Older readers look for
124// integer filenames and cannot replay a recording containing child streams.
125//
126// 0x115 -> 0x116: event stream filenames are fixed-size SHA-256 names and each
127// data/debug stream begins with its complete process-tree identity. Older
128// readers cannot skip or validate these headers.
129pub(crate) const RECORD_VERSION: RecordVersion = RecordVersion(0x116);
130
131/// The highest RECORD_VERSION this project has ever shipped.
132///
133/// RECORD_VERSION must never move BACKWARD onto it. This is a COMPILE-TIME
134/// assertion rather than a test on purpose: it is decidable from two constants,
135/// so it should break the build rather than a test run, and it cannot be
136/// skipped, filtered or left unrun.
137///
138/// WHY IT EXISTS AT ALL. The rejection set in `record_version_requires_an_exact_match`
139/// used to be a list of specific versions, and `!compatible_with(0x10a)` was
140/// catching a backward move BY ACCIDENT. Replacing that list with a window derived
141/// from RECORD_VERSION fixes its silent narrowing but cannot catch a regression,
142/// because a derived window re-derives from whatever the constant currently says.
143/// Dropping the list without this floor would have traded a stale check for a
144/// weaker one.
145///
146/// The failure it guards is near, not hypothetical: a long-stale branch that bumped
147/// 0x109 -> 0x10a while main advanced to 0x10e regresses the constant the moment its
148/// conflict is resolved by taking the branch side. The build would then stamp the
149/// current schema with a label an older reader already claims, and the exact-match
150/// gate would accept a stream whose shape it does not know -- the desynchronization
151/// the version exists to prevent.
152///
153/// RAISE THIS IN THE SAME COMMIT THAT RAISES RECORD_VERSION.
154const HIGHEST_SHIPPED_RECORD_VERSION: u32 = 0x116;
155
156const _: () = assert!(
157    RECORD_VERSION.0 >= HIGHEST_SHIPPED_RECORD_VERSION,
158    "RECORD_VERSION is BELOW HIGHEST_SHIPPED_RECORD_VERSION. A recording made by this \
159     build would carry a label an older reader already claims. If this fired during a \
160     rebase, the version constant was resolved BACKWARD -- merge it forward."
161);

163/// Metadata associated with the recording. This is serialized as a JSON file.
164#[derive(Debug, Serialize, Deserialize)]
165pub struct Metadata {
166    /// The real path to the program.
167    pub exe: PathBuf,
168    /// The name of the program.
169    pub program: String,
170    /// The first argument passed to the program.
171    pub arg0: String,
172    /// Program arguments (not including arg0).
173    pub args: Vec<String>,
174    /// The working directory of the program.
175    pub current_dir: PathBuf,
176    /// The hostname in the UTS namespace used by the program.
177    pub hostname: Option<String>,
178    /// The domainname in the UTS namespace used by the program.
179    pub domainname: Option<String>,
180    /// The environment variables used by the program.
181    pub envs: BTreeMap<String, String>,
182    /// Hermit record/replay version.
183    pub version: RecordVersion,
184    /// Recording-namespace mount roots proven to be Hermit-owned.
185    ///
186    /// Replay consumes the recorder's raw syscall buffers, so it must apply the
187    /// recording-time raw mount IDs rather than IDs from the fresh replay
188    /// namespace.  Older recordings default to no private-root rewrites.
189    #[serde(default)]
190    pub mountinfo_root_rewrites: Vec<MountInfoRootRewrite>,
191    /// Recording-namespace raw mount IDs in canonical row order.
192    #[serde(default)]
193    pub mountinfo_mount_ids: Vec<u64>,
194    /// Whether `mountinfo_mount_ids` is a producer-owned snapshot, including a
195    /// valid empty snapshot.
196    #[serde(default)]
197    pub mountinfo_mount_ids_captured: bool,
198    /// Recording-time raw fdinfo mount IDs absent from mountinfo, in the order
199    /// Detcore first observed them.
200    #[serde(default)]
201    pub fdinfo_unlisted_mount_ids: Vec<u64>,
202}
203
204impl Metadata {
205    /// Creates a new metadata object, populating it with information about a
206    /// command.
207    pub fn new(command: &Command) -> Result<Self, Error> {
208        let exe = command.find_program()?;
209
210        let program = command.get_program().to_string_lossy().into_owned();
211        let arg0 = command.get_arg0().to_string_lossy().into_owned();
212
213        let args = command
214            .get_args()
215            .map(|arg| arg.to_string_lossy().into_owned())
216            .collect();
217
218        let envs = command
219            .get_captured_envs()
220            .into_iter()
221            .map(|(k, v)| {
222                (
223                    k.to_string_lossy().into_owned(),
224                    v.to_string_lossy().into_owned(),
225                )
226            })
227            .collect();
228
229        let current_dir = command
230            .get_current_dir()
231            .map_or_else(|| env::current_dir().unwrap(), ToOwned::to_owned);
232
233        let hostname = command
234            .get_hostname()
235            .map(|s| s.to_string_lossy().into_owned());
236        let domainname = command
237            .get_domainname()
238            .map(|s| s.to_string_lossy().into_owned());
239
240        Ok(Self {
241            exe,
242            program,
243            arg0,
244            args,
245            current_dir,
246            hostname,
247            domainname,
248            envs,
249            version: RECORD_VERSION,
250            mountinfo_root_rewrites: Vec::new(),
251            mountinfo_mount_ids: Vec::new(),
252            mountinfo_mount_ids_captured: false,
253            fdinfo_unlisted_mount_ids: Vec::new(),
254        })
255    }
256
257    /// Constructs a command from the metadata.
258    pub fn command(&self) -> Command {
259        // NOTE: We bypass the normal $PATH search here by passing in the
260        // absolute path to the program directly.
261        let mut command = Command::new(&self.exe);
262        command.arg0(&self.arg0);
263        command.args(&self.args);
264        command.env_clear();
265        command.envs(&self.envs);
266        command.current_dir(&self.current_dir);
267
268        if let Some(hostname) = &self.hostname {
269            command.hostname(hostname);
270        }
271
272        if let Some(domainname) = &self.domainname {
273            command.domainname(domainname);
274        }
275
276        command
277    }
278}

280pub fn record_or_replay_config(data: &Path) -> detcore::Config {
281    // NOTE: Record and replay should use the exact same Detcore configuration.
282    // Callers add the completed producer namespace's mountinfo order and
283    // unlisted fdinfo mount-ID order after this common base is built, so replay
284    // uses recording-time raw IDs rather than IDs from its fresh container.
285    //
286    // WHY THIS IS NOT `hermit run --strict`, WRITTEN HERE ON PURPOSE.
287    //
288    // `virtualize_time: false` below is DELIBERATE, and the rationale used to live only
289    // in the separate dev-hermit workspace -- not in this repository, beside the code it
290    // governs. The cost of that was measured: three agents in one night read this line,
291    // searched this repo's source, git log, docs/ and issues, found nothing, and could
292    // not tell a design decision from a determinism bug; two coordinators then spent
293    // hours treating it as a candidate defect. A decision recorded in a different
294    // repository from the code it governs is, in practice, undocumented. Hence this
295    // comment. See rrnewton/hermit#2295.
296    //
297    // WHAT RECORD/REPLAY ACTUALLY GUARANTEES. Replay re-executes a recording against the
298    // recorded syscall data, so what must be reproducible is THIS recording's replay --
299    // not agreement between two independent recordings. Time is therefore left real: the
300    // recording captures what the guest actually observed, and replay returns those
301    // recorded values. `hermit record start --verify` records once, replays that
302    // recording, and compares the two; its success message says exactly that ("replay
303    // matched recording") and does not claim more.
304    //
305    // WHAT IT DOES NOT GUARANTEE, which is the part that misleads readers. Because time
306    // is not virtualized, two INDEPENDENT recordings of the same program observe
307    // different clock values. Demonstrated: `hermit run -- date` twice returns the
308    // identical virtual epoch, while `hermit record start -- date` twice returns real
309    // wall-clock times seconds apart. Replay fidelity says nothing about that, and
310    // the focused `independent_mountinfo_recordings_are_canonical` test compares
311    // two independent mountinfo recordings after replacing exactly field 3, the
312    // major:minor device column. With `virtualize_metadata: false`, Linux may
313    // legitimately allocate a different anonymous-block-device minor for the
314    // private procfs mount in each container. That test still compares mount ID,
315    // parent ID, root, mountpoint, options, optional fields, separator,
316    // filesystem type, source, and superblock options byte-for-byte. It does not
317    // establish general cross-recording determinism, and a green `--verify`
318    // remains only a replay-fidelity result.
319    //
320    // This configuration differs from `hermit run --strict` on four of the five
321    // properties that define it (see run.rs: only `sequentialize_threads` matches).
322    // Any claim that recording is "strict" in that sense is wrong; the `--strict` flag
323    // on `hermit record` is accepted and ignored purely for command-line compatibility.
324    let default_config: detcore::Config = Default::default();
325    let mut config = detcore::Config {
326        // Record and replay are determinism claims, so an unsupported syscall
327        // must invalidate the operation instead of being recorded from or
328        // replayed against the live host.
329        panic_on_unsupported_syscalls: true,
330        // Return a typed error through Reverie rather than unwinding across the
331        // backend callback. The tracer owns process-tree cleanup on that error.
332        exit_on_unsupported_syscall: true,
333        shutdown_on_unsupported_syscall: false,
334        unsupported_syscall_report_fd: None,
335        panic_on_rcb_overshoot: false,
336        sequentialize_threads: true,
337        runs_post_fork: default_config.runs_post_fork,
338        // Record/replay keeps a partial Detcore subscription. Complete coverage
339        // of the Determinized classification begins in v0x10a; madvise policy
340        // semantics begin in v0x102.
341        passthru_opt: true,
342        deterministic_io: false,
343        virtualize_time: crate::RECORD_REPLAY_VIRTUALIZES_TIME,
344        virtualize_metadata: false,
345        mountinfo_root_rewrites: Vec::new(),
346        mountinfo_device_rewrites: Vec::new(),
347        mountinfo_mount_ids: Vec::new(),
348        mountinfo_mount_ids_captured: false,
349        fdinfo_unlisted_mount_ids: Vec::new(),
350        virtualize_cpuid: true,
351        cpuid_virtualized_by_backend: false,
352        backend_supports_madvise: true,
353        discover_live_file_metadata: false,
354        use_thread_local_clock_reads: false,
355        detect_host_clock_futex_timeouts: false,
356        syscall_clobbers_virtualized_by_backend: false,
357        cancel_killed_thread_rpcs: false,
358        backend_reports_physical_process_exits: false,
359        backend_serializes_fork_children: false,
360        backend_dispatches_thread_tools: true,
361        backend_tracks_process_children: true,
362        backend_runs_exit_robust_list: true,
363        backend_requires_thread_directed_process_signals: false,
364        backend_supports_parked_write_signal_interruption: true,
365        backend_virtualizes_capability_prctls: false,
366        backend_defers_vfork_child_registration: false,
367        has_uts_namespace: true,
368        // The path to the directory where syscalls will be recorded.
369        replay_data: Some(data.to_path_buf()),
370        clock_multiplier: None,
371        epoch: default_config.epoch,
372        gdbserver: false,
373        gdbserver_port: default_config.gdbserver_port,
374        kill_daemons: default_config.kill_daemons,
375        max_timeslice: default_config.max_timeslice,
376        target_timeslice: default_config.target_timeslice,
377        seed: default_config.seed,
378        rng_seed: default_config.rng_seed,
379        imprecise_timers: false,
380        chaos: false,
381        sigint_instakill: false,
382        warn_non_zero_binds: false,
383        sched_heuristic: Default::default(),
384        sched_seed: default_config.sched_seed,
385        recordreplay_modes: true,
386        record_preemptions: false,
387        record_preemptions_to: None,
388        replay_preemptions_from: None,
389        replay_schedule_from: None,
390        replay_exhausted_panic: false,
391        die_on_desync: true,
392        stacktrace_event: Vec::new(),
393        stacktrace_signal: None,
394        preemption_stacktrace: false,
395        preemption_stacktrace_log_file: None,
396        stop_after_turn: None,
397        stop_after_iter: None,
398        debug_externalize_sockets: false,
399        debug_futex_mode: BlockingMode::Precise,
400        sched_sticky_random_param: 0.0,
401        no_rcb_time: false,
402        detlog_heap: false,
403        detlog_stack: false,
404        detlog_regs: false,
405        detlog_io_buffers: crate::RECORD_REPLAY_HASHES_IO_BUFFERS,
406        detlog_regs_cadence: 1,
407        sysinfo_uptime_offset: 120,
408        memory: default_config.memory,
409        interrupt_at: vec![],
410        happens_before: None,
411        fuzz_futexes: false,
412        chaos_target_races: false,
413        chaos_per_thread_slowdown: false,
414        chaos_slowdown_max_factor: 10.0,
415        chaos_epoch_length_ns: 0,
416        fuzz_seed: None,
417    };
418    if config.max_timeslice.is_some() && !reverie_ptrace::is_perf_supported() {
419        tracing::warn!(
420            "Hardware perf counters are not supported on this machine. Records/Replays may randomly fail!"
421        );
422        config.max_timeslice = None;
423    }
424    config
425}

434    #[test]
435    fn record_version_requires_an_exact_match() {
436        assert!(RECORD_VERSION.compatible_with(&RECORD_VERSION));
437
438        // DERIVED FROM THE COMPATIBILITY RULE, NOT ENUMERATED.
439        //
440        // This assertion used to be a list of specific versions -- 0x10a, 0x10c,
441        // 0x105, 0x110. A list SILENTLY NARROWS every time RECORD_VERSION
442        // advances: the named values drift away from the boundary that matters,
443        // and the test keeps passing while checking less and less. It cannot
444        // fail for the reason its name gives once the value has moved past the
445        // range someone happened to write down.
446        //
447        // `compatible_with` is exact equality, so the property to assert is
448        // "every OTHER version is refused". Re-deriving the cases from
449        // RECORD_VERSION itself means the window travels with the constant and
450        // cannot go stale.
451        let current = RECORD_VERSION.0;
452        for delta in 1..=16u32 {
453            let older = RecordVersion(current - delta);
454            assert!(
455                !RECORD_VERSION.compatible_with(&older),
456                "recorder at {current:#x} must refuse a recording made at {:#x}",
457                older.0
458            );
459            let newer = RecordVersion(current + delta);
460            assert!(
461                !RECORD_VERSION.compatible_with(&newer),
462                "recorder at {current:#x} must refuse a recording made at {:#x}",
463                newer.0
464            );
465        }
466    }
