25/// Hermit record version. Recorded as part of hermit-record, hermit-replay
26/// will check this version and will fail if hermit-record version is newer.
27#[derive(
28    Debug,
29    Clone,
30    Copy,
31    Eq,
32    PartialEq,
33    Ord,
34    PartialOrd,
35    Deserialize,
36    Serialize

38#[repr(transparent)]
39pub struct RecordVersion(u32);
40impl RecordVersion {
41    /// Check if the recorder/replayer version is compatible with a given
42    /// recording (trace).
43    pub fn compatible_with(&self, other: &RecordVersion) -> bool {
44        self == other
45    }

48/// hermit record/replay version.
49// NB: Increase the version number when there are breaking changes, i.e.:
50// when new syscalls or event schemas are added.
51//
52// AUTONOMOUS-BOT-IMPLEMENTED
53// TODO-HUMAN-REVIEW(#2373)
54// 0x10b -> 0x10c: flock(2) stopped being a Detcore no-op and now reaches
55// `record_or_replay`, so a post-fix run emits a `Return` event per flock call.
56// Replay re-issues calls for materialized descriptors and fails closed when it
57// cannot reproduce the lock side effect. A 0x10b recording contains NO flock
58// event at all -- the old handler returned Ok(0) before ever reaching the
59// recorder, even though flock was already classified Determinized -- so
60// replaying one under this build would read the next thread event for every
61// flock and desynchronize the stream. The version gate must refuse it.
62//
63// TODO-HUMAN-REVIEW(#2370)
64// 0x10d -> 0x10e: record/replay now carries successful exec image/path state and
65// an explicit ppoll timeout-pointer shape. Older readers cannot interpret those
66// events without desynchronizing the stream, so the format advances once.
67//
68// TODO-HUMAN-REVIEW(#2272)
69// 0x10c -> 0x10d: `Ppoll` becomes its OWN event rather than sharing `Poll`'s,
70// because ppoll additionally copies out a timeout and must preserve it on
71// EINTR and on a partial EFAULT copy-out; and `Poll` itself now records the
72// partial `revents` copy-out an error return leaves behind. The recorded
73// stream therefore carries a different event shape for both syscalls.
74//
75// ONE increment covers BOTH halves deliberately. This branch bumped twice --
76// once per format change -- but a reader either understands the new stream or
77// it does not, so what matters is that the version differs from every stream
78// a different shape was written under. Landing two increments would imply a
79// 0x10d recording exists that this build can read and it cannot: no build
80// ever wrote one.
81//
82// ⚠️ THIS BUMP IS WHY THE REBASE COULD NOT SIMPLY TAKE EITHER SIDE. This
83// change was authored against 0x10a and bumped to 0x10b; main has since gone
84// to 0x10c for the unrelated flock work. Keeping the branch's 0x10b would
85// move the version BACKWARDS and let a 0x10c flock recording be replayed by a
86// build whose ppoll events have a different shape. Keeping main's 0x10c
87// unchanged would be worse: the format would change with no bump at all, so a
88// recording made here would claim 0x10c while containing a `Ppoll` event the
89// 0x10c reader does not know -- exactly the desynchronization the paragraph
90// above exists to prevent. The version must go FORWARD once more.
91//
92// AUTONOMOUS-BOT-IMPLEMENTED
93// TODO-HUMAN-REVIEW(#2407)
94// 0x10e -> 0x10f: pidfd_getfd(2) is now explicitly subscribed and recorded.
95// Earlier builds let the syscall bypass the recorder, so a successful call
96// changed the guest descriptor table without a corresponding stream event.
97// The replayer now consumes one exact pidfd_getfd event and reissues the
98// syscall to reproduce the kernel side effect; accepting a 0x10e stream would
99// therefore consume the next event at every pidfd_getfd call.
100//
101// 0x10f -> 0x110: metadata now carries the recording namespace's proven
102// Hermit-owned mount IDs. Detcore applies that map after both the recorder's
103// raw read and the replayer's ReadV2 copyout. An older recording has no map and
104// would replay different guest-visible mountinfo bytes under this sanitizer,
105// so accepting it would violate replay fidelity even though the event enum did
106// not change.
107//
108// 0x110 -> 0x111: metadata now also carries the recording namespace's full
109// raw mount-ID order. That makes fdinfo's mnt_id use the same canonical identity
110// as mountinfo during replay instead of consulting the unrelated replay
111// namespace or collapsing all descriptors to one mount.
112// 0x111 -> 0x113: fdinfo mount identities are now keyed by each observed raw
113// mount ID rather than broad descriptor classes. Metadata carries only the
114// completed namespace's mountinfo row/parent order; pseudo-filesystem IDs that
115// are absent from mountinfo are allocated afterward in deterministic guest
116// observation order. Replay must rebuild that mapping from recording-time raw
117// bytes, or pidfs/nsfs/anon_inodefs can be collapsed or rejected. The draft
118// 0x112 format was never shipped.
119//
120// 0x113 -> 0x114: recordings persist the completed guest's mountinfo order and
121// unlisted fdinfo mount-ID order, and event streams use stable recording-local
122// thread IDs rather than host PID allocation. An older reader cannot rebuild
123// either mapping reliably for a plain public record API recording.
124//
125// 0x114 -> 0x115: event stream filenames now encode deterministic process-tree
126// pedigrees instead of process-local allocation order. Older readers look for
127// integer filenames and cannot replay a recording containing child streams.
128//
129// 0x115 -> 0x116: event stream filenames are fixed-size SHA-256 names and each
130// data/debug stream begins with its complete process-tree identity. Older
131// readers cannot skip or validate these headers.
132// 0x116 -> 0x119: nonpositioned random-device readv now uses Detcore's
133// shared canonical cursor and no longer emits or consumes recorder ReadvV2
134// events. Old event streams cannot be replayed under this routing.
135// 0x117 and 0x118 identify incompatible retained RNG, clock, and network
136// recordings from parallel development; do not reinterpret those identities.
137// 0x119 -> 0x11a: random-device preadv/preadv2 now use Detcore's canonical
138// generator, including the shared cursor for preadv2(-1). They no longer emit
139// or consume ReadvV2 events; accepting an older recording would desynchronize
140// the event stream and replace its recorded random bytes.
141// Early rejection of invalid negative offsets also stops emitting ReadvV2
142// error events for non-random descriptors.
143// 0x11a -> 0x11b: sysinfo uptime rounds exact elapsed logical time up, removing
144// dependence on the calendar epoch's fraction. Detcore recomputes sysinfo during
145// recording/replay, even with virtualize_time=false; replaying an older stream
146// under the new projection could change guest control flow and event consumption.
147// Sysinfo remains determinized rather than gaining a recorded event.
148// 0x11b -> 0x11c: /proc/uptime renders the floor of exact elapsed logical time
149// instead of the difference of the separately truncated clock and epoch, so a
150// fractional epoch no longer moves it (or /proc/stat's idle ticks derived from
151// it) by one second. /proc/stat btime is now derived from the fixed boot
152// instant rather than from the clock minus uptime. Detcore recomputes these
153// fields over the recorded procfs bytes during recording and replay, so an older
154// stream replayed under the new projection could show the guest different bytes.
155// 0x11c -> 0x11d: with time not virtualized, which is always the case for
156// record/replay, detcore passes clock reads to record/replay instead of refusing
157// them. Streams now carry Timespec and Timeofday events for clock_gettime,
158// gettimeofday and time, and clock_getres gains a recorder event (Timespec, or
159// Return for a NULL `res`). An older reader would desynchronize on these events.
160// 0x11d -> 0x11e: guest-semantic madvise advice, which detcore used to refuse
161// with ENOSYS, now reaches the recorder and replayer and emits a Madvise event.
162// 0x11e -> 0x11f: faccessat, faccessat2, chdir, getcwd and the legacy path
163// mutations (rename, link, symlink, chmod, chown, lchown, mknod, rmdir) are
164// recorded instead of running live. A successful chdir or fchdir carries a
165// Bytes event naming the directory it entered. An older reader would
166// desynchronize on these events.
167// 0x11f -> 0x120: descriptor syncs and metadata changes (fsync, fdatasync,
168// syncfs, fchmod, fchown, fsetxattr, fremovexattr) record a Return event, the
169// xattr queries (getxattr, lgetxattr, fgetxattr and the listxattr family)
170// record a Bytes event (Return for a zero-size query), and zero-count read and
171// pread64 calls are recorded instead of running live. An older reader would
172// desynchronize on these events.
173pub(crate) const RECORD_VERSION: RecordVersion = RecordVersion(0x120);
174
175/// The highest RECORD_VERSION this project has ever shipped.
176///
177/// RECORD_VERSION must never move BACKWARD onto it. This is a COMPILE-TIME
178/// assertion rather than a test on purpose: it is decidable from two constants,
179/// so it should break the build rather than a test run, and it cannot be
180/// skipped, filtered or left unrun.
181///
182/// WHY IT EXISTS AT ALL. The rejection set in `record_version_requires_an_exact_match`
183/// used to be a list of specific versions, and `!compatible_with(0x10a)` was
184/// catching a backward move BY ACCIDENT. Replacing that list with a window derived
185/// from RECORD_VERSION fixes its silent narrowing but cannot catch a regression,
186/// because a derived window re-derives from whatever the constant currently says.
187/// Dropping the list without this floor would have traded a stale check for a
188/// weaker one.
189///
190/// The failure it guards is near, not hypothetical: a long-stale branch that bumped
191/// 0x109 -> 0x10a while main advanced to 0x10e regresses the constant the moment its
192/// conflict is resolved by taking the branch side. The build would then stamp the
193/// current schema with a label an older reader already claims, and the exact-match
194/// gate would accept a stream whose shape it does not know -- the desynchronization
195/// the version exists to prevent.
196///
197/// RAISE THIS IN THE SAME COMMIT THAT RAISES RECORD_VERSION.
198const HIGHEST_SHIPPED_RECORD_VERSION: u32 = 0x120;
199
200const _: () = assert!(
201    RECORD_VERSION.0 >= HIGHEST_SHIPPED_RECORD_VERSION,
202    "RECORD_VERSION is BELOW HIGHEST_SHIPPED_RECORD_VERSION. A recording made by this \
203     build would carry a label an older reader already claims. If this fired during a \
204     rebase, the version constant was resolved BACKWARD -- merge it forward."
205);

207/// Metadata associated with the recording. This is serialized as a JSON file.
208#[derive(Debug, Serialize, Deserialize)]
209pub struct Metadata {
210    /// The real path to the program.
211    pub exe: PathBuf,
212    /// The name of the program.
213    pub program: String,
214    /// The first argument passed to the program.
215    pub arg0: String,
216    /// Program arguments (not including arg0).
217    pub args: Vec<String>,
218    /// The working directory of the program.
219    pub current_dir: PathBuf,
220    /// The hostname in the UTS namespace used by the program.
221    pub hostname: Option<String>,
222    /// The domainname in the UTS namespace used by the program.
223    pub domainname: Option<String>,
224    /// The environment variables used by the program.
225    pub envs: BTreeMap<String, String>,
226    /// Hermit record/replay version.
227    pub version: RecordVersion,
228    /// Actual guest proc mount mode after setup. Missing in older recordings or
229    /// when the guest's proc mount could not be inspected; diagnostic only.
230    #[serde(default)]
231    pub proc_readonly: Option<bool>,
232    /// Recording-namespace mount roots proven to be Hermit-owned.
233    ///
234    /// Replay consumes the recorder's raw syscall buffers, so it must apply the
235    /// recording-time raw mount IDs rather than IDs from the fresh replay
236    /// namespace.  Older recordings default to no private-root rewrites.
237    #[serde(default)]
238    pub mountinfo_root_rewrites: Vec<MountInfoRootRewrite>,
239    /// Recording-namespace raw mount IDs in canonical row order.
240    #[serde(default)]
241    pub mountinfo_mount_ids: Vec<u64>,
242    /// Whether `mountinfo_mount_ids` is a producer-owned snapshot, including a
243    /// valid empty snapshot.
244    #[serde(default)]
245    pub mountinfo_mount_ids_captured: bool,
246    /// Recording-time raw fdinfo mount IDs absent from mountinfo, in the order
247    /// Detcore first observed them.
248    #[serde(default)]
249    pub fdinfo_unlisted_mount_ids: Vec<u64>,
250    /// Whether the recording ran in an isolated network namespace with only
251    /// loopback (`record start --network local`). `None` means the recorder
252    /// did not say, which is every recording made before this field existed;
253    /// those recordings ran on the host network. An autopilot replay uses the
254    /// same network, so syscalls the replayer re-executes see the same
255    /// interfaces and ports they were recorded against.
256    #[serde(default)]
257    pub local_networking: Option<bool>,
258}

260impl Metadata {
261    /// Creates a new metadata object, populating it with information about a
262    /// command.
263    pub fn new(command: &Command) -> Result<Self, Error> {
264        let exe = command.find_program()?;
265
266        let program = command.get_program().to_string_lossy().into_owned();
267        let arg0 = command.get_arg0().to_string_lossy().into_owned();
268
269        let args = command
270            .get_args()
271            .map(|arg| arg.to_string_lossy().into_owned())
272            .collect();
273
274        let envs = command
275            .get_captured_envs()
276            .into_iter()
277            .map(|(k, v)| {
278                (
279                    k.to_string_lossy().into_owned(),
280                    v.to_string_lossy().into_owned(),
281                )
282            })
283            .collect();
284
285        let current_dir = command
286            .get_current_dir()
287            .map_or_else(|| env::current_dir().unwrap(), ToOwned::to_owned);
288
289        let hostname = command
290            .get_hostname()
291            .map(|s| s.to_string_lossy().into_owned());
292        let domainname = command
293            .get_domainname()
294            .map(|s| s.to_string_lossy().into_owned());
295
296        Ok(Self {
297            exe,
298            program,
299            arg0,
300            args,
301            current_dir,
302            hostname,
303            domainname,
304            envs,
305            version: RECORD_VERSION,
306            proc_readonly: None,
307            mountinfo_root_rewrites: Vec::new(),
308            mountinfo_mount_ids: Vec::new(),
309            mountinfo_mount_ids_captured: false,
310            fdinfo_unlisted_mount_ids: Vec::new(),
311            local_networking: None,
312        })
313    }
314
315    /// Reads the metadata of the recording stored in `dir`.
316    pub fn load(dir: &Path) -> Result<Self, Error> {
317        let metadata_path = dir.join(METADATA_NAME);
318        serde_json::from_reader(
319            fs::File::open(&metadata_path)
320                .with_context(|| format!("Failed to open {:?}", metadata_path))?,
321        )
322        .with_context(|| format!("Failed to parse {:?}", metadata_path))
323    }
324
325    /// Constructs a command from the metadata.
326    pub fn command(&self) -> Command {
327        // NOTE: We bypass the normal $PATH search here by passing in the
328        // absolute path to the program directly.
329        let mut command = Command::new(&self.exe);
330        command.arg0(&self.arg0);
331        command.args(&self.args);
332        command.env_clear();
333        command.envs(&self.envs);
334        command.current_dir(&self.current_dir);
335
336        if let Some(hostname) = &self.hostname {
337            command.hostname(hostname);
338        }
339
340        if let Some(domainname) = &self.domainname {
341            command.domainname(domainname);
342        }
343
344        command
345    }
346}

348pub fn record_or_replay_config(data: &Path) -> detcore::Config {
349    // NOTE: Record and replay should use the exact same Detcore configuration.
350    // Callers add the completed producer namespace's mountinfo order and
351    // unlisted fdinfo mount-ID order after this common base is built, so replay
352    // uses recording-time raw IDs rather than IDs from its fresh container.
353    //
354    // WHY THIS IS NOT `hermit run --strict`, WRITTEN HERE ON PURPOSE.
355    //
356    // `virtualize_time: false` below is DELIBERATE, and the rationale used to live only
357    // in the separate dev-hermit workspace -- not in this repository, beside the code it
358    // governs. The cost of that was measured: three agents in one night read this line,
359    // searched this repo's source, git log, docs/ and issues, found nothing, and could
360    // not tell a design decision from a determinism bug; two coordinators then spent
361    // hours treating it as a candidate defect. A decision recorded in a different
362    // repository from the code it governs is, in practice, undocumented. Hence this
363    // comment. See rrnewton/hermit#2295.
364    //
365    // WHAT RECORD/REPLAY ACTUALLY GUARANTEES. Replay re-executes a recording against the
366    // recorded syscall data, so what must be reproducible is THIS recording's replay --
367    // not agreement between two independent recordings. Time is therefore left real: the
368    // recording captures what the guest actually observed, and replay returns those
369    // recorded values. `hermit record start --verify` records once, replays that
370    // recording, and compares the two; its success message says exactly that ("replay
371    // matched recording") and does not claim more.
372    //
373    // WHAT IT DOES NOT GUARANTEE, which is the part that misleads readers. Because time
374    // is not virtualized, two INDEPENDENT recordings of the same program observe
375    // different clock values. Demonstrated: `hermit run -- date` twice returns the
376    // identical virtual epoch, while `hermit record start -- date` twice returns real
377    // wall-clock times seconds apart. Replay fidelity says nothing about that, and
378    // the focused `independent_mountinfo_recordings_are_canonical` test compares
379    // two independent mountinfo recordings after replacing exactly field 3, the
380    // major:minor device column. With `virtualize_metadata: false`, Linux may
381    // legitimately allocate a different anonymous-block-device minor for the
382    // private procfs mount in each container. That test still compares mount ID,
383    // parent ID, root, mountpoint, options, optional fields, separator,
384    // filesystem type, source, and superblock options byte-for-byte. It does not
385    // establish general cross-recording determinism, and a green `--verify`
386    // remains only a replay-fidelity result.
387    //
388    // This configuration differs from `hermit run --strict` on four of the five
389    // properties that define it (see run.rs: only `sequentialize_threads` matches).
390    // Any claim that recording is "strict" in that sense is wrong; the `--strict` flag
391    // on `hermit record` is accepted and ignored purely for command-line compatibility.
392    let default_config: detcore::Config = Default::default();
393    let mut config = detcore::Config {
394        // Record and replay are determinism claims, so an unsupported syscall
395        // must invalidate the operation instead of being recorded from or
396        // replayed against the live host.
397        panic_on_unsupported_syscalls: true,
398        // Return a typed error through Reverie rather than unwinding across the
399        // backend callback. The tracer owns process-tree cleanup on that error.
400        exit_on_unsupported_syscall: true,
401        shutdown_on_unsupported_syscall: false,
402        unsupported_syscall_report_fd: None,
403        panic_on_rcb_overshoot: false,
404        sequentialize_threads: true,
405        runs_post_fork: default_config.runs_post_fork,
406        // Record/replay keeps a partial Detcore subscription. Complete coverage
407        // of the Determinized classification begins in v0x10a; madvise policy
408        // semantics begin in v0x102.
409        passthru_opt: true,
410        deterministic_io: false,
411        virtualize_time: crate::RECORD_REPLAY_VIRTUALIZES_TIME,
412        virtualize_metadata: false,
413        mountinfo_root_rewrites: Vec::new(),
414        mountinfo_device_rewrites: Vec::new(),
415        mountinfo_mount_ids: Vec::new(),
416        mountinfo_mount_ids_captured: false,
417        fdinfo_unlisted_mount_ids: Vec::new(),
418        virtualize_cpuid: true,
419        cpuid_virtualized_by_backend: false,
420        backend_supports_madvise: true,
421        discover_live_file_metadata: false,
422        use_thread_local_clock_reads: false,
423        detect_host_clock_futex_timeouts: false,
424        syscall_clobbers_virtualized_by_backend: false,
425        cancel_killed_thread_rpcs: false,
426        backend_reports_physical_process_exits: false,
427        backend_serializes_fork_children: false,
428        backend_dispatches_thread_tools: true,
429        backend_tracks_process_children: true,
430        backend_runs_exit_robust_list: true,
431        backend_requires_thread_directed_process_signals: false,
432        backend_is_kvm: false,
433        kvm_shared_dequeue_timers: false,
434        backend_supports_parked_write_signal_interruption: true,
435        backend_virtualizes_capability_prctls: false,
436        backend_defers_vfork_child_registration: false,
437        has_uts_namespace: true,
438        // The path to the directory where syscalls will be recorded.
439        replay_data: Some(data.to_path_buf()),
440        clock_multiplier: None,
441        epoch: default_config.epoch,
442        gdbserver: false,
443        gdbserver_port: default_config.gdbserver_port,
444        kill_daemons: default_config.kill_daemons,
445        max_timeslice: default_config.max_timeslice,
446        target_timeslice: default_config.target_timeslice,
447        seed: default_config.seed,
448        rng_seed: default_config.rng_seed,
449        imprecise_timers: false,
450        chaos: false,
451        sigint_instakill: false,
452        warn_non_zero_binds: false,
453        sched_heuristic: Default::default(),
454        sched_seed: default_config.sched_seed,
455        network_trace: Default::default(),
456        recordreplay_modes: true,
457        record_preemptions: false,
458        record_preemptions_to: None,
459        replay_preemptions_from: None,
460        replay_schedule_from: None,
461        replay_exhausted_panic: false,
462        die_on_desync: true,
463        stacktrace_event: Vec::new(),
464        stacktrace_signal: None,
465        preemption_stacktrace: false,
466        preemption_stacktrace_log_file: None,
467        stop_after_turn: None,
468        stop_after_iter: None,
469        debug_externalize_sockets: false,
470        debug_futex_mode: BlockingMode::Precise,
471        sched_sticky_random_param: 0.0,
472        no_rcb_time: false,
473        detlog_heap: false,
474        detlog_stack: false,
475        detlog_regs: false,
476        detlog_io_buffers: crate::RECORD_REPLAY_HASHES_IO_BUFFERS,
477        detlog_regs_cadence: 1,
478        sysinfo_uptime_offset: 120,
479        memory: default_config.memory,
480        interrupt_at: vec![],
481        happens_before: None,
482        fuzz_futexes: false,
483        chaos_target_races: false,
484        chaos_per_thread_slowdown: false,
485        chaos_slowdown_max_factor: 10.0,
486        chaos_epoch_length_ns: 0,
487        fuzz_seed: None,
488    };
489    if config.max_timeslice.is_some() && !reverie_ptrace::is_perf_supported() {
490        tracing::warn!(
491            "Hardware perf counters are not supported on this machine. Records/Replays may randomly fail!"
492        );
493        config.max_timeslice = None;
494    }
495    config
496}

505    #[test]
506    fn record_version_requires_an_exact_match() {
507        assert!(RECORD_VERSION.compatible_with(&RECORD_VERSION));
508
509        // DERIVED FROM THE COMPATIBILITY RULE, NOT ENUMERATED.
510        //
511        // This assertion used to be a list of specific versions -- 0x10a, 0x10c,
512        // 0x105, 0x110. A list SILENTLY NARROWS every time RECORD_VERSION
513        // advances: the named values drift away from the boundary that matters,
514        // and the test keeps passing while checking less and less. It cannot
515        // fail for the reason its name gives once the value has moved past the
516        // range someone happened to write down.
517        //
518        // `compatible_with` is exact equality, so the property to assert is
519        // "every OTHER version is refused". Re-deriving the cases from
520        // RECORD_VERSION itself means the window travels with the constant and
521        // cannot go stale.
522        let current = RECORD_VERSION.0;
523        for delta in 1..=16u32 {
524            let older = RecordVersion(current - delta);
525            assert!(
526                !RECORD_VERSION.compatible_with(&older),
527                "recorder at {current:#x} must refuse a recording made at {:#x}",
528                older.0
529            );
530            let newer = RecordVersion(current + delta);
531            assert!(
532                !RECORD_VERSION.compatible_with(&newer),
533                "recorder at {current:#x} must refuse a recording made at {:#x}",
534                newer.0
535            );
536        }
537    }

558    /// RECORDING DOES NOT VIRTUALIZE TIME, AND THE VERDICT NOW SAYS SO.
559    ///
560    /// `virtualize_time: false` here is deliberate (see the rationale block on
561    /// `record_or_replay_config`), and it is what makes a green
562    /// `record start --verify` mean something weaker than a green
563    /// `run --verify`: the replay reproduced *that recording*, not that the guest
564    /// is deterministic across invocations. Ported from the residual of
565    /// hermit#2269.
566    ///
567    /// Pinned against the shared constant rather than a literal, because the
568    /// value is now read in two places — the config the run uses, and the
569    /// `ComparisonOptions` that discloses it in the report. If those drifted, the
570    /// report would describe a time policy the run did not use, which is the very
571    /// defect the disclosure exists to close.
572    #[test]
573    fn recording_does_not_virtualize_time_as_documented() {
574        let config = record_or_replay_config(Path::new("replay-data"));
575        assert!(
576            !config.virtualize_time,
577            "record/replay must not virtualize time; a green replay verdict would \
578             otherwise be mistaken for a determinism result"
579        );
580        assert_eq!(
581            config.virtualize_time,
582            crate::RECORD_REPLAY_VIRTUALIZES_TIME,
583            "the config the run uses and the constant the report discloses must be \
584             the same decision, not two that happen to agree"
585        );
586    }

588    #[test]
589    fn record_and_replay_subscribe_every_determinized_syscall() {
590        let config = record_or_replay_config(Path::new("replay-data"));
591        let record = <detcore::Detcore<crate::recorder::Recorder> as Tool>::subscriptions(&config);
592        let replay = <detcore::Detcore<crate::replayer::Replayer> as Tool>::subscriptions(&config);
593
594        for (phase, subscriptions) in [("record", record), ("replay", replay)] {
595            let delivered = subscriptions.iter_syscalls().collect::<Vec<_>>();
596            let missing = detcore::all_pinned_syscalls()
597                .filter(|sysno| detcore::is_determinized_syscall(*sysno))
598                .filter(|sysno| !delivered.contains(sysno))
599                .collect::<Vec<_>>();
600            assert!(
601                missing.is_empty(),
602                "{phase} lets Determinized syscalls bypass Detcore: {}",
603                missing
604                    .iter()
605                    .map(|sysno| sysno.to_string())
606                    .collect::<Vec<_>>()
607                    .join(" ")
608            );
609            assert!(
610                delivered.contains(&Sysno::syslog),
611                "{phase} must deliver syslog to its deterministic Detcore handler"
612            );
613            // chdir is PassThrough for Detcore but its result depends on the
614            // replay chroot, so record/replay subscribes it (hermit#3598);
615            // umask, from the same PassThrough group, stays the sentinel that
616            // record/replay does not subscribe everything.
617            assert!(
618                delivered.contains(&Sysno::chdir) && delivered.contains(&Sysno::getcwd),
619                "{phase} must deliver chdir and getcwd so replay uses recorded results"
620            );
621            // A replayed descriptor may be a placeholder, so these must
622            // return recorded results rather than run
623            // (https://github.com/rrnewton/hermit/issues/3591).
624            for sysno in [
625                Sysno::fsync,
626                Sysno::fdatasync,
627                Sysno::syncfs,
628                Sysno::fchmod,
629                Sysno::fchown,
630                Sysno::fsetxattr,
631                Sysno::fremovexattr,
632                Sysno::setxattr,
633                Sysno::lsetxattr,
634                Sysno::removexattr,
635                Sysno::lremovexattr,
636                Sysno::getxattr,
637                Sysno::lgetxattr,
638                Sysno::fgetxattr,
639                Sysno::listxattr,
640                Sysno::llistxattr,
641                Sysno::flistxattr,
642            ] {
643                assert!(
644                    delivered.contains(&sysno),
645                    "{phase} must deliver {sysno} so replay uses its recorded result"
646                );
647            }
648            assert!(
649                !delivered.contains(&Sysno::umask),
650                "{phase} must leave unlisted PassThrough umask unsubscribed"
651            );
652        }
653    }
