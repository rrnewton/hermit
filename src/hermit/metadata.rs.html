21/// Hermit record version. Recorded as part of hermit-record, hermit-replay
22/// will check this version and will fail if hermit-record version is newer.
23#[derive(
24    Debug,
25    Clone,
26    Copy,
27    Eq,
28    PartialEq,
29    Ord,
30    PartialOrd,
31    Deserialize,
32    Serialize
33)]
34#[repr(transparent)]
35pub struct RecordVersion(u32);
36impl RecordVersion {
37    /// Check if the recorder/replayer version is compatible with a given
38    /// recording (trace).
39    pub fn compatible_with(&self, other: &RecordVersion) -> bool {
40        self == other
41    }
42}
43
44/// hermit record/replay version.
45// NB: Increase the version number when there are breaking changes, i.e.:
46// when new syscalls or event schemas are added.
47pub(crate) const RECORD_VERSION: RecordVersion = RecordVersion(0x109);

49/// Metadata associated with the recording. This is serialized as a JSON file.
50#[derive(Debug, Serialize, Deserialize)]
51pub struct Metadata {
52    /// The real path to the program.
53    pub exe: PathBuf,
54    /// The name of the program.
55    pub program: String,
56    /// The first argument passed to the program.
57    pub arg0: String,
58    /// Program arguments (not including arg0).
59    pub args: Vec<String>,
60    /// The working directory of the program.
61    pub current_dir: PathBuf,
62    /// The hostname in the UTS namespace used by the program.
63    pub hostname: Option<String>,
64    /// The domainname in the UTS namespace used by the program.
65    pub domainname: Option<String>,
66    /// The environment variables used by the program.
67    pub envs: BTreeMap<String, String>,
68    /// Hermit record/replay version.
69    pub version: RecordVersion,
70}
71
72impl Metadata {
73    /// Creates a new metadata object, populating it with information about a
74    /// command.
75    pub fn new(command: &Command) -> Result<Self, Error> {
76        let exe = command.find_program()?;
77
78        let program = command.get_program().to_string_lossy().into_owned();
79        let arg0 = command.get_arg0().to_string_lossy().into_owned();
80
81        let args = command
82            .get_args()
83            .map(|arg| arg.to_string_lossy().into_owned())
84            .collect();
85
86        let envs = command
87            .get_captured_envs()
88            .into_iter()
89            .map(|(k, v)| {
90                (
91                    k.to_string_lossy().into_owned(),
92                    v.to_string_lossy().into_owned(),
93                )
94            })
95            .collect();
96
97        let current_dir = command
98            .get_current_dir()
99            .map_or_else(|| env::current_dir().unwrap(), ToOwned::to_owned);
100
101        let hostname = command
102            .get_hostname()
103            .map(|s| s.to_string_lossy().into_owned());
104        let domainname = command
105            .get_domainname()
106            .map(|s| s.to_string_lossy().into_owned());
107
108        Ok(Self {
109            exe,
110            program,
111            arg0,
112            args,
113            current_dir,
114            hostname,
115            domainname,
116            envs,
117            version: RECORD_VERSION,
118        })
119    }
120
121    /// Constructs a command from the metadata.
122    pub fn command(&self) -> Command {
123        // NOTE: We bypass the normal $PATH search here by passing in the
124        // absolute path to the program directly.
125        let mut command = Command::new(&self.exe);
126        command.arg0(&self.arg0);
127        command.args(&self.args);
128        command.env_clear();
129        command.envs(&self.envs);
130        command.current_dir(&self.current_dir);
131
132        if let Some(hostname) = &self.hostname {
133            command.hostname(hostname);
134        }
135
136        if let Some(domainname) = &self.domainname {
137            command.domainname(domainname);
138        }
139
140        command
141    }
142}

145pub fn record_or_replay_config(data: &Path) -> detcore::Config {
146    // NOTE: Record and replay should use the exact same detcore
147    // configuration. Otherwise, the behavior of the program could diverge
148    // during replay.
149    let default_config: detcore::Config = Default::default();
150    let mut config = detcore::Config {
151        panic_on_unsupported_syscalls: false,
152        exit_on_unsupported_syscall: false,
153        shutdown_on_unsupported_syscall: false,
154        unsupported_syscall_report_fd: None,
155        panic_on_rcb_overshoot: false,
156        sequentialize_threads: true,
157        runs_post_fork: default_config.runs_post_fork,
158        // Record/replay keeps partial Detcore subscription; madvise policy semantics
159        // begin in v0x102.
160        passthru_opt: true,
161        deterministic_io: false,
162        virtualize_time: false,
163        virtualize_metadata: false,
164        virtualize_cpuid: true,
165        cpuid_virtualized_by_backend: false,
166        backend_supports_madvise: true,
167        discover_live_file_metadata: false,
168        use_thread_local_clock_reads: false,
169        detect_host_clock_futex_timeouts: false,
170        syscall_clobbers_virtualized_by_backend: false,
171        cancel_killed_thread_rpcs: false,
172        backend_reports_physical_process_exits: false,
173        backend_serializes_fork_children: false,
174        backend_dispatches_thread_tools: true,
175        backend_requires_thread_directed_process_signals: false,
176        backend_virtualizes_capability_prctls: false,
177        backend_defers_vfork_child_registration: false,
178        has_uts_namespace: true,
179        // The path to the directory where syscalls will be recorded.
180        replay_data: Some(data.to_path_buf()),
181        clock_multiplier: None,
182        epoch: default_config.epoch,
183        gdbserver: false,
184        gdbserver_port: default_config.gdbserver_port,
185        kill_daemons: default_config.kill_daemons,
186        max_timeslice: default_config.max_timeslice,
187        target_timeslice: default_config.target_timeslice,
188        seed: default_config.seed,
189        rng_seed: default_config.rng_seed,
190        imprecise_timers: false,
191        chaos: false,
192        sigint_instakill: false,
193        warn_non_zero_binds: false,
194        sched_heuristic: Default::default(),
195        sched_seed: default_config.sched_seed,
196        recordreplay_modes: true,
197        record_preemptions: false,
198        record_preemptions_to: None,
199        replay_preemptions_from: None,
200        replay_schedule_from: None,
201        replay_exhausted_panic: false,
202        die_on_desync: true,
203        stacktrace_event: Vec::new(),
204        stacktrace_signal: None,
205        preemption_stacktrace: false,
206        preemption_stacktrace_log_file: None,
207        stop_after_turn: None,
208        stop_after_iter: None,
209        debug_externalize_sockets: false,
210        debug_futex_mode: BlockingMode::Precise,
211        sched_sticky_random_param: 0.0,
212        no_rcb_time: false,
213        detlog_heap: false,
214        detlog_stack: false,
215        detlog_regs: false,
216        detlog_regs_cadence: 1,
217        sysinfo_uptime_offset: 120,
218        memory: 1024 * 1024 * 1024,
219        interrupt_at: vec![],
220        happens_before: None,
221        fuzz_futexes: false,
222        chaos_target_races: false,
223        chaos_per_thread_slowdown: false,
224        chaos_slowdown_max_factor: 10.0,
225        chaos_epoch_length_ns: 0,
226        fuzz_seed: None,
227    };
228    if config.max_timeslice.is_some() && !reverie_ptrace::is_perf_supported() {
229        tracing::warn!(
230            "Hardware perf counters are not supported on this machine. Records/Replays may randomly fail!"
231        );
232        config.max_timeslice = None;
233    }
234    config
235}

237#[cfg(test)]
238mod tests {
239    use super::*;
240
241    #[test]
242    fn record_version_requires_an_exact_match() {
243        assert!(RECORD_VERSION.compatible_with(&RECORD_VERSION));
244        assert!(!RECORD_VERSION.compatible_with(&RecordVersion(0x105)));
245        assert!(!RECORD_VERSION.compatible_with(&RecordVersion(0x110)));
246    }
247
248    #[test]
249    fn record_and_replay_preserve_partial_subscriptions() {
250        assert!(record_or_replay_config(Path::new("replay-data")).passthru_opt);
251    }
252
253    #[test]
254    fn record_version_rejects_pre_madvise_policy_streams() {
255        assert!(!RECORD_VERSION.compatible_with(&RecordVersion(0x104)));
256        assert!(!RECORD_VERSION.compatible_with(&RecordVersion(0x102)));
257    }
