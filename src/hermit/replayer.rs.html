49fn capture_guest_fd(pid: Pid, fd: libc::c_int) -> (Option<std::os::fd::OwnedFd>, Option<String>) {
50    match crate::fd::duplicate_guest_fd(pid, fd) {
51        Ok(duplicate) => (Some(duplicate), None),
52        Err(error) if error.raw_os_error() == Some(libc::EBADF) => (None, None),
53        Err(error) => (None, Some(error.to_string())),
54    }
55}

57fn replay_root(pid: Pid) -> Option<PathBuf> {
58    std::fs::canonicalize(format!("/proc/{}/root", pid.as_raw())).ok()
59}

81fn remember_materialized_file(pid: Pid, fd: libc::c_int) {
82    let Ok(metadata) = std::fs::metadata(format!("/proc/{}/fd/{fd}", pid.as_raw())) else {
83        return;
84    };
85    if metadata.file_type().is_file() {
86        materialized_files()
87            .lock()
88            .unwrap_or_else(|poisoned| poisoned.into_inner())
89            .insert(ReplayFileIdentity::from_metadata(&metadata));
90    }
91}

98/// A Reverie tool that replays syscalls. Note that only syscalls that cannot be
99/// made deterministic are forwarded to this tool.
100#[derive(Default, Serialize, Deserialize)]
101pub struct Replayer {
102    // Keep track of the data directory. Each thread uses this path to open its
103    // event stream.
104    data: PathBuf,
105    /// Duplicates of this guest process's captured output endpoints.
106    #[serde(skip)]
107    stdout: Option<std::os::fd::OwnedFd>,
108    #[serde(skip)]
109    stderr: Option<std::os::fd::OwnedFd>,
110    /// Preserve replayed write ordering independently for each captured stream.
111    #[serde(skip)]
112    stdout_output_lock: tokio::sync::Mutex<()>,
113    #[serde(skip)]
114    stderr_output_lock: tokio::sync::Mutex<()>,
115    #[serde(skip)]
116    stdout_error: Option<String>,
117    #[serde(skip)]
118    stderr_error: Option<String>,
119}

126    fn new(pid: Pid, cfg: &<Self::GlobalState as GlobalTool>::Config) -> Self {
127        let (stdout, stdout_error) = capture_guest_fd(pid, libc::STDOUT_FILENO);
128        let (stderr, stderr_error) = capture_guest_fd(pid, libc::STDERR_FILENO);
129        Self {
130            data: cfg.replay_data.as_ref().unwrap().clone(),
131            stdout,
132            stderr,
133            stdout_output_lock: tokio::sync::Mutex::new(()),
134            stderr_output_lock: tokio::sync::Mutex::new(()),
135            stdout_error,
136            stderr_error,
137        }
138    }

140    fn init_thread_state(
141        &self,
142        child: Tid,
143        _parent: Option<(Tid, &Self::ThreadState)>,
144    ) -> Self::ThreadState {
145        // We have to unwrap because there is now way to handle errors here.
146        EventReader::open(&self.data, child).unwrap_or_else(|err| {
147            panic!(
148                "Failed to open {:?} for thread {}: {}",
149                self.data, child, err
150            )
151        })
152    }

154    fn subscriptions(config: &<Self::GlobalState as GlobalTool>::Config) -> Subscription {
155        // Subscribe to the exact same events as the recorder does.
156        crate::recorder::Recorder::subscriptions(config)
157    }

159    async fn handle_syscall_event<G: Guest<Self>>(
160        &self,
161        guest: &mut G,
162        syscall: Syscall,
163    ) -> Result<i64, Error> {
164        self.expect_syscall(guest, syscall);
165
166        // NOTE: This match statement should be identical to the one in the
167        // recorder. Otherwise, our recorder and replayer will disagree about
168        // how syscalls should be handled.
169        //
170        // FIXME: Figure out a way to avoid duplicate code. (Merge record/replay
171        // into a single tool?)
172        Ok(match syscall {
173            // We must let through execve without any modification. Recording
174            // events for these is hard because execve only returns upon
175            // failure.
176            Syscall::Execve(_) => guest.inject(syscall).await,
177            Syscall::Execveat(_) => guest.inject(syscall).await,
178            Syscall::Brk(_) => self.let_through(guest, syscall).await,
179            Syscall::Mprotect(_) => self.let_through(guest, syscall).await,
180            Syscall::ArchPrctl(_) => {
181                // To properly handle arch_prctl, we should prevent calls from
182                // using ARCH_SET_CPUID since we already do that for the
183                // tracees. However, it is rare for programs to use
184                // ARCH_SET_CPUID. For all other arch_prctl subfunctions, we
185                // should let it through.
186                self.let_through(guest, syscall).await
187            }
188            Syscall::Read(syscall) => self.handle_read(guest, syscall).await,
189            Syscall::Pread64(syscall) => self.handle_pread64(guest, syscall).await,
190            Syscall::Readv(syscall) => {
191                self.handle_readv_family(
192                    guest,
193                    syscall.iov().map(|a| a.as_raw()),
194                    syscall.len(),
195                    syscall.into(),
196                )
197                .await
198            }
199            Syscall::Preadv(syscall) => {
200                self.handle_readv_family(
201                    guest,
202                    syscall.iov().map(|a| a.as_raw()),
203                    syscall.iov_len(),
204                    syscall.into(),
205                )
206                .await
207            }
208            Syscall::Preadv2(syscall) => {
209                self.handle_readv_family(
210                    guest,
211                    syscall.iov().map(|a| a.as_raw()),
212                    syscall.iov_len() as usize,
213                    syscall.into(),
214                )
215                .await
216            }
217            Syscall::Recvfrom(syscall) => self.handle_recvfrom(guest, syscall).await,
218            Syscall::Recvmsg(syscall) => self.handle_recvmsg(guest, syscall).await,
219            Syscall::Write(syscall) => self.handle_write_family(guest, syscall.into()).await,
220            Syscall::Pwrite64(syscall) => self.handle_write_family(guest, syscall.into()).await,
221            Syscall::Writev(syscall) => self.handle_write_family(guest, syscall.into()).await,
222            Syscall::Pwritev(syscall) => self.handle_write_family(guest, syscall.into()).await,
223            Syscall::Pwritev2(syscall) => self.handle_write_family(guest, syscall.into()).await,
224            Syscall::Access(_) => self.handle_simple(guest, syscall).await,
225            Syscall::Lseek(_) => self.handle_optional_fd_position(guest, syscall).await,
226            Syscall::Stat(syscall) => self.handle_stat_family(guest, syscall.into()).await,
227            Syscall::Fstat(syscall) => self.handle_stat_family(guest, syscall.into()).await,
228            Syscall::Lstat(syscall) => self.handle_stat_family(guest, syscall.into()).await,
229            Syscall::Newfstatat(syscall) => self.handle_stat_family(guest, syscall.into()).await,
230            Syscall::Statfs(syscall) => self.handle_statfs(guest, syscall.buf()).await,
231            Syscall::Fstatfs(syscall) => self.handle_statfs(guest, syscall.buf()).await,
232            Syscall::Statx(syscall) => self.handle_statx(guest, syscall).await,
233            Syscall::Getdents(syscall) => self.handle_getdents(guest, syscall).await,
234            Syscall::Getdents64(syscall) => self.handle_getdents64(guest, syscall).await,
235            Syscall::Mmap(syscall) => self.handle_mmap(guest, syscall).await,
236            Syscall::Munmap(_) => self.let_through(guest, syscall).await,
237            Syscall::Open(call) => {
238                self.handle_virtual_fd_create(guest, syscall, call.flags())
239                    .await
240            }
241            Syscall::Openat(call) => {
242                self.handle_virtual_fd_create(guest, syscall, call.flags())
243                    .await
244            }
245            Syscall::Openat2(call) => self.handle_openat2(guest, call).await,
246            Syscall::Close(_) => self.handle_close(guest, syscall).await,
247            Syscall::Fchdir(call) => self.handle_fchdir(guest, call).await,
248            Syscall::Fadvise64(_) => self.handle_simple(guest, syscall).await,
249            Syscall::Flock(_) => self.handle_simple(guest, syscall).await,
250            Syscall::Ftruncate(syscall) => self.handle_ftruncate(guest, syscall),
251            Syscall::Dup(_) => {
252                self.handle_replayed_side_effect(guest, syscall, "dup")
253                    .await
254            }
255            Syscall::Dup2(_) => self.handle_dup2(guest, syscall).await,
256            Syscall::Dup3(_) => {
257                self.handle_replayed_side_effect(guest, syscall, "dup3")
258                    .await
259            }
260            Syscall::Ioctl(syscall) => self.handle_ioctl(guest, syscall).await,
261            Syscall::Socket(_) => {
262                self.handle_replayed_side_effect(guest, syscall, "socket")
263                    .await
264            }
265            // AUTONOMOUS-BOT-IMPLEMENTED
266            // TODO-HUMAN-REVIEW(PR-979): recreate and validate the pidfd on
267            // replay. pidfd_open returns a real kernel fd that later modeled
268            // descriptor ops (fcntl/poll/waitid/close) act on, so we re-inject
269            // it and assert the fd matches the recorded value, catching
270            // fd-allocation or target-lifetime drift rather than blindly
271            // substituting the recorded number.
272            Syscall::PidfdOpen(_) => {
273                self.handle_replayed_side_effect(guest, syscall, "pidfd_open")
274                    .await
275            }
276            Syscall::ClockGettime(syscall) => self.handle_clock_gettime(guest, syscall).await,
277            Syscall::Gettimeofday(syscall) => self.handle_gettimeofday(guest, syscall).await,
278            Syscall::Settimeofday(_) => self.handle_simple(guest, syscall).await,
279            Syscall::Time(syscall) => self.handle_time(guest, syscall).await,
280            Syscall::Setsockopt(_) => {
281                self.handle_replayed_side_effect(guest, syscall, "setsockopt")
282                    .await
283            }
284            Syscall::Fcntl(call)
285                if matches!(
286                    call.cmd(),
287                    FcntlCmd::F_DUPFD(_) | FcntlCmd::F_DUPFD_CLOEXEC(_)
288                ) =>
289            {
290                self.handle_replayed_side_effect(guest, syscall, "fcntl_dupfd")
291                    .await
292            }
293            Syscall::Fcntl(call) if matches!(call.cmd(), FcntlCmd::F_SETFD(_)) => {
294                self.handle_replayed_side_effect(guest, syscall, "fcntl_setfd")
295                    .await
296            }
297            Syscall::Fcntl(_) => self.handle_simple(guest, syscall).await,
298            Syscall::Connect(_) => self.handle_simple(guest, syscall).await,
299            Syscall::Sendto(_) => self.handle_simple(guest, syscall).await,
300            Syscall::Sendmsg(_) => self.handle_simple(guest, syscall).await,
301            Syscall::Poll(syscall) => self.handle_poll(guest, syscall).await,
302            Syscall::Ppoll(syscall) => self.handle_ppoll(guest, syscall).await,
303            Syscall::EpollWait(syscall) => self.handle_epoll_wait(guest, syscall).await,
304            Syscall::Getsockopt(syscall) => self.handle_sockopt_family(guest, syscall.into()).await,
305            Syscall::Getpeername(syscall) => {
306                self.handle_sockopt_family(guest, syscall.into()).await
307            }
308            Syscall::Getsockname(syscall) => {
309                self.handle_sockopt_family(guest, syscall.into()).await
310            }
311            Syscall::Getrandom(syscall) => self.handle_getrandom(guest, syscall).await,
312            Syscall::Readlink(syscall) => self.handle_readlink(guest, syscall).await,
313            // AUTONOMOUS-BOT-IMPLEMENTED
314            // TODO-HUMAN-REVIEW(#653)
315            Syscall::Mkdir(_) => {
316                self.handle_replayed_side_effect(guest, syscall, "mkdir")
317                    .await
318            }
319            Syscall::Unlink(_) => self.handle_optional_path_removal(guest, syscall).await,
320            Syscall::Unlinkat(call) => self.handle_unlinkat(guest, call).await,
321            Syscall::Mkdirat(_)
322            | Syscall::Mknodat(_)
323            | Syscall::Fchownat(_)
324            | Syscall::Linkat(_)
325            | Syscall::Renameat(_)
326            | Syscall::Renameat2(_)
327            | Syscall::Symlinkat(_)
328            | Syscall::Fchmodat(_)
329            | Syscall::Utimensat(_) => self.handle_confined_path_mutation(guest, syscall).await,
330            // AUTONOMOUS-BOT-IMPLEMENTED
331            Syscall::Other(Sysno::close_range, _) => self.handle_close_range(guest, syscall).await,
332            unsupported => return Ok(guest.inject_with_retry(unsupported).await?),
333        }?)
334    }

346    pub(super) async fn reserve_replay_fd<G: Guest<Self>>(
347        &self,
348        guest: &mut G,
349        fd: i32,
350        cloexec: bool,
351    ) {
352        let flags = if cloexec {
353            EfdFlags::EFD_CLOEXEC
354        } else {
355            EfdFlags::empty()
356        };
357        let placeholder = guest
358            .inject_with_retry(Eventfd2::new().with_count(0).with_flags(flags))
359            .await
360            .unwrap_or_else(|error| {
361                panic!("could not reserve replay FD {fd} with an eventfd: {error}")
362            });
363        if placeholder != i64::from(fd) {
364            let _ = guest.inject(Close::new().with_fd(placeholder as i32)).await;
365            panic!(
366                "replay FD namespace diverged: expected slot {fd}, placeholder returned {placeholder}"
367            );
368        }
369    }
370    pub(super) fn fd_is_in_replay_root(&self, pid: Pid, fd: libc::c_int) -> bool {
371        let path = format!("/proc/{}/fd/{fd}", pid.as_raw());
372        let Ok(metadata) = std::fs::metadata(&path) else {
373            return false;
374        };
375        if metadata.file_type().is_file() {
376            return materialized_files()
377                .lock()
378                .unwrap_or_else(|poisoned| poisoned.into_inner())
379                .contains(&ReplayFileIdentity::from_metadata(&metadata));
380        }
381        if !metadata.file_type().is_dir() {
382            return false;
383        }
384        let Some(root) = replay_root(pid) else {
385            return false;
386        };
387        std::fs::canonicalize(path).is_ok_and(|path| path.starts_with(root))
388    }

390    // AUTONOMOUS-BOT-IMPLEMENTED
391    // TODO-HUMAN-REVIEW(#662): Audit replay path confinement.
392    fn open_path_in_replay_root<G: Guest<Self>>(
393        &self,
394        guest: &G,
395        syscall: Syscall,
396    ) -> Option<PathBuf> {
397        let (dirfd, path) = match syscall {
398            Syscall::Open(call) => (
399                libc::AT_FDCWD,
400                call.path().and_then(|path| path.read(&guest.memory()).ok()),
401            ),
402            Syscall::Openat(call) => (
403                call.dirfd(),
404                call.path().and_then(|path| path.read(&guest.memory()).ok()),
405            ),
406            _ => return None,
407        };
408        let path = path?;
409        let root = replay_root(guest.pid())?;
410
411        let candidate = if let Ok(relative) = path.strip_prefix(Path::new("/")) {
412            root.join(relative)
413        } else {
414            let base = if dirfd == libc::AT_FDCWD {
415                format!("/proc/{}/cwd", guest.pid().as_raw())
416            } else {
417                format!("/proc/{}/fd/{dirfd}", guest.pid().as_raw())
418            };
419            let base = std::fs::canonicalize(base).ok()?;
420            if !base.starts_with(&root) {
421                return None;
422            }
423            base.join(path)
424        };
425
426        let resolved = match std::fs::canonicalize(&candidate) {
427            Ok(path) => path,
428            Err(_) => {
429                let parent = std::fs::canonicalize(candidate.parent()?).ok()?;
430                parent.join(candidate.file_name()?)
431            }
432        };
433        resolved.starts_with(&root).then_some(resolved)
434    }

436    // AUTONOMOUS-BOT-IMPLEMENTED
437    // TODO-HUMAN-REVIEW(#662): Audit replay-root file and directory materialization.
438    async fn handle_virtual_fd_create<G: Guest<Self>>(
439        &self,
440        guest: &mut G,
441        syscall: Syscall,
442        flags: OFlag,
443    ) -> Result<i64, Errno> {
444        let event = next_event!(guest, Open)?;
445        let recorded = event.result;
446        if let Ok(fd) = recorded {
447            let candidate = event
448                .materialize
449                .then(|| self.open_path_in_replay_root(guest, syscall))
450                .flatten();
451            if let Some(candidate) = &candidate {
452                if flags.contains(OFlag::O_DIRECTORY) {
453                    let _ = std::fs::create_dir_all(candidate);
454                } else if flags.contains(OFlag::O_CREAT)
455                    && let Some(parent) = candidate.parent()
456                {
457                    let _ = std::fs::create_dir_all(parent);
458                }
459            }
460            let materialize = candidate.as_ref().is_some_and(|candidate| {
461                if flags.contains(OFlag::O_TMPFILE) {
462                    return candidate.is_dir();
463                }
464                match std::fs::metadata(candidate) {
465                    Ok(metadata) if metadata.file_type().is_dir() => true,
466                    Ok(metadata) if metadata.file_type().is_file() => materialized_files()
467                        .lock()
468                        .unwrap_or_else(|poisoned| poisoned.into_inner())
469                        .contains(&ReplayFileIdentity::from_metadata(&metadata)),
470                    Err(error)
471                        if error.kind() == std::io::ErrorKind::NotFound
472                            && flags.contains(OFlag::O_CREAT) =>
473                    {
474                        true
475                    }
476                    _ => false,
477                }
478            });
479            if materialize {
480                match guest.inject_with_retry(syscall).await {
481                    Ok(actual) => {
482                        assert_eq!(
483                            actual, fd,
484                            "replay materialized open returned a different descriptor"
485                        );
486                        remember_materialized_file(guest.pid(), actual as libc::c_int);
487                    }
488                    Err(error) => {
489                        tracing::debug!(
490                            ?syscall,
491                            %error,
492                            "replay path unavailable; reserving a virtual descriptor"
493                        );
494                        self.reserve_replay_fd(guest, fd as i32, flags.contains(OFlag::O_CLOEXEC))
495                            .await;
496                    }
497                }
498            } else {
499                self.reserve_replay_fd(guest, fd as i32, flags.contains(OFlag::O_CLOEXEC))
500                    .await;
501            }
502        }
503        recorded
504    }
505    fn dirfd_is_confined(&self, pid: Pid, fd: libc::c_int) -> bool {
506        fd == libc::AT_FDCWD || self.fd_is_in_replay_root(pid, fd)
507    }
508
509    fn path_mutation_dirfds_are_confined(&self, pid: Pid, syscall: Syscall) -> bool {
510        match syscall {
511            Syscall::Mkdirat(call) => self.dirfd_is_confined(pid, call.dirfd()),
512            Syscall::Mknodat(call) => self.dirfd_is_confined(pid, call.dirfd()),
513            Syscall::Fchownat(call) => self.dirfd_is_confined(pid, call.dirfd()),
514            Syscall::Fchmodat(call) => self.dirfd_is_confined(pid, call.dirfd()),
515            Syscall::Utimensat(call) => self.dirfd_is_confined(pid, call.dirfd()),
516            Syscall::Symlinkat(call) => self.dirfd_is_confined(pid, call.newdirfd()),
517            Syscall::Linkat(call) => {
518                self.dirfd_is_confined(pid, call.olddirfd())
519                    && self.dirfd_is_confined(pid, call.newdirfd())
520            }
521            Syscall::Renameat(call) => {
522                self.dirfd_is_confined(pid, call.olddirfd())
523                    && self.dirfd_is_confined(pid, call.newdirfd())
524            }
525            Syscall::Renameat2(call) => {
526                self.dirfd_is_confined(pid, call.olddirfd())
527                    && self.dirfd_is_confined(pid, call.newdirfd())
528            }
529            _ => false,
530        }
531    }

533    async fn handle_openat2<G: Guest<Self>>(
534        &self,
535        guest: &mut G,
536        syscall: reverie::syscalls::Openat2,
537    ) -> Result<i64, Errno> {
538        let recorded = next_event!(guest, Return);
539        if let Ok(fd) = recorded {
540            if self.dirfd_is_confined(guest.pid(), syscall.dirfd()) {
541                match guest.inject_with_retry(syscall).await {
542                    Ok(actual) => {
543                        assert_eq!(actual, fd, "replayed openat2 returned a different fd")
544                    }
545                    Err(error) => {
546                        tracing::debug!(%error, "replay openat2 path unavailable");
547                        self.reserve_replay_fd(guest, fd as i32, false).await;
548                    }
549                }
550            } else {
551                self.reserve_replay_fd(guest, fd as i32, false).await;
552            }
553        }
554        recorded
555    }
556
557    async fn handle_confined_path_mutation<G: Guest<Self>>(
558        &self,
559        guest: &mut G,
560        syscall: Syscall,
561    ) -> Result<i64, Errno> {
562        let recorded = next_event!(guest, Return);
563        if let Ok(expected) = recorded
564            && self.path_mutation_dirfds_are_confined(guest.pid(), syscall)
565        {
566            match guest.inject_with_retry(syscall).await {
567                Ok(actual) => assert_eq!(
568                    actual, expected,
569                    "replayed path mutation returned a different result"
570                ),
571                Err(error @ (Errno::ENOENT | Errno::ENOTDIR | Errno::EROFS)) => {
572                    tracing::debug!(?syscall, %error, "replay path mutation kept virtual");
573                }
574                Err(error) => {
575                    panic!(
576                        "replayed path mutation {syscall:?} failed after recording returned {expected}: {error}"
577                    );
578                }
579            }
580        }
581        recorded
582    }

612    async fn handle_optional_path_removal<G: Guest<Self>>(
613        &self,
614        guest: &mut G,
615        syscall: Syscall,
616    ) -> Result<i64, Errno> {
617        let recorded = next_event!(guest, Return);
618        if let Ok(expected) = recorded {
619            match guest.inject_with_retry(syscall).await {
620                Ok(actual) => assert_eq!(
621                    actual, expected,
622                    "replayed path removal returned a different result"
623                ),
624                Err(error @ (Errno::ENOENT | Errno::ENOTDIR)) => {
625                    tracing::debug!(
626                        ?syscall,
627                        %error,
628                        "replay path unavailable; keeping path removal virtual"
629                    );
630                }
631                Err(error) => {
632                    panic!(
633                        "replayed path removal {syscall:?} failed after recording returned {expected}: {error}"
634                    );
635                }
636            }
637        }
638        recorded
639    }

641    async fn handle_optional_fd_position<G: Guest<Self>>(
642        &self,
643        guest: &mut G,
644        syscall: Syscall,
645    ) -> Result<i64, Errno> {
646        let Syscall::Lseek(call) = syscall else {
647            unreachable!("descriptor-position handler received {syscall:?}");
648        };
649        if !self.fd_is_in_replay_root(guest.pid(), call.fd()) {
650            return self.handle_simple(guest, syscall).await;
651        }
652        if matches!(call.whence(), Whence::SEEK_DATA | Whence::SEEK_HOLE) {
653            let recorded = next_event!(guest, Return);
654            if let Ok(expected) = recorded {
655                let duplicate = crate::fd::duplicate_guest_fd(guest.pid(), call.fd())
656                    .unwrap_or_else(|error| {
657                        panic!("failed to duplicate replay file for extent seek: {error}")
658                    });
659                // SAFETY: duplicate owns the same open-file description and the
660                // recorded extent seek returned this nonnegative offset.
661                let actual =
662                    unsafe { libc::lseek(duplicate.as_raw_fd(), expected, libc::SEEK_SET) };
663                assert_eq!(
664                    actual, expected,
665                    "failed to restore replay position after extent seek"
666                );
667            }
668            return recorded;
669        }
670        let recorded = next_event!(guest, Return);
671        if let Ok(expected) = recorded {
672            match guest.inject_with_retry(syscall).await {
673                Ok(actual) => assert_eq!(
674                    actual, expected,
675                    "replayed descriptor position returned a different offset"
676                ),
677                Err(error @ (Errno::EBADF | Errno::ESPIPE)) => {
678                    tracing::debug!(
679                        ?syscall,
680                        %error,
681                        "replay descriptor is virtual; keeping position change virtual"
682                    );
683                }
684                Err(error) => {
685                    panic!(
686                        "replayed descriptor position {syscall:?} failed after recording returned {expected}: {error}"
687                    );
688                }
689            }
690        }
691        recorded
692    }

694    /// Replays the recorded result of `close` while preserving its physical FD
695    /// namespace effect.
696    async fn handle_close<G: Guest<Self>>(
697        &self,
698        guest: &mut G,
699        syscall: Syscall,
700    ) -> Result<i64, Errno> {
701        let recorded = next_event!(guest, Return);
702
703        // Linux releases a descriptor even when close reports EINTR, EIO,
704        // ENOSPC, or EDQUOT. EBADF leaves the namespace unchanged, while
705        // ERESTARTSYS means Reverie must restart the injection first.
706        if !matches!(recorded, Err(Errno::EBADF | Errno::ERESTARTSYS))
707            && let Err(error) = guest.inject(syscall).await
708        {
709            if error == Errno::EBADF {
710                // Some replayed descriptor sources, notably SCM_RIGHTS,
711                // currently have no physical peer.
712                tracing::debug!(?syscall, "replayed close had no physical descriptor");
713            } else {
714                tracing::warn!(
715                    ?error,
716                    "physical close during replay differed from the recorded result"
717                );
718            }
719        }
720
721        recorded
722    }

724    // Check if we received the expected syscall or not.
725    fn expect_syscall<G: Guest<Self>>(&self, guest: &mut G, syscall: Syscall) {
726        let thread = guest.tid();
727        let next_count = guest.thread_state().count + 1;
728        let debug_event = guest
729            .thread_state_mut()
730            .next_debug_event()
731            .unwrap_or_else(|source| {
732                panic!(
733                    "Replay syscall stream ended unexpectedly for recording {} on thread {} at event {} while the guest executed {:?}: {}",
734                    self.data.display(),
735                    thread,
736                    next_count,
737                    syscall,
738                    source,
739                )
740            });
741
742        // Compare only the argument registers the syscall actually uses. Reverie
743        // keeps all six raw registers in every typed syscall and derives
744        // `PartialEq` over them, so unused registers (which hold arbitrary
745        // leftover guest values) would otherwise produce false desyncs for any
746        // syscall with fewer than six arguments.
747        if normalize_unused_args(debug_event.syscall()) == normalize_unused_args(syscall) {
748            return;
749        }
750
751        if guest.is_root_thread() {
752            // execve and execveat for the root thread are special cases. Even
753            // when ASLR is turned off, these can have different pointer values
754            // than what we originally recorded because the pointers originate
755            // outside of the current address space.
756            match syscall {
757                Syscall::Execve(_) | Syscall::Execveat(_) => return,
758                _ => {}
759            }
760        }
761
762        let error = DesyncError {
763            thread,
764            count: guest.thread_state().count,
765            actual: DebugEvent::new(syscall, &guest.memory()),
766            expected: debug_event,
767        };
768        let summary = error.summary(&self.data, 16, 4).to_string();
769        let report = match error.generate_report(&self.data) {
770            Ok(report) => format!("Full desynchronization report: {}", report.display()),
771            Err(report_error) => {
772                format!("Could not write the full desynchronization report: {report_error}")
773            }
774        };
775
776        panic!(
777            "Replay diverged from recording {} on thread {} at syscall event {}. Re-record the workload with the same Hermit build after diagnosing the mismatch.\n{}\n{}",
778            self.data.display(),
779            thread,
780            error.count,
781            summary,
782            report,
783        );
784    }

786    /// Called for syscalls to explicitly let through. This should only be called
787    /// for syscalls that cannot be recorded and are necessary for the program to
788    /// function correctly. Examples of syscalls that fall into this category are
789    /// ones that help with memory management (e.g., `brk`, `mprotect`, or
790    /// `munmap`) or process management (e.g., `fork`, `vfork`, `clone`).
791    ///
792    /// For these syscalls, we just care about detecting dsynchronization and
793    /// simply inject them to let them through.
794    async fn let_through<G: Guest<Self>>(
795        &self,
796        guest: &mut G,
797        syscall: Syscall,
798    ) -> Result<i64, Errno> {
799        // NOTE: Must use `inject_with_retry` here. Otherwise, we may end up
800        // introducing non-determinism into the replay and popping multiple
801        // syscall events.
802        guest.inject_with_retry(syscall).await
803    }

805    /// Handles a syscall whose only value we care about is the return value
806    /// (i.e., simple syscalls).
807    ///
808    /// For recording, this means we only record the return value of the syscall.
809    /// For replay, this means we substitute the return value in lieu of actually
810    /// performing the injection.
811    ///
812    /// The syscall must have two properties satisfied for this to be called:
813    ///  1. The syscall must only have "input" arguments. That is, all arguments
814    ///     must either be values or const pointers.
815    ///  2. The execution of the program must not depend on anything else other
816    ///     than the return value of the syscall. For example, `mmap` would violate
817    ///     this rule since it affects later memory access.
818    ///
819    /// There are many syscalls who satisfy these two requirements.
820    async fn handle_simple<G: Guest<Self>>(
821        &self,
822        guest: &mut G,
823        _syscall: Syscall,
824    ) -> Result<i64, Errno> {
825        next_event!(guest, Return)
826    }

828    async fn handle_dup2<G: Guest<Self>>(
829        &self,
830        guest: &mut G,
831        syscall: Syscall,
832    ) -> Result<i64, Errno> {
833        let recorded = next_event!(guest, Return);
834        if recorded.is_ok() {
835            let actual = guest.inject_with_retry(syscall).await;
836            // Some source descriptors are virtual: open-family syscalls replay
837            // their recorded return value without creating a live kernel fd.
838            // Preserve that behavior when there is nothing to duplicate in the
839            // replay process.
840            if actual != Err(Errno::EBADF) {
841                assert_eq!(actual, recorded, "dup2 fd-table mutation diverged");
842            }
843        }
844        recorded
845    }
