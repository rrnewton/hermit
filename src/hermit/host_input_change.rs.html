9//! Name a host file that changed during one verification run.
10//!
11//! Each run of `hermit run --verify` records the host identity of every file
12//! the guest opens, in order, with the open that found it (see
13//! `detcore_model::host_input`). A file that a run opens several times shows a
14//! PATTERN of identities: each open finds an identity it has found at that
15//! path before, or a new one. A file the guest itself creates, rewrites or
16//! replaces changes at the same opens in both runs, even though its host inode
17//! numbers and times differ between them. So the two runs' patterns agree
18//! unless something outside the guest changed a file underneath one run: that
19//! run finds a new identity at an open where the other run found a known one.
20//!
21//! Only one kind of change is named: the file was REPLACED, so the open found
22//! a different host inode (`dev`, `ino`). Hermit mints deterministic inodes in
23//! the order it first sees host inodes, so a replacement gives that run one
24//! more inode and shifts every later deterministic inode number, whatever the
25//! guest does with the file. A change of size or time alone is not named.
26//!
27//! A pattern difference is evidence of a host change only if the runs had not
28//! diverged yet. A guest that behaves differently in the two runs can, for
29//! example, replace a file in one run only, and then the patterns differ too,
30//! but that behaviour shows in the compared log before the open. So the
31//! difference names the cause of a divergence only when the comparison
32//! compared everything (one strict enough for bitwise parity) and the open
33//! that found it is recorded, in BOTH runs' logs, before the first divergent
34//! record ([`explains_divergence`]): up to that record both runs did the same
35//! things, so the guest did not cause the difference.
36//!
37//! Positions are records in the log, not times. On a backend whose log is not
38//! one stream in time order, such as SaBRe, which appends the records its
39//! guest forwards after the coordinator's own, each part is still in time
40//! order, and the comparator reports the first difference in the earliest
41//! part. An open in a later part than that difference is therefore refused,
42//! and an open before it in the same part did happen before it.
43//!
44//! Nor is a change named when the guest itself rebound that path, or a
45//! directory above it, in either run: renamed to or from it, linked or
46//! unlinked it, or created something there. Those operations can replace the
47//! file without the compared log showing any difference (a rename onto a name
48//! that already names the same file succeeds and changes nothing), so a
49//! replacement there may be the guest's own.
50//!
51//! Nothing else is concluded. The walk stops at the first open whose path,
52//! thread or syscall differs between the runs and at the first difference that
53//! is not a replacement, an incomplete log names nothing, an open whose finish
54//! record a log does not show exactly once is not placed, and paths below
55//! `/proc`, `/sys` and `/dev` are skipped: those are kernel objects whose
56//! metadata is not file content and which Hermit virtualizes separately.

70/// One run's host-input log.
71#[derive(Clone, Debug, Default, Eq, PartialEq)]
72pub struct HostInputs {
73    /// The records, in the order the run made the opens.
74    pub records: Vec<HostInputRecord>,
75    /// The paths the guest itself rebound in its file namespace during the
76    /// run ([`HostMutationRecord`]).
77    pub mutated: Vec<String>,
78    /// Whether every line parsed and the log ends with the
79    /// [`HostInputLogEnd`] line counting exactly these lines. Detcore writes
80    /// the log once, when the run ends, so a run that did not end normally, a
81    /// failed write or a cut file leaves it incomplete.
82    pub complete: bool,
83}
84
85/// Read one run's log. A missing or empty file is an incomplete log.
86pub fn read_host_inputs(path: &Path) -> std::io::Result<HostInputs> {
87    match std::fs::read_to_string(path) {
88        Ok(text) => Ok(parse_host_inputs(&text)),
89        Err(error) if error.kind() == std::io::ErrorKind::NotFound => Ok(HostInputs::default()),
90        Err(error) => Err(error),
91    }
92}

94fn parse_host_inputs(text: &str) -> HostInputs {
95    let mut lines: Vec<&str> = text.lines().collect();
96    let end = lines
97        .pop()
98        .and_then(|line| serde_json::from_str::<HostInputLogEnd>(line).ok());
99    let mut records = Vec::new();
100    let mut mutated = Vec::new();
101    let mut parsed = true;
102    for line in &lines {
103        if let Ok(record) = serde_json::from_str::<HostInputRecord>(line) {
104            records.push(record);
105        } else if let Ok(mutation) = serde_json::from_str::<HostMutationRecord>(line) {
106            mutated.push(mutation.mutated);
107        } else {
108            parsed = false;
109        }
110    }
111    let complete = parsed && end.is_some_and(|end| end.records == lines.len() as u64);
112    HostInputs {
113        records,
114        mutated,
115        complete,
116    }
117}

119/// Whether the guest may have rebound `path`, or a directory above it, in its
120/// file namespace: one of `mutated` is `path` or an ancestor of it. Detcore
121/// reports `/`, the ancestor of every absolute path, for a change whose path
122/// it could not establish. An open whose path could not be made absolute
123/// cannot be matched against those changes at all, so it counts as rebound
124/// too.
125fn guest_rebound(mutated: &[String], path: &str) -> bool {
126    !Path::new(path).is_absolute()
127        || mutated
128            .iter()
129            .any(|rebound| Path::new(path).starts_with(Path::new(rebound)))
130}

132/// Kernel objects rather than host input files; see the module documentation.
133fn is_kernel_object(path: &str) -> bool {
134    ["/proc", "/sys", "/dev"].iter().any(|root| {
135        path == *root
136            || path
137                .strip_prefix(root)
138                .is_some_and(|rest| rest.starts_with('/'))
139    })
140}

142/// The opens one run has made so far: for each path, the distinct
143/// identities found there and the identity found at its latest open.
144#[derive(Default)]
145struct Pattern<'a> {
146    seen: HashMap<&'a str, (Vec<HostFileIdentity>, HostFileIdentity)>,
147}
148
149impl<'a> Pattern<'a> {
150    /// Record an open. Returns whether its identity is one this run has not
151    /// found at the path before, and the identity of the path's previous open.
152    fn open(&mut self, record: &'a HostInputRecord) -> (bool, Option<HostFileIdentity>) {
153        match self.seen.get_mut(record.path.as_str()) {
154            None => {
155                self.seen.insert(
156                    record.path.as_str(),
157                    (vec![record.identity], record.identity),
158                );
159                (true, None)
160            }
161            Some((distinct, latest)) => {
162                let previous = std::mem::replace(latest, record.identity);
163                let new = !distinct.contains(&record.identity);
164                if new {
165                    distinct.push(record.identity);
166                }
167                (new, Some(previous))
168            }
169        }
170    }
171}

173/// A pattern difference between the runs: in `run`, the open that thread
174/// `dtid` made as its syscall number `syscall` found `path` changed from
175/// `before` to `after`, while the other run found a known identity there.
176#[derive(Clone, Debug, Eq, PartialEq)]
177pub struct PatternDifference {
178    pub run: VerificationRun,
179    pub path: String,
180    pub dtid: u64,
181    pub syscall: u64,
182    pub before: HostFileIdentity,
183    pub after: HostFileIdentity,
184}
185
186impl PatternDifference {
187    /// The typed outcome, for a difference that [`explains_divergence`].
188    pub fn into_infrastructure_error(self) -> InfrastructureError {
189        InfrastructureError::HostInputChanged {
190            run: self.run,
191            path: self.path,
192            before: self.before,
193            after: self.after,
194        }
195    }
196}

198/// The first open, in order, at which exactly one run found an identity it
199/// had not found at that path before while the other run found one it had,
200/// when that identity is a different host inode.
201///
202/// Returns `None` when either log is incomplete, when the patterns agree up to
203/// the first open whose path, thread or syscall number differs, when both
204/// runs found an unfamiliar identity at the same open, which no single run
205/// explains, or when the first such difference keeps the host inode.
206pub fn find_pattern_difference(run1: &HostInputs, run2: &HostInputs) -> Option<PatternDifference> {
207    if !run1.complete || !run2.complete {
208        return None;
209    }
210    let mut pattern1 = Pattern::default();
211    let mut pattern2 = Pattern::default();
212    for (left, right) in run1.records.iter().zip(&run2.records) {
213        if (&left.path, left.dtid, left.syscall) != (&right.path, right.dtid, right.syscall) {
214            return None;
215        }
216        if is_kernel_object(&left.path) {
217            continue;
218        }
219        let (new1, previous1) = pattern1.open(left);
220        let (new2, previous2) = pattern2.open(right);
221        let (run, previous, record) = match (new1, new2) {
222            (true, false) => (VerificationRun::Run1, previous1, left),
223            (false, true) => (VerificationRun::Run2, previous2, right),
224            _ => continue,
225        };
226        // Both runs find a new identity at a path's first open, so only a
227        // path seen before reaches this point.
228        let before = previous?;
229        if (before.dev, before.ino) == (record.identity.dev, record.identity.ino) {
230            return None;
231        }
232        // The guest may have replaced it itself, through operations whose
233        // effects the compared log does not show (a rename onto a name that
234        // already names the same file succeeds and changes nothing).
235        if [run1, run2]
236            .iter()
237            .any(|run| guest_rebound(&run.mutated, &record.path))
238        {
239            return None;
240        }
241        return Some(PatternDifference {
242            run,
243            path: record.path.clone(),
244            dtid: record.dtid,
245            syscall: record.syscall,
246            before,
247            after: record.identity,
248        });
249    }
250    None
251}

253/// Where an open sits in a run's log (`contents`): the position, as the
254/// comparator counts records (`detcore::logdiff::log_record_events`), of the
255/// record on which thread `dtid` finished its syscall number `syscall`, an
256/// open. `None` when the log does not show that record exactly once: a log
257/// without it says nothing, and a thread number and syscall count can repeat
258/// (a thread that execs from a non-leader thread takes the leader's number and
259/// keeps its own count), so a repeated one is ambiguous; also `None` for a log
260/// the comparator could not read either.
261///
262/// The open's record is a `syscall_result` whose Hermit-written prefix reads
263/// `[syscall][detcore, dtid D] finish syscall #N: ` followed by `openat(`,
264/// `open(` or `creat(`, the calls Detcore's openat handler serves. The prefix
265/// is read where it first occurs in the record, before any guest-chosen text
266/// such as a path.
267pub fn open_position(contents: &str, dtid: u64, syscall: u64) -> Option<usize> {
268    let mut found = None;
269    for (position, human, event) in detcore::logdiff::log_record_events(contents).ok()? {
270        if matches!(event, Some(DetLogEvent::SyscallResult { .. }))
271            && finished_open(human) == Some((dtid, syscall))
272        {
273            if found.is_some() {
274                return None;
275            }
276            found = Some(position);
277        }
278    }
279    found
280}

282/// The thread and that thread's syscall number of a finished open's record,
283/// read from the Hermit-written prefix where it first occurs.
284fn finished_open(human: &str) -> Option<(u64, u64)> {
285    const PREFIX: &str = "[syscall][detcore, dtid ";
286    let rest = &human[human.find(PREFIX)? + PREFIX.len()..];
287    let (dtid, rest) = rest.split_once("] finish syscall #")?;
288    let (syscall, call) = rest.split_once(": ")?;
289    let numeric = |digits: &str| {
290        (!digits.is_empty() && digits.bytes().all(|byte| byte.is_ascii_digit()))
291            .then(|| digits.parse().ok())
292            .flatten()
293    };
294    ["openat(", "open(", "creat("]
295        .iter()
296        .any(|name| call.starts_with(name))
297        .then_some((numeric(dtid)?, numeric(syscall)?))
298}

300/// Whether a pattern difference explains a divergence: the comparison was one
301/// strict enough for bitwise parity (`complete_comparison`), and the open that
302/// found the difference (placed by [`open_position`] in each run's log) is
303/// recorded before the first divergent record in both logs
304/// (`first_divergent_positions`, the comparator's positions of that record),
305/// so the two runs had done the same things up to that open. Without every
306/// position nothing is attributed.
307pub fn explains_divergence(
308    complete_comparison: bool,
309    open_positions: [Option<usize>; 2],
310    first_divergent_positions: Option<(Option<usize>, Option<usize>)>,
311) -> bool {
312    let Some((Some(left), Some(right))) = first_divergent_positions else {
313        return false;
314    };
315    complete_comparison
316        && matches!(open_positions, [Some(open1), Some(open2)] if open1 < left && open2 < right)
317}

336    fn opens(records: &[(&str, u64)]) -> HostInputs {
337        HostInputs {
338            records: records

341                .map(|(index, (path, ino))| HostInputRecord {

347                .collect(),
348            mutated: Vec::new(),
349            complete: true,
350        }

545    /// Only a structured record of a finished open, read at Hermit's own
546    /// prefix and found exactly once, places an open. A thread number and
547    /// syscall count that repeat (a non-leader exec takes the leader's number

572    /// compared everything and its open is recorded, in both runs, before
573    /// the first divergent record. A guest that behaved differently first,
574    /// such as one that replaced the file in one run only, has diverged by
575    /// the open, and is not explained.

577    fn only_an_open_before_the_divergence_in_both_runs_explains_it() {

580        assert!(explains_divergence(true, [Some(39), Some(41)], divergent));
581        assert!(!explains_divergence(false, [Some(10), Some(10)], divergent));
582        assert!(!explains_divergence(true, [Some(40), Some(10)], divergent));
583        assert!(!explains_divergence(true, [Some(10), Some(42)], divergent));
584        assert!(!explains_divergence(true, [None, Some(10)], divergent));
585        assert!(!explains_divergence(true, [Some(10), None], divergent));
586        assert!(!explains_divergence(true, [Some(10), Some(10)], None));
587        assert!(!explains_divergence(
588            true,
589            [Some(10), Some(10)],
590            Some((None, Some(42)))

592        assert!(!explains_divergence(
593            true,
594            [Some(10), Some(10)],
595            Some((Some(40), None))

599    /// A replacement of a file whose path, or a directory above it, the
600    /// guest rebound in either run is not named: the guest may have made it
601    /// with operations whose effects the compared log does not show.
602    #[test]
603    fn a_replacement_where_the_guest_rebound_the_path_is_not_named() {
604        let run1 = opens(&[("/data/F", 1), ("/data/F", 2)]);
605        let run2 = opens(&[("/data/F", 2), ("/data/F", 2)]);
606        assert!(find_pattern_difference(&run1, &run2).is_some());
607        for (in_run1, rebound) in [(true, "/data/F"), (false, "/data/F"), (false, "/data")] {
608            let (mut run1, mut run2) = (run1.clone(), run2.clone());
609            let run = if in_run1 { &mut run1 } else { &mut run2 };
610            run.mutated.push(rebound.into());
611            assert_eq!(find_pattern_difference(&run1, &run2), None, "{rebound}");
612        }
613        let mut unrelated = run1.clone();
614        unrelated.mutated = vec!["/data/F.tmp".into(), "/tmp".into(), "/data/Fx".into()];
615        assert!(find_pattern_difference(&unrelated, &run2).is_some());
616        assert!(guest_rebound(&["/".into()], "/etc/ld.so.cache"));
617        assert!(!guest_rebound(&["/etc/ld.so".into()], "/etc/ld.so.cache"));
618        // An open recorded under a path that could not be made absolute
619        // (through a directory descriptor with no known path) cannot be
620        // matched against the guest's changes, so it is never named, with or
621        // without them.
622        assert!(guest_rebound(&[], "F"));
623        assert!(guest_rebound(&["/".into()], "F"));
624        let relative1 = opens(&[("F", 1), ("F", 2)]);
625        let relative2 = opens(&[("F", 2), ("F", 2)]);
626        assert_eq!(find_pattern_difference(&relative1, &relative2), None);
627    }

645    /// replacement whose new inode shifts Hermit's numbering, so it is not

647    #[test]
