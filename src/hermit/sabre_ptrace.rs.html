90fn should_replace_signal_diagnostic(
91    existing: Option<(Signal, Option<i32>, bool)>,
92    signal: Signal,
93    si_code: Option<i32>,
94) -> bool {
95    // SaBRe handles its reserved SIGILL markers in-process. For an unknown hardware SIGILL it
96    // restores SIG_DFL and raises SIGILL again, producing a second ptrace stop from SI_TKILL.
97    // Preserve the original kernel fault context across that re-raise, but let a later hardware
98    // fault replace it so a successfully handled marker cannot leave stale diagnostics behind.
99    if let Some((Signal::SIGILL, Some(existing_code), existing_is_marker)) = existing
100        && signal == Signal::SIGILL
101        && existing_code > 0
102        && !existing_is_marker
103        && si_code.is_none_or(|code| code <= 0)
104    {
105        return false;
106    }
107    true
108}

110fn is_sabre_sigill_marker(instruction_bytes: Option<&[u8; 16]>) -> bool {
111    instruction_bytes
112        .is_some_and(|bytes| matches!(&bytes[..2], [0x0f, 0xff] | [0x0f, 0x0b] | [0x0f, 0x0c]))
113}

115fn final_physical_exit(status: &WaitStatus) -> Option<(Pid, ExitStatus)> {
116    match *status {
117        WaitStatus::Exited(pid, code) => Some((pid, ExitStatus::from_raw(code << 8))),
118        WaitStatus::Signaled(pid, signal, core_dumped) => {
119            let raw = signal as i32 | if core_dumped { 0x80 } else { 0 };
120            Some((pid, ExitStatus::from_raw(raw)))
121        }
122        _ => None,
123    }
124}

126/// Identity of one address space (`mm`). Assigned by this supervisor rather
127/// than read from the kernel, which exposes no stable mm identifier.
128#[derive(Clone, Copy, Debug, Eq, Hash, PartialEq, PartialOrd, Ord)]
129struct AddressSpaceId(u64);
130
131/// Page-classification cache, keyed by ADDRESS SPACE rather than by tracee.
132///
133/// The classification of a page is a property of the mm that maps it, not of
134/// the thread that happened to fault on it. Keying by `Pid` was unsound: under
135/// `CLONE_VM` every thread in a group shares one mm, so a mutation observed on
136/// thread A evicted only A's entries and a sibling B kept serving the
137/// pre-mutation verdict for the same page -- letting a raw syscall at a page
138/// that is no longer trusted bypass both redirection and accounting, which is
139/// exactly the invisible-site class this evidence exists to close.
140struct MappingCache {
141    entries: HashMap<(AddressSpaceId, usize), MappingClassification>,
142    /// Which address space each tracee currently runs in.
143    spaces: HashMap<Pid, AddressSpaceId>,
144    next: u64,
145}
146
147impl MappingCache {
148    fn new(root: Pid) -> Self {
149        Self {
150            entries: HashMap::new(),
151            spaces: HashMap::from([(root, AddressSpaceId(0))]),
152            next: 1,
153        }
154    }
155
156    fn fresh(&mut self) -> AddressSpaceId {
157        let id = AddressSpaceId(self.next);
158        self.next += 1;
159        id
160    }
161
162    /// The address space `pid` runs in. An unknown tracee gets a fresh space:
163    /// we have never cached anything for it, so it can hold nothing stale.
164    fn space_of(&mut self, pid: Pid) -> AddressSpaceId {
165        if let Some(id) = self.spaces.get(&pid) {
166            return *id;
167        }
168        let id = self.fresh();
169        self.spaces.insert(pid, id);
170        id
171    }
172
173    fn get(&mut self, pid: Pid, page: usize) -> Option<&MappingClassification> {
174        let space = self.space_of(pid);
175        self.entries.get(&(space, page))
176    }
177
178    fn insert(&mut self, pid: Pid, page: usize, classification: MappingClassification) {
179        let space = self.space_of(pid);
180        self.entries.insert((space, page), classification);
181    }
182
183    /// Drop every cached page of the address space `pid` runs in -- including
184    /// entries cached while a SIBLING thread was executing.
185    fn invalidate_address_space(&mut self, pid: Pid) {
186        let space = self.space_of(pid);
187        self.entries.retain(|(cached, _), _| *cached != space);
188    }
189
190    /// `child` shares `parent`'s address space (CLONE_VM / vfork).
191    fn share_address_space(&mut self, parent: Pid, child: Pid) {
192        let space = self.space_of(parent);
193        self.spaces.insert(child, space);
194    }
195
196    /// `child` gets its own address space (fork).
197    fn new_address_space(&mut self, child: Pid) {
198        let space = self.fresh();
199        self.spaces.insert(child, space);
200    }
201
202    /// `pid` exec'd: the old mm is gone, so discard it and start a fresh one.
203    fn replace_address_space(&mut self, pid: Pid) {
204        self.invalidate_address_space(pid);
205        let space = self.fresh();
206        self.spaces.insert(pid, space);
207    }
208
209    /// `pid` exited. Its cached pages stay valid for any sibling still sharing
210    /// the mm, so they are dropped only once the last user is gone.
211    fn forget(&mut self, pid: Pid) {
212        let Some(space) = self.spaces.remove(&pid) else {
213            return;
214        };
215        if !self.spaces.values().any(|other| *other == space) {
216            self.entries.retain(|(cached, _), _| *cached != space);
217        }
218    }
219
220    /// Place a newly announced `child` in the right address space.
221    ///
222    /// `shares_mm` is an OBSERVATION of the child that now exists -- see
223    /// `mm_sharing` -- not a prediction from the clone arguments. An earlier
224    /// revision read `CLONE_VM` out of the tracee's `clone_args` at the clone3
225    /// syscall-entry stop, which is before the kernel copies that buffer: a
226    /// concurrently runnable sibling thread could rewrite it in the window and
227    /// the recorded intent then disagreed with the child the kernel actually
228    /// made, in both directions. Demonstrated with a tracer that rewrites the
229    /// buffer after the entry read; `kcmp(KCMP_VM)` at this stop reported the
230    /// opposite of the entry value each time.
231    fn admit_child(&mut self, parent: Pid, child: Pid, shares_mm: MmSharing) {
232        match shares_mm {
233            MmSharing::Shared => self.share_address_space(parent, child),
234            MmSharing::Private => self.new_address_space(child),
235        }
236    }
237}

284    fn run(mut self) -> Result<(ExitStatus, PathEvidence), Error> {
285        ptrace::attach(self.root).context("failed to attach SaBRe supervisor worker")?;
286        match waitpid(self.root, Some(WaitPidFlag::__WALL))? {
287            WaitStatus::Stopped(pid, Signal::SIGSTOP) if pid == self.root => {}
288            status => {
289                return Err(anyhow!(
290                    "unexpected SaBRe supervisor attach stop: {status:?}"
291                ));
292            }
293        }
294        tracing::trace!(
295            target: "hermit::sabre::fallback",
296            tid = self.root.as_raw(),
297            "received supervisor attach stop",
298        );
299        self.set_options(self.root)
300            .context("failed to set options on the initial SaBRe tracee")?;
301        ptrace::syscall(self.root, None)
302            .context("failed to resume the initial SaBRe tracee with PTRACE_SYSCALL")?;
303
304        let mut root_status = None;
305        while !self.tracees.is_empty() {
306            let status = match waitpid(Pid::from_raw(-1), Some(WaitPidFlag::__WALL)) {
307                Ok(status) => status,
308                Err(nix::errno::Errno::EINTR) => continue,
309                Err(error) => return Err(error.into()),
310            };
311            tracing::trace!(
312                target: "hermit::sabre::fallback",
313                ?status,
314                "received ptrace wait status",
315            );
316            if let Some((pid, exit_status)) = final_physical_exit(&status) {
317                if let WaitStatus::Signaled(_, signal, _) = status
318                    && let Some(diagnostic) = self
319                        .signal_diagnostics
320                        .get(&pid)
321                        .filter(|diagnostic| diagnostic.signal == signal)
322                {
323                    let instruction_bytes = diagnostic.instruction_bytes.as_ref().map(|bytes| {
324                        bytes
325                            .iter()
326                            .map(|byte| format!("{byte:02x}"))
327                            .collect::<Vec<_>>()
328                            .join(" ")
329                    });
330                    tracing::error!(
331                        target: "hermit::sabre::fallback",
332                        tid = pid.as_raw(),
333                        ?signal,
334                        si_code = diagnostic.si_code,
335                        si_errno = diagnostic.si_errno,
336                        rip = format!("{:#x}", diagnostic.registers.rip),
337                        rsp = format!("{:#x}", diagnostic.registers.rsp),
338                        fault_address = diagnostic
339                            .fault_address
340                            .map(|address| format!("{address:#x}")),
341                        mapping = diagnostic
342                            .mapping
343                            .as_ref()
344                            .map(|mapping| mapping.line.as_str()),
345                        file_offset = diagnostic
346                            .mapping
347                            .as_ref()
348                            .map(|mapping| format!("{:#x}", mapping.file_offset)),
349                        relative_offset = diagnostic
350                            .mapping
351                            .as_ref()
352                            .map(|mapping| format!("{:#x}", mapping.relative_offset)),
353                        instruction_bytes,
354                        rax = format!("{:#x}", diagnostic.registers.rax),
355                        rbx = format!("{:#x}", diagnostic.registers.rbx),
356                        rcx = format!("{:#x}", diagnostic.registers.rcx),
357                        rdx = format!("{:#x}", diagnostic.registers.rdx),
358                        rsi = format!("{:#x}", diagnostic.registers.rsi),
359                        rdi = format!("{:#x}", diagnostic.registers.rdi),
360                        rbp = format!("{:#x}", diagnostic.registers.rbp),
361                        r8 = format!("{:#x}", diagnostic.registers.r8),
362                        r9 = format!("{:#x}", diagnostic.registers.r9),
363                        r10 = format!("{:#x}", diagnostic.registers.r10),
364                        r11 = format!("{:#x}", diagnostic.registers.r11),
365                        r12 = format!("{:#x}", diagnostic.registers.r12),
366                        r13 = format!("{:#x}", diagnostic.registers.r13),
367                        r14 = format!("{:#x}", diagnostic.registers.r14),
368                        r15 = format!("{:#x}", diagnostic.registers.r15),
369                        orig_rax = format!("{:#x}", diagnostic.registers.orig_rax),
370                        eflags = format!("{:#x}", diagnostic.registers.eflags),
371                        cs = format!("{:#x}", diagnostic.registers.cs),
372                        ss = format!("{:#x}", diagnostic.registers.ss),
373                        fs_base = format!("{:#x}", diagnostic.registers.fs_base),
374                        gs_base = format!("{:#x}", diagnostic.registers.gs_base),
375                        "SaBRe tracee terminated by a fatal signal",
376                    );
377                }
378                self.remove_tracee(pid);
379                self.physical_exit_observer
380                    .complete_physical_process_exit(pid.as_raw());
381                if pid == self.root {
382                    root_status = Some(exit_status);
383                }
384                continue;
385            }
386            match status {
387                WaitStatus::PtraceSyscall(pid) => self.handle_syscall_stop(pid)?,
388                WaitStatus::PtraceEvent(pid, _, event) => self.handle_ptrace_event(pid, event)?,
389                WaitStatus::Stopped(pid, signal) => {
390                    if signal == Signal::SIGSTOP && pid != self.root {
391                        self.states.entry(pid).or_default();
392                        self.tracees.insert(pid);
393                        self.set_options(pid)?;
394                        self.resume(pid, None)?;
395                    } else {
396                        if !matches!(signal, Signal::SIGSTOP | Signal::SIGCHLD) {
397                            let registers = ptrace::getregs(pid).ok();
398                            let rip = registers.as_ref().map_or(0, |registers| registers.rip);
399                            let captures_fault_context = matches!(
400                                signal,
401                                Signal::SIGSEGV
402                                    | Signal::SIGILL
403                                    | Signal::SIGBUS
404                                    | Signal::SIGFPE
405                                    | Signal::SIGABRT
406                            );
407                            if captures_fault_context {
408                                let siginfo = ptrace::getsiginfo(pid).ok();
409                                let si_code = siginfo.as_ref().map(|info| info.si_code);
410                                let existing =
411                                    self.signal_diagnostics.get(&pid).map(|diagnostic| {
412                                        (
413                                            diagnostic.signal,
414                                            diagnostic.si_code,
415                                            is_sabre_sigill_marker(
416                                                diagnostic.instruction_bytes.as_ref(),
417                                            ),
418                                        )
419                                    });
420                                if should_replace_signal_diagnostic(existing, signal, si_code) {
421                                    if let Some(registers) = registers {
422                                        let fault_address = siginfo
423                                            .as_ref()
424                                            .filter(|info| info.si_code > 0)
425                                            .map(|info| unsafe { info.si_addr() as usize });
426                                        let mapping = fs::read_to_string(format!(
427                                            "/proc/{}/maps",
428                                            pid.as_raw()
429                                        ))
430                                        .ok()
431                                        .and_then(|maps| mapping_diagnostic(&maps, rip as usize));
432                                        let instruction_bytes =
433                                            read_diagnostic_bytes(pid, rip as usize).ok();
434                                        self.signal_diagnostics.insert(
435                                            pid,
436                                            SignalDiagnostic {
437                                                signal,
438                                                si_code,
439                                                si_errno: siginfo
440                                                    .as_ref()
441                                                    .map(|info| info.si_errno),
442                                                fault_address,
443                                                mapping,
444                                                instruction_bytes,
445                                                registers,
446                                            },
447                                        );
448                                    } else {
449                                        self.signal_diagnostics.remove(&pid);
450                                    }
451                                }
452                            }
453                            tracing::debug!(
454                                target: "hermit::sabre::fallback",
455                                tid = pid.as_raw(),
456                                ?signal,
457                                rip = format!("{rip:#x}"),
458                                "forwarding signal to tracee",
459                            );
460                        }
461                        self.resume(pid, Some(signal))?;
462                    }
463                }
464                WaitStatus::Exited(..) | WaitStatus::Signaled(..) => unreachable!(),
465                WaitStatus::Continued(_) | WaitStatus::StillAlive => {}
466            }
467        }
468
469        let status = root_status.ok_or_else(|| anyhow!("SaBRe root tracee disappeared"))?;
470        let mut trusted_shared_objects = self
471            .trusted_shared_objects
472            .into_iter()
473            .map(|path| path.display().to_string())
474            .collect::<Vec<_>>();
475        trusted_shared_objects.sort();
476        Ok((
477            status,
478            PathEvidence {
479                schema: 1,
480                guest_rpc_observed: self.readiness.load(Ordering::Acquire),
481                ptrace_fallback_sites: self.patched_sites.len(),
482                trusted_shared_object_sites: self.trusted_shared_object_sites.len(),
483                trusted_shared_objects,
484            },
485        ))
486    }

502    fn handle_syscall_stop(&mut self, pid: Pid) -> Result<(), Error> {
503        let syscall_info = get_syscall_info(pid)?;
504        tracing::trace!(
505            target: "hermit::sabre::fallback",
506            tid = pid.as_raw(),
507            op = syscall_info.op,
508            "decoded ptrace syscall stop",
509        );
510        match syscall_info.op {
511            libc::PTRACE_SYSCALL_INFO_ENTRY => {
512                let mut regs = ptrace::getregs(pid)?;
513                let site = regs
514                    .rip
515                    .checked_sub(SYSCALL_INSN.len() as u64)
516                    .ok_or_else(|| anyhow!("invalid syscall RIP {:#x} in tracee {pid}", regs.rip))?
517                    as usize;
518                let bytes = read_two_bytes(pid, site)?;
519                let fallback_ready = self.fallback_ready()?;
520                if pid != self.root {
521                    tracing::trace!(
522                        target: "hermit::sabre::fallback",
523                        tid = pid.as_raw(),
524                        nr = regs.orig_rax,
525                        site = format!("{site:#x}"),
526                        raw = bytes == SYSCALL_INSN,
527                        "child syscall entry",
528                    );
529                }
530                if bytes == SYSCALL_INSN && fallback_ready {
531                    let mapping = self.classify_mapping(pid, site)?;
532                    if let Some(path) = mapping.trusted_shared_object {
533                        self.trusted_shared_object_sites.insert((pid, site));
534                        self.trusted_shared_objects.insert(path);
535                    }
536                    if !mapping.trusted {
537                        write_two_bytes(pid, site, SABRE_SYSCALL_MARKER)?;
538                        let syscall = regs.orig_rax;
539                        regs.orig_rax = u64::MAX;
540                        ptrace::setregs(pid, regs)?;
541                        self.states.entry(pid).or_default().pending_patch =
542                            Some(PendingPatch { site, syscall });
543                        self.patched_sites.insert((pid, site));
544                        tracing::debug!(
545                            target: "hermit::sabre::fallback",
546                            tid = pid.as_raw(),
547                            address = site,
548                            "redirecting raw syscall instruction through the SaBRe handler",
549                        );
550                    }
551                }
552            }
553            libc::PTRACE_SYSCALL_INFO_EXIT => {
554                // A cached verdict describes the mapping that occupied that page
555                // when it was classified. mmap/munmap/mremap/mprotect/brk can
556                // replace or re-permission that page in-process, so a page
557                // previously classified as trusted can come to host a
558                // completely different raw-syscall site. Drop the whole
559                // ADDRESS SPACE this tracee runs in whenever it mutates that
560                // space; the next site there is reclassified against live
561                // /proc/<pid>/maps. Whole-space is coarse but correct --
562                // per-page would be an optimisation, not a correctness
563                // requirement.
564                let exit_regs = ptrace::getregs(pid)?;
565                if mutates_address_space(exit_regs.orig_rax) {
566                    // Evict the whole ADDRESS SPACE, not just this thread's
567                    // entries: CLONE_VM siblings share one mm, so a sibling
568                    // would otherwise keep serving a pre-mutation verdict for
569                    // the very page this syscall just replaced.
570                    self.mapping_cache.invalidate_address_space(pid);
571                }
572                if let Some(pending) = self.states.entry(pid).or_default().pending_patch.take() {
573                    let mut regs = exit_regs;
574                    regs.rax = pending.syscall;
575                    regs.orig_rax = pending.syscall;
576                    regs.rip = pending.site as u64;
577                    ptrace::setregs(pid, regs)?;
578                }
579            }
580            _ => {}
581        }
582        ptrace::syscall(pid, None)?;
583        Ok(())
584    }

586    fn handle_ptrace_event(&mut self, pid: Pid, event: libc::c_int) -> Result<(), Error> {
587        if matches!(
588            event,
589            libc::PTRACE_EVENT_CLONE | libc::PTRACE_EVENT_FORK | libc::PTRACE_EVENT_VFORK
590        ) {
591            let child = Pid::from_raw(ptrace::getevent(pid)? as i32);
592            self.tracees.insert(child);
593            self.states.entry(child).or_default();
594            // Ask the kernel whether the child it just made shares this mm.
595            // Both tasks are alive at THIS stop and nowhere later, which is the
596            // only window in which the question can be answered by observation
597            // rather than predicted from the clone arguments.
598            let shares_mm = mm_sharing(pid, child);
599            self.mapping_cache.admit_child(pid, child, shares_mm);
600        } else if event == libc::PTRACE_EVENT_EXEC {
601            // exec installs a brand-new mm and tears down the old one.
602            self.mapping_cache.replace_address_space(pid);
603            self.states.insert(pid, TraceeState::default());
604            self.signal_diagnostics.remove(&pid);
605        }
606        self.resume(pid, None)
607    }

613    fn fallback_ready(&mut self) -> Result<bool, Error> {
614        let ready = self.readiness.load(Ordering::Acquire);
615        if ready && !self.ready_observed {

619                "SaBRe fallback readiness observed",

621        }

626        &mut self,
627        pid: Pid,
628        address: usize,

630        let page = address & !4095usize;

632            return Ok(classification.clone());
633        }

640            |entry| classify_mapping(&entry, self.sabre_id.as_ref(), self.plugin_id.as_ref()),
641        );
642        self.mapping_cache.insert(pid, page, classification.clone());
643        Ok(classification)
644    }

654// nix 0.30.1 passes a null `addr` to PTRACE_GET_SYSCALL_INFO, but Linux
655// defines that argument as the size of the output buffer. Use the kernel ABI
656// directly until the upstream wrapper supplies the required size.
657fn get_syscall_info(pid: Pid) -> Result<libc::ptrace_syscall_info, Error> {
658    let mut info = std::mem::MaybeUninit::<libc::ptrace_syscall_info>::zeroed();
659    let size = std::mem::size_of::<libc::ptrace_syscall_info>();
660    let written = unsafe {
661        libc::ptrace(
662            libc::PTRACE_GET_SYSCALL_INFO,
663            pid.as_raw(),
664            size as *mut libc::c_void,
665            info.as_mut_ptr().cast::<libc::c_void>(),
666        )
667    };
668    if written < 0 {
669        return Err(std::io::Error::last_os_error().into());
670    }
671    if written == 0 {
672        return Err(anyhow!(
673            "PTRACE_GET_SYSCALL_INFO returned no data for {pid}"
674        ));
675    }
676    Ok(unsafe { info.assume_init() })
677}

679fn mapping_line(maps: &str, address: usize) -> Option<&str> {
680    maps.lines().find_map(|line| {
681        let mut fields = line.split_whitespace();
682        let range = fields.next()?;
683        let mut limits = range.split('-');
684        let start = usize::from_str_radix(limits.next()?, 16).ok()?;
685        let end = usize::from_str_radix(limits.next()?, 16).ok()?;
686        (start <= address && address < end).then_some(line)
687    })
688}

690/// Identity of a mapped file: its `inode` plus the canonicalized path the
691/// object resolves to. This names the object itself rather than a path anyone
692/// can reproduce with the same basename.
693///
694/// `dev` is deliberately NOT part of the identity. It looks like the obvious
695/// second component and it is the one that does not survive the comparison:
696/// `stat(2)` reports the filesystem's `st_dev`, while `/proc/<pid>/maps` prints
697/// the SUPERBLOCK device, and on btrfs those differ because every subvolume
698/// gets its own anonymous `st_dev`. Measured on one file on a btrfs developer
699/// box: `stat()` gives `st_dev = 46` (rendered `00:2e`) while the maps row for
700/// the very same mapping of that very same inode prints `00:2d`. Comparing
701/// them made the SaBRe loader unrecognisable as infrastructure, so the
702/// supervisor patched a UD0 marker into the loader's own text and the guest
703/// died with SIGSEGV. `inode` is the discriminating component and it agrees;
704/// the canonicalized path is what rules out an inode collision across
705/// filesystems, which is the only way two distinct files can share one.
706#[derive(Clone, Debug, Eq, PartialEq)]
707struct FileId {
708    inode: u64,
709    path: PathBuf,
710}
711
712/// One parsed `/proc/<pid>/maps` row: `range perms offset dev inode path`.
713struct MappingEntry<'a> {
714    inode: Option<u64>,
715    path: &'a str,
716}
717
718impl MappingEntry<'_> {
719    /// `None` for anonymous or pseudo mappings (inode 0), which have no file
720    /// identity to bind to and therefore can never match a launched object.
721    fn file_id(&self) -> Option<FileId> {
722        match self.inode {
723            Some(inode) if inode != 0 => {
724                // The kernel already prints an absolute, resolved path here;
725                // canonicalizing anyway keeps both sides of the comparison in
726                // the same form. A path that no longer resolves (a deleted or
727                // replaced file) yields `None`, which fails CLOSED: no identity
728                // means no match and therefore no exemption.
729                let path = fs::canonicalize(Path::new(self.path)).ok()?;
730                Some(FileId { inode, path })
731            }
732            _ => None,
733        }
734    }
735}
736
737/// Identity of a path this supervisor launched. `None` when the file cannot be
738/// stat'ed, which fails CLOSED: an unknown identity matches no mapping, so the
739/// exemption is withheld rather than granted.
740fn launched_file_id(path: &Path) -> Option<FileId> {
741    use std::os::unix::fs::MetadataExt;
742    let metadata = fs::metadata(path).ok()?;
743    let canonical = fs::canonicalize(path).ok()?;
744    Some(FileId {
745        inode: metadata.ino(),
746        path: canonical,
747    })
748}

750fn mapping_entry(maps: &str, address: usize) -> Option<MappingEntry<'_>> {
751    let mut fields = mapping_line(maps, address)?.split_whitespace();
752    fields.next()?; // range
753    fields.next()?; // perms
754    fields.next()?; // offset
755    // dev: the superblock device column is parsed past but deliberately NOT
756    // stored — it is not part of file identity, because btrfs disagrees with
757    // stat's `st_dev` for the same inode (see the `FileId` doc above).
758    fields.next()?; // dev
759    let inode = fields.next()?;
760    Some(MappingEntry {
761        inode: inode.parse::<u64>().ok(),
762        path: fields.next().unwrap_or(""),
763    })
764}

766fn mapping_diagnostic(maps: &str, address: usize) -> Option<MappingDiagnostic> {
767    let line = mapping_line(maps, address)?;
768    let mut fields = line.split_whitespace();
769    let mut limits = fields.next()?.split('-');
770    let start = usize::from_str_radix(limits.next()?, 16).ok()?;
771    fields.next()?;
772    let mapping_offset = usize::from_str_radix(fields.next()?, 16).ok()?;
773    let relative_offset = address.checked_sub(start)?;
774    Some(MappingDiagnostic {
775        line: line.to_owned(),
776        relative_offset,
777        file_offset: mapping_offset.checked_add(relative_offset)?,
778    })
779}

781/// Syscalls that can replace, move, unmap or re-permission a page in the

800/// Whether two tasks share one address space.
801#[derive(Clone, Copy, Debug, Eq, PartialEq)]
802enum MmSharing {
803    Shared,
804    Private,
805}
806
807/// `kcmp(2)`'s "compare address spaces" selector. `KCMP_FILE` is 0 and
808/// `KCMP_VM` is 1; getting that backwards silently compares file descriptor 0,
809/// which two related tasks almost always share, so every answer comes back
810/// "shared".
811const KCMP_VM: libc::c_int = 1;
812
813/// Ask the KERNEL whether `child` shares `parent`'s address space.
814///
815/// This is an observation of the tasks that now exist, taken at the
816/// `PTRACE_EVENT_*` stop where both are alive. It deliberately replaces the
817/// earlier approach of predicting the answer from clone arguments:
818///
819///   * `clone(2)` carries its flags in a register, which a sibling cannot
820///     change while the caller is stopped -- that read was sound, and
821///   * `clone3(2)` carries them in a USER BUFFER that the kernel copies AFTER
822///     the syscall-entry stop. Sibling threads stay runnable across that stop,
823///     so the buffer can change in the window and the prediction can be wrong
824///     in either direction. Measured with a tracer that rewrites the buffer
825///     after the entry read: pre-copy flags said SHARED where the child was
826///     private, and said private where the child was SHARED.
827///
828/// Rather than keep two rules with different trust properties, every clone
829/// shape is now answered the same way, by the kernel, after the fact.
830///
831/// `kcmp` needs both tasks alive: called after either exits it fails with
832/// `ESRCH`. Any failure -- including a kernel built without
833/// `CONFIG_CHECKPOINT_RESTORE`, where the syscall is `ENOSYS` -- falls back to
834/// `Shared`, which over-evicts. Over-eviction costs one re-read of
835/// `/proc/<pid>/maps`; under-eviction leaves a stale trusted verdict that a raw
836/// syscall can hide behind.
837fn mm_sharing(parent: Pid, child: Pid) -> MmSharing {
838    // SAFETY: kcmp takes only scalars and returns a scalar.
839    let result = unsafe {
840        libc::syscall(
841            libc::SYS_kcmp,
842            parent.as_raw() as libc::c_long,
843            child.as_raw() as libc::c_long,
844            KCMP_VM as libc::c_long,
845            0 as libc::c_long,
846            0 as libc::c_long,
847        )
848    };
849    interpret_kcmp_vm(result)
850}
851
852/// `kcmp` returns 0 when the two address spaces are the SAME, an ordering
853/// value (1/2/3) when they differ, and -1 on error. Split out so the mapping --
854/// including the fail-safe on error -- is testable without spawning tasks.
855fn interpret_kcmp_vm(result: libc::c_long) -> MmSharing {
856    match result {
857        0 => MmSharing::Shared,
858        r if r > 0 => MmSharing::Private,
859        // Errors fail SAFE: assume sharing rather than risk a stale verdict.
860        _ => MmSharing::Shared,
861    }
862}

864/// Kernel-supplied mappings whose raw syscall sites are causally identified as
865/// infrastructure the guest cannot have rewritten. `[vdso]` and `[vsyscall]`
866/// are mapped by the kernel and are not writable, so SaBRe cannot expand a
867/// syscall site inside them and the supervisor must not try to patch one.
868///
869/// They are exempt from REDIRECTION only — never from ACCOUNTING. A raw syscall
870/// executed here did not traverse the measured in-guest SaBRe handler, so it is
871/// still recorded as a trusted-native site and still makes the cell ineligible.
872/// Exempting a mapping from both redirection and counting is the one
873/// combination that makes a real raw syscall disappear from the evidence.
874const CAUSAL_KERNEL_MAPPINGS: [&str; 2] = ["[vdso]", "[vsyscall]"];
875
876fn classify_mapping(
877    entry: &MappingEntry<'_>,
878    sabre: Option<&FileId>,
879    plugin: Option<&FileId>,
880) -> MappingClassification {
881    let path = entry.path;
882    if path.starts_with('[') {
883        // Only the causally identified kernel mappings above are exempt from
884        // redirection, and even those are counted. Every other bracket-named
885        // mapping -- `[stack]`, `[heap]`, or any future kernel name -- is NOT
886        // infrastructure: fall through to `trusted: false` so the site is
887        // redirected through the SaBRe handler and counted as a fallback site.
888        if CAUSAL_KERNEL_MAPPINGS.contains(&path) {
889            return MappingClassification {
890                trusted: true,
891                trusted_shared_object: Some(PathBuf::from(path)),
892            };
893        }
894        return MappingClassification {
895            trusted: false,
896            trusted_shared_object: None,
897        };
898    }
899    let path = Path::new(path.strip_suffix(" (deleted)").unwrap_or(path));
900    // Bind the infrastructure exemption to the identity of the objects this
901    // supervisor actually launched (inode plus canonical path, both resolved
902    // from the same `/proc/<pid>/maps` line), not to their basenames. A
903    // basename match would
904    // exempt any unrelated mapping that happens to be called `sabre` or
905    // `libdetcore_sabre.so`.
906    let mapped = entry.file_id();
907    let infrastructure = matches!((mapped.as_ref(), sabre), (Some(seen), Some(known)) if seen == known)
908        || matches!((mapped.as_ref(), plugin), (Some(seen), Some(known)) if seen == known);
909    if infrastructure {
910        return MappingClassification {
911            trusted: true,
912            trusted_shared_object: None,
913        };
914    }
915    // SaBRe owns shared-library rewriting. A raw syscall that still reaches the
916    // supervisor from another shared object ran outside the measured in-guest
917    // interception path. Keep trusting it for runtime safety, but report it so
918    // compatibility accounting cannot credit the cell to SaBRe.
919    let trusted_shared_object = path
920        .file_name()
921        .is_some_and(|name| name.to_string_lossy().contains(".so"))
922        .then(|| path.to_path_buf());
923    MappingClassification {
924        trusted: trusted_shared_object.is_some(),
925        trusted_shared_object,
926    }
927}

929fn read_two_bytes(pid: Pid, address: usize) -> Result<[u8; 2], Error> {
930    let word_size = std::mem::size_of::<libc::c_long>();
931    let aligned = address & !(word_size - 1);
932    let offset = address - aligned;
933    let first = ptrace::read(pid, aligned as ptrace::AddressType)?.to_ne_bytes();
934    if offset + 1 < word_size {
935        Ok([first[offset], first[offset + 1]])
936    } else {
937        let second = ptrace::read(pid, (aligned + word_size) as ptrace::AddressType)?.to_ne_bytes();
938        Ok([first[offset], second[0]])
939    }
940}

942fn read_diagnostic_bytes(pid: Pid, address: usize) -> Result<[u8; 16], Error> {

947        chunk.copy_from_slice(&word.to_ne_bytes()[..chunk.len()]);
948    }

952fn write_two_bytes(pid: Pid, address: usize, bytes: [u8; 2]) -> Result<(), Error> {
953    let word_size = std::mem::size_of::<libc::c_long>();
954    let aligned = address & !(word_size - 1);
955    let offset = address - aligned;
956    let mut first = ptrace::read(pid, aligned as ptrace::AddressType)?.to_ne_bytes();
957    first[offset] = bytes[0];
958    if offset + 1 < word_size {
959        first[offset + 1] = bytes[1];
960        ptrace::write(
961            pid,
962            aligned as ptrace::AddressType,
963            libc::c_long::from_ne_bytes(first),
964        )?;
965    } else {
966        ptrace::write(
967            pid,
968            aligned as ptrace::AddressType,
969            libc::c_long::from_ne_bytes(first),
970        )?;
971        let second_address = aligned + word_size;
972        let mut second = ptrace::read(pid, second_address as ptrace::AddressType)?.to_ne_bytes();
973        second[0] = bytes[1];
974        ptrace::write(
975            pid,
976            second_address as ptrace::AddressType,
977            libc::c_long::from_ne_bytes(second),
978        )?;
979    }
980    Ok(())
981}

983pub async fn run(
984    mut command: std::process::Command,
985    sabre: PathBuf,
986    plugin: PathBuf,
987    readiness: Arc<AtomicBool>,
988    physical_exit_observer: Arc<detcore::GlobalState>,
989    capture_output: bool,
990) -> Result<Output, Error> {
991    if capture_output {
992        command.stdout(Stdio::piped()).stderr(Stdio::piped());
993    }
994    // Spawn before creating the blocking supervisor worker. A worker thread consumes a task ID
995    // in the guest PID namespace; creating it first shifts the root guest from PID 3 to PID 4 and
996    // makes otherwise identical ptrace and SaBRe programs observe different process identities.
997    let child = spawn_tracee(command)?;
998    let root = Pid::from_raw(child.id() as i32);
999    match waitpid(root, Some(WaitPidFlag::__WALL))? {
1000        WaitStatus::Stopped(pid, Signal::SIGTRAP) if pid == root => {}
1001        status => return Err(anyhow!("unexpected initial SaBRe ptrace stop: {status:?}")),
1002    }
1003    // Ptrace ownership belongs to the individual tracer task, not its thread group. Leave the
1004    // tracee stopped while handing ownership from this async caller to the blocking supervisor.
1005    // Injecting SIGSTOP as part of detach prevents any guest instruction from running in between.
1006    ptrace::detach(root, Some(Signal::SIGSTOP))
1007        .context("failed to hand SaBRe tracee to supervisor worker")?;
1008    tokio::task::spawn_blocking(move || {
1009        run_blocking(child, sabre, plugin, readiness, physical_exit_observer)
1010    })
1011    .await
1012    .context("SaBRe ptrace supervisor task panicked")?
1013}

1015fn spawn_tracee(mut command: std::process::Command) -> Result<std::process::Child, Error> {
1016    // TODO-HUMAN-REVIEW(PR-845): Review SaBRe launch-time ASLR disabling.
1017    // PTRACE_TRACEME makes exec stop with SIGTRAP. A pre-exec SIGSTOP would
1018    // deadlock std::process::Command on its exec error pipe. personality(2)
1019    // is async-signal-safe and survives the SaBRe and guest execs.
1020    unsafe {
1021        command.pre_exec(|| {
1022            let current = libc::personality(0xffff_ffff);
1023            if current == -1 {
1024                return Err(std::io::Error::last_os_error());
1025            }
1026            let deterministic = current as libc::c_ulong | libc::ADDR_NO_RANDOMIZE as libc::c_ulong;
1027            if libc::personality(deterministic) == -1 {
1028                return Err(std::io::Error::last_os_error());
1029            }
1030            ptrace::traceme().map_err(std::io::Error::from)
1031        });
1032    }
1033
1034    command
1035        .spawn()
1036        .context("failed to spawn ptraced SaBRe guest")
1037}

1039fn run_blocking(
1040    mut child: std::process::Child,
1041    sabre: PathBuf,
1042    plugin: PathBuf,
1043    readiness: Arc<AtomicBool>,
1044    physical_exit_observer: Arc<detcore::GlobalState>,
1045) -> Result<Output, Error> {
1046    let root = Pid::from_raw(child.id() as i32);
1047    let stdout = child.stdout.take();
1048    let stderr = child.stderr.take();
1049    drop(child);
1050
1051    let stdout_thread = std::thread::spawn(move || read_pipe(stdout));
1052    let stderr_thread = std::thread::spawn(move || read_pipe(stderr));
1053    let supervised = Supervisor::new(root, sabre, plugin, readiness, physical_exit_observer).run();
1054    if supervised.is_err() {
1055        let _ = nix::sys::signal::kill(root, Signal::SIGKILL);
1056    }
1057    let stdout = stdout_thread
1058        .join()
1059        .map_err(|_| anyhow!("SaBRe stdout reader panicked"))??;
1060    let stderr = stderr_thread
1061        .join()
1062        .map_err(|_| anyhow!("SaBRe stderr reader panicked"))??;
1063    let (status, path_evidence) = supervised?;
1064    Ok(Output {
1065        status,
1066        stdout,
1067        stderr,
1068        path_evidence,
1069    })
1070}

1534    /// Fork a child of the requested shape, run `check` while it is alive, then
1535    /// reap it. Real tasks, because the property under test is what the KERNEL
1536    /// did -- a hand-built pid pair would test nothing.
1537    fn with_child<F: FnOnce(Pid, Pid)>(share_mm: bool, check: F) {
1538        const STACK: usize = 1 << 20;
1539        let mut stack = vec![0u8; STACK];
1540        let child = if share_mm {
1541            // CLONE_VM|SIGCHLD: shares the mm AND is reported as
1542            // PTRACE_EVENT_FORK, the shape that defeats an event-kind rule.
1543            let top = unsafe { stack.as_mut_ptr().add(STACK) } as *mut libc::c_void;
1544            extern "C" fn sleeper(_: *mut libc::c_void) -> libc::c_int {
1545                unsafe { libc::pause() };
1546                0
1547            }
1548            unsafe {
1549                libc::clone(
1550                    sleeper,
1551                    top,
1552                    libc::CLONE_VM | libc::SIGCHLD,
1553                    std::ptr::null_mut(),
1554                )
1555            }
1556        } else {
1557            match unsafe { libc::fork() } {
1558                0 => unsafe {
1559                    libc::pause();
1560                    libc::_exit(0)
1561                },
1562                pid => pid,
1563            }
1564        };
1565        assert!(child > 0, "failed to create the {share_mm} child");
1566        let parent = Pid::from_raw(unsafe { libc::getpid() });
1567        let result = std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| {
1568            check(parent, Pid::from_raw(child))
1569        }));
1570        unsafe {
1571            libc::kill(child, libc::SIGKILL);
1572            libc::waitpid(child, std::ptr::null_mut(), libc::__WALL);
1573        }
1574        if let Err(payload) = result {
1575            std::panic::resume_unwind(payload);
1576        }
1577    }
