9//! Expensive one-time host qualifications, cached in the user's cache
10//! directory (`$XDG_CACHE_HOME/hermit/`, by default `~/.cache/hermit/`), so
11//! that a later run reuses the verdict instead of measuring the host again
12//! (<https://github.com/rrnewton/hermit/issues/4515>).
13//!
14//! Each qualification is one file holding one verdict and the key it was
15//! measured under: the kernel release, the boot, the CPU (vendor, family,
16//! model, stepping, microcode), the Hermit binary that measured it, this
17//! cache's schema and the qualification's own inputs. A reboot, a kernel or
18//! CPU change, a microcode update, a rebuilt or reinstalled Hermit, or new
19//! inputs miss and measure again. Only verdicts that depend on the host alone
20//! belong here: nothing in the key can tell a chroot or container from the
21//! host, so a check whose answer depends on the caller's sandbox (namespaces,
22//! ptrace, seccomp) must run every time instead. Each qualification decides
23//! which of its verdicts are stable facts of the key and so may be stored;
24//! a verdict host load or timing can change is never one of them. The cache
25//! is best effort: a missing, unreadable, oversized,
26//! non-regular, corrupt or unwritable file is a miss, never a failure.
27

39/// Bumped whenever what a cached verdict means changes, which discards every
40/// older file.
41const SCHEMA: u32 = 1;
42
43/// The largest cache file read; a larger one is a miss.
44const MAX_FILE_BYTES: u64 = 1 << 20;

46/// The host facts a cached verdict holds for.
47#[derive(Clone, Debug, Default, PartialEq, Eq, Serialize, Deserialize)]
48pub(crate) struct HostKey {
49    pub(crate) kernel_release: String,
50    pub(crate) boot_id: String,
51    pub(crate) cpu: String,
52    pub(crate) microcode: String,
53    /// The running Hermit binary's GNU build ID (or, without one, its
54    /// device, inode, size and modification time): a qualification's own
55    /// code is part of what it certifies, so a verdict measured by another
56    /// build is never reused, even one copied in place with its mtime kept.
57    pub(crate) hermit_binary: String,
58}

60fn read_trimmed(path: &str) -> String {
61    fs::read_to_string(path)
62        .map(|text| text.trim().to_owned())
63        .unwrap_or_default()

66impl HostKey {
67    /// This host's key: `uname`, the boot id, CPUID and the microcode
68    /// revision, a few constant-time reads.
69    pub(crate) fn current() -> Self {
70        Self {
71            kernel_release: nix::sys::utsname::uname()
72                .map(|uts| uts.release().to_string_lossy().into_owned())

82/// The running executable's GNU build ID (`NT_GNU_BUILD_ID`), read from its
83/// own `PT_NOTE` segments in memory, as lowercase hex.
84fn build_id() -> Option<String> {
85    unsafe extern "C" fn first_object(
86        info: *mut libc::dl_phdr_info,
87        _size: usize,
88        data: *mut libc::c_void,
89    ) -> libc::c_int {
90        // SAFETY: dl_iterate_phdr passes a valid info for the object it
91        // reports, and `data` is the `Option<String>` below. The first object
92        // it reports is the executable; its notes are mapped where its
93        // program headers say.
94        unsafe {
95            let info = &*info;
96            let found = &mut *(data as *mut Option<String>);
97            let headers = std::slice::from_raw_parts(info.dlpi_phdr, usize::from(info.dlpi_phnum));
98            for header in headers
99                .iter()
100                .filter(|header| header.p_type == libc::PT_NOTE)
101            {
102                let start = (info.dlpi_addr as usize + header.p_vaddr as usize) as *const u8;
103                let notes = std::slice::from_raw_parts(start, header.p_memsz as usize);
104                if let Some(id) = gnu_build_id(notes) {
105                    *found = Some(id);
106                    break;
107                }
108            }
109        }
110        1
111    }
112    let mut found: Option<String> = None;
113    // SAFETY: the callback only reads the reported object's mapped notes.
114    unsafe {
115        libc::dl_iterate_phdr(
116            Some(first_object),
117            (&mut found as *mut Option<String>).cast::<libc::c_void>(),
118        );
119    }
120    found
121}

123/// The `NT_GNU_BUILD_ID` descriptor among ELF `notes`, as lowercase hex.
124fn gnu_build_id(notes: &[u8]) -> Option<String> {
125    let word = |at: usize| -> Option<usize> {
126        Some(u32::from_ne_bytes(notes.get(at..at + 4)?.try_into().ok()?) as usize)
127    };
128    let mut at = 0;
129    while at + 12 <= notes.len() {
130        let (name_size, desc_size, kind) = (word(at)?, word(at + 4)?, word(at + 8)?);
131        let name = at + 12;
132        let desc = name + name_size.next_multiple_of(4);
133        let next = desc + desc_size.next_multiple_of(4);
134        if next > notes.len() {
135            return None;
136        }
137        if kind == 3 && notes.get(name..name + name_size)? == b"GNU\0" {
138            let id = &notes[desc..desc + desc_size];
139            return Some(id.iter().map(|byte| format!("{byte:02x}")).collect());
140        }
141        at = next;
142    }
143    None
144}

146/// `/proc/self/exe`'s device, inode, size and modification time.
147fn binary_identity() -> String {
148    use std::os::unix::fs::MetadataExt;
149    fs::metadata("/proc/self/exe")
150        .map(|m| {
151            format!(
152                "{}:{}:{}:{}.{}",
153                m.dev(),
154                m.ino(),
155                m.size(),
156                m.mtime(),
157                m.mtime_nsec()
158            )
159        })
160        .unwrap_or_default()
161}

163/// The CPU's vendor, family, model and stepping, from CPUID leaves 0 and 1.
164#[cfg(target_arch = "x86_64")]
165fn cpu_signature() -> String {
166    // CPUID leaves 0 and 1 exist on every x86_64 CPU.
167    let (leaf0, leaf1) = (std::arch::x86_64::__cpuid(0), std::arch::x86_64::__cpuid(1));
168    signature_of([leaf0.ebx, leaf0.edx, leaf0.ecx], leaf1.eax)
169}
170
171/// The vendor, family, model and stepping that CPUID leaf 0's vendor words
172/// and leaf 1's EAX name, as Linux's `/proc/cpuinfo` decodes them.
173fn signature_of(vendor_words: [u32; 3], eax: u32) -> String {
174    let vendor: Vec<u8> = vendor_words
175        .iter()
176        .flat_map(|word| word.to_le_bytes())
177        .collect();
178    let base_family = (eax >> 8) & 0xf;
179    let family = if base_family == 0xf {
180        base_family + ((eax >> 20) & 0xff)
181    } else {
182        base_family
183    };
184    let model = if base_family == 0x6 || base_family == 0xf {
185        ((eax >> 12) & 0xf0) | ((eax >> 4) & 0xf)
186    } else {
187        (eax >> 4) & 0xf
188    };
189    format!(
190        "{} family {family:#x} model {model:#x} stepping {:#x}",
191        String::from_utf8_lossy(&vendor),
192        eax & 0xf
193    )
194}

197fn cpu_signature() -> String {
198    String::new()

211/// Where a qualification named `name` is cached, or `None` without a home
212/// directory. The file name carries [`binary_suffix`], so two Hermit
213/// builds sharing one cache directory keep separate verdicts instead of
214/// overwriting each other's on every run (issue 452); the
215/// host key inside the file still binds the verdict to the binary that
216/// measured it.
217pub(crate) fn default_path(name: &str) -> Option<PathBuf> {
218    dirs::cache_dir().map(|dir| {
219        dir.join("hermit")
220            .join(format!("{name}-{}.json", binary_suffix()))
221    })
222}
223
224/// A short filename-safe token naming the running Hermit binary: the
225/// first 16 hex digits of its GNU build ID, or, without one, its file
226/// identity ([`binary_identity`]) with every non-alphanumeric byte
227/// replaced by `-`, truncated. Stable across processes of one binary;
228/// distinct binaries get distinct tokens with overwhelming likelihood,
229/// and a collision only costs a cache miss, because the entry's host
230/// key still has to match exactly ([`load`]).
231pub(crate) fn binary_suffix() -> &'static str {
232    static SUFFIX: std::sync::OnceLock<String> = std::sync::OnceLock::new();
233    SUFFIX.get_or_init(|| match build_id() {
234        Some(id) => id.chars().take(16).collect(),
235        None => binary_identity()
236            .chars()
237            .map(|c| if c.is_ascii_alphanumeric() { c } else { '-' })
238            .take(32)
239            .collect(),
240    })
241}

243/// The directory holding the probe's per-user coordination files (its
244/// lock and shared verdict): the user's runtime directory
245/// (`$XDG_RUNTIME_DIR`) where there is a usable one, else the
246/// temporary directory. The runtime directory is the same for every
247/// process of one user on one host, where `TMPDIR` is routinely set
248/// per test or per sandbox; coordination files under a per-test
249/// `TMPDIR` serialize nothing, and concurrent probes then switch
250/// each other out until they all refuse. A runtime directory that
251/// is missing, not a directory, owned by another user, or not
252/// writable is not usable: sandboxes routinely forward a stale
253/// `XDG_RUNTIME_DIR` naming no such directory, and coordination
254/// files there would fail on every run, so those fall back too.
255pub(crate) fn coordination_dir() -> PathBuf {
256    coordination_dir_from(dirs::runtime_dir(), nix::unistd::geteuid().as_raw())
257}
258
259/// [`coordination_dir`] with the runtime-directory candidate and the
260/// effective user named, for tests.
261fn coordination_dir_from(runtime_dir: Option<PathBuf>, euid: u32) -> PathBuf {
262    match runtime_dir {
263        Some(dir) if runtime_dir_usable(&dir, euid) => dir,
264        _ => std::env::temp_dir(),
265    }
266}
267
268/// Whether `dir` can hold this user's coordination files: it exists,
269/// is a directory, is owned by `euid`, and this process may create
270/// files in it (write plus search permission).
271fn runtime_dir_usable(dir: &Path, euid: u32) -> bool {
272    use std::os::unix::fs::MetadataExt;
273    let Ok(metadata) = fs::metadata(dir) else {
274        return false;
275    };
276    metadata.is_dir()
277        && metadata.uid() == euid
278        && nix::unistd::access(
279            dir,
280            nix::unistd::AccessFlags::W_OK | nix::unistd::AccessFlags::X_OK,
281        )
282        .is_ok()
283}

285/// The verdict cached at `path` for `host` and `inputs`, if there is one.
286pub(crate) fn load<T: DeserializeOwned>(path: &Path, host: &HostKey, inputs: &str) -> Option<T> {
287    let bytes = read_bounded(path)?;
288    let entry: Entry<T> = serde_json::from_slice(&bytes).ok()?;
289    (entry.schema == SCHEMA && entry.host == *host && entry.inputs == inputs)
290        .then_some(entry.verdict)
291}

293/// Caches `verdict` at `path` for `host` and `inputs`, by an atomic rename of
294/// a file in the same directory, and says whether it did. A failure leaves
295/// the cache as it was.
296pub(crate) fn store<T: Serialize>(path: &Path, host: &HostKey, inputs: &str, verdict: &T) -> bool {
297    let entry = Entry {
298        schema: SCHEMA,
299        host: host.clone(),
300        inputs: inputs.to_owned(),
301        verdict,
302    };
303    write_atomically(path, &entry).is_ok()
304}

306/// Whether a verdict could be stored at `path`: its directory exists, or can
307/// be made, and this process may write it.
308pub(crate) fn writable(path: &Path) -> bool {
309    path.parent().is_some_and(|directory| {
310        fs::create_dir_all(directory).is_ok()
311            && nix::unistd::access(directory, nix::unistd::AccessFlags::W_OK).is_ok()
312    })
313}

315/// Runs `f` holding an exclusive `flock` on `<path>.lock`, so that processes
316/// that miss at once measure one at a time and the later ones can find the
317/// first one's verdict. Without a lock file that is a regular file `f` runs
318/// unlocked; the open neither blocks on a FIFO nor follows a link.
319pub(crate) fn locked<R>(path: &Path, f: impl FnOnce() -> R) -> R {
320    use std::os::fd::AsRawFd;
321    let lock = path.parent().and_then(|directory| {
322        fs::create_dir_all(directory).ok()?;
323        let file = fs::OpenOptions::new()
324            .create(true)
325            .truncate(false)
326            .write(true)
327            .custom_flags(libc::O_NOFOLLOW | libc::O_NONBLOCK)
328            .open(path.with_extension("lock"))
329            .ok()?;
330        if !file.metadata().ok()?.is_file() {
331            return None;
332        }
333        // SAFETY: the descriptor is open for the duration of the call.
334        (unsafe { libc::flock(file.as_raw_fd(), libc::LOCK_EX) } == 0).then_some(file)
335    });
336    let result = f();
337    drop(lock);
338    result
339}

341/// Reads `path` if it is a regular file of at most [`MAX_FILE_BYTES`] that
342/// this process's user owns. The open neither blocks on a FIFO nor follows a
343/// symbolic link.
344fn read_bounded(path: &Path) -> Option<Vec<u8>> {
345    read_bounded_as(path, nix::unistd::geteuid().as_raw())
346}
347
348/// [`read_bounded`] for a file that user `owner` must own.
349fn read_bounded_as(path: &Path, owner: u32) -> Option<Vec<u8>> {
350    use std::os::unix::fs::MetadataExt;
351    let file = fs::OpenOptions::new()
352        .read(true)
353        .custom_flags(libc::O_NONBLOCK | libc::O_NOFOLLOW)
354        .open(path)
355        .ok()?;
356    let metadata = file.metadata().ok()?;
357    if !metadata.is_file() || metadata.len() > MAX_FILE_BYTES || metadata.uid() != owner {
358        return None;
359    }
360    // A file that grows after the check is cut at the cap, so it does not
361    // parse and is a miss.
362    let mut bytes = Vec::new();
363    file.take(MAX_FILE_BYTES).read_to_end(&mut bytes).ok()?;
364    Some(bytes)
365}
366
367fn write_atomically<T: Serialize>(path: &Path, entry: &T) -> std::io::Result<()> {
368    let directory = path
369        .parent()
370        .ok_or_else(|| std::io::Error::other("the cache path has no directory"))?;
371    fs::create_dir_all(directory)?;
372    let mut temporary = tempfile::NamedTempFile::new_in(directory)?;
373    temporary.write_all(&serde_json::to_vec_pretty(entry).map_err(std::io::Error::other)?)?;
374    temporary.persist(path).map_err(|error| error.error)?;
375    Ok(())
376}

621    /// A usable runtime directory is the coordination directory — this
622    /// is what serializes probes across per-test `TMPDIR` isolation,
623    /// so it must fail if `coordination_dir` ever reverts to the
624    /// temporary directory unconditionally. A missing, non-directory
625    /// or other-owned candidate, and no candidate at all, fall back
626    /// to the temporary directory instead of failing every run.
627    #[test]
628    fn a_usable_runtime_dir_is_the_coordination_dir_and_an_unusable_one_falls_back() {
629        let me = nix::unistd::geteuid().as_raw();
630        let directory = tempfile::tempdir().unwrap();
631        assert_eq!(
632            coordination_dir_from(Some(directory.path().to_owned()), me),
633            directory.path(),
634            "a usable runtime directory must be chosen over temp_dir()"
635        );
636        let temp = std::env::temp_dir();
637        assert_eq!(coordination_dir_from(None, me), temp);
638        assert_eq!(
639            coordination_dir_from(Some(directory.path().join("missing")), me),
640            temp,
641            "a missing runtime directory falls back"
642        );
643        let file = directory.path().join("a-file");
644        fs::write(&file, b"not a directory").unwrap();
645        assert_eq!(
646            coordination_dir_from(Some(file), me),
647            temp,
648            "a non-directory runtime path falls back"
649        );
650        assert_eq!(
651            coordination_dir_from(Some(directory.path().to_owned()), me.wrapping_add(1)),
652            temp,
653            "a runtime directory owned by another user falls back"
654        );
655    }
