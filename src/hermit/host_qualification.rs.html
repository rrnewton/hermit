9//! Expensive one-time host qualifications, cached in the user's cache
10//! directory (`$XDG_CACHE_HOME/hermit/`, by default `~/.cache/hermit/`), so
11//! that a later run reuses the verdict instead of measuring the host again
12//! (<https://github.com/rrnewton/hermit/issues/4515>).
13//!
14//! Each qualification is one file holding one verdict and the key it was
15//! measured under: the kernel release, the boot, the CPU (vendor, family,
16//! model, stepping, microcode), the Hermit binary that measured it, this
17//! cache's schema and the qualification's own inputs. A reboot, a kernel or
18//! CPU change, a microcode update, a rebuilt or reinstalled Hermit, or new
19//! inputs miss and measure again. Only verdicts that depend on the host alone
20//! belong here: nothing in the key can tell a chroot or container from the
21//! host, so a check whose answer depends on the caller's sandbox (namespaces,
22//! ptrace, seccomp) must run every time instead. Only a passing verdict is
23//! stored. The cache is best effort: a missing, unreadable, oversized,
24//! non-regular, corrupt or unwritable file is a miss, never a failure.
25

37/// Bumped whenever what a cached verdict means changes, which discards every
38/// older file.
39const SCHEMA: u32 = 1;
40
41/// The largest cache file read; a larger one is a miss.
42const MAX_FILE_BYTES: u64 = 1 << 20;
43
44/// The host facts a cached verdict holds for.
45#[derive(Clone, Debug, Default, PartialEq, Eq, Serialize, Deserialize)]
46pub(crate) struct HostKey {
47    pub(crate) kernel_release: String,
48    pub(crate) boot_id: String,
49    pub(crate) cpu: String,
50    pub(crate) microcode: String,
51    /// The running Hermit binary's GNU build ID (or, without one, its
52    /// device, inode, size and modification time): a qualification's own
53    /// code is part of what it certifies, so a verdict measured by another
54    /// build is never reused, even one copied in place with its mtime kept.
55    pub(crate) hermit_binary: String,
56}

58fn read_trimmed(path: &str) -> String {
59    fs::read_to_string(path)
60        .map(|text| text.trim().to_owned())
61        .unwrap_or_default()
62}

64impl HostKey {
65    /// This host's key: `uname`, the boot id, CPUID and the microcode
66    /// revision, a few constant-time reads.
67    pub(crate) fn current() -> Self {
68        Self {
69            kernel_release: nix::sys::utsname::uname()
70                .map(|uts| uts.release().to_string_lossy().into_owned())
71                .unwrap_or_default(),
72            boot_id: read_trimmed("/proc/sys/kernel/random/boot_id"),
73            cpu: cpu_signature(),
74            microcode: read_trimmed("/sys/devices/system/cpu/cpu0/microcode/version"),
75            hermit_binary: build_id().unwrap_or_else(binary_identity),
76        }
77    }
78}

80/// The running executable's GNU build ID (`NT_GNU_BUILD_ID`), read from its
81/// own `PT_NOTE` segments in memory, as lowercase hex.
82fn build_id() -> Option<String> {
83    unsafe extern "C" fn first_object(
84        info: *mut libc::dl_phdr_info,
85        _size: usize,
86        data: *mut libc::c_void,
87    ) -> libc::c_int {
88        // SAFETY: dl_iterate_phdr passes a valid info for the object it
89        // reports, and `data` is the `Option<String>` below. The first object
90        // it reports is the executable; its notes are mapped where its
91        // program headers say.
92        unsafe {
93            let info = &*info;
94            let found = &mut *(data as *mut Option<String>);
95            let headers = std::slice::from_raw_parts(info.dlpi_phdr, usize::from(info.dlpi_phnum));
96            for header in headers
97                .iter()
98                .filter(|header| header.p_type == libc::PT_NOTE)
99            {
100                let start = (info.dlpi_addr as usize + header.p_vaddr as usize) as *const u8;
101                let notes = std::slice::from_raw_parts(start, header.p_memsz as usize);
102                if let Some(id) = gnu_build_id(notes) {
103                    *found = Some(id);
104                    break;
105                }
106            }
107        }
108        1
109    }
110    let mut found: Option<String> = None;
111    // SAFETY: the callback only reads the reported object's mapped notes.
112    unsafe {
113        libc::dl_iterate_phdr(
114            Some(first_object),
115            (&mut found as *mut Option<String>).cast::<libc::c_void>(),
116        );
117    }
118    found
119}

121/// The `NT_GNU_BUILD_ID` descriptor among ELF `notes`, as lowercase hex.
122fn gnu_build_id(notes: &[u8]) -> Option<String> {
123    let word = |at: usize| -> Option<usize> {
124        Some(u32::from_ne_bytes(notes.get(at..at + 4)?.try_into().ok()?) as usize)
125    };
126    let mut at = 0;
127    while at + 12 <= notes.len() {
128        let (name_size, desc_size, kind) = (word(at)?, word(at + 4)?, word(at + 8)?);
129        let name = at + 12;
130        let desc = name + name_size.next_multiple_of(4);
131        let next = desc + desc_size.next_multiple_of(4);
132        if next > notes.len() {
133            return None;
134        }
135        if kind == 3 && notes.get(name..name + name_size)? == b"GNU\0" {
136            let id = &notes[desc..desc + desc_size];
137            return Some(id.iter().map(|byte| format!("{byte:02x}")).collect());
138        }
139        at = next;
140    }
141    None
142}

144/// `/proc/self/exe`'s device, inode, size and modification time.
145fn binary_identity() -> String {
146    use std::os::unix::fs::MetadataExt;
147    fs::metadata("/proc/self/exe")
148        .map(|m| {
149            format!(
150                "{}:{}:{}:{}.{}",
151                m.dev(),
152                m.ino(),
153                m.size(),
154                m.mtime(),
155                m.mtime_nsec()
156            )
157        })
158        .unwrap_or_default()
159}

161/// The CPU's vendor, family, model and stepping, from CPUID leaves 0 and 1.
162#[cfg(target_arch = "x86_64")]
163fn cpu_signature() -> String {
164    // CPUID leaves 0 and 1 exist on every x86_64 CPU.
165    let (leaf0, leaf1) = (std::arch::x86_64::__cpuid(0), std::arch::x86_64::__cpuid(1));
166    signature_of([leaf0.ebx, leaf0.edx, leaf0.ecx], leaf1.eax)
167}
168
169/// The vendor, family, model and stepping that CPUID leaf 0's vendor words
170/// and leaf 1's EAX name, as Linux's `/proc/cpuinfo` decodes them.
171fn signature_of(vendor_words: [u32; 3], eax: u32) -> String {
172    let vendor: Vec<u8> = vendor_words
173        .iter()
174        .flat_map(|word| word.to_le_bytes())
175        .collect();
176    let base_family = (eax >> 8) & 0xf;
177    let family = if base_family == 0xf {
178        base_family + ((eax >> 20) & 0xff)
179    } else {
180        base_family
181    };
182    let model = if base_family == 0x6 || base_family == 0xf {
183        ((eax >> 12) & 0xf0) | ((eax >> 4) & 0xf)
184    } else {
185        (eax >> 4) & 0xf
186    };
187    format!(
188        "{} family {family:#x} model {model:#x} stepping {:#x}",
189        String::from_utf8_lossy(&vendor),
190        eax & 0xf
191    )
192}

194#[cfg(not(target_arch = "x86_64"))]
195fn cpu_signature() -> String {
196    String::new()
197}

199/// One qualification's file.
200#[derive(Debug, Serialize, Deserialize)]
201struct Entry<T> {
202    schema: u32,
203    host: HostKey,
204    /// The qualification's own inputs, as it describes them.
205    inputs: String,
206    verdict: T,
207}

209/// Where a qualification named `name` is cached, or `None` without a home
210/// directory.
211pub(crate) fn default_path(name: &str) -> Option<PathBuf> {
212    dirs::cache_dir().map(|dir| dir.join("hermit").join(format!("{name}.json")))
213}

215/// The verdict cached at `path` for `host` and `inputs`, if there is one.
216pub(crate) fn load<T: DeserializeOwned>(path: &Path, host: &HostKey, inputs: &str) -> Option<T> {
217    let bytes = read_bounded(path)?;
218    let entry: Entry<T> = serde_json::from_slice(&bytes).ok()?;
219    (entry.schema == SCHEMA && entry.host == *host && entry.inputs == inputs)
220        .then_some(entry.verdict)
221}
222
223/// Caches `verdict` at `path` for `host` and `inputs`, by an atomic rename of
224/// a file in the same directory, and says whether it did. A failure leaves
225/// the cache as it was.
226pub(crate) fn store<T: Serialize>(path: &Path, host: &HostKey, inputs: &str, verdict: &T) -> bool {
227    let entry = Entry {
228        schema: SCHEMA,
229        host: host.clone(),
230        inputs: inputs.to_owned(),
231        verdict,
232    };
233    write_atomically(path, &entry).is_ok()
234}

236/// Whether a verdict could be stored at `path`: its directory exists, or can
237/// be made, and this process may write it.
238pub(crate) fn writable(path: &Path) -> bool {
239    path.parent().is_some_and(|directory| {
240        fs::create_dir_all(directory).is_ok()
241            && nix::unistd::access(directory, nix::unistd::AccessFlags::W_OK).is_ok()
242    })
243}

245/// Runs `f` holding an exclusive `flock` on `<path>.lock`, so that processes
246/// that miss at once measure one at a time and the later ones can find the
247/// first one's verdict. Without a lock file that is a regular file `f` runs
248/// unlocked; the open neither blocks on a FIFO nor follows a link.
249pub(crate) fn locked<R>(path: &Path, f: impl FnOnce() -> R) -> R {
250    use std::os::fd::AsRawFd;
251    let lock = path.parent().and_then(|directory| {
252        fs::create_dir_all(directory).ok()?;
253        let file = fs::OpenOptions::new()
254            .create(true)
255            .truncate(false)
256            .write(true)
257            .custom_flags(libc::O_NOFOLLOW | libc::O_NONBLOCK)
258            .open(path.with_extension("lock"))
259            .ok()?;
260        if !file.metadata().ok()?.is_file() {
261            return None;
262        }
263        // SAFETY: the descriptor is open for the duration of the call.
264        (unsafe { libc::flock(file.as_raw_fd(), libc::LOCK_EX) } == 0).then_some(file)
265    });
266    let result = f();
267    drop(lock);
268    result
269}

271/// Reads `path` if it is a regular file of at most [`MAX_FILE_BYTES`] that
272/// this process's user owns. The open neither blocks on a FIFO nor follows a
273/// symbolic link.
274fn read_bounded(path: &Path) -> Option<Vec<u8>> {
275    read_bounded_as(path, nix::unistd::geteuid().as_raw())
276}
277
278/// [`read_bounded`] for a file that user `owner` must own.
279fn read_bounded_as(path: &Path, owner: u32) -> Option<Vec<u8>> {
280    use std::os::unix::fs::MetadataExt;
281    let file = fs::OpenOptions::new()
282        .read(true)
283        .custom_flags(libc::O_NONBLOCK | libc::O_NOFOLLOW)
284        .open(path)
285        .ok()?;
286    let metadata = file.metadata().ok()?;
287    if !metadata.is_file() || metadata.len() > MAX_FILE_BYTES || metadata.uid() != owner {
288        return None;
289    }
290    // A file that grows after the check is cut at the cap, so it does not
291    // parse and is a miss.
292    let mut bytes = Vec::new();
293    file.take(MAX_FILE_BYTES).read_to_end(&mut bytes).ok()?;
294    Some(bytes)
295}
296
297fn write_atomically<T: Serialize>(path: &Path, entry: &T) -> std::io::Result<()> {
298    let directory = path
299        .parent()
300        .ok_or_else(|| std::io::Error::other("the cache path has no directory"))?;
301    fs::create_dir_all(directory)?;
302    let mut temporary = tempfile::NamedTempFile::new_in(directory)?;
303    temporary.write_all(&serde_json::to_vec_pretty(entry).map_err(std::io::Error::other)?)?;
304    temporary.persist(path).map_err(|error| error.error)?;
305    Ok(())
306}

399    /// Each key field reads what it names, compared with an independent read
400    /// of the same source: `/proc/sys/kernel/osrelease`, the boot id file,
401    /// `/proc/cpuinfo`'s vendor, family, model, stepping and microcode, and
402    /// this test binary's own metadata.
403    #[test]
404    fn the_current_key_reads_this_host() {
405        let host = HostKey::current();
406        assert_eq!(
407            host.kernel_release,
408            read_trimmed("/proc/sys/kernel/osrelease")
409        );
410        let boot = fs::read_to_string("/proc/sys/kernel/random/boot_id").unwrap();
411        assert_eq!(host.boot_id, boot.trim());
412        assert!(!host.boot_id.is_empty());
413        let cpuinfo = fs::read_to_string("/proc/cpuinfo").unwrap();
414        let field = |name: &str| {
415            cpuinfo
416                .lines()
417                .find_map(|line| {
418                    let (key, value) = line.split_once(':')?;
419                    (key.trim() == name).then(|| value.trim().to_owned())
420                })
421                .unwrap_or_default()
422        };
423        #[cfg(target_arch = "x86_64")]
424        {
425            let number = |name: &str| field(name).parse::<u32>().unwrap();
426            assert_eq!(
427                host.cpu,
428                format!(
429                    "{} family {:#x} model {:#x} stepping {:#x}",
430                    field("vendor_id"),
431                    number("cpu family"),
432                    number("model"),
433                    number("stepping")
434                )
435            );
436        }
437        let sysfs = read_trimmed("/sys/devices/system/cpu/cpu0/microcode/version");
438        assert_eq!(host.microcode, sysfs);
439        if !field("microcode").is_empty() {
440            assert_eq!(
441                u64::from_str_radix(host.microcode.trim_start_matches("0x"), 16).unwrap(),
442                u64::from_str_radix(field("microcode").trim_start_matches("0x"), 16).unwrap()
443            );
444        }
445        // The build ID read from memory equals the one the binary on disk
446        // carries, parsed independently. A binary built without a GNU
447        // build ID (the validate profile's test binaries are one) falls
448        // back to the file identity, exactly as `HostKey::current` does.
449        use object::Object;
450        let exe = fs::File::open(std::env::current_exe().unwrap()).unwrap();
451        let cache = object::read::ReadCache::new(exe);
452        let parsed = object::File::parse(&cache).unwrap();
453        match parsed.build_id().unwrap() {
454            Some(on_disk) => {
455                let hex: String = on_disk.iter().map(|byte| format!("{byte:02x}")).collect();
456                assert_eq!(host.hermit_binary, hex);
457            }
458            None => assert_eq!(host.hermit_binary, binary_identity()),
459        }
460    }

462    #[test]
463    fn the_build_id_is_found_among_notes() {
464        let note = |name: &[u8], kind: u32, desc: &[u8]| {
465            let mut bytes = Vec::new();
466            bytes.extend((name.len() as u32).to_ne_bytes());
467            bytes.extend((desc.len() as u32).to_ne_bytes());
468            bytes.extend(kind.to_ne_bytes());
469            bytes.extend(name);
470            bytes.resize(bytes.len().next_multiple_of(4), 0);

472            bytes.resize(bytes.len().next_multiple_of(4), 0);

478        assert_eq!(gnu_build_id(&note(b"GNU\0", 1, &[1, 2])), None);
