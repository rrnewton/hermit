68fn bounded_file(mut file: &File, maximum: usize) -> Result<Vec<u8>> {
69    let mut bytes = Vec::new();
70    Read::by_ref(&mut file)
71        .take(maximum as u64 + 1)
72        .read_to_end(&mut bytes)?;
73    ensure!(
74        bytes.len() <= maximum,
75        "bootstrap input exceeds declared bound"
76    );
77    Ok(bytes)
78}

80fn read_path(path: impl AsRef<Path>, maximum: usize) -> Result<Vec<u8>> {
81    bounded_file(&File::open(path)?, maximum)

84fn word(bytes: &[u8], offset: usize) -> Result<usize> {
85    let end = offset
86        .checked_add(8)
87        .ok_or_else(|| anyhow!("word offset overflow"))?;
88    Ok(u64::from_le_bytes(
89        bytes
90            .get(offset..end)
91            .ok_or_else(|| anyhow!("short word"))?
92            .try_into()?,
93    ) as usize)
94}

107impl Map {
108    fn contains(&self, address: usize, size: usize) -> bool {
109        address >= self.start && address.checked_add(size).is_some_and(|end| end <= self.end)
110    }
111
112    fn displays_path(&self, path: &Path) -> bool {
113        // Linux maps escapes newline as \\012 but leaves a literal backslash
114        // unchanged. Never decode this ambiguous display into an object path.
115        let mut displayed = Vec::new();
116        for byte in path.as_os_str().as_bytes() {
117            if *byte == b'\n' {
118                displayed.extend_from_slice(b"\\012");
119            } else {
120                displayed.push(*byte);
121            }
122        }
123        self.path == displayed
124    }
125
126    fn mapped_path(&self, pid: Pid) -> Result<PathBuf> {
127        ensure!(
128            self.path.starts_with(b"/") && self.inode != 0,
129            "unsupported anonymous early getrandom source"
130        );
131        // Ordinary path displays are already raw bytes. Only ambiguous kernel
132        // displays need map_files; do not add that proc capability requirement
133        // to an otherwise unambiguous legacy path.
134        if !self.path.windows(4).any(|part| part == b"\\012") && !self.path.ends_with(b" (deleted)")
135        {
136            return Ok(PathBuf::from(OsStr::from_bytes(&self.path)));
137        }
138        // The magic link returns actual pathname bytes. Authenticate its
139        // display and then the opened FD/device/inode/root-relative contents;
140        // a newline and a literal \\012 alone cannot select the same object.
141        let path = std::fs::read_link(format!(
142            "/proc/{pid}/map_files/{:x}-{:x}",
143            self.start, self.end
144        ))?;
145        ensure!(
146            self.displays_path(&path),
147            "mapped object path display changed"
148        );
149        Ok(path)
150    }
151}

153fn maps(pid: Pid) -> Result<Vec<Map>> {
154    let bytes = read_path(format!("/proc/{pid}/maps"), MAX_MAPS)?;
155    bytes
156        .strip_suffix(b"\n")
157        .unwrap_or(&bytes)
158        .split(|byte| *byte == b'\n')
159        .map(|line| {
160            // Only the five structural fields are whitespace-separated text.
161            // The remaining pathname may contain spaces, kernel escapes or
162            // non-UTF8 bytes in an unrelated mapping. Retain those bytes and
163            // authenticate pathname bytes only when selecting that object.
164            let mut rest = line;
165            let mut fields = Vec::with_capacity(5);
166            for _ in 0..5 {
167                rest = rest.trim_ascii_start();
168                let end = rest
169                    .iter()
170                    .position(u8::is_ascii_whitespace)
171                    .unwrap_or(rest.len());
172                ensure!(end > 0, "missing bootstrap mapping field");
173                fields.push(std::str::from_utf8(&rest[..end])?);
174                rest = &rest[end..];
175            }
176            let (start, end) = fields[0]
177                .split_once('-')
178                .ok_or_else(|| anyhow!("bad map range"))?;
179            let row = Map {
180                start: usize::from_str_radix(start, 16)?,
181                end: usize::from_str_radix(end, 16)?,
182                offset: usize::from_str_radix(fields[2], 16)?,
183                device: fields[3].to_owned(),
184                inode: fields[4].parse()?,
185                path: rest.trim_ascii_start().to_vec(),
186                permissions: fields[1].to_owned(),
187            };
188            ensure!(
189                row.start < row.end && row.permissions.len() == 4,
190                "invalid bootstrap map"
191            );
192            Ok(row)
193        })
194        .collect()
195}

197fn containing(rows: &[Map], address: usize, size: usize) -> Result<&Map> {
198    let mut found = rows.iter().filter(|row| row.contains(address, size));
199    let first = found
200        .next()
201        .ok_or_else(|| anyhow!("bootstrap extent is not in one mapping"))?;
202    ensure!(found.next().is_none(), "ambiguous bootstrap extent");
203    Ok(first)
204}

206fn open_under_root(pid: Pid, path: &Path) -> Result<File> {
207    let path = path.as_os_str().as_bytes();
208    ensure!(path.starts_with(b"/"), "nonabsolute bootstrap object path");
209    let components: Vec<_> = path[1..].split(|byte| *byte == b'/').collect();
210    ensure!(
211        !components.is_empty()
212            && components
213                .iter()
214                .all(|c| !c.is_empty() && *c != b"." && *c != b".."),
215        "noncanonical bootstrap object path"
216    );
217    // Follow only the proc magic link to this owned task's root. Every actual
218    // path component thereafter is no-follow and relative to a held directory.
219    let mut file = File::open(format!("/proc/{pid}/root"))?;
220    for (i, part) in components.iter().enumerate() {
221        let name = CString::new(*part)?;
222        let flags = libc::O_CLOEXEC
223            | libc::O_NOFOLLOW
224            | if i + 1 < components.len() {
225                libc::O_DIRECTORY
226            } else {
227                0
228            };
229        let fd = unsafe { libc::openat(file.as_raw_fd(), name.as_ptr(), flags, 0) };
230        if fd < 0 {
231            return Err(std::io::Error::last_os_error().into());
232        }
233        file = unsafe { File::from_raw_fd(fd) };
234    }
235    Ok(file)
236}

243fn require_supported_snapshot_filesystem(file: &File) -> Result<()> {
244    let mut fs = std::mem::MaybeUninit::<libc::statfs>::uninit();
245    if unsafe { libc::fstatfs(file.as_raw_fd(), fs.as_mut_ptr()) } != 0 {
246        return Err(std::io::Error::last_os_error().into());
247    }
248    let fs = unsafe { fs.assume_init() };
249    // XFS_SUPER_MAGIC from linux/magic.h is not exported by this libc version.
250    // Limit this storage path to qualified regular filesystems; this is not a
251    // claim about their physical block-device backing. The mapped snapshot
252    // pages must be reclaimable independently of the supervisor's heap.
253    const XFS_SUPER_MAGIC: libc::c_long = 0x58465342;
254    ensure!(
255        matches!(
256            fs.f_type,
257            libc::BTRFS_SUPER_MAGIC | libc::EXT4_SUPER_MAGIC | XFS_SUPER_MAGIC
258        ),
259        "large bootstrap snapshot requires a supported cache filesystem (unqualified filesystem {:#x})",
260        fs.f_type
261    );
262    Ok(())
263}

265impl DiskSnapshot {
266    fn new(file: &File, length: usize) -> Result<Self> {
267        ensure!(
268            length <= isize::MAX as usize,
269            "bootstrap snapshot exceeds address space"
270        );
271        // Reuse the existing Hermit cache placement, never cwd or a new
272        // hardcoded scratch directory. Small objects do not need this cache.
273        let data = super::HermitData::new();
274        ensure!(
275            data.data_dir().is_absolute(),
276            "bootstrap snapshot cache must be absolute"
277        );
278        let parent = data.data_dir().join("tmp");
279        std::fs::create_dir_all(&parent)?;
280        let directory = tempfile::Builder::new()
281            .prefix("sabre-snapshot-")
282            .permissions(std::fs::Permissions::from_mode(0o700))
283            .tempdir_in(parent)?;
284        let mut output = tempfile::tempfile_in(directory.path())?;
285        require_supported_snapshot_filesystem(&output)?;
286        let mut buffer = [0; 64 * 1024];
287        let mut offset = 0;
288        while offset < length {
289            let size = buffer.len().min(length - offset);
290            file.read_exact_at(&mut buffer[..size], offset as u64)?;
291            output.write_all(&buffer[..size])?;
292            offset += size;
293        }
294        ensure!(
295            file.metadata()?.len() == length as u64,
296            "bootstrap object size changed during snapshot"
297        );
298        // Reopen read-only, then discard the writer. After mmap discard even
299        // the read-only FD: the guest shares this supervisor's PID namespace
300        // and must not obtain a snapshot alias through /proc/<parent>/fd.
301        let readonly = File::open(format!("/proc/self/fd/{}", output.as_raw_fd()))?;
302        drop(output);
303        let address = unsafe {
304            libc::mmap(
305                std::ptr::null_mut(),
306                length,
307                libc::PROT_READ,
308                libc::MAP_PRIVATE,
309                readonly.as_raw_fd(),
310                0,
311            )
312        };
313        if address == libc::MAP_FAILED {
314            return Err(std::io::Error::last_os_error().into());
315        }
316        let snapshot = Self {
317            address: address as usize,
318            length,
319        };
320        drop(readonly);
321        directory.close()?;
322        Ok(snapshot)
323    }

325    fn as_slice(&self) -> &[u8] {
326        // The anonymous copied backing has no writable handle or exposed
327        // pathname. It is owned solely by this read-only mapping until Drop,
328        // under the same trusted-supervisor assumption as the former Vec.
329        unsafe { std::slice::from_raw_parts(self.address as *const u8, self.length) }
330    }
331}

339enum OriginalBytes {
340    Memory(Vec<u8>),
341    Disk(DiskSnapshot),
342}
343
344impl OriginalBytes {
345    fn new(file: &File, length: usize) -> Result<Self> {
346        if length <= IN_MEMORY_SNAPSHOT_LIMIT {
347            Ok(Self::Memory(bounded_file(file, IN_MEMORY_SNAPSHOT_LIMIT)?))
348        } else {
349            Ok(Self::Disk(DiskSnapshot::new(file, length)?))
350        }
351    }
352
353    fn as_slice(&self) -> &[u8] {
354        match self {
355            Self::Memory(bytes) => bytes,
356            Self::Disk(snapshot) => snapshot.as_slice(),
357        }
358    }
359
360    fn matches_file(&self, file: &File) -> Result<bool> {
361        if file.metadata()?.len() != self.len() as u64 {
362            return Ok(false);
363        }
364        let mut buffer = [0; 64 * 1024];
365        for (index, expected) in self.chunks(buffer.len()).enumerate() {
366            let actual = &mut buffer[..expected.len()];
367            file.read_exact_at(actual, (index * 64 * 1024) as u64)?;
368            if actual != expected {
369                return Ok(false);
370            }
371        }
372        Ok(file.metadata()?.len() == self.len() as u64)
373    }
374}

392impl HeldObject {
393    fn open(path: &Path) -> Result<Self> {
394        let held = Self::open_file(path)?;
395        let elf = object::File::parse(held.bytes.as_slice())?;
396        ensure!(
397            elf.format() == object::BinaryFormat::Elf
398                && elf.architecture() == object::Architecture::X86_64
399                && elf.is_little_endian(),
400            "bootstrap requires x86-64 ELF"
401        );
402        Ok(held)
403    }
404
405    fn open_file(path: &Path) -> Result<Self> {
406        let path = std::fs::canonicalize(path)?;
407        let file = File::open(&path)?;
408        let meta = file.metadata()?;
409        ensure!(
410            meta.is_file() && meta.len() > 0,
411            "unsupported bootstrap ELF object size"
412        );
413        let length = usize::try_from(meta.len())?;
414        let bytes = OriginalBytes::new(&file, length)?;
415        ensure!(
416            bytes.matches_file(&file)?,
417            "bootstrap object changed during snapshot"
418        );
419        // Compare maps-device to maps-device for this exact held FD. Btrfs's
420        // maps superblock device need not equal its subvolume st_dev.
421        let address = unsafe {
422            libc::mmap(
423                std::ptr::null_mut(),
424                PAGE,
425                libc::PROT_READ,
426                libc::MAP_PRIVATE,
427                file.as_raw_fd(),
428                0,
429            )
430        };
431        ensure!(
432            address != libc::MAP_FAILED,
433            "cannot map held bootstrap object"
434        );
435        let observed = (|| {
436            let own = maps(Pid::this())?;
437            let row = containing(&own, address as usize, 1)?;
438            ensure!(
439                row.displays_path(&path) && row.inode == meta.ino() && row.offset == 0,
440                "held bootstrap FD mapping mismatch"
441            );
442            Ok::<_, anyhow::Error>(row.device.clone())
443        })();
444        let unmapped = unsafe { libc::munmap(address, PAGE) };
445        ensure!(unmapped == 0, "failed to unmap bootstrap identity mapping");
446        Ok(Self {
447            path,
448            file,
449            bytes,
450            device: observed?,
451            inode: meta.ino(),
452        })
453    }

463             device={} inode={} permissions={} expected_device={} expected_inode={}",
464            row.start,

466            row.offset,

468            row.inode,

477        let other = open_under_root(pid, &self.path)?;
478        let a = self.file.metadata()?;

491    fn bias(&self, pid: Pid, rows: &[Map]) -> Result<usize> {
492        let elf = object::File::parse(self.bytes.as_slice())?;
493        let mut common: Option<BTreeSet<usize>> = None;
494        for row in rows.iter().filter(|r| self.matches_mapping(r)) {
495            self.authenticate(pid, row)?;
496            if !row.permissions.contains('x') {
497                continue;
498            }
499            let mut candidates = BTreeSet::new();
500            for segment in elf.segments() {
501                let object::SegmentFlags::Elf { p_flags } = segment.flags() else {
502                    continue;
503                };
504                if p_flags & object::elf::PF_X == 0 {
505                    continue;
506                }
507                let (offset, size) = segment.file_range();
508                if size == 0 {
509                    continue;
510                }
511                let lo = offset as usize & !(PAGE - 1);
512                let end = (offset as usize)
513                    .checked_add(size as usize)
514                    .and_then(|n| n.checked_add(PAGE - 1))
515                    .ok_or_else(|| anyhow!("ELF segment overflow"))?
516                    & !(PAGE - 1);
517                if row.offset >= lo && row.offset < end {
518                    let virtual_start = (segment.address() as usize & !(PAGE - 1))
519                        .checked_add(row.offset - lo)
520                        .ok_or_else(|| anyhow!("ELF bias overflow"))?;
521                    if let Some(bias) = row.start.checked_sub(virtual_start) {
522                        candidates.insert(bias);
523                    }
524                }
525            }
526            ensure!(
527                !candidates.is_empty(),
528                "executable mapping is not backed by an executable ELF load segment"
529            );
530            common = Some(match common {
531                None => candidates,
532                Some(old) => old.intersection(&candidates).copied().collect(),
533            });
534        }
535        let biases =
536            common.ok_or_else(|| anyhow!("missing executable bootstrap object mapping"))?;
537        ensure!(biases.len() == 1, "ambiguous bootstrap ELF load bias");
538        Ok(*biases.first().unwrap())
539    }

541    /// Select the instance containing an observed executable address. The
542    /// loader and guest may map the same PT_INTERP file at different bases;
543    /// unrelated instances cannot contribute constraints to this address.
544    fn bias_at(&self, pid: Pid, rows: &[Map], anchor: usize, role: &str) -> Result<usize> {
545        let result = (|| {
546            let row = containing(rows, anchor, 1)?;
547            self.authenticate(pid, row)?;
548            ensure!(
549                row.permissions.contains('x'),
550                "anchor mapping is not executable"
551            );
552            let elf = object::File::parse(self.bytes.as_slice())?;
553            let mut candidates = BTreeSet::new();
554            for segment in elf.segments() {
555                let object::SegmentFlags::Elf { p_flags } = segment.flags() else {
556                    continue;
557                };
558                if p_flags & object::elf::PF_X == 0 {
559                    continue;
560                }
561                let (offset, size) = segment.file_range();
562                let offset = usize::try_from(offset)?;
563                let size = usize::try_from(size)?;
564                let virtual_start = usize::try_from(segment.address())?;
565                let file_anchor = row
566                    .offset
567                    .checked_add(anchor - row.start)
568                    .ok_or_else(|| anyhow!("anchor file offset overflow"))?;
569                if file_anchor < offset || file_anchor - offset >= size {
570                    continue;
571                }
572                let relative = virtual_start
573                    .checked_add(file_anchor - offset)
574                    .ok_or_else(|| anyhow!("anchor virtual address overflow"))?;
575                if let Some(bias) = anchor.checked_sub(relative)
576                    && bias.is_multiple_of(PAGE)
577                {
578                    candidates.insert(bias);
579                }
580            }
581            ensure!(
582                candidates.len() == 1,
583                "missing/ambiguous anchored ELF load bias"
584            );
585            Ok(*candidates.first().unwrap())
586        })();
587        result.map_err(|error: anyhow::Error| anyhow!("{role} anchor={anchor:#x}: {error:#}"))
588    }

590    fn program_headers(
591        &self,
592        pid: Pid,
593        rows: &[Map],
594        bias: usize,
595        address: usize,
596        count: usize,
597        size: usize,
598    ) -> Result<()> {
599        let result = (|| {
600            let phoff = word(&self.bytes, 32)?;
601            let phnum = u16::from_le_bytes(self.bytes[56..58].try_into()?) as usize;
602            ensure!(
603                count == phnum && size == 56,
604                "final program header count/size changed"
605            );
606            let length = phnum
607                .checked_mul(56)
608                .ok_or_else(|| anyhow!("program header extent overflow"))?;
609            let elf = object::File::parse(self.bytes.as_slice())?;
610            let first = elf
611                .segments()
612                .next()
613                .ok_or_else(|| anyhow!("missing first ELF load"))?;
614            let (offset, _) = first.file_range();
615            let virtual_address = usize::try_from(first.address())?
616                .checked_add(
617                    phoff
618                        .checked_sub(usize::try_from(offset)?)
619                        .ok_or_else(|| anyhow!("program headers precede first ELF load"))?,
620                )
621                .ok_or_else(|| anyhow!("program header virtual address overflow"))?;
622            ensure!(
623                bias.checked_add(virtual_address) == Some(address),
624                "final program header address differs from anchored ELF"
625            );
626            let row = containing(rows, address, length)?;
627            if self.matches_mapping(row) {
628                self.authenticate(pid, row)?;
629            } else {
630                // library_buf_get_original replaces precisely the first file
631                // page after saving the original image. Authenticate only the
632                // PHDR bytes there, anchored by the still-file-backed entry.
633                // This is not authority for anonymous instructions or frames.
634                let page = bias
635                    .checked_add(usize::try_from(first.address())? & !(PAGE - 1))
636                    .ok_or_else(|| anyhow!("copied ELF page address overflow"))?;
637                let object::SegmentFlags::Elf { p_flags } = first.flags() else {
638                    return Err(anyhow!("missing first ELF load flags"));
639                };
640                let permissions = format!(
641                    "{}{}{}p",
642                    if p_flags & object::elf::PF_R != 0 {
643                        'r'
644                    } else {
645                        '-'
646                    },
647                    if p_flags & object::elf::PF_W != 0 {
648                        'w'
649                    } else {
650                        '-'
651                    },
652                    if p_flags & object::elf::PF_X != 0 {
653                        'x'
654                    } else {
655                        '-'
656                    }
657                );
658                ensure!(
659                    offset == 0
660                        && row.start == page
661                        && page.checked_add(PAGE) == Some(row.end)
662                        && row.offset == 0
663                        && row.inode == 0
664                        && row.device == "00:00"
665                        && row.path.is_empty()
666                        && row.permissions == permissions,
667                    "unrecognized copied PHDR mapping: map={:#x}..{:#x} offset={:#x} \
668                     device={} inode={} permissions={} expected_page={page:#x} expected_permissions={permissions}",
669                    row.start,
670                    row.end,
671                    row.offset,
672                    row.device,
673                    row.inode,
674                    row.permissions
675                );
676                self.authenticate_path(pid)?;
677            }
678            let expected = self
679                .bytes
680                .get(
681                    phoff
682                        ..phoff
683                            .checked_add(length)
684                            .ok_or_else(|| anyhow!("program header range overflow"))?,
685                )
686                .ok_or_else(|| anyhow!("short program headers"))?;
687            ensure!(
688                self.at_virtual(virtual_address, length, false)? == expected
689                    && remote_bytes(pid, address, length)? == expected,
690                "final program headers differ from held ELF"
691            );
692            Ok(())
693        })();
694        result.map_err(|error: anyhow::Error| {
695            anyhow!("program PHDR address={address:#x} load_bias={bias:#x}: {error:#}")
696        })
697    }
698
699    fn at_virtual(&self, address: usize, length: usize, readonly: bool) -> Result<&[u8]> {
700        let elf = object::File::parse(self.bytes.as_slice())?;
701        let end = address
702            .checked_add(length)
703            .ok_or_else(|| anyhow!("ELF extent overflow"))?;
704        let mut found = None;
705        for segment in elf.segments() {
706            let (offset, size) = segment.file_range();
707            let lo = segment.address() as usize;
708            let hi = lo
709                .checked_add(size as usize)
710                .ok_or_else(|| anyhow!("ELF segment overflow"))?;
711            if address >= lo && end <= hi {
712                let object::SegmentFlags::Elf { p_flags } = segment.flags() else {
713                    continue;
714                };
715                ensure!(
716                    !readonly
717                        || (p_flags & object::elf::PF_R != 0 && p_flags & object::elf::PF_W == 0),
718                    "bootstrap descriptor is in a writable ELF load segment"
719                );
720                ensure!(found.is_none(), "ambiguous ELF file extent");
721                let start = (offset as usize)
722                    .checked_add(address - lo)
723                    .ok_or_else(|| anyhow!("ELF file offset overflow"))?;
724                found = self.bytes.get(
725                    start
726                        ..start
727                            .checked_add(length)
728                            .ok_or_else(|| anyhow!("ELF file range overflow"))?,
729                );
730            }
731        }
732        found.ok_or_else(|| anyhow!("bootstrap ELF extent is not fully file-backed"))
733    }
734
735    fn layout(&self) -> Result<FrameLayout> {
736        let elf = object::File::parse(self.bytes.as_slice())?;
737        let definitions: BTreeSet<_> = elf
738            .symbols()
739            .chain(elf.dynamic_symbols())
740            .filter(|s| s.name().ok() == Some(LAYOUT_SYMBOL) && !s.is_undefined())
741            .map(|s| {
742                (
743                    s.address() as usize,
744                    s.size() as usize,
745                    s.kind() == object::SymbolKind::Data,
746                )
747            })
748            .collect();
749        ensure!(
750            definitions.len() == 1,
751            "missing/ambiguous bootstrap frame descriptor"
752        );
753        let (address, size, data) = *definitions.first().unwrap();
754        ensure!(
755            data && size == 9 * 8,
756            "wrong bootstrap frame descriptor type/extent"
757        );
758        FrameLayout::decode(self.at_virtual(address, size, true)?)
759    }
760
761    fn symbol(&self, name: &str) -> Result<usize> {
762        let elf = object::File::parse(self.bytes.as_slice())?;
763        let symbols: BTreeSet<_> = elf
764            .symbols()
765            .chain(elf.dynamic_symbols())
766            .filter(|s| s.name().ok() == Some(name) && !s.is_undefined())
767            .map(|s| s.address() as usize)
768            .collect();
769        ensure!(
770            symbols.len() == 1,
771            "missing/ambiguous bootstrap ELF symbol {name}"
772        );
773        Ok(*symbols.first().unwrap())
774    }

841fn stat_generation(pid: Pid, bytes: &[u8]) -> Result<u64> {
842    // Linux comm is raw bytes and may itself contain ") ". Only the PID
843    // prefix and the numeric fields following its final delimiter are text.
844    let end = bytes
845        .windows(2)
846        .rposition(|pair| pair == b") ")
847        .ok_or_else(|| anyhow!("malformed bootstrap process stat"))?;
848    let prefix = &bytes[..end];
849    let owner_end = prefix
850        .windows(2)
851        .position(|pair| pair == b" (")
852        .ok_or_else(|| anyhow!("missing process stat owner"))?;
853    ensure!(
854        std::str::from_utf8(&prefix[..owner_end])?.parse::<i32>()? == pid.as_raw(),
855        "bootstrap stat owner mismatch"
856    );
857    let start = bytes[end + 2..]
858        .split(u8::is_ascii_whitespace)
859        .filter(|field| !field.is_empty())
860        .nth(19)
861        .ok_or_else(|| anyhow!("missing process generation"))?;
862    Ok(std::str::from_utf8(start)?.parse()?)
863}

871fn script_words(bytes: &[u8]) -> Result<Vec<&[u8]>> {
872    // Match the loader's single BINPRM_BUF_SIZE/fgets/strtok resolution.
873    if !bytes.starts_with(b"#!") {
874        return Ok(Vec::new());
875    }
876    let line = &bytes[..bytes.len().min(255)];
877    let line = &line[..line.iter().position(|b| *b == b'\n').unwrap_or(line.len())];
878    let mut tokens = line[2..]
879        .split(|b| matches!(b, b' ' | b'\n' | b'\r' | b'\t'))
880        .filter(|part| !part.is_empty());
881    let path = tokens
882        .next()
883        .ok_or_else(|| anyhow!("missing SaBRe script interpreter"))?;
884    // Preserve the existing interpreter-path NUL refusal. C strtok ends the
885    // optional argument at its first NUL and never considers later tokens.
886    ensure!(!path.contains(&0), "NUL in SaBRe script interpreter");
887    let mut words = vec![path];
888    if let Some(token) = tokens.next() {
889        let token = &token[..token
890            .iter()
891            .position(|byte| *byte == 0)
892            .unwrap_or(token.len())];
893        if !token.is_empty() {
894            words.push(token);
895        }
896    }
897    Ok(words)
898}

900fn elf_interpreter(bytes: &[u8]) -> Result<Option<PathBuf>> {
901    // PT_INTERP is a program header, not a PT_LOAD segment.
902    let phoff = word(bytes, 32)?;
903    let phnum = u16::from_le_bytes(
904        bytes
905            .get(56..58)
906            .ok_or_else(|| anyhow!("short ELF header"))?
907            .try_into()?,
908    ) as usize;
909    ensure!(
910        phnum <= 128 && bytes.get(54..56) == Some(&56u16.to_le_bytes()),
911        "unsupported ELF program headers"
912    );
913    ensure!(
914        phnum > 0
915            && phoff
916                .checked_add(phnum * 56)
917                .is_some_and(|end| end <= bytes.len()),
918        "program headers outside held ELF"
919    );
920    let mut path = None;
921    for i in 0..phnum {
922        let p = phoff
923            .checked_add(i * 56)
924            .ok_or_else(|| anyhow!("program header overflow"))?;
925        if bytes.get(p..p + 4) == Some(&3u32.to_le_bytes()) {
926            ensure!(path.is_none(), "duplicate guest interpreter");
927            let offset = word(bytes, p + 8)?;
928            let size = word(bytes, p + 32)?;
929            ensure!((2..=4096).contains(&size), "invalid guest interpreter path");
930            let bytes = bytes
931                .get(
932                    offset
933                        ..offset
934                            .checked_add(size)
935                            .ok_or_else(|| anyhow!("interpreter extent overflow"))?,
936                )
937                .ok_or_else(|| anyhow!("short interpreter path"))?;
938            ensure!(
939                bytes.last() == Some(&0) && !bytes[..size - 1].contains(&0),
940                "invalid interpreter terminator"
941            );
942            path = Some(PathBuf::from(OsStr::from_bytes(&bytes[..size - 1])));
943        }
944    }
945    Ok(path)
946}

948// The initial exec stop precedes the loader's first instruction. Bound string
949// capture by the actual kernel-created stack mapping, with one non-overlapping
950// extent per argument. No smaller per-string limit is imposed. Only argv is
951// retained; environment values are not retained or included in diagnostics.
952fn initial_arguments(pid: Pid, rows: &[Map], stack: usize) -> Result<Vec<Vec<u8>>> {
953    let row = containing(rows, stack, 8)?;
954    ensure!(
955        row.path == b"[stack]" && row.permissions.starts_with("rw"),
956        "initial argv is not on the owned writable stack"
957    );
958    let argc = word(&remote_bytes(pid, stack, 8)?, 0)?;
959    // run_sabre prepends the loader, plugin and delimiter to the existing
960    // supported guest argc (4096). IMAGE retains that original guest bound.
961    ensure!(
962        (4..=4099).contains(&argc),
963        "unsupported initial loader argc"
964    );
965    let vector_size = (argc + 2) * 8;
966    ensure!(
967        row.contains(stack, vector_size),
968        "initial argv vector leaves stack"
969    );
970    let vector = remote_bytes(pid, stack, vector_size)?;
971    ensure!(
972        word(&vector, (argc + 1) * 8)? == 0,
973        "missing initial argv terminator"
974    );
975    let mut previous_end = stack + vector_size;
976    let mut arguments = Vec::with_capacity(argc);
977    for index in 0..argc {
978        let pointer = word(&vector, (index + 1) * 8)?;
979        ensure!(
980            pointer >= previous_end && row.contains(pointer, 1),
981            "initial argv string overlaps or leaves kernel stack: index={index}"
982        );
983        let mut value = Vec::new();
984        let mut at = pointer;
985        loop {
986            ensure!(
987                at < row.end,
988                "unterminated initial argv string: index={index}"
989            );
990            let chunk = remote_bytes(pid, at, (row.end - at).min(PAGE))?;
991            if let Some(end) = chunk.iter().position(|byte| *byte == 0) {
992                value.extend_from_slice(&chunk[..=end]);
993                previous_end = at + end + 1;
994                break;
995            }
996            value.extend_from_slice(&chunk);
997            at += chunk.len();
998        }
999        arguments.push(value);
1000    }
1001    Ok(arguments)
1002}

1004fn guest_arguments(initial: Vec<Vec<u8>>, script: Option<&[u8]>) -> Result<Vec<Vec<u8>>> {
1005    // find_client_path_idx scans after argv[0], and stops at the first exact
1006    // delimiter. Later literal "--" arguments belong to the client unchanged.
1007    let delimiter = initial
1008        .iter()
1009        .skip(1)
1010        .position(|value| value == b"--\0")
1011        .map(|index| index + 1)
1012        .ok_or_else(|| anyhow!("initial loader argv lacks delimiter"))?;
1013    ensure!(
1014        delimiter + 1 < initial.len(),
1015        "initial loader argv lacks client"
1016    );
1017    let mut expected = Vec::new();
1018    if let Some(script) = script {
1019        for token in script_words(script)? {
1020            let mut value = token.to_vec();
1021            value.push(0);
1022            expected.push(value);
1023        }
1024    }
1025    expected.extend(initial.into_iter().skip(delimiter + 1));
1026    ensure!(
1027        !expected.is_empty() && expected.len() <= 4096,
1028        "unsupported bootstrap argc"
1029    );
1030    Ok(expected)
1031}

1033fn authenticate_arguments(
1034    pid: Pid,
1035    rows: &[Map],
1036    vector: &[u8],
1037    expected: &[Vec<u8>],
1038) -> Result<usize> {
1039    let argc = word(vector, 0)?;
1040    ensure!(
1041        argc > 0 && argc <= 4096 && argc == expected.len(),
1042        "final argv count changed"
1043    );
1044    for (index, value) in expected.iter().enumerate() {
1045        let pointer = word(vector, (index + 1) * 8)?;
1046        let row = containing(rows, pointer, value.len()).map_err(|error| {
1047            anyhow!("final argv mapping index={index} address={pointer:#x}: {error:#}")
1048        })?;
1049        ensure!(
1050            row.permissions.starts_with('r'),
1051            "final argv mapping is not readable: index={index}"
1052        );
1053        // Values include the terminating NUL. Comparing the full retained
1054        // extent rejects truncation, extension, reordered arguments and an
1055        // unrelated readable pointer without assuming strings live on stack.
1056        let mut offset = 0;
1057        while offset < value.len() {
1058            let length = (value.len() - offset).min(PAGE);
1059            ensure!(
1060                remote_bytes(pid, pointer + offset, length)? == value[offset..offset + length],
1061                "final argv bytes/order changed: index={index}"
1062            );
1063            offset += length;
1064        }
1065    }
1066    let at = (argc + 1) * 8;
1067    ensure!(word(vector, at)? == 0, "missing final argv terminator");
1068    Ok(at + 8)
1069}

1204fn auxv_random(bytes: &[u8]) -> Result<usize> {
1205    ensure!(bytes.len().is_multiple_of(16), "short initial kernel auxv");
1206    let mut random = None;
1207    for pair in bytes.as_chunks::<16>().0 {
1208        let kind = word(pair, 0)?;
1209        let value = word(pair, 8)?;
1210        if kind == libc::AT_NULL as usize {
1211            ensure!(value == 0, "invalid auxv terminator");
1212            return random.ok_or_else(|| anyhow!("initial kernel auxv has no AT_RANDOM"));
1213        }
1214        if kind == libc::AT_RANDOM as usize {
1215            ensure!(
1216                value != 0 && random.replace(value).is_none(),
1217                "ambiguous initial kernel AT_RANDOM"
1218            );
1219        }
1220    }
1221    Err(anyhow!("unterminated initial kernel auxv"))
1222}

1224fn initial_vdso<'a>(rows: &'a [Map], auxv: &[u8]) -> Result<Option<&'a Map>> {
1225    ensure!(auxv.len().is_multiple_of(16), "short initial kernel auxv");
1226    let mut address = None;
1227    let mut terminated = false;
1228    for pair in auxv.as_chunks::<16>().0 {
1229        let kind = word(pair, 0)?;
1230        let value = word(pair, 8)?;
1231        if kind == libc::AT_NULL as usize {
1232            ensure!(value == 0, "invalid auxv terminator");
1233            terminated = true;
1234            break;
1235        }
1236        if kind == libc::AT_SYSINFO_EHDR as usize {
1237            ensure!(
1238                address.replace(value).is_none(),
1239                "ambiguous initial vDSO auxv"
1240            );
1241        }
1242    }
1243    ensure!(terminated, "unterminated initial kernel auxv");
1244    let mut candidates = rows.iter().filter(|row| row.path == b"[vdso]");
1245    let mapping = candidates.next();
1246    ensure!(candidates.next().is_none(), "ambiguous initial kernel vDSO");
1247    match (address.filter(|value| *value != 0), mapping) {
1248        (None, None) => Ok(None),
1249        (Some(address), Some(row)) => {
1250            ensure!(address == row.start, "initial vDSO auxv/mapping mismatch");
1251            ensure!(
1252                row.permissions == "r-xp" && row.inode == 0 && row.offset == 0,
1253                "unsupported initial vDSO mapping"
1254            );
1255            Ok(Some(row))
1256        }
1257        _ => Err(anyhow!("initial vDSO auxv/mapping mismatch")),
1258    }
1259}

1261impl Bootstrap {
1262    pub(super) fn new(root: Pid, launch: Launch) -> Result<Self> {
1263        let generation = generation(root)?;
1264        ensure!(generation != 0, "invalid initial process generation");
1265        let rows = maps(root)?;
1266        let expected_arguments = if launch.initializes_random() {
1267            let registers = nix::sys::ptrace::getregs(root)?;
1268            guest_arguments(
1269                initial_arguments(root, &rows, registers.rsp as usize)?,
1270                launch.script.as_ref().map(|script| script.bytes.as_slice()),
1271            )?
1272        } else {
1273            // Static legacy TAKE has no IMAGE/argv proof. Preserve its
1274            // existing admission instead of imposing the dynamic argc bound.
1275            Vec::new()
1276        };
1277        let auxv = read_path(format!("/proc/{root}/auxv"), 4096)?;
1278        // The caller still owns the initial attach stop. Capture pristine
1279        // bytes now; absence requires agreement between maps and kernel auxv,
1280        // and cannot stand in for an unreadable or already rewritten vDSO.
1281        let vdso = initial_vdso(&rows, &auxv)?
1282            .map(|mapping| {
1283                let bytes = remote_bytes(root, mapping.start, mapping.end - mapping.start)?;
1284                let elf = object::File::parse(bytes.as_slice())?;
1285                ensure!(
1286                    elf.architecture() == object::Architecture::X86_64 && elf.is_little_endian(),
1287                    "invalid initial kernel vDSO ELF"
1288                );
1289                Ok::<_, anyhow::Error>(VdsoSnapshot {
1290                    mapping: mapping.clone(),
1291                    bytes,
1292                })
1293            })
1294            .transpose()?;
1295        let initial_random = auxv_random(&auxv)?;
1296        ensure!(
1297            self::generation(root)? == generation,
1298            "initial process generation changed"
1299        );
1300        let prng = root_prng(launch.config.rng_seed());
1301        Ok(Self {
1302            launch,
1303            root,
1304            generation,
1305            image: None,
1306            prng,
1307            taken: false,
1308            sigill: None,
1309            initial_random,
1310            expected_arguments,
1311            vdso,
1312            other_objects: Vec::new(),
1313            continuations: BTreeMap::new(),
1314            static_reexec_observed: false,
1315        })
1316    }

1326    fn syscall_site(&self, pid: Pid, rows: &[Map], site: usize) -> Result<usize> {
1327        let loader = &self.launch.loader;
1328        let bias = loader.bias_at(pid, rows, site, "loader private syscall")?;
1329        let symbol = loader.symbol(bootstrap::SYSCALL_SYMBOL)?;
1330        ensure!(
1331            site == bias
1332                .checked_add(symbol)
1333                .ok_or_else(|| anyhow!("loader syscall address overflow"))?,
1334            "private bootstrap request is not at the retained loader instruction"
1335        );
1336        let row = containing(rows, site, 2)?;
1337        ensure!(
1338            row.permissions.contains('x'),
1339            "private bootstrap instruction is not executable"
1340        );
1341        loader.authenticate(pid, row)?;
1342        ensure!(
1343            loader.at_virtual(symbol, 2, false)? == [0x0f, 0x05]
1344                && remote_bytes(pid, site, 2)? == [0x0f, 0x05],
1345            "private bootstrap instruction bytes changed"
1346        );
1347        // Recheck the complete immutable descriptor in its live readonly load
1348        // mapping as well as the pre-launch held-file extraction.
1349        let descriptor = loader.symbol(LAYOUT_SYMBOL)?;
1350        let address = bias
1351            .checked_add(descriptor)
1352            .ok_or_else(|| anyhow!("descriptor address overflow"))?;
1353        let mapping = containing(rows, address, 72)?;
1354        ensure!(
1355            mapping.permissions.starts_with("r-") && !mapping.permissions.contains('w'),
1356            "live bootstrap descriptor is writable"
1357        );
1358        loader.authenticate(pid, mapping)?;
1359        ensure!(
1360            remote_bytes(pid, address, 72)? == loader.at_virtual(descriptor, 72, true)?,
1361            "live bootstrap descriptor differs from held object"
1362        );
1363        Ok(bias)
1364    }

1366    fn image(&mut self, pid: Pid, rows: &[Map], stack: usize, entry: usize) -> Result<i64> {
1367        ensure!(
1368            self.image.is_none() && !self.taken,
1369            "duplicate bootstrap IMAGE"
1370        );
1371        ensure!(stack.is_multiple_of(16), "unaligned final guest stack");
1372        let stack_map = containing(rows, stack, 8)?;
1373        ensure!(
1374            stack_map.permissions.starts_with("rw") && stack_map.path == b"[stack]",
1375            "final guest stack is not the owned writable stack"
1376        );
1377        let length = (stack_map.end - stack).min(64 * 1024);
1378        let bytes = remote_bytes(pid, stack, length)?;
1379        let mut at = authenticate_arguments(pid, rows, &bytes, &self.expected_arguments)?;
1380        let mut env_count = 0;
1381        loop {
1382            let pointer = word(&bytes, at)?;
1383            at += 8;
1384            if pointer == 0 {
1385                break;
1386            }
1387            env_count += 1;
1388            ensure!(
1389                env_count <= 4096 && stack_map.contains(pointer, 1),
1390                "unsupported final environment extent"
1391            );
1392        }
1393        let mut aux = std::collections::BTreeMap::new();
1394        let mut terminated = false;
1395        for _ in 0..128 {
1396            let kind = word(&bytes, at)?;
1397            let value = word(&bytes, at + 8)?;
1398            at += 16;
1399            if kind == libc::AT_NULL as usize {
1400                ensure!(value == 0, "bad final auxv terminator");
1401                terminated = true;
1402                break;
1403            }
1404            ensure!(
1405                aux.insert(kind, value).is_none(),
1406                "duplicate final auxv entry"
1407            );
1408        }
1409        ensure!(terminated, "unterminated final auxv");
1410        let get = |key: u64| {
1411            aux.get(&(key as usize))
1412                .copied()
1413                .ok_or_else(|| anyhow!("missing final auxv field {key}"))
1414        };
1415        let random = get(libc::AT_RANDOM)?;
1416        ensure!(
1417            random == self.initial_random && stack_map.contains(random, 16),
1418            "final AT_RANDOM is not the original owned writable target"
1419        );
1420        let program = &self.launch.program;
1421        let bias = program.bias_at(pid, rows, get(libc::AT_ENTRY)?, "program entry")?;
1422        let elf = object::File::parse(program.bytes.as_slice())?;
1423        let expected_entry = bias
1424            .checked_add(elf.entry() as usize)
1425            .ok_or_else(|| anyhow!("guest entry overflow"))?;
1426        ensure!(
1427            get(libc::AT_ENTRY)? == expected_entry,
1428            "final guest entry differs from held ELF"
1429        );
1430        let entry_map = containing(rows, expected_entry, 1)?;
1431        program.authenticate(pid, entry_map)?;
1432        ensure!(
1433            entry_map.permissions.contains('x'),
1434            "guest entry not executable"
1435        );
1436        program.program_headers(
1437            pid,
1438            rows,
1439            bias,
1440            get(libc::AT_PHDR)?,
1441            get(libc::AT_PHNUM)?,
1442            get(libc::AT_PHENT)?,
1443        )?;
1444        self.launch.authenticate_script(pid)?;
1445        let interpreter = self
1446            .launch
1447            .interpreter
1448            .as_ref()
1449            .ok_or_else(|| anyhow!("initial static image must not request random IMAGE"))?;
1450        let interp_bias = interpreter.bias_at(pid, rows, entry, "guest interpreter entry")?;
1451        let interp = object::File::parse(interpreter.bytes.as_slice())?;
1452        ensure!(
1453            entry
1454                == interp_bias
1455                    .checked_add(interp.entry() as usize)
1456                    .ok_or_else(|| anyhow!("interpreter entry overflow"))?,
1457            "loader final entry is not held guest interpreter"
1458        );
1459        let row = containing(rows, entry, 1)?;
1460        interpreter.authenticate(pid, row)?;
1461        ensure!(
1462            row.permissions.contains('x'),
1463            "interpreter entry not executable"
1464        );
1465        initialize_auxv(
1466            &mut self.prng,
1467            RemoteMemory(pid),
1468            AddrMut::from_raw(random).ok_or_else(|| anyhow!("null final AT_RANDOM"))?,
1469            detcore::types::DetTid::from_raw(pid.as_raw()),
1470        )?;
1471        self.image = Some(InitialImage {
1472            pid: pid.as_raw(),
1473            start_time_ticks: self.generation,
1474            at_random: random,
1475        });
1476        Ok(0)
1477    }

1479    fn original_bytes(
1480        &mut self,
1481        pid: Pid,
1482        rows: &[Map],
1483        site: usize,
1484        length: usize,
1485    ) -> Result<Vec<u8>> {
1486        ensure!(length <= 128, "original instruction read exceeds bound");
1487        let row = containing(rows, site, length)?.clone();
1488        ensure!(
1489            row.permissions.contains('x'),
1490            "early getrandom source is not executable"
1491        );
1492        if row.path == b"[vdso]" {
1493            let vdso = self
1494                .vdso
1495                .as_ref()
1496                .ok_or_else(|| anyhow!("vDSO origin lacks an initial snapshot"))?;
1497            ensure!(row == vdso.mapping, "initial kernel vDSO mapping changed");
1498            let offset = site - vdso.mapping.start;
1499            Ok(vdso
1500                .bytes
1501                .get(offset..offset + length)
1502                .ok_or_else(|| anyhow!("short original vDSO instruction range"))?
1503                .to_vec())
1504        } else {
1505            let object = if self.launch.program.matches_mapping(&row) {
1506                &self.launch.program
1507            } else if let Some(interpreter) = self.launch.interpreter.as_ref()
1508                && interpreter.matches_mapping(&row)
1509            {
1510                interpreter
1511            } else {
1512                if !self
1513                    .other_objects
1514                    .iter()
1515                    .any(|object| object.matches_mapping(&row))
1516                {
1517                    ensure!(
1518                        self.other_objects.len() < MAX_OBJECTS,
1519                        "too many early bootstrap source objects"
1520                    );
1521                    let object = HeldObject::open(&row.mapped_path(pid)?)?;
1522                    object.authenticate(pid, &row)?;
1523                    self.other_objects.push(object);
1524                }
1525                self.other_objects
1526                    .iter()
1527                    .find(|object| object.matches_mapping(&row))
1528                    .ok_or_else(|| anyhow!("bootstrap source object identity changed"))?
1529            };
1530            object.authenticate(pid, &row)?;
1531            let bias = object.bias_at(pid, rows, site, "original instruction")?;
1532            let relative = site
1533                .checked_sub(bias)
1534                .ok_or_else(|| anyhow!("source bias underflow"))?;
1535            Ok(object.at_virtual(relative, length, false)?.to_vec())
1536        }
1537    }

1542            "early source was not a syscall in the authenticated image"

1547    /// Save only a real kernel signal-delivery observation. A later private
1548    /// request's argument values or supplied stack pointer cannot create it.
1549    pub(super) fn signal(&mut self, pid: Pid, signal: nix::sys::signal::Signal) -> Result<()> {
1550        ensure!(
1551            self.sigill.take().is_none(),
1552            "intervening signal invalidated bootstrap SIGILL provenance"
1553        );
1554        if self.taken
1555            || !self.launch.initializes_random()
1556            || signal != nix::sys::signal::Signal::SIGILL
1557        {
1558            return Ok(());
1559        }
1560        self.same_owner(pid)?;
1561        let regs = nix::sys::ptrace::getregs(pid)?;
1562        if regs.rax != libc::SYS_getrandom as u64
1563            || remote_bytes(pid, regs.rip as usize, 2)? != [0x0f, 0xff]
1564        {
1565            return Ok(());
1566        }
1567        ensure!(
1568            self.image.is_some(),
1569            "getrandom SIGILL before authenticated IMAGE"
1570        );
1571        let info = nix::sys::ptrace::getsiginfo(pid)?;
1572        ensure!(
1573            info.si_code > 0 && unsafe { info.si_addr() } as usize == regs.rip as usize,
1574            "SIGILL is not a kernel fault at the rewritten source"
1575        );
1576        let rows = maps(pid)?;
1577        let mapping = self.original_syscall(pid, &rows, regs.rip as usize)?;
1578        self.sigill = Some(SigillOrigin {
1579            generation: self.generation,
1580            registers: regs,
1581            mapping,
1582        });
1583        Ok(())
1584    }

1586    fn origin(
1587        &mut self,
1588        pid: Pid,
1589        rows: &[Map],
1590        arguments: [usize; 3],
1591        wrapper: usize,
1592        loader_bias: usize,
1593    ) -> Result<()> {
1594        let layout = self.launch.layout;
1595        if let Some(saved) = self.sigill.take() {
1596            ensure!(
1597                saved.generation == self.generation,
1598                "stale bootstrap SIGILL generation"
1599            );
1600            let regs = saved.registers;
1601            ensure!(
1602                [regs.rdi as usize, regs.rsi as usize, regs.rdx as usize] == arguments,
1603                "SIGILL original arguments differ from forwarded request"
1604            );
1605            ensure!(
1606                self.original_syscall(pid, rows, regs.rip as usize)? == saved.mapping
1607                    && remote_bytes(pid, regs.rip as usize, 2)? == [0x0f, 0xff],
1608                "SIGILL source changed before forwarding"
1609            );
1610            let address = wrapper
1611                .checked_add(layout.scratch_return)
1612                .ok_or_else(|| anyhow!("SIGILL return pointer overflow"))?;
1613            let row = containing(rows, address, 8)?;
1614            ensure!(
1615                row.path == b"[stack]" && row.permissions.starts_with("rw"),
1616                "SIGILL return word is not on owned stack"
1617            );
1618            let returned = word(&remote_bytes(pid, address, 8)?, 0)?;
1619            ensure!(
1620                returned == regs.rip as usize + 2,
1621                "SIGILL return word differs from saved kernel continuation"
1622            );
1623            return Ok(());
1624        }
1625        let row = containing(rows, wrapper, layout.size)?;
1626        ensure!(
1627            row.path == b"[stack]"
1628                && row.permissions.starts_with("rw")
1629                && wrapper.is_multiple_of(8),
1630            "ordinary bootstrap frame is not on owned stack"
1631        );
1632        let frame = remote_bytes(pid, wrapper, layout.size)?;
1633        ensure!(
1634            [
1635                word(&frame, layout.rdi)?,
1636                word(&frame, layout.rsi)?,
1637                word(&frame, layout.rdx)?
1638            ] == arguments,
1639            "assembly frame arguments differ from forwarded request"
1640        );
1641        let returned = word(&frame, layout.architectural_return)?;
1642        let site = returned
1643            .checked_sub(2)
1644            .ok_or_else(|| anyhow!("architectural return underflow"))?;
1645        self.original_syscall(pid, rows, site)?;
1646        let scratch = word(&frame, layout.scratch_return)?;
1647        // Real rewriter.c's syscall trampoline: saved return points at the
1648        // red-zone-restoring LEA, 33 bytes after its PUSH/LEA/body begins.
1649        let start = scratch
1650            .checked_sub(33)
1651            .ok_or_else(|| anyhow!("scratch return underflow"))?;
1652        let mapping = containing(rows, start, 41)?;
1653        ensure!(
1654            mapping.permissions.contains('x'),
1655            "scratch continuation is not executable"
1656        );
1657        let code = remote_bytes(pid, start, 41)?;
1658        ensure!(
1659            code[..4] == [0x50, 0x48, 0x8d, 0x05]
1660                && code[8..11] == [0x50, 0x48, 0xb8]
1661                && code[19..33]
1662                    == [
1663                        0x50, 0x48, 0x8d, 0x05, 0x06, 0, 0, 0, 0x48, 0x87, 0x44, 0x24, 0x10, 0xc3
1664                    ]
1665                && code[33..] == [0x48, 0x8d, 0xa4, 0x24, 0x80, 0, 0, 0],
1666            "unrecognized full-frame scratch trampoline"
1667        );
1668        let displacement = i32::from_le_bytes(code[4..8].try_into()?) as i64;
1669        ensure!(
1670            (start as i64)
1671                .checked_add(8)
1672                .and_then(|n| n.checked_add(displacement))
1673                == Some(returned as i64),
1674            "scratch trampoline names a different syscall continuation"
1675        );
1676        let handler = word(&code, 11)?;
1677        let bias = loader_bias;
1678        let mut handlers = BTreeSet::new();
1679        for symbol in ["handle_syscall", "handle_syscall_loader"] {
1680            handlers.insert(
1681                bias.checked_add(self.launch.loader.symbol(symbol)?)
1682                    .ok_or_else(|| anyhow!("handler address overflow"))?,
1683            );
1684        }
1685        ensure!(
1686            handlers.contains(&handler),
1687            "scratch trampoline does not call the held loader wrapper"
1688        );
1689        // Bind the source jump to this exact scratch body, including the real
1690        // SYSCALL clobber/red-zone prefix and both relocated byte sequences.
1691        // The production rewriter keeps five instructions in its ring; four
1692        // preceding/following x86 instructions occupy at most 4 * 15 bytes.
1693        // A matching frame and an unrelated executable byte pattern alone are
1694        // not evidence that this original instruction enters that trampoline.
1695        let prefix = start
1696            .checked_sub(17)
1697            .ok_or_else(|| anyhow!("scratch prefix underflow"))?;
1698        let prefix_bytes = remote_bytes(pid, prefix, 17)?;
1699        ensure!(
1700            prefix_bytes[..13]
1701                == [
1702                    0x48, 0x8d, 0x64, 0x24, 0x80, 0x90, 0x90, 0x9c, 0x41, 0x5b, 0x48, 0x8d, 0x0d,
1703                ]
1704                && (start as i64)
1705                    .checked_add(i32::from_le_bytes(prefix_bytes[13..17].try_into()?) as i64)
1706                    == Some(returned as i64),
1707            "scratch prefix does not preserve the original SYSCALL continuation"
1708        );
1709        let source_map = containing(rows, site, 2)?;
1710        let mut matched = 0;
1711        for pre in 0..=60 {
1712            let Some(jump) = site.checked_sub(pre) else {
1713                continue;
1714            };
1715            let Some(destination) = prefix.checked_sub(pre) else {
1716                continue;
1717            };
1718            if !source_map.contains(jump, 5) {
1719                continue;
1720            }
1721            let entry = remote_bytes(pid, jump, 5)?;
1722            if entry[0] != 0xe9
1723                || (jump as i64 + 5).checked_add(i32::from_le_bytes(entry[1..5].try_into()?) as i64)
1724                    != Some(destination as i64)
1725            {
1726                continue;
1727            }
1728            ensure!(
1729                containing(rows, destination, pre + 17)?
1730                    .permissions
1731                    .contains('x'),
1732                "scratch entry is not executable"
1733            );
1734            if pre != 0
1735                && self.original_bytes(pid, rows, jump, pre)?
1736                    != remote_bytes(pid, destination, pre)?
1737            {
1738                continue;
1739            }
1740            for post in 0..=60 {
1741                if pre + 2 + post < 5 || !source_map.contains(jump, pre + 2 + post) {
1742                    continue;
1743                }
1744                let after = start + 41 + post;
1745                if !mapping.contains(after, 5) {
1746                    continue;
1747                }
1748                let exit = remote_bytes(pid, after, 5)?;
1749                if exit[0] != 0xe9
1750                    || (after as i64 + 5)
1751                        .checked_add(i32::from_le_bytes(exit[1..5].try_into()?) as i64)
1752                        != Some((returned + post) as i64)
1753                {
1754                    continue;
1755                }
1756                if post != 0
1757                    && self.original_bytes(pid, rows, returned, post)?
1758                        != remote_bytes(pid, start + 41, post)?
1759                {
1760                    continue;
1761                }
1762                if pre + 2 + post > 5
1763                    && remote_bytes(pid, jump + 5, pre + 2 + post - 5)?
1764                        .iter()
1765                        .any(|byte| *byte != 0x90)
1766                {
1767                    continue;
1768                }
1769                matched += 1;
1770            }
1771        }
1772        ensure!(
1773            matched == 1,
1774            "missing/ambiguous rewritten source-to-scratch linkage"
1775        );
1776        ensure!(
1777            remote_bytes(pid, site, 2)? != [0x0f, 0x05]
1778                && remote_bytes(pid, site, 2)? != [0x0f, 0xff],
1779            "ordinary assembly path lacks a rewritten source site"
1780        );
1781        Ok(())
1782    }

1784    /// None leaves the existing supervisor path untouched. Protocol ownership,
1785    /// phase and shape errors abort the run; syscall errors remain Linux results.
1786    pub(super) fn request(
1787        &mut self,
1788        pid: Pid,
1789        regs: &libc::user_regs_struct,
1790    ) -> Result<Option<i64>> {
1791        let private =
1792            regs.orig_rax == libc::SYS_prctl as u64 && regs.rdi == bootstrap::PRCTL_OPTION;
1793        if !private {
1794            ensure!(
1795                self.sigill.take().is_none(),
1796                "intervening syscall invalidated bootstrap SIGILL provenance"
1797            );
1798            return Ok(None);
1799        }
1800        let continuation = self.continuations.get(&pid).copied();
1801        if let Some(image) = continuation {
1802            ensure!(
1803                generation(pid)? == image.start_time_ticks,
1804                "continuation owner or process generation changed"
1805            );
1806        } else {
1807            self.same_owner(pid)?;
1808        }
1809        let rows = maps(pid)?;
1810        let site = (regs.rip as usize)
1811            .checked_sub(2)
1812            .ok_or_else(|| anyhow!("private syscall RIP underflow"))?;
1813        let loader_bias = self.syscall_site(pid, &rows, site)?;
1814        let initial_static = !self.taken && !self.launch.initializes_random();
1815        if continuation.is_some() || initial_static {
1816            ensure!(
1817                self.sigill.is_none()
1818                    && regs.rsi == bootstrap::TAKE_STATE
1819                    && regs.r8 == bootstrap::VERSION
1820                    && regs.r9 == 0,
1821                "legacy continuation requires the exact TAKE protocol"
1822            );
1823            let image = if let Some(image) = continuation {
1824                image
1825            } else {
1826                // Static clients initialize their plugin before final-stack
1827                // compaction. Authenticate the actual mapped held program,
1828                // while leaving their ordinary auxv/ThreadState path alone.
1829                self.launch.authenticate_script(pid)?;
1830                let program = &self.launch.program;
1831                let bias = program.bias(pid, &rows)?;
1832                let elf = object::File::parse(program.bytes.as_slice())?;
1833                let entry = bias
1834                    .checked_add(elf.entry() as usize)
1835                    .ok_or_else(|| anyhow!("static entry overflow"))?;
1836                let row = containing(&rows, entry, 1)?;
1837                ensure!(
1838                    row.permissions.contains('x'),
1839                    "held static entry not executable"
1840                );
1841                program.authenticate(pid, row)?;
1842                InitialImage {
1843                    pid: pid.as_raw(),
1844                    start_time_ticks: self.generation,
1845                    at_random: self.initial_random,
1846                }
1847            };
1848            self.check_current_auxv(pid, &rows, image)?;
1849            let state = if initial_static {
1850                LoaderState::InitialStaticLegacy
1851            } else {
1852                LoaderState::ObservedExecContinuation
1853            };
1854            let bytes = encode_continuation(&self.launch.config, image, state)?;
1855            let result = Self::write_take(pid, regs, &bytes)?;
1856            if result > 0 {
1857                if initial_static {
1858                    self.taken = true;
1859                }
1860                self.continuations.remove(&pid);
1861            }
1862            return Ok(Some(result));
1863        }
1864        ensure!(!self.taken, "bootstrap state already handed off");
1865        let result = match regs.rsi {
1866            bootstrap::IMAGE => {
1867                ensure!(
1868                    self.sigill.is_none() && regs.r8 == bootstrap::VERSION && regs.r9 == 0,
1869                    "invalid IMAGE protocol shape"
1870                );
1871                self.image(pid, &rows, regs.rdx as usize, regs.r10 as usize)
1872                    .map_err(|error| anyhow!("bootstrap IMAGE authentication: {error:#}"))?
1873            }
1874            bootstrap::GETRANDOM => {
1875                ensure!(self.image.is_some(), "early getrandom before IMAGE");
1876                let args = [regs.rdx as usize, regs.r10 as usize, regs.r8 as usize];
1877                self.origin(pid, &rows, args, regs.r9 as usize, loader_bias)?;
1878                let call = Syscall::from_raw(
1879                    Sysno::getrandom,
1880                    SyscallArgs::new(args[0], args[1], args[2], 0, 0, 0),
1881                );
1882                let Syscall::Getrandom(call) = call else {
1883                    unreachable!()
1884                };
1885                match getrandom(
1886                    &mut self.prng,
1887                    RemoteMemory(pid),
1888                    detcore::types::DetTid::from_raw(pid.as_raw()),
1889                    call,
1890                ) {
1891                    Ok(n) => n,
1892                    Err(e) => -(e.into_raw() as i64),
1893                }
1894            }
1895            bootstrap::TAKE_STATE => {
1896                ensure!(
1897                    self.sigill.is_none() && regs.r8 == bootstrap::VERSION && regs.r9 == 0,
1898                    "invalid TAKE protocol shape"
1899                );
1900                let image = self.image.ok_or_else(|| anyhow!("TAKE before IMAGE"))?;
1901                self.check_current_auxv(pid, &rows, image)?;
1902                let bytes = encode_initial_state(&self.launch.config, image, &self.prng)?;
1903                let result = Self::write_take(pid, regs, &bytes)?;
1904                if result > 0 {
1905                    self.taken = true;
1906                }
1907                result
1908            }
1909            _ => return Err(anyhow!("unknown loader bootstrap operation")),
1910        };
1911        Ok(Some(result))
1912    }

1914    fn write_take(pid: Pid, regs: &libc::user_regs_struct, bytes: &[u8]) -> Result<i64> {
1915        let capacity = regs.r10 as usize;
1916        let destination = AddrMut::from_raw(regs.rdx as usize);
1917        if capacity == 0 || capacity > bootstrap::MAX_STATE_BYTES || destination.is_none() {
1918            return Ok(-libc::EINVAL as i64);
1919        }
1920        if bytes.len() > capacity {
1921            return Ok(-libc::EMSGSIZE as i64);
1922        }
1923        Ok(
1924            match RemoteMemory(pid).write_exact(destination.unwrap(), bytes) {
1925                Ok(()) => bytes.len() as i64,
1926                Err(e) => -(e.into_raw() as i64),
1927            },
1928        )
1929    }

1931    fn check_current_auxv(&self, pid: Pid, rows: &[Map], image: InitialImage) -> Result<()> {
1932        ensure!(
1933            generation(pid)? == image.start_time_ticks
1934                && pid.as_raw() == image.pid
1935                && auxv_random(&read_path(format!("/proc/{pid}/auxv"), 4096)?)? == image.at_random,
1936            "loader handoff image or kernel auxv changed"
1937        );
1938        let row = containing(rows, image.at_random, 16)?;
1939        ensure!(
1940            row.path == b"[stack]" && row.permissions.starts_with("rw"),
1941            "loader handoff AT_RANDOM is not the owned writable stack"

1950    /// Called only for a real kernel event on the supervisor's owned lineage,
1951    /// before it resumes that image. An exec replaces any unconsumed earlier
1952    /// image proof; a mere matching pid/start or user-supplied payload cannot
1953    /// create one. Entries live no longer than the existing owned tracee set.
1954    pub(super) fn event(&mut self, pid: Pid, event: libc::c_int) -> Result<()> {
1955        ensure!(
1956            self.sigill.take().is_none(),
1957            "intervening ptrace event invalidated bootstrap SIGILL provenance"
1958        );
1959        if event == libc::PTRACE_EVENT_EXEC {
1960            let image = InitialImage {
1961                pid: pid.as_raw(),
1962                start_time_ticks: generation(pid)?,
1963                at_random: auxv_random(&read_path(format!("/proc/{pid}/auxv"), 4096)?)?,
1964            };
1965            self.check_current_auxv(pid, &maps(pid)?, image)?;
1966            if self.taken {
1967                self.continuations.insert(pid, image);
1968                return Ok(());
1969            }
1970            if !self.launch.initializes_random() && !self.static_reexec_observed {
1971                self.same_owner(pid)?;
1972                // The static loader may exec its dynamic linker once to
1973                // preload the plugin. This does not authorize arbitrary initial
1974                // dynamic execution: TAKE must still prove the original held
1975                // static program and exact loader mapping before succeeding.
1976                self.static_reexec_observed = true;
1977                self.initial_random = image.at_random;
1978                return Ok(());
1979            }
1980        }
1981        ensure!(
1982            self.taken
1983                || !matches!(
1984                    event,
1985                    libc::PTRACE_EVENT_CLONE
1986                        | libc::PTRACE_EVENT_FORK
1987                        | libc::PTRACE_EVENT_VFORK
1988                        | libc::PTRACE_EVENT_EXEC
1989                ),
1990            "clone/fork/exec before initial random handoff is unsupported"
1991        );
1992        Ok(())
1993    }
