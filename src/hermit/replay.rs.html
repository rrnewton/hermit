48/// Represents a replay that is currently running.
49pub struct Replay {
50    // The running tracee.
51    tracer: Tracer,
52
53    // The chroot. When dropped, everything in this directory will be
54    // recursively deleted.
55    chroot: TempChroot,
56
57    // Retires this replay root's physical-file trust entries on every exit
58    // path, including cancellation or a spawn/wait error.
59    _materialization_scope: crate::replayer::ReplayMaterializationScope,
60}

62impl Replay {
63    /// Spawns a new replay using the provided base directory where the replay
64    /// data is stored.
65    pub async fn spawn(
66        dir: &Path,
67        capture_output: bool,
68        gdbserver: Option<u16>,
69        mounts: &[Mount],
70    ) -> Result<Self, Error> {
71        let metadata_path = dir.join(METADATA_NAME);
72
73        let metadata: Metadata = serde_json::from_reader(
74            fs::File::open(&metadata_path)
75                .with_context(|| format!("Failed to open {:?}", metadata_path))?,
76        )
77        .with_context(|| format!("Failed to parse {:?}", metadata_path))?;
78
79        let recording_version = &metadata.version;
80        let replayer_version = &RECORD_VERSION;
81        if !replayer_version.compatible_with(recording_version) {
82            return Err(anyhow::anyhow!(format!(
83                "Version mismatch, recording version {:?}, replayer version {:?}",
84                recording_version, replayer_version
85            )));
86        }
87
88        let mut command = metadata.command();
89
90        if capture_output {
91            command.stdin(Stdio::null());
92            command.stdout(Stdio::piped());
93            command.stderr(Stdio::piped());
94        }
95
96        let mut config = record_or_replay_config(dir);
97        // Recorder events contain the raw bytes from the recording namespace.
98        // Reapply Detcore's sanitizer with the recording-time mount IDs, not
99        // the unrelated IDs of this fresh replay namespace.
100        config.mountinfo_root_rewrites = metadata.mountinfo_root_rewrites.clone();
101        config.mountinfo_mount_ids = metadata.mountinfo_mount_ids.clone();
102        config.mountinfo_mount_ids_captured = metadata.mountinfo_mount_ids_captured;
103        config.fdinfo_unlisted_mount_ids = metadata.fdinfo_unlisted_mount_ids.clone();
104        let sequentialize_threads = config.sequentialize_threads;
105
106        let (chroot, bootstrap_program, materialization_scope, replay_mounts) =
107            prepare_chroot(dir, &metadata, mounts)
108                .context("Failed to create chroot environment")?;
109        // `Command::program` resets argv[0], so restore the recorded value.
110        command.program(&bootstrap_program).arg0(&metadata.arg0);
111
112        for mount in replay_mounts {
113            command.mount(mount);
114        }
115
116        // bind mount fbcode otherwise many program can fail to execve due to missing
117        // shared libraries. This path only exists on Meta hosts; skip it elsewhere
118        // (e.g. generic external CI runners) where the missing source would make
119        // mount(2) fail with ENOENT.
120        //
121        // The bind-mount target directory is created here, in the parent process,
122        // rather than via `Mount::touch_target()`. `touch_target` defers directory
123        // creation to the cloned child immediately before `execve`, where
124        // reverie-process runs it on a fixed 4 KiB clone stack (see
125        // reverie-process `clone.rs`). Its `create_dir_all`/`touch_path` helpers
126        // each place a `[0; PATH_MAX]` (4 KiB) buffer on that stack, overflowing
127        // it and corrupting the `envp` pointer that the child then passes to
128        // `execve`. That made the guest's initial `execve` fail with `EFAULT` on
129        // every Meta-host replay (recording spawns without mounts, so it was
130        // unaffected), so replay diverged from the recording at syscall #1.
131        // Pre-creating the target keeps the child's pre-exec path allocation-free.
132        let fbcode = Path::new("/usr/local/fbcode");
133        if fbcode.exists() {
134            // prepare_chroot created this target before recorded symlinks were
135            // installed, so the controller-visible mount path cannot escape.
136            let target = chroot.path().join("usr/local/fbcode");
137            command.mount(Mount::bind(fbcode, &target).readonly());
138            command.mount(
139                Mount::new(target)
140                    .flags(MountFlags::MS_BIND | MountFlags::MS_REMOUNT | MountFlags::MS_RDONLY),
141            );
142        }
143
144        // Keep process-relative executable aliases (`/proc/self/exe`,
145        // `/proc/thread-self/exe`, and `/proc/self/fd/N`) attached to the
146        // replayed guest. Resolving those spellings in the controller would
147        // select Hermit itself. The target is pre-created to avoid allocation
148        // in reverie-process's small pre-exec clone stack.
149        let proc_target = chroot.path().join("proc");
150        command.mount(Mount::proc().target(&proc_target));
151
152        command.chroot(chroot.path());
153
154        let mut builder = reverie_ptrace::TracerBuilder::<ReplayTool>::new(command).config(config);
155        if let Some(port) = gdbserver {
156            builder = builder.gdbserver(port);
157        }
158        if sequentialize_threads {
159            // Inform gdbserver not to serialize guests because this is
160            // done by detcore already.
161            builder = builder.sequentialized_guest();
162        }
163        let tracer = builder.spawn().await?;
164
165        Ok(Self {
166            tracer,
167            chroot,
168            _materialization_scope: materialization_scope,
169        })
170    }
171
172    /// Waits for the replay to finish and returns its exit status.
173    pub async fn wait(self) -> Result<ExitStatus, reverie::Error> {
174        let (exit_status, global_state) = self.tracer.wait().await?;
175        self.chroot.remove()?;
176        global_state.clean_up(false, &None).await;
177        Ok(exit_status)
178    }
179
180    /// Waits for the replay to finish and collects its output.
181    pub async fn wait_with_output(self) -> Result<Output, reverie::Error> {
182        let (output, global_state) = self.tracer.wait_with_output().await?;
183        self.chroot.remove()?;
184        global_state.clean_up(false, &None).await;
185        Ok(output)
186    }
187}

189fn prepare_replay_mounts(chroot: &TempChroot, mounts: &[Mount]) -> io::Result<Vec<Mount>> {
190    mounts
191        .iter()
192        .map(|mount| {
193            let target = chroot.relpath(mount.get_target());
194            let source_is_file = match mount.get_source() {
195                Some(source) => fs::metadata(source)?.is_file(),
196                None => false,
197            };
198            if source_is_file {
199                if let Some(parent) = target.parent() {
200                    fs::create_dir_all(parent)?;
201                }
202                match fs::symlink_metadata(&target) {
203                    Ok(metadata) if metadata.file_type().is_file() => {}
204                    Ok(_) => {
205                        return Err(io::Error::new(
206                            io::ErrorKind::AlreadyExists,
207                            format!(
208                                "replay mount target {} is not a regular file",
209                                target.display()
210                            ),
211                        ));
212                    }
213                    Err(error) if error.kind() == io::ErrorKind::NotFound => {
214                        fs::OpenOptions::new()
215                            .write(true)
216                            .create_new(true)
217                            .open(&target)?;
218                    }
219                    Err(error) => return Err(error),
220                }
221            } else {
222                fs::create_dir_all(&target)?;
223            }
224            Ok(mount.clone().target(target))
225        })
226        .collect()
227}

229/// Creates the temporary chroot directory.
230fn prepare_chroot(
231    dir: &Path,
232    metadata: &Metadata,
233    mounts: &[Mount],
234) -> io::Result<(
235    TempChroot,
236    PathBuf,
237    crate::replayer::ReplayMaterializationScope,
238    Vec<Mount>,
239)> {
240    let chroot = TempChroot::new_in(dir)?;
241
242    // Mount targets must be created before recorded paths install symlinks.
243    // prepare_replay_mounts uses controller-side path traversal, which is safe
244    // only while this new chroot is empty; afterward a recorded absolute
245    // symlink could redirect target creation outside the replay root.
246    let replay_mounts = prepare_replay_mounts(&chroot, mounts)?;
247    let fbcode = Path::new("/usr/local/fbcode");
248    if fbcode.exists() {
249        chroot.create_dir_all(fbcode)?;
250    }
251    let replay_root = crate::record_replay_path::open_directory_path(chroot.path())?;
252    let materialization_scope =
253        crate::replayer::ReplayMaterializationScope::new(replay_root.as_raw_fd())?;
254
255    // These are replay-owned mount/alias points, so create them while the
256    // chroot is still empty. No recorded symlink topology has been installed
257    // yet, making TempChroot's path-based helpers safe here.
258    chroot.create_dir_all(Path::new("/proc"))?;
259    chroot.create_dir_all(Path::new("/dev"))?;
260    chroot.symlink(Path::new("/proc/self/fd"), Path::new("/dev/fd"))?;
261
262    let bootstrap = read_bootstrap_exec_event(dir, metadata)?;
263    let ExecTarget::Materialize(target) = &bootstrap.target else {
264        return Err(io::Error::new(
265            io::ErrorKind::InvalidData,
266            "recorded bootstrap exec does not contain materializable path topology",
267        ));
268    };
269    let target_path = PathBuf::from(OsString::from_vec(target.path.clone()));
270    let root_snapshot = fs::canonicalize(
271        dir.join(EXEC_FILES_NAME)
272            .join(bootstrap.executable.digest.to_string()),
273    )?;
274    let canonical_chroot = fs::canonicalize(chroot.path())?;
275    let controller_relative =
276        padded_bootstrap_path(&canonical_chroot, bootstrap.request.path.len())?;
277    let bootstrap_program = canonical_chroot.join(&controller_relative);
278
279    // `TracerBuilder` validates the program before entering the chroot, so use
280    // a padded private copy backed by the immutable recording snapshot. The
281    // replayer overwrites this trapped pathname buffer with the recorded path
282    // before Linux sees the exec, so kernel-built argv/auxv/stack state remains
283    // identical to recording.
284    stage_recorded_exec_image(
285        &replay_root,
286        &replay_root,
287        &root_snapshot,
288        &controller_relative,
289        &[],
290        &bootstrap.executable,
291        true,
292    )?;
293    stage_recorded_exec_image(
294        &replay_root,
295        &replay_root,
296        &root_snapshot,
297        &bootstrap_program,
298        &[],
299        &bootstrap.executable,
300        true,
301    )?;
302    stage_recorded_exec_image(
303        &replay_root,
304        &replay_root,
305        &root_snapshot,
306        &target_path,
307        &target.symlinks,
308        &bootstrap.executable,
309        false,
310    )?;
311
312    crate::record_replay_path::ensure_directory_path_follow_final(
313        &replay_root,
314        &replay_root,
315        &metadata.current_dir,
316    )?;
317    let replay_cwd = crate::record_replay_path::resolve_existing_path(
318        &replay_root,
319        &replay_root,
320        &metadata.current_dir,
321        false,
322    )?
323    .object;
324    for dependency in &bootstrap.dependencies {
325        let path = PathBuf::from(OsString::from_vec(dependency.path.clone()));
326        let start = match &dependency.base {
327            crate::event::ExecMaterializationBase::Root => &replay_root,
328            crate::event::ExecMaterializationBase::Cwd => &replay_cwd,
329            crate::event::ExecMaterializationBase::DirectoryFd(fd) => {
330                return Err(io::Error::new(
331                    io::ErrorKind::InvalidData,
332                    format!("bootstrap dependency requires unavailable directory descriptor {fd}"),
333                ));
334            }
335        };
336        stage_recorded_exec_image(
337            &replay_root,
338            start,
339            &dir.join(EXEC_FILES_NAME)
340                .join(dependency.image.digest.to_string()),
341            &path,
342            &dependency.symlinks,
343            &dependency.image,
344            false,
345        )?;
346    }
347
348    ensure_replay_standard_directories(&replay_root)?;
349
350    Ok((
351        chroot,
352        bootstrap_program,
353        materialization_scope,
354        replay_mounts,
355    ))
356}

358/// Chooses a controller-visible pathname whose byte length is at least the
359/// recorded exec pathname, allowing the trapped pathname buffer to be safely
360/// overwritten in place before kernel injection. Every component remains
361/// within Linux NAME_MAX and the complete path remains below PATH_MAX.
362fn padded_bootstrap_path(chroot: &Path, minimum_absolute_len: usize) -> io::Result<PathBuf> {
363    const COMPONENT_LIMIT: usize = 255;
364    const FINAL_COMPONENT_MINIMUM: usize = 3;
365
366    let mut relative = PathBuf::from(".hermit-record-replay-bootstrap");
367    loop {
368        let prefix_len = chroot.join(&relative).as_os_str().as_bytes().len() + 1;
369        let final_len = minimum_absolute_len
370            .saturating_sub(prefix_len)
371            .max(FINAL_COMPONENT_MINIMUM);
372        if final_len <= COMPONENT_LIMIT {
373            relative.push("e".repeat(final_len));
374            break;
375        }
376        let directory_len = (final_len - COMPONENT_LIMIT).min(COMPONENT_LIMIT);
377        relative.push("p".repeat(directory_len));
378    }
379    let absolute_len = chroot.join(&relative).as_os_str().as_bytes().len();
380    if absolute_len < minimum_absolute_len || absolute_len >= libc::PATH_MAX as usize {
381        return Err(io::Error::new(
382            io::ErrorKind::InvalidInput,
383            format!(
384                "cannot construct padded replay bootstrap path of at least {minimum_absolute_len} bytes"
385            ),
386        ));
387    }
388    Ok(relative)
389}

410/// Reads the root thread's first successful exec without consuming the stream
411/// used by the runtime replayer. Valid pre-exec events remain in that independent
412/// runtime stream. The first successful exec is validated immediately so a later
413/// exec cannot mask a corrupt bootstrap record.
414fn read_bootstrap_exec_event(dir: &Path, metadata: &Metadata) -> io::Result<ExecEvent> {
415    let mut events = EventReader::open(dir, &EventStreamId::root())?;
416    let mut count = 0_u64;
417    let exec = loop {
418        let event = events.next_event().map_err(|error| {
419            io::Error::new(
420                io::ErrorKind::InvalidData,
421                format!("no successful root exec after {count} events: {error}"),
422            )
423        })?;
424        count = count.checked_add(1).ok_or_else(|| {
425            io::Error::new(
426                io::ErrorKind::InvalidData,
427                "root event count overflow before successful exec",
428            )
429        })?;
430        if let Ok(SyscallEvent::Exec(exec)) = event.event {
431            break exec;
432        }
433    };
434    let expected_path = metadata.exe.as_os_str().as_bytes();
435    if exec.request.dirfd != libc::AT_FDCWD
436        || exec.request.flags != 0
437        || exec.request.path != expected_path
438    {
439        return Err(io::Error::new(
440            io::ErrorKind::InvalidData,
441            format!(
442                "recorded bootstrap exec request does not match metadata executable {:?}",
443                metadata.exe
444            ),
445        ));
446    }
447    match &exec.target {
448        ExecTarget::Materialize(materialization)
449            if matches!(
450                &materialization.base,
451                crate::event::ExecMaterializationBase::Root
452            ) && materialization.path == expected_path => {}
453        _ => {
454            return Err(io::Error::new(
455                io::ErrorKind::InvalidData,
456                "recorded bootstrap exec is missing root-relative path topology",
457            ));
458        }
459    }
460    Ok(exec)
461}

463/// Materializes one immutable recorded image at its original guest pathname.
464/// The source is opened without following symlinks and verified by digest;
465/// destination traversal is confined to the pinned replay root and follows
466/// only the symlinks captured in the bootstrap event.
467fn stage_recorded_exec_image(
468    replay_root: &OwnedFd,
469    start: &OwnedFd,
470    source_path: &Path,
471    guest_path: &Path,
472    symlinks: &[crate::record_replay_path::ResolvedSymlink],
473    image: &ExecImage,
474    require_absent: bool,
475) -> io::Result<()> {
476    if guest_path.as_os_str().is_empty() {
477        return Err(io::Error::new(
478            io::ErrorKind::InvalidInput,
479            "bootstrap replay path is empty",
480        ));
481    }
482    let mut source = OpenOptions::new()
483        .read(true)
484        .custom_flags(libc::O_CLOEXEC | libc::O_NOFOLLOW)
485        .open(source_path)?;
486    let metadata = source.metadata()?;
487    if !metadata.file_type().is_file() {
488        return Err(io::Error::new(
489            io::ErrorKind::InvalidData,
490            format!("bootstrap replay source is not a regular file: {source_path:?}"),
491        ));
492    }
493    let digest = detcore::Digest::digest_reader(&mut source)?;
494    if digest != image.digest {
495        return Err(io::Error::new(
496            io::ErrorKind::InvalidData,
497            format!(
498                "recorded bootstrap image digest mismatch: expected {}, found {}",
499                image.digest, digest
500            ),
501        ));
502    }
503    source.seek(SeekFrom::Start(0))?;
504    if require_absent {
505        match crate::record_replay_path::resolve_existing_path(
506            replay_root,
507            start,
508            guest_path,
509            false,
510        ) {
511            Err(error) if error.kind() == io::ErrorKind::NotFound => {}
512            Err(error) => return Err(error),
513            Ok(_) => {
514                return Err(io::Error::new(
515                    io::ErrorKind::AlreadyExists,
516                    format!("private replay bootstrap path already exists: {guest_path:?}"),
517                ));
518            }
519        }
520    }
521    crate::record_replay_path::materialize_regular_file(
522        replay_root,
523        start,
524        guest_path,
525        symlinks,
526        &mut source,
527        image.digest,
528        image.mode,
529    )?;
530    let staged =
531        crate::record_replay_path::resolve_existing_path(replay_root, start, guest_path, false)?;
532    crate::replayer::remember_materialized_object(
533        replay_root.as_raw_fd(),
534        staged.object.as_raw_fd(),
535    )
536}
