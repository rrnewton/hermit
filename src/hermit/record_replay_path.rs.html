80/// Names the tracee thread's current working directory as the thread itself
81/// would see it: the procfs cwd link with the thread's root prefix removed.
82/// Recording keeps this resolved name for a successful `chdir`, so replay can
83/// enter the same directory even when the spelled path went through a host
84/// symlink that the replay root does not contain.
85pub(crate) fn process_cwd_path(tid: Pid) -> io::Result<PathBuf> {
86    // procfs appends " (deleted)" to a removed directory's name. That string is
87    // not a path, and replay cannot reproduce a deleted working directory.
88    let cwd_fd = open_process_cwd(tid)?;
89    let mut stat = MaybeUninit::<libc::stat>::uninit();
90    if unsafe { libc::fstat(cwd_fd.as_raw_fd(), stat.as_mut_ptr()) } != 0 {
91        return Err(io::Error::last_os_error());
92    }
93    if unsafe { stat.assume_init() }.st_nlink == 0 {
94        return Err(io::Error::other(format!(
95            "working directory of thread {tid} has been removed"
96        )));
97    }
98    let root = std::fs::read_link(format!("/proc/{}/root", tid.as_raw()))?;
99    let cwd = std::fs::read_link(format!("/proc/{}/cwd", tid.as_raw()))?;
100    let relative = cwd.strip_prefix(&root).map_err(|_| {
101        io::Error::other(format!(
102            "tracee cwd {cwd:?} is not beneath its root {root:?}"
103        ))
104    })?;
105    Ok(Path::new("/").join(relative))
106}

135fn open_proc_path(path: &str, flags: libc::c_int) -> io::Result<OwnedFd> {
136    let path = CString::new(path).expect("proc descriptor path cannot contain NUL");
137    // Intentionally follow the procfs magic link: the returned fd pins the
138    // tracee object, after which all confinement checks are descriptor based.
139    let fd = unsafe { libc::open(path.as_ptr(), flags | libc::O_CLOEXEC) };
140    owned_fd(fd)
141}
142
143fn owned_fd(fd: RawFd) -> io::Result<OwnedFd> {
144    if fd < 0 {
145        Err(io::Error::last_os_error())
146    } else {
147        // SAFETY: a nonnegative return from open/openat/fcntl is a new owned fd.
148        Ok(unsafe { OwnedFd::from_raw_fd(fd) })
149    }
150}

152fn duplicate_fd(fd: RawFd) -> io::Result<OwnedFd> {
153    owned_fd(unsafe { libc::fcntl(fd, libc::F_DUPFD_CLOEXEC, 3) })
154}

188fn mkdir_at(parent: RawFd, component: &OsStr) -> io::Result<()> {
189    let component = component_cstring(component)?;
190    let result = unsafe { libc::mkdirat(parent, component.as_ptr(), 0o777) };

198fn open_or_create_directory_at(parent: RawFd, component: &OsStr) -> io::Result<OwnedFd> {
199    match open_directory_at(parent, component) {
200        Ok(directory) => Ok(directory),
201        Err(error) if error.kind() == io::ErrorKind::NotFound => {
202            match mkdir_at(parent, component) {
203                Ok(()) => {}
204                Err(error) if error.kind() == io::ErrorKind::AlreadyExists => {}
205                Err(error) => return Err(error),
206            }
207            open_directory_at(parent, component)
208        }
209        Err(error) => Err(error),
210    }
211}

213fn read_link_at(parent: RawFd, component: &OsStr) -> io::Result<OsString> {
214    let component = component_cstring(component)?;
215    let mut buffer = vec![0u8; 256];
216    loop {
217        let length = unsafe {
218            libc::readlinkat(
219                parent,
220                component.as_ptr(),
221                buffer.as_mut_ptr().cast(),
222                buffer.len(),
223            )
224        };
225        if length < 0 {
226            return Err(io::Error::last_os_error());
227        }
228        let length = length as usize;
229        if length < buffer.len() {
230            buffer.truncate(length);
231            return Ok(OsString::from_vec(buffer));
232        }
233        buffer.resize(buffer.len() * 2, 0);
234    }
235}

237fn create_symlink_at(parent: RawFd, component: &OsStr, target: &OsStr) -> io::Result<()> {
238    let component = component_cstring(component)?;
239    let target = component_cstring(target)?;
240    let result = unsafe { libc::symlinkat(target.as_ptr(), parent, component.as_ptr()) };
241    if result == 0 {
242        Ok(())
243    } else {
244        Err(io::Error::last_os_error())
245    }
246}

258fn directory_identity(fd: RawFd) -> io::Result<(libc::dev_t, libc::ino_t)> {
259    let mut stat = MaybeUninit::<libc::stat>::zeroed();
260    if unsafe { libc::fstat(fd, stat.as_mut_ptr()) } != 0 {
261        return Err(io::Error::last_os_error());
262    }
263    // SAFETY: fstat initialized the complete stat structure on success.
264    let stat = unsafe { stat.assume_init() };
265    Ok((stat.st_dev, stat.st_ino))
266}

293/// Whether `directory` is on procfs, whose symlinks include "magic" links
294/// such as `/proc/<pid>/root` and `/proc/<pid>/fd/<n>`. Their text is not
295/// where Linux resolves them: the kernel jumps to the linked object itself,
296/// which may be outside the replay root, so text resolution cannot confine them.
297fn is_on_procfs(directory: RawFd) -> io::Result<bool> {
298    let mut filesystem = MaybeUninit::<libc::statfs>::zeroed();
299    if unsafe { libc::fstatfs(directory, filesystem.as_mut_ptr()) } != 0 {
300        return Err(io::Error::last_os_error());
301    }
302    // SAFETY: fstatfs initialized the complete statfs structure on success.
303    let filesystem = unsafe { filesystem.assume_init() };
304    Ok(filesystem.f_type == libc::PROC_SUPER_MAGIC)
305}
306
307/// Refuses to follow a symlink in a procfs directory, as
308/// `RESOLVE_NO_MAGICLINKS` does, with the same ELOOP.
309fn refuse_procfs_symlink(directory: RawFd) -> io::Result<()> {
310    if is_on_procfs(directory)? {
311        return Err(io::Error::from_raw_os_error(libc::ELOOP));
312    }
313    Ok(())
314}

320/// Reopens an O_PATH descriptor for reading without consulting its original
321/// pathname. This preserves the selected object across rename/unlink races.
322pub(crate) fn open_readable_fd(fd: RawFd) -> io::Result<File> {
323    let path = CString::new(format!("/proc/self/fd/{fd}"))
324        .expect("controller descriptor path cannot contain NUL");
325    let readable = unsafe { libc::open(path.as_ptr(), libc::O_RDONLY | libc::O_CLOEXEC) };
326    owned_fd(readable).map(File::from)
327}

333/// Checks ancestry using only pinned directory descriptors. No pathname string
334/// is used as a security or confinement boundary.
335pub(crate) fn directory_is_beneath(root: &OwnedFd, directory: &OwnedFd) -> io::Result<bool> {
336    let mut current = duplicate_fd(directory.as_raw_fd())?;
337    for _ in 0..MAX_DIRECTORY_DEPTH {
338        if same_directory(current.as_raw_fd(), root.as_raw_fd())? {
339            return Ok(true);
340        }
341        let parent = open_directory_at(current.as_raw_fd(), OsStr::new(".."))?;
342        if same_directory(current.as_raw_fd(), parent.as_raw_fd())? {
343            return Ok(false);
344        }
345        current = parent;
346    }
347    Err(io::Error::new(
348        io::ErrorKind::InvalidData,
349        "directory ancestry exceeded safety limit",
350    ))
351}

353fn append_components(queue: &mut VecDeque<PendingComponent>, path: &Path) -> io::Result<()> {
354    for component in path.components().rev() {
355        let component = match component {
356            Component::RootDir => PendingComponent::Root,
357            Component::CurDir => continue,
358            Component::ParentDir => PendingComponent::Parent,
359            Component::Normal(component) => PendingComponent::Normal(component.to_os_string()),
360            Component::Prefix(_) => {
361                return Err(io::Error::new(
362                    io::ErrorKind::InvalidInput,
363                    "non-Unix guest pathname",
364                ));
365            }
366        };
367        queue.push_front(component);
368    }
369    Ok(())
370}

372fn initial_directory(root: &OwnedFd, start: &OwnedFd, path: &Path) -> io::Result<OwnedFd> {
373    if !path.is_absolute() && !directory_is_beneath(root, start)? {
374        return Err(io::Error::new(
375            io::ErrorKind::PermissionDenied,
376            "relative pathname base is outside replay root",
377        ));
378    }
379    if path.is_absolute() {
380        duplicate_fd(root.as_raw_fd())
381    } else {
382        duplicate_fd(start.as_raw_fd())
383    }
384}
385
386fn parent_directory(root: &OwnedFd, current: &OwnedFd) -> io::Result<OwnedFd> {
387    if same_directory(current.as_raw_fd(), root.as_raw_fd())? {
388        return duplicate_fd(root.as_raw_fd());
389    }
390    let parent = open_directory_at(current.as_raw_fd(), OsStr::new(".."))?;
391    if !directory_is_beneath(root, &parent)? {
392        return Err(io::Error::new(
393            io::ErrorKind::PermissionDenied,
394            "pathname traversal escaped replay root",
395        ));
396    }
397    Ok(parent)
398}

400/// Resolves an existing pathname using descriptor-only traversal. Symlinks are
401/// followed in the same order as Linux, so `link/..` first traverses `link` and
402/// only then applies the parent component. The final symlink can optionally be
403/// returned without following it for `AT_SYMLINK_NOFOLLOW` classification.
404pub(crate) fn resolve_existing_path(
405    root: &OwnedFd,
406    start: &OwnedFd,
407    path: &Path,
408    nofollow_final: bool,
409) -> io::Result<ResolvedPath> {
410    resolve_existing_path_impl(root, start, path, nofollow_final, false)
411}
412
413/// As [`resolve_existing_path`], but fails with ELOOP instead of following a
414/// symlink on procfs. Use it when the answer decides whether the guest's own
415/// kernel may perform a real filesystem operation on the path: the kernel
416/// follows procfs magic links to their objects, which this text resolution
417/// would place inside the replay root.
418pub(crate) fn resolve_existing_path_without_procfs_symlinks(
419    root: &OwnedFd,
420    start: &OwnedFd,
421    path: &Path,
422) -> io::Result<ResolvedPath> {
423    resolve_existing_path_impl(root, start, path, false, true)
424}

426fn resolve_existing_path_impl(
427    root: &OwnedFd,
428    start: &OwnedFd,
429    path: &Path,
430    nofollow_final: bool,
431    refuse_procfs_symlinks: bool,
432) -> io::Result<ResolvedPath> {
433    if path.as_os_str().is_empty() {
434        return Err(io::Error::from_raw_os_error(libc::ENOENT));
435    }
436
437    let mut current = initial_directory(root, start, path)?;
438    let mut pending = VecDeque::new();
439    append_components(&mut pending, path)?;
440    let mut followed_symlinks = 0usize;
441    let mut lookup_index = 0u64;
442    let mut symlinks = Vec::new();
443
444    while let Some(component) = pending.pop_front() {
445        match component {
446            PendingComponent::Root => current = duplicate_fd(root.as_raw_fd())?,
447            PendingComponent::Parent => current = parent_directory(root, &current)?,
448            PendingComponent::Normal(component) => {
449                lookup_index = lookup_index.checked_add(1).ok_or_else(|| {
450                    io::Error::new(io::ErrorKind::InvalidData, "pathname lookup count overflow")
451                })?;
452                let object = open_path_at(current.as_raw_fd(), &component)?;
453                let identity = file_identity(object.as_raw_fd())?;
454                let final_component = pending.is_empty();
455                if is_symlink(identity) && !(final_component && nofollow_final) {
456                    if refuse_procfs_symlinks {
457                        refuse_procfs_symlink(current.as_raw_fd())?;
458                    }
459                    followed_symlinks += 1;
460                    if followed_symlinks > MAX_SYMLINKS {
461                        return Err(io::Error::from_raw_os_error(libc::ELOOP));
462                    }
463                    let target = read_link_at(current.as_raw_fd(), &component)?;
464                    symlinks.push(ResolvedSymlink {
465                        lookup_index,
466                        target: target.as_bytes().to_vec(),
467                    });
468                    append_components(&mut pending, Path::new(&target))?;
469                    continue;
470                }
471                if final_component {
472                    return Ok(ResolvedPath { object, symlinks });
473                }
474                if !is_directory(identity) {
475                    return Err(io::Error::from_raw_os_error(libc::ENOTDIR));
476                }
477                current = object;
478            }
479        }
480    }
481
482    Ok(ResolvedPath {
483        object: current,
484        symlinks,
485    })
486}

488fn ensure_expected_symlink(
489    parent: RawFd,
490    component: &OsStr,
491    expected: &ResolvedSymlink,
492) -> io::Result<OsString> {
493    let expected_target = OsString::from_vec(expected.target.clone());
494    match read_link_at(parent, component) {
495        Ok(actual) if actual == expected_target => Ok(actual),
496        Ok(actual) => Err(io::Error::new(
497            io::ErrorKind::InvalidData,
498            format!(
499                "recorded exec symlink target mismatch at {component:?}: expected {expected_target:?}, found {actual:?}"
500            ),
501        )),
502        Err(error) if error.kind() == io::ErrorKind::NotFound => {
503            create_symlink_at(parent, component, &expected_target)?;
504            Ok(expected_target)
505        }
506        Err(error) => Err(io::Error::new(
507            io::ErrorKind::InvalidData,
508            format!("recorded exec symlink collides at {component:?}: {error}"),
509        )),
510    }
511}

513fn temporary_name() -> OsString {
514    let sequence = TEMPORARY_FILE_SEQUENCE.fetch_add(1, Ordering::Relaxed);
515    OsString::from(format!(".hermit-exec-{}-{sequence}", std::process::id()))
516}

518fn create_temporary_file(parent: RawFd) -> io::Result<(OsString, OwnedFd)> {
519    for _ in 0..128 {
520        let name = temporary_name();
521        let name_c = component_cstring(&name)?;
522        let fd = unsafe {
523            libc::openat(
524                parent,
525                name_c.as_ptr(),
526                libc::O_WRONLY | libc::O_CREAT | libc::O_EXCL | libc::O_NOFOLLOW | libc::O_CLOEXEC,
527                0o600,
528            )
529        };
530        match owned_fd(fd) {
531            Ok(fd) => return Ok((name, fd)),
532            Err(error) if error.kind() == io::ErrorKind::AlreadyExists => continue,
533            Err(error) => return Err(error),
534        }
535    }
536    Err(io::Error::new(
537        io::ErrorKind::AlreadyExists,
538        "could not allocate confined exec staging file",
539    ))
540}

542/// Creates a collision-free regular file directly beneath a pinned replay
543/// root. The returned component can be opened by the stopped guest and then
544/// unlinked without ever resolving a controller pathname through guest-created
545/// symlinks.
546pub(crate) fn create_root_temporary_regular_file(
547    root: &OwnedFd,
548    source: &mut File,
549    expected_digest: detcore::Digest,
550    mode: u32,
551) -> io::Result<OsString> {
552    let (name, temporary_fd) = create_temporary_file(root.as_raw_fd())?;
553    let result = (|| {
554        source.seek(SeekFrom::Start(0))?;
555        let mut output = File::from(temporary_fd);
556        io::copy(source, &mut output)?;
557        if unsafe { libc::fchmod(output.as_raw_fd(), mode as libc::mode_t) } != 0 {
558            return Err(io::Error::last_os_error());
559        }
560        output.sync_all()?;
561        let actual = detcore::Digest::digest_reader(open_readable_fd(output.as_raw_fd())?)?;
562        if actual != expected_digest {
563            return Err(io::Error::new(
564                io::ErrorKind::InvalidData,
565                format!(
566                    "temporary recorded exec digest mismatch: expected {expected_digest}, found {actual}"
567                ),
568            ));
569        }
570        Ok(())
571    })();
572    if let Err(error) = result {
573        let _ = unlink_at(root.as_raw_fd(), &name);
574        return Err(error);
575    }
576    Ok(name)
577}

583fn replace_regular_file_at(
584    parent: RawFd,
585    component: &OsStr,
586    source: &mut File,
587    mode: u32,
588) -> io::Result<()> {
589    let (temporary_name, temporary_fd) = create_temporary_file(parent)?;
590    let result = (|| {
591        source.seek(SeekFrom::Start(0))?;
592        let mut output = File::from(temporary_fd);
593        io::copy(source, &mut output)?;
594        if unsafe { libc::fchmod(output.as_raw_fd(), mode as libc::mode_t) } != 0 {
595            return Err(io::Error::last_os_error());
596        }
597        output.sync_all()?;
598        let temporary = component_cstring(&temporary_name)?;
599        let destination = component_cstring(component)?;
600        if unsafe { libc::renameat(parent, temporary.as_ptr(), parent, destination.as_ptr()) } != 0
601        {
602            return Err(io::Error::last_os_error());
603        }
604        Ok(())
605    })();
606    if result.is_err() {
607        let _ = unlink_at(parent, &temporary_name);
608    }
609    result
610}

612fn materialize_regular_file_at(
613    parent: RawFd,
614    component: &OsStr,
615    source: &mut File,
616    expected_digest: detcore::Digest,
617    mode: u32,
618) -> io::Result<()> {
619    match open_path_at(parent, component) {
620        Ok(existing) => {
621            let identity = file_identity(existing.as_raw_fd())?;
622            if !is_regular_file(identity) {
623                return Err(io::Error::new(
624                    io::ErrorKind::InvalidData,
625                    format!("recorded exec destination is not a regular file: {component:?}"),
626                ));
627            }
628            let actual_mode = identity.mode & 0o7777;
629            let digest = detcore::Digest::digest_reader(open_readable_fd(existing.as_raw_fd())?)?;
630            if digest == expected_digest && actual_mode == mode {
631                return Ok(());
632            }
633            replace_regular_file_at(parent, component, source, mode)
634        }
635        Err(error) if error.kind() == io::ErrorKind::NotFound => {
636            replace_regular_file_at(parent, component, source, mode)
637        }
638        Err(error) => Err(error),
639    }
640}

642/// Materializes one recorded regular file beneath a pinned replay root. Every
643/// parent component is opened with `O_NOFOLLOW`; only symlinks explicitly
644/// captured during recording are created/followed, and their targets must match
645/// byte-for-byte. Final replacement uses renameat relative to a pinned parent,
646/// so even a racing symlink cannot redirect writes outside the replay root.
647pub(crate) fn materialize_regular_file(
648    root: &OwnedFd,
649    start: &OwnedFd,
650    path: &Path,
651    symlinks: &[ResolvedSymlink],
652    source: &mut File,
653    expected_digest: detcore::Digest,
654    mode: u32,
655) -> io::Result<()> {
656    if path.as_os_str().is_empty() {
657        return Err(io::Error::from_raw_os_error(libc::ENOENT));
658    }
659    let mut current = initial_directory(root, start, path)?;
660    let mut pending = VecDeque::new();
661    append_components(&mut pending, path)?;
662    let mut symlink_index = 0usize;
663    let mut lookup_index = 0u64;
664    let mut followed_symlinks = 0usize;
665
666    while let Some(component) = pending.pop_front() {
667        match component {
668            PendingComponent::Root => current = duplicate_fd(root.as_raw_fd())?,
669            PendingComponent::Parent => current = parent_directory(root, &current)?,
670            PendingComponent::Normal(component) => {
671                lookup_index = lookup_index.checked_add(1).ok_or_else(|| {
672                    io::Error::new(io::ErrorKind::InvalidData, "pathname lookup count overflow")
673                })?;
674                if let Some(expected) = symlinks.get(symlink_index)
675                    && expected.lookup_index == lookup_index
676                {
677                    let target =
678                        ensure_expected_symlink(current.as_raw_fd(), &component, expected)?;
679                    symlink_index += 1;
680                    followed_symlinks += 1;
681                    if followed_symlinks > MAX_SYMLINKS {
682                        return Err(io::Error::from_raw_os_error(libc::ELOOP));
683                    }
684                    append_components(&mut pending, Path::new(&target))?;
685                    continue;
686                }
687                if symlinks
688                    .get(symlink_index)
689                    .is_some_and(|expected| expected.lookup_index < lookup_index)
690                {
691                    return Err(io::Error::new(
692                        io::ErrorKind::InvalidData,
693                        "recorded exec symlink lookup order is invalid",
694                    ));
695                }
696                if pending.is_empty() {
697                    materialize_regular_file_at(
698                        current.as_raw_fd(),
699                        &component,
700                        source,
701                        expected_digest,
702                        mode,
703                    )?;
704                    if symlink_index != symlinks.len() {
705                        return Err(io::Error::new(
706                            io::ErrorKind::InvalidData,
707                            "recorded exec has unused symlink topology",
708                        ));
709                    }
710                    return Ok(());
711                }
712                current = match open_or_create_directory_at(current.as_raw_fd(), &component) {
713                    Ok(directory) => directory,
714                    Err(error) => {
715                        if read_link_at(current.as_raw_fd(), &component).is_ok() {
716                            return Err(io::Error::new(
717                                io::ErrorKind::InvalidData,
718                                format!(
719                                    "unexpected symlink in recorded exec destination at {component:?}"
720                                ),
721                            ));
722                        }
723                        return Err(error);
724                    }
725                };
726            }
727        }
728    }
729
730    Err(io::Error::new(
731        io::ErrorKind::InvalidInput,
732        "recorded exec path did not contain a file component",
733    ))
734}

736/// Ensures that `path` names a directory beneath `root`, interpreting each
737/// component with Linux-like symlink and `..` ordering. Intermediate symlinks
738/// are followed; the final component must be a directory rather than a symlink.
739/// Missing directory components are created relative to pinned parent fds.
740pub(crate) fn ensure_directory_path(
741    root: &OwnedFd,
742    start: &OwnedFd,
743    path: &Path,
744) -> io::Result<()> {
745    ensure_directory_path_impl(root, start, path, false, false)
746}
747
748/// As [`ensure_directory_path`], but follows an existing final symlink. This
749/// matches `chdir(2)` and mount-target lookup, which resolve the final
750/// component, while retaining descriptor-rooted confinement.
751pub(crate) fn ensure_directory_path_follow_final(
752    root: &OwnedFd,
753    start: &OwnedFd,
754    path: &Path,
755) -> io::Result<()> {
756    ensure_directory_path_impl(root, start, path, true, false)
757}
758
759/// As [`ensure_directory_path_follow_final`], but fails with ELOOP instead of
760/// following a symlink on procfs (see
761/// [`resolve_existing_path_without_procfs_symlinks`]).
762pub(crate) fn ensure_directory_path_follow_final_without_procfs_symlinks(
763    root: &OwnedFd,
764    start: &OwnedFd,
765    path: &Path,
766) -> io::Result<()> {
767    ensure_directory_path_impl(root, start, path, true, true)
768}
769
770fn ensure_directory_path_impl(
771    root: &OwnedFd,
772    start: &OwnedFd,
773    path: &Path,
774    follow_final_symlink: bool,
775    refuse_procfs_symlinks: bool,
776) -> io::Result<()> {
777    if !path.is_absolute() && !directory_is_beneath(root, start)? {
778        return Err(io::Error::new(
779            io::ErrorKind::PermissionDenied,
780            "relative directory base is outside replay root",
781        ));
782    }
783
784    let mut current = if path.is_absolute() {
785        duplicate_fd(root.as_raw_fd())?
786    } else {
787        duplicate_fd(start.as_raw_fd())?
788    };
789    let mut pending = VecDeque::new();
790    append_components(&mut pending, path)?;
791    let mut followed_symlinks = 0usize;
792
793    while let Some(component) = pending.pop_front() {
794        match component {
795            PendingComponent::Root => current = duplicate_fd(root.as_raw_fd())?,
796            PendingComponent::Parent => {
797                if same_directory(current.as_raw_fd(), root.as_raw_fd())? {
798                    continue;
799                }
800                let parent = open_directory_at(current.as_raw_fd(), OsStr::new(".."))?;
801                if !directory_is_beneath(root, &parent)? {
802                    return Err(io::Error::new(
803                        io::ErrorKind::PermissionDenied,
804                        "directory traversal escaped replay root",
805                    ));
806                }
807                current = parent;
808            }
809            PendingComponent::Normal(component) => {
810                match open_or_create_directory_at(current.as_raw_fd(), &component) {
811                    Ok(directory) => current = directory,
812                    Err(open_error) => {
813                        let target = match read_link_at(current.as_raw_fd(), &component) {
814                            Ok(target) if !pending.is_empty() || follow_final_symlink => target,
815                            Ok(_) => {
816                                return Err(io::Error::new(
817                                    io::ErrorKind::InvalidData,
818                                    "final replay directory component is a symlink",
819                                ));
820                            }
821                            Err(_) => return Err(open_error),
822                        };
823                        if refuse_procfs_symlinks {
824                            refuse_procfs_symlink(current.as_raw_fd())?;
825                        }
826                        followed_symlinks += 1;
827                        if followed_symlinks > MAX_SYMLINKS {
828                            return Err(io::Error::from_raw_os_error(libc::ELOOP));
829                        }
830                        append_components(&mut pending, Path::new(&target))?;
831                    }
832                }
833            }
834        }
835    }
836    Ok(())
837}
