30/// Stable event-stream identity derived from the guest process tree.
31///
32/// Each component after the root is the parent's deterministic child ordinal.
33/// Unlike a process-local counter, this remains collision-free when Reverie
34/// forks the tool itself and each process receives its own tool instance.
35#[derive(Clone, Debug, Default, Eq, PartialEq, Serialize, Deserialize)]
36pub(crate) struct EventStreamId(Vec<u64>);
37
38impl EventStreamId {
39    pub(crate) fn root() -> Self {
40        Self(Vec::new())
41    }
42
43    pub(crate) fn child(&self, ordinal: u64) -> Self {
44        let mut path = self.0.clone();
45        path.push(ordinal);
46        Self(path)
47    }
48
49    fn file_name(&self) -> String {
50        let mut identity = b"hermit-event-stream-v1\0".to_vec();
51        for ordinal in &self.0 {
52            identity.extend_from_slice(&ordinal.to_be_bytes());
53        }
54        format!("stream-{}", detcore::Digest::new(&identity))
55    }
56}
57
58impl fmt::Display for EventStreamId {
59    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
60        write!(f, "{}", detcore::ROOT_DETPID.as_raw())?;
61        for ordinal in &self.0 {
62            write!(f, ".{ordinal}")?;
63        }
64        Ok(())
65    }
66}

68/// Allocates child ordinals within one deterministic parent stream.
69///
70/// The counter belongs to the parent thread state rather than the Recorder or
71/// Replayer process instance. When Reverie forks a tool process, the child
72/// receives its already-assigned stream identity and a fresh descendant
73/// counter, while the continuing parent retains the incremented counter.
74#[derive(Default)]
75pub(crate) struct ChildEventStreamIds(AtomicU64);
76
77impl ChildEventStreamIds {
78    pub(crate) fn next(&self, parent: &EventStreamId) -> EventStreamId {
79        let ordinal = self
80            .0
81            .try_update(Ordering::Relaxed, Ordering::Relaxed, |value| {
82                value.checked_add(1)
83            })
84            .unwrap_or_else(|_| panic!("record/replay child stream ordinal overflow"));
85        parent.child(ordinal)
86    }
87}

89#[derive(Serialize, Deserialize)]
90struct EventStreamHeader {
91    stream_id: EventStreamId,
92}
93
94fn decode_stream_header(
95    reader: &mut io::BufReader<fs::File>,
96    expected: &EventStreamId,
97) -> io::Result<()> {
98    let header: EventStreamHeader =
99        bincode::serde::decode_from_std_read(reader, bincode::config::legacy()).map_err(
100            |error| {
101                io::Error::new(
102                    io::ErrorKind::InvalidData,
103                    format!("invalid event-stream identity header: {error}"),
104                )
105            },
106        )?;
107    if header.stream_id != *expected {
108        return Err(io::Error::new(
109            io::ErrorKind::InvalidData,
110            format!(
111                "event-stream identity mismatch: expected {expected}, found {}",
112                header.stream_id
113            ),
114        ));
115    }
116    Ok(())
117}
118
119fn encode_stream_header(
120    writer: &mut io::BufWriter<fs::File>,
121    stream_id: &EventStreamId,
122) -> io::Result<()> {
123    bincode::serde::encode_into_std_write(
124        EventStreamHeader {
125            stream_id: stream_id.clone(),
126        },
127        writer,
128        bincode::config::legacy(),
129    )
130    .map(|_| ())
131    .map_err(|error| {
132        io::Error::new(
133            io::ErrorKind::InvalidData,
134            format!("failed to encode event-stream identity header: {error}"),
135        )
136    })
137}

139impl Serialize for ChildEventStreamIds {
140    fn serialize<S>(&self, serializer: S) -> Result<S::Ok, S::Error>
141    where
142        S: serde::Serializer,
143    {
144        serializer.serialize_u64(self.0.load(Ordering::Relaxed))
145    }
146}
147
148impl<'de> Deserialize<'de> for ChildEventStreamIds {
149    fn deserialize<D>(deserializer: D) -> Result<Self, D::Error>
150    where
151        D: serde::Deserializer<'de>,
152    {
153        Ok(Self(AtomicU64::new(u64::deserialize(deserializer)?)))
154    }

157/// An event to help with debugging, but is not actually necessary for the
158/// functionality of record/replay.
159#[derive(Debug, Serialize, Deserialize)]
160pub struct DebugEvent {
161    /// The raw syscall.
162    syscall: (Sysno, SyscallArgs),
163
164    /// The pretty, displayable version of the syscall.
165    pretty: String,
166
167    /// Exec pathname bytes and lookup context. Raw syscall registers only hold
168    /// a pointer, so this snapshot prevents a changed failed exec request from
169    /// silently consuming an event that happened to return the same errno.
170    exec_request: Option<DebugExecRequest>,
171}
172
173#[derive(Debug, Eq, PartialEq, Serialize, Deserialize)]
174struct DebugExecRequest {
175    dirfd: libc::c_int,
176    path: Result<Vec<u8>, Errno>,
177    flags: libc::c_int,
178}
179
180impl DebugEvent {
181    /// Constructs a new `DebugEvent`.
182    pub fn new<M: MemoryAccess>(syscall: Syscall, memory: &M) -> Self {
183        let exec_request = match syscall {
184            Syscall::Execve(call) => {
185                let call: reverie::syscalls::Execveat = call.into();
186                Some(DebugExecRequest {
187                    dirfd: call.dirfd(),
188                    path: call
189                        .path()
190                        .ok_or(Errno::EFAULT)
191                        .and_then(|path| path.read(memory))
192                        .map(|path| path.as_os_str().as_bytes().to_vec()),
193                    flags: call.flags(),
194                })
195            }
196            Syscall::Execveat(call) => Some(DebugExecRequest {
197                dirfd: call.dirfd(),
198                path: call
199                    .path()
200                    .ok_or(Errno::EFAULT)
201                    .and_then(|path| path.read(memory))
202                    .map(|path| path.as_os_str().as_bytes().to_vec()),
203                flags: call.flags(),
204            }),
205            _ => None,
206        };
207        Self {
208            syscall: syscall.into_parts(),
209            pretty: format!("{}", syscall.display(memory)),
210            exec_request,
211        }
212    }
213
214    /// Returns the syscall associated with this debug event.
215    pub fn syscall(&self) -> Syscall {
216        Syscall::from_raw(self.syscall.0, self.syscall.1)
217    }
218
219    pub(crate) fn exec_request_matches(&self, other: &Self) -> bool {
220        self.exec_request == other.exec_request
221    }
222
223    #[cfg(test)]
224    pub(crate) fn for_test(sysno: Sysno, pretty: &str) -> Self {
225        Self {
226            syscall: (sysno, SyscallArgs::new(0, 0, 0, 0, 0, 0)),
227            pretty: pretty.to_owned(),
228            exec_request: None,
229        }
230    }
231}
232
233impl fmt::Display for DebugEvent {
234    fn fmt(&self, f: &mut fmt::Formatter) -> fmt::Result {
235        f.write_str(&self.pretty)
236    }
237}

239/// The number of argument registers the x86-64 kernel actually reads for
240/// `sysno`. Registers beyond this are not part of the syscall's ABI and can hold
241/// arbitrary leftover values in the guest, so they must be excluded when
242/// comparing a replayed syscall against its recorded counterpart.
243///
244/// Reverie stores all six raw registers in every typed syscall and derives
245/// `PartialEq` over them, so a naive `Syscall == Syscall` compares unused
246/// registers too. That produces false desyncs for any syscall with fewer than
247/// six arguments (e.g. `statfs`, which uses two).
248///
249/// Returns `None` for syscalls without an entry; callers then fall back to
250/// comparing all six registers (the conservative pre-existing behavior). Only
251/// syscalls that record/replay subscribes to (see `recorder::subscriptions`)
252/// flow through the comparator, so those are the ones covered here.
253///
254/// These are true kernel arities, which for a few syscalls exceed the number of
255/// typed fields reverie exposes (`open`, `openat`, `ioctl`, `socket`, and
256/// `fcntl` fold meaningful arguments into typed enums). Using the kernel arity
257/// -- not the typed field count -- guarantees we never zero a meaningful
258/// argument and therefore never mask a genuine divergence.
259fn kernel_arg_count(sysno: Sysno) -> Option<u8> {
260    use reverie::syscalls::Sysno::*;
261    Some(match sysno {
262        close | fchdir | dup | time | unlink => 1,
263        access | stat | fstat | lstat | dup2 | clock_gettime | gettimeofday | settimeofday
264        | mkdir | statfs | fstatfs | ftruncate | kill | listen | rt_sigpending => 2,
265        mprotect | read | readv | write | writev | lseek | getdents | getdents64 | dup3 | ioctl
266        | socket | fcntl | connect | sendmsg | poll | getpeername | getsockname | getrandom
267        | readlink | unlinkat | open | execve | close_range | tgkill => 3,
268        pread64 | pwrite64 | newfstatat | fadvise64 | openat => 4,
269        statx | pwritev | preadv | ppoll | setsockopt | getsockopt | execveat | prctl => 5,
270        recvfrom | sendto | pwritev2 | preadv2 | mmap => 6,
271        _ => return None,
272    })
273}
274
275/// Returns `syscall` with any argument registers beyond its kernel arity zeroed.
276/// This makes the record/replay desync comparison ignore unused registers, which
277/// are not part of the syscall and may legitimately differ between record and
278/// replay. Syscalls without a known arity are returned unchanged (all six
279/// registers still compared).
280pub(crate) fn normalize_unused_args(syscall: Syscall) -> Syscall {
281    let (sysno, args) = syscall.into_parts();
282    let Some(used) = kernel_arg_count(sysno) else {
283        return syscall;
284    };
285    let mut raw = [
286        args.arg0, args.arg1, args.arg2, args.arg3, args.arg4, args.arg5,
287    ];
288    for reg in raw.iter_mut().skip(usize::from(used)) {
289        *reg = 0;
290    }
291    Syscall::from_raw(
292        sysno,
293        SyscallArgs::new(raw[0], raw[1], raw[2], raw[3], raw[4], raw[5]),
294    )
295}

297/// A stream of syscall events.
298#[derive(Serialize, Deserialize)]
299pub struct EventReader {
300    // The file where events are stored.
301    //
302    // NOTE: This field isn't serializable/deserializable, so we have to skip it
303    // for now. With an in-guest backend, we'd need to implement this manually
304    // to support state migration.
305    #[serde(skip, default = "default_reader")]
306    reader: io::BufReader<fs::File>,
307
308    // The file where raw syscalls are stored. This is used for detecting
309    // desynchronization bugs. This is stored in a separate file so that we can
310    // easily turn this on or off to shift the balance on debuggability and
311    // performance.
312    #[serde(skip, default = "default_reader")]
313    debug_events: io::BufReader<fs::File>,
314
315    // The number of events read so far. Useful for debugging purposes.
316    pub count: u64,
317}
318
319fn default_reader() -> io::BufReader<fs::File> {
320    unimplemented!("Serialization is not yet implemented")
321}
322
323impl EventReader {
324    /// Opens an existing event stream.
325    pub fn open(path: &Path, stream_id: &EventStreamId) -> io::Result<Self> {
326        let file_name = stream_id.file_name();
327        let mut reader = io::BufReader::new(fs::File::open(path.join("thread").join(&file_name))?);
328        let mut debug_events = io::BufReader::new(fs::File::open(
329            path.join("thread").join(format!("{file_name}.debug")),
330        )?);
331        decode_stream_header(&mut reader, stream_id)?;
332        decode_stream_header(&mut debug_events, stream_id)?;
333        Ok(Self {
334            reader,
335            debug_events,
336            count: 0,
337        })
338    }
339
340    /// Reads the next event from the stream. Returns an error if there are no
341    /// more events to consume.
342    pub fn next_event(&mut self) -> Result<Event, bincode::error::DecodeError> {
343        bincode::serde::decode_from_std_read(&mut self.reader, bincode::config::legacy())
344    }
345
346    /// Reads the next syscall from the syscall stream.
347    pub fn next_debug_event(&mut self) -> Result<DebugEvent, bincode::error::DecodeError> {
348        let debug_event = bincode::serde::decode_from_std_read(
349            &mut self.debug_events,
350            bincode::config::legacy(),
351        )?;
352        self.count += 1;
353        Ok(debug_event)
354    }
355}
356
357impl Default for EventReader {
358    fn default() -> Self {
359        panic!("Thread state should be explicitly initialized in init_thread_state")
360    }
361}

363/// A stream of syscall events.
364#[derive(Serialize, Deserialize)]
365pub struct EventWriter {
366    // The file where events are stored.
367    //
368    // NOTE: This field isn't serializable/deserializable, so we have to skip it
369    // for now. With an in-guest backend, we'd need to implement this manually
370    // to support state migration.
371    #[serde(skip, default = "default_writer")]
372    writer: io::BufWriter<fs::File>,
373
374    // The file where syscalls are stored. This is used for debugging purposes.
375    #[serde(skip, default = "default_writer")]
376    debug_events: io::BufWriter<fs::File>,
377}
378
379fn default_writer() -> io::BufWriter<fs::File> {
380    unimplemented!("Serialization is not yet implemented")
381}
382
383impl EventWriter {
384    /// Creates a new event stream.
385    pub fn create(path: &Path, stream_id: &EventStreamId) -> io::Result<Self> {
386        let path = path.join("thread");
387
388        fs::create_dir_all(&path)?;
389        let file_name = stream_id.file_name();
390
391        let create = |path| {
392            fs::OpenOptions::new()
393                .write(true)
394                .create_new(true)
395                .open(path)
396        };
397
398        let mut writer = io::BufWriter::new(create(path.join(&file_name))?);
399        let mut debug_events = io::BufWriter::new(create(path.join(format!("{file_name}.debug")))?);
400        encode_stream_header(&mut writer, stream_id)?;
401        encode_stream_header(&mut debug_events, stream_id)?;
402        Ok(Self {
403            writer,
404            debug_events,
405        })
406    }
407
408    /// Writes an event to the end of the stream.
409    pub fn push_event(&mut self, event: Event) -> Result<(), bincode::error::EncodeError> {
410        bincode::serde::encode_into_std_write(&event, &mut self.writer, bincode::config::legacy())
411            .map(|_| ())
412    }
413
414    /// Writes a debug event to the end of the stream.
415    pub fn push_debug_event(
416        &mut self,
417        event: DebugEvent,
418    ) -> Result<(), bincode::error::EncodeError> {
419        bincode::serde::encode_into_std_write(
420            &event,
421            &mut self.debug_events,
422            bincode::config::legacy(),
423        )
424        .map(|_| ())
425    }
426}
427
428impl Default for EventWriter {
429    fn default() -> Self {
430        panic!("Thread state should be explicitly initialized in init_thread_state")
431    }
432}

452    #[test]
453    fn unused_args_do_not_cause_desync() {
454        // statfs(path, buf) uses two arguments; registers 2..6 are unused and may
455        // hold arbitrary leftover values. The raw compare (the bug) sees them as
456        // different, but the normalized compare must not.
457        let clean = raw(Sysno::statfs, SyscallArgs::new(0x1000, 0x2000, 0, 0, 0, 0));
458        let garbage = raw(
459            Sysno::statfs,
460            SyscallArgs::new(0x1000, 0x2000, 0xdead, 0xbeef, 0, 0xcafe),
461        );
462        assert_ne!(clean, garbage, "raw comparison should differ (the bug)");
463        assert_eq!(
464            normalize_unused_args(clean),
465            normalize_unused_args(garbage),
466            "normalized statfs must ignore unused argument registers"
467        );
468    }

525        let root = EventStreamId::root();
526        let recording = ChildEventStreamIds::default();
527        let replay = ChildEventStreamIds::default();
528        let first = recording.next(&root);
529        let second = recording.next(&root);

534        assert_eq!(second, replay.next(&root));

536        let serialized = serde_json::to_vec(&recording).expect("serialize child ordinal");

539        assert_eq!(restored.next(&root).to_string(), "3.2");
540
541        let grandchildren = ChildEventStreamIds::default();
542        assert_eq!(grandchildren.next(&first).to_string(), "3.0.0");
543    }
