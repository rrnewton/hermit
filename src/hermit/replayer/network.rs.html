24fn write_bytes<M: MemoryAccess>(
25    memory: &mut M,
26    pointer: *mut libc::c_void,
27    bytes: &[u8],
28) -> Result<(), Errno> {
29    if bytes.is_empty() {
30        return Ok(());
31    }
32    let address = AddrMut::<u8>::from_raw(pointer as usize).ok_or(Errno::EFAULT)?;
33    memory.write_exact(address.cast(), bytes)
34}

36fn read_iovecs<M: MemoryAccess>(
37    memory: &M,
38    message: &libc::msghdr,
39) -> Result<Vec<libc::iovec>, Errno> {
40    if message.msg_iovlen == 0 {
41        return Ok(Vec::new());
42    }
43    let address = Addr::from_raw(message.msg_iov as usize).ok_or(Errno::EFAULT)?;
44    let mut iovecs = vec![
45        libc::iovec {
46            iov_base: std::ptr::null_mut(),
47            iov_len: 0,
48        };
49        message.msg_iovlen
50    ];
51    memory.read_values(address, &mut iovecs)?;
52    Ok(iovecs)
53}

55fn cmsg_align(length: usize) -> Option<usize> {
56    let alignment = std::mem::size_of::<usize>();
57    length
58        .checked_add(alignment - 1)
59        .map(|value| value & !(alignment - 1))
60}

62fn scm_rights_fds(control: &[u8]) -> Vec<i32> {
63    let header_size = std::mem::size_of::<libc::cmsghdr>();
64    let data_offset = cmsg_align(header_size).unwrap();
65    let mut offset: usize = 0;
66    let mut fds = Vec::new();
67
68    while offset
69        .checked_add(header_size)
70        .is_some_and(|end| end <= control.len())
71    {
72        // The recorded control buffer has native cmsghdr layout but may not be
73        // aligned as a Vec<u8>, so read the header without assuming alignment.
74        let header = unsafe {
75            std::ptr::read_unaligned(control.as_ptr().add(offset).cast::<libc::cmsghdr>())
76        };
77        let length = header.cmsg_len;
78        let Some(end) = offset.checked_add(length) else {
79            break;
80        };
81        if length < data_offset || end > control.len() {
82            break;
83        }
84
85        if header.cmsg_level == libc::SOL_SOCKET && header.cmsg_type == libc::SCM_RIGHTS {
86            let (fd_bytes, _) =
87                control[offset + data_offset..end].as_chunks::<{ std::mem::size_of::<i32>() }>();
88            for bytes in fd_bytes {
89                let fd = i32::from_ne_bytes(*bytes);
90                if fd >= 0 {
91                    fds.push(fd);
92                }
93            }
94        }
95
96        let Some(aligned_length) = cmsg_align(length) else {
97            break;
98        };
99        let Some(next) = offset.checked_add(aligned_length) else {
100            break;
101        };
102        if next <= offset {
103            break;
104        }
105        offset = next;
106    }
107    fds
108}

110impl Replayer {
111    pub(super) async fn handle_epoll_wait<G: Guest<Self>>(
112        &self,
113        guest: &mut G,
114        syscall: EpollWait,
115    ) -> Result<i64, Errno> {
116        let event = next_event!(guest, EpollWait)?;
117        if event.replay_kernel_side_effect {
118            let actual = guest.inject(syscall).await;
119            assert_eq!(
120                actual,
121                Ok(event.updated as i64),
122                "replayed epoll_wait kernel side effect diverged"
123            );
124        }
125        assert_eq!(
126            event.events.len(),
127            event.updated * std::mem::size_of::<libc::epoll_event>()
128        );
129        assert!(event.updated <= syscall.maxevents() as usize);
130
131        if !event.events.is_empty() {
132            guest
133                .memory()
134                .write_exact(syscall.events().ok_or(Errno::EFAULT)?.cast(), &event.events)?;
135        }
136        Ok(event.updated as i64)
137    }
138
139    pub(super) async fn handle_poll<G: Guest<Self>>(
140        &self,
141        guest: &mut G,
142        syscall: Poll,
143    ) -> Result<i64, Errno> {
144        let event = next_event!(guest, Poll)?;
145
146        let nfds = syscall.nfds() as usize;
147
148        assert_eq!(event.fds.len(), nfds);
149
150        // Write out the recorded fds (if any).
151        if let Some(addr) = syscall.fds() {
152            guest.memory().write_values(addr, &event.fds)?;
153        }
154
155        Ok(event.updated as i64)
156    }
157
158    pub(super) async fn handle_ppoll<G: Guest<Self>>(
159        &self,
160        guest: &mut G,
161        syscall: Ppoll,
162    ) -> Result<i64, Errno> {
163        // `ppoll` shares `poll`'s recorded output (the updated pollfd array and
164        // return count), so it reuses the `Poll` event. We restore every
165        // recorded `revents` field without consulting live descriptors, and we
166        // never inject the call, so the recorded temporary signal mask has no
167        // replay effect (a recorded `EINTR` is reproduced via the event's
168        // `Result`, handled before we get here).
169        let event = next_event!(guest, Poll)?;
170
171        let nfds = syscall.nfds() as usize;
172        assert_eq!(event.fds.len(), nfds);
173
174        // Write out the recorded fds (if any). `Ppoll::fds()` is typed as
175        // `AddrMut<libc::pollfd>`; cast to the layout-compatible `PollFd`.
176        if let Some(addr) = syscall.fds() {
177            guest
178                .memory()
179                .write_values(addr.cast::<PollFd>(), &event.fds)?;
180        }
181
182        Ok(event.updated as i64)
183    }
184
185    pub(super) async fn handle_sockopt_family<G: Guest<Self>>(
186        &self,
187        guest: &mut G,
188        syscall: SockOptFamily,
189    ) -> Result<i64, Errno> {
190        let event = next_event!(guest, SockOpt)?;
191
192        // A NULL value buffer is valid when the recorded value is empty.
193        if let Some(address) = syscall.value() {
194            guest
195                .memory()
196                .write_exact(address.cast::<u8>(), &event.value)?;
197        } else if !event.value.is_empty() {
198            return Err(Errno::EFAULT);
199        }
200
201        // Write out the length parameter.
202        guest
203            .memory()
204            .write_value(syscall.value_len().ok_or(Errno::EFAULT)?, &event.length)?;
205
206        Ok(0)
207    }

209    pub(super) async fn handle_recvmsg<G: Guest<Self>>(
210        &self,
211        guest: &mut G,
212        syscall: Recvmsg,
213    ) -> Result<i64, Errno> {
214        let event = next_event!(guest, Recvmsg)?;
215        let cloexec = syscall.flags() & libc::MSG_CMSG_CLOEXEC != 0;
216        for fd in scm_rights_fds(&event.control) {
217            self.reserve_replay_fd(guest, fd, cloexec).await;
218        }
219
220        let message_address = syscall.msg().ok_or(Errno::EFAULT)?;
221        let mut message: libc::msghdr = guest.memory().read_value(message_address)?;
222        let iovecs = read_iovecs(&guest.memory(), &message)?;
223        assert_eq!(iovecs.len(), event.iovs.len());
224
225        for (iovec, bytes) in iovecs.into_iter().zip(&event.iovs) {
226            assert!(bytes.len() <= iovec.iov_len);
227            write_bytes(&mut guest.memory(), iovec.iov_base, bytes)?;
228        }
229
230        assert!(event.name.len() <= message.msg_namelen as usize);
231        assert!(event.control.len() <= message.msg_controllen);
232        write_bytes(&mut guest.memory(), message.msg_name, &event.name)?;
233        write_bytes(&mut guest.memory(), message.msg_control, &event.control)?;
234
235        message.msg_namelen = event.name_len;
236        message.msg_controllen = event.control_len;
237        message.msg_flags = event.flags;
238        guest.memory().write_value(message_address, &message)?;
239
240        Ok(event.result)
241    }

243    pub(super) async fn handle_recvfrom<G: Guest<Self>>(
244        &self,
245        guest: &mut G,
246        syscall: Recvfrom,
247    ) -> Result<i64, Errno> {
248        let buf = next_event!(guest, Bytes)?;
249
250        assert!(buf.len() <= syscall.len());
251
252        // Write out the buffer.
253        guest
254            .memory()
255            .write_exact(syscall.buf().unwrap(), &buf)
256            .unwrap();
257        Ok(buf.len() as i64)
258    }
