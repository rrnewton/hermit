32fn replay_pollfds<M: MemoryAccess>(
33    memory: &mut M,
34    fds_address: Option<AddrMut<'_, PollFd>>,
35    nfds: usize,
36    result: Result<i64, Errno>,
37    fds_pointer_present: bool,
38    fds: Option<Vec<PollFd>>,
39) -> Result<(), Errno> {
40    assert_eq!(
41        fds_address.is_some(),
42        fds_pointer_present,
43        "recorded pollfd pointer shape diverged during replay"
44    );
45
46    if matches!(result, Ok(_) | Err(Errno::EINTR)) {
47        assert_eq!(fds.is_some(), fds_pointer_present);
48        if let Ok(updated) = result {
49            assert!(updated >= 0);
50            assert!((updated as usize) <= nfds);
51        }
52    } else if !matches!(result, Err(Errno::EFAULT)) {
53        assert!(fds.is_none());
54    }
55
56    if let Some(fds) = fds {
57        assert!(fds.len() <= nfds);
58        if matches!(result, Ok(_) | Err(Errno::EINTR)) {
59            assert_eq!(fds.len(), nfds);
60        }
61        let address = fds_address.expect("recorded pollfd output requires a pointer");
62        let write_result = memory.write_values(address, &fds);
63        if matches!(result, Err(Errno::EFAULT)) {
64            // The same page boundary that made Linux return EFAULT can make
65            // this replay write fault after restoring an earlier prefix. Keep
66            // going so ppoll can also restore its captured timeout, then return
67            // the recorded errno unchanged.
68            if let Err(error) = write_result {
69                tracing::trace!(?error, "partial pollfd replay write returned an error");
70            }
71        } else {
72            write_result?;
73        }
74    }
75    Ok(())
76}

78fn replay_poll_event<M: MemoryAccess>(
79    memory: &mut M,
80    fds_address: Option<AddrMut<'_, PollFd>>,
81    nfds: usize,
82    event: PollEvent,
83) -> Result<i64, Errno> {
84    let PollEvent {
85        result,
86        fds_pointer_present,
87        fds,
88    } = event;
89    replay_pollfds(memory, fds_address, nfds, result, fds_pointer_present, fds)?;
90    result
91}

93fn replay_ppoll_event<M: MemoryAccess>(
94    memory: &mut M,
95    fds_address: Option<AddrMut<'_, libc::pollfd>>,
96    timeout_address: Option<AddrMut<'_, Timespec>>,
97    nfds: usize,
98    event: PpollEvent,
99) -> Result<i64, Errno> {
100    let PpollEvent {
101        result,
102        fds_pointer_present,
103        fds,
104        timeout_pointer_present,
105        timeout,
106    } = event;
107
108    replay_pollfds(
109        memory,
110        fds_address.map(|address| address.cast::<PollFd>()),
111        nfds,
112        result,
113        fds_pointer_present,
114        fds,
115    )?;
116
117    assert_eq!(
118        timeout_address.is_some(),
119        timeout_pointer_present,
120        "recorded ppoll timeout pointer shape diverged during replay"
121    );
122    if !matches!(result, Err(Errno::EFAULT)) {
123        assert_eq!(timeout.is_some(), timeout_pointer_present);
124    }
125    if let Some(timeout) = timeout {
126        let address = timeout_address.expect("recorded ppoll timeout requires a pointer");
127        // Linux preserves the ppoll result when remaining-time copyout faults.
128        // Restore the exact captured value when possible, but never replace the
129        // recorded result with a replay-only memory error.
130        if let Err(error) = memory.write_value(address, &timeout) {
131            tracing::trace!(?error, "ppoll timeout replay write returned an error");
132        }
133    }
134
135    result
136}

138fn write_bytes<M: MemoryAccess>(
139    memory: &mut M,
140    pointer: *mut libc::c_void,
141    bytes: &[u8],
142) -> Result<(), Errno> {
143    if bytes.is_empty() {
144        return Ok(());
145    }
146    let address = AddrMut::<u8>::from_raw(pointer as usize).ok_or(Errno::EFAULT)?;
147    memory.write_exact(address.cast(), bytes)
148}

150fn replay_select_event<M: MemoryAccess>(
151    memory: &mut M,
152    nfds: i32,
153    fd_set_addresses: [Option<AddrMut<'_, libc::fd_set>>; 3],
154    timeout_address: Option<AddrMut<'_, u8>>,
155    event: SelectEvent,
156) -> Result<i64, Errno> {
157    let SelectEvent {
158        result,
159        fd_sets,
160        timeout,
161    } = event;
162    let length = fd_set_bytes(nfds);
163    for (address, bytes) in fd_set_addresses.into_iter().zip(fd_sets) {
164        let Some(bytes) = bytes else {
165            assert!(
166                address.is_none() || result.is_err(),
167                "recorded select omitted a set the kernel copied out"
168            );
169            continue;
170        };
171        assert!(bytes.len() <= length);
172        let address = address.expect("recorded select set output requires a pointer");
173        let write_result = write_bytes(memory, address.as_raw() as *mut libc::c_void, &bytes);
174        if matches!(result, Err(Errno::EFAULT)) {
175            // As for ppoll: the fault that ended the recorded copy-out may end
176            // this one too. The recorded errno is still the result.
177            if let Err(error) = write_result {
178                tracing::trace!(?error, "partial select set replay write returned an error");
179            }
180        } else {
181            write_result?;
182        }
183    }
184    if let Some(timeout) = timeout {
185        let address = timeout_address.expect("recorded select timeout requires a pointer");
186        // Linux keeps the select result when the remaining-time write faults.
187        if let Err(error) = write_bytes(memory, address.as_raw() as *mut libc::c_void, &timeout) {
188            tracing::trace!(?error, "select timeout replay write returned an error");
189        }
190    }
191    result
192}

194fn cmsg_align(length: usize) -> Option<usize> {
195    let alignment = std::mem::size_of::<usize>();
196    length
197        .checked_add(alignment - 1)
198        .map(|value| value & !(alignment - 1))
199}

201fn scm_rights_fds(control: &[u8]) -> Vec<i32> {
202    let header_size = std::mem::size_of::<libc::cmsghdr>();
203    let data_offset = cmsg_align(header_size).unwrap();
204    let mut offset: usize = 0;
205    let mut fds = Vec::new();
206
207    while offset
208        .checked_add(header_size)
209        .is_some_and(|end| end <= control.len())
210    {
211        // The recorded control buffer has native cmsghdr layout but may not be
212        // aligned as a Vec<u8>, so read the header without assuming alignment.
213        let header = unsafe {
214            std::ptr::read_unaligned(control.as_ptr().add(offset).cast::<libc::cmsghdr>())
215        };
216        let length = header.cmsg_len;
217        let Some(end) = offset.checked_add(length) else {
218            break;
219        };
220        if length < data_offset || end > control.len() {
221            break;
222        }
223
224        if header.cmsg_level == libc::SOL_SOCKET && header.cmsg_type == libc::SCM_RIGHTS {
225            let (fd_bytes, _) =
226                control[offset + data_offset..end].as_chunks::<{ std::mem::size_of::<i32>() }>();
227            for bytes in fd_bytes {
228                let fd = i32::from_ne_bytes(*bytes);
229                if fd >= 0 {
230                    fds.push(fd);
231                }
232            }
233        }
234
235        let Some(aligned_length) = cmsg_align(length) else {
236            break;
237        };
238        let Some(next) = offset.checked_add(aligned_length) else {
239            break;
240        };
241        if next <= offset {
242            break;
243        }
244        offset = next;
245    }
246    fds
247}

250    pub(super) async fn handle_epoll_wait<G: Guest<Self>>(
251        &self,
252        guest: &mut G,
253        syscall: EpollWait,
254    ) -> Result<i64, Errno> {
255        let event = next_event!(guest, EpollWait)?;
256        if event.replay_kernel_side_effect {
257            let actual = guest.inject(syscall).await;
258            assert_eq!(
259                actual,
260                Ok(event.updated as i64),
261                "replayed epoll_wait kernel side effect diverged"
262            );
263        }
264        assert_eq!(
265            event.events.len(),
266            event.updated * std::mem::size_of::<libc::epoll_event>()
267        );
268        assert!(event.updated <= syscall.maxevents() as usize);
269
270        if !event.events.is_empty() {
271            guest
272                .memory()
273                .write_exact(syscall.events().ok_or(Errno::EFAULT)?.cast(), &event.events)?;
274        }
275        Ok(event.updated as i64)
276    }

338    // TODO-HUMAN-REVIEW(PR-3871): Replay accepted connections from the log.
339    pub(super) async fn handle_accept4<G: Guest<Self>>(
340        &self,
341        guest: &mut G,
342        syscall: Accept4,
343    ) -> Result<i64, Errno> {
344        let event = next_event!(guest, Accept)?;
345
346        // Reproduce the recorded copy-out before any descriptor exists, in
347        // the order record used (Linux 7.1's): the length first, then the
348        // address prefix. These writes succeeded during recording into the
349        // same address space, so a failure here means replay has diverged.
350        if let Some(addr_len) = event.addr_len {
351            let len_address = syscall
352                .addrlen()
353                .expect("replay diverged: recorded accept4 wrote addrlen, replay passed none");
354            guest
355                .memory()
356                .write_value(len_address.cast::<libc::socklen_t>(), &addr_len)
357                .unwrap_or_else(|error| {
358                    panic!("replay diverged: cannot write the recorded accept4 addrlen: {error}")
359                });
360        }
361        if !event.addr.is_empty() {
362            let sockaddr = syscall.sockaddr().expect(
363                "replay diverged: recorded accept4 wrote a peer address, replay passed none",
364            );
365            guest
366                .memory()
367                .write_exact(sockaddr.cast::<u8>(), &event.addr)
368                .unwrap_or_else(|error| {
369                    panic!(
370                        "replay diverged: cannot write the recorded accept4 peer address: {error}"
371                    )
372                });
373        }
374
375        // A copy-out that failed during recording released the connection
376        // without installing a descriptor.
377        let fd = event.result?;
378
379        // Replay does not recreate the connection's peer, so a placeholder
380        // holds the recorded descriptor number and every recorded operation on
381        // it is served from the log, as for descriptors received over
382        // `SCM_RIGHTS`.
383        let cloexec = syscall.flags().contains(SockFlag::SOCK_CLOEXEC);
384        self.reserve_accept_stand_in(guest, fd, cloexec).await;
385
386        Ok(i64::from(fd))
387    }

389    pub(super) async fn handle_sockopt_family<G: Guest<Self>>(
390        &self,
391        guest: &mut G,
392        syscall: SockOptFamily,
393    ) -> Result<i64, Errno> {
394        let event = next_event!(guest, SockOpt)?;
395
396        // A NULL value buffer is valid when the recorded value is empty.
397        if let Some(address) = syscall.value() {
398            guest
399                .memory()
400                .write_exact(address.cast::<u8>(), &event.value)?;
401        } else if !event.value.is_empty() {
402            return Err(Errno::EFAULT);
403        }
404
405        // Write out the length parameter.
406        guest
407            .memory()
408            .write_value(syscall.value_len().ok_or(Errno::EFAULT)?, &event.length)?;
409
410        Ok(0)
411    }

413    pub(super) async fn handle_recvmsg<G: Guest<Self>>(
414        &self,
415        guest: &mut G,
416        syscall: Recvmsg,
417    ) -> Result<i64, Errno> {
418        let event = next_event!(guest, Recvmsg)?;
419        let cloexec = syscall.flags() & libc::MSG_CMSG_CLOEXEC != 0;
420        for fd in scm_rights_fds(&event.control) {
421            self.reserve_replay_fd(guest, fd, cloexec).await;
422        }
423
424        let message_address = syscall.msg().ok_or(Errno::EFAULT)?;
425        let mut message: libc::msghdr = guest.memory().read_value(message_address)?;
426        let iovecs = crate::read_iovecs(&guest.memory(), &message)?;
427        assert_eq!(iovecs.len(), event.iovs.len());
428
429        for (iovec, bytes) in iovecs.into_iter().zip(&event.iovs) {
430            assert!(bytes.len() <= iovec.iov_len);
431            write_bytes(&mut guest.memory(), iovec.iov_base, bytes)?;
432        }
433
434        assert!(event.name.len() <= message.msg_namelen as usize);
435        assert!(event.control.len() <= message.msg_controllen);
436        write_bytes(&mut guest.memory(), message.msg_name, &event.name)?;
437        write_bytes(&mut guest.memory(), message.msg_control, &event.control)?;
438
439        message.msg_namelen = event.name_len;
440        message.msg_controllen = event.control_len;
441        message.msg_flags = event.flags;
442        guest.memory().write_value(message_address, &message)?;
443
444        Ok(event.result)
445    }

447    pub(super) async fn handle_recvfrom<G: Guest<Self>>(
448        &self,
449        guest: &mut G,
450        syscall: Recvfrom,
451    ) -> Result<i64, Errno> {
452        let buf = next_event!(guest, Bytes)?;
453
454        assert!(buf.len() <= syscall.len());
455
456        // Write out the buffer.
457        guest
458            .memory()
459            .write_exact(syscall.buf().unwrap(), &buf)
460            .unwrap();
461        Ok(buf.len() as i64)
462    }

857    #[test]
858    fn replay_select_restores_only_the_recorded_prefix_before_efault() {
859        // Recorded: the read set was partly copied out (its first long), then
860        // the kernel faulted before reaching the write set.
861        let mut readfds: libc::fd_set = unsafe { std::mem::zeroed() };
862        let mut writefds: libc::fd_set = unsafe { std::mem::zeroed() };
863        unsafe {
864            libc::FD_SET(3, &mut readfds);
865            libc::FD_SET(100, &mut readfds);
866            libc::FD_SET(5, &mut writefds);
867        }
868        let event = SelectEvent {
869            result: Err(Errno::EFAULT),
870            fd_sets: [Some(vec![0b0100_0000, 0, 0, 0, 0, 0, 0, 0]), None, None],
871            timeout: None,
872        };
873        let result = replay_select_event(
874            &mut LocalMemory::new(),
875            128,
876            [
877                AddrMut::from_raw((&mut readfds as *mut libc::fd_set) as usize),
878                AddrMut::from_raw((&mut writefds as *mut libc::fd_set) as usize),
879                None,
880            ],
881            None,
882            event,
883        );
884
885        assert_eq!(result, Err(Errno::EFAULT));
886        assert!(!unsafe { libc::FD_ISSET(3, &readfds) });
887        assert!(unsafe { libc::FD_ISSET(6, &readfds) });
888        // Past the recorded prefix, and the set the kernel never reached,
889        // keep the guest's own bytes.
890        assert!(unsafe { libc::FD_ISSET(100, &readfds) });
891        assert!(unsafe { libc::FD_ISSET(5, &writefds) });
892    }
