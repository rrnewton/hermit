34fn replay_pollfds<M: MemoryAccess>(
35    memory: &mut M,
36    fds_address: Option<AddrMut<'_, PollFd>>,
37    nfds: usize,
38    result: Result<i64, Errno>,
39    fds_pointer_present: bool,
40    fds: Option<Vec<PollFd>>,
41) -> Result<(), Errno> {
42    assert_eq!(
43        fds_address.is_some(),
44        fds_pointer_present,
45        "recorded pollfd pointer shape diverged during replay"
46    );
47
48    if matches!(result, Ok(_) | Err(Errno::EINTR)) {
49        assert_eq!(fds.is_some(), fds_pointer_present);
50        if let Ok(updated) = result {
51            assert!(updated >= 0);
52            assert!((updated as usize) <= nfds);
53        }
54    } else if !matches!(result, Err(Errno::EFAULT)) {
55        assert!(fds.is_none());
56    }
57
58    if let Some(fds) = fds {
59        assert!(fds.len() <= nfds);
60        if matches!(result, Ok(_) | Err(Errno::EINTR)) {
61            assert_eq!(fds.len(), nfds);
62        }
63        let address = fds_address.expect("recorded pollfd output requires a pointer");
64        let write_result = memory.write_values(address, &fds);
65        if matches!(result, Err(Errno::EFAULT)) {
66            // The same page boundary that made Linux return EFAULT can make
67            // this replay write fault after restoring an earlier prefix. Keep
68            // going so ppoll can also restore its captured timeout, then return
69            // the recorded errno unchanged.
70            if let Err(error) = write_result {
71                tracing::trace!(?error, "partial pollfd replay write returned an error");
72            }
73        } else {
74            write_result?;
75        }
76    }
77    Ok(())
78}

80fn replay_poll_event<M: MemoryAccess>(
81    memory: &mut M,
82    fds_address: Option<AddrMut<'_, PollFd>>,
83    nfds: usize,
84    event: PollEvent,
85) -> Result<i64, Errno> {
86    let PollEvent {
87        result,
88        fds_pointer_present,
89        fds,
90    } = event;
91    replay_pollfds(memory, fds_address, nfds, result, fds_pointer_present, fds)?;
92    result
93}

95fn replay_ppoll_event<M: MemoryAccess>(
96    memory: &mut M,
97    fds_address: Option<AddrMut<'_, libc::pollfd>>,
98    timeout_address: Option<AddrMut<'_, Timespec>>,
99    nfds: usize,
100    event: PpollEvent,
101) -> Result<i64, Errno> {
102    let PpollEvent {
103        result,
104        fds_pointer_present,
105        fds,
106        timeout_pointer_present,
107        timeout,
108    } = event;
109
110    replay_pollfds(
111        memory,
112        fds_address.map(|address| address.cast::<PollFd>()),
113        nfds,
114        result,
115        fds_pointer_present,
116        fds,
117    )?;
118
119    assert_eq!(
120        timeout_address.is_some(),
121        timeout_pointer_present,
122        "recorded ppoll timeout pointer shape diverged during replay"
123    );
124    if !matches!(result, Err(Errno::EFAULT)) {
125        assert_eq!(timeout.is_some(), timeout_pointer_present);
126    }
127    if let Some(timeout) = timeout {
128        let address = timeout_address.expect("recorded ppoll timeout requires a pointer");
129        // Linux preserves the ppoll result when remaining-time copyout faults.
130        // Restore the exact captured value when possible, but never replace the
131        // recorded result with a replay-only memory error.
132        if let Err(error) = memory.write_value(address, &timeout) {
133            tracing::trace!(?error, "ppoll timeout replay write returned an error");
134        }
135    }
136
137    result
138}

140fn write_bytes<M: MemoryAccess>(
141    memory: &mut M,
142    pointer: *mut libc::c_void,
143    bytes: &[u8],
144) -> Result<(), Errno> {
145    if bytes.is_empty() {
146        return Ok(());
147    }
148    let address = AddrMut::<u8>::from_raw(pointer as usize).ok_or(Errno::EFAULT)?;
149    memory.write_exact(address.cast(), bytes)
150}

152fn replay_select_event<M: MemoryAccess>(
153    memory: &mut M,
154    nfds: i32,
155    fd_set_addresses: [Option<AddrMut<'_, libc::fd_set>>; 3],
156    timeout_address: Option<AddrMut<'_, u8>>,
157    event: SelectEvent,
158) -> Result<i64, Errno> {
159    let SelectEvent {
160        result,
161        fd_sets,
162        timeout,
163    } = event;
164    let length = fd_set_bytes(nfds);
165    for (address, bytes) in fd_set_addresses.into_iter().zip(fd_sets) {
166        let Some(bytes) = bytes else {
167            assert!(
168                address.is_none() || result.is_err(),
169                "recorded select omitted a set the kernel copied out"
170            );
171            continue;
172        };
173        assert!(bytes.len() <= length);
174        let address = address.expect("recorded select set output requires a pointer");
175        let write_result = write_bytes(memory, address.as_raw() as *mut libc::c_void, &bytes);
176        if matches!(result, Err(Errno::EFAULT)) {
177            // As for ppoll: the fault that ended the recorded copy-out may end
178            // this one too. The recorded errno is still the result.
179            if let Err(error) = write_result {
180                tracing::trace!(?error, "partial select set replay write returned an error");
181            }
182        } else {
183            write_result?;
184        }
185    }
186    if let Some(timeout) = timeout {
187        let address = timeout_address.expect("recorded select timeout requires a pointer");
188        // Linux keeps the select result when the remaining-time write faults.
189        if let Err(error) = write_bytes(memory, address.as_raw() as *mut libc::c_void, &timeout) {
190            tracing::trace!(?error, "select timeout replay write returned an error");
191        }
192    }
193    result
194}

196fn cmsg_align(length: usize) -> Option<usize> {
197    let alignment = std::mem::size_of::<usize>();
198    length
199        .checked_add(alignment - 1)
200        .map(|value| value & !(alignment - 1))
201}

203fn scm_rights_fds(control: &[u8]) -> Vec<i32> {
204    let header_size = std::mem::size_of::<libc::cmsghdr>();
205    let data_offset = cmsg_align(header_size).unwrap();
206    let mut offset: usize = 0;
207    let mut fds = Vec::new();
208
209    while offset
210        .checked_add(header_size)
211        .is_some_and(|end| end <= control.len())
212    {
213        // The recorded control buffer has native cmsghdr layout but may not be
214        // aligned as a Vec<u8>, so read the header without assuming alignment.
215        let header = unsafe {
216            std::ptr::read_unaligned(control.as_ptr().add(offset).cast::<libc::cmsghdr>())
217        };
218        let length = header.cmsg_len;
219        let Some(end) = offset.checked_add(length) else {
220            break;
221        };
222        if length < data_offset || end > control.len() {
223            break;
224        }
225
226        if header.cmsg_level == libc::SOL_SOCKET && header.cmsg_type == libc::SCM_RIGHTS {
227            let (fd_bytes, _) =
228                control[offset + data_offset..end].as_chunks::<{ std::mem::size_of::<i32>() }>();
229            for bytes in fd_bytes {
230                let fd = i32::from_ne_bytes(*bytes);
231                if fd >= 0 {
232                    fds.push(fd);
233                }
234            }
235        }
236
237        let Some(aligned_length) = cmsg_align(length) else {
238            break;
239        };
240        let Some(next) = offset.checked_add(aligned_length) else {
241            break;
242        };
243        if next <= offset {
244            break;
245        }
246        offset = next;
247    }
248    fds
249}

297    /// Serves an `epoll_wait`-family `syscall` from the recording, writing the
298    /// recorded events to `events`.
299    async fn replay_epoll_wait<G: Guest<Self>>(
300        &self,
301        guest: &mut G,
302        syscall: Syscall,
303        events: Option<usize>,
304        maxevents: libc::c_int,
305    ) -> Result<i64, Errno> {
306        let event = next_event!(guest, EpollWait)?;
307        if event.replay_kernel_side_effect {
308            let actual = guest.inject(syscall).await;
309            assert_eq!(
310                actual,
311                Ok(event.updated as i64),
312                "replayed epoll_wait kernel side effect diverged"
313            );
314        }
315        assert_eq!(
316            event.events.len(),
317            event.updated * std::mem::size_of::<libc::epoll_event>()
318        );
319        assert!(event.updated <= maxevents as usize);
320
321        if !event.events.is_empty() {
322            let address = events
323                .and_then(AddrMut::<u8>::from_raw)
324                .ok_or(Errno::EFAULT)?;
325            guest.memory().write_exact(address, &event.events)?;
326        }
327        Ok(event.updated as i64)
328    }

390    // TODO-HUMAN-REVIEW(PR-3871): Replay accepted connections from the log.
391    pub(super) async fn handle_accept4<G: Guest<Self>>(
392        &self,
393        guest: &mut G,
394        syscall: Accept4,
395    ) -> Result<i64, Errno> {
396        let event = next_event!(guest, Accept)?;
397
398        // Reproduce the recorded copy-out before any descriptor exists, in
399        // the order record used (Linux 7.1's): the length first, then the
400        // address prefix. These writes succeeded during recording into the
401        // same address space, so a failure here means replay has diverged.
402        if let Some(addr_len) = event.addr_len {
403            let len_address = syscall
404                .addrlen()
405                .expect("replay diverged: recorded accept4 wrote addrlen, replay passed none");
406            guest
407                .memory()
408                .write_value(len_address.cast::<libc::socklen_t>(), &addr_len)
409                .unwrap_or_else(|error| {
410                    panic!("replay diverged: cannot write the recorded accept4 addrlen: {error}")
411                });
412        }
413        if !event.addr.is_empty() {
414            let sockaddr = syscall.sockaddr().expect(
415                "replay diverged: recorded accept4 wrote a peer address, replay passed none",
416            );
417            guest
418                .memory()
419                .write_exact(sockaddr.cast::<u8>(), &event.addr)
420                .unwrap_or_else(|error| {
421                    panic!(
422                        "replay diverged: cannot write the recorded accept4 peer address: {error}"
423                    )
424                });
425        }
426
427        // A copy-out that failed during recording released the connection
428        // without installing a descriptor.
429        let fd = event.result?;
430
431        // Replay does not recreate the connection's peer, so a placeholder
432        // holds the recorded descriptor number and every recorded operation on
433        // it is served from the log, as for descriptors received over
434        // `SCM_RIGHTS`.
435        let cloexec = syscall.flags().contains(SockFlag::SOCK_CLOEXEC);
436        self.reserve_accept_stand_in(guest, fd, cloexec).await;
437
438        Ok(i64::from(fd))
439    }

441    pub(super) async fn handle_sockopt_family<G: Guest<Self>>(
442        &self,
443        guest: &mut G,
444        syscall: SockOptFamily,
445    ) -> Result<i64, Errno> {
446        let event = next_event!(guest, SockOpt)?;
447
448        // A NULL value buffer is valid when the recorded value is empty.
449        if let Some(address) = syscall.value() {
450            guest
451                .memory()
452                .write_exact(address.cast::<u8>(), &event.value)?;
453        } else if !event.value.is_empty() {
454            return Err(Errno::EFAULT);
455        }
456
457        // Write out the length parameter.
458        guest
459            .memory()
460            .write_value(syscall.value_len().ok_or(Errno::EFAULT)?, &event.length)?;
461
462        Ok(0)
463    }

465    pub(super) async fn handle_recvmsg<G: Guest<Self>>(
466        &self,
467        guest: &mut G,
468        syscall: Recvmsg,
469    ) -> Result<i64, Errno> {
470        let event = next_event!(guest, Recvmsg)?;
471        let cloexec = syscall.flags() & libc::MSG_CMSG_CLOEXEC != 0;
472        for fd in scm_rights_fds(&event.control) {
473            self.reserve_replay_fd(guest, fd, cloexec).await;
474        }
475
476        let message_address = syscall.msg().ok_or(Errno::EFAULT)?;
477        let mut message: libc::msghdr = guest.memory().read_value(message_address)?;
478        let iovecs = crate::read_iovecs(&guest.memory(), &message)?;
479        assert_eq!(iovecs.len(), event.iovs.len());
480
481        for (iovec, bytes) in iovecs.into_iter().zip(&event.iovs) {
482            assert!(bytes.len() <= iovec.iov_len);
483            write_bytes(&mut guest.memory(), iovec.iov_base, bytes)?;
484        }
485
486        assert!(event.name.len() <= message.msg_namelen as usize);
487        assert!(event.control.len() <= message.msg_controllen);
488        write_bytes(&mut guest.memory(), message.msg_name, &event.name)?;
489        write_bytes(&mut guest.memory(), message.msg_control, &event.control)?;
490
491        message.msg_namelen = event.name_len;
492        message.msg_controllen = event.control_len;
493        message.msg_flags = event.flags;
494        guest.memory().write_value(message_address, &message)?;
495
496        Ok(event.result)
497    }

499    pub(super) async fn handle_recvfrom<G: Guest<Self>>(
500        &self,
501        guest: &mut G,
502        syscall: Recvfrom,
503    ) -> Result<i64, Errno> {
504        let buf = next_event!(guest, Bytes)?;
505
506        assert!(buf.len() <= syscall.len());
507
508        // Write out the buffer.
509        guest
510            .memory()
511            .write_exact(syscall.buf().unwrap(), &buf)

513        Ok(buf.len() as i64)

909    #[test]
910    fn replay_select_restores_only_the_recorded_prefix_before_efault() {
911        // Recorded: the read set was partly copied out (its first long), then
912        // the kernel faulted before reaching the write set.
913        let mut readfds: libc::fd_set = unsafe { std::mem::zeroed() };
914        let mut writefds: libc::fd_set = unsafe { std::mem::zeroed() };
915        unsafe {
916            libc::FD_SET(3, &mut readfds);
917            libc::FD_SET(100, &mut readfds);
918            libc::FD_SET(5, &mut writefds);
919        }
920        let event = SelectEvent {
921            result: Err(Errno::EFAULT),
922            fd_sets: [Some(vec![0b0100_0000, 0, 0, 0, 0, 0, 0, 0]), None, None],
923            timeout: None,
924        };
925        let result = replay_select_event(
926            &mut LocalMemory::new(),
927            128,
928            [
929                AddrMut::from_raw((&mut readfds as *mut libc::fd_set) as usize),
930                AddrMut::from_raw((&mut writefds as *mut libc::fd_set) as usize),
931                None,
932            ],
933            None,
934            event,
935        );
936
937        assert_eq!(result, Err(Errno::EFAULT));
938        assert!(!unsafe { libc::FD_ISSET(3, &readfds) });
939        assert!(unsafe { libc::FD_ISSET(6, &readfds) });
940        // Past the recorded prefix, and the set the kernel never reached,
941        // keep the guest's own bytes.
942        assert!(unsafe { libc::FD_ISSET(100, &readfds) });
943        assert!(unsafe { libc::FD_ISSET(5, &writefds) });
944    }
