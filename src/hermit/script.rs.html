9use std::ffi::OsStr;
10use std::ffi::OsString;
11use std::fs;
12use std::io::Read;

17// Defined in kernel include/linux/binfmts.h
18const BINPRM_BUF_SIZE: usize = 256;
19
20/// Unix shebang, see [Wikipedia](https://en.wikipedia.org/wiki/Shebang_(Unix)).
21#[derive(Debug, Eq, PartialEq)]
22pub struct Shebang {
23    program: PathBuf,
24    args: Vec<OsString>,
25}
26
27impl Shebang {
28    // Source of truth: fs/binfmt_script.c, function load_script().
29    pub(crate) fn from_buf(buf: &[u8]) -> Option<Self> {
30        if !buf.starts_with(b"#!") {
31            return None;
32        }
33
34        let mut i = 2;
35        while i < buf.len() {
36            if buf[i] == b' ' || buf[i] == b'\t' {
37                i += 1;
38            } else {
39                break;
40            }
41        }
42
43        let mut j = i;
44        while j < buf.len() {
45            if b" \t\r\n".contains(&buf[j]) {
46                break;
47            } else {
48                j += 1;
49            }
50        }
51
52        let program = PathBuf::from(OsStr::from_bytes(&buf[i..j]));
53
54        i = j;
55        while j < buf.len() {
56            if b"\n".contains(&buf[j]) {
57                break;
58            } else {
59                j += 1;
60            }
61        }
62
63        let args = String::from_utf8_lossy(&buf[i..j])
64            .split_ascii_whitespace()
65            .map(OsString::from)
66            .collect::<Vec<_>>();
67
68        Some(Shebang { program, args })
69    }
70
71    /// Extract the interpreter path without checking whether it exists.
72    ///
73    /// Callers must resolve the path in the filesystem namespace that will
74    /// execute the script.
75    pub fn interpreter_from_buf(buf: &[u8]) -> Option<PathBuf> {
76        Self::from_buf(buf).map(|shebang| shebang.program)
77    }
78
79    /// Parse exe/interpreter from a script contains a shebang, whenever possible.
80    pub fn new<P: AsRef<Path>>(path: P) -> Option<Self> {
81        let mut buf = Vec::new();
82        let nb = fs::File::open(path)
83            .and_then(|f| {
84                let mut handle = f.take(BINPRM_BUF_SIZE as u64);
85                handle.read_to_end(&mut buf)
86            })
87            .ok()?;
88        // Pass a slice up to the number of bytes read to the lookup function. We don't want to pass
89        // the extra zeroes.
90        let shebang = Self::from_buf(&buf[..nb])?;
91        let metadata = fs::metadata(&shebang.program).ok()?;
92        if metadata.is_file() {
93            Some(shebang)
94        } else {
95            None
96        }
97    }
98
99    /// Get interpreter from shebang
100    pub fn interpreter(&self) -> &Path {
101        &self.program
102    }
103
104    /// Get intepreter arguments from shebang.
105    pub fn args(&self) -> impl Iterator<Item = &OsStr> {
106        self.args.iter().map(OsStr::new)
107    }
108
109    /// Convert shebang into (intepreter, args).
110    pub fn into_parts(self) -> (PathBuf, Vec<OsString>) {
111        (self.program, self.args)
112    }
113}

115/// How the kernel's script handler reads the start of a file, for checks that
116/// must inspect exactly the interpreter the kernel executes.
117// Only the in-guest LiteInst program check uses it.
118#[cfg(any(test, feature = "liteinst"))]
119#[derive(Debug, Eq, PartialEq)]
120pub(crate) enum KernelScript {
121    /// The file does not start with `#!`; the kernel tries its other handlers.
122    NotScript,
123    /// The script handler runs no interpreter: the `#!` line names none, or it
124    /// has no newline and the interpreter path runs to the end of the
125    /// [`BINPRM_BUF_SIZE`] bytes the kernel reads, so it may be truncated.
126    /// Measured on Linux 7.1: `#!\n`, `#! \t\n` and a 254-byte path with no
127    /// newline fail with `ENOEXEC`, so the kernel tries its other handlers;
128    /// `#!` alone and `#!\0/bin/sh\n`, whose name a NUL leaves empty, fail the
129    /// `execve` with `EACCES`.
130    Declined,
131    /// The kernel executes this interpreter path, byte for byte.
132    Interpreter(PathBuf),
133}
134
135#[cfg(any(test, feature = "liteinst"))]
136impl KernelScript {
137    /// Source of truth: `load_script()` in `fs/binfmt_script.c`; the errors
138    /// noted on [`KernelScript::Declined`] were measured on Linux 7.1. Unlike
139    /// [`Shebang::from_buf`], which its existing callers keep,
140    /// this ends the interpreter path only at a space, tab or NUL: a carriage
141    /// return is part of the path, as it is for the kernel.
142    pub(crate) fn from_buf(buf: &[u8]) -> Self {
143        // The kernel reads the file's first BINPRM_BUF_SIZE bytes into a buffer
144        // that is NUL-padded when the file is shorter.
145        let mut header = [0_u8; BINPRM_BUF_SIZE];
146        let len = buf.len().min(BINPRM_BUF_SIZE);
147        header[..len].copy_from_slice(&buf[..len]);
148        if !header.starts_with(b"#!") {
149            return Self::NotScript;
150        }
151        let last = BINPRM_BUF_SIZE - 1;
152        // `strnchr` stops at the first NUL.
153        let newline = header
154            .iter()
155            .take_while(|byte| **byte != 0)
156            .position(|byte| *byte == b'\n');
157        let line_end = match newline {
158            Some(newline) => newline,
159            None => {
160                // Without a newline the interpreter path must be followed by a
161                // space, tab or NUL within the buffer, or it may be truncated.
162                let Some(start) = (2..=last).find(|&index| !matches!(header[index], b' ' | b'\t'))
163                else {
164                    return Self::Declined;
165                };
166                if !header[start..=last]
167                    .iter()
168                    .any(|byte| matches!(byte, b' ' | b'\t' | 0))
169                {
170                    return Self::Declined;
171                }
172                // The kernel overwrites the buffer's last byte with a NUL.
173                last
174            }
175        };
176        let line = &header[2..line_end];
177        let start = line
178            .iter()
179            .position(|byte| !matches!(byte, b' ' | b'\t'))
180            .unwrap_or(line.len());
181        let name = &line[start..];
182        let name = &name[..name
183            .iter()
184            .position(|byte| matches!(byte, b' ' | b'\t' | 0))
185            .unwrap_or(name.len())];
186        if name.is_empty() {
187            Self::Declined
188        } else {
189            Self::Interpreter(PathBuf::from(OsStr::from_bytes(name)))
190        }
191    }
192}

198    #[test]
199    fn shebang_invalid_shebang() {
200        assert_eq!(Shebang::from_buf(b"! /bin/bash"), None);

203    #[test]
204    fn shebang_empty_interpreter() {
205        for input in [b"#!".as_slice(), b"#!\n", b"#! \t\n"] {
206            assert_eq!(
207                Shebang::from_buf(input).map(|shebang| shebang.program),
208                Some(PathBuf::new()),
209                "input: {input:?}"
210            );
211        }

214    #[test]
215    fn shebang_bin_bash() {
216        assert_eq!(
217            Shebang::from_buf(b"#! /bin/bash").map(|s| s.program),
218            Some(PathBuf::from("/bin/bash"))
219        );
220        assert_eq!(
221            Shebang::from_buf(b"#!  \t/bin/bash").map(|s| s.program),
222            Some(PathBuf::from("/bin/bash"))
223        );
224        assert_eq!(
225            Shebang::from_buf(b"#!/bin/bash\nfoobar").map(|s| s.program),
226            Some(PathBuf::from("/bin/bash"))
227        );
228        assert_eq!(
229            Shebang::from_buf(b"#!/bin/bash").map(|s| s.program),
230            Some(PathBuf::from("/bin/bash"))
231        );

287    fn kernel_interpreter(path: &[u8]) -> KernelScript {
288        KernelScript::Interpreter(PathBuf::from(OsStr::from_bytes(path)))

291    #[test]
292    fn kernel_script_reads_the_interpreter_as_load_script_does() {
293        let cases: [(&[u8], KernelScript); 7] = [
294            (b"\x7fELF\x02\x01\x01", KernelScript::NotScript),
295            (b"! /bin/bash", KernelScript::NotScript),
296            // A carriage return is part of the path, unlike in Shebang.
297            (b"#!/tmp/interp\r\n", kernel_interpreter(b"/tmp/interp\r")),
298            (
299                b"#! \t/usr/bin/env python3 -c\nbody",
300                kernel_interpreter(b"/usr/bin/env"),
301            ),
302            // A NUL ends the path.
303            (b"#!/bin/sh\0junk\n", kernel_interpreter(b"/bin/sh")),
304            // A short file is NUL-padded, which ends the path.
305            (b"#!/bin/bash", kernel_interpreter(b"/bin/bash")),
306            (b"#!relative arg\n", kernel_interpreter(b"relative")),
307        ];
308        for (input, expected) in cases {
309            assert_eq!(KernelScript::from_buf(input), expected, "input: {input:?}");
310        }
311    }
312
313    #[test]
314    fn kernel_script_declines_a_missing_or_truncated_interpreter() {
315        for input in [b"#!".as_slice(), b"#!\n", b"#! \t\n", b"#!\0/bin/sh\n"] {
316            assert_eq!(
317                KernelScript::from_buf(input),
318                KernelScript::Declined,
319                "input: {input:?}"
320            );
321        }
322        // No newline and no space, tab or NUL in the first 256 bytes: the path
323        // may be truncated, so the kernel declines it, whatever follows.
324        let mut long = b"#!/".to_vec();
325        long.resize(BINPRM_BUF_SIZE, b'a');
326        assert_eq!(KernelScript::from_buf(&long), KernelScript::Declined);
327        long.extend_from_slice(b" arg\n");
328        assert_eq!(KernelScript::from_buf(&long), KernelScript::Declined);
329        // A space in the last byte ends the path inside the buffer.
330        long.truncate(BINPRM_BUF_SIZE);
331        long[BINPRM_BUF_SIZE - 1] = b' ';
332        let expected = long[2..BINPRM_BUF_SIZE - 1].to_vec();
333        assert_eq!(KernelScript::from_buf(&long), kernel_interpreter(&expected));
334    }
