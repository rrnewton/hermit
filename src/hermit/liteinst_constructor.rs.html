24fn invalid(message: &str) -> io::Error {
25    io::Error::new(io::ErrorKind::InvalidData, message)
26}

28fn file_span(bytes: &[u8], offset: u64, size: u64) -> io::Result<Range<usize>> {
29    let start = usize::try_from(offset).map_err(|_| invalid("file extent offset overflow"))?;
30    let size = usize::try_from(size).map_err(|_| invalid("file extent size overflow"))?;
31    let end = start
32        .checked_add(size)
33        .ok_or_else(|| invalid("file extent end overflow"))?;
34    bytes
35        .get(start..end)
36        .ok_or_else(|| invalid("file extent is truncated"))?;
37    Ok(start..end)
38}

40fn loaded_file_span(
41    bytes: &[u8],
42    elf: &Elf<'_>,
43    address: u64,
44    size: u64,
45) -> io::Result<Range<usize>> {
46    let end = address
47        .checked_add(size)
48        .ok_or_else(|| invalid("loaded extent end overflow"))?;
49    let mut found = None;
50    for segment in &elf.program_headers {
51        if segment.p_type != program_header::PT_LOAD {
52            continue;
53        }
54        let top = segment
55            .p_vaddr
56            .checked_add(segment.p_filesz)
57            .ok_or_else(|| invalid("loaded file extent overflow"))?;
58        if address < segment.p_vaddr || end > top {
59            continue;
60        }
61        if size == 0 || segment.p_filesz > segment.p_memsz || found.is_some() {
62            return Err(invalid("loaded file extent is empty or ambiguous"));
63        }
64        let offset = segment
65            .p_offset
66            .checked_add(address - segment.p_vaddr)
67            .ok_or_else(|| invalid("loaded file offset overflow"))?;
68        found = Some(file_span(bytes, offset, size)?);
69    }
70    found.ok_or_else(|| invalid("loaded extent has no complete file backing"))
71}

73fn tag(elf: &Elf<'_>, wanted: u64) -> io::Result<Option<u64>> {
74    let dynamic = elf
75        .dynamic
76        .as_ref()
77        .ok_or_else(|| invalid("Detcore has no dynamic table"))?;
78    let mut entries = dynamic.dyns.iter().filter(|entry| entry.d_tag == wanted);
79    let value = entries.next().map(|entry| entry.d_val);
80    if entries.next().is_some() {
81        return Err(invalid("duplicate constructor dynamic tag"));
82    }
83    Ok(value)
84}

86// x86-64 ABI write widths. COPY and unknown relocation operations cannot prove
87// a sole constructor writer and are unsupported in this discovery branch.
88fn relocation_width(kind: u32) -> io::Result<u64> {
89    match kind {
90        0 | 35 => Ok(0), // NONE, TLSDESC_CALL (no data write)
91        1 | 6..=8 | 16..=18 | 24..=25 | 27..=31 | 33 | 37..=38 => Ok(8),
92        2..=4 | 9..=11 | 19..=23 | 26 | 32 | 34 | 41..=42 => Ok(4),
93        12..=13 => Ok(2),
94        14..=15 => Ok(1),
95        36 => Ok(16), // TLSDESC
96        _ => Err(invalid("unsupported constructor relocation writer")),
97    }
98}

100/// This discovers the strict public binding/registration shape. It does not
101/// prove the full constructor bytes, native stack transition, loader layouts,
102/// allocator or preparation closure; those require the shared qualifier.
103pub(crate) fn validate(bytes: &[u8], elf: &Elf<'_>) -> io::Result<()> {
104    if !elf.is_64
105        || !elf.little_endian
106        || elf.header.e_type != header::ET_DYN
107        || elf.header.e_machine != header::EM_X86_64
108    {
109        return Err(invalid("Detcore is not a little-endian x86-64 DSO"));
110    }
111    let mut roots = elf
112        .dynsyms
113        .iter()
114        .filter(|symbol| elf.dynstrtab.get_at(symbol.st_name) == Some(ROOT));
115    let root = roots
116        .next()
117        .ok_or_else(|| invalid("Detcore constructor root is absent"))?;
118    if roots.next().is_some()
119        || root.st_bind() != sym::STB_GLOBAL
120        || root.st_visibility() != sym::STV_PROTECTED
121        || root.st_type() != sym::STT_FUNC
122        || root.st_shndx == section_header::SHN_UNDEF as usize
123        || root.st_size != 16
124        || root.st_value == 0
125    {
126        return Err(invalid(
127            "Detcore constructor root is not unique strong PROTECTED FUNC16",
128        ));
129    }
130    let root_end = root
131        .st_value
132        .checked_add(root.st_size)
133        .ok_or_else(|| invalid("Detcore constructor root extent overflow"))?;
134    let section = elf
135        .section_headers
136        .get(root.st_shndx)
137        .ok_or_else(|| invalid("Detcore constructor root section is absent"))?;
138    if section.sh_type != section_header::SHT_PROGBITS
139        || section.sh_flags & u64::from(section_header::SHF_EXECINSTR) == 0
140        || root.st_value < section.sh_addr
141        || section
142            .sh_addr
143            .checked_add(section.sh_size)
144            .is_none_or(|end| root_end > end)
145    {
146        return Err(invalid(
147            "Detcore constructor root has no executable complete section",
148        ));
149    }
150    let root_offset = section
151        .sh_offset
152        .checked_add(root.st_value - section.sh_addr)
153        .ok_or_else(|| invalid("Detcore constructor root file offset overflow"))?;
154    if loaded_file_span(bytes, elf, root.st_value, root.st_size)?
155        != file_span(bytes, root_offset, root.st_size)?
156        || !elf.program_headers.iter().any(|segment| {
157            segment.p_type == program_header::PT_LOAD
158                && segment.p_flags & (program_header::PF_R | program_header::PF_X)
159                    == (program_header::PF_R | program_header::PF_X)
160                && segment.p_flags & program_header::PF_W == 0
161                && root.st_value >= segment.p_vaddr
162                && segment
163                    .p_vaddr
164                    .checked_add(segment.p_filesz)
165                    .is_some_and(|end| root_end <= end)
166        })
167    {
168        return Err(invalid(
169            "Detcore constructor root has no unique file-backed RX mapping",
170        ));
171    }
172    let mut dynamics = elf
173        .program_headers
174        .iter()
175        .filter(|segment| segment.p_type == program_header::PT_DYNAMIC);
176    let dynamic_segment = dynamics
177        .next()
178        .ok_or_else(|| invalid("Detcore dynamic mapping is absent"))?;
179    if dynamics.next().is_some()
180        || dynamic_segment.p_filesz == 0
181        || !dynamic_segment.p_filesz.is_multiple_of(16)
182        || loaded_file_span(
183            bytes,
184            elf,
185            dynamic_segment.p_vaddr,
186            dynamic_segment.p_filesz,
187        )? != file_span(bytes, dynamic_segment.p_offset, dynamic_segment.p_filesz)?
188    {
189        return Err(invalid(
190            "Detcore dynamic mapping is ambiguous or physically inconsistent",
191        ));
192    }
193    let mut arrays = elf
194        .section_headers
195        .iter()
196        .filter(|section| section.sh_type == section_header::SHT_INIT_ARRAY);
197    let array = arrays
198        .next()
199        .ok_or_else(|| invalid("Detcore has no constructor array"))?;
200    if arrays.next().is_some()
201        || array.sh_size == 0
202        || !array.sh_addr.is_multiple_of(8)
203        || !array.sh_offset.is_multiple_of(8)
204        || !array.sh_size.is_multiple_of(8)
205        || !matches!(array.sh_entsize, 0 | 8)
206        || array.sh_flags & u64::from(section_header::SHF_ALLOC) == 0
207    {
208        return Err(invalid(
209            "Detcore constructor array is empty, ambiguous or unaligned",
210        ));
211    }
212    let end = array
213        .sh_addr
214        .checked_add(array.sh_size)
215        .ok_or_else(|| invalid("Detcore constructor array extent overflow"))?;
216    let array_file = file_span(bytes, array.sh_offset, array.sh_size)?;
217    if tag(elf, dynamic::DT_INIT_ARRAY)? != Some(array.sh_addr)
218        || tag(elf, dynamic::DT_INIT_ARRAYSZ)? != Some(array.sh_size)
219        || loaded_file_span(bytes, elf, array.sh_addr, array.sh_size)? != array_file
220    {
221        return Err(invalid(
222            "Detcore constructor dynamic/section/file identity differs",
223        ));
224    }
225    for packed in [35, 36, 37] {
226        // DT_RELRSZ, DT_RELR, DT_RELRENT
227        if tag(elf, packed)?.is_some() {
228            return Err(invalid(
229                "packed constructor relocation writers are unsupported",
230            ));
231        }
232    }
233    if tag(elf, dynamic::DT_PREINIT_ARRAY)?.is_some_and(|value| value != 0)
234        || tag(elf, dynamic::DT_PREINIT_ARRAYSZ)?.is_some_and(|value| value != 0)
235        || tag(elf, dynamic::DT_INIT)?
236            .is_some_and(|entry| (root.st_value..root_end).contains(&entry))
237    {
238        return Err(invalid("Detcore has a competing constructor registration"));
239    }
240    let mut slots = vec![None; array_file.len() / 8];
241    for relocation in elf
242        .dynrelas
243        .iter()
244        .chain(elf.dynrels.iter())
245        .chain(elf.pltrelocs.iter())
246    {
247        let width = relocation_width(relocation.r_type)?;
248        let writer_end = relocation
249            .r_offset
250            .checked_add(width)
251            .ok_or_else(|| invalid("constructor relocation writer overflow"))?;
252        if width == 0 || relocation.r_offset >= end || writer_end <= array.sh_addr {
253            continue;
254        }
255        if width != 8
256            || relocation.r_offset < array.sh_addr
257            || writer_end > end
258            || !(relocation.r_offset - array.sh_addr).is_multiple_of(8)
259        {
260            return Err(invalid(
261                "constructor array has a partial or unaligned writer",
262            ));
263        }
264        if relocation.r_type != reloc::R_X86_64_RELATIVE || relocation.r_sym != 0 {
265            return Err(invalid("constructor slot is not symbol-zero RELATIVE"));
266        }
267        let target = relocation
268            .r_addend
269            .and_then(|addend| u64::try_from(addend).ok())
270            .ok_or_else(|| invalid("constructor slot has no nonnegative RELA addend"))?;
271        let index = usize::try_from((relocation.r_offset - array.sh_addr) / 8)
272            .map_err(|_| invalid("constructor slot index overflow"))?;
273        if slots[index].replace(target).is_some() {
274            return Err(invalid("constructor slot has duplicate relocation writers"));
275        }
276        if target != root.st_value && (root.st_value..root_end).contains(&target) {
277            return Err(invalid("constructor slot enters the root interior"));
278        }
279    }
280    if slots.iter().any(Option::is_none) {
281        return Err(invalid("constructor slot has no sole RELATIVE relocation"));
282    }
283    if slots
284        .iter()
285        .filter(|target| **target == Some(root.st_value))
286        .count()
287        != 1
288    {
289        return Err(invalid(
290            "Detcore has no unique exact-root RELATIVE constructor slot",
291        ));
292    }
293    Ok(())
294}

309    fn word(bytes: &mut [u8], offset: usize, value: u16) {
310        bytes[offset..offset + 2].copy_from_slice(&value.to_le_bytes());
311    }
312
313    fn dword(bytes: &mut [u8], offset: usize, value: u32) {
314        bytes[offset..offset + 4].copy_from_slice(&value.to_le_bytes());
315    }
316
317    fn qword(bytes: &mut [u8], offset: usize, value: u64) {
318        bytes[offset..offset + 8].copy_from_slice(&value.to_le_bytes());
319    }

321    fn section(bytes: &mut [u8], index: usize, fields: (u32, u64, u64, u64, u64, u64)) {
322        let (kind, flags, address, offset, size, entry_size) = fields;
323        let at = SECTIONS + index * 64;
324        dword(bytes, at + 4, kind);
325        qword(bytes, at + 8, flags);
326        qword(bytes, at + 16, address);
327        qword(bytes, at + 24, offset);
328        qword(bytes, at + 32, size);
329        qword(bytes, at + 48, 8);
330        qword(bytes, at + 56, entry_size);
331    }

333    fn image(entries: usize) -> Vec<u8> {
334        let mut bytes = vec![0; 0x800];
335        bytes[..7].copy_from_slice(b"\x7fELF\x02\x01\x01");
336        word(&mut bytes, 16, header::ET_DYN);
337        word(&mut bytes, 18, header::EM_X86_64);
338        dword(&mut bytes, 20, 1);
339        qword(&mut bytes, 32, 64);
340        qword(&mut bytes, 40, SECTIONS as u64);
341        word(&mut bytes, 52, 64);
342        word(&mut bytes, 54, 56);
343        word(&mut bytes, 56, 3);
344        word(&mut bytes, 58, 64);
345        word(&mut bytes, 60, 6);
346        for (index, kind, flags, offset, address, size) in [
347            (0, program_header::PT_LOAD, 5, 0, 0x1000, 0x200),
348            (1, program_header::PT_LOAD, 6, 0x200, 0x2000, 0x380),
349            (
350                2,
351                program_header::PT_DYNAMIC,
352                6,
353                DYNAMIC as u64,
354                0x2040,
355                176,
356            ),
357        ] {
358            let at = 64 + index * 56;
359            dword(&mut bytes, at, kind);
360            dword(&mut bytes, at + 4, flags);
361            qword(&mut bytes, at + 8, offset);
362            qword(&mut bytes, at + 16, address);
363            qword(&mut bytes, at + 32, size);
364            qword(&mut bytes, at + 40, size);
365            qword(&mut bytes, at + 48, 1);
366        }
367        bytes[0x100..0x110].copy_from_slice(&[0x90; 16]);
368        let strings = format!("\0{ROOT}\0");
369        bytes[0x360..0x360 + strings.len()].copy_from_slice(strings.as_bytes());
370        for (index, (kind, value)) in [
371            (dynamic::DT_STRTAB, 0x2160),
372            (dynamic::DT_SYMTAB, 0x2100),
373            (dynamic::DT_STRSZ, strings.len() as u64),
374            (dynamic::DT_SYMENT, 24),
375            (dynamic::DT_HASH, 0x21a0),
376            (dynamic::DT_RELA, 0x2200),
377            (dynamic::DT_RELASZ, entries as u64 * 24),

379            (dynamic::DT_INIT_ARRAY, ARRAY_ADDRESS),
380            (dynamic::DT_INIT_ARRAYSZ, entries as u64 * 8),
381            (dynamic::DT_NULL, 0),
382        ]
383        .into_iter()
384        .enumerate()
385        {
386            qword(&mut bytes, DYNAMIC + index * 16, kind);
387            qword(&mut bytes, DYNAMIC + index * 16 + 8, value);
388        }
389        dword(&mut bytes, SYMBOL, 1);
390        bytes[SYMBOL + 4] = (sym::STB_GLOBAL << 4) | sym::STT_FUNC;
391        bytes[SYMBOL + 5] = sym::STV_PROTECTED;
392        word(&mut bytes, SYMBOL + 6, 1);
393        qword(&mut bytes, SYMBOL + 8, ROOT_ADDRESS);
394        qword(&mut bytes, SYMBOL + 16, 16);
395        for (index, value) in [1, 2, 1, 0, 0].into_iter().enumerate() {
396            dword(&mut bytes, 0x3a0 + index * 4, value);
397        }
398        for index in 0..entries {
399            qword(
400                &mut bytes,
401                RELA + index * 24,
402                ARRAY_ADDRESS + index as u64 * 8,
403            );
404            qword(
405                &mut bytes,
406                RELA + index * 24 + 8,
407                u64::from(reloc::R_X86_64_RELATIVE),
408            );
409            qword(&mut bytes, RELA + index * 24 + 16, ROOT_ADDRESS);
410        }
411        section(
412            &mut bytes,

427        );
428        section(
429            &mut bytes,
430            3,
431            (section_header::SHT_DYNSYM, 2, 0x2100, 0x300, 48, 24),

462    fn discover(bytes: &[u8]) -> io::Result<()> {
463        let mut file = tempfile::NamedTempFile::new()?;
464        file.write_all(bytes)?;
465        crate::validate_preload_constructor_library(file.path(), "unit Detcore", &[ROOT], ROOT)
466            .map(|_| ())
467    }
468
469    fn refused(bytes: &[u8], reason: &str) {
470        let error = discover(bytes).expect_err("invalid constructor was discovered");
471        assert_eq!(error.kind(), io::ErrorKind::InvalidData);
472        assert!(
473            error.to_string().contains(reason),
474            "{error}; expected {reason}"
475        );
476    }

479    fn protected_relative_constructor_with_zero_raw_slot_is_discovered() {
480        let bytes = image(1);
481        assert_eq!(&bytes[0x200..0x208], &[0; 8]);
482        discover(&bytes).expect("local RELATIVE addend must be resolved without raw pointers");
483        // LLD may leave this advisory section field absent. The x86-64 ABI,

486        qword(&mut absent_entry_size, SECTIONS + 2 * 64 + 56, 0);
487        discover(&absent_entry_size)
488            .expect("absent section entry size must retain ABI pointer width");
