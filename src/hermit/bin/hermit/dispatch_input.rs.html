3//! Dispatch input custody, used only by the pre-fbinit executable entry.
4//!
5//! The frame conveys only an existing stdin reservation/error. It grants no
6//! private role, signal permission, resource admission, verdict or writer
7//! authority. Shared-input manifests remain untouched for the target's normal
8//! single adoption before descriptor retirement.

42/// Raw Unix environment at the single-threaded executable boundary. Keep
43/// duplicate entries, their order, empty values and non-UTF8 bytes. A map or
44/// reconstructed `vars_os` iterator is not the exec protocol's authority.
45pub(super) struct Environment {
46    entries: Vec<Vec<u8>>,
47}
48
49impl Environment {
50    /// Called before fbinit, workers or any environment mutation by the app.
51    pub(super) fn capture() -> Result<Self> {
52        let limit = unsafe { libc::sysconf(libc::_SC_ARG_MAX) };
53        ensure!(limit > 0, "cannot bound the current Unix environment");
54        let limit = usize::try_from(limit)?;
55        let table = unsafe { environ };
56        if table.is_null() {
57            return Ok(Self {
58                entries: Vec::new(),
59            });
60        }
61        let mut entries = Vec::new();
62        let mut total = 0usize;
63        for index in 0..=limit {
64            // SAFETY: this is the runtime's NUL-terminated environment table;
65            // the executable entry is single-threaded and does not mutate it.
66            let pointer = unsafe { *table.add(index) };
67            if pointer.is_null() {
68                return Ok(Self { entries });
69            }
70            let remaining = limit
71                .checked_sub(total)
72                .context("Unix environment exceeds the kernel argument bound")?;
73            let length = unsafe { libc::strnlen(pointer, remaining.saturating_add(1)) };
74            ensure!(
75                length <= remaining,
76                "Unix environment exceeds the kernel argument bound"
77            );
78            total = total
79                .checked_add(length + 1)
80                .context("Unix environment length overflow")?;
81            ensure!(
82                total <= limit,
83                "Unix environment exceeds the kernel argument bound"
84            );
85            entries
86                .push(unsafe { std::slice::from_raw_parts(pointer.cast::<u8>(), length) }.to_vec());
87        }
88        bail!("Unix environment has no bounded terminator")
89    }
90
91    pub(super) fn locator(&self) -> Result<Option<&OsStr>> {
92        let prefix = format!("{LOCATOR_ENV}=");
93        let mut values = self
94            .entries
95            .iter()
96            .filter_map(|entry| entry.strip_prefix(prefix.as_bytes()));
97        let value = values.next();
98        ensure!(
99            values.next().is_none(),
100            "dispatch input locator occurs more than once"
101        );
102        Ok(value.map(OsStr::from_bytes))
103    }
104
105    pub(super) fn digest(&self) -> String {
106        let mut framed = b"hermit-dispatch-raw-unix-environment/v1\0".to_vec();
107        framed.extend_from_slice(&(self.entries.len() as u64).to_le_bytes());
108        for entry in &self.entries {
109            framed.extend_from_slice(&(entry.len() as u64).to_le_bytes());
110            framed.extend_from_slice(entry);
111        }
112        detcore::Digest::new(&framed).to_string()
113    }
114
115    pub(super) fn require_unchanged(&self) -> Result<()> {
116        ensure!(
117            Self::capture()?.entries == self.entries,
118            "raw Unix environment changed during executable dispatch; no backend was substituted"
119        );
120        Ok(())
121    }
122
123    /// The identity probe uses an empty environment. Loader injection cannot
124    /// be silently cleared for final exec or admitted as if the probe had
125    /// observed those constructors. Refuse it, preserving the caller's bytes.
126    pub(super) fn require_probe_environment(&self) -> Result<()> {
127        for entry in &self.entries {
128            if let Some(index) = entry.iter().position(|byte| *byte == b'=') {
129                let name = &entry[..index];
130                let value = &entry[index + 1..];
131                ensure!(
132                    !name.starts_with(b"LD_") || value.is_empty(),
133                    "external dispatch cannot qualify dynamic-loader LD_* overrides; unset them for this invocation or use the backend executable directly"
134                );
135            }
136        }
137        Ok(())
138    }
139
140    pub(super) fn inherited(&self, value: &OsStr) -> Result<Vec<std::ffi::CString>> {
141        ensure!(
142            self.locator()?.is_none(),
143            "a dispatch handoff cannot be forwarded recursively"
144        );
145        let mut entries = self
146            .entries
147            .iter()
148            .map(|entry| std::ffi::CString::new(entry.as_slice()))
149            .collect::<std::result::Result<Vec<_>, _>>()?;
150        let mut locator = format!("{LOCATOR_ENV}=").into_bytes();
151        locator.extend_from_slice(value.as_bytes());
152        entries.push(std::ffi::CString::new(locator)?);
153        Ok(entries)
154    }
155
156    fn original(&self, fd: RawFd, expected_digest: &str) -> Result<Self> {
157        let locator = format!("{LOCATOR_ENV}={fd}").into_bytes();
158        ensure!(
159            self.locator()?.is_some() && self.entries.last() == Some(&locator),
160            "dispatch input locator is not the single appended handoff descriptor"
161        );
162        let original = Self {
163            entries: self.entries[..self.entries.len() - 1].to_vec(),
164        };
165        ensure!(
166            original.digest() == expected_digest,
167            "raw Unix environment changed across executable dispatch"
168        );
169        Ok(original)
170    }
171
172    pub(super) fn restore_after_adoption(&self) -> Result<()> {
173        // SAFETY: the executable facade is still before fbinit/workers. The
174        // validated protocol appended only this name; no original entry has it.
175        unsafe { std::env::remove_var(LOCATOR_ENV) };
176        self.require_unchanged()
177    }
178}

184    inode: u64,
185    mode: u32,
186    special_device: u64,
187    status_flags: i32,
188}

190fn descriptor_identity(fd: RawFd) -> Result<DescriptorIdentity> {
191    let mut stat = std::mem::MaybeUninit::<libc::stat>::uninit();
192    // SAFETY: fstat initializes the supplied stat on success; no data is read
193    // from the descriptor and its open-file-description offset is unchanged.
194    ensure!(
195        unsafe { libc::fstat(fd, stat.as_mut_ptr()) } == 0,
196        "cannot identify original stdin reservation: {}",
197        std::io::Error::last_os_error()
198    );
199    let stat = unsafe { stat.assume_init() };
200    let flags = unsafe { libc::fcntl(fd, libc::F_GETFL) };
201    ensure!(
202        flags >= 0,
203        "cannot query original stdin status: {}",
204        std::io::Error::last_os_error()
205    );
206    Ok(DescriptorIdentity {
207        device: stat.st_dev,
208        inode: stat.st_ino,
209        mode: stat.st_mode,
210        special_device: stat.st_rdev,
211        status_flags: flags,
212    })
213}

215fn descriptor_flags(fd: RawFd) -> Result<i32> {
216    let flags = unsafe { libc::fcntl(fd, libc::F_GETFD) };
217    ensure!(
218        flags >= 0,
219        "cannot query inherited descriptor: {}",
220        std::io::Error::last_os_error()
221    );
222    Ok(flags)
223}

225fn set_flags(fd: RawFd, flags: i32) -> Result<()> {
226    ensure!(
227        unsafe { libc::fcntl(fd, libc::F_SETFD, flags) } == 0,
228        "cannot restore dispatch descriptor ownership: {}",
229        std::io::Error::last_os_error()
230    );
231    Ok(())
232}

234fn process_generation_of(pid: u32) -> Result<u64> {
235    use std::io::Read;
236    let mut bytes = Vec::new();
237    File::open(format!("/proc/{pid}/stat"))?
238        .take(8193)
239        .read_to_end(&mut bytes)?;
240    ensure!(
241        bytes.len() <= 8192,
242        "process generation record exceeds its bound"
243    );
244    // comm is arbitrary Linux bytes; only its numeric suffix is UTF8.
245    let boundary = bytes
246        .windows(2)
247        .rposition(|bytes| bytes == b") ")
248        .context("process generation has no comm boundary")?;
249    let fields = std::str::from_utf8(&bytes[boundary + 2..])?;
250    // The suffix starts at field3 (state); starttime is field22.
251    fields
252        .split_whitespace()
253        .nth(19)
254        .context("process generation has no starttime")?
255        .parse()
256        .context("process generation has invalid starttime")
257}
258
259fn process_generation() -> Result<u64> {
260    process_generation_of(std::process::id())
261}

275/// `waitid` equivalents of a pending wait, WIFSTOPPED and terminal
276/// WIFEXITED/WIFSIGNALED. None of the latter can satisfy mapping custody.
277pub(super) enum StopObservation {
278    Pending,
279    Stopped,
280    Terminated { code: i32, status: i32 },
281}
282
283fn pidfd_pid(fd: RawFd) -> Result<u32> {
284    use std::io::Read;
285    let mut bytes = Vec::new();
286    File::open(format!("/proc/self/fdinfo/{fd}"))?
287        .take(8193)
288        .read_to_end(&mut bytes)?;
289    ensure!(bytes.len() <= 8192, "pidfd identity exceeds its bound");
290    let text = std::str::from_utf8(&bytes)?;
291    let mut pids = text.lines().filter_map(|line| line.strip_prefix("Pid:\t"));
292    let pid = pids
293        .next()
294        .context("dispatch cleanup descriptor is not a pidfd")?
295        .parse()?;
296    ensure!(pids.next().is_none(), "dispatch pidfd repeats its identity");
297    Ok(pid)
298}

300impl ProbeReaper {
301    pub(super) fn observe_stop(fd: RawFd, pid: u32) -> Result<StopObservation> {
302        let mut info = std::mem::MaybeUninit::<libc::siginfo_t>::zeroed();
303        ensure!(
304            unsafe {
305                libc::waitid(
306                    libc::P_PIDFD,
307                    fd as libc::id_t,
308                    info.as_mut_ptr(),
309                    libc::WSTOPPED | libc::WEXITED | libc::WNOHANG | libc::WNOWAIT,
310                )
311            } == 0,
312            "cannot observe held-probe stop: {}",
313            std::io::Error::last_os_error()
314        );
315        let info = unsafe { info.assume_init() };
316        let actual_pid = unsafe { info.si_pid() };
317        if actual_pid == 0 {
318            return Ok(StopObservation::Pending);
319        }
320        ensure!(
321            actual_pid as u32 == pid,
322            "held-probe stop wait returned another child"
323        );
324        let status = unsafe { info.si_status() };
325        match info.si_code {
326            libc::CLD_STOPPED => {
327                ensure!(
328                    status == libc::SIGSTOP,
329                    "held probe stopped for an unexpected signal: {status}"
330                );
331                Ok(StopObservation::Stopped)
332            }
333            code @ (libc::CLD_EXITED | libc::CLD_KILLED | libc::CLD_DUMPED) => {
334                Ok(StopObservation::Terminated { code, status })
335            }
336            code => bail!("unexpected held-probe wait event: code={code}, status={status}"),
337        }
338    }
339
340    pub(super) fn require_stopped(fd: RawFd, pid: u32) -> Result<()> {
341        match Self::observe_stop(fd, pid)? {
342            StopObservation::Stopped => Ok(()),
343            StopObservation::Pending => bail!("held-probe stop has not been observed"),
344            StopObservation::Terminated { code, status } => {
345                bail!("held probe ended before executable custody: code={code}, status={status}")
346            }
347        }
348    }
349
350    pub(super) fn from_parent(fd: RawFd, pid: u32) -> Result<Self> {
351        ensure!(
352            fd > 2 && !(198..=204).contains(&fd),
353            "invalid held-probe pidfd"
354        );
355        ensure!(
356            pidfd_pid(fd)? == pid,
357            "held probe pidfd belongs to another process"
358        );
359        Self::require_stopped(fd, pid)?;
360        Ok(Self {
361            fd,
362            pid,
363            start_ticks: process_generation_of(pid)?,
364            descriptor: descriptor_identity(fd)?,
365        })
366    }
367
368    fn bind_child(&self) -> Result<()> {
369        ensure!(
370            self.fd > 2 && !(198..=204).contains(&self.fd),
371            "invalid inherited probe pidfd"
372        );
373        ensure!(
374            pidfd_pid(self.fd)? == self.pid && process_generation_of(self.pid)? == self.start_ticks,
375            "held probe process generation changed"
376        );
377        ensure!(
378            descriptor_identity(self.fd)? == self.descriptor,
379            "inherited held-probe pidfd changed"
380        );
381        // P_PIDFD wait is also an ownership check: an unrelated process is
382        // not our child and cannot be signaled/reaped by this protocol.
383        self.observe(false)?;
384        Ok(())
385    }
386
387    fn observe(&self, reap: bool) -> Result<Option<(i32, i32)>> {
388        let mut info = std::mem::MaybeUninit::<libc::siginfo_t>::zeroed();
389        let flags = libc::WEXITED | libc::WNOHANG | if reap { 0 } else { libc::WNOWAIT };
390        ensure!(
391            unsafe {
392                libc::waitid(
393                    libc::P_PIDFD,
394                    self.fd as libc::id_t,
395                    info.as_mut_ptr(),
396                    flags,
397                )
398            } == 0,
399            "cannot observe this exact held-probe child: {}",
400            std::io::Error::last_os_error()
401        );
402        let info = unsafe { info.assume_init() };
403        let pid = unsafe { info.si_pid() };
404        if pid == 0 {
405            return Ok(None);
406        }
407        ensure!(
408            pid as u32 == self.pid,
409            "held-probe wait returned another child"
410        );
411        Ok(Some((info.si_code, unsafe { info.si_status() })))
412    }

414    fn complete_after(&self, validate_input: impl FnOnce() -> Result<()>) -> Result<()> {
415        self.bind_child()?;
416        // Claim the descriptor only after its exact process/generation and
417        // current child relationship are proven. Untrusted/mismatched frame
418        // FDs are never signaled or adopted by a refusal path.
419        let owner = unsafe { OwnedFd::from_raw_fd(self.fd) };
420        let mut reaped = false;
421        let result = (|| -> Result<()> {
422            Self::require_stopped(self.fd, self.pid)?;
423            ensure!(
424                descriptor_flags(self.fd)? & libc::FD_CLOEXEC == 0,
425                "held-probe pidfd was not transferred through exec"
426            );
427            validate_input()?;
428            // The parent's control endpoint was CLOEXEC. Its closure releases the
429            // old mapping only after this image's exec has admitted its executable.
430            // No input owner is adopted until that exact child is cleanly reaped.
431            set_flags(self.fd, libc::FD_CLOEXEC)?;
432            ensure!(
433                unsafe {
434                    libc::syscall(
435                        libc::SYS_pidfd_send_signal,
436                        self.fd,
437                        libc::SIGCONT,
438                        std::ptr::null::<libc::siginfo_t>(),
439                        0,
440                    )
441                } == 0,
442                "cannot release the exact held probe after exec: {}",
443                std::io::Error::last_os_error()
444            );
445            let started = Instant::now();
446            while started.elapsed() < Duration::from_secs(2) {
447                if let Some((code, status)) = self.observe(false)? {
448                    ensure!(
449                        self.observe(true)? == Some((code, status)),
450                        "held-probe reap changed its status"
451                    );
452                    reaped = true;
453                    ensure!(
454                        code == libc::CLD_EXITED && status == 0,
455                        "held-probe custody ended abnormally: code={code}, status={status}"
456                    );
457                    return Ok(());
458                }
459                std::thread::sleep(Duration::from_millis(10));
460            }
461            bail!("held-probe completion exceeded2seconds")
462        })();
463        // Preserve the original refusal even if cleanup succeeds. Handle every
464        // error after the trusted binding, including flag/SIGCONT/wait errors.
465        let cleanup = if result.is_err() && !reaped {
466            self.kill_and_reap()
467        } else {
468            Ok(())
469        };
470        drop(owner);
471        result.with_context(|| format!("held-probe completion refused; cleanup={cleanup:?}"))
472    }
473
474    fn kill_and_reap(&self) -> Result<()> {
475        let signal = unsafe {
476            libc::syscall(
477                libc::SYS_pidfd_send_signal,
478                self.fd,
479                libc::SIGKILL,
480                std::ptr::null::<libc::siginfo_t>(),
481                0,
482            )
483        };
484        let signal_error = (signal < 0).then(std::io::Error::last_os_error);
485        let started = Instant::now();
486        while started.elapsed() < Duration::from_secs(2) {
487            if let Some(expected) = self.observe(false)? {
488                ensure!(
489                    self.observe(true)? == Some(expected),
490                    "held-probe cleanup changed its exact status"
491                );
492                if let Some(error) = signal_error
493                    && error.raw_os_error() != Some(libc::ESRCH)
494                {
495                    return Err(error).context("held-probe cleanup signal refused after reaping");
496                }
497                return Ok(());
498            }
499            std::thread::sleep(Duration::from_millis(10));
500        }
501        bail!("held-probe cleanup could not reap within2seconds; signal_error={signal_error:?}")
502    }
503}

505fn argv_digest(argv: &[OsString]) -> Result<String> {
506    // Length framing preserves empty/non-UTF8 operands and every byte after
507    // `--`. No original argv member is rewritten by this input protocol.
508    let mut framed = b"hermit-dispatch-raw-unix-argv/v1\0".to_vec();
509    framed.extend_from_slice(&(argv.len() as u64).to_le_bytes());
510    for argument in argv {
511        let bytes = argument.as_os_str().as_bytes();
512        ensure!(!bytes.contains(&0), "Unix argv contains NUL");
513        framed.extend_from_slice(&(bytes.len() as u64).to_le_bytes());
514        framed.extend_from_slice(bytes);
515    }
516    Ok(detcore::Digest::new(&framed).to_string())
517}

519fn public_argv(argv: &[OsString]) -> Result<()> {
520    ensure!(!argv.is_empty(), "dispatch input has no argv0");
521    // Existing private entries require their exact argv1 marker. The input
522    // frame cannot be used to enter one of those separately owned protocols.
523    ensure!(
524        !argv
525            .get(1)
526            .is_some_and(|arg| arg.as_os_str().as_bytes().starts_with(b"__hermit")),
527        "dispatch input cannot authorize a private executable role"
528    );
529    Ok(())
530}

534struct Frame {
535    schema: String,
536    pid: u32,
537    start_ticks: u64,
538    argv_sha256: String,
539    environment_sha256: String,
540    target: ExecutableIdentity,

548fn valid_encoding(value: i32) -> bool {
549    value >= 3 || (-4096..=-2).contains(&value)
550}

552fn sealed_frame(bytes: &[u8]) -> Result<OwnedFd> {
553    ensure!(
554        bytes.len() <= MAX_FRAME_BYTES,
555        "dispatch stdin frame exceeds its bound"
556    );
557    let original = super::verify_inputs::new_unsealed_memfd(c"hermit-dispatch-stdin")?;
558    // Keep the protocol frame outside descriptors reserved by the existing
559    // native-input adapter. This does not relocate the original stdin owner.
560    let raw = unsafe { libc::fcntl(original.as_raw_fd(), libc::F_DUPFD_CLOEXEC, 205) };
561    ensure!(
562        raw >= 0,
563        "cannot allocate dispatch input descriptor: {}",
564        std::io::Error::last_os_error()
565    );
566    let owner = unsafe { OwnedFd::from_raw_fd(raw) };
567    let mut writer = File::from(owner.try_clone()?);
568    writer.write_all(bytes)?;
569    drop(writer);
570    ensure!(
571        unsafe { libc::fcntl(owner.as_raw_fd(), libc::F_ADD_SEALS, SEALS) } == 0,
572        "cannot seal dispatch input frame: {}",
573        std::io::Error::last_os_error()
574    );
575    Ok(owner)
576}

578/// Affine parent-side ownership. `Drop` restores the original reservation on
579/// abandonment. The caller must use `restore` after an exec error so any flag
580/// restoration error is included in its ordinary refusal, never a fallback.
581pub(super) struct Transfer {
582    original: i32,
583    descriptor: Option<OwnedFd>,
584    original_flags: Option<i32>,
585    frame: OwnedFd,
586    armed: bool,
587    restored: bool,
588}
589
590impl Transfer {
591    pub(super) fn prepare(
592        argv: &[OsString],
593        environment: &Environment,
594        target: ExecutableIdentity,
595        core: CoreCompatibility,
596        probe: ProbeReaper,
597    ) -> Result<Self> {
598        public_argv(argv)?;
599        ensure!(
600            matches!(target, ExecutableIdentity::Backend { .. }),
601            "dispatch input requires a compiled backend target"
602        );
603        core.validate()?;
604        let original = super::STARTUP_STDIN.load(Ordering::Acquire);
605        ensure!(
606            valid_encoding(original),
607            "original startup stdin is not available exactly once"
608        );
609        let original_flags = if original >= 3 {
610            Some(descriptor_flags(original)?)
611        } else {
612            None
613        };
614        if let Some(flags) = original_flags {
615            ensure!(
616                flags & libc::FD_CLOEXEC != 0,
617                "startup stdin reservation is already inherited by another transfer"
618            );
619        }
620        let frame = Frame {
621            schema: SCHEMA.to_owned(),
622            pid: std::process::id(),
623            start_ticks: process_generation()?,
624            argv_sha256: argv_digest(argv)?,
625            environment_sha256: environment.digest(),
626            target,
627            core,
628            original,
629            descriptor: if original >= 3 {
630                Some(descriptor_identity(original)?)
631            } else {
632                None
633            },
634            probe,
635        };
636        let frame = sealed_frame(&serde_json::to_vec(&frame)?)?;
637        super::STARTUP_STDIN
638            .compare_exchange(
639                original,
640                super::STDIN_TAKEN,
641                Ordering::AcqRel,
642                Ordering::Acquire,
643            )
644            .map_err(|_| {
645                anyhow::anyhow!("startup stdin owner changed during dispatch preparation")
646            })?;
647        // Only after the atomic transfer succeeds does this guard own the FD.
648        let descriptor = (original >= 3).then(|| unsafe { OwnedFd::from_raw_fd(original) });
649        Ok(Self {
650            original,
651            descriptor,
652            original_flags,
653            frame,
654            armed: false,
655            restored: false,
656        })
657    }

659    /// The last pre-exec step, after probe/custody validation. No process is
660    /// spawned while the original stdin descriptor is inheritable.
661    pub(super) fn arm(&mut self) -> Result<(&'static str, OsString)> {
662        ensure!(!self.armed, "dispatch stdin transfer was armed twice");
663        if let (Some(fd), Some(flags)) = (&self.descriptor, self.original_flags) {
664            set_flags(fd.as_raw_fd(), flags & !libc::FD_CLOEXEC)?;
665        }
666        if let Err(error) = set_flags(self.frame.as_raw_fd(), 0) {
667            if let (Some(fd), Some(flags)) = (&self.descriptor, self.original_flags) {
668                set_flags(fd.as_raw_fd(), flags)
669                    .context("restore stdin after frame inheritance refusal")?;
670            }
671            return Err(error);
672        }
673        self.armed = true;
674        Ok((LOCATOR_ENV, self.frame.as_raw_fd().to_string().into()))
675    }
676
677    pub(super) fn environment(&self) -> (&'static str, OsString) {
678        (LOCATOR_ENV, self.frame.as_raw_fd().to_string().into())
679    }
680
681    fn restore_inner(&mut self) -> Result<()> {
682        if self.restored {
683            return Ok(());
684        }
685        if let (Some(fd), Some(flags)) = (&self.descriptor, self.original_flags) {
686            set_flags(fd.as_raw_fd(), flags)?;
687        }
688        set_flags(self.frame.as_raw_fd(), libc::FD_CLOEXEC)?;
689        super::STARTUP_STDIN
690            .compare_exchange(
691                super::STDIN_TAKEN,
692                self.original,
693                Ordering::AcqRel,
694                Ordering::Acquire,
695            )
696            .map_err(|_| anyhow::anyhow!("startup stdin owner changed before exec restoration"))?;
697        if let Some(fd) = self.descriptor.take() {
698            let _ = fd.into_raw_fd();
699        }
700        self.restored = true;
701        Ok(())
702    }
703
704    pub(super) fn restore(mut self) -> Result<()> {
705        self.restore_inner()
706    }
707}
708
709impl Drop for Transfer {
710    fn drop(&mut self) {
711        // Best effort during unwinding/abandonment. An explicit exec error uses
712        // restore() above; it cannot report successful restoration by Drop.
713        let _ = self.restore_inner();
714    }
715}

717fn read_frame(fd: RawFd) -> Result<Frame> {
718    ensure!(
719        fd > 2 && !(198..=204).contains(&fd),
720        "invalid dispatch input descriptor"
721    );
722    ensure!(
723        descriptor_flags(fd)? & libc::FD_CLOEXEC == 0,
724        "dispatch input descriptor was not inherited across exec"
725    );
726    let seals = unsafe { libc::fcntl(fd, libc::F_GET_SEALS) };
727    ensure!(
728        seals >= 0 && seals & SEALS == SEALS,
729        "dispatch input frame is not immutable"
730    );
731    let identity = descriptor_identity(fd)?;
732    ensure!(
733        identity.mode & libc::S_IFMT == libc::S_IFREG,
734        "dispatch input frame is not a sealed regular file"
735    );
736    let mut stat = std::mem::MaybeUninit::<libc::stat>::uninit();
737    ensure!(
738        unsafe { libc::fstat(fd, stat.as_mut_ptr()) } == 0,
739        "cannot size dispatch input frame: {}",
740        std::io::Error::last_os_error()
741    );
742    let size = unsafe { stat.assume_init() }.st_size;
743    ensure!(
744        size > 0 && size as u64 <= MAX_FRAME_BYTES as u64,
745        "dispatch input frame exceeds its bound or is empty"
746    );
747    let mut bytes = vec![0u8; size as usize];
748    // pread never changes the shared offset, including for a sealed frame.
749    let mut offset = 0;
750    while offset < bytes.len() {
751        let count = unsafe {
752            libc::pread(
753                fd,
754                bytes[offset..].as_mut_ptr().cast(),
755                bytes.len() - offset,
756                offset as libc::off_t,
757            )
758        };
759        if count < 0 && std::io::Error::last_os_error().raw_os_error() == Some(libc::EINTR) {
760            continue;
761        }
762        ensure!(
763            count > 0,
764            "cannot read complete dispatch input frame: {}",
765            std::io::Error::last_os_error()
766        );
767        offset += count as usize;
768    }
769    let frame: Frame = serde_json::from_slice(&bytes).context("invalid dispatch input frame")?;
770    ensure!(
771        frame.schema == SCHEMA && valid_encoding(frame.original),
772        "unsupported dispatch input encoding"
773    );
774    ensure!(
775        (frame.original >= 3) == frame.descriptor.is_some(),
776        "dispatch input descriptor/error encoding is incomplete"
777    );
778    Ok(frame)
779}
780
781fn restore_original_stdin_closedness(original: i32) -> Result<()> {
782    if original != -libc::EBADF - 1 {
783        return Ok(());
784    }
785    // Only the fully authenticated handoff's exact original EBADF encoding
786    // authorizes this close. Rust startup may have replaced fd0 with /dev/null,
787    // even when the preceding exec inherited a physically closed descriptor.
788    // Never retry close: on Linux an error can follow descriptor retirement.
789    if unsafe { libc::close(libc::STDIN_FILENO) } < 0 {
790        let error = std::io::Error::last_os_error();
791        if error.raw_os_error() != Some(libc::EBADF) {
792            return Err(error).context("cannot restore original closed stdin after dispatch");
793        }
794    }
795    Ok(())
796}

798/// Target-side entry, before fbinit/private handlers/retirement. This never
799/// reads fd0 and never adopts a shared-input manifest. The returned environment
800/// is the exact original environment, excluding only our appended locator.
801pub(super) fn adopt(
802    fd: RawFd,
803    argv: &[OsString],
804    environment: &Environment,
805    compiled: &ExecutableIdentity,
806    core: &CoreCompatibility,
807) -> Result<Environment> {
808    public_argv(argv)?;
809    ensure!(
810        matches!(compiled, ExecutableIdentity::Backend { .. }),
811        "a core executable cannot redispatch an input handoff"
812    );
813    let frame = read_frame(fd)?;
814    ensure!(
815        frame.pid == std::process::id() && frame.start_ticks == process_generation()?,
816        "dispatch input belongs to another process generation"
817    );
818    ensure!(
819        &frame.target == compiled && frame.argv_sha256 == argv_digest(argv)?,
820        "dispatch input is bound to another executable or raw argv"
821    );
822    let original_environment = environment.original(fd, &frame.environment_sha256)?;
823    frame.core.require_exact(core)?;
824    let fresh = super::STARTUP_STDIN.load(Ordering::Acquire);
825    ensure!(
826        valid_encoding(fresh),
827        "target binary did not reserve startup stdin exactly once"
828    );
829    ensure!(
830        fresh != fd && frame.original != fd,
831        "dispatch frame aliases a stdin owner"
832    );
833    ensure!(
834        frame.probe.fd != fd && frame.probe.fd != fresh && frame.probe.fd != frame.original,
835        "held-probe cleanup aliases an input owner"
836    );
837    frame.probe.complete_after(|| {
838        if let Some(expected) = &frame.descriptor {
839            ensure!(
840                frame.original != fresh,
841                "dispatch cannot adopt its fresh sanitized-stdin capture"
842            );
843            ensure!(
844                descriptor_flags(frame.original)? & libc::FD_CLOEXEC == 0,
845                "original stdin reservation was not inherited across exec"
846            );
847            ensure!(
848                &descriptor_identity(frame.original)? == expected,
849                "original stdin descriptor identity/status changed across dispatch"
850            );
851        }
852        Ok(())
853    })?;
854    // All validation is complete before either input descriptor is adopted.
855    set_flags(fd, libc::FD_CLOEXEC)?;
856    let frame_owner = unsafe { OwnedFd::from_raw_fd(fd) };
857    let original_owner = if frame.original >= 3 {
858        set_flags(frame.original, libc::FD_CLOEXEC)?;
859        Some(unsafe { OwnedFd::from_raw_fd(frame.original) })
860    } else {
861        None
862    };
863    super::STARTUP_STDIN
864        .compare_exchange(
865            fresh,
866            super::STDIN_TAKEN,
867            Ordering::AcqRel,
868            Ordering::Acquire,
869        )
870        .map_err(|_| anyhow::anyhow!("target startup stdin owner changed during handoff"))?;
871    if fresh >= 3 {
872        drop(unsafe { OwnedFd::from_raw_fd(fresh) });
873    }
874    if let Some(original) = original_owner {
875        let _ = original.into_raw_fd();
876    }
877    super::STARTUP_STDIN.store(frame.original, Ordering::Release);
878    restore_original_stdin_closedness(frame.original)?;
879    drop(frame_owner);
880    Ok(original_environment)
881}

883pub(super) fn locator(value: &OsStr) -> Result<RawFd> {
884    let bytes = value.as_bytes();
885    ensure!(
886        !bytes.is_empty() && bytes.iter().all(u8::is_ascii_digit),
887        "dispatch input locator is not a descriptor"
888    );
889    let fd: RawFd = std::str::from_utf8(bytes)?.parse()?;
890    if fd <= 2 || (198..=204).contains(&fd) {
891        bail!("dispatch input locator names a reserved descriptor");
892    }
893    Ok(fd)
894}
