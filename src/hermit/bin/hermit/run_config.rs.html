9//! Run configuration files: `hermit run --config FILE` and
10//! `hermit run --save-config FILE`.
11//!
12//! A run config is a YAML document (JSON is accepted, being YAML) that names
13//! `hermit run` options by their long flag names:
14//!
15//! ```yaml
16//! schema: hermit-run-config/v1
17//! hermit-version: 0.4.1 (2026-10-08, source revision not embedded)
18//! global:                 # options before the subcommand: hermit [GLOBAL] run
19//!   backend: ptrace
20//!   log: info
21//! run:                    # options of `hermit run`
22//!   epoch: 2026-10-08T17:33:00.123456789+00:00
23//!   seed: 42
24//!   verify: true          # a flag: `true` gives it, `false` is the same as omitting it
25//!   env: [FOO=1, HOME]    # a repeatable option: a list, one entry per occurrence
26//! program: /bin/echo
27//! args: [hello]
28//! ```
29//!
30//! The contract is that a run config means exactly what typing its options
31//! would. Loading turns every entry into the command-line option it names
32//! (`--seed=42`, `--env=FOO=1`, ...) and hands the result to the same clap
33//! parser as a typed command line, so value syntax, conflicts and
34//! requirements are checked once, in one place, for both. Nothing here knows
35//! what any particular option means; a new `hermit run` option is part of
36//! the format the moment it exists. The `run_config` tests require a
37//! round-trip sample for every option, so an option that does not survive
38//! save and load fails a test rather than a reproduction.
39//!
40//! Precedence, highest first: an option given on the command line, the
41//! file, the option's environment variable (`HERMIT_EPOCH`, `HERMIT_LOG`, ...),
42//! its default. A command-line option replaces the file's value for that
43//! option, a repeatable option's whole list included. A guest program on the
44//! command line replaces the file's `program` and `args` together. The file
45//! outranks the environment because it names its values explicitly, while
46//! an inherited variable is exactly the kind of implicit input it records.
47//!
48//! `--save-config` writes the options the invocation was given, on the
49//! command line or through their environment variables, plus the implicit
50//! inputs Hermit resolved for the run: an epoch sampled from the host clock
51//! or adopted from a replayed recording, and the seed `--seed-from` chose.
52//! It does not write defaults: an explicit default can collide with clap's
53//! conflict rules (`--network=local` beside `--no-namespace`), so the file
54//! reproduces the run under the hermit that wrote it, whose version it
55//! records. The host environment that `--base-env=host` passes through is
56//! not recorded; `--base-env=minimal` or `--base-env=empty` with `--env`
57//! makes the guest environment explicit.
58//!
59//! Unknown keys are refused, and so is a `schema` other than [`SCHEMA`]. The
60//! top-level `seccomp:` key is reserved for a future syscall policy section
61//! and is refused until that exists.

100/// One run config document. See the module documentation for the format.
101#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
102#[serde(deny_unknown_fields)]
103pub(crate) struct RunConfig {
104    /// Always [`SCHEMA`].
105    pub(crate) schema: String,
106    /// The version of the hermit that wrote the file. Informational: it is
107    /// never checked.
108    #[serde(
109        rename = "hermit-version",
110        default,
111        skip_serializing_if = "Option::is_none"
112    )]
113    pub(crate) hermit_version: Option<String>,
114    /// Options that come before the subcommand, by long name.
115    #[serde(default, skip_serializing_if = "BTreeMap::is_empty")]
116    pub(crate) global: BTreeMap<String, Value>,
117    /// Options of `hermit run`, by long name.
118    #[serde(default, skip_serializing_if = "BTreeMap::is_empty")]
119    pub(crate) run: BTreeMap<String, Value>,
120    /// The guest program.
121    #[serde(default, skip_serializing_if = "Option::is_none")]
122    pub(crate) program: Option<String>,
123    /// The guest program's arguments.
124    #[serde(default, skip_serializing_if = "Vec::is_empty")]
125    pub(crate) args: Vec<String>,
126    /// Reserved for a future syscall policy section; refused when present.
127    #[serde(default, skip_serializing_if = "Option::is_none")]
128    seccomp: Option<Value>,
129}

145    pub(crate) fn to_yaml(&self) -> Result<String, Error> {
146        let body = serde_yaml::to_string(self).context("cannot encode the run config as YAML")?;
147        if let Some(backend) = self.global.get("backend").and_then(Value::as_str)
148            && super::backend_hook::validate_name(backend).is_ok()
149        {

152                 # backend executable and `run --config FILE`; options given on that command\n\
153                 # line replace the ones here.\n{body}"
154            ));
155        }
156        Ok(format!(
157            "# Written by `hermit run --save-config`. Reproduce the run with\n\
158             # `hermit run --config FILE`; options given on that command line\n\
159             # replace the ones here.\n{body}"
160        ))
161    }

164/// Parses a run config document, refusing a missing or different `schema`,
165/// unknown top-level keys and the reserved `seccomp` section. Option keys are
166/// checked when the document is applied to a command ([`expand_argv`]).
167pub(crate) fn parse(text: &str) -> Result<RunConfig, Error> {
168    let document: Value =
169        serde_yaml::from_str(text).context("the file is not a YAML (or JSON) document")?;
170    let Value::Mapping(map) = &document else {
171        anyhow::bail!(
172            "a run config is a mapping that starts with `schema: {SCHEMA}`; this document is a \
173             {}",
174            kind(&document)
175        );
176    };
177    match map.get("schema") {
178        None => anyhow::bail!(
179            "the document has no `schema` key; a run config starts with `schema: {SCHEMA}`. \
180             Write one with `hermit run --save-config FILE`."
181        ),
182        Some(Value::String(schema)) if schema == SCHEMA => {}
183        Some(other) => {
184            let written_by = map
185                .get("hermit-version")
186                .and_then(Value::as_str)
187                .map(|version| format!(" (it was written by hermit {version})"))
188                .unwrap_or_default();
189            anyhow::bail!(
190                "the document has schema {}{written_by}, but this hermit reads only `{SCHEMA}`. \
191                 Load it with the hermit that wrote it, or write it again with this hermit's \
192                 `hermit run --save-config FILE`.",
193                display_value(other)
194            );
195        }
196    }
197    const KEYS: &str = "`schema`, `hermit-version`, `global`, `run`, `program` and `args`";
198    for key in map.keys() {
199        if !matches!(
200            key.as_str(),
201            Some("schema" | "hermit-version" | "global" | "run" | "program" | "args" | "seccomp")
202        ) {
203            anyhow::bail!(
204                "{} is not a run config key; the keys are {KEYS}",
205                display_value(key)
206            );
207        }
208    }
209    let config: RunConfig = serde_yaml::from_value(document).with_context(|| {
210        format!("a run config's keys are {KEYS}, described in docs/USER_GUIDE.md")
211    })?;
212    if config.seccomp.is_some() {
213        anyhow::bail!(
214            "the `seccomp` section is reserved for a syscall policy that this hermit does not \
215             support yet; remove it from the file"
216        );
217    }
218    Ok(config)
219}

242/// [`from_manifest`] for the manifest read from `path`. A network
243/// recording's rerun replays its trace: the trace the manifest names,
244/// checked against the digest it recorded, becomes `--replay-networking`. A
245/// syscall recording is replayed with `hermit replay`, not rerun.
246pub(crate) fn from_manifest_at(
247    manifest: &crate::app::manifest::Manifest,
248    path: &Path,
249) -> Result<RunConfig, Error> {
250    use crate::app::manifest::Kind;
251    let mut config = from_manifest(manifest)?;
252    match manifest.kind {
253        Kind::InputsOnly | Kind::History => {}
254        Kind::Network => {
255            let trace = crate::app::manifest::checked_network_trace(manifest, path)?;
256            let trace = trace.to_str().ok_or_else(|| {
257                anyhow::anyhow!("network trace path {} is not UTF-8", trace.display())
258            })?;
259            config
260                .run
261                .insert(REPLAY_NETWORKING_KEY.to_owned(), trace.into());
262        }
263        Kind::FullSyscall => anyhow::bail!(
264            "{} describes a syscall recording, which `hermit run --from` does not rerun; replay \
265             it with: hermit replay {}",
266            path.display(),
267            crate::app::manifest::directory_of(path).display()
268        ),
269    }
270    Ok(config)
271}

273/// The run config a manifest's `inputs` and `command` stand for, which
274/// `hermit run --from` loads as `--config` loads a file.
275pub(crate) fn from_manifest(manifest: &crate::app::manifest::Manifest) -> Result<RunConfig, Error> {
276    let section = |entries: &BTreeMap<String, serde_json::Value>| {
277        entries
278            .iter()
279            .map(|(long, value)| {
280                serde_json::from_value(value.clone())
281                    .map(|value| (long.clone(), value))
282                    .with_context(|| {
283                        format!("the manifest's value for {long} is not an option value")
284                    })
285            })
286            .collect::<Result<BTreeMap<String, Value>, Error>>()
287    };
288    let mut config = RunConfig::new();
289    config.hermit_version = Some(manifest.hermit.version.clone());
290    config.global = section(&manifest.inputs.global)?;
291    config.run = section(&manifest.inputs.run)?;
292    config.program = manifest.command.program.clone();
293    config.args = manifest.command.args.clone();
294    Ok(config)
295}

316/// Writes `config` to `path` as `--save-config` does, and reads the text back
317/// so that a document that would not load is an error now rather than when
318/// someone tries to reproduce the run.
319pub(crate) fn write(path: &Path, config: &RunConfig) -> Result<(), Error> {
320    let text = config.to_yaml()?;
321    let reread = parse(&text).context("the run config this hermit encoded does not parse back")?;
322    if reread != *config {
323        anyhow::bail!(
324            "the run config this hermit encoded parses back differently:\nwritten: \
325             {config:?}\nread back: {reread:?}"
326        );
327    }
328    std::fs::write(path, text)
329        .with_context(|| format!("cannot write --save-config {}", path.display()))
330}

332/// How an option takes its value, and so how a run config spells it.
333#[derive(Clone, Copy, Debug, PartialEq, Eq)]
334enum Shape {
335    /// `key: true` gives the flag; `false` omits it.
336    Flag,
337    /// `key: N` gives the flag N times.
338    Count,
339    /// `key: VALUE`, one value.
340    One,
341    /// `key: [VALUE, ...]`, one entry per occurrence.
342    Many,
343}
344
345/// One option a run config section may name.
346#[derive(Clone, Debug)]
347struct OptionSpec {
348    id: String,
349    long: String,
350    shape: Shape,
351}
352
353fn shape(arg: &Arg) -> Option<Shape> {
354    match arg.get_action() {
355        ArgAction::SetTrue | ArgAction::SetFalse => Some(Shape::Flag),
356        ArgAction::Count => Some(Shape::Count),
357        ArgAction::Set => Some(Shape::One),
358        ArgAction::Append => Some(Shape::Many),
359        _ => None,
360    }
361}

363/// The options of `command` that a run config can name: every option with
364/// a long name, in declaration order. Positional arguments are the
365/// document's `program` and `args`.
366fn option_specs(command: &Command) -> Vec<OptionSpec> {
367    command
368        .get_arguments()
369        .filter(|arg| !arg.is_positional())
370        .filter_map(|arg| {
371            Some(OptionSpec {
372                id: arg.get_id().as_str().to_owned(),
373                long: arg.get_long()?.to_owned(),
374                shape: shape(arg)?,
375            })
376        })
377        .collect()
378}

380/// Options that belong to one invocation rather than to the run it
381/// configures, so a run config neither holds nor accepts them.
382fn invocation_only(id: &str) -> Option<&'static str> {
383    match id {
384        CONFIG_ID => Some("a run config cannot load another run config"),
385        FROM_ID => Some("a run config cannot load a manifest"),
386        MANIFEST_ID => Some(
387            "whether to write a manifest is a choice of the invocation, not part of the run; \
388             pass --manifest on the command line",
389        ),
390        SAVE_CONFIG_ID => Some(
391            "where to save a config is a choice of the invocation, not part of the run; pass \
392             --save-config on the command line",
393        ),
394        _ => None,
395    }
396}

404/// The run config holding the options of a parsed `hermit run` invocation
405/// that were given on the command line or through an environment variable.
406/// `global_matches` and `command` are the whole `hermit` invocation;
407/// `run_matches` is its `run` subcommand.
408pub(crate) fn capture(
409    command: &Command,
410    global_matches: &ArgMatches,
411    run_matches: &ArgMatches,
412) -> Result<RunConfig, Error> {
413    let mut config = RunConfig::new();
414    config.global = capture_section(command, global_matches)?;
415    config.run = capture_section(run_command(command), run_matches)?;
416    let positional = |id: &str| -> Result<Vec<String>, Error> {
417        run_matches
418            .get_raw(id)
419            .into_iter()
420            .flatten()
421            .map(|value| utf8(value, id))
422            .collect()
423    };
424    config.program = positional(PROGRAM_ID)?.pop();
425    config.args = positional(ARGS_ID)?;
426    Ok(config)
427}

429/// The global options `matches` took from the command line, respelled as
430/// `--long[=VALUE]` tokens, for a command that reruns as `hermit [GLOBAL]
431/// run --from` (`hermit replay`). Options from the environment are left to
432/// the environment the rerun inherits.
433pub(crate) fn global_tokens(
434    command: &Command,
435    matches: &ArgMatches,
436) -> Result<Vec<OsString>, Error> {
437    let specs = option_specs(command);
438    let mut section = capture_section(command, matches)?;
439    section.retain(|long, _| {
440        specs.iter().any(|spec| {
441            spec.long == *long && matches.value_source(&spec.id) == Some(ValueSource::CommandLine)
442        })
443    });
444    file_tokens(Section::Global, &section, command, |_| false)
445}

447fn capture_section(
448    command: &Command,
449    matches: &ArgMatches,
450) -> Result<BTreeMap<String, Value>, Error> {
451    let mut section = BTreeMap::new();
452    for spec in option_specs(command) {
453        if invocation_only(&spec.id).is_some()
454            || !matches!(
455                matches.value_source(&spec.id),
456                Some(ValueSource::CommandLine | ValueSource::EnvVariable)
457            )
458        {
459            continue;
460        }
461        let raw = || -> Result<Vec<Value>, Error> {
462            matches
463                .get_raw(&spec.id)
464                .into_iter()
465                .flatten()
466                .map(|value| utf8(value, &spec.long).map(|value| scalar(&value)))
467                .collect()
468        };
469        let value = match spec.shape {
470            Shape::Flag => Value::Bool(true),
471            Shape::Count => Value::Number(u64::from(matches.get_count(&spec.id)).into()),
472            Shape::One => raw()?
473                .pop()
474                .with_context(|| format!("--{} was given without a value", spec.long))?,
475            Shape::Many => Value::Sequence(raw()?),
476        };
477        section.insert(spec.long, value);
478    }
479    Ok(section)
480}

491/// A command-line value as YAML: a canonical unsigned integer becomes a
492/// number, anything else a string, so the value loads back as the same text.
493fn scalar(text: &str) -> Value {
494    match text.parse::<u64>() {
495        Ok(number) if number.to_string() == text => Value::Number(number.into()),
496        _ => Value::String(text.to_owned()),
497    }

500/// Rewrites a `hermit [GLOBAL] run --config FILE [OPTIONS] [-- PROGRAM ARGS]`
501/// invocation into the command line it means: the file's options and guest,
502/// less every option the command line gives itself, followed by the
503/// command line's own options and guest. Returns `None`, leaving the
504/// invocation to the ordinary parser, when it is not a `hermit run` that
505/// names `--config`, or when it does not parse even leniently (an unknown
506/// option, `--help`), which that parser then reports.
507#[cfg(test)]
508pub(crate) fn expand_argv(
509    command: &Command,
510    argv: &[OsString],
511) -> Result<Option<Vec<OsString>>, Error> {
512    expand_argv_for_application(command, argv, None)
513}

515/// A separately linked executable owns its selected backend identity. Saved
516/// configs/manifests use the existing `global.backend` string to bind that
517/// identity: only the matching executable may consume it without a built-in
518/// `--backend` enum. Ordinary Hermit still parses the value normally and
519/// refuses an unknown backend instead of silently choosing ptrace.
520/// An external executable also refuses files with no saved backend identity;
521/// the ordinary executable retains its implicit builtin default.
522pub(super) fn expand_argv_for_application(
523    command: &Command,
524    argv: &[OsString],
525    application_backend: Option<&str>,
526) -> Result<Option<Vec<OsString>>, Error> {
527    expand_argv_with_selection(command, argv, application_backend)
528}

530/// The selected executable parses the complete option schema. Only its shared
531/// global selector is consumed here, using the same option names/arity as run
532/// config expansion. The original Unix argv used for exec is never modified.
533/// A matching explicit selector is consumed for this executable's parser; it
534/// cannot override a missing/different saved executable backend identity.
535pub(super) fn application_arguments(
536    command: &Command,
537    argv: &[OsString],
538    application: &str,
539) -> Result<(Vec<OsString>, bool), Error> {
540    super::backend_hook::validate_name(application)?;
541    let mut normalized = None;
542    if let Ok(mut prefix) = command_prefix(command, argv)
543        && let Some(backend) = prefix.backend()?
544    {
545        anyhow::ensure!(
546            backend == OsStr::new(application),
547            "this executable selects backend `{application}`, but the command line selects {}; \
548             use the executable for that backend instead",
549            backend.to_string_lossy(),
550        );
551        let (_, command_index) = prefix.command.as_ref().ok_or_else(|| {
552            anyhow::anyhow!("an executable-selected --backend requires a subcommand")
553        })?;
554        // Re-spell only the parsed global prefix. Raw guest operands, including
555        // non-UTF8 values and every byte after `--`, remain the original tail.
556        prefix.matches.remove_many::<OsString>("backend");
557        let mut parsed = vec![argv.first().cloned().unwrap_or_else(|| "hermit".into())];
558        parsed.extend(given_tokens(command, &prefix.matches));
559        parsed.extend_from_slice(&argv[*command_index..]);
560        normalized = Some(parsed);
561    }
562    let input = normalized.as_deref().unwrap_or(argv);
563    let expanded = expand_argv_with_selection(command, input, Some(application))?;
564    let config_applied = expanded.is_some();
565    Ok((expanded.unwrap_or_else(|| input.to_vec()), config_applied))
566}

568fn expand_argv_with_selection(
569    command: &Command,
570    argv: &[OsString],
571    application_backend: Option<&str>,
572) -> Result<Option<Vec<OsString>>, Error> {
573    let names = |long: &str| {
574        let with_value = format!("{long}=");
575        argv.iter()
576            .any(|arg| arg == long || arg.to_str().is_some_and(|arg| arg.starts_with(&with_value)))
577    };
578    if !names("--config") && !names("--from") {
579        return Ok(None);
580    }
581    let Ok(matches) = lenient_command(command).try_get_matches_from(argv) else {
582        return Ok(None);
583    };
584    let Some(run_matches) = matches.subcommand_matches(RUN) else {
585        return Ok(None);
586    };
587    let Some((path, mut config)) = input_config(run_matches, true)? else {
588        return Ok(None);
589    };
590    if let Some(application) = application_backend {
591        let backend = config.global.remove("backend").with_context(|| {
592            format!(
593                "{} has no saved backend identity, but this executable selects `{application}`; \
594                 use the executable for the saved backend instead",
595                path.display(),
596            )
597        })?;
598        anyhow::ensure!(
599            backend.as_str() == Some(application),
600            "{} selects backend {}, but this executable selects `{application}`; \
601             use the executable for the saved backend instead",
602            path.display(),
603            display_value(&backend),
604        );
605    }
606    let run = run_command(command);
607    let given =
608        |matches: &ArgMatches, id: &str| matches.value_source(id) == Some(ValueSource::CommandLine);
609    let in_section = |section: Section, entries, matches| {
610        file_tokens(section, entries, command, |id| given(matches, id))
611            .with_context(|| format!("cannot load run config {}", path.display()))
612    };
613
614    let mut expanded = vec![argv.first().cloned().unwrap_or_else(|| "hermit".into())];
615    expanded.extend(in_section(Section::Global, &config.global, &matches)?);
616    expanded.extend(given_tokens(command, &matches));
617    expanded.push(RUN.into());
618    expanded.extend(in_section(Section::Run, &config.run, run_matches)?);
619    expanded.extend(given_tokens(run, run_matches));
620    let guest: Vec<OsString> = if run_matches.contains_id(PROGRAM_ID) {
621        [PROGRAM_ID, ARGS_ID]
622            .into_iter()
623            .flat_map(|id| run_matches.get_many::<OsString>(id).into_iter().flatten())
624            .cloned()
625            .collect()
626    } else {
627        config
628            .program
629            .iter()
630            .chain(&config.args)
631            .map(OsString::from)
632            .collect()
633    };
634    if !guest.is_empty() {
635        expanded.push("--".into());
636        expanded.extend(guest);
637    }
638    Ok(Some(expanded))
639}

641/// Read just the explicitly located config/manifest. Dispatch selection uses
642/// the same resolver without emitting the ordinary rerun-directory note;
643/// complete expansion in the selected executable still owns that note.
644fn input_config(
645    run_matches: &ArgMatches,
646    note_directory: bool,
647) -> Result<Option<(PathBuf, RunConfig)>, Error> {
648    let given_paths = |id: &str| -> Vec<&OsString> {
649        run_matches
650            .get_many::<OsString>(id)
651            .into_iter()
652            .flatten()
653            .collect()
654    };
655    let (configs, manifests) = (given_paths(CONFIG_ID), given_paths(FROM_ID));
656    let input = match (configs.as_slice(), manifests.as_slice()) {
657        ([], []) => return Ok(None),
658        ([path], []) => (PathBuf::from(path), read(Path::new(path))?),
659        ([], [spec]) => {
660            use crate::app::manifest::Target;
661            let path = match crate::app::manifest::resolve(&crate::app::manifest::data_dir(), spec)?
662            {
663                Target::Manifest(path) => path,
664                Target::Recording { data_dir, id } => anyhow::bail!(
665                    "{} is a syscall recording, which `hermit run --from` does not rerun; \
666                     replay it with: hermit replay {}",
667                    spec.to_string_lossy(),
668                    data_dir.join(id.to_string()).display()
669                ),
670            };
671            let manifest = crate::app::manifest::read(&path)?;
672            if note_directory {
673                note_another_directory(&manifest);
674            }
675            (path.clone(), from_manifest_at(&manifest, &path)?)
676        }
677        ([_, ..], [_, ..]) => {
678            anyhow::bail!("--config and --from both name a run's options; give one of them")
679        }
680        (paths, []) | ([], paths) => anyhow::bail!(
681            "--{} was given {} times; give one",
682            if configs.is_empty() { "from" } else { "config" },
683            paths.len()
684        ),
685    };
686    Ok(Some(input))
687}

710/// Parse only the actual global prefix. Treat the subcommand and all of its
711/// raw operands as an external tail; no backend-specific option arity can
712/// affect which global --backend this parser owns.
713fn command_prefix(command: &Command, argv: &[OsString]) -> Result<CommandPrefix, Error> {
714    let matches = lenient_scope(command, false)
715        .allow_external_subcommands(true)
716        .external_subcommand_value_parser(ValueParser::os_string())
717        .try_get_matches_from(argv)?;
718    let selected = if let Some((name, tail)) = matches.subcommand() {
719        let count = tail.get_many::<OsString>("").into_iter().flatten().count();
720        let index = argv
721            .len()
722            .checked_sub(count + 1)
723            .ok_or_else(|| anyhow::anyhow!("shared subcommand has an invalid raw-argv boundary"))?;
724        anyhow::ensure!(
725            argv.get(index).is_some_and(|arg| arg == name),
726            "shared subcommand does not match its original raw-argv boundary"
727        );
728        Some((name.to_owned(), index))
729    } else {
730        None
731    };
732    Ok(CommandPrefix {
733        matches,
734        command: selected,
735    })
736}

739/// a verdict: the selected executable still parses its complete option schema.

746/// Select only from known shared option grammar and structured config fields.
747/// Unknown extension-option arity and an undelimited guest are deliberately
748/// unsupported wrapper forms: use the selected executable directly for them.
749/// Neither an unknown option value nor a guest operand is scanned for control
750/// flags, config paths or report destinations.
751pub(super) fn external_selection(
752    command: &Command,
753    argv: &[OsString],
754) -> Result<Option<ExternalSelection>, Error> {
755    use clap::ValueEnum;
756
757    let Ok(prefix) = command_prefix(command, argv) else {
758        return Ok(None);
759    };
760    let explicit = prefix
761        .backend()?
762        .map(|value| {
763            value.to_str().ok_or_else(|| {
764                anyhow::anyhow!("backend identity must be UTF8; raw guest operands need not be")
765            })
766        })
767        .transpose()?;
768    let builtin = |name| <hermit::Backend as ValueEnum>::from_str(name, false).is_ok();
769    if explicit.is_some_and(builtin) {
770        return Ok(None);
771    }
772    let Some((name, index)) = &prefix.command else {
773        return Ok(None);
774    };
775    if name != RUN {
776        anyhow::ensure!(
777            explicit.is_none(),
778            "external backend selection currently supports `run`; use the backend executable directly for other commands"
779        );
780        return Ok(None);
781    }
782    let options = match shared_options_before_guest(command, argv, *index) {
783        Ok(matches) => matches,
784        Err(_) if explicit.is_none() => return Ok(None),
785        Err(error) => return Err(error).context("external dispatch cannot prove shared option ownership; use the backend executable directly, or keep only known shared options before an explicit `--` guest boundary"),
786    };
787    let run = options
788        .subcommand_matches(RUN)
789        .ok_or_else(|| anyhow::anyhow!("shared selection lost its run subcommand"))?;
790    let saved = match input_config(run, false) {
791        Ok(saved) => saved,
792        // Until a config actually identifies an external backend, keep its
793        // ordinary parser/descriptor/error ordering. An explicit external
794        // selector owns the early refusal instead; it must never fall back.
795        Err(_) if explicit.is_none() => return Ok(None),
796        Err(error) => return Err(error),
797    };
798    let configured = saved
799        .as_ref()
800        .and_then(|(_, config)| config.global.get("backend"));
801    let selected = match (explicit, configured) {
802        (Some(name), _) => Some(name),
803        (None, Some(Value::String(name))) => Some(name.as_str()),
804        (None, Some(value)) => anyhow::bail!(
805            "`global.backend` must be a backend name, not {}",
806            display_value(value)
807        ),
808        (None, None) => None,
809    };
810    let Some(selected) = selected else {
811        return Ok(None);
812    };
813    if builtin(selected) {
814        return Ok(None);
815    }
816    super::backend_hook::validate_name(selected)?;
817    hermit::compatibility::ExecutableIdentity::backend(selected)?;
818    if saved.is_some() {
819        anyhow::ensure!(
820            configured.and_then(Value::as_str) == Some(selected),
821            "saved backend identity must exist and match executable-selected backend `{selected}`; use the executable for the saved backend instead"
822        );
823    }
824    let pending_verification =
825        pending_verification_path(run, saved.as_ref().map(|(_, config)| config))?;
826    Ok(Some(ExternalSelection {
827        backend: selected.to_owned(),
828        pending_verification,
829    }))
830}

832fn shared_options_before_guest(
833    command: &Command,
834    argv: &[OsString],
835    command_index: usize,
836) -> Result<ArgMatches, Error> {
837    let parser = || shared_selection_parser(command);
838    if let Ok(matches) = parser().try_get_matches_from(argv) {
839        return Ok(matches);
840    }
841    // A candidate boundary is accepted only when the entire known-option
842    // prefix parses WITHOUT positionals. `--` used as a known option's value
843    // therefore cannot be mistaken for the guest delimiter.
844    for index in command_index + 1..argv.len() {
845        if argv[index] == "--"
846            && let Ok(matches) = parser().try_get_matches_from(&argv[..index])
847        {
848            return Ok(matches);
849        }
850    }
851    anyhow::bail!("ambiguous extension option or undelimited guest arguments")
852}

855    lenient_scope(command, false).subcommand(lenient_scope(run_command(command), false))

858/// A failed dispatch still stamps a report whose ownership is provable before
859/// an unknown option/operand. Select the longest COMPLETE shared-option prefix,
860/// without positionals or an unknown-option catch-all. Nothing after that
861/// prefix is interpreted, even if it resembles --config/--verify-json.
862pub(super) fn external_refusal_pending(
863    command: &Command,
864    argv: &[OsString],
865) -> Result<Option<PathBuf>, Error> {
866    use clap::ValueEnum;
867    let Ok(prefix) = command_prefix(command, argv) else {
868        return Ok(None);
869    };
870    let explicit = prefix.backend()?.and_then(OsStr::to_str);
871    let builtin = |name| <hermit::Backend as ValueEnum>::from_str(name, false).is_ok();
872    if explicit.is_some_and(builtin) {
873        return Ok(None);
874    }
875    let Some((name, index)) = &prefix.command else {
876        return Ok(None);
877    };
878    if name != RUN {
879        return Ok(None);
880    }
881    let mut parsed = None;
882    for end in (*index + 1..=argv.len()).rev() {
883        if let Ok(matches) = shared_selection_parser(command).try_get_matches_from(&argv[..end]) {
884            parsed = Some(matches);
885            break;
886        }
887    }
888    let Some(parsed) = parsed else {
889        return Ok(None);
890    };
891    let run = parsed
892        .subcommand_matches(RUN)
893        .context("shared prefix lost its run command")?;
894    let saved = match input_config(run, false) {
895        Ok(saved) => saved,
896        // A missing/invalid file cannot grant a report path. If the explicit
897        // external selector is proven, the known CLI fields still can.
898        Err(_) if explicit.is_some() => None,
899        Err(_) => return Ok(None),
900    };
901    let configured = saved
902        .as_ref()
903        .and_then(|(_, config)| config.global.get("backend"))
904        .and_then(Value::as_str);
905    let Some(selected) = explicit.or(configured) else {
906        return Ok(None);
907    };
908    if builtin(selected) {
909        return Ok(None);
910    }
911    pending_verification_path(run, saved.as_ref().map(|(_, config)| config))
912}

914/// Select only the two common failure-report fields, with the existing CLI >
915/// file precedence and scalar rules. Unknown backend config keys are left for
916/// the selected executable's COMPLETE parser; they cannot grant report-path or
917/// verification authority. No report is written by selection itself.
918fn pending_verification_path(
919    run: &ArgMatches,
920    config: Option<&RunConfig>,
921) -> Result<Option<PathBuf>, Error> {
922    let given = |id| run.value_source(id) == Some(ValueSource::CommandLine);
923    let verify = if given("verify") {
924        run.get_count("verify") != 0
925    } else {
926        match config.and_then(|config| config.run.get("verify")) {
927            None | Some(Value::Bool(false)) => false,
928            Some(Value::Bool(true)) => true,
929            Some(value) => anyhow::bail!(
930                "`run.verify` must be true or false, not {}",
931                display_value(value)
932            ),
933        }
934    };
935    if !verify {
936        return Ok(None);
937    }
938    if given("verify_json") {
939        let paths: Vec<_> = run
940            .get_many::<OsString>("verify_json")
941            .into_iter()
942            .flatten()
943            .collect();
944        anyhow::ensure!(
945            paths.len() == 1,
946            "--verify-json must name exactly one report destination"
947        );
948        return Ok(Some(PathBuf::from(paths[0])));
949    }
950    config
951        .and_then(|config| config.run.get("verify-json"))
952        .map(|value| {
953            scalar_text(value).map(PathBuf::from).ok_or_else(|| {
954                anyhow::anyhow!(
955                    "`run.verify-json` must be a single path value, not {}",
956                    display_value(value)
957                )
958            })
959        })
960        .transpose()
961}

963/// A copy of the `hermit` command that accepts every option of the global
964/// level and of `run` with any value and in any combination, keeping only
965/// the names and value arity. It finds what the command line gave without
966/// applying the rules that the file's options may be what satisfies, such as
967/// `--verify-strict` requiring `--verify`.
968fn lenient_command(command: &Command) -> Command {
969    lenient_scope(command, true)
970        .allow_external_subcommands(true)
971        .subcommand(lenient_scope(run_command(command), true))
972}

974fn lenient_scope(source: &Command, include_positionals: bool) -> Command {
975    let positionals = source
976        .get_arguments()
977        .filter(|arg| arg.is_positional())
978        .count();
979    let mut index = 0;
980    let mut copy = Command::new(source.get_name().to_owned())
981        .disable_help_flag(true)
982        .disable_version_flag(true)
983        .trailing_var_arg(include_positionals && source.is_trailing_var_arg_set());
984    for arg in source.get_arguments() {
985        let mut lenient = Arg::new(arg.get_id().as_str().to_owned());
986        if arg.is_positional() {
987            if !include_positionals {
988                continue;
989            }
990            index += 1;
991            lenient = lenient.index(index).value_parser(ValueParser::os_string());
992            lenient = if index == positionals {
993                lenient
994                    .action(ArgAction::Append)
995                    .num_args(0..)
996                    .allow_hyphen_values(true)
997                    .trailing_var_arg(source.is_trailing_var_arg_set())
998            } else {
999                lenient.action(ArgAction::Set)
1000            };
1001            copy = copy.arg(lenient);
1002            continue;
1003        }
1004        if let Some(long) = arg.get_long() {
1005            lenient = lenient.long(long.to_owned());
1006        }
1007        if let Some(short) = arg.get_short() {
1008            lenient = lenient.short(short);
1009        }
1010        if let Some(aliases) = arg.get_all_aliases() {
1011            lenient = lenient.aliases(aliases.into_iter().map(str::to_owned));
1012        }
1013        if let Some(shorts) = arg.get_all_short_aliases() {
1014            lenient = lenient.short_aliases(shorts);
1015        }
1016        // A global option may follow the subcommand, as on a typed command line.
1017        lenient = lenient.global(arg.is_global_set());
1018        lenient = match shape(arg) {
1019            Some(Shape::Flag | Shape::Count) => lenient.action(ArgAction::Count),
1020            _ => lenient
1021                .action(ArgAction::Append)
1022                .value_parser(ValueParser::os_string())
1023                .num_args(1)
1024                .allow_hyphen_values(arg.is_allow_hyphen_values_set())
1025                .require_equals(arg.is_require_equals_set()),
1026        };
1027        copy = copy.arg(lenient);
1028    }
1029    copy
1030}

1032/// The command-line options of a lenient parse, respelled as `--long[=VALUE]`
1033/// tokens, `--config` left out.
1034fn given_tokens(command: &Command, matches: &ArgMatches) -> Vec<OsString> {
1035    let mut tokens = Vec::new();
1036    for spec in option_specs(command) {
1037        if spec.id == CONFIG_ID
1038            || spec.id == FROM_ID
1039            || matches.value_source(&spec.id) != Some(ValueSource::CommandLine)
1040        {
1041            continue;
1042        }
1043        match spec.shape {
1044            Shape::Flag | Shape::Count => {
1045                for _ in 0..matches.get_count(&spec.id) {
1046                    tokens.push(format!("--{}", spec.long).into());
1047                }
1048            }
1049            Shape::One | Shape::Many => {
1050                for value in matches.get_many::<OsString>(&spec.id).into_iter().flatten() {
1051                    tokens.push(valued_token(&spec.long, value));
1052                }
1053            }
1054        }
1055    }
1056    tokens
1057}
1058
1059fn valued_token(long: &str, value: &OsStr) -> OsString {
1060    let mut token = OsString::from(format!("--{long}="));
1061    token.push(value);
1062    token
1063}

1065/// The two option sections of a run config.
1066#[derive(Clone, Copy, Debug, PartialEq, Eq)]
1067enum Section {
1068    /// `global:`, the options before the subcommand.
1069    Global,
1070    /// `run:`, the options of `hermit run`.
1071    Run,
1072}
1073
1074impl Section {
1075    fn name(self) -> &'static str {
1076        match self {
1077            Section::Global => "global",
1078            Section::Run => RUN,
1079        }
1080    }
1081
1082    /// The command whose options this section names, given the whole
1083    /// `hermit` command.
1084    fn command(self, hermit: &Command) -> &Command {
1085        match self {
1086            Section::Global => hermit,
1087            Section::Run => run_command(hermit),
1088        }
1089    }
1090
1091    fn other(self) -> Section {
1092        match self {
1093            Section::Global => Section::Run,
1094            Section::Run => Section::Global,

1096    }

1098    fn help(self) -> &'static str {
1099        match self {
1100            Section::Global => "`hermit --help`",

1106/// The command-line tokens for one section of a run config, skipping the
1107/// options `given` reports the command line gives itself. `hermit` is the
1108/// whole `hermit` command.
1109fn file_tokens(
1110    section: Section,
1111    entries: &BTreeMap<String, Value>,
1112    hermit: &Command,
1113    given: impl Fn(&str) -> bool,
1114) -> Result<Vec<OsString>, Error> {
1115    let specs = option_specs(section.command(hermit));
1116    let mut tokens = Vec::new();
1117    for (key, value) in entries {
1118        let Some(spec) = specs.iter().find(|spec| spec.long == *key) else {
1119            anyhow::bail!("{}", unknown_key(section, key, hermit));
1120        };
1121        let section = section.name();
1122        if let Some(reason) = invocation_only(&spec.id) {
1123            anyhow::bail!("`{section}.{key}` is not allowed in a run config: {reason}");
1124        }
1125        if given(&spec.id) {
1126            continue;
1127        }
1128        let wrong = |expected: &str| {
1129            anyhow::anyhow!(
1130                "`{section}.{key}` must be {expected}, not {}",
1131                display_value(value)
1132            )
1133        };
1134        match (spec.shape, value) {
1135            (Shape::Flag, Value::Bool(true)) => tokens.push(format!("--{key}").into()),
1136            (Shape::Flag, Value::Bool(false)) => {}
1137            (Shape::Flag, _) => {
1138                return Err(wrong(
1139                    "`true` (a flag; write `true` to give it, or leave the key out)",
1140                ));
1141            }
1142            (Shape::Count, Value::Number(number)) => {
1143                let count = number
1144                    .as_u64()
1145                    .ok_or_else(|| wrong("a count (a non-negative integer)"))?;
1146                for _ in 0..count {
1147                    tokens.push(format!("--{key}").into());
1148                }
1149            }
1150            (Shape::Count, _) => return Err(wrong("a count (a non-negative integer)")),
1151            (Shape::One, Value::Sequence(_)) => {
1152                return Err(wrong(&format!(
1153                    "one value (--{key} is not repeatable, so it takes no list)"
1154                )));
1155            }
1156            (Shape::One, value) => {
1157                let text = scalar_text(value).ok_or_else(|| wrong("a single value"))?;
1158                tokens.push(valued_token(key, OsStr::new(&text)));
1159            }
1160            (Shape::Many, Value::Sequence(values)) => {
1161                for value in values {
1162                    let text =
1163                        scalar_text(value).ok_or_else(|| wrong("a list of single values"))?;
1164                    tokens.push(valued_token(key, OsStr::new(&text)));
1165                }
1166            }
1167            (Shape::Many, value) => {
1168                let text = scalar_text(value).ok_or_else(|| wrong("a list of single values"))?;
1169                tokens.push(valued_token(key, OsStr::new(&text)));
1170            }
1171        }
1172    }
1173    Ok(tokens)
1174}

1176/// The command-line text of a YAML scalar, or `None` for a null, list or
1177/// mapping.
1178fn scalar_text(value: &Value) -> Option<String> {
1179    match value {
1180        Value::String(text) => Some(text.clone()),
1181        Value::Number(number) => Some(number.to_string()),
1182        Value::Bool(flag) => Some(flag.to_string()),
1183        Value::Null | Value::Sequence(_) | Value::Mapping(_) | Value::Tagged(_) => None,
1184    }
1185}

1187/// Why `key` names no option of `section`, with the spelling that would.
1188/// `hermit` is the whole `hermit` command.
1189fn unknown_key(section: Section, key: &str, hermit: &Command) -> String {
1190    let command = section.command(hermit);
1191    let names = |section: Section, key: &str| {
1192        option_specs(section.command(hermit))
1193            .iter()
1194            .any(|spec| spec.long == key)
1195    };
1196    let by_alias = command.get_arguments().find(|arg| {
1197        arg.get_all_aliases()
1198            .is_some_and(|aliases| aliases.contains(&key))
1199    });
1200    let hyphenated = key.replace('_', "-");
1201    let other = section.other();
1202    let suggestion = if let Some(long) = by_alias.and_then(Arg::get_long) {
1203        format!("`{key}` is an alias; write the option's name, `{long}`")
1204    } else if names(section, &hyphenated) {
1205        format!("keys are long option names, with hyphens: write `{hyphenated}`")
1206    } else if names(other, key) || names(other, &hyphenated) {
1207        format!(
1208            "it is an option of the `{}` section; move it there",
1209            other.name()

1583    fn parse_at(argv: &[OsString], seconds: u64) -> (crate::app::Args, ArgMatches) {
1584        let matches = crate::app::Args::command()
1585            .try_get_matches_from(argv)
1586            .unwrap_or_else(|error| panic!("{argv:?} should parse: {error}"));
1587        let args = crate::app::args_from_matches_with_clock(&matches, || {
1588            UNIX_EPOCH + Duration::from_secs(seconds)
1589        })
1590        .unwrap_or_else(|error| panic!("{argv:?} should parse: {error}"));
1591        (args, matches)
1592    }

1613    /// Parses `argv`, saves it as `--save-config` would, loads the file
1614    /// through `--config` at a later clock, and returns both parses and the
1615    /// saved document.
1616    fn save_and_load(argv: &[OsString]) -> (Parsed, Parsed, RunConfig) {
1617        let (mut args, matches) = parse_at(argv, SAVING_CLOCK);
1618        let config = capture(
1619            &crate::app::Args::command(),
1620            &matches,
1621            matches.subcommand_matches(RUN).unwrap(),
1622        )
1623        .unwrap();
1624        let crate::app::Subcommand::Run(run) = &mut args.command else {
1625            panic!("{argv:?} is not a `run` command");
1626        };
1627        run.set_config_capture(config);
1628        let saved = run.saved_run_config().unwrap();
1629        let directory = tempfile::tempdir().unwrap();
1630        let path = directory.path().join("run.yaml");
1631        write(&path, &saved).unwrap();
1632        assert_eq!(read(&path).unwrap(), saved);
1633        let (reloaded, _) = parse_at(&load(&path, &[]), LOADING_CLOCK);
1634        (parsed(&args), parsed(&reloaded), saved)
1635    }
