9//! Exec-owned VM teardown helper. No code here runs in the container's
10//! raw-clone child.
11//!
12//! The helper is this same binary re-executed in the private
13//! `__hermit-vm-teardown-reap` role, spawned by [`ensure_installed`]
14//! before the first KVM container clone (the shape the GDB watch
15//! helper uses). It holds one end of a Unix socket; the other end is
16//! installed in the library as the process's VM teardown sink
17//! (`hermit::vm_teardown`), inherited by the container child at the
18//! clone. Each dropped KVM backend forwards duplicates of its vCPU
19//! and VM descriptors there, and this process closes them — vCPU
20//! first — on a small worker pool, acknowledging each bundle. The
21//! kernel VM destroy (an RCU grace-period wait in `close`) therefore
22//! runs here, not on a guest exit path or in the exiting CLI process.
23//!
24//! Lifetime: the helper serves until the socket reaches EOF, which
25//! needs every sender copy closed — the parent's installed sink and
26//! each container child's inherited copy — then drains its queue and
27//! exits. Past the run's own exit it therefore lives at most for one
28//! drain of what was already adopted (concurrent destroys overlap).
29//! Dying earlier is safe at every point: whatever descriptors it
30//! holds close in its own exit path, so those VMs are still
31//! destroyed, and later adoptions in the run see the dead socket and
32//! fall back to synchronous destroys. It deliberately keeps the
33//! run's process group and default signal dispositions (SIGPIPE
34//! aside): a terminal interrupt, a supervisor's group or cgroup
35//! kill, ends it like any other member of the run, with the same
36//! kernel-close fallback.

59pub(super) const ENTRY: &str = "__hermit-vm-teardown-reap";

61/// Concurrent destroys overlap their RCU grace-period waits. The pool
62/// must at least match the concurrency it replaces: without a sink,
63/// every guest child destroys its VM on its own host thread, so a
64/// fork-heavy run already has as many destroys in flight as it has
65/// unjoined children (measured: a 4-worker pool at a 16-bundle bound
66/// raised the fork_loop slope from 6.5 to 15.6 ms/fork inside one CPU
67/// quota, because the backlog then serialized at adoption instead).
68/// Eight concurrent closes measured ~3.6x the throughput of serial
69/// ones in the Phase-0 probe.
70const WORKERS: usize = 16;
71
72fn run(control_fd: RawFd) -> Result<(), Error> {
73    anyhow::ensure!(control_fd >= 3, "invalid VM teardown helper descriptor");
74    // SAFETY: the private exec entry transfers this inherited
75    // descriptor exactly once.
76    let stream = unsafe { UnixStream::from_raw_fd(control_fd) };
77    stream
78        .peer_addr()
79        .context("VM teardown helper descriptor is not a Unix socket")?;
80    // SAFETY: the descriptor is owned above; fcntl touches no memory.
81    if unsafe { libc::fcntl(stream.as_raw_fd(), libc::F_SETFD, libc::FD_CLOEXEC) } < 0 {
82        return Err(io::Error::last_os_error()).context("protect VM teardown helper descriptor");
83    }
84    super::retire_inherited_descriptors(&[stream.as_fd()])
85        .context("retire unsolicited VM teardown helper descriptors")?;
86    // Name the process for leftover audits: a bounded helper with this
87    // comm may outlive its run by one drain, and no other process.
88    // SAFETY: the name is a NUL-terminated literal within the kernel's
89    // 15-byte comm limit; prctl only reads it.
90    unsafe {
91        libc::prctl(
92            libc::PR_SET_NAME,
93            c"hermit-vm-reap".as_ptr() as libc::c_ulong,
94            0,
95            0,
96            0,
97        );
98        // Acknowledgements to a sender that died mid-run must report
99        // EPIPE, not kill the drain with SIGPIPE.
100        libc::signal(libc::SIGPIPE, libc::SIG_IGN);
101    }
102
103    let ack = Arc::new(Mutex::new(
104        stream.try_clone().context("clone VM teardown socket")?,
105    ));
106    let (sender, receiver) = mpsc::channel::<(OwnedFd, OwnedFd)>();
107    let receiver = Arc::new(Mutex::new(receiver));
108    let mut workers = Vec::new();
109    for _ in 0..WORKERS {
110        let receiver = Arc::clone(&receiver);
111        let ack = Arc::clone(&ack);
112        workers.push(
113            std::thread::Builder::new()
114                .name("vm-teardown-reap".into())
115                .spawn(move || {
116                    loop {
117                        let bundle = {
118                            let receiver = receiver.lock().unwrap_or_else(|p| p.into_inner());
119                            receiver.recv()
120                        };
121                        let Ok((vcpu, vm)) = bundle else { return };
122                        drop(vcpu);
123                        drop(vm);
124                        let mut ack = ack.lock().unwrap_or_else(|p| p.into_inner());
125                        if ack.write_all(&[ACK_BYTE]).is_err() {
126                            return;
127                        }
128                    }
129                })
130                .context("spawn VM teardown worker")?,
131        );
132    }
133    {
134        let mut ack = ack.lock().unwrap_or_else(|p| p.into_inner());
135        let _ = ack.write_all(&[READY_BYTE]);
136    }
137    let mut result = Ok(());
138    loop {
139        match recv_vm_bundle(stream.as_raw_fd()) {
140            Ok(Some(bundle)) => {
141                if sender.send(bundle).is_err() {
142                    result = Err(Error::msg("VM teardown workers stopped early"));
143                    break;
144                }
145            }
146            Ok(None) => break,
147            Err(error) => {
148                result = Err(error).context("receive VM teardown bundle");
149                break;
150            }
151        }
152    }
153    // EOF (or a fatal receive error): closing the channel lets the
154    // workers drain what was already adopted, then stop.
155    drop(sender);
156    for worker in workers {
157        let _ = worker.join();
158    }
159    result
160}

162/// Called before normal CLI parsing or logging. Returning None is the
163/// normal CLI path; a recognized but malformed private entry is an
164/// explicit error.
165pub(super) fn maybe_run() -> Option<Result<(), Error>> {
166    let mut args = std::env::args_os().skip(1);
167    if args.next().as_deref() != Some(std::ffi::OsStr::new(ENTRY)) {
168        return None;
169    }
170    Some((|| {
171        let control: RawFd = args
172            .next()
173            .context("missing VM teardown helper descriptor")?
174            .into_string()
175            .map_err(|_| anyhow::anyhow!("non-text VM teardown helper descriptor"))?
176            .parse()
177            .context("invalid VM teardown helper descriptor")?;
178        anyhow::ensure!(
179            args.next().is_none(),
180            "unexpected VM teardown helper arguments"
181        );
182        run(control)
183    })())
184}

189pub(super) fn helper_command(control: RawFd) -> Command {
190    // Re-exec this exact image even if its on-disk path was replaced or
191    // unlinked; resolving current_exe() would reopen whichever image
192    // now owns that name.
193    let mut command = Command::new("/proc/self/exe");
194    #[cfg(not(test))]
195    command.arg(ENTRY).arg(control.to_string());
196    // A libtest executable does not enter the production main. Re-exec
197    // an explicit test role which runs the identical helper.
198    #[cfg(test)]
199    command
200        .arg("--exact")
201        .arg("app::vm_teardown_helper::tests::exec_helper_entry")
202        .arg("--nocapture")
203        .arg("--test-threads=1")
204        .env(TEST_FD, control.to_string());
205    // The helper must not hold the run's stdio open: an inherited
206    // stdout/stderr pipe would delay its EOF until the helper drains,
207    // for exactly the readers this teardown exists to unblock.
208    command
209        .stdin(Stdio::null())
210        .stdout(Stdio::null())
211        .stderr(Stdio::null());
212    command
213}

215/// Spawns the helper and returns the sender end of its socket plus the
216/// child handle. The caller decides what owns each; dropping the
217/// child handle detaches it (it is reaped by init or a subreaper
218/// after this process exits).
219fn spawn_helper() -> Result<(std::process::Child, UnixStream), Error> {
220    let (sender, helper_end) = UnixStream::pair().context("create VM teardown helper socket")?;
221    let control = helper_end.as_raw_fd();
222    let mut launcher = helper_command(control);
223    // The only pre_exec work is async-signal-safe fcntl on the one
224    // owned descriptor: it is not inheritable in this process, only
225    // in the exec'd helper.
226    unsafe {
227        launcher.pre_exec(move || {
228            if libc::fcntl(control, libc::F_SETFD, 0) < 0 {
229                return Err(io::Error::last_os_error());
230            }
231            Ok(())
232        });
233    }
234    let child = launcher.spawn().context("start VM teardown helper")?;
235    drop(helper_end);
236    Ok((child, sender))
237}

239/// Installs the process's VM teardown sink, spawning the helper on
240/// first use. Call before the first container clone of a KVM launch;
241/// later calls do nothing. A helper that cannot start is a warning,
242/// never a failed run: with no sink installed, backends destroy
243/// their VMs synchronously at drop, as they did before.
244pub(super) fn ensure_installed() {
245    if hermit::vm_teardown::vm_teardown_sink_installed() {
246        return;
247    }
248    match spawn_helper() {
249        Ok((child, sender)) => {
250            if hermit::vm_teardown::install_vm_teardown_sink(sender) {
251                // Deliberately detached: the helper outlives this
252                // process by at most one drain (see the module docs)
253                // and is then reaped by init or a subreaper. Waiting
254                // for it here would put the teardown back on the exit
255                // path this exists to remove.
256                drop(child);
257            } else {
258                // Another thread installed first; this helper's
259                // socket was dropped by the refused install, so it
260                // sees EOF and exits. Reap it rather than detach a
261                // process nobody adopted.
262                let mut child = child;
263                let _ = child.wait();
264            }
265        }
266        Err(error) => {
267            crate::app::tracing::diagnostic_eprintln!(
268                "hermit: WARNING: the VM teardown helper did not start ({error:#}); \
269                 KVM VM destruction stays on the dropping threads"
270            );
271        }
272    }
273}

315    fn pipe() -> (OwnedFd, OwnedFd) {
316        let mut fds = [0; 2];
317        // SAFETY: `fds` is a valid two-element array for pipe2 to fill.
318        assert_eq!(unsafe { libc::pipe2(fds.as_mut_ptr(), libc::O_CLOEXEC) }, 0);
319        // SAFETY: pipe2 just created both descriptors for this call.
320        unsafe { (OwnedFd::from_raw_fd(fds[0]), OwnedFd::from_raw_fd(fds[1])) }
321    }
322
323    fn is_eof(fd: &OwnedFd) -> bool {
324        let mut poll = libc::pollfd {
325            fd: fd.as_raw_fd(),
326            events: libc::POLLHUP,
327            revents: 0,
328        };
329        // SAFETY: `poll` is one valid pollfd for the duration of the call.
330        let ready = unsafe { libc::poll(&mut poll, 1, 0) };
331        ready == 1 && poll.revents & libc::POLLHUP != 0
332    }

334    fn read_byte(stream: &mut UnixStream, deadline: Instant) -> u8 {
335        stream
336            .set_read_timeout(Some(
337                deadline
338                    .saturating_duration_since(Instant::now())
339                    .max(Duration::from_millis(1)),
340            ))
341            .unwrap();
342        let mut byte = [0u8; 1];
343        stream.read_exact(&mut byte).expect("read helper byte");
344        byte[0]
345    }

347    /// The exec'd helper closes forwarded descriptors, acknowledges,
348    /// and exits 0 once the sender closes (EOF after drain).
349    #[test]
350    fn helper_process_closes_bundles_and_exits_at_eof() {
351        let (mut child, mut sender) = spawn_helper().expect("spawn helper");
352        let deadline = Instant::now() + Duration::from_secs(30);
353        assert_eq!(read_byte(&mut sender, deadline), READY_BYTE);
354        let (read_a, write_a) = pipe();
355        let (read_b, write_b) = pipe();
356        send_vm_bundle(&sender, write_a.as_raw_fd(), write_b.as_raw_fd()).unwrap();
357        drop(write_a);
358        drop(write_b);
359        assert_eq!(read_byte(&mut sender, deadline), ACK_BYTE);
360        assert!(is_eof(&read_a) && is_eof(&read_b));
361        drop(sender);
362        let status = child.wait().expect("reap helper");
363        assert!(status.success(), "helper exit: {status}");
364    }
