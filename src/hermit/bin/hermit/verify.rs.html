60/// The exact native dimensions used by both live adapters and retained-run
61/// comparison. The full validated snapshot stays with the physical adapter.
62pub(super) trait NativeStatistics: std::fmt::Debug {
63    fn intercepted_syscalls(&self) -> u64;
64    fn rewritten_syscalls(&self) -> u64;
65    fn stdin_reads(&self) -> u64;
66    fn memory_hash_fold(&self) -> u64;
67}
68
69pub(super) fn same_native_observable_behavior(
70    first: &impl NativeStatistics,
71    second: &impl NativeStatistics,
72) -> bool {
73    // The counted-branch clock is a separate exact dimension, not an exemption.
74    first.intercepted_syscalls() == second.intercepted_syscalls()
75        && first.rewritten_syscalls() == second.rewritten_syscalls()
76        && first.stdin_reads() == second.stdin_reads()
77        && first.memory_hash_fold() == second.memory_hash_fold()
78}

80pub(super) fn native_branch_clock_mismatch(label: &str, first: u64, second: u64) -> Option<String> {
81    (first != second).then(|| {
82        format!(
83            "{label} verification failed: counted-branch clocks differed between runs ({first} != {second}); \
84             the clock is a deterministic function of the executed instruction stream"
85        )
86    })
87}

89pub(super) fn native_initialization_record_mismatch(
90    label: &str,
91    first: usize,
92    second: usize,
93) -> Option<String> {
94    (first != second).then(|| {
95        format!(
96            "{label} verification failed: authenticated process-image initialization counts differed between runs ({first} != {second})"
97        )
98    })
99}

101#[derive(Debug)]
102pub(super) struct NativeSummaryComparison {
103    failure: Option<String>,
104}
105
106impl NativeSummaryComparison {
107    #[cfg(test)]
108    pub(super) fn compare(
109        first_stats: &impl NativeStatistics,
110        second_stats: &impl NativeStatistics,
111        first_initialization_records: usize,
112        second_initialization_records: usize,
113    ) -> Self {
114        Self::compare_named(
115            "DBT",
116            first_stats,
117            second_stats,
118            first_initialization_records,
119            second_initialization_records,
120        )
121    }
122
123    fn compare_named(
124        label: &str,
125        first_stats: &impl NativeStatistics,
126        second_stats: &impl NativeStatistics,
127        first_initialization_records: usize,
128        second_initialization_records: usize,
129    ) -> Self {
130        let mut failures = Vec::new();
131        if !same_native_observable_behavior(first_stats, second_stats) {
132            failures.push(format!(
133                "{label} verification failed: typed native Detcore statistics differed \
134                 ({first_stats:?} != {second_stats:?})"
135            ));
136        }
137        if let Some(failure) = native_initialization_record_mismatch(
138            label,
139            first_initialization_records,
140            second_initialization_records,
141        ) {
142            failures.push(failure);
143        }
144        Self {
145            failure: (!failures.is_empty()).then(|| failures.join("; ")),
146        }
147    }
148
149    pub(super) fn requires_log_retention(&self) -> bool {
150        self.failure.is_some()
151    }
152
153    pub(super) fn apply(self, outcome: &mut VerificationOutcome) -> Option<String> {
154        if self.failure.is_some() {
155            outcome.verdict = Verdict::Diverged;
156            outcome.no_result_reason = None;
157        }
158        self.failure
159    }
160}

162pub(super) fn record_native_branch_clock_comparison(
163    label: &str,
164    outcome: &mut VerificationOutcome,
165    comparison: DbtCountedBranchComparison,
166) -> Option<String> {
167    match native_branch_clock_mismatch(label, comparison.left, comparison.right) {
168        Some(message) => {
169            outcome.dbt_counted_branches = Some(comparison);
170            outcome.verdict = Verdict::Diverged;
171            outcome.no_result_reason = None;
172            Some(message)
173        }
174        None if outcome.verdict == Verdict::NoResult => {
175            // Equal clocks never promote a refused common comparison.
176            outcome.dbt_counted_branches = None;
177            None
178        }
179        None => {
180            outcome.dbt_counted_branches = Some(comparison);
181            None
182        }
183    }
184}

186/// Core-owned exact evidence policy, applied around the sole common comparator.
187pub(super) struct NativeEvidenceComparison {
188    label: String,
189    branches: DbtCountedBranchComparison,
190    summary: NativeSummaryComparison,
191}
192
193impl NativeEvidenceComparison {
194    pub(super) fn new(
195        backend: &str,
196        first: &super::backend_hook::NativeFacts,
197        second: &super::backend_hook::NativeFacts,
198    ) -> Self {
199        // Preserve the existing DBT diagnostics and wire field exactly. A new
200        // executable uses its own validated identity, never a fabricated enum.
201        let label = if backend == "dbt" { "DBT" } else { backend }.to_owned();
202        Self {
203            branches: DbtCountedBranchComparison {
204                left: first.counted_branches,
205                right: second.counted_branches,
206            },
207            summary: NativeSummaryComparison::compare_named(
208                &label,
209                first,
210                second,
211                first.initialization_records,
212                second.initialization_records,
213            ),
214            label,
215        }
216    }
217
218    pub(super) fn configure(&self, options: &mut ComparisonOptions) {
219        options.strictness = LogCompareStrictness::Canonical;
220        let mismatch = !self.branches.matched() || self.summary.requires_log_retention();
221        options.keep_logs |= mismatch;
222        options.match_overridden |= mismatch;
223    }
224
225    pub(super) fn apply(self, outcome: &mut VerificationOutcome) -> Vec<String> {
226        let summary = self.summary.apply(outcome);
227        let branches = record_native_branch_clock_comparison(&self.label, outcome, self.branches);
228        branches.into_iter().chain(summary).collect()
229    }
230}

276pub(crate) struct ComparisonOptions {
277    /// Controls only how much diff *output* is printed (a larger syscall-history
278    /// window), NOT the comparison semantics. Comparison strictness is carried
279    /// separately in [`Self::strictness`] so a quiet run can still be
280    /// bitwise-strict — the two knobs were historically conflated behind a single
281    /// `verbose` flag, which made the only bitwise comparison also the loudest.
282    pub verbose: bool,
283    /// How strictly the internal event stream is compared. This is the
284    /// condition the verdict rests on, and is recorded verbatim in the resulting
285    /// [`VerificationOutcome`] so a consumer can tell a stripped match from a
286    /// bitwise one.
287    pub strictness: LogCompareStrictness,
288    pub compare_logs: bool,
289    /// Compare DEBUG/TRACE diagnostics in addition to the canonical INFO
290    /// envelope. This is reserved for the explicit `--verify-verbose` diagnostic
291    /// mode; an ordinary `--verify-strict` verdict must not depend on diagnostic
292    /// events merely because the caller requested that they be captured.
293    pub diagnostic_full_trace: bool,
294    /// Whether the compared records contain the CONTENT of syscall output
295    /// buffers, i.e. whether the run hashed them. That is ON BY DEFAULT; it is
296    /// false only when the caller passed `--no-detlog-io-buffers`.
297    ///
298    /// This is not a comparator setting; it describes what the records being
299    /// compared can possibly show. See [`ComparisonSpec::compare_io_buffers`].
300    pub compare_io_buffers: bool,
301    /// Keep captured logs at their selected paths after comparison.
302    ///
303    /// After a match, only the first side's log is kept, as the golden log,
304    /// and the second side's log, which compared equal to it under the
305    /// selected policy, is deleted. After a divergence, a
306    /// no-result, an error, or a match that did not compare the logs, both
307    /// logs are kept, because they are the evidence of what differed or of
308    /// what was never checked.
309    pub keep_logs: bool,
310    /// The caller has already observed a result that replaces a comparator
311    /// match: a skid overshoot, which is an infrastructure error, or a DBT
312    /// counted-branch or statistics divergence. A match is then not the
313    /// terminal verdict, so the logs are retained as for a divergence, and
314    /// [`Self::keep_logs`] does not delete the second side's log.
315    pub match_overridden: bool,
316    /// Where an implicitly retained failed comparison is stored and bounded.
317    /// This is absent when the caller explicitly requested `--keep-logs`; an
318    /// explicit evidence directory remains entirely caller-owned.
319    pub failed_log_retention: Option<FailedVerifyLogRetention>,
320    /// The logs belong to the caller (two `hermit run --verify-run`
321    /// directories): the comparison neither moves, deletes nor announces
322    /// them, whatever it finds, and [`Self::keep_logs`] and
323    /// [`Self::failed_log_retention`] do not apply. The caller hands the
324    /// comparison links to them, so a log the comparison drops on an early
325    /// error is only the link.
326    pub caller_owns_logs: bool,
327    /// Typed, versioned record envelope applied before selecting messages.
328    /// Its policy identity is serialized beside the verdict.
329    pub record_envelope: RecordEnvelope,
330    /// Whether the Detcore configuration under which BOTH sides ran virtualized
331    /// time. See [`ComparisonSpec::virtualize_time`] for what it changes about
332    /// the meaning of a match.
333    pub virtualize_time: bool,
334}

360/// The exact comparison that produced a [`Verdict`], carried beside it so a bare
361/// "verified" can always say *which* comparison certified it.
362///
363/// The high-level [`Self::strictness`] and the concrete flags it expands to are
364/// both recorded: a JSON consumer keying a bitwise-parity ratchet on the verdict
365/// can require `strip_lines == false`, `full_trace == true`, and an INFO-or-
366/// stronger [`Self::log_scope`] directly, rather than having to know how a
367/// strictness label maps onto the diff engine.
368#[derive(Debug, Clone, Copy, PartialEq, Eq)]
369pub struct ComparisonSpec {
370    /// The strictness label the comparison ran under.
371    pub strictness: LogCompareStrictness,
372    /// Human-readable name for the exact comparison policy. Derived in the same
373    /// exhaustive match as the diff flags so evidence cannot name a different
374    /// policy from the one that ran.
375    pub display_name: &'static str,
376    /// Whether the internal event stream was compared at all. When
377    /// `false` only stdout/stderr/exit status were compared and the strictness
378    /// fields describe a log comparison that did not run — a consumer must not
379    /// read such a verdict as bitwise log parity.
380    pub compare_logs: bool,
381    /// Whether the compared records contain the CONTENT of syscall output
382    /// buffers, not merely the syscalls' return values.
383    ///
384    /// ⚠️ WHEN THIS IS `false`, A MATCH IS NOT A CONTENT-PARITY RESULT, and a
385    /// consumer must not read it as one. Reverie types many output buffers as
386    /// bare pointers, so the record prints the ADDRESS and not the bytes --
387    /// `reverie-syscalls/src/syscalls.rs` carries standing TODOs saying exactly
388    /// that for `Read` and `Write`. Two runs whose buffers differ while their
389    /// return values agree therefore produce character-identical records and
390    /// compare equal.
391    ///
392    /// That is not hypothetical. A `recvmsg` of a netlink RTM_GETLINK dump
393    /// returns a stable `Ok(1468)` on every run while four bytes of its payload
394    /// vary (host kernel jiffies and a kernel-randomised timer). Measured: the
395    /// same guest reports `verdict: matched, bitwise_parity: true` with this
396    /// `false`, and `verdict: diverged` with it `true`. On one QEMU/Linux boot,
397    /// 44.1% of syscalls (278,824 of 632,228) move bytes through a buffer whose
398    /// content the record does not show.
399    ///
400    /// True by default, and set false only by `--no-detlog-io-buffers`. It is
401    /// recorded here rather than inferred so a consumer can require it instead
402    /// of assuming it.
403    pub compare_io_buffers: bool,
404    /// The message envelope selected from the captured log. The compared-message
405    /// counts refer exactly to this scope.
406    pub log_scope: ComparedLogScope,
407    /// Versioned policy selecting which parsed records entered the message
408    /// envelope. A caller-defined predicate is disclosed but cannot qualify for
409    /// parity.
410    pub record_envelope: RecordEnvelopePolicy,
411    /// Whether the Detcore configuration under which BOTH sides ran virtualized
412    /// time.
413    ///
414    /// ⚠️ A `bitwise_parity: true` FROM A COMPARISON WITH `virtualize_time:
415    /// false` IS NOT A DETERMINISM RESULT, and a consumer must not read it as
416    /// one. It says the second side reproduced the first. If the guest read a
417    /// host-derived value such as the real monotonic clock, that value was
418    /// captured and faithfully reproduced, and a separate invocation would
419    /// capture a different one.
420    ///
421    /// This is not hypothetical. `hermit run --verify` executes the guest twice
422    /// with virtual time on, so a match is evidence about the guest; `hermit
423    /// record start --verify` runs under
424    /// `hermit_cli::metadata::record_or_replay_config`, which deliberately sets
425    /// `virtualize_time: false`, so a match says only that the replay reproduced
426    /// that recording. Both previously reported `bitwise_parity: true` with
427    /// nothing in the report to tell them apart, which is what this field fixes.
428    ///
429    /// It lives on [`ComparisonSpec`] rather than on the report so it is absent
430    /// exactly when there was no comparison: `VerificationReport::comparison` is
431    /// `None` for [`Verdict::NoResult`], so a run that never compared anything
432    /// cannot publish a time policy describing a comparison that did not happen.
433    pub virtualize_time: bool,
434    /// Concrete: were numeric values, addresses, tmp paths, and timestamps
435    /// normalized away wholesale before diffing (the lossy [`Stripped`] path)?
436    ///
437    /// [`Stripped`]: LogCompareStrictness::Stripped
438    pub strip_lines: bool,
439    /// Concrete: were host memory addresses canonicalized to first-appearance
440    /// ordinals before diffing? Unlike
441    /// [`Self::strip_lines`] this is lossless for parity: it discards only the
442    /// raw pointer value, keeping identity, order, and aliasing.
443    pub canonicalize_addresses: bool,
444    /// Concrete: was the complete parity observation envelope compared (vs. the
445    /// legacy deterministic subset)? For `BitwiseInfoV1`, that complete envelope
446    /// is every INFO record admitted by [`Self::record_envelope`], and
447    /// [`Self::log_scope`] records whether the explicit diagnostic full-trace
448    /// superset was requested.
449    pub full_trace: bool,
450    /// Concrete: was everything OTHER than the stripped prefix and the
451    /// canonicalized addresses compared exactly (virtual-time timestamps,
452    /// syscall inputs/results, counts, sizes, flags)? True for the parity policy;
453    /// false whenever a lossy normalization (e.g. [`Self::strip_lines`]) ran.
454    pub exact_remainder: bool,
455    /// Versioned tokens for every prefix STRIPPED before comparison. The parity
456    /// policy permits exactly `["real-wall-clock-prefix/v1"]`; a lossy stripped
457    /// comparison additionally lists the wholesale-normalization token. Recorded
458    /// (not inferred) so a consumer can see precisely what was discarded.
459    pub stripped_prefixes: &'static [&'static str],
460    /// Versioned tokens for every CANONICALIZATION applied before comparison. The
461    /// parity policy requires exactly
462    /// `["host-address-to-first-appearance-ordinal/v1"]`.
463    pub canonicalizations: &'static [&'static str],
464    /// Concrete: were any `--ignore-lines` substring filters applied, dropping
465    /// matching log lines before the comparison? Bitwise parity requires none.
466    pub ignore_lines: bool,
467    /// Concrete: were `COMMIT` messages excluded from the comparison? Bitwise
468    /// parity requires them included.
469    pub skip_commit: bool,
470    /// Concrete: were `DETLOG` messages (or any DETLOG class) excluded from the
471    /// comparison? Bitwise parity requires the full event stream.
472    pub skip_detlog: bool,
473}

475impl ComparisonSpec {
476    /// Build the spec (and, implicitly, the concrete diff flags) from the
477    /// requested strictness and whether logs are compared at all. This is the
478    /// single place the strictness label maps onto `strip_lines`/`full_trace`,
479    /// so the flags the diff engine sees and the flags the verdict reports can
480    /// never drift apart.
481    pub fn new(
482        strictness: LogCompareStrictness,
483        compare_logs: bool,
484        diagnostic_full_trace: bool,
485        compare_io_buffers: bool,
486        record_envelope: RecordEnvelopePolicy,
487        virtualize_time: bool,
488    ) -> Self {
489        // Map the strictness label onto the concrete diff flags AND the versioned
490        // policy tokens in one place, so the flags the engine sees, the tokens
491        // the verdict reports, and the strictness label can never drift apart.
492        let (
493            strip_lines,
494            canonicalize_addresses,
495            full_trace,
496            exact_remainder,
497            log_scope,
498            stripped_prefixes,
499            canonicalizations,
500            display_name,
501        ) = match strictness {
502            // Lossy wholesale normalization: numbers/addresses/paths/timestamps
503            // erased; the remainder is NOT compared exactly.
504            LogCompareStrictness::Stripped => {
505                debug_assert!(!diagnostic_full_trace);
506                (
507                    true,
508                    false,
509                    false,
510                    false,
511                    ComparedLogScope::Deterministic,
512                    &[STRIP_WALL_CLOCK_PREFIX_V1, STRIP_UNSAFE_NORMALIZATION_V1][..],
513                    // Under stripping, addresses are ERASED (to a single `<ADDR>`
514                    // token), not canonicalized; there is no ordinal preserved.
515                    &[][..],
516                    "Stripped",
517                )
518            }
519            // Canonical parity: strip only the wall-clock prefix,
520            // canonicalize addresses, and compare every INFO message admitted
521            // by the selected named record envelope exactly.
522            // The explicit verbose diagnostic mode compares the all-level
523            // superset without changing the canonicalization policy.
524            LogCompareStrictness::Canonical => (
525                false,
526                true,
527                true,
528                true,
529                if diagnostic_full_trace {
530                    ComparedLogScope::FullTrace
531                } else {
532                    ComparedLogScope::Info
533                },
534                PARITY_STRIPPED_PREFIXES,
535                PARITY_CANONICALIZATIONS,
536                if diagnostic_full_trace {
537                    "BitwiseFullTraceV1"
538                } else {
539                    "BitwiseInfoV1"
540                },
541            ),
542        };
543        ComparisonSpec {
544            strictness,
545            display_name,
546            compare_logs,
547            compare_io_buffers,
548            log_scope,
549            record_envelope,
550            virtualize_time,
551            strip_lines,
552            canonicalize_addresses,
553            full_trace,
554            exact_remainder,
555            stripped_prefixes,
556            canonicalizations,
557            // The `--verify` code paths never expose the diff engine's line
558            // filters, so the comparison they produce applies none. These are
559            // recorded (not merely assumed) so a parity consumer can *require*
560            // their absence rather than trust that no CLI surface enables them;
561            // `default_log_diff_opts_apply_no_line_filters` binds these values to
562            // the actual `LogDiffOpts` the engine sees.
563            ignore_lines: false,
564            skip_commit: false,
565            skip_detlog: false,
566        }
567    }
568
569    /// The `LogComparisonMode` this spec selects for the diff engine.
570    fn log_comparison_mode(&self) -> LogComparisonMode {
571        match self.log_scope {
572            ComparedLogScope::Deterministic => LogComparisonMode::Deterministic,
573            ComparedLogScope::Info => LogComparisonMode::Info,
574            ComparedLogScope::FullTrace => LogComparisonMode::FullTrace,
575        }
576    }
577
578    /// Whether the log engine and serialized policy are exactly the shared
579    /// `BitwiseInfoV1` comparison. Observation properties such as output-buffer
580    /// hashing and virtual time are recorded by that policy but do not select a
581    /// different parser or normalization.
582    fn uses_bitwise_info_v1(&self) -> bool {
583        self.strictness == LogCompareStrictness::Canonical
584            && self.compare_logs
585            && self.log_scope == ComparedLogScope::Info
586            && self.record_envelope == RecordEnvelopePolicy::AllRecordsV1
587            && !self.strip_lines
588            && self.canonicalize_addresses
589            && self.full_trace
590            && self.exact_remainder
591            && self.stripped_prefixes == PARITY_STRIPPED_PREFIXES
592            && self.canonicalizations == PARITY_CANONICALIZATIONS
593            && !self.ignore_lines
594            && !self.skip_commit
595            && !self.skip_detlog
596    }
597
598    /// Does this comparison satisfy the `BitwiseInfoV1` parity contract a
599    /// determinism / record-replay ratchet must require before it may read a
600    /// `Matched` verdict as *true parity*? A bare `verified` is not enough:
601    /// `verified` can rest on a stripped compare, an opaque filtered subset, or
602    /// an output-only fallback, all of which normalize or omit exactly the data
603    /// (virtual-time timestamps, raw syscall argument/result values, whole event
604    /// classes) that parity exists to check.
605    ///
606    /// This requires the EXACT `BitwiseInfoV1` policy shape, not merely
607    /// "not stripped": a generic `strip_lines = false` is inadmissible on its
608    /// own. All clauses must hold:
609    /// - the full INFO event stream within the named record envelope (or the
610    ///   explicit all-level diagnostic superset) was compared
611    ///   ([`Self::full_trace`] and [`Self::log_scope`]), which carries exact
612    ///   virtual timestamps and syscall argument/result values;
613    /// - no lossy wholesale normalization ran (`!strip_lines`) and the remainder
614    ///   was compared exactly ([`Self::exact_remainder`]);
615    /// - addresses were CANONICALIZED, not erased ([`Self::canonicalize_addresses`]),
616    ///   so allocation-order and aliasing differences are still detectable;
617    /// - the versioned policy tokens are exactly the parity set — only the
618    ///   wall-clock prefix stripped, only address-ordinal canonicalization
619    ///   applied — so a future extra strip/canonicalization cannot silently pass;
620    /// - the record predicate is one of the explicitly named, versioned
621    ///   canonical envelopes ([`Self::record_envelope`]); an opaque
622    ///   caller-defined predicate cannot qualify;
623    /// - no ad hoc ignore/skip filter dropped any line or event class
624    ///   (`!ignore_lines && !skip_commit && !skip_detlog`);
625    /// - syscall output-buffer content hashes were present in the records
626    ///   ([`Self::compare_io_buffers`]);
627    /// - the internal log stream was actually compared, not skipped for an
628    ///   output-only fallback ([`Self::compare_logs`]).
629    ///
630    /// A consumer asking for parity must reject `Matched` under every weaker
631    /// comparison or observation envelope; this predicate is that single
632    /// acceptance rule.
633    pub fn is_bitwise_parity(&self) -> bool {
634        self.compare_io_buffers
635            && self.compare_logs
636            && self.full_trace
637            && matches!(
638                self.log_scope,
639                ComparedLogScope::Info | ComparedLogScope::FullTrace
640            )
641            && !self.strip_lines
642            && self.canonicalize_addresses
643            && self.exact_remainder
644            && self.stripped_prefixes == PARITY_STRIPPED_PREFIXES
645            && self.canonicalizations == PARITY_CANONICALIZATIONS
646            && self.record_envelope.is_canonical()
647            && !self.ignore_lines
648            && !self.skip_commit
649            && !self.skip_detlog
650    }
651}

697/// The full outcome of comparing two runs: the verification [`Verdict`] plus the
698/// guest exit status, so a caller never has to infer either one from the other.
699#[derive(Debug, Clone)]
700pub struct VerificationOutcome {
701    pub verdict: Verdict,
702    /// Why verification did not reach a verdict. None for a completed match or
703    /// divergence; current no-result producers must provide a typed reason.
704    pub no_result_reason: Option<NoResultReason>,
705    /// The log comparator's exact refusal (a log cut at its size bound), kept
706    /// even when an observed output or exit-status difference selected
707    /// [`Verdict::Diverged`] and so left [`Self::no_result_reason`] empty. The
708    /// ordinary report does not carry it, because the report schema allows a
709    /// `no_result_reason` only beside a `no_result` verdict;
710    /// [`write_skid_overshoot_verification_json`] reads it so that an overshoot
711    /// cannot erase the refusal.
712    pub refused_log_comparison: Option<String>,
713    /// Exit status of the second (replay / repeat) run, propagated verbatim.
714    pub guest_status: ExitStatus,
715    /// The exact common output/log comparison used for [`Self::verdict`],
716    /// carried so a consumer never has to assume which comparison a "matched"
717    /// rests on. Backend-specific dimensions are carried separately below.
718    pub comparison: ComparisonSpec,
719    /// How many log messages the comparison actually compared, or `None` when
720    /// the log comparison was not run at all (output-only fallback). `None` and
721    /// `Some(0/0)` are both "no log evidence" and neither can support parity.
722    pub compared_log_messages: Option<ComparedLogCounts>,
723    /// Exact guest output and disposition for both runs.
724    pub compared_outputs: ComparedOutputs,
725    /// Typed whole-process DBT branch clocks, when that backend completed both
726    /// runs and produced readable statistics. `None` for non-DBT runs and when
727    /// verification did not reach a verdict.
728    pub dbt_counted_branches: Option<DbtCountedBranchComparison>,
729    /// Runtime totals for the two compared executions when their summaries were readable.
730    pub runtime: Option<VerificationRuntime>,
731    /// Scheduler turn at the first log divergence, when a preceding COMMIT
732    /// identified the turn.
733    pub first_divergent_scheduler_turn: Option<u64>,
734    /// The first different record's position in each run's log, as
735    /// `detcore::logdiff::log_record_events` counts records (`None` for a log
736    /// that ended before it), when the logs were compared and differ. Not part
737    /// of the report: `hermit::host_input_change` places an open against it.
738    pub first_divergent_record_positions: Option<(Option<usize>, Option<usize>)>,
739    /// Virtual nanoseconds at that same COMMIT, when the log recorded them.
740    pub first_divergent_virtual_nanoseconds: Option<u64>,
741    /// 1-based index of the first differing compared record.
742    ///
743    /// The two fields above are the position of the preceding scheduler COMMIT,
744    /// so when no COMMIT precedes the differing record they both collapse to the
745    /// origin and say nothing about how far the run got. This one is the
746    /// LOCATION of the divergence rather than a bound on it, and it shares a
747    /// unit with `compared_log_messages`, so `record / compared` is the fraction
748    /// of the log that was deterministic.
749    pub first_divergent_record: Option<usize>,
750    /// How many syscalls the guest had COMPLETED when the divergence appeared,
751    /// read from detcore's own `finish syscall #N` counter.
752    ///
753    /// The fourth unit, and the one closest to what a person debugging actually
754    /// pictures: "the guest got 37 syscalls in" is more legible than a record
755    /// index. It is NOT interchangeable with `first_divergent_record` -- one
756    /// counts guest work, the other counts compared log records, and the two
757    /// move at completely different rates.
758    ///
759    /// `null` when the logs matched, when no comparison ran, or when no syscall
760    /// had completed before the divergence -- which is a real state, not a
761    /// missing value: a run can diverge during startup.
762    pub first_divergent_syscall: Option<u64>,
763    /// First differing compared message from the left execution, with only the
764    /// separately recorded syscall number, scheduler turn, and committed time
765    /// removed.
766    pub first_divergent_left_message: Option<String>,
767    /// Corresponding first differing compared message from the right execution.
768    pub first_divergent_right_message: Option<String>,
769}

771impl VerificationOutcome {
772    /// Did verification pass, independent of the guest exit code?
773    pub fn verified(&self) -> bool {
774        self.verdict == Verdict::Matched
775    }
776
777    /// Collapse the outcome to the historical process-exit convention: a match
778    /// propagates the guest exit status, a divergence exits one, and a refused
779    /// comparison remains an error. Callers that need to separate the verdict
780    /// from the guest exit code must read [`Self::verdict`] / [`Self::verified`]
781    /// (or the `--verify-json` report) *before* calling this.
782    pub fn into_exit_status(self) -> Result<ExitStatus, Error> {
783        match self.verdict {
784            Verdict::Matched => Ok(self.guest_status),
785            Verdict::Diverged => Ok(ExitStatus::Exited(HERMIT_VERIFICATION_DIVERGENCE_EXIT)),
786            // Still an error, so the historical nonzero process exit is
787            // unchanged -- but carry the reason instead of replacing it with a
788            // generic message.
789            Verdict::NoResult => match self.no_result_reason {
790                Some(NoResultReason::ComparisonRefused { detail }) => Err(Error::msg(format!(
791                    "Verification did not reach a verdict: {detail}"
792                ))),
793                Some(reason) => Err(Error::msg(format!(
794                    "Verification did not reach a verdict: {reason:?}"
795                ))),
796                None => Err(Error::msg("Verification did not reach a verdict.")),
797            },
798            Verdict::InfrastructureError => {
799                Err(Error::msg("Verification recorded an infrastructure error."))
800            }
801        }
802    }
803}

847pub(crate) fn verification_report(outcome: &VerificationOutcome) -> VerificationReport {
848    VerificationReport {
849        verified: outcome.verified(),
850        // Bitwise parity is a conjunction: the runs matched AND the
851        // comparison was strict enough for the match to *mean* bitwise
852        // identity. A `Diverged` verdict is never bitwise parity.
853        // Three-way conjunction: the runs matched, the comparison was
854        // strict enough for the match to *mean* bitwise identity, AND that
855        // comparison actually consumed log evidence. The third conjunct is
856        // not redundant: an empty-vs-empty log comparison reports "no
857        // difference" under the strictest possible spec, so without a
858        // nonzero count a run that produced no DETLOG at all would certify
859        // as bitwise parity.
860        bitwise_parity: outcome.verified()
861            && outcome.comparison.is_bitwise_parity()
862            && outcome
863                .compared_log_messages
864                .is_some_and(|counts| counts.is_nonzero()),
865        verdict: outcome.verdict,
866        no_result_reason: outcome.no_result_reason.clone(),
867        infrastructure_error: None,
868        comparison: (outcome.verdict != Verdict::NoResult)
869            .then(|| comparison_report(&outcome.comparison)),
870        compared_log_messages: if outcome.verdict == Verdict::NoResult {
871            None
872        } else {
873            outcome
874                .compared_log_messages
875                .map(|counts| ComparedLogMessages {
876                    left: u64::try_from(counts.left).expect("compared log count fits u64"),
877                    right: u64::try_from(counts.right).expect("compared log count fits u64"),
878                })
879        },
880        compared_outputs: (outcome.verdict != Verdict::NoResult)
881            .then(|| outcome.compared_outputs.clone()),
882        dbt_counted_branches: if outcome.verdict == Verdict::NoResult {
883            None
884        } else {
885            outcome.dbt_counted_branches
886        },
887        runtime: outcome.runtime.clone(),
888        guest_exit_code: outcome.guest_status.code(),
889        guest_signal: outcome.guest_status.signal(),
890        first_divergent_scheduler_turn: outcome.first_divergent_scheduler_turn,
891        first_divergent_virtual_nanoseconds: outcome.first_divergent_virtual_nanoseconds,
892        first_divergent_record: outcome
893            .first_divergent_record
894            .map(|record| u64::try_from(record).expect("divergent record index fits u64")),
895        first_divergent_syscall: outcome.first_divergent_syscall,
896        first_divergent_left_message: outcome.first_divergent_left_message.clone(),
897        first_divergent_right_message: outcome.first_divergent_right_message.clone(),
898        exact_branch_counter: None,
899    }
900}

936}

938/// Write a completed comparison that cannot be accepted because precise PMU
939/// timer delivery passed its target. The comparison remains available for
940/// diagnosis, while the canonical verdict makes the refusal unambiguous.
941///
942/// A comparison the comparator itself refused (a log cut at its size bound)
943/// is written as that refusal instead: verdict `no_result` with its
944/// `comparison_refused` reason. That holds whether the comparator selected
945/// `no_result` or, because an output or exit-status difference was also
946/// observed, `diverged`; the overshoot already made that difference
947/// unattributable, and the refusal must not disappear with it. The report
948/// schema allows a `no_result_reason` only beside a `no_result` verdict, so
949/// one report cannot carry both causes, and converting this one to an
950/// overshoot would erase the refusal. The refusal is the failure a fresh
951/// attempt could hide: a reader that may retry a pure overshoot
952/// (<https://github.com/rrnewton/hermit/issues/1845>) must not mistake this
953/// attempt for one. The overshoot itself is still reported, by the
954/// `HERMIT_SKID_OVERSHOOT` lines on stderr and by the caller's skid-overshoot
955/// exit status.
956pub fn write_skid_overshoot_verification_json(
957    path: &Path,
958    outcome: &VerificationOutcome,
959    count: u64,
960) -> Result<(), Error> {
961    assert!(
962        count > 0,
963        "a skid-overshoot report requires a positive count"
964    );
965    let refusal = outcome.refused_log_comparison.clone().or_else(|| {
966        match (&outcome.verdict, &outcome.no_result_reason) {
967            (Verdict::NoResult, Some(NoResultReason::ComparisonRefused { detail })) => {
968                Some(detail.clone())
969            }
970            _ => None,
971        }
972    });
973    if let Some(detail) = refusal {
974        let refused = VerificationOutcome {
975            verdict: Verdict::NoResult,
976            no_result_reason: Some(NoResultReason::ComparisonRefused { detail }),
977            ..outcome.clone()
978        };
979        return write_report_json(path, &verification_report(&refused));
980    }
981    let mut report = verification_report(outcome);
982    report.verified = false;
983    report.bitwise_parity = false;
984    report.verdict = Verdict::InfrastructureError;
985    report.no_result_reason = None;
986    report.infrastructure_error = Some(InfrastructureError::SkidOvershoot { count });
987    report.dbt_counted_branches = None;
988    write_report_json(path, &report)
989}

1091/// Execute one real verification run and replace the pending stamp only when
1092/// the container classifier established an unexpected child exit. Neither a
1093/// returned guest Output nor an unclassified error is that observation.
1094pub(crate) fn run_verification_execution<T>(
1095    path: Option<&Path>,
1096    run: VerificationRun,
1097    execute: impl FnOnce() -> Result<T, Error>,
1098) -> Result<T, Error> {
1099    execute().map_err(|error| {
1100        let Some(disposition) = container_failure_of(&error) else {
1101            return error;
1102        };
1103        let Some(path) = path else { return error };
1104        match write_report_json(path, &container_failed_report(run, disposition)) {
1105            Ok(()) => error,
1106            Err(secondary) => {
1107                retain_verification_error(error, "publishing container failure", secondary)
1108            }
1109        }
1110    })
1111}

1126            signal,

1148/// A skid overshoot is not one: it already published its infrastructure
1149/// error. Neither is an untyped error (a bad flag, an unreadable path), which
1150/// leaves the pending stamp.
1151pub(crate) fn run_failure_of(error: &Error) -> Option<RunFailure> {
1152    if detcore::ReplayDepartureError::in_error(error).is_some() {
1153        Some(RunFailure::PolicyRefusal)
1154    } else if error.downcast_ref::<hermit::SkidOvershootError>().is_some() {
1155        None
1156    } else if error
1157        .downcast_ref::<super::run::DeterminismLossRecorded>()
1158        .is_some()
1159    {
1160        Some(RunFailure::DeterminismLoss)
1161    } else if error.downcast_ref::<hermit::BackendRunFailure>().is_some() {
1162        Some(RunFailure::BackendFailure)
1163    } else if error
1164        .downcast_ref::<super::container::PolicyRefusal>()
1165        .is_some()
1166    {
1167        Some(RunFailure::PolicyRefusal)
1168    } else if matches!(
1169        error.downcast_ref::<super::container::SchedulerOutcome>(),
1170        Some(super::container::SchedulerOutcome(
1171            detcore::TerminalOutcome::GuestDeadlock
1172        ))
1173    ) {
1174        // Not a virtual timeout: `--verify` with the bound is refused on
1175        // every backend that cannot unwind the guest, and on ptrace a stopped
1176        // run completes and is compared.
1177        Some(RunFailure::Deadlock)
1178    } else {
1179        None
1180    }
1181}

1183/// The run_failed detail keeps the failure's message chain, bounded so one
1184/// pathological message cannot grow the report without limit.
1185const RUN_FAILED_DETAIL_MAX_BYTES: usize = 4096;
1186
1187/// Replace the pending stamp with `run_failed` when `error` is a typed failure
1188/// of verification run `run` ([`RunFailure`]), and return `error` unchanged.
1189/// Call it only where `run` had been started: a refusal before any run (the
1190/// host's seccomp filter, an inexact branch counter) is a host condition, not
1191/// a result of this guest on this backend.
1192pub(crate) fn publish_run_failure(
1193    path: Option<&Path>,
1194    run: VerificationRun,
1195    error: Error,
1196) -> Error {
1197    let (Some(path), Some(failure)) = (path, run_failure_of(&error)) else {
1198        return error;
1199    };
1200    match write_report_json(path, &run_failed_report(run, failure, format!("{error:#}"))) {
1201        Ok(()) => error,
1202        Err(secondary) => retain_verification_error(error, "publishing run failure", secondary),
1203    }
1204}

1206pub(crate) fn run_failed_report(
1207    run: VerificationRun,
1208    failure: RunFailure,
1209    mut detail: String,
1210) -> VerificationReport {
1211    if detail.len() > RUN_FAILED_DETAIL_MAX_BYTES {
1212        let mut end = RUN_FAILED_DETAIL_MAX_BYTES;
1213        while !detail.is_char_boundary(end) {
1214            end -= 1;
1215        }
1216        detail.truncate(end);
1217    }
1218    let mut report = VerificationReport::no_result();
1219    report.no_result_reason = Some(NoResultReason::RunFailed {
1220        run,
1221        failure,
1222        detail,
1223    });
1224    report
1225}

1309/// [`retain_logs_after_verification_error`] for a run that the log cap ended.
1310/// The logs are kept as for any other failure, but the report that names them
1311/// is attempted without waiting and is left out when stderr cannot take it at
1312/// once (see [`write_without_waiting`]). Through `eprintln!` it would wait on a
1313/// full stderr pipe, and panic, exiting 101 instead of 123, on one whose reader
1314/// has gone (round-3 review of https://github.com/rrnewton/hermit/pull/3686,
1315/// finding 4).
1316///
1317/// Lines are packed into writes of at most `PIPE_BUF` bytes, so that a pipe
1318/// takes each write whole or not at all; a line longer than that is left out
1319/// rather than cut.
1320fn retain_logs_after_log_cap<const N: usize>(
1321    mut primary: Error,
1322    logs: [(&str, TempPath); N],
1323) -> Error {
1324    let mut lines = vec![":: Verification logs retained:\n".to_owned()];
1325    // Attempt every requested retention, even if an earlier one fails.
1326    for (label, log) in logs {
1327        match log.keep() {
1328            Ok(path) => lines.push(format!("::   {label}: {}\n", path.display())),
1329            Err(secondary) => {
1330                primary = retain_verification_error(
1331                    primary,
1332                    "retaining verification log",
1333                    secondary.into(),
1334                );
1335            }
1336        }
1337    }
1338    let mut chunk = String::new();
1339    for line in lines.iter().filter(|line| line.len() <= libc::PIPE_BUF) {
1340        if chunk.len() + line.len() > libc::PIPE_BUF {
1341            write_without_waiting(libc::STDERR_FILENO, chunk.as_bytes());
1342            chunk.clear();
1343        }
1344        chunk.push_str(line);
1345    }
1346    if !chunk.is_empty() {
1347        write_without_waiting(libc::STDERR_FILENO, chunk.as_bytes());
1348    }
1349    primary
1350}

1352/// Write `report` to `path` atomically: a reader concurrent with the write sees
1353/// either the old contents or the complete new record, never a truncated one.
1354/// The directory to stage the temporary record in: always the one the target
1355/// lives in, so `persist` is a same-filesystem rename.
1356///
1357/// `Path::parent` returns an EMPTY path for a bare filename, not `"."`. Falling
1358/// back to the system temp directory for that case (as this did) puts the
1359/// staged file on a different filesystem than the target whenever `TMPDIR` and
1360/// the working directory differ -- the common case, e.g. tmpfs `/tmp` beside a
1361/// btrfs checkout. `persist` then fails with `EXDEV` and the caller returns an
1362/// error, leaving whatever the file previously held: a stale
1363/// `{verified:true}` survives precisely the invocation that was supposed to
1364/// overwrite it. A bare filename means the working directory, so say so.
1365fn staging_directory(path: &Path) -> &Path {
1366    match path.parent() {
1367        Some(parent) if !parent.as_os_str().is_empty() => parent,
1368        _ => Path::new("."),
1369    }
1370}

1372pub fn write_report_json(path: &Path, report: &VerificationReport) -> Result<(), Error> {
1373    use std::io::Write as _;
1374
1375    let report = with_exact_branch_counter(report, EXACT_BRANCH_COUNTER.get());
1376    let json = serde_json::to_string(&report)?;
1377    // Same directory as the target so the rename below stays within one
1378    // filesystem and is therefore atomic.
1379    let mut temp = NamedTempFile::new_in(staging_directory(path))
1380        .with_context(|| format!("creating a temporary file beside {}", path.display()))?;
1381    writeln!(temp, "{json}")
1382        .with_context(|| format!("writing verification verdict for {}", path.display()))?;
1383    temp.flush()
1384        .with_context(|| format!("flushing verification verdict for {}", path.display()))?;
1385    temp.persist(path)
1386        .map_err(|e| e.error)
1387        .with_context(|| format!("publishing verification verdict to {}", path.display()))?;
1388    Ok(())
1389}

1444pub fn temp_log_files_in(
1445    name1: &str,
1446    name2: &str,
1447    directory: Option<&Path>,
1448) -> io::Result<(NamedTempFile, NamedTempFile)> {
1449    let fallback = std::env::var_os(VERIFY_TMPDIR_ENV).filter(|value| !value.is_empty());

1464            }),
1465            (None, None) => builder.tempfile(),
1466        }
1467    };

1469    let file2 = create(name2)?;

1471    Ok((file1, file2))

1531/// How many of a run's last INFO DETLOG records [`signal_termination_report`]
1532/// shows.
1533const SIGNAL_REPORT_RECORDS: usize = 20;
1534/// How much of the end of a run's log [`signal_termination_report`] reads.
1535const SIGNAL_REPORT_TAIL_BYTES: u64 = 4 << 20;
1536/// The longest record [`signal_termination_report`] shows before cutting it.
1537const SIGNAL_REPORT_LINE_BYTES: usize = 400;
1538
1539/// What the guest of a run was doing when a signal terminated it: the last
1540/// INFO DETLOG records of its log, the records `--verify` compares, for
1541/// printing to stderr. `None` unless `status` is a termination by a signal.
1542/// The DEBUG records a verification log also holds (register dumps, scheduler
1543/// steps) would otherwise push the guest's last system calls out of the
1544/// report.
1545///
1546/// A verify log lives where `--verify-log-dir` put it, which for a test is
1547/// usually inside a disposable checkout, and `--keep-logs` keeps it only as
1548/// long as that checkout. Stderr is what a test runner keeps, so a crash
1549/// under `--verify` that the caller never reproduces is still diagnosable
1550/// from these records; see https://github.com/rrnewton/hermit/issues/3833,
1551/// where it was not.
1552///
1553/// The wait status carries only the signal and whether a core was dumped, not
1554/// the faulting address or thread, so the records are the evidence of where.
1555/// Each is shown without its machine-readable suffix and cut at
1556/// [`SIGNAL_REPORT_LINE_BYTES`], so the report stays a few kilobytes.
1557pub(crate) fn signal_termination_report(
1558    label: &str,
1559    status: ExitStatus,
1560    log: &Path,
1561) -> Option<String> {
1562    let (number, name, core_dumped) = match status {
1563        ExitStatus::Signaled(signal, core_dumped) => {
1564            (signal as i32, format!("{signal:?}"), core_dumped)
1565        }
1566        // A realtime signal, which nix's Signal cannot name
1567        // (https://github.com/rrnewton/reverie/issues/997).
1568        ExitStatus::SignaledRaw(signal, core_dumped) => {
1569            (signal, format!("SIG{signal}"), core_dumped)
1570        }
1571        ExitStatus::Exited(_) => return None,
1572    };
1573    let core = if core_dumped { ", core dumped" } else { "" };
1574    let heading = format!(":: The {label} guest was terminated by signal {number} ({name}{core}).");
1575    let tail = match read_log_tail(log, SIGNAL_REPORT_TAIL_BYTES) {
1576        Ok(tail) => tail,
1577        Err(error) => {
1578            return Some(format!(
1579                "{heading} Its log {} could not be read: {error}\n",
1580                log.display()
1581            ));
1582        }
1583    };
1584    let records: Vec<&str> = tail
1585        .lines()
1586        .filter(|line| is_info_detlog(line))
1587        .map(|line| {
1588            // The last separator, as the DETLOG reader splits: the human
1589            // text before it can hold the same characters, in a pathname.
1590            line.rsplit_once(detcore::detlog::RECORD_SEPARATOR)
1591                .map_or(line, |(human, _)| human)
1592        })
1593        .collect();
1594    if records.is_empty() {
1595        return Some(format!(
1596            "{heading} The last {} bytes of its log {} hold no INFO DETLOG record.\n",
1597            SIGNAL_REPORT_TAIL_BYTES,
1598            log.display()
1599        ));
1600    }
1601    let shown = &records[records.len().saturating_sub(SIGNAL_REPORT_RECORDS)..];
1602    let mut report = format!(
1603        "{heading} Its last {} INFO DETLOG records, from {}:\n",
1604        shown.len(),
1605        log.display()
1606    );
1607    for record in shown {
1608        report.push_str("::   ");
1609        report.push_str(cut_at_char_boundary(record, SIGNAL_REPORT_LINE_BYTES));
1610        if record.len() > SIGNAL_REPORT_LINE_BYTES {
1611            report.push_str(" [cut]");
1612        }
1613        report.push('\n');
1614    }
1615    Some(report)
1616}

1618/// The [`signal_termination_report`]s of a verification's runs, prepared
1619/// while each run's log exists, because the comparison consumes the logs.
1620///
1621/// When this is dropped they are queued, unless [`Self::verified`] discarded
1622/// them first. So every way a verification ends without verifying its runs
1623/// reports the signals that ended them: a status `--verify-allow` refuses, a
1624/// comparison that rejects runs whose statuses `--verify-allow=both` or
1625/// `failure` admitted, or an error. A verification that verifies its runs has
1626/// nothing to explain and prints nothing.
1627///
1628/// `main` writes the queue with [`write_queued_signal_reports`], after
1629/// everything else hermit says, so the report can neither delay nor replace
1630/// the verification's own disposition: nothing waits behind it, and it never
1631/// waits itself.
1632#[must_use]
1633#[derive(Default)]
1634pub(crate) struct SignalTerminationReports(Vec<String>);
1635
1636/// The reports of verifications that ended unverified, for
1637/// [`write_queued_signal_reports`].
1638static QUEUED_SIGNAL_REPORTS: std::sync::Mutex<Vec<String>> = std::sync::Mutex::new(Vec::new());
1639
1640/// Write the queued [`SignalTerminationReports`] to stderr. `main` calls this
1641/// last, just before hermit exits.
1642///
1643/// No write waits for the reader of stderr, with or without
1644/// `--max-log-bytes`: each goes through
1645/// [`hermit::nonwaiting_write::write_without_waiting_for_a_reader`], and the
1646/// first one that stderr refuses or that fails ends the report. A pipe nobody
1647/// reads therefore gets what fits and hermit exits; the line that filled it
1648/// may be cut.
1649pub(crate) fn write_queued_signal_reports() {
1650    let reports = std::mem::take(
1651        &mut *QUEUED_SIGNAL_REPORTS
1652            .lock()
1653            .unwrap_or_else(std::sync::PoisonError::into_inner),
1654    );
1655    write_signal_reports(libc::STDERR_FILENO, &reports);
1656}
1657
1658/// Write `reports` to `fd` for [`write_queued_signal_reports`], one line per
1659/// write.
1660///
1661/// ⚠️ ONE WRITE OF A WHOLE REPORT CAN PUT NOTHING IN A PIPE THAT HAS ROOM. A
1662/// non-blocking write to a Linux pipe adds to the pipe's last page only the
1663/// part of the write past its last whole page, and only when all of that part
1664/// fits there; the rest needs an empty page. A report of 7.8 KB written to a
1665/// one-page pipe that already holds 700 bytes is therefore refused whole,
1666/// although 3.4 KB of the page are free: whether its first line arrived
1667/// depended on how much hermit had said before it, and the no-perf-counters
1668/// warning a GitHub-hosted runner prints was enough to lose it. Records are
1669/// cut at [`SIGNAL_REPORT_LINE_BYTES`], far less than a page, so each line is
1670/// added to the last page while it fits.
1671fn write_signal_reports(fd: libc::c_int, reports: &[String]) {
1672    for line in reports
1673        .iter()
1674        .flat_map(|report| report.split_inclusive('\n'))
1675    {
1676        let mut rest = line.as_bytes();
1677        while !rest.is_empty() {
1678            match hermit::nonwaiting_write::write_without_waiting_for_a_reader(fd, rest) {
1679                Ok(written) if written > 0 => rest = &rest[written..],
1680                _ => return,
1681            }
1682        }
1683    }
1684}

1704        QUEUED_SIGNAL_REPORTS
1705            .lock()
1706            .unwrap_or_else(std::sync::PoisonError::into_inner)
1707            .append(&mut self.0);
1708    }
1709}

1723/// The last `max` bytes of `path`, from the first line that starts within
1724/// them, with invalid UTF-8 replaced.
1725fn read_log_tail(path: &Path, max: u64) -> io::Result<String> {
1726    use std::io::Read;
1727    use std::io::Seek;
1728    use std::io::SeekFrom;
1729
1730    let mut file = File::open(path)?;
1731    let start = file.metadata()?.len().saturating_sub(max);
1732    // From one byte earlier, which tells whether `start` begins a line.
1733    file.seek(SeekFrom::Start(start.saturating_sub(1)))?;
1734    let mut bytes = Vec::new();
1735    file.read_to_end(&mut bytes)?;
1736    if start > 0 {
1737        // Drop that byte, and the rest of the line it ends or is part of.
1738        let first_line_end = bytes.iter().position(|&byte| byte == b'\n');
1739        bytes.drain(..first_line_end.map_or(bytes.len(), |end| end + 1));
1740    }
1741    Ok(String::from_utf8_lossy(&bytes).into_owned())
1742}

1790fn retained_comparison_dirs(root: &Path) -> io::Result<Vec<(PathBuf, std::time::SystemTime)>> {
1791    let mut directories = Vec::new();
1792    for entry in fs::read_dir(root)? {
1793        let entry = entry?;
1794        let name = entry.file_name();
1795        let Some(name) = name.to_str() else {
1796            continue;
1797        };
1798        if !name.starts_with(FAILED_VERIFY_LOG_DIR_PREFIX) {
1799            continue;
1800        }
1801        let metadata = fs::symlink_metadata(entry.path())?;
1802        if !metadata.file_type().is_dir() || metadata.uid() != unsafe { libc::geteuid() } {
1803            continue;
1804        }
1805        directories.push((entry.path(), metadata.modified()?));
1806    }
1807    directories.sort_by(|(left_path, left_time), (right_path, right_time)| {
1808        right_time
1809            .cmp(left_time)
1810            .then_with(|| right_path.cmp(left_path))
1811    });
1812    Ok(directories)
1813}

1816    for entry in fs::read_dir(root)? {

1823            && !name.starts_with(FAILED_VERIFY_LOG_PENDING_PREFIX)

1826        }

1828        if metadata.file_type().is_dir() && metadata.uid() == unsafe { libc::geteuid() } {
1829            fs::remove_dir_all(entry.path())?;

1835fn retire_failed_verification_logs(root: &Path, current: &Path, keep: usize) -> io::Result<usize> {
1836    let directories = retained_comparison_dirs(root)?;
1837    let mut kept = BTreeSet::from([current.to_path_buf()]);
1838    for (path, _) in &directories {
1839        if kept.len() >= keep.max(1) {
1840            break;
1841        }
1842        kept.insert(path.clone());
1843    }
1844
1845    let mut retired = 0;
1846    for (path, _) in directories {
1847        if kept.contains(&path) {
1848            continue;
1849        }
1850        let lock = match open_retention_lock(&path.join(FAILED_VERIFY_LOG_LOCK)) {
1851            Ok(lock) => lock,
1852            Err(error) => {
1853                eprintln!(
1854                    "WARNING: could not inspect retained verification logs {}: {error}",
1855                    path.display()
1856                );
1857                continue;
1858            }
1859        };
1860        match flock(&lock, libc::LOCK_EX | libc::LOCK_NB) {
1861            Ok(()) => {}
1862            Err(error) if error.kind() == io::ErrorKind::WouldBlock => continue,
1863            Err(error) => {
1864                eprintln!(
1865                    "WARNING: could not lock retained verification logs {}: {error}",
1866                    path.display()
1867                );
1868                continue;
1869            }
1870        }
1871        let name = path
1872            .file_name()
1873            .and_then(|name| name.to_str())
1874            .ok_or_else(|| io::Error::other("retained verification directory has no name"))?;
1875        let retiring = tempfile::Builder::new()
1876            .prefix(&format!("{FAILED_VERIFY_LOG_RETIRING_PREFIX}{name}-"))
1877            .rand_bytes(12)
1878            .tempdir_in(root)?
1879            .keep();
1880        fs::remove_dir(&retiring)?;
1881        fs::rename(&path, &retiring)?;
1882        fs::remove_dir_all(&retiring)?;
1883        retired += 1;
1884    }
1885    Ok(retired)
1886}

1888fn retain_failed_verification_logs(
1889    logs: [(&str, TempPath); 2],
1890    retention: &FailedVerifyLogRetention,
1891) -> Result<Vec<PathBuf>, Error> {
1892    fs::create_dir_all(&retention.root).with_context(|| {
1893        format!(
1894            "could not create failed verification log directory {}",
1895            retention.root.display()
1896        )
1897    })?;
1898    let root = fs::canonicalize(&retention.root).with_context(|| {
1899        format!(
1900            "could not resolve failed verification log directory {}",
1901            retention.root.display()
1902        )
1903    })?;
1904    let root_lock = open_retention_lock(&root.join(FAILED_VERIFY_LOG_ROOT_LOCK))?;
1905    flock(&root_lock, libc::LOCK_EX)?;
1906    finish_interrupted_retirements(&root)?;
1907
1908    let pending = tempfile::Builder::new()
1909        .prefix(FAILED_VERIFY_LOG_PENDING_PREFIX)
1910        .rand_bytes(12)
1911        .tempdir_in(&root)?
1912        .keep();
1913    let pending_name = pending
1914        .file_name()
1915        .and_then(|name| name.to_str())
1916        .ok_or_else(|| io::Error::other("failed verification directory has no name"))?;
1917    let final_name = pending_name
1918        .strip_prefix(FAILED_VERIFY_LOG_PENDING_PREFIX)
1919        .ok_or_else(|| io::Error::other("failed verification directory has an invalid name"))?;
1920    let completed = root.join(format!("{FAILED_VERIFY_LOG_DIR_PREFIX}{final_name}"));
1921    let pair_lock = open_retention_lock(&pending.join(FAILED_VERIFY_LOG_LOCK))?;
1922    flock(&pair_lock, libc::LOCK_EX)?;
1923
1924    let mut retained = Vec::with_capacity(logs.len());
1925    for (index, (label, log)) in logs.into_iter().enumerate() {
1926        let original_name = log
1927            .file_name()
1928            .and_then(|name| name.to_str())
1929            .unwrap_or(if index == 0 { "run1.log" } else { "run2.log" });
1930        let destination = pending.join(original_name);
1931        persist_log(log, &destination)?;
1932        retained.push((label, destination));
1933    }
1934    fs::rename(&pending, &completed)?;
1935    for (label, path) in &mut retained {
1936        *path = completed.join(path.file_name().expect("retained log has a name"));
1937        eprintln!("::   {label}: {}", path.display());
1938    }
1939    drop(pair_lock);
1940
1941    let retired = retire_failed_verification_logs(&root, &completed, retention.keep)?;
1942    if retired > 0 {
1943        eprintln!(
1944            ":: Retired {retired} older failed verification comparison(s); keeping the newest {} plus comparisons with active readers",
1945            retention.keep.max(1)
1946        );
1947    }
1948    Ok(retained.into_iter().map(|(_, path)| path).collect())
1949}

1951pub(crate) fn lock_failed_verification_logs_for_read(
1952    paths: impl IntoIterator<Item = PathBuf>,
1953) -> io::Result<Vec<File>> {
1954    let mut directories = BTreeSet::new();
1955    for path in paths {
1956        let path = match fs::canonicalize(path) {
1957            Ok(path) => path,
1958            // Follow mode also accepts logs their writers have not created yet.
1959            Err(error) if error.kind() == io::ErrorKind::NotFound => continue,
1960            Err(error) => return Err(error),
1961        };
1962        let Some(directory) = path.parent() else {
1963            continue;
1964        };
1965        let Some(name) = directory.file_name().and_then(|name| name.to_str()) else {
1966            continue;
1967        };
1968        if name.starts_with(FAILED_VERIFY_LOG_DIR_PREFIX)
1969            && directory.join(FAILED_VERIFY_LOG_LOCK).is_file()
1970        {
1971            directories.insert(directory.to_path_buf());
1972        }
1973    }
1974    let mut locks = Vec::with_capacity(directories.len());
1975    for directory in directories {
1976        let lock = open_retention_lock(&directory.join(FAILED_VERIFY_LOG_LOCK))?;
1977        flock(&lock, libc::LOCK_SH)?;
1978        locks.push(lock);
1979    }
1980    Ok(locks)
1981}

1991fn compare_two_runs_with_unsupported_scan(
1992    first: ComparedRun<'_>,
1993    second: ComparedRun<'_>,
1994    options: ComparisonOptions,
1995    scan_unsupported_syscalls: impl Fn(&Path) -> io::Result<BTreeSet<String>>,
1996) -> Result<VerificationOutcome, Error> {
1997    let ComparedRun {
1998        output: out1,
1999        log: log1,
2000        label: label1,
2001    } = first;
2002    let ComparedRun {
2003        output: out2,
2004        log: log2,
2005        label: label2,
2006    } = second;
2007    let output_evidence = |output: &Output| ComparedOutput {
2008        exit_code: output.status.code(),
2009        signal: output.status.signal(),
2010        stdout_sha256: detcore::Digest::new(&output.stdout).to_string(),
2011        stdout_bytes: u64::try_from(output.stdout.len()).expect("guest stdout length fits u64"),
2012        stderr_sha256: detcore::Digest::new(&output.stderr).to_string(),
2013        stderr_bytes: u64::try_from(output.stderr.len()).expect("guest stderr length fits u64"),
2014    };
2015    let compared_outputs = ComparedOutputs {
2016        left: output_evidence(out1),
2017        right: output_evidence(out2),
2018    };
2019    let compared_labels = ComparisonSideLabels::new(label1, label2);
2020    let mut failed = false;
2021    // A difference that was actually OBSERVED, as opposed to a comparison that
2022    // was refused. Only an observed difference can justify a `Diverged`
2023    // verdict; see the verdict selection at the end of this function.
2024    let mut observed_divergence = false;
2025    // The log comparator's exact reason for declining to produce a verdict.
2026    let mut comparison_refusal_reason = None;
2027    // None until the log comparison actually runs; stays None on the
2028    // output-only fallback so the report can distinguish
2029    // "compared nothing" from "compared and matched".
2030    let mut compared_log_messages: Option<ComparedLogCounts> = None;
2031    let mut first_divergent_scheduler_turn = None;
2032    let mut first_divergent_record_positions = None;
2033    let mut first_divergent_virtual_nanoseconds = None;
2034    let mut first_divergent_record = None;
2035    let mut first_divergent_syscall = None;
2036    let mut first_divergent_left_message = None;
2037    let mut first_divergent_right_message = None;
2038
2039    // Resolve the strictness label to concrete diff flags once, and carry the
2040    // resulting spec through to the verdict so the returned outcome records
2041    // exactly which comparison certified it.
2042    let spec = ComparisonSpec::new(
2043        options.strictness,
2044        options.compare_logs,
2045        options.diagnostic_full_trace,
2046        options.compare_io_buffers,
2047        options.record_envelope.policy(),
2048        options.virtualize_time,
2049    );
2050
2051    if out1.stdout != out2.stdout {
2052        failed = true;
2053        observed_divergence = true;
2054        eprintln!("Mismatch in stdout between {label1} and {label2}:");
2055        let str1 = String::from_utf8_lossy(&out1.stdout);
2056        let str2 = String::from_utf8_lossy(&out2.stdout);
2057        if str1.lines().count() > 1 {
2058            display_diff(&str1, &str2);
2059        } else {
2060            eprintln!("{}", Comparison::new(&str1, &str2));
2061        }
2062    }
2063
2064    if out1.stderr != out2.stderr {
2065        failed = true;
2066        observed_divergence = true;
2067        eprintln!("Mismatch in stderr between {label1} and {label2}:");
2068        let str1 = String::from_utf8_lossy(&out1.stderr);
2069        let str2 = String::from_utf8_lossy(&out2.stderr);
2070        if str1.lines().count() > 1 {
2071            display_diff(&str1, &str2);
2072        } else {
2073            eprintln!("{}", Comparison::new(&str1, &str2));
2074        }
2075    }
2076
2077    let log_processing_result = (|| -> Result<(), Error> {
2078        if options.compare_logs {
2079            eprintln!(
2080                ":: {}",
2081                "Comparing captured verification logs...".yellow().bold()
2082            );
2083            // The comparison semantics come from `spec` (strip_lines + mode); only
2084            // the printed syscall-history depth still tracks `verbose`. Historically
2085            // both were flipped together, so the sole bitwise comparison was also the
2086            // loudest — decoupling them lets a quiet run be bitwise-strict.
2087            let diff_options = logdiff::LogDiffOpts {
2088                strip_lines: spec.strip_lines,
2089                // Thread canonical address normalization from the spec so the parity
2090                // (`Canonical`) policy actually rewrites host addresses to ordinals
2091                // in the engine; without this the verdict would REPORT
2092                // `canonicalize_addresses = true` while the diff ran with the raw
2093                // addresses — the exact proxy/binding drift the spec exists to close.
2094                canonicalize_addresses: spec.canonicalize_addresses,
2095                comparison: spec.log_comparison_mode(),
2096                side_labels: compared_labels.clone(),
2097                // These logs were produced by this invocation, so accepting the
2098                // historical prose-only shape would hide a producer regression.
2099                // Standalone log-diff keeps that compatibility path for retained
2100                // older logs; a current verification result requires Detcore's
2101                // structured event records.
2102                require_structured_events: true,
2103                syscall_history: if options.verbose { 10 } else { 5 },
2104                // Thread the filter facts from the spec so what the verdict *reports*
2105                // (`spec.skip_commit`/`spec.skip_detlog`) is exactly what the diff
2106                // engine *does*; the remaining filters stay at their no-op defaults.
2107                skip_commit: spec.skip_commit,
2108                skip_detlog: spec.skip_detlog,
2109                ..Default::default()
2110            };
2111            // Bind the spec's recorded filter-absence to the engine's real defaults:
2112            // if `LogDiffOpts::default()` ever grew a filtering default, the spec
2113            // would silently misreport "no filters", so refuse to run in that case.
2114            debug_assert!(
2115                diff_options.ignore_lines.is_empty() != spec.ignore_lines,
2116                "ComparisonSpec.ignore_lines must match the diff engine's ignore_lines"
2117            );
2118
2119            let summary = if spec.uses_bitwise_info_v1() {
2120                // This path cannot be weakened by caller-selected record or
2121                // normalization filters while its report names BitwiseInfoV1.
2122                // Diagnostic settings preserve the existing console output but
2123                // cannot alter the verdict.
2124                let mut diagnostic_output = std::io::stderr();
2125                logdiff::try_compare_bitwise_info_v1_with_diagnostics(
2126                    log1.as_ref(),
2127                    log2.as_ref(),
2128                    compared_labels.clone(),
2129                    logdiff::BitwiseInfoV1Diagnostics {
2130                        difference_limit: diff_options.limit,
2131                        syscall_history: diff_options.syscall_history,
2132                        no_color: diff_options.no_color,
2133                        print_logs: diff_options.print_logs,
2134                    },
2135                    &mut diagnostic_output,
2136                )?
2137            } else {
2138                logdiff::try_log_diff_detailed_with_filter(
2139                    log1.as_ref(),
2140                    log2.as_ref(),
2141                    &diff_options,
2142                    options.record_envelope.predicate(),
2143                )?
2144            };
2145            compared_log_messages = Some(ComparedLogCounts {
2146                left: summary.compared_left,
2147                right: summary.compared_right,
2148            });
2149            // A recvmsg whose buffers could not be observed after the call is
2150            // logged as an `unobserved` buffer-digest record and compared like
2151            // any other; every verification path reports how many there were,
2152            // so the gap is never silent.
2153            let unobserved = [log1.as_ref(), log2.as_ref()].map(|log: &std::path::Path| {
2154                logdiff::count_unobserved_receives(log)
2155                    .map_or_else(|_| "unreadable".to_string(), |count| count.to_string())
2156            });
2157            if unobserved.iter().any(|count| count != "0") {
2158                eprintln!(
2159                    ":: recvmsg calls whose buffers were unobservable after the call (compared as 'unobserved' buffer-digest entries): run1={}, run2={}",
2160                    unobserved[0], unobserved[1]
2161                );
2162            }
2163            if summary.diff_found {
2164                failed = true;
2165                if let Some(refusal_reason) = summary.refusal_reason.clone() {
2166                    comparison_refusal_reason = Some(refusal_reason);
2167                } else {
2168                    observed_divergence = true;
2169                }
2170                first_divergent_scheduler_turn = summary.first_divergent_scheduler_turn;
2171                first_divergent_record_positions = summary.first_divergent_record_positions;
2172                first_divergent_virtual_nanoseconds = summary.first_divergent_virtual_nanoseconds;
2173                // Set inside this `diff_found` arm, matching its two siblings
2174                // above.
2175                //
2176                // The arm is DEFENCE, not the source of the guarantee, and the
2177                // difference matters to whoever edits this next: a matching pair
2178                // already yields `None` from the comparator, because
2179                // `first_divergent_record` is computed from `first_different`,
2180                // which is `None` when nothing differed. Deleting this arm would
2181                // therefore NOT produce a `0` today, and no test would catch its
2182                // removal -- verified by deleting the equivalent gate in
2183                // `logdiff.rs` and watching
2184                // `identical_logs_have_no_first_divergent_record` still pass.
2185                //
2186                // It is kept because it costs nothing and it is what holds the
2187                // invariant if the comparator ever starts reporting a position
2188                // on a match.
2189                first_divergent_record = summary.first_divergent_record;
2190                first_divergent_syscall = summary.first_divergent_syscall;
2191                first_divergent_left_message = summary.first_divergent_left_message.clone();
2192                first_divergent_right_message = summary.first_divergent_right_message.clone();
2193                if summary.refusal_reason.is_none() {
2194                    eprintln!(
2195                        ":: {}",
2196                        format!("Log differences found between {label1} and {label2}.")
2197                            .red()
2198                            .bold()
2199                    );
2200                }
2201            }
2202        } else {
2203            eprintln!(
2204                ":: Comparing guest output and exit status only; internal logs were not compared"
2205            );
2206        }
2207
2208        if out1.status != out2.status {
2209            failed = true;
2210            observed_divergence = true;
2211            eprintln!(
2212                "Mismatch in exit status between {label1} and {label2}: {}",
2213                Comparison::new(&out1.status, &out2.status)
2214            );
2215        }
2216
2217        if !failed {
2218            let mut unsupported = scan_unsupported_syscalls(log1.as_ref())?;
2219            unsupported.extend(scan_unsupported_syscalls(log2.as_ref())?);
2220            if let Some(message) = detcore::format_unsupported_syscall_warning(&unsupported) {
2221                eprintln!("WARNING: {message}");
2222            }
2223        }
2224        Ok(())
2225    })();
2226
2227    if options.caller_owns_logs {
2228        // Links to the caller's logs: dropping them leaves the logs.
2229        drop((log1, log2));
2230        log_processing_result?;
2231    } else {
2232        if let Err(error) = log_processing_result {
2233            if options.keep_logs || failed {
2234                if let Some(retention) = &options.failed_log_retention {
2235                    eprintln!(":: Verification logs retained:");
2236                    retain_failed_verification_logs([(label1, log1), (label2, log2)], retention)?;
2237                } else {
2238                    retain_verification_logs([(label1, log1), (label2, log2)])?;
2239                }
2240            }
2241            return Err(error);
2242        }
2243
2244        // Divergence historically retained both diagnostics, and it still does.
2245        // `--keep-logs` extends retention to successful comparisons, but a match
2246        // needs only one log: once the two logs compared equal under the selected
2247        // policy, the first is kept as the golden log and the second is deleted.
2248        // It is deleted only when the logs were actually compared and no result
2249        // the caller already observed overrides the match; every other outcome
2250        // keeps both, as the failure path always has.
2251        let keep_golden_log_only = options.keep_logs
2252            && !failed
2253            && !options.match_overridden
2254            && compared_log_messages.is_some();
2255        if keep_golden_log_only {
2256            retain_verification_logs([(label1, log1)])?;
2257            let matched_log = log2.to_path_buf();
2258            log2.close().with_context(|| {
2259                format!(
2260                    "could not delete {label2}'s log {}, which matched {label1}'s",
2261                    matched_log.display()
2262                )
2263            })?;
2264            // Deliberately not in the `::   <label>: <path>` shape printed for a
2265            // retained log: the deleted file has no path left to report.
2266            eprintln!(

2338/// Name only relaxations in the comparison itself. Execution policy such as
2339/// time or metadata virtualization remains in the typed report fields; folding
2340/// it into this string would make `none` depend on an open-ended list of run
2341/// options instead of the comparator contract this evidence describes.
2342fn comparison_relaxations(comparison: &ComparisonSpec) -> Option<String> {
2343    let policy_shape_matches = match comparison.strictness {
2344        LogCompareStrictness::Stripped => {
2345            comparison.display_name == "Stripped"
2346                && comparison.strip_lines
2347                && !comparison.canonicalize_addresses
2348                && !comparison.full_trace
2349                && !comparison.exact_remainder
2350                && comparison.log_scope == ComparedLogScope::Deterministic
2351                && comparison.stripped_prefixes
2352                    == [STRIP_WALL_CLOCK_PREFIX_V1, STRIP_UNSAFE_NORMALIZATION_V1]
2353                && comparison.canonicalizations.is_empty()
2354        }
2355        LogCompareStrictness::Canonical => {
2356            !comparison.strip_lines
2357                && comparison.canonicalize_addresses
2358                && comparison.full_trace
2359                && comparison.exact_remainder
2360                && matches!(
2361                    (comparison.log_scope, comparison.display_name),
2362                    (ComparedLogScope::Info, "BitwiseInfoV1")
2363                        | (ComparedLogScope::FullTrace, "BitwiseFullTraceV1")
2364                )
2365                && comparison.stripped_prefixes == PARITY_STRIPPED_PREFIXES
2366                && comparison.canonicalizations == PARITY_CANONICALIZATIONS
2367        }
2368    };
2369    if !policy_shape_matches
2370        || !comparison.record_envelope.is_canonical()
2371        || comparison.ignore_lines
2372        || comparison.skip_commit
2373        || comparison.skip_detlog
2374    {
2375        return None;
2376    }
2377
2378    let mut relaxations = Vec::new();
2379    if comparison.strictness == LogCompareStrictness::Stripped {
2380        relaxations.push(STRIP_UNSAFE_NORMALIZATION_V1);
2381    }
2382    if !comparison.compare_io_buffers {
2383        relaxations.push("--no-detlog-io-buffers");
2384    }
2385    Some(if relaxations.is_empty() {
2386        "none".to_owned()
2387    } else {
2388        relaxations.join(",")
2389    })
2390}

2440/// The phrase that claims bitwise parity on the console. It is printed only

2447/// leaves uncovered.
2448pub(crate) static CPU_PROFILE_V3_RAN: std::sync::atomic::AtomicBool =

2451/// Write the announcement for `outcome` to `out`.
2452///
2453/// On a match, the caller's success message is followed by a line that states
2454/// what the match rests on. The success message is kept byte-for-byte as the
2455/// callers pass it, because tests and harnesses in this repository and in the
2456/// parent workspace match `Success: deterministic. Determinism verified.` as a
2457/// substring and, in at least one place, as an exact line. That message is
2458/// printed for the lossy `Stripped` comparison of a plain `--verify` as well as
2459/// for the canonical comparison of `--verify-strict`, so on its own it cannot
2460/// tell a reader which one ran.
2461fn write_verification_announcement(
2462    out: &mut impl io::Write,
2463    outcome: &VerificationOutcome,
2464    second_run: SecondRun,
2465    success_message: &str,
2466    failure_message: &str,
2467) -> io::Result<()> {
2468    // Announce the comparison only after any requested machine-readable report
2469    // has been durably published. Output-only fallbacks did not compare an
2470    // internal log and therefore must not claim a comparison policy.
2471    if let Some(evidence) = comparison_evidence_line(&outcome.comparison) {
2472        writeln!(out, "{evidence}")?;
2473    }
2474    match outcome.verdict {
2475        Verdict::Matched => {
2476            writeln!(out, ":: {}", success_message.green().bold())?;
2477            writeln!(
2478                out,
2479                ":: {}",
2480                matched_comparison_qualifier(outcome, second_run)
2481            )?;
2482            // What a match under x86-64-v3 does not cover
2483            // (https://github.com/rrnewton/hermit/issues/4243).
2484            if CPU_PROFILE_V3_RAN.load(std::sync::atomic::Ordering::SeqCst) {
2485                writeln!(out, ":: {}", detcore_model::cpu_profile::V3_RESIDUAL)?;
2486            }
2487        }
2488        Verdict::Diverged => writeln!(out, ":: {}", failure_message.red().bold())?,
2489        Verdict::NoResult => writeln!(
2490            out,
2491            ":: {}",
2492            "No result: the comparison was refused, so this is neither a match nor a difference."
2493                .red()
2494                .bold()
2495        )?,
2496        Verdict::InfrastructureError => writeln!(
2497            out,
2498            ":: {}",
2499            "Infrastructure error: the result is neither a match nor a product difference."
2500                .red()
2501                .bold()
2502        )?,
2503    }
2504    Ok(())
2505}

2507/// Say, in plain words, what a `Matched` verdict compared.
2508///
2509/// The bitwise claim is derived from [`verification_report`], the same
2510/// predicate that sets `bitwise_parity` in `--verify-json`, so the console and
2511/// the machine-readable report cannot disagree. Stdout and stderr are always
2512/// compared byte for byte by [`compare_two_runs`]; the guest's log records are
2513/// compared only when `compare_logs` is set, under the spec's strictness and
2514/// only within its `log_scope`. When time was not virtualized, the line also
2515/// says what that leaves unshown, which depends on `second_run`.
2516fn matched_comparison_qualifier(outcome: &VerificationOutcome, second_run: SecondRun) -> String {
2517    let comparison = &outcome.comparison;
2518    let scope = match comparison.log_scope {
2519        ComparedLogScope::Deterministic => {
2520            "only the DETLOG and scheduler COMMIT records of the guest's log"
2521        }
2522        ComparedLogScope::Info => "the INFO records of the guest's log",
2523        ComparedLogScope::FullTrace => "every record of the guest's log",
2524    };
2525    let records = outcome.compared_log_messages.map_or_else(
2526        || scope.to_owned(),
2527        |counts| format!("{scope} ({} vs {})", counts.left, counts.right),
2528    );
2529    let compared = if verification_report(outcome).bitwise_parity {
2530        format!(
2531            "Compared: stdout and stderr byte for byte, and {records} exactly apart from \
2532             the wall-clock prefix and canonicalized addresses, under {}: \
2533             {BITWISE_PARITY_CLAIM}.",
2534            comparison.display_name
2535        )
2536    } else if !comparison.compare_logs {
2537        "Compared: stdout and stderr only. The guest's log records were not compared, \
2538         so this is not a bitwise comparison."
2539            .to_owned()
2540    } else if comparison.strictness == LogCompareStrictness::Stripped {
2541        format!(
2542            "Compared: stdout and stderr byte for byte, and {records} under the lossy \
2543             Stripped comparison, which erases numbers, addresses, paths and timestamps \
2544             before comparing. This is not a bitwise comparison; add --verify-strict \
2545             for one."
2546        )
2547    } else {
2548        format!(
2549            "Compared: stdout and stderr byte for byte, and {records} under {}, but \
2550             without all the evidence the parity contract requires (see \
2551             `bitwise_parity` in --verify-json). This is not a bitwise comparison.",
2552            comparison.display_name
2553        )
2554    };
2555    match (comparison.virtualize_time, second_run) {
2556        (true, _) => compared,
2557        (false, SecondRun::Replay) => format!("{compared} {TIME_NOT_VIRTUALIZED_NOTE}"),
2558        (false, SecondRun::Rerun) => format!("{compared} {RERUN_TIME_NOT_VIRTUALIZED_NOTE}"),
2559    }
2560}

3408    /// Run the test named `test` in a new process of this test binary, with
3409    /// descriptor 2 a copy of `stderr`, the write end of a pipe. That test
3410    /// starts with [`retention_child_entry`], so the child calls
3411    /// `retain_logs_after_verification_error` with the `primary` error for
3412    /// one log it creates in `dir`. Wait for it for at most
3413    /// [`RETENTION_CHILD_BOUND`].
3414    ///
3415    /// The child runs with `--nocapture`. A forked child would inherit
3416    /// libtest's output capture when this test runs under `cargo test`
3417    /// without `--nocapture`, and its `eprint!` would then never reach its
3418    /// descriptor 2.
3419    ///
3420    /// The child keeps SIGPIPE ignored, as hermit's outer process does: the
3421    /// Rust runtime ignores it at startup and hermit never restores it. A
3422    /// write to a pipe without a reader therefore fails with EPIPE, which
3423    /// `eprintln!` turns into a panic. Unit tests never enable the exit
3424    /// bound that `main` arms for a capped run
3425    /// (`tracing::bound_log_cap_exit`), so what this observes is the report
3426    /// itself, not that bound covering for it.
3427    fn retention_report_in_child(
3428        test: &str,
3429        primary: &str,
3430        stderr: &std::os::fd::OwnedFd,
3431        dir: &Path,
3432    ) -> RetentionChild {
3433        use std::os::unix::process::ExitStatusExt as _;
3434        // The Command, and with it this process's copy of `stderr`, is
3435        // dropped at the end of this statement.
3436        let mut child = std::process::Command::new(std::env::current_exe().unwrap())
3437            .args([test, "--exact", "--nocapture", "--test-threads=1"])
3438            .env(RETENTION_CHILD_PRIMARY, primary)
3439            .env(RETENTION_CHILD_DIR, dir)
3440            .stdin(std::process::Stdio::null())
3441            .stdout(std::process::Stdio::null())
3442            .stderr(stderr.try_clone().unwrap())
3443            .spawn()
3444            .unwrap();
3445        let start = std::time::Instant::now();
3446        let status = loop {
3447            if let Some(status) = child.try_wait().unwrap() {
3448                break status;
3449            }
3450            if start.elapsed() >= RETENTION_CHILD_BOUND {
3451                child.kill().unwrap();
3452                child.wait().unwrap();
3453                return RetentionChild::StillRunning;
3454            }
3455            std::thread::sleep(std::time::Duration::from_millis(10));
3456        };
3457        assert_ne!(
3458            status.code(),
3459            Some(RETENTION_CHILD_OUTPUT_CAPTURED),
3460            "eprint! in the child did not reach its descriptor 2"
3461        );
3462        match status.code() {
3463            Some(RETENTION_CHILD_RETURNED) => RetentionChild::Returned,
3464            Some(101) => RetentionChild::Panicked,
3465            _ => RetentionChild::Other(status.into_raw()),
3466        }
3467    }

3469    /// The body of the child of [`retention_report_in_child`]; returns its
3470    /// exit code.
3471    fn retention_report_child_body(primary: fn() -> Error, dir: &Path) -> i32 {
3472        // SAFETY: the calls below only change this process's SIGPIPE
3473        // disposition and descriptors, and read into a local buffer. The
3474        // test harness's other thread is waiting for this test and writes
3475        // nothing to descriptor 2 meanwhile.
3476        unsafe {
3477            libc::signal(libc::SIGPIPE, libc::SIG_IGN);
3478            let stderr = libc::fcntl(libc::STDERR_FILENO, libc::F_DUPFD_CLOEXEC, 3);
3479            let mut probe = [0; 2];
3480            if stderr < 0
3481                || libc::pipe2(probe.as_mut_ptr(), libc::O_NONBLOCK | libc::O_CLOEXEC) != 0
3482                || libc::dup2(probe[1], libc::STDERR_FILENO) < 0
3483            {
3484                return 6;
3485            }
3486            eprint!("probe");
3487            let mut seen = [0u8; 8];
3488            if libc::read(probe[0], seen.as_mut_ptr().cast(), seen.len()) != 5
3489                || &seen[..5] != b"probe"
3490            {
3491                return RETENTION_CHILD_OUTPUT_CAPTURED;
3492            }
3493            if libc::dup2(stderr, libc::STDERR_FILENO) < 0 {
3494                return 6;
3495            }
3496        }
3497        let Ok(log) = tempfile::NamedTempFile::new_in(dir) else {
3498            return 6;
3499        };
3500        let log = log.into_temp_path();
3501        let path = log.to_path_buf();
3502        let class = super::super::failure_exit_code(&primary());
3503        match std::panic::catch_unwind(std::panic::AssertUnwindSafe(|| {
3504            retain_logs_after_verification_error(primary(), [("run 1", log)])
3505        })) {
3506            Err(_) => 101,
3507            Ok(error) if super::super::failure_exit_code(&error) != class => 3,
3508            // A TempPath that was not kept deletes its file when dropped.
3509            Ok(_) if !path.exists() => 4,
3510            Ok(_) => RETENTION_CHILD_RETURNED,
3511        }
3512    }

5591    // as true bitwise parity ONLY under a full-INFO, unstripped, named canonical
5592    // record envelope and reject every weaker or opaque policy. This brackets
5593    // both sides: each canonical envelope fires, while stripped, output-only,
5594    // caller-defined, and ad hoc ignore/skip variants are refused.
5595    /// A GREEN REPORT SAYS WHAT IT COMPARED AND WHETHER TIME WAS VIRTUAL — AND A
5596    /// REPORT THAT COMPARED NOTHING SAYS NEITHER.
5597    ///
5598    /// Ported from the residual of hermit#2269. `bitwise_parity: true` meant two
5599    /// different things depending on the subcommand and the report could not tell
5600    /// them apart: `run --verify` executes the guest twice with virtual time on,
5601    /// so a match is evidence about the guest, while `record start --verify` runs
5602    /// with `virtualize_time: false`, so a match says only that the replay
5603    /// reproduced that recording. The prose in `record start --help` says this;
5604    /// until now the ARTIFACT did not.
5605    ///
5606    /// ⚠️ THE FIRST ASSERTION IS THE POINT: parity and virtual time are
5607    /// INDEPENDENT. A comparison can qualify for bitwise parity with
5608    /// `virtualize_time: false`, and that combination is exactly the one a
5609    /// consumer must not read as a determinism result.
5610    ///
5611    /// The second half is why the field lives on `ComparisonSpec` and not on the
5612    /// report: `comparison` is `None` for `NoResult`, so a run that compared
5613    /// nothing cannot publish a time policy describing a comparison that did not
5614    /// happen. Putting it on the report would have re-created, one field over, the
5615    /// "value present, meaning absent" defect that hermit#2498's typed
