9//! OCI image rootfs materialization for `hermit run --image`.
10//!
11//! This is the *filesystem half* of hermit-as-container-runtime. Hermit already
12//! owns the *namespace half* (PID/mount/user namespaces via reverie-process);
13//! this module bolts on deterministic FILE INPUTS by materializing a rootfs from
14//! a pinned OCI image (ideally by digest) and letting `run` chroot into it.
15//!
16//! Podman is the image manager: images are fetched into the local
17//! containers/storage store by `podman pull` (or `hermit oci pull`), and this
18//! module materializes a rootfs by copying the merged overlay layers out of
19//! that store (see [`crate::app::podman_store`]). Hermit never pulls on the run
20//! path; a reference the store does not hold is refused with the remedy.
21

31/// Pseudo-filesystem mount points the guest expects to exist inside the root but
32/// that an image layer often omits (busybox, distroless, and `scratch`-derived
33/// images ship no `/proc` or `/sys`). We create them so reverie can mount the
34/// deterministic `/proc` into the chrooted root.
35///
36/// `dev/pts` is here rather than in [`DEV_BIND_TARGETS`] because it receives a
37/// whole filesystem (a fresh `devpts` instance), not a single bind.
38const REQUIRED_DIRS: &[&str] = &["proc", "sys", "dev", "dev/pts", "tmp"];

40/// Character devices the guest gets as bind mounts of the host node, relative to
41/// the image root. See `container::image_container` for the mounts themselves and
42/// for why each is safe to expose; this list only has to guarantee the mount
43/// *targets* exist, because the image root is remounted read-only before the
44/// binds are applied and a bind mount cannot create its own target.
45///
46/// An unprivileged user namespace cannot `mknod` a character device, so binding
47/// the host node is the only way to give the guest one. That is also what
48/// rootless podman and runc do.
49pub(super) const DEV_BIND_TARGETS: &[&str] = &["null", "zero", "full", "random", "urandom"];

51/// Extra device placeholder created in every rootfs but bound only for the
52/// KVM backend (whose VMM process opens `/dev/kvm` after chroot). For other

54pub(super) const KVM_DEV_TARGETS: &[&str] = &["kvm"];

56/// Conventional `/dev` symlinks, as `(link, target)`. These need no privilege and
57/// no mount: they resolve through `/proc`, which Hermit already replaces with its
58/// deterministic instance inside the image root.
59///
60/// `ptmx` completes the fresh `devpts` instance mounted at `/dev/pts`; the
61/// kernel's devpts documentation names exactly this symlink (or a bind) as the
62/// supported way to expose a per-instance `ptmx`. `fd`, `stdin`, `stdout` and
63/// `stderr` are what bash process substitution (`<(...)`) and countless build
64/// scripts open.
65const DEV_SYMLINKS: &[(&str, &str)] = &[
66    ("ptmx", "pts/ptmx"),
67    ("fd", "/proc/self/fd"),
68    ("stdin", "/proc/self/fd/0"),
69    ("stdout", "/proc/self/fd/1"),
70    ("stderr", "/proc/self/fd/2"),
71];

73/// Version of the materialized-rootfs *layout* (which mount targets and symlinks
74/// the tree must contain), mixed into the cache key.
75///
76/// A cached tree is reused whenever its readiness marker exists, so a tree
77/// materialized by an older Hermit would silently lack the `/dev` targets added
78/// here and the binds would fail at mount time. Versioning the key
79/// re-materializes instead. The key is used rather than the marker because an
80/// image root is often mode `0555` and can carry entries owned by unmapped
81/// sub-UIDs, so hermit cannot reliably rewrite or delete a tree it already
82/// produced -- but it can always create a new one beside it. Superseded trees are
83/// left on disk; clearing `~/.cache/hermit/oci-rootfs` reclaims them.
84const ROOTFS_LAYOUT_VERSION: &str = "rootfs-layout-v4-dev";

93/// Reduce an image reference to the identity the rootfs cache is keyed by.
94///
95/// A tag is a mutable pointer: keying the cache by the raw reference string
96/// lets a moved tag silently alias an older rootfs. When a local podman store
97/// is available the reference is therefore resolved to the store's canonical
98/// image ID first (a local-only lookup that never contacts a registry), so
99/// `podman pull` moving a tag produces a fresh cache entry. References that
100/// are already an image ID, and references the store cannot resolve (the
101/// store cannot resolve it), fall back to the reference itself.
102pub(crate) fn canonical_image_ref(image_ref: &str) -> String {
103    // Local image inputs (oci-dir:/docker-archive: with a rooted path) and
104    // bare-relative prefixed values are never store references: they key the
105    // cache by their verified content identity in `cache_dir_for`, and the
106    // podman probe below must never run for them.
107    if !matches!(
108        crate::app::image_source::parse_image_source(image_ref),
109        crate::app::image_source::ImageSourceRef::StoreReference
110    ) {
111        return image_ref.to_string();
112    }
113    let candidate = image_ref
114        .trim()
115        .strip_prefix("sha256:")
116        .unwrap_or(image_ref.trim());
117    if candidate.len() == 64 && candidate.chars().all(|c| c.is_ascii_hexdigit()) {
118        return candidate.to_ascii_lowercase();
119    }
120    if let Ok(Some(store)) = crate::app::podman_store::PodmanStore::probe()
121        && let Ok(resolved) = store.resolve(image_ref)
122    {
123        return resolved.id;
124    }
125    image_ref.to_string()
126}

128/// A SHA-256 digest explicitly pinned in a store reference, without the
129/// algorithm prefix. Accepts `name@sha256:<hex>`, `sha256:<hex>` and a bare
130/// 64-hex image ID.
131fn pinned_sha256(image_ref: &str) -> Option<String> {
132    let candidate = image_ref.trim().rsplit_once("@sha256:").map_or_else(
133        || {
134            image_ref
135                .trim()
136                .strip_prefix("sha256:")
137                .unwrap_or(image_ref.trim())
138        },
139        |(_, digest)| digest,
140    );
141    (candidate.len() == 64 && candidate.chars().all(|c| c.is_ascii_hexdigit()))
142        .then(|| candidate.to_ascii_lowercase())
143}

145/// Convert a store resolution into the identity a manifest records,
146/// refusing a pinned reference that does not name the resolved content.
147/// A `name@sha256:` pin is accepted when it equals the selected image ID,
148/// the selected platform-manifest digest, or one of the store's recorded
149/// repo digests (the multi-arch index case). A bare ID/`sha256:<id>` pin
150/// must equal the selected image ID.
151fn image_from_resolved(
152    image_ref: &str,
153    resolved: &crate::app::podman_store::ResolvedImage,
154) -> Result<crate::app::manifest::Image, Error> {
155    use crate::app::manifest::Image;
156    use crate::app::manifest::ImageSource;
157
158    if let Some(pinned) = pinned_sha256(image_ref) {
159        let selected_manifest = resolved
160            .digest
161            .as_deref()
162            .and_then(|digest| digest.strip_prefix("sha256:"));
163        let repo_matches = resolved.repo_digests.iter().any(|repo_digest| {
164            repo_digest
165                .rsplit_once("@sha256:")
166                .is_some_and(|(_, digest)| digest.eq_ignore_ascii_case(&pinned))
167        });
168        let pinned_is_manifest_reference = image_ref.contains("@sha256:");
169        let matches = if pinned_is_manifest_reference {
170            pinned == resolved.id
171                || selected_manifest.is_some_and(|digest| digest.eq_ignore_ascii_case(&pinned))
172                || repo_matches
173        } else {
174            pinned == resolved.id
175        };
176        if !matches {
177            return Err(Error::msg(format!(
178                "image {image_ref:?} pins sha256:{pinned}, but the local store selected                  image id {} with manifest digest {}; refusing to record an                  unverified image identity in the run manifest",
179                resolved.id,
180                resolved.digest.as_deref().unwrap_or("<none>")
181            )));
182        }
183    }
184    Ok(Image {
185        source: ImageSource::PodmanStore,
186        reference: Some(resolved.reference.clone()),
187        identity: resolved.id.clone(),
188        manifest_digest: resolved.digest.clone(),
189        image_id: Some(resolved.id.clone()),
190    })
191}

193/// Convert a loaded local image's verified namespaced identity into the
194/// manifest representation.
195fn image_from_local_identity(
196    kind: crate::app::image_source::LocalImageKind,
197    identity: &str,
198) -> Result<crate::app::manifest::Image, Error> {
199    use crate::app::image_source::LocalImageKind;
200    use crate::app::manifest::Image;
201    use crate::app::manifest::ImageSource;
202
203    let digest = identity
204        .rsplit_once("sha256:")
205        .map(|(_, digest)| digest.to_owned())
206        .filter(|digest| {
207            digest.len() == 64
208                && digest
209                    .bytes()
210                    .all(|b| b.is_ascii_digit() || (b'a'..=b'f').contains(&b))
211        })
212        .ok_or_else(|| {
213            Error::msg(format!(
214                "local image identity {identity:?} does not carry a lowercase sha256 digest"
215            ))
216        })?;
217    Ok(match kind {
218        LocalImageKind::OciDir => Image {
219            source: ImageSource::OciDir,
220            reference: None,
221            identity: digest.clone(),
222            manifest_digest: Some(format!("sha256:{digest}")),
223            image_id: None,
224        },
225        LocalImageKind::DockerArchive => Image {
226            source: ImageSource::DockerArchive,
227            reference: None,
228            identity: digest.clone(),
229            manifest_digest: None,
230            image_id: Some(digest),
231        },
232    })
233}

235/// Resolve an `--image` value to the content identity a run manifest
236/// records. Store references must resolve successfully in the local store;
237/// a digest spelled in the reference is only a cross-check, never a
238/// substitute for resolution against actual local content.
239pub(crate) fn resolved_image_identity(
240    image_ref: &str,
241) -> Result<crate::app::manifest::Image, Error> {
242    use crate::app::image_source::ImageSourceRef;
243
244    match crate::app::image_source::parse_image_source(image_ref) {
245        ImageSourceRef::Local { kind, path } => {
246            let identity = crate::app::image_source::local_identity(kind, &path)?;
247            image_from_local_identity(kind, &identity)
248        }
249        ImageSourceRef::BareRelative { kind, remainder } => Err(
250            crate::app::image_source::bare_relative_error(kind, &remainder),
251        ),
252        ImageSourceRef::StoreReference => {
253            let store = crate::app::podman_store::PodmanStore::probe()?
254                .ok_or_else(|| {
255                    Error::msg(format!(
256                        "cannot resolve image {image_ref:?} for the run manifest:                          no local podman store is available"
257                    ))
258                })?;
259            let resolved = store.resolve(image_ref)?;
260            image_from_resolved(image_ref, &resolved)
261        }
262    }
263}

265/// Compute the cache directory that holds materialized rootfs trees. Keyed by
266/// the SHA-256 of the complete image reference so distinct references cannot
267/// alias after filesystem sanitization.
268fn rootfs_cache_dir(image_ref: &str) -> Result<PathBuf, Error> {
269    let base = if let Ok(xdg) = std::env::var("XDG_CACHE_HOME") {
270        PathBuf::from(xdg)
271    } else if let Ok(home) = std::env::var("HOME") {
272        PathBuf::from(home).join(".cache")
273    } else {
274        std::env::temp_dir()
275    };
276    // NUL-separate the two fields so no image reference can collide with a
277    // different (reference, layout) pair by concatenation.
278    let mut keyed = Vec::from(image_ref.as_bytes());
279    keyed.push(0);
280    keyed.extend_from_slice(ROOTFS_LAYOUT_VERSION.as_bytes());
281    let key = Digest::new(&keyed).to_string();
282    Ok(base.join("hermit").join("oci-rootfs").join(key))
283}

285/// The cache directory for a verified local-image identity
286/// (`oci-dir@sha256:...` / `docker-archive@sha256:...`). The key input is
287/// domain-separated with a NUL-delimited prefix: a command-line store
288/// reference cannot contain a NUL byte, so no raw store reference string
289/// (for example the literal `oci-dir@sha256:...`) can ever alias a local
290/// key. Store keys and the layout version mixed in by `rootfs_cache_dir`
291/// are unchanged.
292fn local_rootfs_cache_dir(identity: &str) -> Result<PathBuf, Error> {
293    rootfs_cache_dir(&format!("\0hermit-local-image\0{identity}"))
294}
295
296/// The cache directory for any `--image` value. Store references keep the
297/// canonical-ref keying above; local image inputs key by their verified,
298/// domain-separated content identity (see [`local_rootfs_cache_dir`]) so
299/// two paths holding the same image share a tree and a changed image can
300/// never alias an older one.
301fn cache_dir_for(image_ref: &str) -> Result<PathBuf, Error> {
302    use crate::app::image_source::ImageSourceRef;
303    match crate::app::image_source::parse_image_source(image_ref) {
304        ImageSourceRef::StoreReference => rootfs_cache_dir(&canonical_image_ref(image_ref)),
305        ImageSourceRef::Local { kind, path } => {
306            local_rootfs_cache_dir(&crate::app::image_source::local_identity(kind, &path)?)
307        }
308        ImageSourceRef::BareRelative { kind, remainder } => Err(
309            crate::app::image_source::bare_relative_error(kind, &remainder),
310        ),
311    }
312}

314/// Basenames under `base` whose readiness marker records a local-image
315/// identity. Local materializations are identified by that marker content
316/// and treated as live by prune, since no store listing can name them.
317fn local_live_keys_in(base: &Path) -> std::collections::BTreeSet<String> {
318    let mut keys = std::collections::BTreeSet::new();
319    let Ok(entries) = std::fs::read_dir(base) else {
320        return keys;
321    };
322    for entry in entries.flatten() {
323        let Ok(marker) = std::fs::read_to_string(ready_marker(&entry.path())) else {
324            continue;
325        };
326        let marker = marker.trim();
327        if marker.starts_with("oci-dir@sha256:") || marker.starts_with("docker-archive@sha256:") {
328            keys.insert(entry.file_name().to_string_lossy().into_owned());
329        }
330    }
331    keys
332}

334/// Basenames of cache entries still referenced by `live_refs`, for GC.
335/// Entries whose marker identifies a local image are live as well: prune
336/// has no store listing that could name them, and deleting a local cache
337/// merely forces a re-materialization from a source path a later run may
338/// still hold.
339pub(crate) fn live_cache_keys(
340    live_refs: &[String],
341) -> Result<std::collections::BTreeSet<String>, Error> {
342    let mut keys = std::collections::BTreeSet::new();
343    for r in live_refs {
344        let dir = rootfs_cache_dir(&canonical_image_ref(r))?;
345        if let Some(name) = dir.file_name().and_then(|n| n.to_str()) {
346            keys.insert(name.to_string());
347        }
348    }
349    if let Some(base) = cache_base()? {
350        keys.extend(local_live_keys_in(&base));
351    }
352    Ok(keys)
353}

355/// The oci-rootfs cache base directory, when it exists.
356pub(crate) fn cache_base() -> Result<Option<PathBuf>, Error> {
357    // Any ref works: only the parent of the keyed dir is wanted.
358    let probe = rootfs_cache_dir("probe")?;
359    Ok(probe
360        .parent()
361        .map(|p| p.to_path_buf())
362        .filter(|p| p.is_dir()))
363}

365/// A rootfs is considered already materialized if the marker file is present.
366/// The marker is written last, so its presence means the copy finished.
367///
368/// The marker lives in the cache directory *beside* `rootfs`, not inside it: an
369/// image root is frequently mode `0555` (read-only) and can contain entries
370/// owned by unmapped sub-UIDs, so hermit (running as the plain invoking user,
371/// outside any user namespace) cannot reliably write into the rootfs after
372/// materialization. The cache directory itself is created and owned by hermit,
373/// so the marker is always writable there.
374fn ready_marker(cache: &Path) -> PathBuf {
375    cache.join(".hermit-oci-ready")
376}

388/// The subset of an image's OCI runtime config that image mode applies so the
389/// guest sees the deterministic, digest-pinned environment the image declares.
390#[derive(Default)]
391pub(crate) struct ImageConfig {
392    /// `Config.Env` entries, split into `(key, value)` pairs.
393    pub env: Vec<(String, String)>,
394    /// `Config.WorkingDir`, if the image set a non-empty one.
395    pub workdir: Option<String>,
396    /// `Config.Cmd` (one argument per persisted line).
397    pub cmd: Vec<String>,
398    /// `Config.Entrypoint` (one argument per persisted line).
399    pub entrypoint: Vec<String>,
400}
401
402fn read_image_config_files(
403    env_path: &Path,
404    workdir_path: &Path,
405) -> Result<Option<ImageConfig>, Error> {
406    let env_exists = env_path.try_exists().with_context(|| {
407        format!(
408            "Failed to inspect OCI environment file {}",
409            env_path.display()
410        )
411    })?;
412    let workdir_exists = workdir_path.try_exists().with_context(|| {
413        format!(
414            "Failed to inspect OCI working-directory file {}",
415            workdir_path.display()
416        )
417    })?;
418    if !env_exists && !workdir_exists {
419        return Ok(None);
420    }
421
422    let env = if env_exists {
423        parse_env(&std::fs::read_to_string(env_path).with_context(|| {
424            format!("Failed to read OCI environment file {}", env_path.display())
425        })?)
426    } else {
427        Vec::new()
428    };
429    let workdir = if workdir_exists {
430        parse_workdir(&std::fs::read_to_string(workdir_path).with_context(|| {
431            format!(
432                "Failed to read OCI working-directory file {}",
433                workdir_path.display()
434            )
435        })?)
436    } else {
437        None
438    };
439    Ok(Some(ImageConfig {
440        env,
441        workdir,
442        ..Default::default()
443    }))
444}

446fn read_argv_file(path: &Path) -> Vec<String> {
447    std::fs::read_to_string(path)
448        .map(|text| {

453        })

455}

457/// Read the persisted OCI config (Env + WorkingDir) captured at materialization
458/// time. Missing files yield empty/None so images that declare neither still
459/// run. File *presence*, rather than a non-empty Env, selects the location: an
460/// image may legitimately declare a WorkingDir and no Env.
461///
462/// This is read from two candidate locations, in order, and the first that
463/// yields a non-empty `Env` wins:
464///
465/// 1. The host cache dir beside the rootfs — valid when `guest_command` runs
466///    *before* the chroot (e.g. program validation).
467/// 2. The absolute in-root path (`/.hermit-oci-env`) — valid when
468///    `guest_command` runs *inside* the container, after `chroot(rootfs)`, where
469///    the host cache path (an absolute host path) no longer resolves. The
470///    materializer writes a copy of the config here for exactly this case.
471///
472/// The two-location read is what makes the image-declared environment reach the
473/// guest: the environment that actually runs the program is built in the forked,
474/// chrooted child, so it can only see files that live inside the rootfs.
475pub(crate) fn read_image_config(image_ref: &str) -> Result<ImageConfig, Error> {
476    // A local image's cache dir derives from its source files; inside the
477    // chroot those host paths no longer resolve, so a lookup failure here
478    // falls through to the in-root copy instead of failing the run. A
479    // malformed local source is still refused by name at materialization.
480    if let Ok(cache) = cache_dir_for(image_ref)
481        && let Some(mut host) = read_image_config_files(&env_file(&cache), &workdir_file(&cache))?
482    {
483        host.cmd = read_argv_file(&cache.join(CMD_BASENAME));
484        host.entrypoint = read_argv_file(&cache.join(ENTRYPOINT_BASENAME));
485        return Ok(host);
486    }
487    // Fall back to the in-root copy (post-chroot view).
488    if let Some(mut in_root) = read_image_config_files(
489        &Path::new("/").join(ENV_BASENAME),
490        &Path::new("/").join(WORKDIR_BASENAME),
491    )? {
492        in_root.cmd = read_argv_file(&Path::new("/").join(CMD_BASENAME));
493        in_root.entrypoint = read_argv_file(&Path::new("/").join(ENTRYPOINT_BASENAME));
494        return Ok(in_root);
495    }
496    Ok(ImageConfig::default())
497}

499fn parse_env(contents: &str) -> Vec<(String, String)> {
500    contents
501        .lines()
502        .filter_map(|line| {
503            let line = line.trim_end_matches('\r');
504            if line.is_empty() {
505                return None;
506            }
507            line.split_once('=')
508                .map(|(k, v)| (k.to_string(), v.to_string()))
509        })
510        .collect()
511}

532/// Refuse a materialized tree whose content identity differs from the
533/// identity a run manifest recorded. An absent expected identity means the
534/// run made no manifest claim and imposes no check.
535pub(crate) fn ensure_materialized_identity_matches(
536    image_ref: &str,
537    expected: Option<&crate::app::manifest::Image>,
538    actual: Option<&crate::app::manifest::Image>,
539) -> Result<(), Error> {
540    let Some(expected) = expected else {
541        return Ok(());
542    };
543    let actual = actual.ok_or_else(|| {
544        Error::msg(format!(
545            "image {image_ref:?} materialized without a resolved content identity, \
546             so it cannot be bound to the identity recorded in this run's manifest"
547        ))
548    })?;
549    if expected.same_content_identity(actual) {
550        Ok(())
551    } else {
552        Err(Error::msg(format!(
553            "image {image_ref:?} materialized as {actual:?}, which does not match \
554             the image identity recorded in this run's manifest ({expected:?}); \
555             refusing to run different content than the manifest names"
556        )))
557    }
558}

560/// Materialize the given OCI image reference into a plain, user-owned rootfs
561/// directory and return it with its resolved identity. Reusing a previously
562/// materialized reference is a no-op that returns the cached directory.
563///
564/// `image_ref` must already be in the local podman store, e.g.
565/// `docker.io/library/busybox@sha256:...` (digest-pinned, recommended for
566/// determinism) or `docker.io/library/busybox:latest`, or name a local image
567/// input — `oci-dir:/path` or `docker-archive:/path.tar` with an absolute or
568/// `./`-/`../`-relative path (see [`crate::app::image_source`]).
569pub(crate) fn materialize_rootfs(image_ref: &str) -> Result<MaterializedRootfs, Error> {
570    use crate::app::image_source::ImageSourceRef;
571    match crate::app::image_source::parse_image_source(image_ref) {
572        ImageSourceRef::Local { kind, path } => {
573            return materialize_via_local_source(kind, &path);
574        }
575        ImageSourceRef::BareRelative { kind, remainder } => {
576            return Err(crate::app::image_source::bare_relative_error(
577                kind, &remainder,
578            ));
579        }
580        ImageSourceRef::StoreReference => {}
581    }
582    // Resolve once and use that exact resolution for both the cache key
583    // and the layer copy below, so the identity returned here is the
584    // identity of the tree being executed. A failed resolution preserves
585    // the legacy cache-hit fallback, but yields no identity.
586    let resolved_store = match crate::app::podman_store::PodmanStore::probe()? {
587        Some(store) => store
588            .resolve(image_ref)
589            .ok()
590            .map(|resolved| (store, resolved)),
591        None => None,
592    };
593    let identity = resolved_store
594        .as_ref()
595        .map(|(_, resolved)| image_from_resolved(image_ref, resolved))
596        .transpose()?;
597    let cache = match &resolved_store {
598        Some((_, resolved)) => rootfs_cache_dir(&resolved.id)?,
599        None => rootfs_cache_dir(&canonical_image_ref(image_ref))?,
600    };
601    let rootfs = cache.join("rootfs");
602    if ready_marker(&cache).exists() {
603        tracing::info!(
604            "Reusing cached OCI rootfs for {} at {}",
605            image_ref,
606            rootfs.display()
607        );
608        return Ok(MaterializedRootfs {
609            path: rootfs,
610            identity,
611        });
612    }
613
614    tracing::info!("Materializing OCI rootfs for {}", image_ref);
615    std::fs::create_dir_all(&rootfs)
616        .with_context(|| format!("Failed to create rootfs cache dir {}", rootfs.display()))?;
617
618    // Copy the merged tree out of the local podman store's overlay layers.
619    let materialized = match &resolved_store {
620        Some((store, resolved)) => materialize_via_podman_store(store, resolved, &cache, &rootfs)?,
621        None => false,
622    };
623    if materialized {
624        std::fs::write(ready_marker(&cache), image_ref.as_bytes())
625            .context("Failed to write OCI rootfs readiness marker")?;
626        return Ok(MaterializedRootfs {
627            path: rootfs,
628            identity,
629        });
630    }
631
632    Err(Error::msg(format!(
633        "image {image_ref} is not in the local podman store, so there is no \
634         rootfs to materialize. Fetch it first with `hermit oci pull {image_ref}` \
635         (or `podman pull {image_ref}`), then re-run."
636    )))
637}

639/// Copy the merged rootfs out of the podman store's overlay layers.
640/// Returns `Ok(false)` when no store holds the image. Layers apply
641/// base-first; overlay whiteouts are honored.
642fn materialize_via_podman_store(
643    store: &crate::app::podman_store::PodmanStore,
644    resolved: &crate::app::podman_store::ResolvedImage,
645    cache: &Path,
646    rootfs: &Path,
647) -> Result<bool, Error> {
648    let Ok(layers) = store.overlay_dirs(resolved) else {
649        return Ok(false);
650    };
651    if layers.is_empty() {
652        return Ok(false);
653    }
654    // overlay_dirs is topmost-first; overlay application is base-first.
655    for layer in layers.iter().rev() {
656        apply_overlay_layer(Path::new(layer), rootfs)?;
657    }
658    let config = store.config(resolved)?;
659    finish_materialized_rootfs(cache, rootfs, &config)?;
660    Ok(true)
661}

663/// Remove a cache tree, restoring writable directory modes first: a
664/// partial local materialization can contain read-only (`0555`) layer
665/// directories that plain `remove_dir_all` cannot empty as their owner
666/// without a chmod pass. Symlinks are never followed.
667fn remove_tree(path: &Path) {
668    fn make_writable(dir: &Path) {
669        use std::os::unix::fs::PermissionsExt;
670        let _ = std::fs::set_permissions(dir, std::fs::Permissions::from_mode(0o755));
671        if let Ok(entries) = std::fs::read_dir(dir) {
672            for entry in entries.flatten() {
673                let child = entry.path();
674                if let Ok(meta) = std::fs::symlink_metadata(&child)
675                    && meta.is_dir()
676                    && !meta.file_type().is_symlink()
677                {
678                    make_writable(&child);
679                }
680            }
681        }
682    }
683    match std::fs::symlink_metadata(path) {
684        Ok(meta) if meta.is_dir() && !meta.file_type().is_symlink() => {
685            make_writable(path);
686            let _ = std::fs::remove_dir_all(path);
687        }
688        Ok(_) => {
689            let _ = std::fs::remove_file(path);
690        }
691        Err(_) => {}
692    }
693}

695/// Materialize a local image input (`oci-dir:`/`docker-archive:`) into the
696/// rootfs cache, keyed by its verified content identity. Verification of
697/// every digest and diff_id happens in `image_source::load_local_image`
698/// before anything is applied; podman is never invoked on this path.
699///
700/// Materialization is atomic at cache granularity: layers apply into a
701/// per-identity, per-process staging directory beside the final cache
702/// entry, the readiness marker is written inside staging last, and the
703/// completed staging directory is renamed into place. An interrupted run
704/// therefore leaves only a staging directory — cleaned up on entry — and
705/// never a partial, possibly `0555`-locked tree at the final location.
706/// If a concurrent materializer publishes the same identity first, the
707/// completed tree is reused and the staging copy is discarded.
708fn materialize_via_local_source(
709    kind: crate::app::image_source::LocalImageKind,
710    path: &Path,
711) -> Result<MaterializedRootfs, Error> {
712    let loaded = crate::app::image_source::load_local_image(kind, path)?;
713    let identity = Some(image_from_local_identity(kind, &loaded.identity)?);
714    let cache = local_rootfs_cache_dir(&loaded.identity)?;
715    let rootfs = cache.join("rootfs");
716    if ready_marker(&cache).exists() {
717        tracing::info!(
718            "Reusing cached OCI rootfs for {} at {}",
719            loaded.identity,
720            rootfs.display()
721        );
722        return Ok(MaterializedRootfs {
723            path: rootfs,
724            identity,
725        });
726    }
727    let base = cache
728        .parent()
729        .map(Path::to_path_buf)
730        .ok_or_else(|| Error::msg("OCI rootfs cache dir has no parent".to_string()))?;
731    std::fs::create_dir_all(&base)
732        .with_context(|| format!("Failed to create rootfs cache dir {}", base.display()))?;
733    let key = cache
734        .file_name()
735        .map(|n| n.to_string_lossy().into_owned())
736        .unwrap_or_default();
737    // Clean up staging leftovers on entry: our own earlier staging, and
738    // staging from processes that no longer exist. A live concurrent
739    // materializer's staging (its pid still exists) is left alone.
740    let my_pid = std::process::id();
741    if let Ok(entries) = std::fs::read_dir(&base) {
742        let prefix = format!("{key}.staging-");
743        for entry in entries.flatten() {
744            let name = entry.file_name().to_string_lossy().into_owned();
745            let Some(pid_text) = name.strip_prefix(&prefix) else {
746                continue;
747            };
748            let stale = match pid_text.parse::<u32>() {
749                Ok(pid) => pid == my_pid || !Path::new(&format!("/proc/{pid}")).exists(),
750                Err(_) => false,
751            };
752            if stale {
753                remove_tree(&entry.path());
754            }
755        }
756    }
757    let staging = base.join(format!("{key}.staging-{my_pid}"));
758    let staging_rootfs = staging.join("rootfs");
759    let apply = (|| -> Result<(), Error> {
760        tracing::info!("Materializing OCI rootfs for {}", loaded.identity);
761        std::fs::create_dir_all(&staging_rootfs).with_context(|| {
762            format!(
763                "Failed to create rootfs staging dir {}",
764                staging_rootfs.display()
765            )
766        })?;
767        crate::app::image_source::apply_layers(&loaded, &staging_rootfs)?;
768        finish_materialized_rootfs(&staging, &staging_rootfs, &loaded.config)?;
769        std::fs::write(ready_marker(&staging), loaded.identity.as_bytes())
770            .context("Failed to write OCI rootfs readiness marker")?;
771        Ok(())
772    })();
773    if let Err(e) = apply {
774        remove_tree(&staging);
775        return Err(e);
776    }
777    // Publish. A complete tree that appeared meanwhile (concurrent
778    // materializer) wins; an incomplete final entry can only be a leftover
779    // from before staging existed, because this path never writes to the
780    // final location directly, so it is cleared to unblock the rename.
781    if ready_marker(&cache).exists() {
782        remove_tree(&staging);
783        return Ok(MaterializedRootfs {
784            path: rootfs,
785            identity,
786        });
787    }
788    if std::fs::symlink_metadata(&cache).is_ok() {
789        remove_tree(&cache);
790    }
791    match std::fs::rename(&staging, &cache) {
792        Ok(()) => Ok(MaterializedRootfs {
793            path: rootfs,
794            identity,
795        }),
796        Err(rename_err) => {
797            if ready_marker(&cache).exists() {
798                remove_tree(&staging);
799                Ok(MaterializedRootfs {
800                    path: rootfs,
801                    identity,
802                })
803            } else {
804                Err(Error::from(rename_err).context(format!(
805                    "Failed to move materialized rootfs for {} into place at {}",
806                    loaded.identity,
807                    cache.display()
808                )))
809            }
810        }
811    }
812}

814/// Ensure `rootfs/rel` exists as a chain of real directories, refusing by
815/// name if any component the image planted is a symlink: `create_dir_all`
816/// and later writes would follow it outside the rootfs. Legitimate trees
817/// (real directories or absent components) behave exactly as before.
818fn ensure_real_dir(rootfs: &Path, rel: &str) -> Result<PathBuf, Error> {
819    let mut current = rootfs.to_path_buf();
820    for part in rel.split('/').filter(|part| !part.is_empty()) {
821        current.push(part);
822        match std::fs::symlink_metadata(&current) {
823            Ok(meta) if meta.file_type().is_symlink() => {
824                return Err(Error::msg(format!(
825                    "refusing to write through symlinked directory {part:?} \
826                     (rootfs path {rel:?}): the image planted a symlink where \
827                     Hermit must create its own files, and following it would \
828                     write outside the image root. Re-export the image with \
829                     {rel:?} as a real directory."
830                )));
831            }
832            Ok(meta) if meta.is_dir() => {}
833            Ok(_) => {
834                return Err(Error::msg(format!(
835                    "rootfs path {rel:?} is not a directory in the image, so \
836                     Hermit cannot create its files there. Re-export the \
837                     image with {rel:?} as a directory."
838                )));
839            }
840            Err(e) if e.kind() == std::io::ErrorKind::NotFound => {
841                std::fs::create_dir(&current).with_context(|| {
842                    format!("Failed to create rootfs directory {}", current.display())
843                })?;
844            }
845            Err(e) => {
846                return Err(
847                    Error::from(e).context(format!("Failed to inspect {}", current.display()))
848                );
849            }
850        }
851    }
852    Ok(current)
853}

855/// Write `bytes` at `rootfs/rel` (a single basename or a path whose parents
856/// [`ensure_real_dir`] has already vetted) without ever following a
857/// planted symlink at the final component: a symlink there is replaced
858/// itself, and the file is created with `create_new` so nothing pre-existing
859/// is written through.
860fn write_in_root(rootfs: &Path, rel: &str, bytes: &[u8], what: &str) -> Result<(), Error> {
861    use std::io::Write;
862    let target = rootfs.join(rel);
863    match std::fs::symlink_metadata(&target) {
864        Ok(meta) if meta.is_dir() && !meta.file_type().is_symlink() => {
865            return Err(Error::msg(format!(
866                "refusing to replace directory {rel:?} in the image root \
867                 with {what}. Re-export the image without a directory there."
868            )));
869        }
870        Ok(_) => {
871            std::fs::remove_file(&target)
872                .with_context(|| format!("Failed to replace {}", target.display()))?;
873        }
874        Err(e) if e.kind() == std::io::ErrorKind::NotFound => {}
875        Err(e) => {
876            return Err(Error::from(e).context(format!("Failed to inspect {}", target.display())));
877        }
878    }
879    let mut file = std::fs::OpenOptions::new()
880        .write(true)
881        .create_new(true)
882        .open(&target)
883        .with_context(|| format!("Failed to write {what}"))?;
884    file.write_all(bytes)
885        .with_context(|| format!("Failed to write {what}"))
886}

888/// The finishing steps every materialized rootfs needs, whichever source
889/// produced the tree: mount points, `/etc/hostname`, device placeholder
890/// files and `/dev` symlinks, and the persisted image config (beside the
891/// rootfs for host-side reads, inside it for post-chroot reads). Shared by
892/// the podman-store path and the local image inputs in
893/// [`crate::app::image_source`] so both produce identical trees.
894///
895/// Every write into the tree is symlink-guarded (`lstat`, never
896/// write-through): an image that plants `etc`, `dev`, or a `.hermit-oci-*`
897/// name as a symlink to a host path cannot redirect these writes outside
898/// the rootfs. Required directories are refused by name when symlinked; a
899/// symlinked final component (device placeholder, `/dev` link, in-root
900/// config file) is replaced itself.
901fn finish_materialized_rootfs(
902    cache: &Path,
903    rootfs: &Path,
904    config: &crate::app::podman_store::ImageRunConfig,
905) -> Result<(), Error> {
906    if let Ok(meta) = std::fs::symlink_metadata(rootfs)
907        && meta.file_type().is_symlink()
908    {
909        return Err(Error::msg(format!(
910            "refusing to finish rootfs at {}: it is a symlink, and writing \
911             into it would escape the cache.",
912            rootfs.display()
913        )));
914    }
915    // Mount points, device placeholders, and symlinks, here as plain
916    // filesystem operations on our own cache.
917    {
918        use std::os::unix::fs::PermissionsExt;
919        let _ = std::fs::set_permissions(rootfs, std::fs::Permissions::from_mode(0o755));
920    }
921    for dir in REQUIRED_DIRS {
922        ensure_real_dir(rootfs, dir)
923            .with_context(|| format!("Failed to create rootfs mount point {dir}"))?;
924    }
925    // Container runtimes provide /etc/hostname; images often omit it. An
926    // existing entry of any kind (lstat, so a broken symlink counts) is
927    // the image's own and is left alone.
928    ensure_real_dir(rootfs, "etc")?;
929    let hostname_path = rootfs.join("etc/hostname");
930    if std::fs::symlink_metadata(&hostname_path).is_err() {
931        write_in_root(
932            rootfs,
933            "etc/hostname",
934            b"hermetic-container.local\n",
935            "rootfs /etc/hostname",
936        )?;
937    }
938    for node in DEV_BIND_TARGETS.iter().chain(KVM_DEV_TARGETS.iter()) {
939        write_in_root(
940            rootfs,
941            &format!("dev/{node}"),
942            b"",
943            &format!("device placeholder {node}"),
944        )?;
945    }
946    for (link, target) in DEV_SYMLINKS {
947        let link_path = rootfs.join("dev").join(link);
948        match std::fs::symlink_metadata(&link_path) {
949            Ok(meta) if meta.is_dir() && !meta.file_type().is_symlink() => {
950                return Err(Error::msg(format!(
951                    "refusing to replace directory dev/{link} in the image \
952                     root with a symlink. Re-export the image without a \
953                     directory there."
954                )));
955            }
956            Ok(_) => {
957                std::fs::remove_file(&link_path)
958                    .with_context(|| format!("Failed to replace /dev symlink {link}"))?;
959            }
960            Err(_) => {}
961        }
962        std::os::unix::fs::symlink(target, &link_path)
963            .with_context(|| format!("Failed to create /dev symlink {link}"))?;
964    }
965    // Persist the image config beside and inside the rootfs, as a container runtime does, so later runs never re-read the store for Env/WorkingDir.
966    let env_text: String = config
967        .env
968        .iter()
969        .map(|(k, v)| format!("{k}={v}\n"))
970        .collect();
971    let workdir_text: String = config
972        .workdir
973        .as_deref()
974        .map(|w| format!("{w}\n"))
975        .unwrap_or_default();
976    std::fs::write(env_file(cache), &env_text).context("Failed to write OCI env file")?;
977    std::fs::write(workdir_file(cache), &workdir_text)
978        .context("Failed to write OCI workdir file")?;
979    let argv_text =
980        |argv: &[String]| -> String { argv.iter().map(|arg| format!("{arg}\n")).collect() };
981    std::fs::write(cache.join(CMD_BASENAME), argv_text(&config.cmd))
982        .context("Failed to write OCI cmd file")?;
983    std::fs::write(
984        cache.join(ENTRYPOINT_BASENAME),
985        argv_text(&config.entrypoint),
986    )
987    .context("Failed to write OCI entrypoint file")?;
988    write_in_root(
989        rootfs,
990        CMD_BASENAME,
991        argv_text(&config.cmd).as_bytes(),
992        "in-root OCI cmd file",
993    )?;
994    write_in_root(
995        rootfs,
996        ENTRYPOINT_BASENAME,
997        argv_text(&config.entrypoint).as_bytes(),
998        "in-root OCI entrypoint file",
999    )?;
1000    write_in_root(
1001        rootfs,
1002        ENV_BASENAME,
1003        env_text.as_bytes(),
1004        "in-root OCI env file",
1005    )?;
1006    write_in_root(
1007        rootfs,
1008        WORKDIR_BASENAME,
1009        workdir_text.as_bytes(),
1010        "in-root OCI workdir file",
1011    )?;
1012    Ok(())
1013}

1015/// Apply one overlay layer onto `rootfs`, honoring whiteouts.
1016fn apply_overlay_layer(layer: &Path, rootfs: &Path) -> Result<(), Error> {
1017    // Whiteouts first: `.wh.<name>` deletes `<name>`; `.wh..wh..opq` makes
1018    // the containing directory opaque (clear its current contents).
1019    let mut whiteouts: Vec<PathBuf> = Vec::new();
1020    collect_whiteouts(layer, layer, &mut whiteouts);
1021    for whiteout in &whiteouts {
1022        let rel = whiteout.strip_prefix(layer).unwrap_or(whiteout);
1023        if rel.file_name().and_then(|n| n.to_str()) == Some(".wh..wh..opq") {
1024            let dir = rootfs.join(rel.parent().map(|p| p.to_path_buf()).unwrap_or_default());
1025            if dir.is_dir() {
1026                for entry in std::fs::read_dir(&dir)? {
1027                    let path = entry?.path();
1028                    if path.is_dir() && !path.is_symlink() {
1029                        let _ = std::fs::remove_dir_all(&path);
1030                    } else {
1031                        let _ = std::fs::remove_file(&path);
1032                    }
1033                }
1034            }
1035        } else if let Some(name) = rel
1036            .file_name()
1037            .and_then(|n| n.to_str())
1038            .and_then(|n| n.strip_prefix(".wh."))
1039        {
1040            let target = rootfs
1041                .join(rel.parent().map(|p| p.to_path_buf()).unwrap_or_default())
1042                .join(name);
1043            if target.is_dir() && !target.is_symlink() {
1044                let _ = std::fs::remove_dir_all(&target);
1045            } else {
1046                let _ = std::fs::remove_file(&target);
1047            }
1048        }
1049    }
1050    // Copy the layer's payload (excluding whiteout markers) with `cp -a` so
1051    // modes, symlinks, and hardlinks survive exactly as in the buildah path.
1052    // The copy runs under `podman unshare` because layer entries can be
1053    // owned by mapped sub-UIDs the plain invoking user cannot read.
1054    let started = std::time::Instant::now();
1055    let mut attempt = 0;
1056    loop {
1057        let output = Command::new("podman")
1058            .args(["unshare", "cp", "-a"])
1059            .arg(format!("{}/.", layer.display()))
1060            .arg(format!("{}/", rootfs.display()))
1061            .stdin(Stdio::null())
1062            .output()
1063            .context("Failed to spawn `podman unshare cp` to apply an overlay layer")?;
1064        if output.status.success() {
1065            break;
1066        }
1067        let stderr = String::from_utf8_lossy(&output.stderr);
1068        attempt += 1;
1069        if !crate::app::podman_store::podman_database_locked(&stderr)
1070            || !crate::app::podman_store::wait_for_podman_lock(attempt, started)
1071        {
1072            return Err(Error::msg(format!(
1073                "cp failed applying overlay layer {} (exit {}): {}",
1074                layer.display(),
1075                output.status,
1076                stderr.trim()
1077            )));
1078        }
1079    }
1080    // Drop any whiteout markers the copy carried over.
1081    let mut copied: Vec<PathBuf> = Vec::new();
1082    collect_whiteouts(rootfs, rootfs, &mut copied);
1083    for marker in copied {
1084        let _ = std::fs::remove_file(&marker);
1085    }
1086    Ok(())
1087}

1089fn collect_whiteouts(dir: &Path, root: &Path, out: &mut Vec<PathBuf>) {
1090    let Ok(entries) = std::fs::read_dir(dir) else {
1091        return;
1092    };
1093    for entry in entries.flatten() {
1094        let path = entry.path();
1095        let name = entry.file_name().to_string_lossy().into_owned();
1096        if name.starts_with(".wh.") {
1097            out.push(path.clone());
1098        }
1099        if path.is_dir() && !path.is_symlink() && path != root {
1100            collect_whiteouts(&path, root, out);
1101        }
1102    }
1103}

1105/// Resolve an absolute guest path to its on-host location *inside* the rootfs,
1106/// following symlinks the way the kernel would once the guest has `chroot`ed in.
1107///
1108/// This is necessary because images (nixos/nix is the extreme case) populate
1109/// `/bin/sh`, `/usr/bin/env`, etc. as symlinks whose targets are **absolute**
1110/// guest paths (e.g. `/nix/store/...-bash/bin/bash`). Those targets only resolve
1111/// correctly relative to the image root; a naive `fs::metadata` on the host
1112/// would follow them against the *host* `/` and fail. We therefore walk the path
1113/// component by component, and whenever we hit a symlink we re-root an absolute
1114/// target back onto `rootfs` (a relative target stays relative to the link's
1115/// directory), exactly as a chrooted kernel would. The hop budget guards against
1116/// symlink loops.
1117///
1118/// Returns the final on-host path (which the caller can `stat`/validate). The
1119/// returned path is not guaranteed to exist; the caller performs that check so
1120/// it can produce a task-appropriate error message.
1121pub(crate) fn resolve_in_rootfs(rootfs: &Path, guest_abs: &Path) -> PathBuf {
1122    const MAX_HOPS: usize = 40;
1123
1124    // `current` is an on-host path that always stays within `rootfs`.
1125    let mut current = rootfs.to_path_buf();
1126    // Remaining guest components to process, as a stack (front = next).
1127    let mut pending: Vec<std::ffi::OsString> = guest_abs
1128        .components()
1129        .filter_map(|c| match c {
1130            std::path::Component::Normal(part) => Some(part.to_os_string()),
1131            // Root/prefix are absorbed by starting at `rootfs`; `.`/`..` are
1132            // handled below.
1133            std::path::Component::ParentDir => Some(std::ffi::OsString::from("..")),
1134            _ => None,
1135        })
1136        .rev()
1137        .collect();
1138
1139    let mut hops = 0usize;
1140    while let Some(part) = pending.pop() {
1141        if part == ".." {
1142            // Never let `..` escape above the rootfs.
1143            if current != rootfs {
1144                current.pop();
1145            }
1146            continue;
1147        }
1148        let candidate = current.join(&part);
1149        match std::fs::symlink_metadata(&candidate) {
1150            Ok(meta) if meta.file_type().is_symlink() => {
1151                hops += 1;
1152                if hops > MAX_HOPS {
1153                    // Give up untangling; return best-effort so the caller's
1154                    // existence check reports a clean error.
1155                    return candidate;
1156                }
1157                match std::fs::read_link(&candidate) {
1158                    Ok(target) => {
1159                        if target.is_absolute() {
1160                            // Re-root the absolute target onto the rootfs and
1161                            // re-process its components.
1162                            current = rootfs.to_path_buf();
1163                        }
1164                        // Push target components back onto the stack (relative
1165                        // ones continue from `current`, i.e. the link's dir).
1166                        for comp in target
1167                            .components()
1168                            .filter_map(|c| match c {
1169                                std::path::Component::Normal(p) => Some(p.to_os_string()),
1170                                std::path::Component::ParentDir => {
1171                                    Some(std::ffi::OsString::from(".."))
1172                                }
1173                                _ => None,
1174                            })
1175                            .rev()
1176                        {
1177                            pending.push(comp);
1178                        }
1179                    }
1180                    Err(_) => return candidate,
1181                }
1182            }
1183            _ => {
1184                // Regular file/dir, or nonexistent: advance and let the caller
1185                // decide.
1186                current = candidate;
1187            }
1188        }
1189    }
1190    current
1191}
