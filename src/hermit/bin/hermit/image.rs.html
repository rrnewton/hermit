9//! PROTOTYPE: OCI image rootfs materialization for `hermit run --image`.
10//!
11//! This is the *filesystem half* of hermit-as-container-runtime. Hermit already
12//! owns the *namespace half* (PID/mount/user namespaces via reverie-process);
13//! this module bolts on deterministic FILE INPUTS by materializing a rootfs from
14//! a pinned OCI image (ideally by digest) and letting `run` chroot into it.
15//!
16//! Podman is the image manager: images are fetched into the local
17//! containers/storage store by `podman pull` (or `hermit oci pull`), and this
18//! module materializes a rootfs by copying the merged overlay layers out of
19//! that store (see [`crate::app::podman_store`]). Hermit never pulls on the run
20//! path; a reference the store does not hold is refused with the remedy.
21

31/// Pseudo-filesystem mount points the guest expects to exist inside the root but
32/// that an image layer often omits (busybox, distroless, and `scratch`-derived
33/// images ship no `/proc` or `/sys`). We create them so reverie can mount the
34/// deterministic `/proc` into the chrooted root.
35///
36/// `dev/pts` is here rather than in [`DEV_BIND_TARGETS`] because it receives a
37/// whole filesystem (a fresh `devpts` instance), not a single bind.
38const REQUIRED_DIRS: &[&str] = &["proc", "sys", "dev", "dev/pts", "tmp"];
39
40/// Character devices the guest gets as bind mounts of the host node, relative to
41/// the image root. See `container::image_container` for the mounts themselves and
42/// for why each is safe to expose; this list only has to guarantee the mount
43/// *targets* exist, because the image root is remounted read-only before the
44/// binds are applied and a bind mount cannot create its own target.
45///
46/// An unprivileged user namespace cannot `mknod` a character device, so binding
47/// the host node is the only way to give the guest one. That is also what
48/// rootless podman and runc do.
49pub(super) const DEV_BIND_TARGETS: &[&str] = &["null", "zero", "full", "random", "urandom"];
50
51/// Extra device placeholder created in every rootfs but bound only for the
52/// KVM backend (whose VMM process opens `/dev/kvm` after chroot). For other
53/// backends it remains an empty regular file.
54pub(super) const KVM_DEV_TARGETS: &[&str] = &["kvm"];
55
56/// Conventional `/dev` symlinks, as `(link, target)`. These need no privilege and
57/// no mount: they resolve through `/proc`, which Hermit already replaces with its
58/// deterministic instance inside the image root.
59///
60/// `ptmx` completes the fresh `devpts` instance mounted at `/dev/pts`; the
61/// kernel's devpts documentation names exactly this symlink (or a bind) as the
62/// supported way to expose a per-instance `ptmx`. `fd`, `stdin`, `stdout` and
63/// `stderr` are what bash process substitution (`<(...)`) and countless build
64/// scripts open.
65const DEV_SYMLINKS: &[(&str, &str)] = &[
66    ("ptmx", "pts/ptmx"),
67    ("fd", "/proc/self/fd"),
68    ("stdin", "/proc/self/fd/0"),
69    ("stdout", "/proc/self/fd/1"),
70    ("stderr", "/proc/self/fd/2"),
71];
72
73/// Version of the materialized-rootfs *layout* (which mount targets and symlinks
74/// the tree must contain), mixed into the cache key.
75///
76/// A cached tree is reused whenever its readiness marker exists, so a tree
77/// materialized by an older Hermit would silently lack the `/dev` targets added
78/// here and the binds would fail at mount time. Versioning the key
79/// re-materializes instead. The key is used rather than the marker because an
80/// image root is often mode `0555` and can carry entries owned by unmapped
81/// sub-UIDs, so hermit cannot reliably rewrite or delete a tree it already
82/// produced -- but it can always create a new one beside it. Superseded trees are
83/// left on disk; clearing `~/.cache/hermit/oci-rootfs` reclaims them.
84const ROOTFS_LAYOUT_VERSION: &str = "rootfs-layout-v4-dev";
85
86/// Basename of the captured `Config.Env` file (one `KEY=VALUE` per line).
87const ENV_BASENAME: &str = ".hermit-oci-env";
88/// Basename of the captured `Config.WorkingDir` file (single line).
89const WORKDIR_BASENAME: &str = ".hermit-oci-workdir";
90const CMD_BASENAME: &str = ".hermit-oci-cmd";
91const ENTRYPOINT_BASENAME: &str = ".hermit-oci-entrypoint";

93/// Reduce an image reference to the identity the rootfs cache is keyed by.
94///
95/// A tag is a mutable pointer: keying the cache by the raw reference string
96/// lets a moved tag silently alias an older rootfs. When a local podman store
97/// is available the reference is therefore resolved to the store's canonical
98/// image ID first (a local-only lookup that never contacts a registry), so
99/// `podman pull` moving a tag produces a fresh cache entry. References that
100/// are already an image ID, and references the store cannot resolve (the
101/// store cannot resolve it), fall back to the reference itself.
102pub(crate) fn canonical_image_ref(image_ref: &str) -> String {
103    let candidate = image_ref
104        .trim()
105        .strip_prefix("sha256:")
106        .unwrap_or(image_ref.trim());
107    if candidate.len() == 64 && candidate.chars().all(|c| c.is_ascii_hexdigit()) {
108        return candidate.to_ascii_lowercase();
109    }
110    if let Ok(Some(store)) = crate::app::podman_store::PodmanStore::probe()
111        && let Ok(resolved) = store.resolve(image_ref)
112    {
113        return resolved.id;
114    }
115    image_ref.to_string()
116}

118/// Compute the cache directory that holds materialized rootfs trees. Keyed by
119/// the SHA-256 of the complete image reference so distinct references cannot
120/// alias after filesystem sanitization.
121fn rootfs_cache_dir(image_ref: &str) -> Result<PathBuf, Error> {
122    let base = if let Ok(xdg) = std::env::var("XDG_CACHE_HOME") {
123        PathBuf::from(xdg)
124    } else if let Ok(home) = std::env::var("HOME") {
125        PathBuf::from(home).join(".cache")
126    } else {
127        std::env::temp_dir()
128    };
129    // NUL-separate the two fields so no image reference can collide with a
130    // different (reference, layout) pair by concatenation.
131    let mut keyed = Vec::from(image_ref.as_bytes());
132    keyed.push(0);
133    keyed.extend_from_slice(ROOTFS_LAYOUT_VERSION.as_bytes());
134    let key = Digest::new(&keyed).to_string();
135    Ok(base.join("hermit").join("oci-rootfs").join(key))
136}

138/// Basenames of cache entries still referenced by `live_refs`, for GC.
139pub(crate) fn live_cache_keys(
140    live_refs: &[String],
141) -> Result<std::collections::BTreeSet<String>, Error> {
142    let mut keys = std::collections::BTreeSet::new();
143    for r in live_refs {
144        let dir = rootfs_cache_dir(&canonical_image_ref(r))?;
145        if let Some(name) = dir.file_name().and_then(|n| n.to_str()) {
146            keys.insert(name.to_string());
147        }
148    }
149    Ok(keys)
150}

152/// The oci-rootfs cache base directory, when it exists.
153pub(crate) fn cache_base() -> Result<Option<PathBuf>, Error> {
154    // Any ref works: only the parent of the keyed dir is wanted.
155    let probe = rootfs_cache_dir("probe")?;
156    Ok(probe
157        .parent()
158        .map(|p| p.to_path_buf())
159        .filter(|p| p.is_dir()))
160}

162/// A rootfs is considered already materialized if the marker file is present.
163/// The marker is written last, so its presence means the copy finished.
164///
165/// The marker lives in the cache directory *beside* `rootfs`, not inside it: an
166/// image root is frequently mode `0555` (read-only) and can contain entries
167/// owned by unmapped sub-UIDs, so hermit (running as the plain invoking user,
168/// outside any user namespace) cannot reliably write into the rootfs after
169/// materialization. The cache directory itself is created and owned by hermit,
170/// so the marker is always writable there.
171fn ready_marker(cache: &Path) -> PathBuf {
172    cache.join(".hermit-oci-ready")
173}

185/// The subset of an image's OCI runtime config that the prototype applies so the
186/// guest sees the deterministic, digest-pinned environment the image declares.
187#[derive(Default)]
188pub(crate) struct ImageConfig {
189    /// `Config.Env` entries, split into `(key, value)` pairs.
190    pub env: Vec<(String, String)>,
191    /// `Config.WorkingDir`, if the image set a non-empty one.
192    pub workdir: Option<String>,
193    /// `Config.Cmd` (one argument per persisted line).
194    pub cmd: Vec<String>,
195    /// `Config.Entrypoint` (one argument per persisted line).
196    pub entrypoint: Vec<String>,
197}

199fn read_image_config_files(
200    env_path: &Path,
201    workdir_path: &Path,
202) -> Result<Option<ImageConfig>, Error> {
203    let env_exists = env_path.try_exists().with_context(|| {
204        format!(
205            "Failed to inspect OCI environment file {}",
206            env_path.display()
207        )
208    })?;
209    let workdir_exists = workdir_path.try_exists().with_context(|| {
210        format!(
211            "Failed to inspect OCI working-directory file {}",
212            workdir_path.display()
213        )
214    })?;
215    if !env_exists && !workdir_exists {
216        return Ok(None);
217    }
218
219    let env = if env_exists {
220        parse_env(&std::fs::read_to_string(env_path).with_context(|| {
221            format!("Failed to read OCI environment file {}", env_path.display())
222        })?)
223    } else {
224        Vec::new()
225    };
226    let workdir = if workdir_exists {
227        parse_workdir(&std::fs::read_to_string(workdir_path).with_context(|| {
228            format!(
229                "Failed to read OCI working-directory file {}",
230                workdir_path.display()
231            )
232        })?)
233    } else {
234        None
235    };
236    Ok(Some(ImageConfig {
237        env,
238        workdir,
239        ..Default::default()
240    }))
241}

243fn read_argv_file(path: &Path) -> Vec<String> {
244    std::fs::read_to_string(path)
245        .map(|text| {
246            text.lines()
247                .filter(|line| !line.is_empty())
248                .map(|line| line.to_string())
249                .collect()
250        })
251        .unwrap_or_default()
252}

254/// Read the persisted OCI config (Env + WorkingDir) captured at materialization
255/// time. Missing files yield empty/None so images that declare neither still
256/// run. File *presence*, rather than a non-empty Env, selects the location: an
257/// image may legitimately declare a WorkingDir and no Env.
258///
259/// This is read from two candidate locations, in order, and the first that
260/// yields a non-empty `Env` wins:
261///
262/// 1. The host cache dir beside the rootfs — valid when `guest_command` runs
263///    *before* the chroot (e.g. program validation).
264/// 2. The absolute in-root path (`/.hermit-oci-env`) — valid when
265///    `guest_command` runs *inside* the container, after `chroot(rootfs)`, where
266///    the host cache path (an absolute host path) no longer resolves. The
267///    materializer writes a copy of the config here for exactly this case.
268///
269/// The two-location read is what makes the image-declared environment reach the
270/// guest: the environment that actually runs the program is built in the forked,
271/// chrooted child, so it can only see files that live inside the rootfs.
272pub(crate) fn read_image_config(image_ref: &str) -> Result<ImageConfig, Error> {
273    let cache = rootfs_cache_dir(&canonical_image_ref(image_ref))?;
274    if let Some(mut host) = read_image_config_files(&env_file(&cache), &workdir_file(&cache))? {
275        host.cmd = read_argv_file(&cache.join(CMD_BASENAME));
276        host.entrypoint = read_argv_file(&cache.join(ENTRYPOINT_BASENAME));
277        return Ok(host);
278    }
279    // Fall back to the in-root copy (post-chroot view).
280    if let Some(mut in_root) = read_image_config_files(
281        &Path::new("/").join(ENV_BASENAME),
282        &Path::new("/").join(WORKDIR_BASENAME),
283    )? {
284        in_root.cmd = read_argv_file(&Path::new("/").join(CMD_BASENAME));
285        in_root.entrypoint = read_argv_file(&Path::new("/").join(ENTRYPOINT_BASENAME));
286        return Ok(in_root);
287    }
288    Ok(ImageConfig::default())
289}

291fn parse_env(contents: &str) -> Vec<(String, String)> {
292    contents
293        .lines()
294        .filter_map(|line| {
295            let line = line.trim_end_matches('\r');
296            if line.is_empty() {
297                return None;
298            }
299            line.split_once('=')
300                .map(|(k, v)| (k.to_string(), v.to_string()))
301        })
302        .collect()
303}
304
305fn parse_workdir(contents: &str) -> Option<String> {
306    let trimmed = contents.trim();
307    if trimmed.is_empty() || trimmed == "/" {
308        None
309    } else {
310        Some(trimmed.to_string())
311    }
312}

314/// Materialize the given OCI image reference into a plain, user-owned rootfs
315/// directory and return its path. Reusing a previously materialized reference is
316/// a no-op that returns the cached directory.
317///
318/// `image_ref` must already be in the local podman store, e.g.
319/// `docker.io/library/busybox@sha256:...` (digest-pinned, recommended for
320/// determinism) or `docker.io/library/busybox:latest`.
321pub(crate) fn materialize_rootfs(image_ref: &str) -> Result<PathBuf, Error> {
322    let cache = rootfs_cache_dir(&canonical_image_ref(image_ref))?;
323    let rootfs = cache.join("rootfs");
324    if ready_marker(&cache).exists() {
325        tracing::info!(
326            "Reusing cached OCI rootfs for {} at {}",
327            image_ref,
328            rootfs.display()
329        );
330        return Ok(rootfs);
331    }
332
333    tracing::info!("Materializing OCI rootfs for {}", image_ref);
334    std::fs::create_dir_all(&rootfs)
335        .with_context(|| format!("Failed to create rootfs cache dir {}", rootfs.display()))?;
336
337    // Copy the merged tree out of the local podman store's overlay layers.
338    if materialize_via_podman_store(image_ref, &cache, &rootfs)? {
339        std::fs::write(ready_marker(&cache), image_ref.as_bytes())
340            .context("Failed to write OCI rootfs readiness marker")?;
341        return Ok(rootfs);
342    }
343
344    Err(Error::msg(format!(
345        "image {image_ref} is not in the local podman store, so there is no \
346         rootfs to materialize. Fetch it first with `hermit oci pull {image_ref}` \
347         (or `podman pull {image_ref}`), then re-run."
348    )))
349}

351/// Copy the merged rootfs out of the podman store's overlay layers.
352/// Returns `Ok(false)` when no store holds the image. Layers apply
353/// base-first; overlay whiteouts are honored.
354fn materialize_via_podman_store(
355    image_ref: &str,
356    cache: &Path,
357    rootfs: &Path,
358) -> Result<bool, Error> {
359    let Some(store) = crate::app::podman_store::PodmanStore::probe()? else {
360        return Ok(false);
361    };
362    let Ok(resolved) = store.resolve(image_ref) else {
363        return Ok(false);
364    };
365    let Ok(layers) = store.overlay_dirs(&resolved) else {
366        return Ok(false);
367    };
368    if layers.is_empty() {
369        return Ok(false);
370    }
371    // overlay_dirs is topmost-first; overlay application is base-first.
372    for layer in layers.iter().rev() {
373        apply_overlay_layer(Path::new(layer), rootfs)?;
374    }
375    // Mount points, device placeholders, and symlinks (same set the buildah
376    // path creates, here as plain filesystem operations on our own cache).
377    {
378        use std::os::unix::fs::PermissionsExt;
379        let _ = std::fs::set_permissions(rootfs, std::fs::Permissions::from_mode(0o755));
380    }
381    for dir in REQUIRED_DIRS {
382        std::fs::create_dir_all(rootfs.join(dir))
383            .with_context(|| format!("Failed to create rootfs mount point {dir}"))?;
384    }
385    // Container runtimes provide /etc/hostname; images often omit it.
386    let hostname_path = rootfs.join("etc/hostname");
387    if !hostname_path.exists() {
388        if let Some(parent) = hostname_path.parent() {
389            std::fs::create_dir_all(parent).context("Failed to create rootfs etc directory")?;
390        }
391        std::fs::write(&hostname_path, b"hermetic-container.local\n")
392            .context("Failed to write rootfs /etc/hostname")?;
393    }
394    for node in DEV_BIND_TARGETS.iter().chain(KVM_DEV_TARGETS.iter()) {
395        let target = rootfs.join("dev").join(node);
396        let _ = std::fs::remove_file(&target);
397        std::fs::write(&target, b"")
398            .with_context(|| format!("Failed to create device placeholder {node}"))?;
399    }
400    for (link, target) in DEV_SYMLINKS {
401        let link_path = rootfs.join("dev").join(link);
402        let _ = std::fs::remove_file(&link_path);
403        std::os::unix::fs::symlink(target, &link_path)
404            .with_context(|| format!("Failed to create /dev symlink {link}"))?;
405    }
406    // Persist the image config beside and inside the rootfs, as a container runtime does, so later runs never re-read the store for Env/WorkingDir.
407    let config = store.config(&resolved)?;
408    let env_text: String = config
409        .env
410        .iter()
411        .map(|(k, v)| format!("{k}={v}\n"))
412        .collect();
413    let workdir_text: String = config
414        .workdir
415        .as_deref()
416        .map(|w| format!("{w}\n"))
417        .unwrap_or_default();
418    std::fs::write(env_file(cache), &env_text).context("Failed to write OCI env file")?;
419    std::fs::write(workdir_file(cache), &workdir_text)
420        .context("Failed to write OCI workdir file")?;
421    let argv_text =
422        |argv: &[String]| -> String { argv.iter().map(|arg| format!("{arg}\n")).collect() };
423    std::fs::write(cache.join(CMD_BASENAME), argv_text(&config.cmd))
424        .context("Failed to write OCI cmd file")?;
425    std::fs::write(
426        cache.join(ENTRYPOINT_BASENAME),
427        argv_text(&config.entrypoint),
428    )
429    .context("Failed to write OCI entrypoint file")?;
430    std::fs::write(rootfs.join(CMD_BASENAME), argv_text(&config.cmd))
431        .context("Failed to write in-root OCI cmd file")?;
432    std::fs::write(
433        rootfs.join(ENTRYPOINT_BASENAME),
434        argv_text(&config.entrypoint),
435    )
436    .context("Failed to write in-root OCI entrypoint file")?;
437    std::fs::write(rootfs.join(ENV_BASENAME), &env_text)
438        .context("Failed to write in-root OCI env file")?;
439    std::fs::write(rootfs.join(WORKDIR_BASENAME), &workdir_text)
440        .context("Failed to write in-root OCI workdir file")?;
441    Ok(true)
442}

444/// Apply one overlay layer onto `rootfs`, honoring whiteouts.
445fn apply_overlay_layer(layer: &Path, rootfs: &Path) -> Result<(), Error> {
446    // Whiteouts first: `.wh.<name>` deletes `<name>`; `.wh..wh..opq` makes
447    // the containing directory opaque (clear its current contents).
448    let mut whiteouts: Vec<PathBuf> = Vec::new();
449    collect_whiteouts(layer, layer, &mut whiteouts);
450    for whiteout in &whiteouts {
451        let rel = whiteout.strip_prefix(layer).unwrap_or(whiteout);
452        if rel.file_name().and_then(|n| n.to_str()) == Some(".wh..wh..opq") {
453            let dir = rootfs.join(rel.parent().map(|p| p.to_path_buf()).unwrap_or_default());
454            if dir.is_dir() {
455                for entry in std::fs::read_dir(&dir)? {
456                    let path = entry?.path();
457                    if path.is_dir() && !path.is_symlink() {
458                        let _ = std::fs::remove_dir_all(&path);
459                    } else {
460                        let _ = std::fs::remove_file(&path);
461                    }
462                }
463            }
464        } else if let Some(name) = rel
465            .file_name()
466            .and_then(|n| n.to_str())
467            .and_then(|n| n.strip_prefix(".wh."))
468        {
469            let target = rootfs
470                .join(rel.parent().map(|p| p.to_path_buf()).unwrap_or_default())
471                .join(name);
472            if target.is_dir() && !target.is_symlink() {
473                let _ = std::fs::remove_dir_all(&target);
474            } else {
475                let _ = std::fs::remove_file(&target);
476            }
477        }
478    }
479    // Copy the layer's payload (excluding whiteout markers) with `cp -a` so
480    // modes, symlinks, and hardlinks survive exactly as in the buildah path.
481    // The copy runs under `podman unshare` because layer entries can be
482    // owned by mapped sub-UIDs the plain invoking user cannot read.
483    let started = std::time::Instant::now();
484    let mut attempt = 0;
485    loop {
486        let output = Command::new("podman")
487            .args(["unshare", "cp", "-a"])
488            .arg(format!("{}/.", layer.display()))
489            .arg(format!("{}/", rootfs.display()))
490            .stdin(Stdio::null())
491            .output()
492            .context("Failed to spawn `podman unshare cp` to apply an overlay layer")?;
493        if output.status.success() {
494            break;
495        }
496        let stderr = String::from_utf8_lossy(&output.stderr);
497        attempt += 1;
498        if !crate::app::podman_store::podman_database_locked(&stderr)
499            || !crate::app::podman_store::wait_for_podman_lock(attempt, started)
500        {
501            return Err(Error::msg(format!(
502                "cp failed applying overlay layer {} (exit {}): {}",
503                layer.display(),
504                output.status,
505                stderr.trim()
506            )));
507        }
508    }
509    // Drop any whiteout markers the copy carried over.
510    let mut copied: Vec<PathBuf> = Vec::new();
511    collect_whiteouts(rootfs, rootfs, &mut copied);
512    for marker in copied {
513        let _ = std::fs::remove_file(&marker);
514    }
515    Ok(())
516}
517
518fn collect_whiteouts(dir: &Path, root: &Path, out: &mut Vec<PathBuf>) {
519    let Ok(entries) = std::fs::read_dir(dir) else {
520        return;
521    };
522    for entry in entries.flatten() {
523        let path = entry.path();
524        let name = entry.file_name().to_string_lossy().into_owned();
525        if name.starts_with(".wh.") {
526            out.push(path.clone());
527        }
528        if path.is_dir() && !path.is_symlink() && path != root {
529            collect_whiteouts(&path, root, out);
530        }
531    }
532}

534/// Resolve an absolute guest path to its on-host location *inside* the rootfs,
535/// following symlinks the way the kernel would once the guest has `chroot`ed in.
536///
537/// This is necessary because images (nixos/nix is the extreme case) populate
538/// `/bin/sh`, `/usr/bin/env`, etc. as symlinks whose targets are **absolute**
539/// guest paths (e.g. `/nix/store/...-bash/bin/bash`). Those targets only resolve
540/// correctly relative to the image root; a naive `fs::metadata` on the host
541/// would follow them against the *host* `/` and fail. We therefore walk the path
542/// component by component, and whenever we hit a symlink we re-root an absolute
543/// target back onto `rootfs` (a relative target stays relative to the link's
544/// directory), exactly as a chrooted kernel would. The hop budget guards against
545/// symlink loops.
546///
547/// Returns the final on-host path (which the caller can `stat`/validate). The
548/// returned path is not guaranteed to exist; the caller performs that check so
549/// it can produce a task-appropriate error message.
550pub(crate) fn resolve_in_rootfs(rootfs: &Path, guest_abs: &Path) -> PathBuf {
551    const MAX_HOPS: usize = 40;
552
553    // `current` is an on-host path that always stays within `rootfs`.
554    let mut current = rootfs.to_path_buf();
555    // Remaining guest components to process, as a stack (front = next).
556    let mut pending: Vec<std::ffi::OsString> = guest_abs
557        .components()
558        .filter_map(|c| match c {
559            std::path::Component::Normal(part) => Some(part.to_os_string()),
560            // Root/prefix are absorbed by starting at `rootfs`; `.`/`..` are
561            // handled below.
562            std::path::Component::ParentDir => Some(std::ffi::OsString::from("..")),
563            _ => None,
564        })
565        .rev()
566        .collect();
567
568    let mut hops = 0usize;
569    while let Some(part) = pending.pop() {
570        if part == ".." {
571            // Never let `..` escape above the rootfs.
572            if current != rootfs {
573                current.pop();
574            }
575            continue;
576        }
577        let candidate = current.join(&part);
578        match std::fs::symlink_metadata(&candidate) {
579            Ok(meta) if meta.file_type().is_symlink() => {
580                hops += 1;
581                if hops > MAX_HOPS {
582                    // Give up untangling; return best-effort so the caller's
583                    // existence check reports a clean error.
584                    return candidate;
585                }
586                match std::fs::read_link(&candidate) {
587                    Ok(target) => {
588                        if target.is_absolute() {
589                            // Re-root the absolute target onto the rootfs and
590                            // re-process its components.
591                            current = rootfs.to_path_buf();
592                        }
593                        // Push target components back onto the stack (relative
594                        // ones continue from `current`, i.e. the link's dir).
595                        for comp in target
596                            .components()
597                            .filter_map(|c| match c {
598                                std::path::Component::Normal(p) => Some(p.to_os_string()),
599                                std::path::Component::ParentDir => {
600                                    Some(std::ffi::OsString::from(".."))
601                                }
602                                _ => None,
603                            })
604                            .rev()
605                        {
606                            pending.push(comp);
607                        }
608                    }
609                    Err(_) => return candidate,
610                }
611            }
612            _ => {
613                // Regular file/dir, or nonexistent: advance and let the caller
614                // decide.
615                current = candidate;
616            }
617        }
618    }
619    current
620}
