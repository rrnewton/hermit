28/// How many hex digits of an ID `hermit list` shows; any unique prefix of
29/// at least [`crate::app::manifest::MIN_ID_PREFIX`] names the recording.
30const SHOWN_ID_DIGITS: usize = 8;
31
32/// Command-line options for `hermit list`.
33#[derive(Debug, Parser)]
34pub struct ListOpts {
35    /// Directory where recordings are stored.
36    #[clap(long, value_name = "DIR", env = "HERMIT_DATA_DIR")]
37    data_dir: Option<PathBuf>,
38
39    /// Print the recordings as a JSON array, newest first, each with its
40    /// full `id`, `kind`, `created`, `exit` (null when not recorded),
41    /// `program`, `args` and `path` (its manifest, or for a syscall
42    /// recording without one its directory).
43    #[clap(long)]
44    json: bool,
45}
46
47/// Command-line options for `hermit show`.
48#[derive(Debug, Parser)]
49pub struct ShowOpts {
50    /// The recording to show, by default `last`. It may be `last`,
51    /// `last-run` (the newest of any kind), a recording ID or a unique
52    /// prefix of at least 6 hex digits, or the path of a manifest.json or of
53    /// its directory; `hermit list` lists them.
54    #[clap(value_name = "RECORDING", conflicts_with = "preemptions")]
55    recording: Option<OsString>,
56
57    /// Validate an entire preemption/schedule JSON file and print stored-row
58    /// counts and settings. This does not run a guest or certify replay.
59    #[clap(long, value_name = "FILE", conflicts_with = "recording")]
60    preemptions: Option<PathBuf>,
61
62    /// Directory where recordings are stored.
63    #[clap(long, value_name = "DIR", env = "HERMIT_DATA_DIR")]
64    data_dir: Option<PathBuf>,
65}

67/// One recording, as `hermit list --json` prints it.
68#[derive(Debug, Serialize)]
69struct Entry {
70    id: String,
71    kind: &'static str,
72    created: Option<String>,
73    exit: Option<crate::app::manifest::Exit>,
74    program: Option<String>,
75    args: Vec<String>,
76    path: PathBuf,
77}
78
79impl Entry {
80    fn exit_text(&self) -> String {
81        match &self.exit {
82            Some(crate::app::manifest::Exit {
83                code: Some(code), ..
84            }) => code.to_string(),
85            Some(crate::app::manifest::Exit {
86                signal: Some(signal),
87                ..
88            }) => format!("signal {signal}"),
89            _ => "-".to_owned(),
90        }
91    }
92
93    fn command_text(&self) -> String {
94        std::iter::once(self.program.as_deref().unwrap_or("-"))
95            .chain(self.args.iter().map(String::as_str))
96            .map(|word| shell_words::quote(word).into_owned())
97            .collect::<Vec<_>>()
98            .join(" ")
99    }
100}

102/// Every recording in `data_dir`, newest first. A syscall recording without
103/// a manifest has no `created` time and sorts last. A directory whose
104/// manifest does not load is reported on stderr and left out.
105fn entries(data_dir: &Path) -> Result<Vec<Entry>, Error> {
106    let store = HermitData::with_dir(data_dir);
107    let mut entries = Vec::new();
108    for id in store.try_recordings()? {
109        let directory = data_dir.join(id.to_string());
110        let manifest_path = directory.join(crate::app::manifest::MANIFEST_NAME);
111        if manifest_path.is_file() {
112            match crate::app::manifest::read(&manifest_path) {
113                Ok(manifest) => entries.push(Entry {
114                    id: manifest.id,
115                    kind: manifest.kind.name(),
116                    created: Some(manifest.created),
117                    exit: manifest.exit,
118                    program: manifest.command.program,
119                    args: manifest.command.args,
120                    path: manifest_path,
121                }),
122                Err(error) => eprintln!("hermit list: skipping {}: {error:#}", directory.display()),
123            }
124        } else if directory.join(hermit::RECORDING_METADATA_NAME).is_file() {
125            match store.recording_metadata(id) {
126                Ok(metadata) => entries.push(Entry {
127                    id: id.to_string(),
128                    kind: Kind::FullSyscall.name(),
129                    created: None,
130                    exit: None,
131                    program: Some(metadata.program),
132                    args: metadata.args,
133                    path: directory,
134                }),
135                Err(error) => eprintln!("hermit list: skipping {}: {error:#}", directory.display()),
136            }
137        }
138    }
139    // RFC 3339 UTC times written by one formatter sort as text.
140    entries.sort_by(|left, right| {
141        (right.created.is_some(), &right.created, &left.id).cmp(&(
142            left.created.is_some(),
143            &left.created,
144            &right.id,
145        ))
146    });
147    Ok(entries)
148}

150impl ListOpts {
151    pub fn main(&self) -> Result<ExitStatus, Error> {
152        let store = HermitData::from(self.data_dir.as_ref());
153        let entries = entries(store.data_dir())?;
154        if self.json {
155            println!("{}", serde_json::to_string(&entries)?);
156            return Ok(ExitStatus::SUCCESS);
157        }
158        if entries.is_empty() {
159            eprintln!(
160                "hermit list: {} has no recordings; `hermit record`, and a `hermit run` that \
161                 draws an input from the host, each make one",
162                store.data_dir().display()
163            );
164            return Ok(ExitStatus::SUCCESS);
165        }
166        println!(
167            "{:<8}  {:<12}  {:<20}  {:<10}  COMMAND",
168            "ID", "KIND", "CREATED", "EXIT"
169        );
170        for entry in &entries {
171            println!(
172                "{:<8}  {:<12}  {:<20}  {:<10}  {}",
173                &entry.id[..SHOWN_ID_DIGITS.min(entry.id.len())],
174                entry.kind,
175                entry.created.as_deref().map_or_else(
176                    || "-".to_owned(),
177                    // To the second: `2026-10-13T19:55:00Z`.
178                    |created| format!("{}Z", created.split(['.', 'Z']).next().unwrap_or(created))
179                ),
180                entry.exit_text(),
181                entry.command_text(),
182            );
183        }
184        Ok(ExitStatus::SUCCESS)
185    }
186}

188impl ShowOpts {
189    pub fn main(&self) -> Result<ExitStatus, Error> {
190        if let Some(path) = &self.preemptions {
191            show_preemptions(path, &mut std::io::stdout().lock())?;
192            return Ok(ExitStatus::SUCCESS);
193        }
194        let store = HermitData::from(self.data_dir.as_ref());
195        let spec = self
196            .recording
197            .clone()
198            .unwrap_or_else(|| hermit::LAST.into());
199        let file = match crate::app::manifest::resolve(store.data_dir(), &spec)? {
200            Target::Manifest(path) => {
201                // Refuse what is not a manifest rather than print it as one.
202                crate::app::manifest::read(&path)?;
203                path
204            }
205            Target::Recording { data_dir, id } => {
206                let metadata = data_dir
207                    .join(id.to_string())
208                    .join(hermit::RECORDING_METADATA_NAME);
209                eprintln!(
210                    "hermit show: recording {id} is a syscall recording made before hermit \
211                     wrote manifests; printing its {}",
212                    hermit::RECORDING_METADATA_NAME
213                );
214                metadata
215            }
216        };
217        let text = std::fs::read_to_string(&file)
218            .with_context(|| format!("cannot read {}", file.display()))?;
219        print!("{text}");
220        if !text.ends_with('\n') {
221            println!();
222        }
223        Ok(ExitStatus::SUCCESS)
224    }
225}

227fn show_preemptions(path: &Path, output: &mut impl Write) -> Result<(), Error> {
228    let recording = detcore::preemptions::boundary_recording::Recording::read(path)?;
229    let summary = recording.inspect()?;
230    // No output is exposed until the complete file, EOF and existing typed
231    // validators passed. The summary borrows/counts rows, never loads a clone.
232    serde_json::to_writer(&mut *output, &summary)?;
233    output.write_all(b"\n")?;
234    output.flush()?;
235    Ok(())
236}

243    fn write_manifest(data_dir: &Path, kind: Kind, created: &str) -> hermit::Id {
244        let id = hermit::Id::unique();
245        let mut document = manifest::Manifest::new(
246            id,
247            kind,
248            manifest::Command {
249                program: Some("/bin/echo".to_owned()),
250                args: vec!["two words".to_owned()],
251                cwd: None,
252            },
253            manifest::Inputs::default(),
254        );
255        document.created = created.to_owned();
256        manifest::write(
257            &data_dir.join(id.to_string()).join(manifest::MANIFEST_NAME),
258            &document,
259        )
260        .unwrap();
261        id
262    }

264    #[test]
265    fn a_list_is_newest_first_and_skips_what_is_not_a_recording() {
266        let data = tempfile::tempdir().unwrap();
267        let old = write_manifest(data.path(), Kind::InputsOnly, "2026-10-13T19:55:00.000000Z");
268        let new = write_manifest(data.path(), Kind::Network, "2026-10-13T19:58:00.000000Z");
269        std::fs::create_dir_all(data.path().join("tmp")).unwrap();
270        std::fs::create_dir_all(data.path().join(hermit::Id::unique().to_string())).unwrap();
271        let listed = entries(data.path()).unwrap();
272        let ids: Vec<_> = listed.iter().map(|entry| entry.id.clone()).collect();
273        assert_eq!(ids, [new.to_string(), old.to_string()]);
274        assert_eq!(listed[0].kind, "network");
275        assert_eq!(listed[0].command_text(), "/bin/echo 'two words'");
276        assert_eq!(listed[0].exit_text(), "-");
277    }

279    #[test]
280    fn show_preemptions_validates_the_whole_input_before_printing_counts() {
281        let directory = tempfile::tempdir().unwrap();
282        let path = directory.path().join("history.json");
283        let bytes = br#"{"per_thread":{"3":{"final_prio":1000,"prio_changes":[[10,1000]],"preemption_rcbs":[7],"chaos_epochs":[]}},"global":[],"pipe_wake_latency":0,"signal_state_copy":false}"#;
284        std::fs::write(&path, bytes).unwrap();
285        let mut output = Vec::new();
286        show_preemptions(&path, &mut output).unwrap();
287        let summary: serde_json::Value = serde_json::from_slice(&output).unwrap();
288        assert_eq!(summary["format"], "legacy");
289        assert_eq!(summary["stored_global_rows"], 0);
290        assert_eq!(summary["boundary_rows"], serde_json::Value::Null);
291        assert_eq!(summary["threads"]["3"]["priority_rows"], 1);
292        assert_eq!(summary["threads"]["3"]["rcb_rows"], 1);
293        assert_eq!(summary["threads"]["3"]["final_priority"], 1000);
294        assert_eq!(summary["pipe_wake_latency"], 0);
295        assert_eq!(summary["signal_state_copy"], false);
296        assert_eq!(std::fs::read(&path).unwrap(), bytes);
297
298        let mut malformed = bytes.to_vec();
299        malformed.extend_from_slice(b" true");
300        std::fs::write(&path, &malformed).unwrap();
301        output.clear();
302        assert!(show_preemptions(&path, &mut output).is_err());
303        assert!(output.is_empty());
304        assert_eq!(std::fs::read(&path).unwrap(), malformed);
305    }
