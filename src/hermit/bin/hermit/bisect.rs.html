32/// Bisect two recorded schedules to identify the event ordering that causes a failure.
33#[derive(Debug, Parser)]
34pub struct BisectOpts {
35    /// A recorded schedule whose replay succeeds.
36    #[clap(long, value_name = "SCHEDULE")]
37    good: PathBuf,
38
39    /// A recorded schedule whose replay exhibits the target failure.
40    #[clap(long, value_name = "SCHEDULE")]
41    bad: PathBuf,
42
43    /// Treat stdout matching this regular expression as part of the target failure.
44    #[clap(long, value_name = "REGEX")]
45    target_stdout: Option<Regex>,
46
47    /// Treat stderr matching this regular expression as part of the target failure.
48    #[clap(long, value_name = "REGEX")]
49    target_stderr: Option<Regex>,
50
51    /// Exit code identifying the target failure, 0 to 255, as a shell reports it: the guest's own,
52    /// or 128 plus the signal number when a signal killed it.
53    #[clap(long, default_value = "nonzero", value_name = "NUM|nonzero|any")]
54    target_exit_code: ExitStatusConstraint,
55
56    /// Logging level for replayed guest runs. Above warn, each replay writes its log into a
57    /// file of its own in the workspace.
58    #[clap(short, long, value_name = "LEVEL", env = "HERMIT_LOG")]
59    guest_log: Option<LevelFilter>,
60
61    /// Write the machine-readable race report to this path.
62    #[clap(long, value_name = "PATH")]
63    report_file: Option<PathBuf>,
64
65    /// Use Needleman-Wunsch alignment while selecting midpoint schedules.
66    #[clap(long)]
67    needleman: bool,
68
69    /// Maximum accepted edit-distance jitter in a realized replay schedule.
70    #[clap(long, value_name = "EVENTS")]
71    jitter_dist: Option<usize>,
72
73    /// Number of schedule events to show around the localized race.
74    #[clap(long, value_name = "EVENTS", default_value = "5")]
75    execution_context: usize,
76
77    /// Print replay commands and guest output for each bisection step.
78    #[clap(long, short)]
79    verbose: bool,
80
81    /// The most memory bisect's own process may hold, as `hermit analyze
82    /// --max-memory` takes it: a size, `auto` (a quarter of the memory it may
83    /// use), or `off`.
84    #[clap(long, value_name = "SIZE|auto|off", default_value = "auto")]
85    max_memory: MemoryBudget,
86
87    /// Arguments for the underlying `hermit run`, followed by the program and its arguments.
88    #[clap(value_name = "RUN_ARGS", required = true)]
89    run_args: Vec<String>,
90
91    /// Why this host's retired-branch counter is inexact, if it is (see
92    /// `AnalyzeOpts::inexact_branch_counter`).
93    #[clap(skip = crate::app::host_capabilities::host_inexact_branch_counter as fn() -> Option<String>)]
94    inexact_branch_counter: fn() -> Option<String>,
95}

97/// What [`BisectOpts::prepare`] gives the first replay: the trial
98/// configuration and the good and bad schedules (each stating the RCB
99/// coordinate identity both were checked to share).
100type Prepared = (AnalyzeOpts, PreemptionRecord, PreemptionRecord);
101
102impl BisectOpts {
103    pub fn main(&self, global: &GlobalOpts) -> Result<ExitStatus, Error> {
104        // Before the two schedules are read, the largest allocation.
105        let _watchdog = self.max_memory.enforce("hermit bisect")?;
106        let (mut analyzer, good, bad) = self.prepare(global)?;
107        analyzer.bisect_schedule_pair(good, bad)
108    }
109
110    /// Everything `main` does before the first replay: the refusals, the two
111    /// schedules, and the trial configuration they imply.
112    fn prepare(&self, global: &GlobalOpts) -> Result<Prepared, Error> {
113        let mut analyzer = AnalyzeOpts {
114            target_stdout: self.target_stdout.clone(),
115            target_stderr: self.target_stderr.clone(),
116            target_exit_code: self.target_exit_code.clone(),
117            guest_log: self.guest_log,
118            // Bisect keeps its workspace whatever its log level.
119            guest_log_from_env: false,
120            selfcheck: false,
121            search: false,
122            run_needleman: self.needleman,
123            minimize: false,
124            prefix_search: false,
125            continuation_timeslice: crate::app::analyze::ContinuationTimeslice::Auto,
126            imprecise_search: false,
127            replays_instruction_offsets: false,
128            run1_seed: None,
129            run1_preemptions: None,
130            run1_schedule: None,
131            run2_seed: None,
132            run2_preemptions: None,
133            run2_schedule: None,
134            report_file: self.report_file.clone(),
135            // The "Racing accesses" section is analyze's: bisect's report
136            // keeps its two stacks and runs no extra traced replay.
137            no_racing_accesses: true,
138            race_window: 0,
139            analyze_seed: None,
140            verbose: self.verbose,
141            // The replays log at --guest-log's level (warn without it).
142            trial_log: None,
143            jitter_dist: self.jitter_dist,
144            execution_context: self.execution_context,
145            tmp_dir: None,
146            max_memory: self.max_memory,
147            // Bisect keeps its workspace (`bisect_schedule_pair`), and every
148            // trial's schedules in it, as it has no --keep-tmp to ask.
149            keep_tmp: true,
150            success_exit_code: None,
151            run_arg: Vec::new(),
152            run_args: self.run_args.clone(),
153            // Replays run on the globally selected backend
154            // (`hermit --backend <BACKEND> bisect ...`) and charge the
155            // invocation's shared `--max-log-bytes` budget.
156            backend: global.backend,
157            max_log_bytes: global.max_log_bytes,
158            log_budget: global.log_budget(),
159            inexact_branch_counter: self.inexact_branch_counter,
160            branch_counter_warning: None,
161            // Bisect runs no endpoint to measure: its replays are bounded by
162            // the run arguments' --timeout alone.
163            trial_timeout: crate::app::analyze::TrialTimeout::Off,
164            endpoints: Default::default(),
165            time_skip_sources: Default::default(),
166            // Bisect has no budget and no journal: its replays run as they
167            // did before analyze had them.
168            time_budget: crate::app::analyze::TimeBudget::Off,
169            trial_budget: crate::app::analyze::TrialBudget::Off,
170            resume: None,
171            budget: Default::default(),
172            journal: Default::default(),
173            progress: Default::default(),
174            analyze_seed_drawn: false,
175            partial_report_written: false,
176            resumed_progress: None,
177            trial_runner: Default::default(),
178            hermit_run: Default::default(),
179            trial_schedules_uncapped: Default::default(),
180        };
181        // Before the schedules are read: a refused cap or counter reads and
182        // starts nothing.
183        analyzer.refuse_unsupervised_log_cap()?;
184        analyzer.refuse_options_trials_do_not_apply()?;
185        analyzer.refuse_outputs_trials_overwrite()?;
186        analyzer.refuse_unqualified_trial_timeout()?;
187        analyzer.install_trial_pmu_config()?;
188        analyzer.admit_branch_counter(&mut crate::app::tracing::write_stderr_diagnostic)?;
189        if let Some(notice) =
190            analyzer.small_trial_max_timeslice_notice(crate::app::run::installed_skid_margin)
191        {
192            crate::app::tracing::write_stderr_diagnostic(&notice);
193        }
194
195        // A schedule of a format this Hermit does not read is refused before
196        // anything else is read from either, as a policy refusal: main's
197        // `read_schedule` reports a failed validation as an untyped error.
198        analyzer.refuse_unread_record_formats(&[("--good", &self.good), ("--bad", &self.bad)])?;
199        let good = read_schedule(&self.good, "good")?;
200        let bad = read_schedule(&self.bad, "bad")?;
201        AnalyzeOpts::refuse_recordings_numbered_apart(&[
202            ("--good", &self.good),
203            ("--bad", &self.bad),
204        ])?;
205        // Trials start through `RunOpts::run`, which does not read the epoch
206        // a replayed record carries, so the trials replay from the one the
207        // two schedules were recorded under.
208        analyzer.adopt_recorded_epoch(&[("--good", &self.good), ("--bad", &self.bad)])?;
209        // Every replay charges expired timed waits as the failing schedule did.
210        analyzer.adopt_expired_wait_charge(Some(&self.bad))?;
211        // Every replay registers queued sleeps as the failing schedule did.
212        analyzer.adopt_sleep_registration(Some(&self.bad))?;
213        // Keep the timeslice drawn by the recordings.
214        analyzer.adopt_recorded_timeslice(&[("--good", &self.good), ("--bad", &self.bad)])?;
215        analyzer.refuse_recordings_under_another_pipe_wake_latency(&[
216            ("--good", &self.good),
217            ("--bad", &self.bad),
218        ])?;
219        analyzer
220            .refuse_unadmitted_rcb_histories(&[("--good", &self.good), ("--bad", &self.bad)])?;
221        // Every trial schedule mixes the two, so both must count the same
222        // thing; the records keep the identity they state.
223        combined_identity(
224            (&format!("--good {}", self.good.display()), good.identity()),
225            (&format!("--bad {}", self.bad.display()), bad.identity()),
226        )
227        .map_err(|refusal| {
228            anyhow::Error::new(crate::app::container::PolicyRefusal).context(refusal)
229        })?;
230        // A trial spliced from either schedule's prefix makes the jumps of
231        // global time that prefix contains (`crate::app::time_skip_carry`).
232        {
233            let mut sources = analyzer.time_skip_sources.lock().unwrap();
234            sources.remember(&good);
235            sources.remember(&bad);
236            // Refuse before the first trial, rather than silently downgrading
237            // a format-3 prefix's charging rule to a legacy/format-2 rule.
238            sources.trial_format()?;
239            sources.validate_reserved_inputs()?;
240        }
241        analyzer.adopt_instruction_offsets(&[("--good", &self.good), ("--bad", &self.bad)])?;
242        // Every replay's schedule is written anew, stamped with the replays'
243        // guest IDs, so they are resolved before the first.
244        analyzer.prepare_trial_guest_ids(&[("--good", &self.good), ("--bad", &self.bad)])?;
245        analyzer.adopt_uncapped_slices(&[self.good.as_path(), self.bad.as_path()])?;
246        let (mut good, mut bad) = (good, bad);
247        // Analyze's trial outputs omit optional event timestamps. Put these
248        // admitted search inputs in that same representation: bisection edits
249        // event order, not the original recordings' virtual-time trajectories.
250        // Stripping event times keeps the records' callback checkpoints valid.
251        for event in good
252            .schedevents_iter_mut()
253            .chain(bad.schedevents_iter_mut())
254        {
255            event.end_time = None;
256        }
257        let same_events = good.schedevents().len() == bad.schedevents().len()
258            && good
259                .schedevents()
260                .iter()
261                .zip(bad.schedevents())
262                .all(|(g, b)| g.same_operations(b));
263        if same_events {
264            // An input bisect cannot use, not a failure of Hermit.
265            return Err(
266                crate::app::analyze::AnalyzePrecondition::endpoints_identical(
267                    "the --good and --bad schedules contain identical event traces",
268                )
269                .into(),
270            );
271        }
272        Ok((analyzer, good, bad))
273    }
274}

276fn read_schedule(path: &Path, label: &str) -> anyhow::Result<PreemptionRecord> {
277    let contents = fs::read_to_string(path)
278        .with_context(|| format!("failed to read --{label} schedule {}", path.display()))?;
279    let record: PreemptionRecord =
280        detcore::preemptions::boundary_recording::Recording::decode(contents.as_bytes())
281            .and_then(|recording| recording.into_legacy("bisect"))
282            .with_context(|| format!("failed to parse --{label} schedule {}", path.display()))?;
283    record
284        .validate()
285        .map_err(anyhow::Error::msg)
286        .with_context(|| format!("--{label} schedule {} failed validation", path.display()))?;
287    record.require_faithful_replay().map_err(|error| {
288        anyhow::Error::new(crate::app::container::PolicyRefusal).context(format!(
289            "--{label} schedule {} cannot supply bisection evidence: {error:#}",
290            path.display()
291        ))
292    })?;
293    if !record.contains_schedevents() {
294        bail!(
295            "--{label} schedule {} contains no global schedule events",
296            path.display()
297        );
298    }
299    Ok(record)
300}

502    #[test]
503    fn bisect_search_removes_only_event_times_from_both_inputs() {
504        let first = SchedEvent::syscall(DetTid::from_raw(3), Sysno::write, SyscallPhase::Prehook)
505            .with_start_rip(NonZeroUsize::new(0x1000).unwrap())
506            .with_end_rip(NonZeroUsize::new(0x1002).unwrap());
507        let second = SchedEvent::branches(DetTid::from_raw(5), 4);
508        let good = vec![first, second];
509        let mut timed_good = good.clone();
510        timed_good[0].end_time = Some(LogicalTime::from_nanos(10));
511        let trials_dir = tempfile::tempdir().unwrap();
512
513        // Every distinction other than optional timestamps still reaches the
514        // search, including the order and a genuinely unmatched event.
515        for field in [
516            "order",
517            "thread",
518            "op",
519            "phase",
520            "count",
521            "start_rip",
522            "end_rip",
523            "extra",
524        ] {
525            let mut bad = good.clone();
526            match field {
527                "order" => bad.reverse(),
528                "thread" => bad[0].dettid = DetTid::from_raw(7),
529                "op" => bad[0].op = Op::Syscall(Sysno::read, SyscallPhase::Prehook),
530                "phase" => bad[0].op = Op::Syscall(Sysno::write, SyscallPhase::Posthook),
531                "count" => bad[1].count += 1,
532                "start_rip" => bad[0].start_rip = NonZeroUsize::new(0x2000),
533                "end_rip" => bad[0].end_rip = NonZeroUsize::new(0x2002),
534                "extra" => bad.push(SchedEvent::branches(DetTid::from_raw(7), 1)),
535                _ => unreachable!(),
536            }
537            let timed_bad = bad
538                .iter()
539                .cloned()
540                .map(|event| event.with_time(LogicalTime::from_nanos(20)))
541                .collect();
542            let (mut trials, prepared_good, prepared_bad) = prepare_search_records(
543                &search_record(timed_good.clone()),
544                &search_record(timed_bad),
545            )
546            .unwrap();
547            assert_eq!(prepared_good.schedevents(), &good, "{field}: good input");
548            assert_eq!(prepared_bad.schedevents(), &bad, "{field}: bad input");
549            assert_ne!(
550                prepared_good.schedevents(),
551                prepared_bad.schedevents(),
552                "{field}"

945    /// A synthetic schedule standing in for the `fixture` ("passing" or
946    /// "failing") endpoint: branch events of threads 3 and 5 in an order of
947    /// its own, stating `identity`. It records no run; the checked-in
948    /// flaky_cas recordings' producer is unknown, so they are not relabelled.
949    fn synthetic_schedule(fixture: &str, identity: Option<HistoryIdentity>) -> serde_json::Value {
950        let (three, five) = (detcore::DetTid::from_raw(3), detcore::DetTid::from_raw(5));
951        let order = match fixture {
952            "passing" => [three, five, three],
953            _ => [three, three, five],
954        };
955        let mut writer = detcore::preemptions::PreemptionWriter::new(None);
956        if let Some(identity) = identity {
957            writer = writer.with_identity(identity);
958        }
959        writer.register_thread(three, detcore::DEFAULT_PRIORITY);
960        writer.register_thread(five, detcore::DEFAULT_PRIORITY);
961        for tid in order {
962            writer.insert_schedevent(SchedEvent::branches(tid, 1));
963        }
964        serde_json::from_str(&writer.into_string()).unwrap()
965    }

1267    /// Bisect refuses, before any trial, a `--good` or `--bad` schedule, or a

1272    fn bisect_refuses_schedules_under_another_pipe_wake_latency() {
1273        let dir = tempfile::tempdir().unwrap();
1274        let write = |fixture: &str, name: &str, latency: Option<u64>| {
1275            let mut record = synthetic_schedule(fixture, known());
1276            if let Some(latency) = latency {

1278            }

1283        let prepare = |good: &Path, bad: &Path, run_args: &[&str]| {
1284            let good = format!("--good={}", good.display());

1288            argv.push("/bin/true");

1290            let crate::app::Subcommand::Bisect(options) = args.command else {

1293            options.prepare(&args.global).map(|_| ())
