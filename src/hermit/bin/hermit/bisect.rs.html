30/// Bisect two recorded schedules to identify the event ordering that causes a failure.
31#[derive(Debug, Parser)]
32pub struct BisectOpts {
33    /// A recorded schedule whose replay succeeds.
34    #[clap(long, value_name = "SCHEDULE")]
35    good: PathBuf,
36
37    /// A recorded schedule whose replay exhibits the target failure.
38    #[clap(long, value_name = "SCHEDULE")]
39    bad: PathBuf,
40
41    /// Treat stdout matching this regular expression as part of the target failure.
42    #[clap(long, value_name = "REGEX")]
43    target_stdout: Option<Regex>,
44
45    /// Treat stderr matching this regular expression as part of the target failure.
46    #[clap(long, value_name = "REGEX")]
47    target_stderr: Option<Regex>,
48
49    /// Exit status identifying the target failure.
50    #[clap(long, default_value = "nonzero", value_name = "NUM|nonzero|any")]
51    target_exit_code: ExitStatusConstraint,
52
53    /// Logging level for replayed guest runs.
54    #[clap(short, long, value_name = "LEVEL", env = "HERMIT_LOG")]
55    guest_log: Option<LevelFilter>,
56
57    /// Write the machine-readable race report to this path.
58    #[clap(long, value_name = "PATH")]
59    report_file: Option<PathBuf>,
60
61    /// Use Needleman-Wunsch alignment while selecting midpoint schedules.
62    #[clap(long)]
63    needleman: bool,
64
65    /// Maximum accepted edit-distance jitter in a realized replay schedule.
66    #[clap(long, value_name = "EVENTS")]
67    jitter_dist: Option<usize>,
68
69    /// Number of schedule events to show around the localized race.
70    #[clap(long, value_name = "EVENTS", default_value = "5")]
71    execution_context: usize,
72
73    /// Print replay commands and guest output for each bisection step.
74    #[clap(long, short)]
75    verbose: bool,
76
77    /// The most memory bisect's own process may hold, as `hermit analyze
78    /// --max-memory` takes it: a size, `auto` (a quarter of the memory it may
79    /// use), or `off`.
80    #[clap(long, value_name = "SIZE|auto|off", default_value = "auto")]
81    max_memory: MemoryBudget,
82
83    /// Arguments for the underlying `hermit run`, followed by the program and its arguments.
84    #[clap(value_name = "RUN_ARGS", required = true)]
85    run_args: Vec<String>,
86
87    /// Why this host's retired-branch counter is inexact, if it is (see
88    /// `AnalyzeOpts::inexact_branch_counter`).
89    #[clap(skip = crate::app::host_capabilities::host_inexact_branch_counter as fn() -> Option<String>)]
90    inexact_branch_counter: fn() -> Option<String>,
91}

93impl BisectOpts {
94    pub fn main(&self, global: &GlobalOpts) -> Result<ExitStatus, Error> {
95        // Before the two schedules are read, the largest allocation.
96        let _watchdog = self.max_memory.enforce("hermit bisect")?;
97        let (mut analyzer, good, bad) = self.prepare(global)?;
98        analyzer.bisect_schedule_pair(good, bad)
99    }
100
101    /// Everything `main` does before the first replay: the refusals, the two
102    /// schedules, and the trial configuration they imply.
103    fn prepare(
104        &self,
105        global: &GlobalOpts,
106    ) -> Result<(AnalyzeOpts, Vec<SchedEvent>, Vec<SchedEvent>), Error> {
107        let mut analyzer = AnalyzeOpts {
108            target_stdout: self.target_stdout.clone(),
109            target_stderr: self.target_stderr.clone(),
110            target_exit_code: self.target_exit_code.clone(),
111            guest_log: self.guest_log,
112            selfcheck: false,
113            search: false,
114            run_needleman: self.needleman,
115            minimize: false,
116            prefix_search: false,
117            imprecise_search: false,
118            run1_seed: None,
119            run1_preemptions: None,
120            run1_schedule: None,
121            run2_seed: None,
122            run2_preemptions: None,
123            run2_schedule: None,
124            report_file: self.report_file.clone(),
125            analyze_seed: None,
126            verbose: self.verbose,
127            jitter_dist: self.jitter_dist,
128            execution_context: self.execution_context,
129            tmp_dir: None,
130            max_memory: self.max_memory,
131            success_exit_code: None,
132            run_arg: Vec::new(),
133            run_args: self.run_args.clone(),
134            // Replays run on the globally selected backend
135            // (`hermit --backend <BACKEND> bisect ...`) and charge the
136            // invocation's shared `--max-log-bytes` budget.
137            backend: global.backend,
138            max_log_bytes: global.max_log_bytes,
139            log_budget: global.log_budget(),
140            inexact_branch_counter: self.inexact_branch_counter,
141            branch_counter_warning: None,
142            // Bisect runs no endpoint to measure: its replays are bounded by
143            // the run arguments' --timeout alone.
144            trial_timeout: crate::app::analyze::TrialTimeout::Off,
145            endpoints: Default::default(),
146            time_skip_sources: Default::default(),
147        };
148        // Before the schedules are read: a refused cap or counter reads and
149        // starts nothing.
150        analyzer.refuse_unsupervised_log_cap()?;
151        analyzer.refuse_options_trials_do_not_apply()?;
152        analyzer.refuse_outputs_trials_overwrite()?;
153        analyzer.refuse_unqualified_trial_timeout()?;
154        analyzer.install_trial_pmu_config()?;
155        analyzer.admit_branch_counter(&mut crate::app::tracing::write_stderr_diagnostic)?;
156        if let Some(notice) =
157            analyzer.small_trial_max_timeslice_notice(crate::app::run::installed_skid_margin)
158        {
159            crate::app::tracing::write_stderr_diagnostic(&notice);
160        }
161
162        let good = read_schedule(&self.good, "good")?;
163        let bad = read_schedule(&self.bad, "bad")?;
164        // Each trial's schedule is written anew without an epoch, so the
165        // trials replay from the one the two schedules were recorded under.
166        analyzer.adopt_recorded_epoch(&[("--good", &self.good), ("--bad", &self.bad)])?;
167        // Every replay charges expired timed waits as the failing schedule did.
168        analyzer.adopt_expired_wait_charge(Some(&self.bad))?;
169        // Keep the timeslice drawn by the recordings.
170        analyzer.adopt_recorded_timeslice(&[("--good", &self.good), ("--bad", &self.bad)])?;
171        analyzer.refuse_recordings_under_another_pipe_wake_latency(&[
172            ("--good", &self.good),
173            ("--bad", &self.bad),
174        ])?;
175        // A trial spliced from either schedule's prefix makes the jumps of
176        // global time that prefix contains (`crate::app::time_skip_carry`).
177        {
178            let mut sources = analyzer.time_skip_sources.lock().unwrap();
179            sources.remember(&good);
180            sources.remember(&bad);
181            // Refuse before the first trial, rather than silently downgrading
182            // a format-3 prefix's charging rule to a legacy/format-2 rule.
183            sources.trial_format()?;
184            sources.validate_reserved_inputs()?;
185        }
186        let mut good = good.into_global();
187        let mut bad = bad.into_global();
188        // Analyze's trial outputs omit optional event timestamps. Put these
189        // admitted search inputs in that same representation: bisection edits
190        // event order, not the original recordings' virtual-time trajectories.
191        for event in good.iter_mut().chain(&mut bad) {
192            event.end_time = None;
193        }
194        if good == bad {
195            bail!("the --good and --bad schedules contain identical event traces");
196        }
197        Ok((analyzer, good, bad))
198    }
199}

201fn read_schedule(path: &Path, label: &str) -> anyhow::Result<PreemptionRecord> {
202    let contents = fs::read_to_string(path)
203        .with_context(|| format!("failed to read --{label} schedule {}", path.display()))?;
204    let record: PreemptionRecord = serde_json::from_str(&contents)
205        .with_context(|| format!("failed to parse --{label} schedule {}", path.display()))?;
206    record
207        .validate()
208        .map_err(anyhow::Error::msg)
209        .with_context(|| format!("--{label} schedule {} failed validation", path.display()))?;
210    record.require_faithful_replay().map_err(|error| {
211        anyhow::Error::new(crate::app::container::PolicyRefusal).context(format!(
212            "--{label} schedule {} cannot supply bisection evidence: {error:#}",
213            path.display()
214        ))
215    })?;
216    if !record.contains_schedevents() {
217        bail!(
218            "--{label} schedule {} contains no global schedule events",
219            path.display()
220        );
221    }
222    Ok(record)
223}

238    fn search_record(events: Vec<SchedEvent>) -> serde_json::Value {
239        let record = PreemptionRecord::from_sched_events(events)
240            .with_pipe_wake_latency(Some(3_000_000))
241            .with_signal_state_copy(Some(true));
242        let mut record = serde_json::to_value(record).unwrap();
243        record["epoch"] = serde_json::json!(RECORDED);
244        record
245    }

247    /// Exercise production admission and conversion without launching a trial;
248    /// both successful preparation and refusal must leave the inputs intact.
249    fn prepare_search_records(
250        good: &serde_json::Value,
251        bad: &serde_json::Value,
252    ) -> Result<(AnalyzeOpts, Vec<SchedEvent>, Vec<SchedEvent>), Error> {
253        let dir = tempfile::tempdir().unwrap();
254        let good_path = dir.path().join("good.json");
255        let bad_path = dir.path().join("bad.json");
256        let good_bytes = serde_json::to_vec(good).unwrap();
257        let bad_bytes = serde_json::to_vec(bad).unwrap();
258        fs::write(&good_path, &good_bytes).unwrap();
259        fs::write(&bad_path, &bad_bytes).unwrap();
260        let args = crate::app::Args::try_parse_from([
261            "hermit".to_owned(),
262            "bisect".to_owned(),
263            format!("--good={}", good_path.display()),
264            format!("--bad={}", bad_path.display()),
265            "--".to_owned(),
266            format!("--epoch={RECORDED}"),
267            "--pipe-wake-latency=3000000".to_owned(),
268            "/bin/true".to_owned(),
269        ])
270        .unwrap();
271        let crate::app::Subcommand::Bisect(options) = args.command else {
272            panic!("expected bisect")
273        };
274        let result = options.prepare(&args.global);
275        assert_eq!(fs::read(good_path).unwrap(), good_bytes);
276        assert_eq!(fs::read(bad_path).unwrap(), bad_bytes);
277        result
278    }

314    fn bisect_search_removes_only_event_times_from_both_inputs() {
315        let first = SchedEvent::syscall(DetTid::from_raw(3), Sysno::write, SyscallPhase::Prehook)
316            .with_start_rip(NonZeroUsize::new(0x1000).unwrap())
317            .with_end_rip(NonZeroUsize::new(0x1002).unwrap());
318        let second = SchedEvent::branches(DetTid::from_raw(5), 4);
319        let good = vec![first, second];
320        let mut timed_good = good.clone();
321        timed_good[0].end_time = Some(LogicalTime::from_nanos(10));
322        let trials_dir = tempfile::tempdir().unwrap();
323
324        // Every distinction other than optional timestamps still reaches the
325        // search, including the order and a genuinely unmatched event.
326        for field in [
327            "order",
328            "thread",
329            "op",
330            "phase",
331            "count",
332            "start_rip",
333            "end_rip",
334            "extra",
335        ] {
336            let mut bad = good.clone();
337            match field {
338                "order" => bad.reverse(),
339                "thread" => bad[0].dettid = DetTid::from_raw(7),
340                "op" => bad[0].op = Op::Syscall(Sysno::read, SyscallPhase::Prehook),
341                "phase" => bad[0].op = Op::Syscall(Sysno::write, SyscallPhase::Posthook),
342                "count" => bad[1].count += 1,
343                "start_rip" => bad[0].start_rip = NonZeroUsize::new(0x2000),
344                "end_rip" => bad[0].end_rip = NonZeroUsize::new(0x2002),
345                "extra" => bad.push(SchedEvent::branches(DetTid::from_raw(7), 1)),
346                _ => unreachable!(),
347            }
348            let timed_bad = bad
349                .iter()
350                .cloned()
351                .map(|event| event.with_time(LogicalTime::from_nanos(20)))
352                .collect();
353            let (mut trials, prepared_good, prepared_bad) = prepare_search_records(
354                &search_record(timed_good.clone()),
355                &search_record(timed_bad),
356            )
357            .unwrap();
358            assert_eq!(prepared_good, good, "{field}: good input");
359            assert_eq!(prepared_bad, bad, "{field}: bad input");
360            assert_ne!(prepared_good, prepared_bad, "{field}");
361            trials.tmp_dir = Some(trials_dir.path().to_path_buf());
362            assert_eq!(trials.trial_epoch_for_test(), RECORDED_RFC3339);
363            assert_eq!(trials.trial_pipe_wake_latency(), Some(3_000_000));
364            assert!(trials.trial_signal_state_copy());
365        }
366    }

667    /// Through `prepare`: the trial configuration `hermit bisect` would replay
668    /// these schedules with, or its refusal.
669    fn prepared_trials(
670        dir: &Path,
671        good: Option<&str>,
672        bad: Option<&str>,
673        run_args: &[&str],
674    ) -> Result<AnalyzeOpts, Error> {
675        let (good, bad) = schedules_with_epochs(dir, good, bad);
676        let good = format!("--good={}", good.display());
677        let bad = format!("--bad={}", bad.display());
678        let mut argv = vec!["hermit", "bisect", good.as_str(), bad.as_str(), "--"];
679        argv.extend(run_args);
680        argv.push("/bin/true");
681        let args = crate::app::Args::try_parse_from(&argv).unwrap();
682        let crate::app::Subcommand::Bisect(options) = args.command else {
683            panic!("{argv:?} is not bisect")
684        };

686        analyzer.tmp_dir = Some(dir.to_path_buf());
687        Ok(analyzer)

693    /// Bisect replays from the epoch its schedules were recorded under, as
694    /// `hermit run` does when it replays one
695    /// (https://github.com/rrnewton/hermit/issues/3835). Before, every trial
696    /// started from the run arguments' epoch, the default when none was
697    /// given. The checks run in a child process whose environment fixes
698    /// `HERMIT_EPOCH`, which counts as an explicit epoch: once unset, and once
699    /// set to an epoch other than the recorded one.
700    #[test]
701    fn bisect_replays_from_the_epoch_its_schedules_were_recorded_under() {
702        const CHILD: &str = "HERMIT_BISECT_EPOCH_TEST_CHILD";
703        const OTHER: &str = "2026-01-01T00:00:00Z";
704        let Some(arm) = std::env::var_os(CHILD) else {
705            let name = format!(
706                "{}::bisect_replays_from_the_epoch_its_schedules_were_recorded_under",
707                module_path!().split_once("::").unwrap().1
708            );
709            for arm in ["unset", "set"] {
710                let mut child = std::process::Command::new(std::env::current_exe().unwrap());
711                child
712                    .args(["--exact", &name, "--nocapture"])
713                    .env(CHILD, arm);
714                if arm == "set" {
715                    child.env("HERMIT_EPOCH", OTHER);
716                } else {
717                    child.env_remove("HERMIT_EPOCH");
718                }
719                let output = child.output().unwrap();
720                assert!(output.status.success(), "{arm}: {output:?}");
721                assert!(
722                    String::from_utf8_lossy(&output.stdout).contains("1 passed"),
723                    "{arm}: the child ran no test: {output:?}"
724                );
725            }
726            return;
727        };
728        let dir = tempfile::tempdir().unwrap();
729        let dir = dir.path();
730        if arm == "set" {
731            let error = prepared_trials(dir, Some(RECORDED), Some(RECORDED), &[]).unwrap_err();
732            assert!(error.downcast_ref::<PolicyRefusal>().is_some(), "{error:#}");
733            assert!(
734                error.to_string().contains(&format!(
735                    "the explicit virtual-time epoch 2026-01-01T00:00:00+00:00 (from --epoch or \
736                     HERMIT_EPOCH) differs from the epoch {RECORDED_RFC3339}"
737                )),
738                "{error:#}"
739            );
740            return;
741        }
742
743        // The default epoch, for schedules that carry none.
744        let trials = prepared_trials(dir, None, None, &[]).unwrap();
745        // No `--epoch` is added; `--expired-wait-charge` always is.
746        assert_eq!(trials.run_arg, ["--expired-wait-charge=each"]);
747        assert_eq!(trials.trial_epoch_for_test(), "2026-01-01T00:00:00+00:00");
748
749        // The recorded epoch, from either schedule or both.
750        for (good, bad) in [
751            (Some(RECORDED), Some(RECORDED)),
752            (Some(RECORDED), None),
753            (None, Some(RECORDED)),
754        ] {
755            let trials = prepared_trials(dir, good, bad, &[]).unwrap();
756            assert_eq!(
757                trials.trial_epoch_for_test(),
758                RECORDED_RFC3339,
759                "{good:?} / {bad:?}"
760            );
761        }
762
763        // An explicit epoch equal to the recorded one adds no second --epoch.
764        let trials =
765            prepared_trials(dir, Some(RECORDED), Some(RECORDED), &["--epoch", RECORDED]).unwrap();
766        // No `--epoch` is added; `--expired-wait-charge` always is.
767        assert_eq!(trials.run_arg, ["--expired-wait-charge=each"]);
768        assert_eq!(trials.trial_epoch_for_test(), RECORDED_RFC3339);
769
770        // An explicit epoch that differs is refused, naming the remedy.
771        let error =
772            prepared_trials(dir, Some(RECORDED), Some(RECORDED), &["--epoch", OTHER]).unwrap_err();
773        assert!(error.downcast_ref::<PolicyRefusal>().is_some(), "{error:#}");
774        assert!(
775            error
776                .to_string()
777                .contains(&format!("pass --epoch={RECORDED_RFC3339}")),
778            "{error:#}"
779        );
780    }

861    /// Bisect refuses, before any trial, a `--good` or `--bad` schedule made
862    /// under another `--pipe-wake-latency` than the trials' own
863    /// (https://github.com/rrnewton/hermit/issues/4380), and any schedule in
864    /// its run arguments.
865    #[test]
866    fn bisect_refuses_schedules_under_another_pipe_wake_latency() {
867        let resources = Path::new(env!("CARGO_MANIFEST_DIR")).join("test-resources");
868        let dir = tempfile::tempdir().unwrap();
869        let write = |fixture: &str, name: &str, latency: Option<u64>| {
870            let source = resources.join(format!("flaky_cas_sequence_schedules-{fixture}.json"));
871            let mut record: serde_json::Value =
872                serde_json::from_str(&fs::read_to_string(source).unwrap()).unwrap();
873            if let Some(latency) = latency {
874                record["pipe_wake_latency"] = serde_json::Value::from(latency);
875            }
876            let path = dir.path().join(name);
877            fs::write(&path, record.to_string()).unwrap();
878            path
879        };
880        let prepare = |good: &Path, bad: &Path, run_args: &[&str]| {
881            let good = format!("--good={}", good.display());
882            let bad = format!("--bad={}", bad.display());
883            let mut argv = vec!["hermit", "bisect", good.as_str(), bad.as_str(), "--"];
884            argv.extend(run_args);
885            argv.push("/bin/true");
886            let args = crate::app::Args::try_parse_from(&argv).unwrap();
887            let crate::app::Subcommand::Bisect(options) = args.command else {
888                panic!("{argv:?} is not bisect")
889            };
890            options.prepare(&args.global).map(|_| ())
891        };
892        let good_off = write("passing", "good-off.json", None);
893        let bad_off = write("failing", "bad-off.json", None);
894        let good_zero = write("passing", "good-zero.json", Some(0));
895        let good_three = write("passing", "good-three.json", Some(3_000_000));
896        let bad_three = write("failing", "bad-three.json", Some(3_000_000));
897        let replayed_zero = write("passing", "replayed-zero.json", Some(0));
898        let replay_flag = format!("--replay-schedule-from={}", replayed_zero.display());
899
900        assert!(prepare(&good_off, &bad_off, &[]).is_ok());
901        assert!(prepare(&good_three, &bad_three, &["--pipe-wake-latency=3000000"]).is_ok());
902        for (good, bad, run_args, named) in [
903            (&good_zero, &bad_off, vec![], "--good"),
904            (&good_off, &bad_off, vec!["--pipe-wake-latency=0"], "--good"),
905            (
906                &good_three,
907                &bad_three,
908                vec!["--pipe-wake-latency=0"],
909                "--good",
910            ),
911        ] {
912            let error = prepare(good, bad, &run_args).unwrap_err();
913            assert!(error.downcast_ref::<PolicyRefusal>().is_some(), "{error:#}");
914            let message = format!("{error:#}");
915            assert!(
916                message.contains(named) && message.contains("cannot reproduce the run"),
917                "{run_args:?}: {message}"
918            );
919        }
920        // A schedule in the run arguments is refused before any recording is
921        // read, whatever its latency: no run bisect launches would replay it
922        // (`RunOpts::options_trials_refuse`).
923        // A missing `--good` shows that no schedule was read before it.
924        for good in [good_off.clone(), dir.path().join("missing.json")] {
925            let error = prepare(&good, &bad_off, &[replay_flag.as_str()]).unwrap_err();
926            assert!(error.downcast_ref::<PolicyRefusal>().is_some(), "{error:#}");
927            assert!(
928                format!("{error:#}").contains("refuse --replay-schedule-from in the run arguments"),
929                "{error:#}"
930            );
931        }
932    }
