9use std::path::PathBuf;
10
11use clap::Parser;
12use hermit::Error;
13use hermit::ExitStatus;
14
15use super::container::PolicyRefusal;
16use super::global_opts::GlobalOpts;
17
18/// Arguments for the executable-selected syscall tracing command.
19#[derive(Debug, Parser)]
20pub struct StraceOpts {
21    /// Program to trace.
22    #[clap(value_name = "PROGRAM")]
23    program: PathBuf,
24
25    /// Arguments passed to the traced program.
26    #[clap(
27        value_name = "ARGS",
28        trailing_var_arg = true,
29        allow_hyphen_values = true
30    )]
31    args: Vec<String>,
32}
33
34impl StraceOpts {
35    pub fn main_with_backend<A: super::backend_hook::BackendAdapter>(
36        &self,
37        global: &GlobalOpts,
38        adapter: &A,
39    ) -> Result<ExitStatus, Error> {
40        if global.log.is_some() || global.log_file.is_some() {
41            anyhow::bail!(
42                "the {} strace backend does not support --log or --log-file",
43                A::NAME
44            );
45        }
46        // An executable without a tracer refuses the command itself, so
47        // `--max-log-bytes` neither changes that refusal nor is refused for a
48        // runner that does not exist. Both checks are static facts.
49        if !A::SUPPORTS_STRACE {
50            return Err(Error::new(PolicyRefusal).context(format!(
51                "executable-selected backend `{}` does not support syscall tracing",
52                A::NAME
53            )));
54        }
55        // This physical tracer does not use the core's charged writers or
56        // container owner. Refuse the cap before any adapter/artifact work.
57        if global.max_log_bytes.is_some() {
58            return Err(Error::new(PolicyRefusal).context(format!(
59                "--max-log-bytes cannot be enforced with {} strace: the runner's trace \
60                 output does not pass through hermit's charged writers, and the runner \
61                 is a plain child with no PID namespace, parent-death signal or ptrace \
62                 attachment binding it, so the cap could neither count the trace nor \
63                 stop the guest; drop --max-log-bytes",
64                A::NAME
65            )));
66        }
67        adapter.strace(&self.program, &self.args)
68    }
69}
