37fn write_canonical_info(
38    file: &Path,
39    writer: &mut impl Write,
40    record_envelope: RecordEnvelope,
41) -> std::io::Result<usize> {
42    if record_envelope.policy() == RecordEnvelopePolicy::AllRecordsV1 {
43        let bytes = std::fs::read(file)?;
44        return logdiff::write_bitwise_info_v1_bytes(&bytes, &file.display().to_string(), writer);
45    }
46    logdiff::write_canonical_info_with_filter(file, writer, record_envelope.predicate())
47}

49fn try_log_diff_with_records(
50    left: &Path,
51    right: &Path,
52    options: &logdiff::LogDiffOpts,
53    record_envelope: RecordEnvelope,
54) -> std::io::Result<(logdiff::LogDiffSummary, usize, usize, LogDiffInputs)> {
55    let left_bytes = std::fs::read(left)?;
56    let right_bytes = std::fs::read(right)?;
57    let left = strict_log_text(&left_bytes, &left.display().to_string())?;
58    let right = strict_log_text(&right_bytes, &right.display().to_string())?;
59    let summary = logdiff::log_diff_summary_from_strs_with_filter(
60        left,
61        right,
62        options,
63        &mut std::io::stderr(),
64        record_envelope.predicate(),
65    )?;
66    let identity = |bytes: &[u8]| LogDiffInput {
67        sha256: detcore::Digest::new(bytes).to_string(),
68        bytes: bytes.len() as u64,
69    };
70    Ok((
71        summary,
72        logdiff::record_count(left),
73        logdiff::record_count(right),
74        LogDiffInputs {
75            left: identity(&left_bytes),
76            right: identity(&right_bytes),
77        },
78    ))
79}

81fn strict_log_text<'a>(bytes: &'a [u8], label: &str) -> std::io::Result<&'a str> {
82    std::str::from_utf8(bytes).map_err(|error| {
83        std::io::Error::new(
84            std::io::ErrorKind::InvalidData,
85            format!("{label} is not UTF-8: {error}"),
86        )
87    })
88}

92    right: &Path,

101            difference_limit: options.limit,
102            syscall_history: options.syscall_history,
103            no_color: options.no_color,
104            print_logs: options.print_logs,
105        },
106        &mut diagnostic_output,
107    )

110fn compare_complete_prefix(
111    left: &[u8],
112    right: &[u8],
113    options: &logdiff::LogDiffOpts,
114    writer: &mut impl Write,
115    record_envelope: RecordEnvelope,
116    fixed_bitwise_info_v1: bool,
117) -> std::io::Result<logdiff::PrefixComparison> {
118    if fixed_bitwise_info_v1 {
119        return logdiff::compare_complete_bitwise_info_v1_prefix(
120            left,
121            right,
122            options.side_labels.clone(),
123            logdiff::BitwiseInfoV1Diagnostics {
124                difference_limit: options.limit,
125                syscall_history: options.syscall_history,
126                no_color: options.no_color,
127                print_logs: options.print_logs,
128            },
129            writer,
130        );
131    }
132    let left = strict_log_text(left, "left log")?;
133    let right = strict_log_text(right, "right log")?;
134    logdiff::compare_complete_prefix_with_filter(
135        left,
136        right,
137        options,
138        writer,
139        record_envelope.predicate(),
140    )
141}

143/// Command-line options for the "logdiff" subcommand.
144#[derive(Debug, Parser)]
145pub struct LogDiffCLIOpts {
146    /// A log to print canonically, or the first of two logs to compare.
147    file_a: PathBuf,
148    /// Optional second log. With one input, print the canonical INFO stream
149    /// used by strict verification; with two inputs, compare them.
150    file_b: Option<PathBuf>,
151
152    /// Write one machine-readable JSON line for a two-log comparison.
153    #[clap(long, value_name = "FILE")]
154    json: Option<PathBuf>,
155
156    /// Compare the canonical INFO stream used by --verify-strict within the
157    /// selected record envelope. With --json this comparison is mandatory and
158    /// selected automatically.
159    #[clap(long)]
160    canonical_info: bool,
161
162    /// Versioned record envelope applied before selecting messages. DBT logs
163    /// must opt into their transport envelope explicitly; the default preserves
164    /// every parsed record.
165    #[clap(long, value_enum, default_value = "all-records-v1")]
166    record_envelope: RecordEnvelopeArg,
167
168    /// Compare two runs that are still being written, reporting divergence as
169    /// soon as it appears instead of waiting for both to finish. Only records
170    /// both logs have finished writing are compared, so a half-flushed tail is
171    /// never mistaken for a difference.
172    #[clap(long)]
173    follow: bool,
174
175    /// How often to re-read the logs while following.
176    #[clap(long, value_name = "MILLISECONDS", default_value = "500")]
177    follow_interval_ms: u64,
178
179    /// Give up following after this long. 0 waits indefinitely. Timing out is
180    /// reported as its own outcome, never as agreement.
181    #[clap(long, value_name = "SECONDS", default_value = "300")]
182    follow_timeout_secs: u64,
183
184    /// Consecutive polls with both logs unchanged before they are treated as
185    /// quiescent and following stops.
186    #[clap(long, value_name = "POLLS", default_value = "3")]
187    follow_settle_polls: u32,
188
189    /// Compare two `hermit run --verify-run` directories, given as the two
190    /// inputs, the way `hermit run --verify` compares its two runs, and write
191    /// that comparison's report (the report `--verify --verify-json` writes)
192    /// to FILE. The exit status is the one `hermit run --verify` would exit
193    /// with, and the guest's output is shown as it would show it. The
194    /// comparison is the one both runs' flags chose; runs that chose different
195    /// ones are refused. Where `run --verify` would not have started its
196    /// second run, the second directory need not exist.
197    #[clap(
198        long,
199        value_name = "FILE",
200        conflicts_with_all = ["json", "canonical_info", "follow"]
201    )]
202    verify_json: Option<PathBuf>,
203
204    /// With --verify-json, the guest exit status run 1 must have, as `hermit
205    /// run --verify-allow` requires it. A recording needs none: `hermit
206    /// record start --verify` replays and compares every recording.
207    #[clap(
208        long,
209        value_name = "success|failure|both",
210        default_value = "success",
211        requires = "verify_json"
212    )]
213    verify_allow: VerifyAllow,
214
215    /// With --verify-json, keep the compared logs in DIR as `hermit run
216    /// --verify --keep-logs --verify-log-dir DIR` keeps them: run 1's alone
217    /// after a match, both otherwise. They are links; the --verify-run
218    /// directories keep their own.
219    #[clap(long = "verify-log-dir", value_name = "DIR", requires = "verify_json")]
220    verify_log_dir: Option<PathBuf>,
221
222    #[clap(flatten)]
223    pub more: logdiff::LogDiffOpts,
224}

267    fn run_with_comparator(
268        &self,
269        global: &GlobalOpts,
270        compare: impl FnOnce(
271            [&Path; 2],
272            Option<&Path>,
273            VerifyAllow,
274            Option<&Path>,
275        ) -> Result<ExitStatus, hermit::Error>,
276    ) -> Result<ExitStatus, hermit::Error> {
277        let Some(report) = &self.verify_json else {
278            return Ok(self.main(global));
279        };
280        let Some(file_b) = &self.file_b else {
281            return Err(hermit::Error::msg(
282                "hermit log-diff --verify-json compares two --verify-run directories",
283            ));
284        };
285        compare(
286            [self.file_a.as_path(), file_b.as_path()],
287            Some(report),
288            self.verify_allow,
289            self.verify_log_dir.as_deref(),
290        )
291    }

293    fn one_input_uses_only_canonical_options(&self) -> bool {
294        let defaults = logdiff::LogDiffOpts::default();
295        !self.more.strip_lines
296            && self.more.limit == defaults.limit
297            && self.more.ignore_lines.is_empty()
298            && self.more.syscall_history == defaults.syscall_history
299            && !self.more.no_color
300            && !self.more.skip_commit
301            && !self.more.skip_detlog
302            && !self.more.git_diff
303            && self.more.include_detlogs == defaults.include_detlogs
304    }

306    /// Print one log canonically or compare two logs.
307    pub fn main(&self, _global: &GlobalOpts) -> ExitStatus {
308        let mut inputs = vec![self.file_a.clone()];
309        inputs.extend(self.file_b.iter().cloned());
310        let _retention_locks = match lock_failed_verification_logs_for_read(inputs) {
311            Ok(locks) => locks,
312            Err(error) => {
313                eprintln!("hermit log-diff: could not protect retained logs: {error}");
314                return ExitStatus::Exited(2);
315            }
316        };
317        let record_envelope = self.record_envelope.envelope();
318        eprintln!(
319            "hermit log-diff: record envelope {}",
320            record_envelope.policy().as_str()
321        );
322        let Some(file_b) = &self.file_b else {
323            if self.json.is_some() {
324                eprintln!("hermit log-diff: --json requires two input logs");
325                return ExitStatus::Exited(2);
326            }
327            if !self.one_input_uses_only_canonical_options() {
328                eprintln!(
329                    "hermit log-diff: comparison options require two input logs; one input always prints the canonical INFO stream used by strict verification"
330                );
331                return ExitStatus::Exited(2);
332            }
333            return match write_canonical_info(
334                &self.file_a,
335                &mut std::io::stdout().lock(),
336                record_envelope,
337            ) {
338                Ok(0) => {
339                    eprintln!(
340                        "hermit log-diff: {} contains no comparable INFO messages",
341                        self.file_a.display()
342                    );
343                    ExitStatus::Exited(2)
344                }
345                Ok(_) => ExitStatus::Exited(0),
346                Err(error) => {
347                    eprintln!(
348                        "hermit log-diff: could not canonicalize {}: {error}",
349                        self.file_a.display()
350                    );
351                    ExitStatus::Exited(2)
352                }
353            };
354        };
355
356        let mut options = self.more.clone();
357        if self.canonical_info || self.json.is_some() {
358            if let Err(message) = canonical_comparison_is_unrelaxed(&options) {
359                eprintln!("hermit log-diff: {message}");
360                return ExitStatus::Exited(2);
361            }
362            options.comparison = logdiff::LogComparisonMode::Info;
363            options.canonicalize_addresses = true;
364        }
365        if record_envelope.policy() == RecordEnvelopePolicy::CrossBackendDetcoreV1 {
366            options.require_structured_events = true;
367        }
368
369        if let Some(path) = &self.json
370            && let Err(error) = write_json(
371                path,
372                &pending_json_report(&options, record_envelope.policy()),
373            )
374        {
375            eprintln!(
376                "hermit log-diff: could not initialize JSON report at {}: {error}",
377                path.display()
378            );
379            return ExitStatus::Exited(2);
380        }
381
382        if self.follow {
383            return self.follow_two_runs(file_b, &options, record_envelope);
384        }
385
386        let comparison = if (self.canonical_info || self.json.is_some())
387            && record_envelope.policy() == RecordEnvelopePolicy::AllRecordsV1
388        {
389            try_bitwise_info_v1_with_records(&self.file_a, file_b, &options)
390                .map(|(summary, left, right)| (summary, left, right, None))
391        } else {
392            // The DBT transport envelope remains on its current named-filter
393            // path. It is a distinct canonical envelope, not an all-record
394            // BitwiseInfoV1 comparison.
395            try_log_diff_with_records(&self.file_a, file_b, &options, record_envelope)
396                .map(|(summary, left, right, inputs)| (summary, left, right, Some(inputs)))
397        };
398        let (summary, records_left, records_right, inputs) = match comparison {
399            Ok(result) => result,
400            Err(error) => {
401                eprintln!(
402                    "hermit log-diff: could not compare {} and {}: {error}",
403                    self.file_a.display(),
404                    file_b.display()
405                );
406                // Replace the pending `no_result` with an explicit refusal.
407                // Leaving the pending report would tell a JSON consumer that
408                // no verdict was reached, which reads as "try again" -- but a
409                // refusal is deterministic and re-running changes nothing.
410                if let Some(path) = &self.json {
411                    let mut report = pending_json_report(&options, record_envelope.policy());
412                    report.verdict = LogDiffVerdict::Refused;
413                    report.refusal = Some(error.to_string());
414                    if let Err(write_error) = write_json(path, &report) {
415                        eprintln!(
416                            "hermit log-diff: could not record the refusal in {}: {write_error}",
417                            path.display()
418                        );
419                    }
420                }
421                return ExitStatus::Exited(2);
422            }
423        };
424        let records = LogDiffRecords {
425            compared: records_left.min(records_right),
426            available_left: records_left,
427            available_right: records_right,
428            withheld_incomplete_tail: false,
429        };
430        eprintln!(
431            "hermit log-diff: read {} records in full (left {} | right {}); selected {} | {} messages under {}",
432            records.compared,
433            records_left,
434            records_right,
435            summary.compared_left,
436            summary.compared_right,
437            record_envelope.policy().as_str(),
438        );
439        let mut report = json_report(&summary, &options, records, record_envelope.policy());
440        report.inputs = inputs;
441        if let Some(path) = &self.json
442            && let Err(error) = write_json(path, &report)
443        {
444            eprintln!(
445                "hermit log-diff: could not write JSON report to {}: {error}",
446                path.display()
447            );
448            return ExitStatus::Exited(2);
449        }
450        if let Some(reason) = summary.refusal_reason.as_deref() {
451            eprintln!("hermit log-diff: REFUSED: {reason}");
452            ExitStatus::Exited(2)
453        } else if summary.diff_found {
454            ExitStatus::Exited(HERMIT_VERIFICATION_DIVERGENCE_EXIT)
455        } else if summary.matched_with_evidence() {
456            ExitStatus::Exited(0)
457        } else {
458            eprintln!("hermit log-diff: no comparable messages; refusing to report a match");
459            ExitStatus::Exited(2)
460        }
461    }
462
463    /// Compare two runs while they are still being written.
464    ///
465    /// Exit status is deliberately four-valued, because "they agree", "they
466    /// agree as far as I read", and "I ran out of time before either finished"
467    /// are three different answers and only one of them is good news:
468    ///
469    /// * 0 -- both logs stopped growing and never diverged.
470    /// * 1 -- a divergence was found; the record index bounding it is reported.
471    /// * 2 -- the logs could not be read, or held nothing comparable.
472    /// * 3 -- the timeout expired while at least one run was still writing.
473    fn follow_two_runs(
474        &self,
475        file_b: &Path,
476        options: &logdiff::LogDiffOpts,
477        record_envelope: RecordEnvelope,
478    ) -> ExitStatus {
479        let interval = Duration::from_millis(self.follow_interval_ms.max(1));
480        let settle_polls = self.follow_settle_polls.max(1);
481        let deadline = (self.follow_timeout_secs > 0)
482            .then(|| Instant::now() + Duration::from_secs(self.follow_timeout_secs));
483
484        let fixed_bitwise_info_v1 = (self.canonical_info || self.json.is_some())
485            && record_envelope.policy() == RecordEnvelopePolicy::AllRecordsV1;
486        let mut previous_sizes: Option<(usize, usize)> = None;
487        let mut unchanged_polls = 0u32;
488        let mut last_reported_records = usize::MAX;
489
490        loop {
491            let (left, right) = match (read_growing_log(&self.file_a), read_growing_log(file_b)) {
492                (Ok(left), Ok(right)) => (left, right),
493                (Err(error), _) | (_, Err(error)) => {
494                    eprintln!("hermit log-diff: could not read logs while following: {error}");
495                    return ExitStatus::Exited(2);
496                }
497            };
498
499            // Buffer the comparison transcript: printing it on every poll would
500            // bury the one line that matters.
501            let mut transcript = Vec::new();
502            let comparison = match compare_complete_prefix(
503                &left,
504                &right,
505                options,
506                &mut transcript,
507                record_envelope,
508                fixed_bitwise_info_v1,
509            ) {
510                Ok(comparison) => comparison,
511                Err(error) => {
512                    let reason = error.to_string();
513                    eprintln!("hermit log-diff: comparison failed while following: {reason}");
514                    if fixed_bitwise_info_v1 && let Some(path) = &self.json {
515                        let mut report = pending_json_report(options, record_envelope.policy());
516                        report.verdict = LogDiffVerdict::Refused;
517                        report.refusal = Some(reason);
518                        if let Err(write_error) = write_json(path, &report) {
519                            eprintln!(
520                                "hermit log-diff: could not record the refusal in {}: {write_error}",
521                                path.display()
522                            );
523                        }
524                    }
525                    return ExitStatus::Exited(2);
526                }
527            };
528            let records = LogDiffRecords {
529                compared: comparison.records_compared,
530                available_left: comparison.records_available_left,
531                available_right: comparison.records_available_right,
532                withheld_incomplete_tail: true,
533            };
534
535            if records.compared != last_reported_records {
536                eprintln!(
537                    "hermit log-diff: checked {} complete records (left has {} | right has {}); selected {} | {} messages under {}",
538                    records.compared,
539                    records.available_left,
540                    records.available_right,
541                    comparison.summary.compared_left,
542                    comparison.summary.compared_right,
543                    record_envelope.policy().as_str(),
544                );
545                last_reported_records = records.compared;
546            }
547
548            if let Some(reason) = comparison.summary.refusal_reason.as_deref() {
549                std::io::stderr().write_all(&transcript).ok();
550                eprintln!("hermit log-diff: REFUSED: {reason}");
551                return self.finish_follow(
552                    options,
553                    &comparison.summary,
554                    records,
555                    LogDiffVerdict::Refused,
556                    "refused",
557                    ExitStatus::Exited(2),
558                );
559            } else if comparison.summary.diff_found {
560                std::io::stderr().write_all(&transcript).ok();
561                let first = comparison.summary.first_divergent_record;
562                match first {
563                    Some(index) => eprintln!(
564                        "hermit log-diff: DIVERGED at record {index} of {} compared",
565                        records.compared
566                    ),
567                    None => eprintln!(
568                        "hermit log-diff: DIVERGED within the first {} records compared",
569                        records.compared
570                    ),
571                }
572                return self.finish_follow(
573                    options,
574                    &comparison.summary,
575                    records,
576                    LogDiffVerdict::Diverged,
577                    "diverged",
578                    ExitStatus::Exited(HERMIT_VERIFICATION_DIVERGENCE_EXIT),
579                );
580            }
581
582            let sizes = (left.len(), right.len());
583            if previous_sizes == Some(sizes) {
584                unchanged_polls += 1;
585            } else {
586                unchanged_polls = 0;
587            }
588            previous_sizes = Some(sizes);
589
590            if unchanged_polls >= settle_polls {
591                if comparison.summary.matched_with_evidence() {
592                    eprintln!(
593                        "hermit log-diff: both logs unchanged for {settle_polls} polls; \
594                         identical over {} | {} selected messages from {} complete records",
595                        comparison.summary.compared_left,
596                        comparison.summary.compared_right,
597                        records.compared,
598                    );
599                    return self.finish_follow(
600                        options,
601                        &comparison.summary,
602                        records,
603                        LogDiffVerdict::IdenticalSoFar,
604                        "quiescent",
605                        ExitStatus::Exited(0),
606                    );
607                }
608                eprintln!(
609                    "hermit log-diff: both logs unchanged for {settle_polls} polls after reading {} \
610                     complete records, but selected {} | {} comparable messages; refusing to report a match",
611                    records.compared,
612                    comparison.summary.compared_left,
613                    comparison.summary.compared_right,
614                );
615                return self.finish_follow(
616                    options,
617                    &comparison.summary,
618                    records,
619                    LogDiffVerdict::NoComparableMessages,
620                    "quiescent",
621                    ExitStatus::Exited(2),
622                );
623            }
624
625            if deadline.is_some_and(|deadline| Instant::now() >= deadline) {
626                eprintln!(
627                    "hermit log-diff: timed out after {}s with {} complete records read and {} | {} \
628                     messages compared while at least one run was still writing; NOT a match -- the rest was never read",
629                    self.follow_timeout_secs,
630                    records.compared,
631                    comparison.summary.compared_left,
632                    comparison.summary.compared_right,
633                );
634                return self.finish_follow(
635                    options,
636                    &comparison.summary,
637                    records,
638                    LogDiffVerdict::IdenticalSoFar,
639                    "timeout",
640                    ExitStatus::Exited(3),

649        &self,
650        options: &logdiff::LogDiffOpts,
651        summary: &logdiff::LogDiffSummary,
652        records: LogDiffRecords,
653        verdict: LogDiffVerdict,
654        stopped_because: &'static str,
655        status: ExitStatus,

674        }

679fn canonical_comparison_is_unrelaxed(options: &logdiff::LogDiffOpts) -> Result<(), &'static str> {
680    let defaults = logdiff::LogDiffOpts::default();
681    if options.strip_lines {
682        return Err("canonical INFO comparison conflicts with --unsafe-strip-lines");
683    }
684    if !options.ignore_lines.is_empty() {
685        return Err("canonical INFO comparison does not permit --ignore-lines");
686    }
687    if options.skip_commit || options.skip_detlog {
688        return Err("canonical INFO comparison does not permit message skipping");
689    }
690    if options.git_diff {
691        return Err("canonical INFO JSON does not support --git-diff");
692    }
693    if options.include_detlogs != defaults.include_detlogs {
694        return Err("canonical INFO comparison does not permit DETLOG filtering");
695    }
696    Ok(())
697}

699fn json_report(
700    summary: &logdiff::LogDiffSummary,
701    options: &logdiff::LogDiffOpts,
702    records: LogDiffRecords,
703    record_envelope: RecordEnvelopePolicy,
704) -> LogDiffReport {
705    let verdict = if summary.refusal_reason.is_some() {
706        LogDiffVerdict::Refused
707    } else if summary.diff_found {
708        LogDiffVerdict::Diverged
709    } else if summary.compared_left == 0 && summary.compared_right == 0 {
710        LogDiffVerdict::NoComparableMessages
711    } else {
712        LogDiffVerdict::Matched
713    };
714    let stream = match options.comparison {
715        logdiff::LogComparisonMode::Deterministic => "deterministic",
716        logdiff::LogComparisonMode::Info => "info",
717        logdiff::LogComparisonMode::FullTrace => "full_trace",
718    };
719    let included_detlog_kinds = options
720        .include_detlogs
721        .iter()
722        .map(|kind| match kind {
723            logdiff::DetLogFilter::Syscall => "syscall".to_owned(),
724            logdiff::DetLogFilter::SyscallResult => "syscall_result".to_owned(),
725            logdiff::DetLogFilter::Other => "other".to_owned(),
726        })
727        .collect();
728    LogDiffReport {
729        schema: LOG_DIFF_REPORT_SCHEMA,
730        verdict,
731        refusal: summary.refusal_reason.clone(),
732        selected_messages: LogDiffMessageCounts {
733            left: summary.compared_left,
734            right: summary.compared_right,
735        },
736        records,
737        inputs: None,
738        follow_stopped_because: None,
739        first_divergent_record: summary.first_divergent_record,
740        // A refused or empty comparison measured no prefix at all; report that
741        // as absent rather than as a zero-length match. Both conditions are
742        // checked: a refusal is absent whatever counts it carries.
743        matched_prefix_records: summary.matched_prefix_messages.filter(|_| {
744            summary.refusal_reason.is_none()
745                && (summary.compared_left > 0 || summary.compared_right > 0)
746        }),
747        first_divergent_syscall: summary.first_divergent_syscall,
748        comparison: LogDiffComparison {
749            stream: stream.to_owned(),
750            record_envelope,
751            unsafe_strip_lines: options.strip_lines,
752            canonicalize_host_addresses: options.canonicalize_addresses,
753            require_structured_events: options.require_structured_events,
754            ignored_line_substrings: options.ignore_lines.clone(),
755            skip_commit: options.skip_commit,
756            skip_detlog: options.skip_detlog,
757            included_detlog_kinds,
758            git_diff: options.git_diff,
759        },
760        first_divergent_scheduler_turn: summary.first_divergent_scheduler_turn,
761        first_divergent_virtual_nanoseconds: summary.first_divergent_virtual_nanoseconds,
762        first_divergent_left_message: summary.first_divergent_left_message.clone(),
763        first_divergent_right_message: summary.first_divergent_right_message.clone(),
764    }
765}

809fn write_json(path: &Path, report: &LogDiffReport) -> std::io::Result<()> {
810    let directory = path
811        .parent()
812        .filter(|parent| !parent.as_os_str().is_empty())
813        .unwrap_or_else(|| Path::new("."));
814    let mut temporary = NamedTempFile::new_in(directory)?;
815    serde_json::to_writer(&mut temporary, report).map_err(std::io::Error::other)?;
816    temporary.write_all(b"\n")?;
817    temporary.flush()?;
818    temporary
819        .persist(path)
820        .map_err(|error| error.error)
821        .map(|_| ())
822}
