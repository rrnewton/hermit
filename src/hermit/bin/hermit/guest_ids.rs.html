9//! How a `--no-namespace` run numbers its guest (`--pid-base`).
10//!
11//! Without a PID namespace of its own, a guest would see the host's next free
12//! process and thread IDs, different in every run. Where Hermit may choose
13//! IDs (`CAP_SYS_ADMIN` or `CAP_CHECKPOINT_RESTORE` over its PID namespace:
14//! root, or the root of an enclosing user namespace that owns it), the
15//! guest's first process gets the base ID and Detcore gives each later task
16//! the next one (`GuestIds::Chosen`).
17//!
18//! The ID space is cut into aligned windows of [`window_ids`] IDs. A run
19//! takes the window or windows its IDs start in, checked free of host tasks,
20//! and reserves them for its lifetime ([`Reservation`]), so that two Hermit
21//! runs on one host do not start in the same IDs. IDs past those windows are
22//! neither checked nor reserved (<https://github.com/rrnewton/hermit/issues/4870>).
23//!
24//! A run that asks for chosen IDs, with an explicit `--pid-base` or with
25//! `--strict` (whose base, unless given, is half of `pid_max`, a fact of the
26//! host's configuration rather than its state), is refused when it cannot
27//! have them, naming the cause. Any other run first checks that it may
28//! choose IDs at all, then takes the first free window it can reserve,
29//! starting just behind the kernel's PID allocation cursor (so that the host
30//! reaches it last) and going down, then from the top; it reports the base it
31//! took. With none, or without the capability, it warns and the guest gets
32//! host IDs.
33//!
34//! Chosen IDs also require the host to validate a legacy clone's original
35//! exit signal before the conversion to clone3 can replace it. Admission
36//! probes that behavior; a kernel version string does not establish it.
37

45/// How many IDs a run's window holds: what Hermit checks free and reserves,
46/// and what a guest's tasks stay within unless it creates more (they go on
47/// past it, and an ID found taken then refuses or warns as Detcore's
48/// `inject_with_chosen_id` says). 1024 where `pid_max` is at most 65536: an
49/// Android device's 32768 has 31 windows above [`RESERVED_PIDS`]. 256 where it
50/// is larger: a long-running host's tasks are spread over its whole ID space,
51/// and one with 37,000 threads and `pid_max` 4,194,304 has a free window of
52/// 256 about one time in ten, and of 1024 almost never.
53pub(super) fn window_ids(pid_max: i32) -> i32 {
54    if pid_max <= 65536 { 1024 } else { 256 }
55}
56
57/// Linux's `RESERVED_PIDS`: below this the kernel does not wrap, and no
58/// default window is sought.
59const RESERVED_PIDS: i32 = 300;
60
61/// How the guest of a run is numbered, and the warning to print when it
62/// takes host IDs: from `--no-namespace`, `--pid-base` and `--strict`.
63/// `Err` is a usage error, or the [`PolicyRefusal`] of a run that asked for
64/// chosen IDs and cannot have them. The windows the run takes stay reserved
65/// until this process exits.
66pub(super) fn resolve(
67    no_namespace: bool,
68    required: Option<Required<'_>>,
69    strict: bool,
70) -> Result<(GuestIds, Option<String>), Error> {
71    resolve_with(no_namespace, required, strict, &Host)
72}

74/// A base the run must have, and why.
75#[derive(Clone, Copy, Debug)]
76pub(super) enum Required<'a> {
77    /// The run gave `--pid-base`.
78    PidBase(i32),
79    /// The run replays a record (`source`, an option and a path) whose
80    /// guest IDs were chosen from `base`.
81    Replay { source: &'a str, base: i32 },
82}
83
84impl Required<'_> {
85    fn base(&self) -> i32 {
86        match *self {
87            Required::PidBase(base) | Required::Replay { base, .. } => base,
88        }
89    }
90}

92/// The host facts [`resolve`] reads, replaced in tests.
93trait HostIds {
94    /// `/proc/sys/kernel/pid_max`.
95    fn pid_max(&self) -> Result<i32, String>;
96    /// The IDs of every host task `/proc` shows: each process's threads.
97    fn taken(&self) -> BTreeSet<i32>;
98    /// What holds `id` on the host.
99    fn holder(&self, id: i32) -> String;
100    /// Whether legacy clone validates its original exit signal before the
101    /// chosen-ID conversion may replace that signal.
102    fn legacy_clone_signal_validation(&self) -> Result<(), String>;
103    /// Whether this process can create a child with process ID `id`.
104    fn can_choose(&self, id: i32) -> Result<(), String>;
105    /// Whether this process may choose IDs at all: [`Self::can_choose`],
106    /// where `id` being taken (`EEXIST`) also shows the permission, which
107    /// Linux checks first (`alloc_pid`).
108    fn may_choose(&self, id: i32) -> Result<(), String>;
109    /// The ID the kernel would give a new task next, if known.
110    fn cursor(&self) -> Option<i32>;
111    /// Reserve window number `index` of `size` IDs for this process's
112    /// lifetime; `Err` when another holds it.
113    fn reserve(&self, index: i32, size: i32) -> Result<(), String>;
114}

116struct Host;

118impl HostIds for Host {
119    fn pid_max(&self) -> Result<i32, String> {
120        let text = std::fs::read_to_string("/proc/sys/kernel/pid_max")
121            .map_err(|error| format!("cannot read /proc/sys/kernel/pid_max: {error}"))?;
122        text.trim()
123            .parse()
124            .map_err(|error| format!("cannot parse /proc/sys/kernel/pid_max {text:?}: {error}"))
125    }
126
127    fn taken(&self) -> BTreeSet<i32> {
128        let numbers = |dir: &str| {
129            std::fs::read_dir(dir)
130                .into_iter()
131                .flatten()
132                .flatten()
133                .filter_map(|entry| entry.file_name().to_str()?.parse::<i32>().ok())
134                .collect::<Vec<_>>()
135        };
136        numbers("/proc")
137            .into_iter()
138            .flat_map(|pid| {
139                // A process that exits meanwhile still held its own ID.
140                let mut tasks = numbers(&format!("/proc/{pid}/task"));
141                tasks.push(pid);
142                tasks
143            })
144            .collect()
145    }
146
147    fn holder(&self, id: i32) -> String {
148        detcore::chosen_id_holder(id)
149    }
150
151    fn legacy_clone_signal_validation(&self) -> Result<(), String> {
152        check_legacy_clone_signal_validation(probe_legacy_clone_errno)
153    }
154
155    fn can_choose(&self, id: i32) -> Result<(), String> {
156        probe_chosen_pid(id).map_err(|(_, cause)| cause)
157    }
158
159    fn may_choose(&self, id: i32) -> Result<(), String> {
160        match probe_chosen_pid(id) {
161            Err((Some(libc::EEXIST), _)) => Ok(()),
162            result => result.map_err(|(_, cause)| cause),
163        }
164    }
165
166    fn cursor(&self) -> Option<i32> {
167        probe_cursor()
168    }
169
170    fn reserve(&self, index: i32, size: i32) -> Result<(), String> {
171        Reservation::take(index, size)
172    }
173}

175/// A window of IDs this process holds for its lifetime: an abstract Unix
176/// socket named after it and the PID namespace whose IDs it holds
177/// (`hermit-guest-pid-window:<namespace inode>:<size>:<index>`), bound
178/// with close-on-exec and never closed, so the kernel releases it when Hermit
179/// exits. A second run that binds the same name gets `EADDRINUSE`.
180///
181/// Abstract socket names belong to a network namespace: the reservation
182/// protects runs that share one, which every `--no-namespace` run on a host
183/// does unless it was started inside another (on Android, some services and
184/// apps run in their own). Runs in different network namespaces can still
185/// take the same window; the later one's first process then cannot have its
186/// ID, and the run ends with Hermit's policy-refusal status, naming it.
187struct Reservation;
188
189impl Reservation {
190    fn take(index: i32, size: i32) -> Result<(), String> {
191        use std::os::linux::net::SocketAddrExt;
192        // Runs in different PID namespaces have separate IDs: only those in
193        // this one contend. Without PID namespaces there is only one.
194        let namespace = std::fs::read_link("/proc/self/ns/pid")
195            .ok()
196            .and_then(|link| {
197                let link = link.to_string_lossy().into_owned();
198                link.strip_prefix("pid:[")?
199                    .strip_suffix(']')
200                    .map(str::to_owned)
201            })
202            .unwrap_or_else(|| "0".to_owned());
203        let name = format!("hermit-guest-pid-window:{namespace}:{size}:{index}");
204        let address = std::os::unix::net::SocketAddr::from_abstract_name(name.as_bytes())
205            .map_err(|error| format!("cannot name the reservation {name}: {error}"))?;
206        let socket = std::os::unix::net::UnixDatagram::bind_addr(&address).map_err(|error| {
207            if error.raw_os_error() == Some(libc::EADDRINUSE) {
208                format!(
209                    "another Hermit run holds the {size} IDs from {} (reservation @{name})",
210                    index * size
211                )
212            } else {
213                format!("cannot reserve the IDs from {}: {error}", index * size)
214            }
215        })?;
216        // Held until this process exits.
217        std::mem::forget(socket);
218        Ok(())
219    }
220}

222fn resolve_with(
223    no_namespace: bool,
224    required: Option<Required<'_>>,
225    strict: bool,
226    host: &dyn HostIds,
227) -> Result<(GuestIds, Option<String>), Error> {
228    if !no_namespace {
229        return Ok((GuestIds::Namespace, None));
230    }
231    let pid_max = host.pid_max().map_err(Error::msg)?;
232    // A base the kernel can never give: a usage error when given, a refusal
233    // when a record asks for it (it was recorded on a host with a larger
234    // pid_max).
235    match required {
236        Some(Required::PidBase(base)) if base < 1 || base >= pid_max => anyhow::bail!(
237            "--pid-base={base} is not a process ID this host can give: it must be at least 1 \
238             and below pid_max, {pid_max}"
239        ),
240        Some(Required::Replay { source, base }) if base < 1 || base >= pid_max => {
241            return Err(Error::new(PolicyRefusal).context(format!(
242                "{source} was recorded with guest process IDs chosen from {base}, which this \
243                 host cannot give (its pid_max is {pid_max}), so the replay cannot follow it. \
244                 Record the run again on this host"
245            )));
246        }
247        _ => {}
248    }
249    // The conversion in Detcore preserves early signal validation only on
250    // hosts that demonstrate it. Refuse/fall back before drawing the cursor,
251    // reserving windows or probing a chosen PID; Namespace returned above.
252    if let Err(cause) = host.legacy_clone_signal_validation() {
253        if required.is_none() && !strict {
254            return Ok((
255                GuestIds::Host,
256                Some(host_ids_warning(&format!(
257                    "{cause}. Use a host that passes this validation probe, or omit \
258                     --no-namespace to use a PID namespace"
259                ))),
260            ));
261        }
262        let asked = match required {
263            Some(Required::PidBase(_)) => "--pid-base",
264            Some(Required::Replay { source, .. }) => source,
265            None => "--strict",
266        };
267        return Err(Error::new(PolicyRefusal).context(format!(
268            "{asked} with --no-namespace requires chosen guest process IDs, but Hermit cannot \
269             safely choose them: {cause}. Use a host that passes this validation probe. For a \
270             new recording or run, omit --no-namespace and --pid-base to use a PID namespace"
271        )));
272    }
273    let pid_base = required.map(|required| required.base());
274    let size = window_ids(pid_max);
275    let cursor = host.cursor().unwrap_or(pid_max / 2).clamp(1, pid_max - 1);
276    // The cheap check first: a run that may not choose IDs warns without
277    // reading every host task.
278    if pid_base.is_none()
279        && !strict
280        && let Err(cause) = host.may_choose(cursor)
281    {
282        return Ok((GuestIds::Host, Some(host_ids_warning(&cause))));
283    }
284    let taken = host.taken();
285    let window = |base: i32| base..base.saturating_add(size).min(pid_max);
286    let holder_in = |base: i32| taken.range(window(base)).next().copied();
287    if pid_base.is_some() || strict {
288        let base = pid_base.unwrap_or(pid_max / 2);
289        // Every window the run's first IDs fall in.
290        let windows = base / size..=(window(base).end - 1) / size;
291        let cause = match holder_in(base) {
292            Some(id) => Some(format!(
293                "the {} IDs from {base} must be free, but {} holds one of them",
294                window(base).len(),
295                host.holder(id)
296            )),
297            None => windows
298                .into_iter()
299                .find_map(|index| host.reserve(index, size).err())
300                .or_else(|| host.can_choose(base).err()),
301        };
302        return match cause {
303            None => Ok((
304                GuestIds::Chosen {
305                    base,
306                    refuse_host_ids: true,
307                },
308                None,
309            )),
310            Some(cause) => Err(Error::new(PolicyRefusal).context(match required {
311                Some(Required::Replay { source, .. }) => format!(
312                    "{source} was recorded with guest process IDs chosen from {base}, and this \
313                     replay needs exactly those IDs but cannot have them: {cause}. Stop what \
314                     holds them, run as a user that may choose process IDs \
315                     (CAP_CHECKPOINT_RESTORE or CAP_SYS_ADMIN), or record the run again"
316                ),
317                _ => {
318                    let asked = if pid_base.is_some() {
319                        "--pid-base"
320                    } else {
321                        "--strict"
322                    };
323                    format!(
324                        "{asked} with --no-namespace needs the guest's process IDs chosen from \
325                         {base}, and Hermit cannot choose them: {cause}. Pass a --pid-base whose \
326                         IDs no host task or other run holds, or run as a user that may choose \
327                         process IDs (CAP_CHECKPOINT_RESTORE or CAP_SYS_ADMIN)"
328                    )
329                }
330            })),
331        };
332    }
333    // Windows wholly inside RESERVED_PIDS..pid_max: just behind the cursor's
334    // and down, then from the top down to the cursor's own, last.
335    let first = (RESERVED_PIDS + size - 1) / size;
336    let last = pid_max / size - 1;
337    // A pid_max under two windows (Linux allows down to 301) leaves no whole
338    // window past the reserved IDs: no base is chosen, as when none is free.
339    let free = if first > last {
340        None
341    } else {
342        let own = (cursor / size).clamp(first, last);
343        let order = (first..own).rev().chain((own..=last).rev());
344        order
345            .filter(|index| holder_in(index * size).is_none())
346            .find(|index| host.reserve(*index, size).is_ok())
347    };
348    match free {
349        Some(index) => Ok((
350            GuestIds::Chosen {
351                base: index * size,
352                refuse_host_ids: false,
353            },
354            None,
355        )),
356        None => Ok((
357            GuestIds::Host,
358            Some(host_ids_warning(&format!(
359                "no window of {size} IDs below pid_max, {pid_max}, is free of host tasks and of \
360                 other Hermit runs"
361            ))),
362        )),
363    }
364}

366/// The warning of a run whose guest gets host IDs, for `cause`.
367fn host_ids_warning(cause: &str) -> String {
368    format!(
369        "WARNING: --no-namespace: the guest gets host process IDs, which differ from run to run, \
370         because Hermit cannot choose them: {cause}. Pass --pid-base to refuse such a run instead"
371    )
372}

374/// Admission needs both observations: an invalid signal's EINVAL alone
375/// could instead come from an unrelated validation or policy failure.
376fn check_legacy_clone_signal_validation(
377    mut probe: impl FnMut(i32) -> Result<i32, String>,
378) -> Result<(), String> {
379    let valid = probe(libc::SIGCHLD);
380    let invalid = probe(65);
381    if valid == Ok(libc::EFAULT) && invalid == Ok(libc::EINVAL) {
382        return Ok(());
383    }
384    let describe = |result: Result<i32, String>| match result {
385        Ok(libc::EFAULT) => format!("EFAULT ({})", libc::EFAULT),
386        Ok(libc::EINVAL) => format!("EINVAL ({})", libc::EINVAL),
387        Ok(errno) => format!(
388            "errno {errno} ({})",
389            std::io::Error::from_raw_os_error(errno)
390        ),
391        Err(cause) => format!("probe failed: {cause}"),
392    };
393    Err(format!(
394        "legacy clone exit-signal validation is unsupported or unproven: SIGCHLD probe returned \
395         {}, signal 65 probe returned {}; expected EFAULT ({}) then EINVAL ({})",
396        describe(valid),
397        describe(invalid),
398        libc::EFAULT,
399        libc::EINVAL,
400    ))
401}

403/// Use CLONE_PIDFD with an output address outside the Linux user address
404/// range, so copy_process fails before kernel_clone can wake a new task.
405/// Unlike a null pointer, usize::MAX does not rely on the zero page being
406/// unmapped. No CLONE_PARENT means neither init restrictions nor reparenting
407/// can confuse the probe.
408///
409/// Linux v6.16 copy_process writes the pidfd at kernel/fork.c:2210:
410/// <https://git.kernel.org/pub/scm/linux/kernel/git/stable/linux.git/tree/kernel/fork.c?id=038d61fd642278bab63ee8ef722c50d10ab01e8f#n2210>.
411/// Failure releases the pidfd/PID at line 2430 and returns an error, before
412/// kernel_clone could wake the task at line 2630. Its kernel_clone lacks
413/// original-signal validation, so both probes reach EFAULT. Linux v7.1.3
414/// kernel_clone checks the signal first, yielding EINVAL for 65:
415/// <https://git.kernel.org/pub/scm/linux/kernel/git/stable/linux.git/tree/kernel/fork.c?id=199c9959d3a9b53f346c221757fc7ac507fbac50#n2702>.
416/// This may allocate/release an internal host PID before EFAULT; it allocates
417/// no guest-chosen ID. Unexpected success is reaped and never admits Chosen.
418fn probe_legacy_clone_errno(signal: i32) -> Result<i32, String> {
419    // SAFETY: a fork-like clone without CLONE_VM. The invalid output pointer
420    // should prevent creation; a defensive child path only calls _exit.
421    let child = unsafe {
422        libc::syscall(
423            libc::SYS_clone,
424            (libc::CLONE_PIDFD | signal) as libc::c_ulong,
425            0usize,
426            usize::MAX,
427            0usize,
428            0usize,
429        )
430    };
431    if child == 0 {
432        unsafe { libc::_exit(0) }
433    }
434    if child < 0 {
435        return std::io::Error::last_os_error()
436            .raw_os_error()
437            .ok_or_else(|| "legacy clone failed without an OS error number".to_owned());
438    }
439    let mut status = 0;
440    // SAFETY: wait only for this unexpected child, including a clone child
441    // whose exit signal was not SIGCHLD. Its only userspace action is _exit.
442    while unsafe { libc::waitpid(child as libc::pid_t, &mut status, libc::__WALL) } < 0 {
443        let error = std::io::Error::last_os_error();
444        if error.raw_os_error() != Some(libc::EINTR) {
445            return Err(format!(
446                "legacy clone unexpectedly created child {child}; could not reap it: {error}"
447            ));
448        }
449    }
450    Err(format!(
451        "legacy clone unexpectedly created child {child}; reaped with wait status {status}"
452    ))
453}

455/// The ID the kernel gives a new task next: that of a child made with a
456/// plain fork-like `clone` that exits at once and is reaped here.
457fn probe_cursor() -> Option<i32> {
458    // SAFETY: a fork-like clone without CLONE_VM. The child only calls
459    // _exit, which is async-signal-safe.
460    let child = unsafe { libc::syscall(libc::SYS_clone, libc::SIGCHLD, 0, 0, 0, 0) };
461    if child == 0 {
462        unsafe { libc::_exit(0) }
463    }
464    if child < 0 {
465        return None;
466    }
467    let mut status = 0;
468    // SAFETY: waits for the child created above.
469    while unsafe { libc::waitpid(child as libc::pid_t, &mut status, libc::__WALL) } < 0 {
470        if std::io::Error::last_os_error().raw_os_error() != Some(libc::EINTR) {
471            break;
472        }
473    }
474    i32::try_from(child).ok().map(|child| child + 1)
475}

477/// Whether this process can create a child with process ID `id` in its own
478/// PID namespace: a child made with `clone3`'s `set_tid` that exits at once.
479/// The ID is free again once the child is reaped, which this does before it
480/// returns.
481fn probe_chosen_pid(id: i32) -> Result<(), (Option<i32>, String)> {
482    // struct clone_args through set_tid_size (CLONE_ARGS_SIZE_VER1).
483    let set_tid: libc::pid_t = id;
484    let mut args = [0u64; 10];
485    args[4] = libc::SIGCHLD as u64; // exit_signal
486    args[8] = std::ptr::from_ref(&set_tid) as u64; // set_tid
487    args[9] = 1; // set_tid_size
488    // SAFETY: a fork-like clone3 without CLONE_VM. The child only calls
489    // _exit, which is async-signal-safe.
490    let child = unsafe {
491        libc::syscall(
492            libc::SYS_clone3,
493            args.as_ptr(),
494            std::mem::size_of_val(&args),
495        )
496    };
497    if child == 0 {
498        unsafe { libc::_exit(0) }
499    }
500    if child < 0 {
501        let error = std::io::Error::last_os_error();
502        let errno = error.raw_os_error();
503        return Err((
504            errno,
505            match errno {
506                Some(libc::EPERM) => format!(
507                    "this user may not choose process IDs (clone3 set_tid: {error}): that takes \
508                     CAP_CHECKPOINT_RESTORE or CAP_SYS_ADMIN over the host's PID namespace"
509                ),
510                Some(libc::ENOSYS) => format!(
511                    "clone3 is unavailable (Linux before 5.3, or a seccomp filter refuses it): \
512                     {error}"
513                ),
514                Some(libc::E2BIG) => {
515                    format!("this kernel's clone3 has no set_tid (Linux 5.3 and 5.4): {error}")
516                }
517                _ => format!("clone3 with set_tid {id} failed: {error}"),
518            },
519        ));
520    }
521    let mut status = 0;
522    // SAFETY: waits for the child created above.
523    while unsafe { libc::waitpid(child as libc::pid_t, &mut status, libc::__WALL) } < 0 {
524        let error = std::io::Error::last_os_error();
525        if error.raw_os_error() != Some(libc::EINTR) {
526            return Err((
527                None,
528                format!("cannot reap the probe child {child}: {error}"),
529            ));
530        }
531    }
532    Ok(())
533}

584        }
585        fn can_choose(&self, id: i32) -> Result<(), String> {
586            self.probed.borrow_mut().push(id);
587            self.can_choose.map_err(str::to_owned)
588        }
589        fn may_choose(&self, id: i32) -> Result<(), String> {
590            self.can_choose(id)
591        }
592        fn cursor(&self) -> Option<i32> {
593            self.cursor_reads.set(self.cursor_reads.get() + 1);
594            self.cursor
595        }
596        fn reserve(&self, index: i32, size: i32) -> Result<(), String> {
597            if self.held.borrow_mut().insert(index) {
598                Ok(())
599            } else {
600                Err(format!(
601                    "another Hermit run holds the {size} IDs from {}",
602                    index * size
603                ))
604            }
605        }
606    }

610            base,
611            refuse_host_ids,
612        }

696        let mut signals = Vec::new();
697        assert!(
698            check_legacy_clone_signal_validation(|signal| {
699                signals.push(signal);
700                Ok(if signal == libc::SIGCHLD {
701                    libc::EFAULT
702                } else {
703                    libc::EINVAL
704                })
705            })

707        );

709
710        // The older source's pair, unrelated EINVAL, and failed probes
711        // cannot establish early original-signal validation.

714                [Ok(libc::EFAULT), Ok(libc::EFAULT)],
715                ["EFAULT (14)", "EFAULT (14)"],

718                [Ok(libc::EINVAL), Ok(libc::EINVAL)],
719                ["EINVAL (22)", "EINVAL (22)"],

730                [
731                    "EFAULT (14)",

798    /// Without a base or --strict, a run reserves the free window just behind
799    /// the cursor's, then the next one down, then from the top; a second run
800    /// on the host never gets a window the first holds.
801    #[test]
802    fn the_default_window_is_behind_the_cursor_and_reserved() {
803        let mut host = FakeHost::new();
804        // The cursor 20000 is in window 19 (19456..20480).
805        assert_eq!(
806            resolve_with(true, None, false, &host).unwrap(),
807            (chosen(18 * 1024, false), None)
808        );
809        // Held now: the next run on the host takes the next one down.
810        assert_eq!(
811            resolve_with(true, None, false, &host).unwrap().0,
812            chosen(17 * 1024, false)
813        );
814        // A host task in a window skips it.

816        assert_eq!(
817            resolve_with(true, None, false, &host).unwrap().0,
818            chosen(15 * 1024, false)
819        );
820        // Below the cursor all taken or held: from the top.
821        host.taken.extend((300..15 * 1024).step_by(500));
822        assert_eq!(
823            resolve_with(true, None, false, &host).unwrap().0,
824            chosen(31 * 1024, false)
825        );
826        // The capability is probed at the cursor, before the scan.
827        assert_eq!(host.probed.borrow().last(), Some(&20000));
