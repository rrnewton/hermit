9//! What analyze concludes when it names no event, the conclusions that end it
10//! before it can look for one, and the failures that keep it from concluding.
11//! An [`AnalyzeVerdict`] is a completed analysis (exit 0,
12//! `HERMIT_ANALYZE_VERDICT`); an [`AnalyzePrecondition`] says that the inputs
13//! do not give analyze what it needs (exit
14//! [`hermit::HERMIT_ANALYZE_PRECONDITION_EXIT`], `HERMIT_ANALYZE_PRECONDITION`);
15//! an [`AnalyzeFailure`] is a failure of Hermit or the host, still exit 125,
16//! but with a class line that says which. `main.rs`'s `failure_exit_code` and
17//! `classify_failure` map all three.

26/// What an analysis concluded: the `verdict` field of its report.
27#[derive(
28    Clone,
29    Copy,
30    Debug,
31    Default,
32    PartialEq,
33    Eq,
34    Hash,
35    Serialize,
36    Deserialize
37)]
38#[serde(rename_all = "snake_case")]
39pub enum Verdict {
40    /// The report names the two critical events (`critical_event1` and
41    /// `critical_event2`), and for a timing decision the deciding one.
42    #[default]
43    CriticalEvents,
44    /// The target is reproduced without the target schedule: without chaos,
45    /// the scheduler alone, replaying none of it, meets the target. A prefix
46    /// search needs a run that misses the target to start from, so it
47    /// localizes no event. This does not show that no event decides the
48    /// outcome (an interior prefix may still miss the target). The report
49    /// says so and names no event.
50    ReproducedWithoutSchedule,
51    /// A budget (`--time-budget`, `--trial-budget`) ended the analysis before
52    /// it concluded (exit 120): `partial` says what it found so far, and the
53    /// report names no decision. While the default search runs, both event
54    /// fields name where its closest pair of schedules first differs, which
55    /// is the deciding event or one before it; otherwise they are absent.
56    Partial,
57}
58
59/// Why analyze concluded [`Verdict::ReproducedWithoutSchedule`]: the `verdict_reason`
60/// field of its report, `{"cause": "scheduler_alone"}` (with
61/// `"timing_in_force"` when the trials keep timing settings), or
62/// `{"cause": "timing_semantics", "options": [...], "alternatives": [...],
63/// "trial": "..."}`. A cause a later build adds reads as `unknown`.
64#[derive(Clone, Debug, PartialEq, Eq, Hash, Serialize, Deserialize)]
65#[serde(rename_all = "snake_case", tag = "cause")]
66pub enum VerdictReason {
67    /// Without chaos, the scheduler alone meets the target.
68    SchedulerAlone {
69        /// The timing settings every trial keeps from the recording, when
70        /// there are any: they are in force, but not shown to cause the
71        /// outcome (the empty prefix under their alternatives met the target
72        /// too, or could not be tried).
73        #[serde(default, skip_serializing_if = "Vec::is_empty")]
74        timing_in_force: Vec<String>,
75    },
76    /// Without chaos, the scheduler alone meets the target because of timing
77    /// settings every trial keeps from the recording, its timing semantics:
78    /// the empty prefix under their alternatives (`trial`) missed it.
79    TimingSemantics {
80        /// The settings, as `hermit run` options (`--pipe-wake-latency=N`).
81        options: Vec<String>,
82        /// For each, the other value a new failing run can take
83        /// (`no --pipe-wake-latency`).
84        alternatives: Vec<String>,
85        /// The reproducer of the trial that showed it, in the workspace.
86        trial: String,
87    },
88    /// A cause this build does not know, from a report a later one wrote.
89    #[serde(other)]
90    Unknown,
91}
92
93/// A timing setting every trial keeps from the recording, and the other
94/// value a run can take instead.
95#[derive(Clone, Debug, PartialEq, Eq)]
96pub struct TimingSetting {
97    /// As a `hermit run` option, `--pipe-wake-latency=50000`.
98    pub option: String,
99    /// The other value, as a reader would ask for it, `no
100    /// --pipe-wake-latency`.
101    pub alternative: String,
102    /// The option a run under the alternative drops, `--pipe-wake-latency`.
103    pub drops: &'static str,
104    /// The argument it adds instead, if any: for a setting whose alternative
105    /// is another value, that value (`--pipe-wake-latency=1000`).
106    pub adds: Option<String>,
107}

109/// How the empty prefix fared, continued without chaos as the search's
110/// trials were, under the alternatives of the timing settings in force.
111#[derive(Clone, Debug, PartialEq, Eq)]
112pub enum AlternativeTiming {
113    /// It missed the target: the settings cause the outcome. `trial` names
114    /// its reproducer.
115    Missed { trial: String },
116    /// It met the target too.
117    Met { trial: String },
118    /// It was not tried, for the reason given.
119    Untried { why: String },
120}

122/// A completed analysis that names no event. Analyze writes its report and
123/// exits as for any other completed analysis (0, or `--success-exit-code`),
124/// with the class line [`AnalyzeVerdict::class_line`] on stderr.
125#[derive(Debug, Clone, PartialEq, Eq)]
126pub enum AnalyzeVerdict {
127    /// [`Verdict::ReproducedWithoutSchedule`], with the explanation analyze prints and
128    /// the report carries, and its reason.
129    ReproducedWithoutSchedule {
130        /// What analyze prints, and the report's header.
131        explanation: String,
132        /// The report's `verdict_reason`.
133        reason: VerdictReason,
134        /// The continuation timeslice at which the scheduler alone met the
135        /// target, which a trial under other timing settings uses too.
136        continuation: MaybeTimeslice,
137    },
138}
139
140impl AnalyzeVerdict {
141    /// [`Verdict::ReproducedWithoutSchedule`]: the scheduler alone meets the target,
142    /// continuing at `continuation`, as `explanation` says.
143    pub fn reproduced_without_schedule(
144        explanation: impl Into<String>,
145        continuation: MaybeTimeslice,
146    ) -> Self {
147        AnalyzeVerdict::ReproducedWithoutSchedule {
148            explanation: explanation.into(),
149            reason: VerdictReason::SchedulerAlone {
150                timing_in_force: Vec::new(),
151            },
152            continuation,
153        }
154    }
155
156    /// The continuation timeslice at which the scheduler alone met the
157    /// target.
158    pub fn continuation(&self) -> MaybeTimeslice {
159        match self {
160            AnalyzeVerdict::ReproducedWithoutSchedule { continuation, .. } => *continuation,
161        }
162    }
163
164    /// This verdict, reached by trials that all kept `timing` from the
165    /// recording, given how the empty prefix fared under the alternatives
166    /// (`tried`). Only a trial that missed the target shows the settings
167    /// cause the outcome ([`VerdictReason::TimingSemantics`], "because of");
168    /// otherwise the scheduler alone meets the target "under" them, which
169    /// are in force. Either way, analyzing under the alternatives needs a
170    /// new failing run: analyze does not replay a recording under other
171    /// values (and a record, stamped with its --pipe-wake-latency, is
172    /// refused under another). With no settings, the verdict is unchanged.
173    pub fn with_timing(self, timing: &[TimingSetting], tried: AlternativeTiming) -> Self {
174        let AnalyzeVerdict::ReproducedWithoutSchedule {
175            explanation,
176            continuation,
177            ..
178        } = self;
179        if timing.is_empty() {
180            return AnalyzeVerdict::reproduced_without_schedule(explanation, continuation);
181        }
182        let options: Vec<String> = timing.iter().map(|t| t.option.clone()).collect();
183        let alternatives: Vec<String> = timing.iter().map(|t| t.alternative.clone()).collect();
184        let (opts, alts) = (options.join(" and "), alternatives.join(" and "));
185        // The trial ran at the first continuation the search tried.
186        let at = describe_continuation(continuation);
187        let (explanation, reason) = match tried {
188            AlternativeTiming::Missed { trial } => (
189                format!(
190                    "{explanation} The scheduler alone meets the target because of {opts}, the \
191                     recording's timing semantics, which every trial keeps: the empty prefix \
192                     with {alts}, continued {at}, misses it ({trial} in the workspace). Under \
193                     {opts} the empty prefix already meets the target, so the prefix search has \
194                     no endpoint that misses it to start from and localizes no event; a two-seed \
195                     search against a passing seed can end the same way, as it falls back to \
196                     this search when it establishes no exact one-swap pair of schedules with \
197                     opposite outcomes. To name a deciding event, analyze a new failing run with \
198                     {alts} in the run arguments, if one fails there."
199                ),
200                VerdictReason::TimingSemantics {
201                    options,
202                    alternatives,
203                    trial,
204                },
205            ),
206            AlternativeTiming::Met { trial } => (
207                format!(
208                    "{explanation} The trials keep {opts} from the recording, but the empty \
209                     prefix with {alts}, continued {at}, meets the target too ({trial} in the \
210                     workspace), so the scheduler alone meets it under either."
211                ),
212                VerdictReason::SchedulerAlone {
213                    timing_in_force: options,
214                },
215            ),
216            AlternativeTiming::Untried { why } => (
217                format!(
218                    "{explanation} The scheduler alone meets the target under {opts}, in force \
219                     because every trial keeps them from the recording; whether it does with \
220                     {alts} was not tried: {why}. A new failing run with {alts} in the run \
221                     arguments can tell."
222                ),
223                VerdictReason::SchedulerAlone {
224                    timing_in_force: options,
225                },
226            ),
227        };
228        AnalyzeVerdict::ReproducedWithoutSchedule {
229            explanation,
230            reason,
231            continuation,
232        }
233    }
234
235    /// The report's `verdict`.
236    pub fn verdict(&self) -> Verdict {
237        match self {
238            AnalyzeVerdict::ReproducedWithoutSchedule { .. } => Verdict::ReproducedWithoutSchedule,
239        }
240    }
241
242    /// The report's `verdict_reason`.
243    pub fn reason(&self) -> &VerdictReason {
244        match self {
245            AnalyzeVerdict::ReproducedWithoutSchedule { reason, .. } => reason,
246        }
247    }
248
249    /// The line analyze prints on stderr, which a script can read without
250    /// parsing the report.
251    pub fn class_line(&self) -> String {
252        let class = match self {
253            AnalyzeVerdict::ReproducedWithoutSchedule { .. } => "reproduced-without-schedule",
254        };
255        format!("HERMIT_ANALYZE_VERDICT class={class}")
256    }
257}
258
259impl std::fmt::Display for AnalyzeVerdict {
260    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
261        match self {
262            AnalyzeVerdict::ReproducedWithoutSchedule { explanation, .. } => {
263                f.write_str(explanation)
264            }
265        }
266    }
267}

271/// Which input does not give analyze what it needs.
272#[derive(Clone, Copy, Debug, PartialEq, Eq)]
273pub enum PreconditionClass {
274    /// The target run, or a run of the schedule meant to reproduce it, does
275    /// not meet the target criteria: the first run without `--search`, the
276    /// failing endpoint of a bisection (`hermit bisect`'s `--bad` too) and
277    /// `--minimize`'s last record. (A `--selfcheck` difference is an
278    /// [`AnalyzeFailure`].)
279    TargetNotReproduced,
280    /// The baseline does not miss the target or does not finish: the passing
281    /// endpoint of a bisection (`hermit bisect`'s `--good` too), or the
282    /// baseline run the user chose.
283    BaselineNotReproduced,
284    /// No continuation lets a prefix search compare prefixes of the target
285    /// schedule (`--prefix-search`).
286    NoPrefixContinuation,
287    /// The search has not established an exact one-swap pair of schedules
288    /// with opposite outcomes between the endpoints ([`NoSingleSwap`]).
289    /// Analyze then attempts a prefix search within the target schedule;
290    /// `hermit bisect`, which has no target run to search, ends with this.
291    EndpointsNotSeparable,
292    /// `hermit bisect`'s --good and --bad schedules contain the same events.
293    EndpointsIdentical,
294}
295
296impl PreconditionClass {
297    /// The `class=` of the class line.
298    pub fn name(self) -> &'static str {
299        match self {
300            PreconditionClass::TargetNotReproduced => "target-not-reproduced",
301            PreconditionClass::BaselineNotReproduced => "baseline-not-reproduced",
302            PreconditionClass::NoPrefixContinuation => "no-prefix-continuation",
303            PreconditionClass::EndpointsNotSeparable => "endpoints-not-separable",
304            PreconditionClass::EndpointsIdentical => "endpoints-identical",
305        }
306    }
307
308    /// The class line `main.rs` prints first on stderr.
309    pub fn class_line(self) -> String {
310        format!("HERMIT_ANALYZE_PRECONDITION class={}", self.name())
311    }
312}

314/// The inputs do not give analyze what it needs, so it cannot look for a
315/// deciding event: it ends with exit [`hermit::HERMIT_ANALYZE_PRECONDITION_EXIT`]
316/// and [`PreconditionClass::class_line`], and the explanation says what to
317/// change.
318#[derive(Debug, Clone, PartialEq, Eq)]
319pub struct AnalyzePrecondition {
320    /// Which input.
321    pub class: PreconditionClass,
322    explanation: String,
323}
324
325impl AnalyzePrecondition {
326    /// [`PreconditionClass::TargetNotReproduced`].
327    pub fn target_not_reproduced(explanation: impl Into<String>) -> Self {
328        Self::new(PreconditionClass::TargetNotReproduced, explanation)
329    }
330
331    /// [`PreconditionClass::BaselineNotReproduced`].
332    pub fn baseline_not_reproduced(explanation: impl Into<String>) -> Self {
333        Self::new(PreconditionClass::BaselineNotReproduced, explanation)
334    }
335
336    /// [`PreconditionClass::NoPrefixContinuation`].
337    pub fn no_prefix_continuation(explanation: impl Into<String>) -> Self {
338        Self::new(PreconditionClass::NoPrefixContinuation, explanation)
339    }
340
341    /// [`PreconditionClass::EndpointsIdentical`].
342    pub fn endpoints_identical(explanation: impl Into<String>) -> Self {
343        Self::new(PreconditionClass::EndpointsIdentical, explanation)
344    }
345
346    /// The same conclusion, its explanation followed by `hint`.
347    pub fn with_hint(self, hint: &str) -> Self {
348        Self {
349            explanation: format!("{} {hint}", self.explanation),
350            ..self
351        }
352    }
353
354    fn new(class: PreconditionClass, explanation: impl Into<String>) -> Self {
355        Self {
356            class,
357            explanation: explanation.into(),
358        }
359    }
360}
361
362impl std::fmt::Display for AnalyzePrecondition {
363    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
364        f.write_str(&self.explanation)
365    }
366}

370/// The precondition `error` concludes, if any: an [`AnalyzePrecondition`], or
371/// the schedule search's typed end that it established no exact one-swap
372/// pair of schedules with opposite outcomes, which reaches `main.rs` only
373/// from `hermit bisect`.
374pub fn precondition_class(error: &anyhow::Error) -> Option<PreconditionClass> {
375    if let Some(precondition) = error.downcast_ref::<AnalyzePrecondition>() {
376        return Some(precondition.class);
377    }
378    if error.downcast_ref::<NoSingleSwap>().is_some() {
379        return Some(PreconditionClass::EndpointsNotSeparable);
380    }
381    None
382}

384/// Analyze could not conclude, for a reason in Hermit or the host, not in its
385/// inputs: a failure of Hermit (exit 125), with a class line that says which.
386#[derive(Debug, Clone, PartialEq, Eq)]
387pub enum AnalyzeFailure {
388    /// Replays of one run or schedule did not reproduce it: `--selfcheck`
389    /// found a difference, or a jittered control schedule never replayed the
390    /// same way twice. Determinism, Hermit's or the host's (an inexact branch
391    /// counter), broke.
392    ReplayNotReproducible(String),
393    /// The event-level search did not converge within its passes.
394    SearchDidNotConverge(String),
395}
396
397impl AnalyzeFailure {
398    /// The line `main.rs` prints first on stderr.
399    pub fn class_line(&self) -> String {
400        let class = match self {
401            AnalyzeFailure::ReplayNotReproducible(_) => "replay-not-reproducible",
402            AnalyzeFailure::SearchDidNotConverge(_) => "search-did-not-converge",
403        };
404        format!("HERMIT_INTERNAL_FAILURE class={class}")
405    }
406}
407
408impl std::fmt::Display for AnalyzeFailure {
409    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
410        match self {
411            AnalyzeFailure::ReplayNotReproducible(explanation)
412            | AnalyzeFailure::SearchDidNotConverge(explanation) => f.write_str(explanation),
413        }
414    }
415}

423    /// A verdict serializes as its report field does, and old reports without
424    /// one read as naming the critical events.
425    #[test]
426    fn a_verdict_is_written_in_snake_case() {
427        assert_eq!(
428            serde_json::to_string(&Verdict::ReproducedWithoutSchedule).unwrap(),
429            "\"reproduced_without_schedule\""
430        );
431        assert_eq!(
432            serde_json::to_string(&Verdict::CriticalEvents).unwrap(),
433            "\"critical_events\""
434        );
435        assert_eq!(Verdict::default(), Verdict::CriticalEvents);
436    }

438    /// A reproduced-without-schedule verdict reached under timing settings the
439    /// recording carries says what the trial under their alternatives
440    /// showed: "because of" only when it missed the target, "under" when it
441    /// met it or was not tried; without settings it stays the scheduler
442    /// alone. A cause from a later build reads as unknown.
443    #[test]
444    fn a_verdict_under_timing_settings_says_what_the_trial_showed() {
445        let alone = AnalyzeVerdict::reproduced_without_schedule("Reproduced without it.", None);
446        let latency = [TimingSetting {
447            option: "--pipe-wake-latency=50000".to_owned(),
448            alternative: "no --pipe-wake-latency".to_owned(),
449            drops: "--pipe-wake-latency",
450            adds: None,
451        }];
452        let untried = || AlternativeTiming::Untried {
453            why: "not asked".to_owned(),
454        };
455        assert_eq!(alone.clone().with_timing(&[], untried()), alone);
456        assert_eq!(
457            serde_json::to_value(alone.reason()).unwrap(),
458            serde_json::json!({"cause": "scheduler_alone"})
459        );
460        let trial = "prefix_alternative_timing.repro".to_owned();
461        let shown = alone.clone().with_timing(
462            &latency,
463            AlternativeTiming::Missed {
464                trial: trial.clone(),
465            },
466        );
467        assert_eq!(shown.verdict(), Verdict::ReproducedWithoutSchedule);
468        assert_eq!(
469            shown.to_string(),
470            "Reproduced without it. The scheduler alone meets the target because of \
471             --pipe-wake-latency=50000, the recording's timing semantics, which every trial \
472             keeps: the empty prefix with no --pipe-wake-latency, continued without PMU \
473             timeslices, misses it (prefix_alternative_timing.repro in the workspace). Under \
474             --pipe-wake-latency=50000 the empty prefix already meets the target, so the prefix \
475             search has no endpoint that misses it to start from and localizes no event; a \
476             two-seed search against a passing seed can end the same way, as it falls back to \
477             this search when it establishes no exact one-swap pair of schedules with opposite \
478             outcomes. To name a deciding event, analyze a new failing run with no \
479             --pipe-wake-latency in the run arguments, if one fails there."
480        );
481        // It states the timing experiment and the endpoint, never more.
482        for overclaim in ["no event decides", "needs no chaos decision"] {
483            assert!(
484                !shown.to_string().contains(overclaim),
485                "{overclaim:?}: {shown}"
486            );
487        }
488        assert_eq!(
489            serde_json::to_value(shown.reason()).unwrap(),
490            serde_json::json!({
491                "cause": "timing_semantics",
492                "options": ["--pipe-wake-latency=50000"],
493                "alternatives": ["no --pipe-wake-latency"],
494                "trial": "prefix_alternative_timing.repro",
495            })
496        );
497        let met = alone
498            .clone()
499            .with_timing(&latency, AlternativeTiming::Met { trial });
500        assert!(
501            met.to_string().ends_with(
502                "The trials keep --pipe-wake-latency=50000 from the recording, but the empty \
503                 prefix with no --pipe-wake-latency, continued without PMU timeslices, meets the \
504                 target too (prefix_alternative_timing.repro in the workspace), so the scheduler \
505                 alone meets it under either."
506            ),
507            "{met}"
508        );
509        assert_eq!(
510            serde_json::to_value(met.reason()).unwrap(),
511            serde_json::json!({
512                "cause": "scheduler_alone",
513                "timing_in_force": ["--pipe-wake-latency=50000"],
514            })
515        );
516        let not_tried = alone.with_timing(&latency, untried());
517        assert!(
518            not_tried.to_string().ends_with(
519                "The scheduler alone meets the target under --pipe-wake-latency=50000, in force \
520                 because every trial keeps them from the recording; whether it does with no \
521                 --pipe-wake-latency was not tried: not asked. A new failing run with no \
522                 --pipe-wake-latency in the run arguments can tell."
523            ),
524            "{not_tried}"
525        );
526        assert_eq!(not_tried.reason(), met.reason());
527        let later: VerdictReason =
528            serde_json::from_str(r#"{"cause": "something_new", "detail": 1}"#).unwrap();
529        assert_eq!(later, VerdictReason::Unknown);
530    }

532    /// The schedule search's typed end that it established no exact one-swap
533    /// pair, with or without a swap left, concludes `endpoints-not-separable`; a hint
534    /// keeps the class and adds to the explanation.
535    #[test]
536    fn the_search_ends_that_no_swap_separates_are_a_precondition() {
537        for swaps_left in [0, 1] {
538            let error = anyhow::Error::new(NoSingleSwap {
539                unshared_events: 2,

541                trimmed_failing_passed: false,

543            assert_eq!(
544                precondition_class(&error),
545                Some(PreconditionClass::EndpointsNotSeparable),
546                "{error:#}"
547            );
548        }
549        assert_eq!(precondition_class(&anyhow::anyhow!("other")), None);
550        let hinted = AnalyzePrecondition::target_not_reproduced("did not.").with_hint("Check.");
551        assert_eq!(hinted.class, PreconditionClass::TargetNotReproduced);
552        assert_eq!(hinted.to_string(), "did not. Check.");
553    }
