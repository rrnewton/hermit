9//! `--time-budget` and `--trial-budget`: how much wall-clock time and how
10//! many trials one `hermit analyze` invocation may spend before it reports
11//! what it has found so far.
12//!
13//! The budget decides only when analyze stops. It is checked before every
14//! trial, and a trial gets at most the time that is left as its `--timeout`.
15//! A trial that time cuts is thrown away rather than counted as an outcome
16//! (`BudgetExpired::cut`), so no search decision ever depends on the budget.
17

24use std::time::Instant;

29/// `--time-budget`: wall-clock time for the whole invocation.
30#[derive(Debug, Clone, Copy, PartialEq, Eq)]
31pub enum TimeBudget {
32    /// No limit.
33    Off,
34    /// At most this long.
35    Limit(Duration),
36}
37
38impl FromStr for TimeBudget {
39    type Err = String;
40
41    /// `off`, or a positive whole number of seconds, minutes or hours: `1800`,
42    /// `1800s`, `30m`, `2h`.
43    fn from_str(s: &str) -> Result<Self, Self::Err> {
44        if s == "off" {
45            return Ok(TimeBudget::Off);
46        }
47        let (digits, unit) = match s.char_indices().find(|(_, c)| !c.is_ascii_digit()) {
48            Some((at, _)) => s.split_at(at),
49            None => (s, "s"),
50        };
51        let seconds_per_unit = match unit {
52            "s" => 1,
53            "m" => 60,
54            "h" => 3600,
55            _ => 0,
56        };
57        match digits.parse::<u64>() {
58            Ok(count) if count > 0 && seconds_per_unit > 0 => count
59                .checked_mul(seconds_per_unit)
60                .map(|seconds| TimeBudget::Limit(Duration::from_secs(seconds)))
61                .ok_or_else(|| format!("time budget {s:?} is too large")),
62            _ => Err(format!(
63                "expected `off` or a positive whole number of seconds, minutes or hours (1800, \
64                 1800s, 30m, 2h), got {s:?}"
65            )),
66        }
67    }
68}

70/// `--trial-budget`: how many trials the invocation may run.
71#[derive(Debug, Clone, Copy, PartialEq, Eq)]
72pub enum TrialBudget {
73    /// No limit.
74    Off,
75    /// At most this many.
76    Limit(NonZeroU64),
77}
78
79impl FromStr for TrialBudget {
80    type Err = String;
81
82    fn from_str(s: &str) -> Result<Self, Self::Err> {
83        if s == "off" {
84            return Ok(TrialBudget::Off);
85        }
86        s.parse()
87            .map(TrialBudget::Limit)
88            .map_err(|_| format!("expected `off` or a positive number of trials, got {s:?}"))
89    }
90}

92/// Which budget ran out.
93#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize)]
94#[serde(rename_all = "kebab-case")]
95pub enum BudgetKind {
96    /// `--time-budget`.
97    TimeBudget,
98    /// `--trial-budget`.
99    TrialBudget,
100}
101
102impl BudgetKind {
103    /// The option, as the user spells it.
104    pub fn option(self) -> &'static str {
105        match self {
106            BudgetKind::TimeBudget => "--time-budget",
107            BudgetKind::TrialBudget => "--trial-budget",
108        }
109    }
110
111    /// The `class=` value of the stderr class line.
112    pub fn class(self) -> &'static str {
113        match self {
114            BudgetKind::TimeBudget => "time-budget",
115            BudgetKind::TrialBudget => "trial-budget",
116        }
117    }
118}

120/// The budget ran out: analyze starts no further trial and reports what it
121/// has found so far.
122#[derive(Debug, Clone, PartialEq, Eq)]
123pub struct BudgetExpired {
124    /// Which budget.
125    pub kind: BudgetKind,
126    /// The trial the time budget cut short, if one was running: its outcome
127    /// is not read, and a resumed analysis runs it again.
128    pub cut: Option<String>,
129}
130
131impl fmt::Display for BudgetExpired {
132    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
133        write!(f, "hermit analyze reached its {}", self.kind.option())?;
134        if let Some(run) = &self.cut {
135            write!(f, " during {run}, whose outcome is not counted")?;
136        }
137        Ok(())
138    }
139}
140
141impl std::error::Error for BudgetExpired {}

143/// Where elapsed time comes from: the host's monotonic clock, or a test's.
144type Clock = Arc<dyn Fn() -> Duration + Send + Sync>;
145
146/// The shared state of a budget: every copy of the analyze options (each
147/// run holds one) sees the same.
148struct State {
149    time: Option<Duration>,
150    trials: Option<u64>,
151    /// Trials started so far; a trial answered from a resumed journal does
152    /// not count.
153    started: u64,
154    /// Which budget ran out, once one has.
155    expired: Option<BudgetKind>,
156    /// After expiry: how long the partial report's own replays may still run
157    /// (elapsed time), if any may.
158    grace_until: Option<Duration>,
159}
160
161/// One invocation's budget. Cloning shares it.
162#[derive(Clone)]
163pub struct Budget {
164    state: Arc<Mutex<State>>,
165    clock: Clock,
166}
167
168impl fmt::Debug for Budget {
169    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
170        let state = self.state.lock().unwrap();
171        f.debug_struct("Budget")
172            .field("time", &state.time)
173            .field("trials", &state.trials)
174            .field("started", &state.started)
175            .field("expired", &state.expired)
176            .finish()
177    }
178}
179
180impl Default for Budget {
181    /// No limit.
182    fn default() -> Self {
183        Self::new(TimeBudget::Off, TrialBudget::Off)
184    }
185}

187/// The longest grace period the partial report's stack-trace replays get
188/// after the budget ran out.
189pub const STACK_GRACE_CAP: Duration = Duration::from_secs(120);
190/// The share of `--time-budget` the grace period may take, in percent.
191pub const STACK_GRACE_PERCENT: u32 = 5;
192
193impl Budget {
194    /// A budget counted from now on the host's monotonic clock.
195    pub fn new(time: TimeBudget, trials: TrialBudget) -> Self {
196        let start = Instant::now();
197        Self::with_clock(time, trials, Arc::new(move || start.elapsed()))
198    }
199
200    /// A budget that reads elapsed time from `clock`.
201    pub fn with_clock(time: TimeBudget, trials: TrialBudget, clock: Clock) -> Self {
202        Self {
203            state: Arc::new(Mutex::new(State {
204                time: match time {
205                    TimeBudget::Off => None,
206                    TimeBudget::Limit(limit) => Some(limit),
207                },
208                trials: match trials {
209                    TrialBudget::Off => None,
210                    TrialBudget::Limit(limit) => Some(limit.get()),
211                },
212                started: 0,
213                expired: None,
214                grace_until: None,
215            })),
216            clock,
217        }
218    }
219
220    /// Elapsed time since the budget started.
221    pub fn elapsed(&self) -> Duration {
222        (self.clock)()
223    }
224
225    /// The time limit, if any.
226    pub fn time_limit(&self) -> Option<Duration> {
227        self.state.lock().unwrap().time
228    }
229
230    /// The trial limit, if any.
231    pub fn trial_limit(&self) -> Option<u64> {
232        self.state.lock().unwrap().trials
233    }
234
235    /// How many trials have started.
236    pub fn trials_started(&self) -> u64 {
237        self.state.lock().unwrap().started
238    }
239
240    /// Which budget ran out, if one has.
241    #[cfg(test)]
242    pub fn expired(&self) -> Option<BudgetKind> {
243        self.state.lock().unwrap().expired
244    }
245
246    /// Called before a trial starts: `Err` once a budget has run out, which
247    /// it then stays. Otherwise counts the trial and returns the whole
248    /// seconds of time left to bound it with, if time is limited (at least
249    /// one: a trial bound is a whole positive number of seconds).
250    pub fn start_trial(&self) -> Result<Option<NonZeroU64>, BudgetExpired> {
251        let now = self.elapsed();
252        let mut state = self.state.lock().unwrap();
253        if let Some(kind) = state.expired {
254            return match state.grace_until {
255                Some(until) if now < until => Ok(Some(whole_seconds(until - now))),
256                _ => Err(BudgetExpired { kind, cut: None }),
257            };
258        }
259        if state.trials.is_some_and(|limit| state.started >= limit) {
260            state.expired = Some(BudgetKind::TrialBudget);
261            return Err(BudgetExpired {
262                kind: BudgetKind::TrialBudget,
263                cut: None,
264            });
265        }
266        let left = match state.time {
267            Some(limit) if now >= limit => {
268                state.expired = Some(BudgetKind::TimeBudget);
269                return Err(BudgetExpired {
270                    kind: BudgetKind::TimeBudget,
271                    cut: None,
272                });
273            }
274            Some(limit) => Some(whole_seconds(limit - now)),
275            None => None,
276        };
277        state.started += 1;
278        Ok(left)
279    }
280
281    /// The time budget cut the trial `run` short: it ended by a `--timeout`
282    /// that the time left had shortened. Its outcome came from the budget,
283    /// not from the guest, so it is thrown away.
284    pub fn cut(&self, run: &str) -> BudgetExpired {
285        let mut state = self.state.lock().unwrap();
286        let kind = *state.expired.get_or_insert(BudgetKind::TimeBudget);
287        BudgetExpired {
288            kind,
289            cut: Some(run.to_owned()),
290        }
291    }
292
293    /// The grace period the partial report's stack-trace replays get after
294    /// the budget ran out: min([`STACK_GRACE_CAP`], [`STACK_GRACE_PERCENT`]
295    /// of `--time-budget`), or the cap when only trials were limited.
296    pub fn stack_grace(&self) -> Duration {
297        let state = self.state.lock().unwrap();
298        match state.time {
299            Some(limit) => STACK_GRACE_CAP.min(limit * STACK_GRACE_PERCENT / 100),
300            None => STACK_GRACE_CAP,
301        }
302    }
303
304    /// Open the grace period: until it ends, [`Budget::start_trial`] lets
305    /// trials start again, each bounded by what is left of it.
306    pub fn open_grace(&self, grace: Duration) {
307        let now = self.elapsed();
308        self.state.lock().unwrap().grace_until = Some(now + grace);
309    }
310
311    /// Close the grace period.
312    pub fn close_grace(&self) {
313        self.state.lock().unwrap().grace_until = None;
314    }
315}

317/// `duration` rounded up to whole seconds, at least one.
318fn whole_seconds(duration: Duration) -> NonZeroU64 {
319    let seconds = duration.as_secs() + u64::from(duration.subsec_nanos() > 0);
320    NonZeroU64::new(seconds.max(1)).unwrap()
321}
322
323/// `duration` as the user would write it: `1800 s`, or `30 min` when whole.
324pub fn describe(duration: Duration) -> String {
325    let seconds = duration.as_secs();
326    if seconds >= 60 && seconds.is_multiple_of(60) {
327        format!("{} min", seconds / 60)
328    } else {
329        format!("{seconds} s")
330    }
331}

340    /// A clock a test moves by hand, in whole seconds.
341    fn manual_clock() -> (Arc<AtomicU64>, Clock) {
342        let seconds = Arc::new(AtomicU64::new(0));
343        let read = Arc::clone(&seconds);
344        (
345            seconds,
346            Arc::new(move || Duration::from_secs(read.load(Ordering::SeqCst))),
347        )
348    }

350    #[test]
351    fn a_time_budget_takes_seconds_minutes_or_hours() {
352        for (text, seconds) in [("1800", 1800), ("1800s", 1800), ("30m", 1800), ("2h", 7200)] {
353            assert_eq!(
354                text.parse::<TimeBudget>(),
355                Ok(TimeBudget::Limit(Duration::from_secs(seconds))),
356                "{text}"
357            );
358        }
359        assert_eq!("off".parse::<TimeBudget>(), Ok(TimeBudget::Off));
360        for bad in ["0", "", "30x", "m", "-5", "1.5h", "30 m"] {
361            assert!(bad.parse::<TimeBudget>().is_err(), "{bad:?}");
362        }
363        assert_eq!(
364            "3".parse::<TrialBudget>(),
365            Ok(TrialBudget::Limit(NonZeroU64::new(3).unwrap()))
366        );
367        assert_eq!("off".parse::<TrialBudget>(), Ok(TrialBudget::Off));
368        assert!("0".parse::<TrialBudget>().is_err());
369    }
370
371    /// Each trial gets the time left, rounded up, and none starts once it is
372    /// gone; the budget then stays spent.
373    #[test]
374    fn trials_get_the_time_left_and_none_starts_after_it() {
375        let (now, clock) = manual_clock();
376        let budget = Budget::with_clock(
377            TimeBudget::Limit(Duration::from_secs(100)),
378            TrialBudget::Off,
379            clock,
380        );
381        assert_eq!(budget.start_trial(), Ok(NonZeroU64::new(100)));
382        now.store(99, Ordering::SeqCst);
383        assert_eq!(budget.start_trial(), Ok(NonZeroU64::new(1)));
384        now.store(100, Ordering::SeqCst);
385        let expired = budget.start_trial().unwrap_err();
386        assert_eq!(expired.kind, BudgetKind::TimeBudget);
387        assert_eq!(expired.cut, None);
388        assert_eq!(budget.expired(), Some(BudgetKind::TimeBudget));
389        assert_eq!(budget.trials_started(), 2);
390        // A grace period lets the report's replays run, bounded by it.
391        budget.open_grace(Duration::from_secs(5));
392        assert_eq!(budget.start_trial(), Ok(NonZeroU64::new(5)));
393        now.store(105, Ordering::SeqCst);
394        assert!(budget.start_trial().is_err());
395    }
