9//! Stopping `hermit analyze` on SIGTERM, SIGINT or SIGHUP.
10//!
11//! At their default action these signals kill analyze where it stands. The
12//! trial running dies with it (its container init has a parent-death
13//! signal), but analyze's workspace and the trial's own temporary
14//! directories stay behind. While [`StopSignals`] is held, a stop signal
15//! instead ends the trial running ([`signal_children`]), a trial whose
16//! container init is forked after it exits at once
17//! (`container::exit_if_stop_requested`), and every later trial refuses to
18//! start ([`check`]), so analyze unwinds, removes what it created, and then
19//! dies of the same signal (`AnalyzeOpts::main`).
20//!
21//! The kernel runs the handler on any thread of analyze that does not block
22//! the signal (`complete_signal` and `wants_signal`,
23//! <https://git.kernel.org/pub/scm/linux/kernel/git/stable/linux.git/tree/kernel/signal.c?h=v6.9#n978>),
24//! and a child can be reaped on another thread while it runs. So the handler
25//! never signals a child by its number, which the kernel can give to another
26//! process once the child is reaped: it signals through a pidfd it checks
27//! ([`signal_children`]), which reaches that child or no one.
28//!
29//! The threads that leave the stop signals unblocked are the one that drives
30//! the trials and every thread a trial starts in analyze's process while it
31//! runs, which takes its creator's unblocked mask (`hermit run`'s group-kill
32//! policing thread, `container::police_group_kills`, or a runtime's worker
33//! or blocking-pool thread); the handler is safe on each. Only the memory watchdog's
34//! thread, which analyze starts itself, blocks the stop signals from its
35//! first instruction ([`spawn_blocking_stop_signals`]). Beyond the pidfd, the
36//! handler stores the signal in an atomic ([`check`]) and reads `/proc`,
37//! which reads the same from every thread.
38//!
39//! A signal the handler cannot pass on to a child it found (no descriptor
40//! left for the scan or the pidfd, a pidfd call that fails for any reason
41//! but the child having gone) is never swallowed: analyze says so on stderr
42//! and dies of the signal by its default action, as it would without the
43//! handler, and the trial's container init dies with it by its parent-death
44//! signal. Such a death unwinds nothing, so the workspace stays behind, and so
45//! does the private temporary directory of the trial that was running
46//! ([`signal_children`], [`die_by_default_action`]).

57/// The signals a supervisor, a shell or a terminal sends to stop a process.
58const STOP_SIGNALS: [libc::c_int; 3] = [libc::SIGTERM, libc::SIGINT, libc::SIGHUP];

60/// The process that installed [`on_stop_signal`]. A process forked from it
61/// inherits the handler until it installs its own.
62static HOLDER: AtomicI32 = AtomicI32::new(0);
63
64/// How many children analyze may already have when it installs the handler
65/// and still pass signals on.
66const INHERITED_CAPACITY: usize = 64;
67
68/// The children analyze already had when it installed the handler, 0 in the
69/// unused slots. A process keeps its children across `exec`, so these are
70/// processes whatever ran analyze started (`job & exec hermit analyze ...`,
71/// or the `tee` behind `exec > >(tee log)`), not analyze's to stop. Nothing
72/// reaps them, so their PIDs cannot be reused while analyze runs.
73static INHERITED: [AtomicI32; INHERITED_CAPACITY] =
74    [const { AtomicI32::new(0) }; INHERITED_CAPACITY];

76/// The `/proc` directory the handler scans, opened when it is installed and
77/// kept for the life of the process, deliberately: a scan then needs no
78/// descriptor for it, and no handler still scanning on another thread can
79/// find it closed; -1 until then.
80static PROC_DIR: AtomicI32 = AtomicI32::new(-1);
81
82/// Set while a handler scans `/proc` through [`PROC_DIR`], whose offset
83/// every scan moves.

104fn stopped(recorded: Option<libc::c_int>) -> Result<(), Stopped> {
105    match recorded {
106        None => Ok(()),
107        // Only the three stop signals are recorded.
108        Some(raw) => Err(Stopped(
109            reverie::Signal::try_from(raw).unwrap_or(reverie::Signal::SIGTERM),
110        )),
111    }

114/// Analyze's handler for every stop signal.
115///
116/// In analyze, it records the first signal (`container::request_stop`) and
117/// passes every one on to the children analyze started: the container init
118/// of the trial running, whose own handler exits 128 + signo and so takes the
119/// trial's namespace down. A later signal reaches a trial that started after
120/// an earlier one.
121///
122/// In a process analyze forked that has not installed its own handler yet,
123/// which is a trial's container init during Reverie's container setup, it
124/// exits 128 + signo, as that init's handler will
125/// (`ContainerInitStopSignalBridge` does the same while analyze holds no
126/// handler).
127extern "C" fn on_stop_signal(signal: libc::c_int) {
128    // SAFETY: getpid(2) and _exit(2) are async-signal-safe.
129    let pid = unsafe { libc::getpid() };
130    if pid != HOLDER.load(Ordering::Relaxed) {
131        unsafe { libc::_exit(128 + signal) }
132    }
133    // SAFETY: errno is this thread's; restoring it keeps the code this
134    // handler interrupted from reading a value the handler left.
135    let errno = unsafe { *libc::__errno_location() };
136    container::request_stop(signal);
137    if let Err(undelivered) = signal_children(pid, signal, inherited) {
138        die_by_default_action(signal, undelivered);
139    }
140    unsafe { *libc::__errno_location() = errno };
141}

143/// How long the diagnostic of [`die_by_default_action`] may take before the
144/// signal ends analyze anyway.
145const DIAGNOSTIC_BOUND: libc::timespec = libc::timespec {
146    tv_sec: 1,
147    tv_nsec: 0,
148};
149
150/// End analyze by `signal`'s default action, after saying on stderr that
151/// its handler could not pass the signal on (`undelivered`): the stop is
152/// then what it would be without the handler. The trial's container init
153/// dies with analyze by its parent-death signal. Nothing unwinds, so the
154/// workspace and the running trial's private temporary directory stay.
155/// Async-signal-safe.
156///
157/// Death never waits on the diagnostic. The default action is armed first,
158/// and a timer sends the same signal to the process after
159/// [`DIAGNOSTIC_BOUND`] (`SIGEV_SIGNAL` without `SIGEV_THREAD_ID` is
160/// process-directed, `posix_timer_event`,
161/// <https://git.kernel.org/pub/scm/linux/kernel/git/stable/linux.git/tree/kernel/time/posix-timers.c?h=v6.9#n280>),
162/// before the signal is unblocked on this thread and the diagnostic written:
163/// a write that blocks, on a full pipe whose reader is alive, ends when the
164/// timer's signal kills the process. Without a timer the diagnostic is left
165/// out. Every system call is made directly, with no descriptor opened.
166fn die_by_default_action(signal: libc::c_int, undelivered: Undelivered) -> ! {
167    let mut name = [0u8; 10];
168    let mut errno = [0u8; 10];
169    // SAFETY: sigaction(2), sigemptyset(3), sigaddset(3), pthread_sigmask(3),
170    // timer_create(2), timer_settime(2), write(2), raise(3) and _exit(2) are
171    // async-signal-safe, and every pointer refers to a local that outlives
172    // the call. An all-zero sigaction is SIG_DFL; the kernel writes its
173    // timer id, an int, into `timer`.
174    unsafe {
175        let default: libc::sigaction = std::mem::zeroed();
176        libc::sigaction(signal, &default, std::ptr::null_mut());
177        let mut event: libc::sigevent = std::mem::zeroed();
178        event.sigev_notify = libc::SIGEV_SIGNAL;
179        event.sigev_signo = signal;
180        let mut timer: libc::c_int = 0;
181        let deadline = libc::itimerspec {
182            it_interval: libc::timespec {
183                tv_sec: 0,
184                tv_nsec: 0,
185            },
186            it_value: DIAGNOSTIC_BOUND,
187        };
188        let bounded = libc::syscall(
189            libc::SYS_timer_create,
190            libc::CLOCK_MONOTONIC,
191            &mut event as *mut libc::sigevent,
192            &mut timer as *mut libc::c_int,
193        ) == 0
194            && libc::syscall(
195                libc::SYS_timer_settime,
196                timer,
197                0,
198                &deadline as *const libc::itimerspec,
199                std::ptr::null_mut::<libc::itimerspec>(),
200            ) == 0;
201        let mut unblock: libc::sigset_t = std::mem::zeroed();
202        libc::sigemptyset(&mut unblock);
203        libc::sigaddset(&mut unblock, signal);
204        libc::pthread_sigmask(libc::SIG_UNBLOCK, &unblock, std::ptr::null_mut());
205        if bounded {
206            for part in [
207                &b"hermit analyze: could not pass signal "[..],
208                pid_digits(signal, &mut name).unwrap_or(b"?"),
209                b" on to the run it started (",
210                undelivered.call.as_bytes(),
211                b" failed with errno ",
212                pid_digits(undelivered.errno, &mut errno).unwrap_or(b"?"),
213                b"), so it dies of the signal as it would without its handler, and leaves its \
214                  workspace and the trial's private temporary directory behind\n",
215            ] {
216                libc::write(2, part.as_ptr().cast(), part.len());
217            }
218        }
219        libc::raise(signal);
220        libc::_exit(128 + signal)
221    }
222}

231/// Holds [`on_stop_signal`] installed; dropping it restores the default
232/// action.
233#[must_use = "the stop handlers are removed when this guard is dropped"]
234pub struct StopSignals {
235    /// Which entries of [`STOP_SIGNALS`] this guard installed.
236    installed: [bool; STOP_SIGNALS.len()],
237    /// The exact handler value installed, compared on drop.
238    handler: libc::sighandler_t,
239}
240
241impl StopSignals {
242    /// Install the handler for every stop signal still at its default
243    /// action. One that is ignored (SIGINT in a shell's background job,
244    /// SIGHUP under `nohup`) or handled stays as it is, and so does every
245    /// signal of a process that is PID 1: the container init of a trial sees
246    /// itself as PID 1 too, so the handler could not tell the two apart.
247    /// Nothing is installed either when the handler could not find the run's
248    /// container init ([`record_inherited_children`]), or could not signal it
249    /// safely: where a pidfd cannot be opened, signalled through, or waited on
250    /// ([`usable_pidfds`]). Such an analyze dies of the signal, or ignores
251    /// it, as before.
252    ///
253    /// Call it while no other thread of analyze can take a stop signal: the
254    /// query and the install are two steps. The memory watchdog may run
255    /// already, as it does when `main` installs this, because it blocks the
256    /// stop signals before it starts.
257    pub fn install() -> anyhow::Result<Self> {
258        let handler: extern "C" fn(libc::c_int) = on_stop_signal;
259        let mut guard = Self {
260            installed: [false; STOP_SIGNALS.len()],
261            handler: handler as libc::sighandler_t,
262        };
263        // SAFETY: getpid(2) has no preconditions.
264        let pid = unsafe { libc::getpid() };
265        if pid == 1 {
266            return Ok(guard);
267        }
268        // Before any disposition changes: a handler that recorded a stop but
269        // could not end the run in progress would leave that run going.
270        if !usable_pidfds(pid) || !record_inherited_children(pid) {
271            return Ok(guard);
272        }
273        HOLDER.store(pid, Ordering::Relaxed);
274        // SAFETY: an all-zero sigaction is a valid value, and sigemptyset(3)
275        // and sigaddset(3) only write the mask of this local.
276        let mut action: libc::sigaction = unsafe { std::mem::zeroed() };
277        action.sa_sigaction = guard.handler;
278        // Interrupted system calls resume: the trial's wait ends when its
279        // init exits, not with EINTR.
280        action.sa_flags = libc::SA_RESTART;
281        unsafe {
282            libc::sigemptyset(&mut action.sa_mask);
283            for signal in STOP_SIGNALS {
284                libc::sigaddset(&mut action.sa_mask, signal);
285            }
286        }
287        for (index, signal) in STOP_SIGNALS.into_iter().enumerate() {
288            if current_action(signal)?.sa_sigaction != libc::SIG_DFL {
289                continue;
290            }
291            // SAFETY: the handler performs only async-signal-safe calls, and
292            // `action` outlives the call.
293            if unsafe { libc::sigaction(signal, &action, std::ptr::null_mut()) } == -1 {
294                return Err(std::io::Error::last_os_error())
295                    .with_context(|| format!("cannot install analyze's signal {signal} handler"));
296            }
297            guard.installed[index] = true;
298        }
299        Ok(guard)
300    }
301}

303impl Drop for StopSignals {
304    fn drop(&mut self) {
305        for (index, signal) in STOP_SIGNALS.into_iter().enumerate() {
306            // Undo only this guard's own handler: anything else was installed
307            // after it on purpose.
308            if !self.installed[index]
309                || !current_action(signal).is_ok_and(|current| current.sa_sigaction == self.handler)
310            {
311                continue;
312            }
313            // SAFETY: an all-zero sigaction is SIG_DFL with no flags.
314            let default: libc::sigaction = unsafe { std::mem::zeroed() };
315            unsafe { libc::sigaction(signal, &default, std::ptr::null_mut()) };
316        }
317    }
318}

320/// Record analyze's present children in [`INHERITED`], and say whether the
321/// handler can find the children analyze starts and pass signals on to
322/// them: not when `/proc` belongs to another PID namespace (the
323/// `/proc/self` link names the reader's PID as that `/proc` numbers it,
324/// and `kill` takes PIDs of analyze's own), nor when there are more
325/// children than [`INHERITED`] holds.
326fn record_inherited_children(pid: libc::pid_t) -> bool {
327    let own_proc = std::fs::read_link("/proc/self")
328        .ok()
329        .and_then(|link| link.to_str()?.parse::<libc::pid_t>().ok())
330        == Some(pid);
331    if !own_proc || proc_dir().is_none() {
332        return false;
333    }
334    for slot in &INHERITED {
335        slot.store(0, Ordering::Relaxed);
336    }
337    let mut recorded = 0;
338    let mut fits = true;
339    let scanned = for_each_child(pid, |child| match INHERITED.get(recorded) {
340        Some(slot) => {
341            slot.store(child, Ordering::Relaxed);
342            recorded += 1;
343        }
344        None => fits = false,
345    });
346    scanned.is_ok() && fits
347}

349/// The `/proc` directory [`for_each_child`] scans ([`PROC_DIR`]), opened on
350/// first use; `None` if it cannot be opened.
351fn proc_dir() -> Option<libc::c_int> {
352    let open = PROC_DIR.load(Ordering::Acquire);
353    if open >= 0 {
354        return Some(open);
355    }
356    // SAFETY: open(2) on a NUL-terminated literal.
357    let opened = unsafe {
358        libc::open(
359            c"/proc".as_ptr(),
360            libc::O_RDONLY | libc::O_DIRECTORY | libc::O_CLOEXEC,
361        )
362    };
363    if opened < 0 {
364        return None;
365    }
366    match PROC_DIR.compare_exchange(-1, opened, Ordering::AcqRel, Ordering::Acquire) {
367        Ok(_) => Some(opened),
368        Err(other) => {
369            // SAFETY: `opened` was opened above and is not shared.
370            unsafe { libc::close(opened) };
371            Some(other)
372        }
373    }
374}

376/// The action installed for `signal`, read without changing it.
377fn current_action(signal: libc::c_int) -> anyhow::Result<libc::sigaction> {
378    // SAFETY: a null new action makes sigaction(2) a query, and `current` is
379    // read only on success.
380    let mut current: libc::sigaction = unsafe { std::mem::zeroed() };
381    if unsafe { libc::sigaction(signal, std::ptr::null(), &mut current) } == -1 {
382        return Err(std::io::Error::last_os_error())
383            .with_context(|| format!("cannot read the signal {signal} disposition"));
384    }
385    Ok(current)
386}

388/// A system call that failed while the handler passed a signal on, and its

391pub(super) struct Undelivered {
392    call: &'static str,
393    errno: libc::c_int,

398    fn now(call: &'static str) -> Self {

400        let errno = unsafe { *libc::__errno_location() };

402    }

405/// Send `signal` to every child of `parent` that `skip` does not name;
406/// returns how many it reached, or the call that failed when it may not have
407/// reached one. Async-signal-safe, as [`for_each_child`].
408///
409/// The handler can run on any thread that does not block the signal, and a
410/// child found here can be reaped meanwhile on another thread, and its PID
411/// given to a process analyze did not start. So each child is signalled
412/// through a pidfd ([`signal_child`]), never by its number. A child that is
413/// gone, or a PID that names another process by the time its pidfd is open,
414/// is not an error; a scan that could not read `/proc`, or a pidfd call that
415/// failed otherwise, is, and the handler then ends analyze by the signal's
416/// default action ([`die_by_default_action`]) rather than leave the run
417/// going.
418///
419/// One scan runs at a time: [`PROC_DIR`]'s offset is shared. A handler that
420/// finds another scanning, on another thread, leaves the children to it; a
421/// container init exits on the first stop signal it gets.
422pub(super) fn signal_children(
423    parent: libc::pid_t,
424    signal: libc::c_int,
425    skip: impl Fn(libc::pid_t) -> bool,
426) -> Result<usize, Undelivered> {
427    if SCANNING
428        .compare_exchange(false, true, Ordering::AcqRel, Ordering::Acquire)
429        .is_err()
430    {
431        return Ok(0);
432    }
433    let mut reached = 0;
434    let mut failed = None;
435    let scanned = for_each_child(parent, |child| {
436        if skip(child) || failed.is_some() {
437            return;
438        }
439        #[cfg(test)]
440        tests::in_the_window(child);
441        match signal_child(parent, child, signal) {
442            Ok(true) => reached += 1,
443            Ok(false) => {}
444            Err(undelivered) => failed = Some(undelivered),
445        }
446    });
447    SCANNING.store(false, Ordering::Release);
448    scanned?;
449    match failed {
450        Some(undelivered) => Err(undelivered),
451        None => Ok(reached),
452    }
453}

455/// Send `signal` to `pid` through a pidfd if `pid` is a child of `parent`:
456/// `Ok(true)` if a child received it, `Ok(false)` if `pid` names no child of
457/// `parent` any more, and the failed call otherwise. Async-signal-safe:
458/// system calls on stack buffers.
459///
460/// A pidfd names the process that held `pid` when it was opened, and a signal
461/// sent through it reaches that process or, once it is reaped, none: the
462/// kernel looks the process up by the pidfd's `struct pid`, never by number
463/// (`pidfd_open`,
464/// <https://git.kernel.org/pub/scm/linux/kernel/git/stable/linux.git/tree/kernel/pid.c?h=v6.9#n629>;
465/// `pidfd_send_signal` and `kill_pid_info_type`,
466/// <https://git.kernel.org/pub/scm/linux/kernel/git/stable/linux.git/tree/kernel/signal.c?h=v6.9#n1483>).
467/// Which process that is, is checked through the pidfd too, after it is
468/// open: `waitid(P_PIDFD, ..., WNOHANG | WNOWAIT)` succeeds only for a child
469/// of the caller's thread group, running or exited and not yet reaped, and
470/// reaps nothing; for any other process it fails with ECHILD
471/// (`kernel_waitid_prepare` and `eligible_child`,
472/// <https://git.kernel.org/pub/scm/linux/kernel/git/stable/linux.git/tree/kernel/exit.c?h=v6.9#n1679>).
473/// The check needs no descriptor beyond the pidfd. `parent` other than the
474/// calling process's own PID is checked through `/proc` instead, for tests.
475fn signal_child(
476    parent: libc::pid_t,
477    pid: libc::pid_t,
478    signal: libc::c_int,
479) -> Result<bool, Undelivered> {
480    // SAFETY: pidfd_open(2) has no memory arguments.
481    let opened = unsafe { libc::syscall(libc::SYS_pidfd_open, pid, 0) };
482    let Some(pidfd) = libc::c_int::try_from(opened).ok().filter(|&fd| fd >= 0) else {
483        let undelivered = Undelivered::now("pidfd_open");
484        return match undelivered.errno {
485            libc::ESRCH => Ok(false),
486            _ => Err(undelivered),
487        };
488    };
489    let delivered = is_child(parent, pid, pidfd).and_then(|child| {
490        if !child {
491            return Ok(false);
492        }
493        // SAFETY: pidfd_send_signal(2) with no siginfo; `pidfd` is open.
494        let sent = unsafe {
495            libc::syscall(
496                libc::SYS_pidfd_send_signal,
497                pidfd,
498                signal,
499                std::ptr::null::<libc::siginfo_t>(),
500                0,
501            )
502        };
503        if sent == 0 {
504            return Ok(true);
505        }
506        let undelivered = Undelivered::now("pidfd_send_signal");
507        match undelivered.errno {
508            // It exited after the check.
509            libc::ESRCH => Ok(false),
510            _ => Err(undelivered),
511        }
512    });
513    // SAFETY: `pidfd` is the descriptor opened above.
514    unsafe { libc::close(pidfd) };
515    delivered
516}

518/// Whether `pidfd`, opened for `pid`, names a child of `parent`: through the
519/// pidfd itself for analyze's own children ([`signal_child`]), else (in
520/// tests) by `pid`'s parent in `/proc`.
521fn is_child(
522    parent: libc::pid_t,
523    pid: libc::pid_t,
524    pidfd: libc::c_int,
525) -> Result<bool, Undelivered> {
526    // SAFETY: getpid(2) has no preconditions.
527    if parent != unsafe { libc::getpid() } {
528        let mut digits = [0u8; 10];
529        let Some(name) = pid_digits(pid, &mut digits) else {
530            return Ok(false);
531        };
532        let proc_dir = proc_dir().ok_or_else(|| Undelivered::now("open /proc"))?;
533        return Ok(parent_of(proc_dir, name)? == Some(parent));
534    }
535    if waitid_pidfd(pidfd) == 0 {
536        return Ok(true);
537    }
538    let undelivered = Undelivered::now("waitid");
539    match undelivered.errno {
540        libc::ECHILD => Ok(false),
541        _ => Err(undelivered),
542    }
543}

545/// `waitid(P_PIDFD, pidfd, WEXITED | WNOHANG | WNOWAIT | __WALL)`: 0 if
546/// `pidfd` names a child of the calling process, running or exited and not
547/// yet reaped (reaping nothing), else -1 with errno set (ECHILD for any
548/// other process). Made as the raw system call rather than through the C
549/// library's wrapper, which POSIX does not list as async-signal-safe.
550fn waitid_pidfd(pidfd: libc::c_int) -> libc::c_long {
551    // SAFETY: an all-zero siginfo_t is a valid out-parameter; waitid(2)
552    // writes only `info`, and a null rusage asks for none.
553    let mut info: libc::siginfo_t = unsafe { std::mem::zeroed() };
554    unsafe {
555        libc::syscall(
556            libc::SYS_waitid,
557            libc::P_PIDFD,
558            pidfd,
559            &mut info as *mut libc::siginfo_t,

581/// Whether [`signal_child`] can work here, tried on this process (`pid`): a
582/// pidfd opens (Linux 5.3), a null signal goes through it (5.1), and
583/// `waitid` takes it (`P_PIDFD`, Linux 5.4), answering ECHILD since a process
584/// is not its own child. A seccomp filter can refuse any of the three.
585fn usable_pidfds(pid: libc::pid_t) -> bool {
586    // SAFETY: pidfd_open(2) has no memory arguments.
587    let opened = unsafe { libc::syscall(libc::SYS_pidfd_open, pid, 0) };
588    let Some(pidfd) = libc::c_int::try_from(opened).ok().filter(|&fd| fd >= 0) else {
589        return false;
590    };
591    // SAFETY: pidfd_send_signal(2) with signal 0 and no siginfo sends nothing;
592    // waitid(2) writes only `info`; `pidfd` is open until the close.
593    unsafe {
594        let sends = libc::syscall(
595            libc::SYS_pidfd_send_signal,
596            pidfd,
597            0,
598            std::ptr::null::<libc::siginfo_t>(),
599            0,
600        ) == 0;
601        let waits = waitid_pidfd(pidfd) == -1 && *libc::__errno_location() == libc::ECHILD;
602        libc::close(pidfd);
603        sends && waits
604    }
605}

607/// Spawn `builder`'s thread with SIGTERM, SIGINT and SIGHUP blocked from its
608/// first instruction, so that analyze's stop handler never runs on it. A new
609/// thread starts with its creator's signal mask (`dup_task_struct` copies
610/// it,
611/// <https://git.kernel.org/pub/scm/linux/kernel/git/stable/linux.git/tree/kernel/fork.c?h=v6.9#n1100>),
612/// so they are blocked on the calling thread around the spawn and its own
613/// mask is restored after.
614pub fn spawn_blocking_stop_signals<F, T>(
615    builder: std::thread::Builder,
616    run: F,
617) -> std::io::Result<std::thread::JoinHandle<T>>
618where
619    F: FnOnce() -> T + Send + 'static,
620    T: Send + 'static,
621{
622    // SAFETY: sigemptyset(3) and sigaddset(3) write only this local, and
623    // pthread_sigmask(3) reads and writes the two locals it is given.
624    let mut stop: libc::sigset_t = unsafe { std::mem::zeroed() };
625    let mut before: libc::sigset_t = unsafe { std::mem::zeroed() };
626    unsafe {
627        libc::sigemptyset(&mut stop);
628        for signal in STOP_SIGNALS {
629            libc::sigaddset(&mut stop, signal);
630        }
631    }
632    let blocked = unsafe { libc::pthread_sigmask(libc::SIG_BLOCK, &stop, &mut before) };
633    if blocked != 0 {
634        return Err(std::io::Error::from_raw_os_error(blocked));
635    }
636    let spawned = builder.spawn(run);
637    // SAFETY: `before` is the mask pthread_sigmask(3) returned above.
638    unsafe { libc::pthread_sigmask(libc::SIG_SETMASK, &before, std::ptr::null_mut()) };
639    spawned
640}

642/// Call `visit` with the PID of every child of `parent`; the failed call if
643/// `/proc` could not be read whole, when a child may have been missed. A
644/// process that exits during the scan is not a failure.
645///
646/// Reverie gives the trial's container init's PID out only once the trial
647/// has ended, so a signal handler finds the init here: the processes whose
648/// `/proc/<pid>/stat` names `parent` as their parent (`do_task_stat`,
649/// <https://git.kernel.org/pub/scm/linux/kernel/git/stable/linux.git/tree/fs/proc/array.c?h=v6.9#n534>).
650/// `/proc/<pid>/task/<tid>/children` would be shorter, but kernels built
651/// without CONFIG_PROC_CHILDREN (which comes with checkpoint/restore) leave it
652/// out. The directory is [`PROC_DIR`], read from its start; each `stat` file
653/// takes one descriptor while it is read.
654///
655/// Async-signal-safe: system calls on stack buffers, no allocation, no lock,
656/// no panic.
657fn for_each_child(
658    parent: libc::pid_t,
659    mut visit: impl FnMut(libc::pid_t),
660) -> Result<(), Undelivered> {
661    let proc_dir = proc_dir().ok_or_else(|| Undelivered::now("open /proc"))?;
662    // SAFETY: lseek(2) has no memory arguments.
663    if unsafe { libc::lseek(proc_dir, 0, libc::SEEK_SET) } < 0 {
664        return Err(Undelivered::now("lseek /proc"));
665    }
666    let reclen_at = std::mem::offset_of!(libc::dirent64, d_reclen);
667    let name_at = std::mem::offset_of!(libc::dirent64, d_name);
668    let mut entries = [0u8; 4096];
669    loop {
670        // SAFETY: getdents64(2) writes at most `entries.len()` bytes into it.
671        let filled = unsafe {
672            libc::syscall(
673                libc::SYS_getdents64,
674                proc_dir,
675                entries.as_mut_ptr(),
676                entries.len(),
677            )
678        };
679        if filled < 0 {
680            return Err(Undelivered::now("getdents64 /proc"));
681        }
682        let Some(filled) = usize::try_from(filled).ok().filter(|&filled| filled > 0) else {
683            return Ok(());
684        };
685        let mut records = entries.get(..filled).unwrap_or_default();
686        while let Some(length) = records
687            .get(reclen_at..reclen_at + 2)
688            .map(|bytes| usize::from(u16::from_ne_bytes([bytes[0], bytes[1]])))
689            .filter(|&length| length > name_at)
690        {
691            let Some(record) = records.get(..length) else {
692                break;
693            };
694            let name = &record[name_at..];
695            let name = &name[..name
696                .iter()
697                .position(|&byte| byte == 0)
698                .unwrap_or(name.len())];
699            if let Some(pid) = parse_pid(name)
700                && parent_of(proc_dir, name)? == Some(parent)
701            {
702                visit(pid);
703            }
704            records = &records[length..];
705        }
706    }
707}

709/// The parent of the process named `pid` (its decimal digits) in the open
710/// `/proc` directory `proc_dir`; `None` once it has exited, and the failed
711/// call when its `stat` cannot be read otherwise (no descriptor left).
712fn parent_of(proc_dir: libc::c_int, pid: &[u8]) -> Result<Option<libc::pid_t>, Undelivered> {
713    const STAT: &[u8] = b"/stat\0";
714    let mut path = [0u8; 32];
715    let Some(()) = path
716        .get_mut(..pid.len())
717        .map(|name| name.copy_from_slice(pid))
718        .and_then(|()| path.get_mut(pid.len()..pid.len() + STAT.len()))
719        .map(|tail| tail.copy_from_slice(STAT))
720    else {
721        return Ok(None);
722    };
723    // SAFETY: `path` is NUL-terminated within its length.
724    let file = unsafe {
725        libc::openat(
726            proc_dir,
727            path.as_ptr().cast(),
728            libc::O_RDONLY | libc::O_CLOEXEC,
729        )
730    };
731    if file < 0 {
732        let undelivered = Undelivered::now("open /proc/<pid>/stat");
733        return match undelivered.errno {
734            libc::ENOENT | libc::ESRCH => Ok(None),
735            _ => Err(undelivered),
736        };
737    }
738    let mut stat = [0u8; 512];
739    // SAFETY: read(2) writes at most `stat.len()` bytes into it, and `file`
740    // is the descriptor opened above.
741    let read = unsafe { libc::read(file, stat.as_mut_ptr().cast(), stat.len()) };
742    let undelivered = (read < 0).then(|| Undelivered::now("read /proc/<pid>/stat"));
743    unsafe { libc::close(file) };
744    match undelivered {
745        Some(undelivered) if undelivered.errno == libc::ESRCH => Ok(None),
746        Some(undelivered) => Err(undelivered),
747        None => Ok(stat
748            .get(..usize::try_from(read).unwrap_or_default())
749            .and_then(stat_parent)),
750    }
751}

753/// The parent PID in a `/proc/<pid>/stat` line: the field after the state,
754/// which follows the command name. The name is in parentheses and may itself
755/// hold spaces and parentheses, so the fields are counted from its last
756/// closing one.
757fn stat_parent(stat: &[u8]) -> Option<libc::pid_t> {
758    let name_end = stat.iter().rposition(|&byte| byte == b')')?;
759    let mut fields = stat[name_end + 1..]
760        .split(|&byte| byte == b' ')
761        .filter(|field| !field.is_empty());
762    fields.next()?;
763    parse_pid(fields.next()?)
764}
765
766/// A PID written in decimal digits, and nothing else.
767fn parse_pid(digits: &[u8]) -> Option<libc::pid_t> {
768    if digits.is_empty() {
769        return None;
770    }
771    let zero: libc::pid_t = 0;
772    digits.iter().try_fold(zero, |pid, &digit| {
773        if digit.is_ascii_digit() {
774            pid.checked_mul(10)?
775                .checked_add(libc::pid_t::from(digit - b'0'))
776        } else {
777            None
778        }
779    })

1063    /// The process that holds the handler, PID 2 of its own namespace: it
1064    /// starts a child, has the handler run on a helper thread that blocks no
1065    /// signal, and while the handler is paused at that child reaps it, as
1066    /// the thread that drives the trials reaps a trial's container init,
1067    /// and has its parent give the PID to a process of its own.
1068    fn as_analyze(request: libc::c_int, granted: libc::c_int) -> libc::c_int {
1069        let Ok(guard) = StopSignals::install() else {
1070            return ended::NO_HANDLER;
1071        };
1072        if !guard.installed[0] {
1073            return ended::NO_HANDLER;
1074        }
1075        // SAFETY: this process has one thread; the child only pauses.
1076        let child = unsafe { libc::fork() };
1077        if child == 0 {

1131        // struct clone_args through set_tid_size (CLONE_ARGS_SIZE_VER1).
1132        let set_tid: libc::pid_t = pid;
1133        let mut args = [0u64; 10];
1134        args[4] = libc::SIGCHLD as u64; // exit_signal
1135        args[8] = std::ptr::from_ref(&set_tid) as u64; // set_tid
1136        args[9] = 1; // set_tid_size

1138        // thread; the child calls only async-signal-safe functions.
1139        let child = unsafe {
1140            libc::syscall(
1141                libc::SYS_clone3,
1142                args.as_ptr(),
1143                std::mem::size_of_val(&args),
1144            )
1145        };
1146        if child == 0 {
1147            // SAFETY: an all-zero sigset_t is valid for sigemptyset(3).
1148            let mut none: libc::sigset_t = unsafe { std::mem::zeroed() };
1149            unsafe {
1150                libc::sigemptyset(&mut none);
1151                libc::sigprocmask(libc::SIG_SETMASK, &none, std::ptr::null_mut());

1154                }
1155            }

1157            loop {
1158                unsafe { libc::pause() };
1159            }
1160        }

1164    /// PID 1 of the new namespace: a /proc of its own for the handler to
1165    /// read, the process that holds the handler, and on its request a
1166    /// process of its own with the PID it freed. Whether that process
1167    /// received SIGTERM decides the exit.
1168    fn as_namespace_init() -> libc::c_int {
1169        // SAFETY: mount(2) on NUL-terminated literals.
1170        let private = unsafe {
1171            libc::mount(
1172                std::ptr::null(),
1173                c"/".as_ptr(),
1174                std::ptr::null(),
1175                libc::MS_REC | libc::MS_PRIVATE,
1176                std::ptr::null(),
1177            )
1178        };
1179        let proc = unsafe {
1180            libc::mount(
1181                c"proc".as_ptr(),
1182                c"/proc".as_ptr(),
1183                c"proc".as_ptr(),
1184                libc::MS_NOSUID | libc::MS_NODEV | libc::MS_NOEXEC,
1185                std::ptr::null(),
1186            )
1187        };
1188        if private != 0 || proc != 0 {
1189            eprintln!("mount: {}", std::io::Error::last_os_error());
1190            return ended::NO_PROC_OF_ITS_OWN;
1191        }
1192        let (Some((request, request_by)), Some((granted_by, granted)), Some((ready, ready_by))) =
1193            (pipe(), pipe(), pipe())
1194        else {
1195            return ended::NO_PIPE;
1196        };
1197        // SAFETY: this process has one thread.
1198        let analyze = unsafe { libc::fork() };
1199        if analyze == 0 {
1200            // SAFETY: closes this copy's descriptors of the init's ends.
1201            unsafe {
1202                libc::close(request);
1203                libc::close(granted);
1204                libc::close(ready);
1205                libc::close(ready_by);
1206                libc::_exit(as_analyze(request_by, granted_by))
1207            }
1208        }
1209        // SAFETY: closes the init's copies of the ends it gave away, so that
1210        // a read sees the other end close.
1211        unsafe {
1212            libc::close(request_by);
1213            libc::close(granted_by);
1214        }
1215        if analyze < 0 {
1216            return ended::NO_FORK;
1217        }
1218        let mut freed = [0u8; 4];
1219        if !receive(request, &mut freed) {
1220            return exit_code(analyze);
1221        }
1222        let other = child_with_pid(libc::pid_t::from_ne_bytes(freed), ready_by);
1223        let mut up = [0u8];
1224        if other < 0 {
1225            eprintln!("clone3 set_tid: {}", std::io::Error::last_os_error());
1226        }
1227        if other < 0 || !receive(ready, &mut up) {
1228            // SAFETY: kill(2) on this process's own child.
1229            unsafe { libc::kill(analyze, libc::SIGKILL) };
1230            let _ = exit_code(analyze);
1231            return ended::NO_CHOSEN_PID;
1232        }
1233        send(granted, &[1]);
1234        let analyzed = exit_code(analyze);
1235        // A SIGTERM the handler sent at its default action already made the
1236        // process exit with it (`complete_signal`), and the kernel drops the
1237        // SIGKILL sent to a process that is exiting: so the status names the
1238        // signal that came first.
1239        // SAFETY: kill(2) on this process's own child.
1240        unsafe { libc::kill(other, libc::SIGKILL) };
1241        match exit_code(other) {
1242            code if code == 100 + libc::SIGTERM => ended::THE_REUSED_PID_WAS_SIGNALLED,
1243            code if code == 100 + libc::SIGKILL => analyzed,
1244            _ => ended::THE_REUSED_PID_ENDED,
1245        }
1246    }

1295    /// The process holding the handler in [`stop_without_descriptors`]. It
1296    /// starts a child that dies with it (its parent-death signal, as a
1297    /// trial's container init has) and waits until the child has armed that
1298    /// signal while this process is still its parent, tells the child's PID
1299    /// on `tell`, with `full_stderr` fills its stderr pipe so that a write
1300    /// there blocks, uses up every descriptor it may open, and sends itself
1301    /// SIGTERM. Exiting is the regression: the stop was swallowed.
1302    fn as_analyze_without_descriptors(
1303        says: libc::c_int,
1304        tell: libc::c_int,
1305        full_stderr: bool,
1306    ) -> libc::c_int {
1307        // SAFETY: dup2(2) and close(2) on descriptors this process owns.
1308        unsafe {
1309            libc::dup2(says, 2);
1310            libc::close(says);
1311        }
1312        let Ok(guard) = StopSignals::install() else {
1313            return ended::NO_HANDLER;
1314        };
1315        if !guard.installed[0] {
1316            return ended::NO_HANDLER;
1317        }
1318        let Some((armed, arms)) = pipe() else {
1319            return ended::NO_FORK;
1320        };
1321        // SAFETY: getpid(2) has no preconditions; this process has one
1322        // thread, and the child only arms its signal, reports and pauses.
1323        let parent = unsafe { libc::getpid() };
1324        let child = unsafe { libc::fork() };
1325        if child == 0 {
1326            unsafe {
1327                libc::close(2);
1328                libc::close(tell);
1329                libc::close(armed);
1330                let ready = libc::prctl(libc::PR_SET_PDEATHSIG, libc::SIGKILL) == 0
1331                    && libc::getppid() == parent;
1332                if !ready || !send(arms, b"A") {
1333                    libc::_exit(1);
1334                }
1335                loop {
1336                    libc::pause();
1337                }
1338            }
1339        }
1340        // SAFETY: closes this process's copy of the child's end.
1341        unsafe { libc::close(arms) };
1342        let mut ready = [0u8; 1];
1343        if child < 0 || !receive(armed, &mut ready) || !send(tell, &child.to_ne_bytes()) {
1344            return ended::NO_FORK;
1345        }
1346        // SAFETY: fcntl(2) and write(2) on this process's stderr, a pipe
1347        // whose reader is alive; setrlimit(2) reads `limit`; open(2) on a
1348        // NUL-terminated literal; raise(3) runs the handler on this thread.
1349        unsafe {
1350            libc::close(armed);
1351            libc::close(tell);
1352            if full_stderr {
1353                let flags = libc::fcntl(2, libc::F_GETFL);
1354                libc::fcntl(2, libc::F_SETFL, flags | libc::O_NONBLOCK);
1355                let chunk = [b'x'; 4096];
1356                while libc::write(2, chunk.as_ptr().cast(), chunk.len()) > 0 {}
1357                libc::fcntl(2, libc::F_SETFL, flags);
1358            }
1359            let mut limit: libc::rlimit = std::mem::zeroed();
1360            libc::getrlimit(libc::RLIMIT_NOFILE, &mut limit);
1361            limit.rlim_cur = limit.rlim_cur.min(64);
1362            libc::setrlimit(libc::RLIMIT_NOFILE, &limit);
1363            while libc::open(c"/dev/null".as_ptr(), libc::O_RDONLY) >= 0 {}
1364            libc::raise(libc::SIGTERM);
1365        }
1366        std::mem::forget(guard);
1367        ended::STOP_SWALLOWED
1368    }

1370    /// `pid`'s wait status once it ends, if it does within `bound`; else it
1371    /// is killed through a pidfd (it is this process's child, not yet
1372    /// reaped, so the pidfd names it) and reaped, and `None` returned.
1373    fn wait_within(pid: libc::pid_t, bound: Duration) -> Option<libc::c_int> {
1374        let deadline = std::time::Instant::now() + bound;
1375        let mut raw = 0;
1376        loop {
1377            // SAFETY: `raw` outlives the call; `pid` is this process's child.
1378            let waited = unsafe { libc::waitpid(pid, &mut raw, libc::WNOHANG) };
1379            if waited == pid {
1380                return Some(raw);
1381            }
1382            if waited < 0 || std::time::Instant::now() >= deadline {
1383                break;
1384            }
1385            std::thread::sleep(Duration::from_millis(10));
1386        }
1387        // SAFETY: pidfd_open(2) and pidfd_send_signal(2) without siginfo;
1388        // close(2) of the pidfd opened here; waitpid(2) reaps the child.
1389        unsafe {
1390            let pidfd = libc::syscall(libc::SYS_pidfd_open, pid, 0) as libc::c_int;
1391            if pidfd >= 0 {
1392                libc::syscall(
1393                    libc::SYS_pidfd_send_signal,
1394                    pidfd,
1395                    libc::SIGKILL,
1396                    std::ptr::null::<libc::siginfo_t>(),
1397                    0,
1398                );
1399                libc::close(pidfd);
1400            }
1401            libc::waitpid(pid, &mut raw, 0);
1402        }
1403        None
1404    }

1407    /// say how it ended (its wait status, `None` if it had to be killed after
1408    /// the bound), what it wrote on stderr (read once it and its child have
1409    /// ended, so a full pipe stays full until then), and how its child ended.
1410    fn stop_without_descriptors(
1411        full_stderr: bool,
1412    ) -> (Option<libc::c_int>, String, Option<libc::c_int>) {

1414        // its parent dies, comes to this process to be reaped.
1415        unsafe { libc::prctl(libc::PR_SET_CHILD_SUBREAPER, 1, 0, 0, 0) };
1416        let (Some((said, says)), Some((told, tell))) = (pipe(), pipe()) else {
1417            panic!("pipe: {}", std::io::Error::last_os_error());
1418        };
1419        // SAFETY: the copy's other thread waits for this test and holds no
1420        // lock.
1421        let analyze = unsafe { libc::fork() };
1422        if analyze == 0 {
1423            unsafe {
1424                libc::close(said);
1425                libc::close(told);
1426                libc::_exit(as_analyze_without_descriptors(says, tell, full_stderr))
1427            }
1428        }
1429        assert!(analyze > 0, "fork: {}", std::io::Error::last_os_error());
1430        // SAFETY: closes this process's copies of the ends the child uses.
1431        unsafe {
1432            libc::close(says);
1433            libc::close(tell);
1434        }
1435        let ended = wait_within(analyze, Duration::from_secs(20));
1436        let mut child = [0u8; 4];
1437        let child = receive(told, &mut child)
1438            .then(|| libc::pid_t::from_ne_bytes(child))
1439            .and_then(|child| wait_within(child, Duration::from_secs(20)));
1440        let mut stderr = Vec::new();
1441        let mut chunk = [0u8; 4096];
1442        loop {
1443            // SAFETY: read(2) writes at most `chunk.len()` bytes into it.

1445            let Some(got) = usize::try_from(got).ok().filter(|&got| got > 0) else {
1446                break;
1447            };
1448            stderr.extend_from_slice(&chunk[..got]);
1449        }
1450        // SAFETY: closes the ends this test opened.
1451        unsafe {
1452            libc::close(said);
1453            libc::close(told);
1454        }
1455        let stderr = String::from_utf8_lossy(&stderr);
1456        (ended, stderr.trim_start_matches('x').to_owned(), child)
1457    }
