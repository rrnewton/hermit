9//! The report's "Racing accesses" section (design section 8): the final
10//! replays on each side of the deciding event record an access trace, the
11//! finder ([`races`](super::races)) compares the two, and one more replay of
12//! the target continuation takes the two accesses' stacks.
13//!
14//! The section never fails the analysis: whatever stops it (a backend or mode
15//! the access trace refuses, an unreadable trace) becomes status
16//! `unavailable` with the reason, and a stack replay that goes wrong leaves
17//! the stacks out with a note. A replay that left its schedule before the
18//! deciding event (the two-seed search's interleavings cannot always be
19//! realized) is compared as it ran: from the first event at which the two
20//! replays run different threads, the event the stack recovery uses too.

78/// A final replay that ran with an access trace.
79pub struct TracedReplay {
80    role: &'static str,
81    trace: Result<PathBuf, String>,
82    /// Where it left the schedule it had to follow up to the deciding event,
83    /// if it did.

85    /// The schedule it realized (empty if it recorded none), recorded at
86    /// `realized_path`.
87    pub realized: Vec<SchedEvent>,

90    /// whether it followed its schedule is unknown.
91    schedule_error: Option<String>,

94impl TraceRequest {
95    /// The replay ran and recorded the schedule it realized at
96    /// `realized_path`, which must follow `requested` through its first
97    /// `until` events.
98    pub fn ran(self, requested: &[SchedEvent], realized_path: &Path, until: usize) -> TracedReplay {
99        let read = if realized_path.exists() {
100            Recording::read(realized_path)
101                .and_then(|recording| recording.into_legacy("the racing-accesses section"))
102                .map(PreemptionRecord::into_global)
103        } else {
104            Ok(Vec::new())
105        };
106        match read {
107            Ok(realized) => {
108                let departure = departure(requested, &realized, until);
109                self.ran_with(departure, realized, realized_path)
110            }
111            Err(error) => {
112                let mut replay = self.ran_with(None, Vec::new(), realized_path);
113                replay.schedule_error = Some(format!("{error:#}"));
114                replay
115            }
116        }
117    }

155    /// trace keeps from `--access-trace-from` on, which are the lookback's

181    /// The section for `target` and `baseline`, two replays whose schedules
182    /// agree before `decision`. `stack_run` builds the replay of the target
183    /// continuation the stacks are taken from (it gets the run's name).
184    ///
185    /// When a replay left its schedule before `decision`, the two replays
186    /// are compared as they ran instead, from the first event at which their
187    /// realized schedules run different threads: before it they ran alike,
188    /// so the window lemma holds there, and the traces record what ran. That
189    /// takes `target_met`, the target replay still meeting the target: two
190    /// runs that both pass would otherwise yield a confident pair, so
191    /// without it a replay that left its schedule leaves the section
192    /// unavailable.
193    ///
194    /// Candidates are checked by intervention, a control replay and an
195    /// edited one each for at most [`CHECKED_CANDIDATES`]: `check_run` builds
196    /// a replay that follows the first `replay` events of a schedule of the
197    /// `role` run and then continues as the search's trials did. It gets the
198    /// run's name, `role`, the schedule, how many of its first events are as
199    /// that run realized them (whose callback checkpoints still apply), and
200    /// `replay`.
201    pub(super) fn racing_accesses(
202        &self,
203        decision: usize,
204        target: TracedReplay,
205        baseline: TracedReplay,
206        target_met: bool,
207        stack_run: impl FnOnce(&str) -> anyhow::Result<RunData>,
208        mut check_run: impl FnMut(&str, &str, &[SchedEvent], usize, usize) -> anyhow::Result<RunData>,
209    ) -> RacingAccesses {
210        let mut decision = decision as EventIndex;
211        let realized = if target_met {
212            realized_window(decision, &target, &baseline)
213        } else {
214            None
215        };
216        if let Some((split, _)) = &realized {
217            decision = *split;
218        }
219        let as_ran = realized.is_some();
220        let traces = [&target, &baseline].map(|replay| self.read_trace(decision, replay, as_ran));
221        let [target_trace, baseline_trace] = match traces {
222            [Ok(target), Ok(baseline)] => [target, baseline],
223            [Err(reason), _] | [_, Err(reason)] => {
224                return RacingAccesses::unavailable(decision, reason);
225            }
226        };
227        // A replay that left its schedule before its trace begins left too
228        // early to compare the two as they ran: no --access-trace-from of the
229        // final replays reaches back there.
230        if as_ran {
231            let first = target_trace
232                .trace
233                .header
234                .from
235                .max(baseline_trace.trace.header.from);
236            if decision < first {
237                let role = [&target, &baseline]
238                    .into_iter()
239                    .find(|replay| replay.departure.is_some())
240                    .map_or("target", |replay| replay.role);
241                return RacingAccesses::unavailable(
242                    decision,
243                    format!(
244                        "the {role} replay left its schedule too early: the two replays first \
245                         run different threads at event {decision}, before their traces begin \
246                         (event {first})"
247                    ),
248                );
249            }
250        }
251        // The root process's thread runs the first event of every schedule;
252        // as a thread-group leader its thread id is its process id.
253        let root = target
254            .realized
255            .first()
256            .map(|event| DetPid::from_raw(event.dettid.as_raw()));
257        let criteria = OutcomeCriteria {
258            stdout: self.target_stdout.as_ref(),
259            stderr: self.target_stderr.as_ref(),
260            exit_code: &self.target_exit_code,
261            root,
262        };
263        let Outcome {
264            point: outcome,
265            unknown: unknown_outcome,
266        } = outcome(&target_trace.trace, decision, &criteria);
267        let uncounted_signal =
268            uncounted_signal_outcome(&target_trace.trace, decision, &criteria, outcome.as_ref());
269        let outcome_thread = outcome.as_ref().map(|o| o.thread);
270        // The thread the target replay ran at the deciding event, as it
271        // realized its schedule (which follows the requested one up to there).
272        let thread = usize::try_from(decision)
273            .ok()
274            .and_then(|at| target.realized.get(at))
275            .map(|event| event.dettid);
276        let mut result = find_racing_accesses(
277            &target_trace.trace,
278            &baseline_trace.trace,
279            Window {
280                decision,
281                outcome,
282                thread,
283            },
284            outcome_thread,
285        );
286        result.notes.extend(unknown_outcome);
287        for (role, read) in [("target", &target_trace), ("baseline", &baseline_trace)] {
288            if read.capped {
289                result.notes.push(format!(
290                    "--race-window={} kept that many records of the {role} trace after the \
291                     deciding event; raise it to read further.",
292                    self.race_window
293                ));
294            }
295        }
296        result.notes.extend(uncounted_signal);
297        if let Some((_, note)) = realized {
298            result.notes.insert(0, note);
299        }
300        result.trace_files = vec![target_trace.file.clone(), baseline_trace.file.clone()];
301        if result.status == RaceStatus::Candidates {
302            let mut replay =
303                |name: &str, role: &str, schedule: &[SchedEvent], reversal: &Reversal| {
304                    check_replay(name, role, schedule, reversal, &mut check_run)
305                };
306            for (i, candidate) in result
307                .candidates
308                .iter_mut()
309                .take(CHECKED_CANDIDATES)
310                .enumerate()
311            {
312                candidate.check = Some(check_candidate(
313                    i + 1,
314                    candidate,
315                    [&target, &baseline],
316                    &mut replay,
317                ));
318            }
319            promote_the_deciding_candidate(&mut result);
320        }
321        if matches!(
322            result.status,
323            RaceStatus::Found
324                | RaceStatus::Timing
325                | RaceStatus::Candidates
326                | RaceStatus::FoundByIntervention
327        ) {
328            self.attach_stacks(&mut result, &target_trace.trace, stack_run);
329        }
330        result
331    }

333    /// Read one side's trace, or say why the section cannot use it. With
334    /// `as_ran`, the replays are compared as they ran, so leaving the
335    /// schedule does not disqualify one.
336    fn read_trace(
337        &self,
338        decision: EventIndex,
339        replay: &TracedReplay,
340        as_ran: bool,
341    ) -> Result<ReadTrace, String> {
342        let role = replay.role;
343        if let Some(error) = &replay.schedule_error {
344            return Err(format!(
345                "the {role} replay's realized schedule {} cannot be read ({error}), so whether \
346                 it followed its schedule is unknown",
347                replay.realized_path.display()
348            ));
349        }
350        if let Some(departure) = replay.departure.as_ref().filter(|_| !as_ran) {
351            return Err(format!(
352                "the {role} replay left its schedule at or before the deciding event (it \
353                 {departure}), \
354                 so its accesses are not those of the {role} order"
355            ));
356        }
357        let path = replay
358            .trace
359            .as_ref()
360            .map_err(|refused| format!("the {role} replay records no access trace: {refused}"))?;
361        let (trace, capped) = read_capped(path, decision, self.race_window).map_err(|error| {
362            format!(
363                "the {role} replay's access trace {} cannot be read: {error}",
364                path.display()
365            )
366        })?;
367        Ok(ReadTrace {
368            trace,
369            file: file_name(path),
370            capped,
371        })
372    }

374    /// Take both accesses' stacks from one replay of the target continuation,
375    /// at their entry events. A stack the replay did not take, or took on
376    /// another thread (it left the schedule), is left out with a note.
377    fn attach_stacks(
378        &self,
379        result: &mut RacingAccesses,
380        target: &AccessTrace,
381        stack_run: impl FnOnce(&str) -> anyhow::Result<RunData>,
382    ) {
383        // The reported pair's accesses, or the first candidate's: one replay
384        // takes both. The other candidates rank lower and keep no stacks, so
385        // a report pays for one replay whatever the status.
386        let RacingAccesses {
387            accesses,
388            candidates,
389            notes,
390            ..
391        } = result;
392        let pair: &mut [RacingAccess] = match candidates.first_mut() {
393            Some(first) if accesses.is_empty() => &mut first.accesses,
394            _ => accesses.as_mut_slice(),
395        };
396        let entries: Vec<Option<EventIndex>> = pair
397            .iter()
398            .map(|access| entry_event(target, access.thread, access.nth))
399            .collect();
400        match self.replay_for_access_stacks(&entries, stack_run) {
401            Ok(stacks) => {
402                for ((access, entry), stack) in pair.iter_mut().zip(&entries).zip(stacks) {
403                    if access.ended_first_by.is_some() {
404                        notes.push(format!(
405                            "Thread {}'s {} runs in the baseline only, so the target replay \
406                             takes no stack for it.",
407                            access.thread,
408                            access.call.split('(').next().unwrap_or_default()
409                        ));
410                        continue;
411                    }
412                    match stack {
413                        Some(stack) if stack_thread(&stack) == Some(access.thread) => {
414                            access.stack = Some(stack);
415                        }
416                        Some(_) => notes.push(format!(
417                            "The stack replay ran another thread than thread {} at event {}, so \
418                             that access's stack is left out.",
419                            access.thread,
420                            entry.unwrap_or_default()
421                        )),
422                        None => notes.push(format!(
423                            "The stack replay took no stack for thread {}'s access.",
424                            access.thread
425                        )),
426                    }
427                }
428            }
429            Err(error) => notes.push(format!(
430                "The stacks are not available: the stack replay failed ({error:#})."
431            )),
432        }
433    }

435    /// Launch `stack_run` with stack events at `entries` and read the stacks
436    /// it took, one per entry. A replay that a bound stops or Hermit refuses
437    /// still yields what it took.
438    fn replay_for_access_stacks(
439        &self,
440        entries: &[Option<EventIndex>],
441        stack_run: impl FnOnce(&str) -> anyhow::Result<RunData>,
442    ) -> anyhow::Result<Vec<Option<PrettyBacktrace>>> {
443        let runname = "final_racing_access_stacks";
444        let mut run = stack_run(runname)?;
445        let root = run.root_path();
446        let paths: Vec<PathBuf> = (0..entries.len())
447            .map(|i| root.with_extension(format!("access{}.stack", i + 1)))
448            .collect();
449        let mut events: Vec<(u64, Option<PathBuf>)> = entries
450            .iter()
451            .zip(&paths)
452            .filter_map(|(entry, path)| Some(((*entry)?, Some(path.clone()))))
453            .collect();
454        events.sort_by_key(|(event, _)| *event);
455        run.runopts.det_opts.det_config.stacktrace_event = events;
456        match run.launch() {
457            Ok(()) => {}
458            Err(error)
459                if error.downcast_ref::<RunTimeoutMarker>().is_some()
460                    || guest_reached_a_refusal(&error) =>
461            {
462                eprintln!(
463                    ":: {} the racing-access stack replay did not finish ({error:#}); using the \
464                     stacks it took",
465                    "WARNING:".yellow().bold(),
466                );
467            }
468            Err(error) => return Err(error),
469        }
470        Ok(paths.iter().map(|path| read_stack(path)).collect())
471    }
472}

474/// How a check's replay ended, and where it left the schedule it followed
475/// (or why its recorded schedule gives none).
476type CheckOutcome = (anyhow::Result<TrialEnd>, Result<Option<Departure>, String>);
477
478/// How a check's replay that completed ended. One that did not complete has
479/// no outcome: its launch's error leaves the check unchecked.
480#[derive(Debug, Clone, Copy, PartialEq, Eq)]
481enum TrialEnd {
482    /// It met the target criteria.
483    Met,
484    /// It did not.
485    Missed,
486}

488/// Check `candidate` (the `index`th) by intervention: replay one of the
489/// two runs ([`edit_for`]) with the candidate's turn order reversed
490/// ([`reverse_turns`], which says what it moved), then continue as the
491/// search's trials did. A control replay of the same run without the
492/// reversal, to the same point and continued the same way, comes first:
493/// unless it ends as the run did, a changed outcome could come from where
494/// the replay stops following the schedule rather than from the
495/// reversal. Where a replay left its schedule is read before its outcome.
496fn check_candidate(
497    index: usize,
498    candidate: &RaceCandidate,
499    [target, baseline]: [&TracedReplay; 2],
500    replay: &mut impl FnMut(&str, &str, &[SchedEvent], &Reversal) -> CheckOutcome,
501) -> CandidateCheck {
502    let unchecked =
503        |edited: &str, edit: String, trial: Option<String>, why: String| CandidateCheck {
504            verdict: CheckVerdict::NotChecked,
505            edited: edited.to_string(),
506            edit,
507            moved: None,
508            trial,
509            detail: format!("was not checked: {why}"),
510        };
511    let Some(edit) = edit_for(candidate) else {
512        return unchecked(
513            "target",
514            "its two calls".to_string(),
515            None,
516            "one of them has no event in the target, and the baseline does not run them the \
517             other way round"
518                .to_string(),
519        );
520    };
521    let schedule = if edit.met_decides {
522        &baseline.realized
523    } else {
524        &target.realized
525    };
526    let reversal = match reverse_turns(schedule, edit.earlier, edit.later) {
527        Ok(reversal) => reversal,
528        Err(why) => {
529            return unchecked(
530                edit.role,
531                format!("events #{} and #{}", edit.later, edit.earlier),
532                None,
533                why,
534            );
535        }
536    };
537    let trial = format!("racing_candidate_{index}");
538    let control = format!("{trial}_control");
539    eprintln!(
540        ":: Checking race candidate {index}: reversing {} in the {} run, after a control \
541         replay without the reversal",
542        reversal.edit, edit.role
543    );
544    let (ended, left) = replay(&control, edit.role, schedule, &reversal);
545    if let Err(why) = control_reproduces(ended, left, edit.met_decides) {
546        return CandidateCheck {
547            moved: Some(reversal.moved),
548            ..unchecked(edit.role, reversal.edit, Some(control), why)
549        };
550    }
551    let (ended, left) = replay(&trial, edit.role, &reversal.schedule, &reversal);
552    let (verdict, detail) = verdict_of(ended, left, edit.met_decides);
553    CandidateCheck {
554        verdict,
555        edited: edit.role.to_string(),
556        edit: reversal.edit,
557        moved: Some(reversal.moved),
558        trial: Some(trial),
559        detail,
560    }
561}

567/// Which run a candidate's check edits, and the two events it reverses.
568#[derive(Debug, Clone, Copy, PartialEq, Eq)]
569struct Edit {
570    role: &'static str,
571    /// The event that runs first in the edited run, and the one after it.
572    earlier: usize,
573    later: usize,
574    /// Whether the reversal decides when the replay meets the target (an
575    /// edited baseline, which misses it) rather than when it misses it (an
576    /// edited target).
577    met_decides: bool,
578}
579
580/// The edit that reverses `candidate`'s order: in the target, where its
581/// first access runs before its second; or, when an access has no event in
582/// the target (a wait still pending at the outcome), in the baseline, where
583/// they run the other way round. `None` when neither run holds both.
584fn edit_for(candidate: &RaceCandidate) -> Option<Edit> {
585    let [first, second] = &candidate.accesses;
586    match (
587        first.event,
588        second.event,
589        first.baseline_event,
590        second.baseline_event,
591    ) {
592        (Some(a), Some(b), ..) if a < b => Some(Edit {
593            role: "target",
594            earlier: usize::try_from(a).ok()?,
595            later: usize::try_from(b).ok()?,
596            met_decides: false,
597        }),
598        (_, _, Some(a), Some(b)) if b < a => Some(Edit {
599            role: "baseline",
600            earlier: usize::try_from(b).ok()?,
601            later: usize::try_from(a).ok()?,
602            met_decides: true,
603        }),
604        _ => None,
605    }
606}

626/// The run of consecutive events of one thread in `schedule` that holds
627/// `index`: that thread's turn.
628fn turn_at(schedule: &[SchedEvent], index: usize) -> std::ops::Range<usize> {
629    let thread = schedule[index].dettid;
630    let start = schedule[..index]
631        .iter()
632        .rposition(|event| event.dettid != thread)
633        .map_or(0, |i| i + 1);
634    let end = schedule[index..]
635        .iter()
636        .position(|event| event.dettid != thread)
637        .map_or(schedule.len(), |i| index + i);
638    start..end
639}

641/// Whether `event` is the return of a timed sleep, which cannot come before
642/// its deadline.
643fn is_timed_return(event: &SchedEvent) -> bool {
644    matches!(
645        event.op,
646        Op::Syscall(
647            Sysno::nanosleep | Sysno::clock_nanosleep,
648            SyscallPhase::Posthook
649        )
650    )
651}

653/// `schedule` with the turn holding event `later` (thread Y's) run before
654/// the turn holding event `earlier` (thread X's). Over the span from the
655/// start of X's turn to the end of Y's, Y's events move to its front; when
656/// one of them is a timed sleep's return, which cannot run before its
657/// deadline, X's events move to its back instead. That does not always
658/// help: holding X back pulls the other threads' events earlier, Y's sleep
659/// return among them, which can then still land before its deadline; such
660/// a replay leaves the schedule there, and the check is not_realized,
661/// naming the timed wait ([`not_realized`]). Each thread keeps the order of
662/// its own events, and the moved events cross every event of the other
663/// threads in the span: the result says how many moved and which threads,
664/// besides the pair, they crossed. An error when the two events are on one
665/// thread or not in that order.
666pub(super) fn reverse_turns(
667    schedule: &[SchedEvent],
668    earlier: usize,
669    later: usize,
670) -> Result<Reversal, String> {
671    if earlier >= later || later >= schedule.len() {
672        return Err(format!(
673            "the realized schedule does not hold events #{earlier} and #{later} in that order"
674        ));
675    }
676    let (x, y) = (schedule[earlier].dettid, schedule[later].dettid);
677    if x == y {
678        return Err("the two calls are on one thread".to_string());
679    }
680    let span = turn_at(schedule, earlier).start..turn_at(schedule, later).end;
681    let events = &schedule[span.clone()];
682    let advance = !events.iter().any(|e| e.dettid == y && is_timed_return(e));
683    let (mover, other, past, direction) = if advance {
684        (y, x, earlier, "before")
685    } else {
686        (x, y, later, "after")
687    };
688    let moved: Vec<SchedEvent> = events
689        .iter()
690        .filter(|e| e.dettid == mover)
691        .cloned()
692        .collect();
693    let rest: Vec<SchedEvent> = events
694        .iter()
695        .filter(|e| e.dettid != mover)
696        .cloned()
697        .collect();
698    let mut crossed: Vec<CrossedThread> = Vec::new();
699    for event in rest.iter().filter(|e| e.dettid != other) {
700        match crossed.iter_mut().find(|c| c.thread == event.dettid) {
701            Some(c) => c.events += 1,
702            None => crossed.push(CrossedThread {
703                thread: event.dettid,
704                events: 1,
705            }),
706        }
707    }
708    let mut edited = schedule[..span.start].to_vec();
709    if advance {
710        edited.extend(moved.iter().cloned());
711        edited.extend(rest);
712    } else {
713        edited.extend(rest);
714        edited.extend(moved.iter().cloned());
715    }
716    edited.extend_from_slice(&schedule[span.end..]);
717    let count = |n: usize| format!("{n} event{}", if n == 1 { "" } else { "s" });
718    let crossing = if crossed.is_empty() {
719        "crossing no other thread".to_string()
720    } else {
721        format!(
722            "crossing {}",
723            crossed
724                .iter()
725                .map(|c| format!("thread {} ({})", c.thread, count(c.events)))
726                .collect::<Vec<_>>()
727                .join(", ")
728        )
729    };
730    Ok(Reversal {
731        edit: format!(
732            "moving {} of thread {mover} {direction} thread {other}'s turn at #{past}, {crossing}",
733            count(moved.len())
734        ),
735        moved: MovedEvents {
736            thread: mover,
737            events: moved.len(),
738            direction: direction.to_string(),
739            past_event: past as EventIndex,
740            crossed,
741        },
742        schedule: edited,
743        unchanged: span.start,
744        replay: span.end,
745    })
746}

748/// Replay the first `reversal.replay` events of `schedule` (edited, or not
749/// for a control) of the `role` run with `check_run`, then continue as the
750/// search's trials did: how it ended, and where it left `schedule`, read
751/// from the schedule it recorded.
752fn check_replay(
753    runname: &str,
754    role: &str,
755    schedule: &[SchedEvent],
756    reversal: &Reversal,
757    check_run: &mut impl FnMut(&str, &str, &[SchedEvent], usize, usize) -> anyhow::Result<RunData>,
758) -> CheckOutcome {
759    let mut run = match check_run(runname, role, schedule, reversal.unchanged, reversal.replay) {
760        Ok(run) => run.with_schedule_recording(),
761        Err(error) => {
762            return (
763                Err(error.context("its replay could not be set up")),
764                Ok(None),
765            );
766        }
767    };
768    let path = run.sched_path_out().to_path_buf();
769    // Only a replay that completed has an outcome; one that did not is an
770    // error, which leaves the check unchecked.
771    let ended = run.launch().map(|()| {
772        if run.is_a_match() {
773            TrialEnd::Met
774        } else {
775            TrialEnd::Missed
776        }
777    });
778    let left =
779        realized_schedule(&path).map(|realized| departure(schedule, &realized, reversal.replay));
780    (ended, left)
781}

783/// The schedule a check's replay recorded at `path`, or why it gives none:
784/// absent, unreadable (a replay cut short leaves a partial one) or empty.
785fn realized_schedule(path: &Path) -> Result<Vec<SchedEvent>, String> {
786    if !path.exists() {
787        return Err("its replay recorded no schedule".to_string());
788    }
789    let realized = Recording::read(path)
790        .and_then(|recording| recording.into_legacy("a race candidate's check"))
791        .map(PreemptionRecord::into_global)
792        .map_err(|error| format!("the schedule its replay recorded cannot be read ({error:#})"))?;
793    if realized.is_empty() {
794        return Err("the schedule its replay recorded is empty".to_string());
795    }
796    Ok(realized)
797}

799/// Whether a check's control replay (the run's schedule without the
800/// reversal, replayed to the same point and continued the same way) ended
801/// as the run did: met the target for an edited target, missed it for an
802/// edited baseline (`met_decides`). Otherwise why the check cannot count.
803fn control_reproduces(
804    ended: anyhow::Result<TrialEnd>,
805    left: Result<Option<Departure>, String>,
806    met_decides: bool,
807) -> Result<(), String> {
808    let ended = ended.map_err(|error| format!("its control replay failed ({error:#})"))?;
809    match left {
810        Err(why) => return Err(format!("its control replay: {why}")),
811        Ok(Some(left)) => {
812            return Err(format!(
813                "its control replay, without the reversal, left the schedule: the replay {left}"
814            ));
815        }
816        Ok(None) => {}
817    }
818    match (ended, met_decides) {
819        (TrialEnd::Met, false) | (TrialEnd::Missed, true) => Ok(()),
820        (TrialEnd::Met, true) => Err(
821            "the continuation alone changes the outcome: replayed to the same point without the \
822             reversal and continued, the baseline run meets the target"
823                .to_string(),
824        ),
825        (TrialEnd::Missed, false) => Err(
826            "the continuation alone changes the outcome: replayed to the same point without the \
827             reversal and continued, the target run misses the target"
828                .to_string(),
829        ),
830    }
831}

833/// What a check's replay showed: `left`, where it left the edited schedule,
834/// is read before its outcome; then the outcome decides when it differs
835/// from the edited run's (met for an edited baseline, missed for an edited
836/// target, as `met_decides` says).
837fn verdict_of(
838    ended: anyhow::Result<TrialEnd>,
839    left: Result<Option<Departure>, String>,
840    met_decides: bool,
841) -> (CheckVerdict, String) {
842    let ended = match ended {
843        Ok(ended) => ended,
844        Err(error) => {
845            return (
846                CheckVerdict::NotChecked,
847                format!("was not checked: its replay failed ({error:#})"),
848            );
849        }
850    };
851    match left {
852        Err(why) => return (CheckVerdict::NotChecked, format!("was not checked: {why}")),
853        Ok(Some(left)) => {
854            return (CheckVerdict::NotRealized, not_realized(&left));
855        }
856        Ok(None) => {}
857    }
858    match (ended, met_decides) {
859        (TrialEnd::Met, true) => (
860            CheckVerdict::Decides,
861            "meets the target; the unedited control misses it".to_string(),
862        ),
863        (TrialEnd::Missed, false) => (
864            CheckVerdict::Decides,
865            "misses the target; the unedited control meets it".to_string(),
866        ),
867        (TrialEnd::Met, false) => (
868            CheckVerdict::DoesNotDecide,
869            "still meets the target, as the unedited control does".to_string(),
870        ),
871        (TrialEnd::Missed, true) => (
872            CheckVerdict::DoesNotDecide,
873            "still misses the target, as the unedited control does".to_string(),
874        ),
875    }
876}

878/// Why a check's replay shows nothing: where it left the edited schedule.
879/// Reaching a timed sleep's return early is named, since no reversal that
880/// runs the return sooner can be replayed: the sleep has not expired there.
881fn not_realized(left: &Departure) -> String {
882    match left {
883        Departure::OtherEvent { expected, .. } if is_timed_return(expected) => format!(
884            "cannot be replayed as edited: the edit moves thread {}'s timed wait return before \
885             its deadline (the replay {left})",
886            expected.dettid
887        ),
888        _ => format!("cannot be replayed as edited: the replay {left}"),
889    }
890}

892/// With exactly one candidate whose check decides, report it as found by
893/// intervention: its accesses become the pair, and the candidates stay
894/// listed with their checks.
895fn promote_the_deciding_candidate(result: &mut RacingAccesses) {
896    let deciding: Vec<usize> = result
897        .candidates
898        .iter()
899        .enumerate()
900        .filter(|(_, c)| {
901            c.check
902                .as_ref()
903                .is_some_and(|check| check.verdict == CheckVerdict::Decides)
904        })
905        .map(|(i, _)| i)
906        .collect();
907    let [index] = deciding.as_slice() else {
908        return;
909    };
910    let candidate = &result.candidates[*index];
911    let Some(check) = &candidate.check else {
912        return;
913    };
914    result.reason = format!(
915        "reversing this turn order ({}) in the {} run {} ({}.repro in the workspace), and no \
916         other checked candidate's reversal changes the outcome",
917        check.edit,
918        check.edited,
919        check.detail,
920        check.trial.as_deref().unwrap_or("the check")
921    );
922    result.status = RaceStatus::FoundByIntervention;
923    result.resource = candidate.resource.clone();
924    result.accesses = candidate.accesses.to_vec();
925    result.order = Some(candidate.order.clone());
926}

928/// Where to compare `target` and `baseline` as they ran, when one of them
929/// left its schedule before `decision`: the first event at which their
930/// realized schedules run different threads (or the shorter one ends), with
931/// the note that says so. `None` when both followed their schedules, or a
932/// realized schedule is missing or unreadable.
933fn realized_window(
934    decision: EventIndex,
935    target: &TracedReplay,
936    baseline: &TracedReplay,
937) -> Option<(EventIndex, String)> {
938    let (role, departure) = [target, baseline]
939        .into_iter()
940        .find_map(|replay| Some((replay.role, replay.departure.as_ref()?)))?;
941    if [target, baseline]
942        .iter()
943        .any(|replay| replay.schedule_error.is_some() || replay.realized.is_empty())
944    {
945        return None;
946    }
947    let split = first_thread_difference(&target.realized, &baseline.realized)? as EventIndex;
948    Some((
949        split,
950        format!(
951            "The {role} replay left its schedule at or before the deciding event {decision} (it \
952             {departure}), so the two replays are compared as they ran, from event {split}, the \
953             first at which they run different threads; event numbers here are theirs."
954        ),
955    ))
956}

958/// The first index at which `a` and `b` run different threads, or at which
959/// the shorter one ends; `None` when they run the same threads throughout.
960fn first_thread_difference(a: &[SchedEvent], b: &[SchedEvent]) -> Option<usize> {
961    a.iter()
962        .zip(b)
963        .position(|(x, y)| x.dettid != y.dettid)
964        .or_else(|| (a.len() != b.len()).then(|| a.len().min(b.len())))
965}

967/// The trace at `path`, read up to `window` records whose last event is at or
968/// after `decision`; whether the window stopped the read.
969fn read_capped(
970    path: &Path,
971    decision: EventIndex,
972    window: usize,
973) -> anyhow::Result<(AccessTrace, bool)> {
974    let mut kept = Vec::new();
975    let mut after = 0;
976    let mut capped = false;
977    for line in BufReader::new(File::open(path)?).lines() {
978        let line = line?;
979        let reaches = serde_json::from_str::<TraceRecord>(&line)
980            .is_ok_and(|record| record_reaches(&record, decision));
981        if reaches {
982            if after == window {
983                capped = true;
984                break;
985            }
986            after += 1;
987        }
988        kept.extend_from_slice(line.as_bytes());
989        kept.push(b'\n');
990    }
991    Ok((read_access_trace(&kept[..])?, capped))
992}

994/// Whether a record's last event is at or after `decision` (a call that
995/// never returned reaches every event).
996fn record_reaches(record: &TraceRecord, decision: EventIndex) -> bool {
997    let last = match record {
998        TraceRecord::Syscall(sys) => sys.post.unwrap_or(EventIndex::MAX),
999        TraceRecord::Signal(signal) => signal.event,
1000        TraceRecord::Edge(edge) => edge.to.event,
1001        TraceRecord::WatchHit(hit) => hit.after,
1002        TraceRecord::Header(_) | TraceRecord::End(_) => return false,
1003        TraceRecord::WatchSummary(_) => return false,
1004    };
1005    last >= decision
1006}
