33/// Sanity check that a series of preemptions don't include duplicates and are monotonically increasing.
34fn sanity_preempts(vec: &[(LogicalTime, Priority)]) -> anyhow::Result<()> {
35    let mut set = BTreeSet::new();
36
37    let mut time_last = None;
38    for (ns, prio) in vec {
39        if let Some(last) = time_last
40            && *ns <= last
41        {
42            bail!(
43                "Timestamps failed to monotonically increase ({}), in series:\n {:?}",
44                ns,
45                vec
46            );
47        }
48        assert!(*prio >= FIRST_PRIORITY);
49        assert!(*prio <= LAST_PRIORITY);
50        time_last = Some(*ns);
51        if !set.insert(ns) {
52            bail!(
53                "sanity_preempts: time series of preemptions contained duplicate entries sharing the timestamp {}",
54                ns
55            );
56        }
57    }
58    Ok(())
59}

61impl AnalyzeOpts {
62    /// Iteratively minimize the schedule needed to produce the error.
63    ///
64    /// # Returns
65    /// - Minimized preemption record (in memory),
66    /// - Path of a file containing that same minimized preemption record,
67    /// - Path of the log file that corresponds to the last matching (minimal) run.
68    pub(super) fn minimize(
69        &self,
70        preempts_path: &Path,
71        _global: &GlobalOpts,
72    ) -> anyhow::Result<(PreemptionRecord, PathBuf, PathBuf)> {
73        self.minimize_with(preempts_path, |runname, new_preempts_path| {
74            Ok(self
75                .launch_from_preempts_to_sched(runname, new_preempts_path, None)?
76                .0)
77        })
78    }
79
80    /// The minimization loop behind [`Self::minimize`]. `trial_matches` runs
81    /// one trial with the preemptions in the given file and says whether it
82    /// met the criteria. A trial that cannot finish (for example, one that
83    /// `--max-log-bytes` ended) ends the minimization with its error
84    /// unchanged, so main still reports it by its class.
85    fn minimize_with(
86        &self,
87        preempts_path: &Path,
88        mut trial_matches: impl FnMut(&str, &Path) -> anyhow::Result<bool>,
89    ) -> anyhow::Result<(PreemptionRecord, PathBuf, PathBuf)> {
90        let pr = PreemptionReader::new(preempts_path);
91        let mut tmp_dir = preempts_path.to_path_buf();
92        assert!(tmp_dir.pop());
93
94        let tids: Vec<DetTid> = pr.all_threads();
95        let init_schedule: PreemptionRecord = pr.load_all();
96        super::rundata::RunData::require_complete_record(&init_schedule, preempts_path)?;
97        // An unchanged endpoint is still a transformation input: an old
98        // Chosen record must not acquire provenance through minimization.
99        init_schedule.validate_combined_reserved_inputs(&init_schedule)?;
100        let mut round: u64 = 0;
101
102        // All the preemption points we don't know about yet.  Are they critical?
103        let mut remaining_unknown: BTreeMap<DetTid, _> = BTreeMap::new();
104        let mut init_intervention_count = 0;
105        for (dtid, history) in init_schedule.extract_all() {
106            let vec = history.as_vec();
107            sanity_preempts(&vec)?;
108            init_intervention_count += vec.len();
109            remaining_unknown.insert(dtid, vec);
110        }
111
112        eprintln!(
113            ":: {}",
114            format!(
115                "RootCause: starting with schedule of {} interventions for {} threads.",
116                init_intervention_count,
117                tids.len()
118            )
119            .yellow()
120            .bold()
121        );
122        if self.verbose {
123            eprintln!(
124                "Initial schedule:\n{}",
125                truncated(1000, format!("{}", init_schedule))
126            );
127        }
128
129        // The ones we know for sure are critical.
130        let mut critical_preempts: BTreeMap<DetTid, BTreeSet<(LogicalTime, Priority)>> =
131            BTreeMap::new();
132
133        // We take threads out of consideration if they have no interventions:
134        let all_threads = pr.all_threads();
135        let mut remaining_threads = Vec::new();
136
137        let min_seed = self.analyze_seed.unwrap_or_else(rand::random::<u64>);
138        eprintln!(
139            ":: {}",
140            format!("Minimization search using RNG seed {}", min_seed)
141                .yellow()
142                .bold()
143        );
144        let mut rng = Pcg64Mcg::seed_from_u64(min_seed);
145
146        // The number of preemptions to remove each round:
147        let mut batch_sizes = BTreeMap::new();
148        for tid in all_threads {
149            let len = remaining_unknown.get(&tid).unwrap().len();
150            batch_sizes.insert(tid, std::cmp::max(1, len / 4));
151            if len > 0 {
152                remaining_threads.push(tid)
153            }
154        }
155
156        let mut last_attempt: Option<PreemptionRecord> = None;
157        let mut last_matching_attempt = None;
158        let mut last_matching_pr_file = None;
159        let mut last_matching_log = None;
160
161        // Algorithm: the union of remaining_unknown plus critical_preempts represents our
162        // current point in the space of schedules to explore.
163        loop {
164            round += 1;
165            {
166                let sum: usize = remaining_unknown.values().map(|v| v.len()).sum();
167                let sum2: usize = critical_preempts.values().map(|s| s.len()).sum();
168                eprintln!(
169                    ":: {}",
170                    format!(
171                        "Minimizing schedule, round {}, {} interventions plus {} critical found",
172                        round, sum, sum2
173                    )
174                    .yellow()
175                    .bold(),
176                );
177                if self.verbose {
178                    eprintln!(":: All critical preemptions: {:?}", critical_preempts);
179                }
180                if remaining_threads.is_empty() {
181                    if sum2 == 0 {
182                        eprintln!(
183                            ":: {}",
184                            "Minimized to ZERO interventions.  The base run matches the critera."
185                                .yellow()
186                                .bold()
187                        );
188                    } else {
189                        eprintln!(
190                            ":: {}",
191                            format!("Minimized to {} interventions.", sum2)
192                                .green()
193                                .bold(),
194                        );
195                    }
196                    break;
197                }
198            }
199
200            let selected_ix = rng.random_range(0..remaining_threads.len());
201            let selected_tid = *remaining_threads.get(selected_ix).unwrap();
202            let mut cut = {
203                let batch = batch_sizes.get_mut(&selected_tid).unwrap();
204                assert!(*batch > 0);
205                let preempts = remaining_unknown.get_mut(&selected_tid).unwrap();
206                assert!(!preempts.is_empty());
207                *batch = std::cmp::min(*batch, preempts.len());
208
209                let len = preempts.len();
210                let cut = preempts.split_off(len - *batch);
211                eprintln!(
212                    ":: Shaving {} off of {} preemptions for tid {}: {}",
213                    batch,
214                    len,
215                    selected_tid,
216                    if self.verbose {
217                        truncated(79, format!("{:?}", cut))
218                    } else {
219                        "".to_string()
220                    }
221                );
222                if preempts.is_empty() {
223                    remaining_threads.swap_remove(selected_ix);
224                }
225                cut
226            };
227
228            {
229                let new_preempts_path =
230                    tmp_dir.join(format!("round_{:0wide$}.preempts", round, wide = 3));
231                let pr_new = {
232                    // Expensive... union back in the critical_preempts:
233                    let mut btmap = remaining_unknown.clone();
234                    btmap
235                        .iter_mut()
236                        .try_for_each(|(tid, vec)| -> anyhow::Result<()> {
237                            if let Some(set) = critical_preempts.get(tid) {
238                                for preempt in set {
239                                    vec.push(*preempt);
240                                }
241                                // if cfg!(debug)
242                                {
243                                    sanity_preempts(vec)?;
244                                }
245                            }
246                            Ok(())
247                        })?;
248                    self.stamp_trial_record(
249                        PreemptionRecord::from_vecs(&btmap)
250                            .with_guest_ids(init_schedule.guest_ids().unwrap_or_default())
251                            .with_reserved_input(init_schedule.reserved_input().clone()),
252                    )
253                };
254                if let Err(e) = pr_new.validate() {
255                    bail!(
256                        "Hermit analyzer produced corrupt preemption record, cannot proceed.\n\
257                        Error: {}\n\n\
258                        Corrupt record: {}\n\n\
259                        Last good state: {}",
260                        e,
261                        pr_new,
262                        if let Some(last) = last_attempt {
263                            last.to_string()
264                        } else {
265                            "<none>".to_string()
266                        }
267                    );
268                }
269                pr_new
270                    .write_to_disk(&new_preempts_path)
271                    .map_err(anyhow::Error::msg)?;
272                last_attempt = Some(pr_new);
273
274                let batch = batch_sizes.get_mut(&selected_tid).unwrap();
275                let runname = format!("round_{:0wide$}", round, wide = 3);
276                let log_path = tmp_dir.join(&runname).with_extension("log");
277                if trial_matches(&runname, &new_preempts_path)? {
278                    eprintln!(
279                        ":: {}",
280                        "New run matches criteria, continuing.".green().bold()
281                    );
282                    last_matching_attempt = last_attempt.clone();
283                    last_matching_pr_file = Some(new_preempts_path.clone());
284                    last_matching_log = Some(log_path);
285                    *batch += 1;
286                    continue;
287                } else {
288                    eprintln!(
289                        ":: {}",
290                        "New run fails criteria, backtracking..".red().bold()
291                    );
292                    if *batch == 1 {
293                        let critical = cut.pop().unwrap();
294                        assert!(cut.is_empty());
295                        eprintln!(
296                            ":: Knocked out only one preemption ({:?}), so concluding that one is critical.",
297                            critical
298                        );
299                        let entry = critical_preempts.entry(selected_tid).or_default();
300
301                        assert!(entry.insert(critical));
302                        continue;
303                    } else {
304                        *batch /= 2;
305                        eprintln!(
306                            ":: Batch size for minimizing tid {} reduced to {}",
307                            selected_tid, batch
308                        );
309                        // Hacky way to backtrack one step for now:
310                        {
311                            let preempts = remaining_unknown.get_mut(&selected_tid).unwrap();
312                            preempts.append(&mut cut);
313                        }
314                        continue;
315                    }
316                }
317            }
318        }
319        Ok((
320            last_matching_attempt.expect("at least one run to match the criteria"),
321            last_matching_pr_file.expect("at least one run to match the criteria"),
322            last_matching_log.expect("at least one run to match the criteria"),
323        ))
324    }
325}

334    /// A minimization trial that `--max-log-bytes` ended stops the
335    /// minimization and returns that error, still a `LogCapExceeded`, so main
336    /// reports it as the cap (123). Before, any trial's error panicked here.
337    #[test]
338    fn a_failing_minimize_trial_ends_minimization_with_its_error() {
339        let workspace = tempfile::tempdir().unwrap();
340        let preempts_path = workspace.path().join("target.preempts");
341        // One thread with eight interventions: increasing times, ordinary
342        // priorities. When every trial matches, minimizing it takes three
343        // trials (batches of 2, 3 and 3 interventions).
344        let history: Vec<(LogicalTime, Priority)> = (1..=8)
345            .map(|k| (LogicalTime::from_nanos(k * 1000), FIRST_PRIORITY + k))
346            .collect();
347        PreemptionRecord::from_vecs(&BTreeMap::from([(DetTid::from_raw(3), history)]))
348            .write_to_disk(&preempts_path)
349            .unwrap();
350        let options =
351            AnalyzeOpts::try_parse_from(["analyze", "--analyze-seed=7", "--", "/bin/true"])
352                .unwrap();
353
354        let mut total_trials = 0;
355        options
356            .minimize_with(&preempts_path, |_runname, _trial_preempts| {
357                total_trials += 1;
358                Ok(true)
359            })
360            .unwrap();
361        assert_eq!(total_trials, 3);
362
363        for failing_trial in 1..=total_trials {
364            let mut trials = 0;
365            let result = options.minimize_with(&preempts_path, |_runname, trial_preempts| {
366                trials += 1;
367                assert!(
368                    trial_preempts.exists(),
369                    "each trial's preemption file is written before it runs"
370                );
371                if trials == failing_trial {
372                    return Err(anyhow::Error::new(LogCapExceeded));
373                }
374                // An earlier trial finished and still met the criteria.
375                Ok(true)
376            });
377            let Err(error) = result else {
378                panic!("trial {failing_trial} failed, yet minimization returned a record")
379            };
380            assert!(
381                error.downcast_ref::<LogCapExceeded>().is_some(),
382                "trial {failing_trial}: the error lost its type: {error:#}"
383            );
384            assert_eq!(
385                trials, failing_trial,
386                "minimization ran more trials after trial {failing_trial} failed"
387            );
388        }
389    }

392#[cfg(test)]
393mod replay_departure_tests {
394    use clap::Parser;
395
396    use super::*;
397
398    #[test]
399    fn minimizing_a_marked_record_never_projects_it_into_unmarked_trials() {
400        let directory = tempfile::tempdir().unwrap();
401        let path = directory.path().join("history.json");
402        let history = (1..=8)
403            .map(|i| (LogicalTime::from_nanos(i * 1000), FIRST_PRIORITY + i))
404            .collect();
405        let record = PreemptionRecord::from_vecs(&BTreeMap::from([(DetTid::from_raw(3), history)]));
406        let options =
407            AnalyzeOpts::try_parse_from(["analyze", "--analyze-seed=7", "--", "/bin/true"])
408                .unwrap();
409        for marker in [None, Some(7)] {
410            record
411                .clone()
412                .retain_replay_departure(marker)
413                .write_to_disk(&path)
414                .unwrap();
415            let mut calls = 0;
416            let result = options.minimize_with(&path, |_, _| {
417                calls += 1;
418                Ok(true)
419            });
420            if marker.is_some() {
421                let error = result
422                    .expect_err("marked history must not be reconstructed without its marker");
423                assert!(
424                    error
425                        .downcast_ref::<crate::app::container::PolicyRefusal>()
426                        .is_some(),
427                    "{error:#}"
428                );
429                assert_eq!(calls, 0);
430            } else {
431                result.unwrap();
432                assert_eq!(calls, 3);
433            }
434        }
435    }
