9//! `--log-ring SIZE --log-ring-file FILE`: a flight recorder for Hermit's log.
10//!
11//! WHY. A full `--log=info` log of a real build step is large (817 MB for one
12//! 116-package dnf transaction) and nobody turns it on for every build, so
13//! when a build that ran without it turns out not to reproduce, the evidence
14//! is gone. The ring keeps only the newest SIZE bytes of the INFO log, so it
15//! can stay on in every build, and the newest records are what explains how a
16//! run ended.
17//!
18//! HOW. FILE is a fixed-size file mapped shared into Hermit: a 64-byte header
19//! and a SIZE-byte circular data area. Each tracing record is copied into the
20//! data area as `[u32 length][bytes]` and the header's positions are updated
21//! after it. Because the bytes are in the page cache as they are written, the
22//! file holds the newest records however Hermit ends: a normal exit, an error,
23//! a panic, `_exit` (which the container processes use), or SIGKILL. Nothing
24//! has to run at exit. A record being copied when the process died is not
25//! counted, because the header's end position moves only after the copy.
26//!
27//! READING. `hermit log-diff FILE` decodes a ring file and prints its records
28//! as a log; `hermit log-diff A B` compares two ring files, or a ring file and
29//! an ordinary log. The decoded log starts with a DEBUG record naming how many
30//! older records the ring had dropped.
31
32use std::collections::VecDeque;
33use std::fs::File;
34use std::io;
35use std::io::Write;
36use std::os::fd::AsRawFd;
37use std::sync::Arc;
38use std::sync::Mutex;
39use std::sync::atomic::AtomicU64;
40use std::sync::atomic::Ordering;

42/// The first bytes of a ring file.
43pub(crate) const MAGIC: &[u8; 8] = b"HRMRING1";
44/// Bytes before the data area.
45const HEADER: usize = 64;
46const CAPACITY_AT: usize = 8;
47/// Byte position, counted from the ring's creation, of the oldest record kept.
48const OLDEST_AT: usize = 16;
49/// Byte position just past the newest complete record.
50const END_AT: usize = 24;
51/// How many records were dropped to make room.
52const DROPPED_AT: usize = 32;
53/// Each record's length prefix.
54const LEN: u64 = 4;
55/// The smallest data area: room for a record of a useful size.
56pub(crate) const MIN_CAPACITY: u64 = 4096;
57
58/// The writer side: a mapped ring file and where its records start.
59pub(crate) struct LogRing {
60    base: *mut u8,
61    capacity: u64,
62    /// The start position of every record still in the data area, oldest first.
63    starts: VecDeque<u64>,
64    end: u64,
65    dropped: u64,
66}
67
68// SAFETY: the mapping is owned by this value and only reached through `&mut
69// self`; the `Mutex` in `RingWriter` serializes writers.
70unsafe impl Send for LogRing {}
71
72impl LogRing {
73    /// Size `file` to hold `capacity` bytes of records, map it, and write an
74    /// empty ring's header.
75    pub(crate) fn create(file: &File, capacity: u64) -> io::Result<Self> {
76        if capacity < MIN_CAPACITY {
77            return Err(io::Error::new(
78                io::ErrorKind::InvalidInput,
79                format!("--log-ring must be at least {MIN_CAPACITY} bytes"),
80            ));
81        }
82        let length = HEADER as u64 + capacity;
83        file.set_len(0)?;
84        file.set_len(length)?;
85        // SAFETY: a fresh shared mapping of a file descriptor we own, of the
86        // length just set; it is unmapped only in `Drop`.
87        let base = unsafe {
88            libc::mmap(
89                std::ptr::null_mut(),
90                length as usize,
91                libc::PROT_READ | libc::PROT_WRITE,
92                libc::MAP_SHARED,
93                file.as_raw_fd(),
94                0,
95            )
96        };
97        if base == libc::MAP_FAILED {
98            return Err(io::Error::last_os_error());
99        }
100        let ring = Self {
101            base: base.cast(),
102            capacity,
103            starts: VecDeque::new(),
104            end: 0,
105            dropped: 0,
106        };
107        // SAFETY: the header lies inside the mapping.
108        unsafe { std::ptr::copy_nonoverlapping(MAGIC.as_ptr(), ring.base, MAGIC.len()) };
109        ring.word(CAPACITY_AT).store(capacity, Ordering::Release);
110        Ok(ring)
111    }
112
113    fn word(&self, at: usize) -> &AtomicU64 {
114        // SAFETY: `at` is one of the header's 8-byte aligned slots; the mapping
115        // is page aligned and outlives `self`.
116        unsafe { &*(self.base.add(at).cast::<AtomicU64>()) }
117    }
118
119    fn oldest(&self) -> u64 {
120        self.starts.front().copied().unwrap_or(self.end)
121    }
122
123    /// Copy `bytes` into the data area at byte position `at`, wrapping.
124    fn put(&mut self, at: u64, bytes: &[u8]) {
125        let mut offset = (at % self.capacity) as usize;
126        let mut rest = bytes;
127        while !rest.is_empty() {
128            let room = self.capacity as usize - offset;
129            let (now, later) = rest.split_at(room.min(rest.len()));
130            // SAFETY: `offset + now.len() <= capacity`, inside the data area.
131            unsafe {
132                std::ptr::copy_nonoverlapping(
133                    now.as_ptr(),
134                    self.base.add(HEADER + offset),
135                    now.len(),
136                )
137            };
138            rest = later;
139            offset = 0;
140        }
141    }
142
143    /// Append one record, dropping the oldest records it needs room from. A
144    /// record longer than the data area keeps its last bytes.
145    pub(crate) fn push(&mut self, record: &[u8]) {
146        let most = (self.capacity - LEN) as usize;
147        let record = &record[record.len().saturating_sub(most)..];
148        let need = LEN + record.len() as u64;
149        while self.end + need - self.oldest() > self.capacity {
150            self.starts.pop_front();
151            self.dropped += 1;
152        }
153        // Publish the new oldest record BEFORE overwriting the dropped ones, so
154        // a reader of a process killed mid-copy never decodes overwritten bytes.
155        self.word(OLDEST_AT).store(self.oldest(), Ordering::Release);
156        self.word(DROPPED_AT).store(self.dropped, Ordering::Release);
157        let start = self.end;
158        self.put(start, &(record.len() as u32).to_le_bytes());
159        self.put(start + LEN, record);
160        self.starts.push_back(start);
161        self.end = start + need;
162        if self.starts.len() == 1 {
163            self.word(OLDEST_AT).store(start, Ordering::Release);
164        }
165        self.word(END_AT).store(self.end, Ordering::Release);
166    }
167}
168
169impl Drop for LogRing {
170    fn drop(&mut self) {
171        // SAFETY: the mapping created in `create`, of the same length.
172        unsafe { libc::munmap(self.base.cast(), HEADER + self.capacity as usize) };
173    }
174}

176/// A tracing writer that appends each write, one formatted record, to a ring.
177#[derive(Clone)]
178pub(crate) struct RingWriter(Arc<Mutex<LogRing>>);
179
180impl RingWriter {
181    pub(crate) fn new(ring: LogRing) -> Self {
182        Self(Arc::new(Mutex::new(ring)))
183    }
184}
185
186impl Write for RingWriter {
187    fn write(&mut self, buf: &[u8]) -> io::Result<usize> {
188        self.0
189            .lock()
190            .unwrap_or_else(|poisoned| poisoned.into_inner())
191            .push(buf);
192        Ok(buf.len())
193    }
194
195    fn flush(&mut self) -> io::Result<()> {
196        Ok(())
197    }
198}

200/// The records of a ring file, oldest first, after a DEBUG record that says
201/// how many older ones were dropped; `None` when `bytes` is not a ring file.
202pub(crate) fn decode(bytes: &[u8]) -> Option<io::Result<Vec<u8>>> {
203    if !bytes.starts_with(MAGIC) {
204        return None;
205    }
206    Some(decode_ring(bytes))
207}
208
209fn decode_ring(bytes: &[u8]) -> io::Result<Vec<u8>> {
210    let bad = |what: &str| io::Error::new(io::ErrorKind::InvalidData, format!("ring file: {what}"));
211    if bytes.len() < HEADER {
212        return Err(bad("shorter than its header"));
213    }
214    let word = |at: usize| u64::from_le_bytes(bytes[at..at + 8].try_into().unwrap());
215    let (capacity, oldest, end, dropped) = (
216        word(CAPACITY_AT),
217        word(OLDEST_AT),
218        word(END_AT),
219        word(DROPPED_AT),
220    );
221    if capacity < MIN_CAPACITY || bytes.len() as u64 != HEADER as u64 + capacity {
222        return Err(bad("its size does not match its capacity"));
223    }
224    if oldest > end || end - oldest > capacity {
225        return Err(bad("its positions are inconsistent"));
226    }
227    let data = &bytes[HEADER..];
228    let get = |at: u64, len: u64| -> Vec<u8> {
229        (0..len)
230            .map(|i| data[((at + i) % capacity) as usize])
231            .collect()
232    };
233    let mut records = Vec::new();
234    let mut count = 0u64;
235    let mut at = oldest;
236    while at < end {
237        let len = u32::from_le_bytes(get(at, LEN).try_into().unwrap()) as u64;
238        if at + LEN + len > end {
239            return Err(bad("a record runs past the end"));
240        }
241        records.extend(get(at + LEN, len));
242        at += LEN + len;
243        count += 1;
244    }
245    let mut out = format!(
246        "1970-01-01T00:00:00.000000Z DEBUG hermit::log_ring: --log-ring file: {count} records, {} bytes of {capacity}; {dropped} older records dropped\n",
247        end - oldest
248    )
249    .into_bytes();
250    out.extend(records);
251    Ok(out)
252}

254#[cfg(test)]
255mod tests {
256    use super::*;
257
258    fn ring(capacity: u64) -> (tempfile::NamedTempFile, LogRing) {
259        let file = tempfile::NamedTempFile::new().unwrap();
260        let ring = LogRing::create(file.as_file(), capacity).unwrap();
261        (file, ring)
262    }
263
264    fn records(decoded: &[u8]) -> Vec<String> {
265        String::from_utf8(decoded.to_vec())
266            .unwrap()
267            .lines()
268            .skip(1)
269            .map(str::to_string)
270            .collect()
271    }
272
273    /// The file holds the newest records, whole and in order, without any
274    /// call at exit: the decoder reads what the mapping wrote.
275    #[test]
276    fn the_file_keeps_the_newest_whole_records_in_order() {
277        let (file, mut ring) = ring(MIN_CAPACITY);
278        let mut written = Vec::new();
279        for i in 0..2000 {
280            let record = format!("record {i:05} {}\n", "x".repeat(i % 37));
281            ring.push(record.as_bytes());
282            written.push(record.trim_end_matches('\n').to_string());
283        }
284        let decoded = decode(&std::fs::read(file.path()).unwrap())
285            .unwrap()
286            .unwrap();
287        let kept = records(&decoded);
288        assert!(kept.len() > 10 && kept.len() < 2000, "{}", kept.len());
289        assert_eq!(kept, written[written.len() - kept.len()..]);
290        let header = String::from_utf8(decoded).unwrap();
291        let header = header.lines().next().unwrap();
292        assert!(
293            header.contains(&format!("{} older records dropped", 2000 - kept.len())),
294            "{header}"
295        );
296        let used: u64 = kept.iter().map(|r| LEN + r.len() as u64 + 1).sum();
297        assert!(used <= MIN_CAPACITY, "{used}");
298    }
299
300    /// A record longer than the data area keeps its last bytes, and a file
301    /// that is not a ring is left to the ordinary reader.
302    #[test]
303    fn oversized_records_keep_their_tail_and_other_files_are_not_rings() {
304        let (file, mut ring) = ring(MIN_CAPACITY);
305        let long = format!("{}END\n", "y".repeat(10_000));
306        ring.push(long.as_bytes());
307        let decoded = decode(&std::fs::read(file.path()).unwrap())
308            .unwrap()
309            .unwrap();
310        let kept = records(&decoded);
311        assert_eq!(kept.len(), 1);
312        assert!(kept[0].ends_with("END"));
313        assert_eq!(kept[0].len() as u64, MIN_CAPACITY - LEN - 1);
314        assert!(decode(b"2026-10-22T00:00:00Z  INFO detcore: x\n").is_none());
315        assert!(LogRing::create(file.as_file(), MIN_CAPACITY - 1).is_err());
316    }
