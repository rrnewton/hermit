40/// The `hermit-dap` beside this `hermit` executable.
41pub fn hermit_dap_beside_hermit() -> Result<PathBuf, Error> {
42    let adapter = std::env::current_exe()
43        .context("Failed to locate hermit")?
44        .with_file_name("hermit-dap");
45    if !adapter.is_file() {
46        anyhow::bail!(
47            "--dap-listen needs hermit-dap beside hermit, and {} does not exist; \
48             build it with `cargo build -p hermit-run --bin hermit --bin hermit-dap`",
49            adapter.display()
50        );
51    }
52    Ok(adapter)
53}

63/// Binds `--dap-listen`'s `[HOST:]PORT` (`PORT` alone means loopback), so an
64/// address that does not parse or cannot be listened on is refused here,
65/// before the run starts, and the adapter is handed a socket already
66/// listening.
67pub fn bind(argument: &str) -> Result<TcpListener, Error> {
68    let (host, port) = argument.rsplit_once(':').unwrap_or(("127.0.0.1", argument));
69    let port: u16 = port.parse().map_err(|_| {
70        anyhow::anyhow!("--dap-listen {argument:?} needs a port from 0 to 65535, as [HOST:]PORT")
71    })?;
72    anyhow::ensure!(
73        !host.is_empty(),
74        "--dap-listen {argument:?} needs a host before ':', or just a port"
75    );
76    let address = format!("{host}:{port}");
77    TcpListener::bind(&address).map_err(|error| {
78        anyhow::anyhow!(
79            "--dap-listen cannot listen on {address}: {error}; pass another port, or port 0 \
80             to let the system pick a free one"
81        )
82    })
83}

85/// `command` started so that it dies with this process: SIGKILL when this
86/// process exits, armed before exec, and refused if this process had already
87/// died before it was armed.
88pub fn spawn_owned(command: &mut Command) -> std::io::Result<Child> {
89    // SAFETY: getpid has no arguments; read before fork, compared after.
90    let parent = unsafe { libc::getpid() };
91    // SAFETY: only async-signal-safe calls (prctl, getppid) between fork and
92    // exec; they touch no memory of the parent.
93    unsafe {
94        command.pre_exec(move || {
95            if libc::prctl(libc::PR_SET_PDEATHSIG, libc::SIGKILL) == -1 {
96                return Err(std::io::Error::last_os_error());
97            }
98            // A parent that died before the line above never sends the
99            // signal: the child was already reparented. Notice that here.
100            if libc::getppid() != parent {
101                return Err(std::io::Error::from_raw_os_error(libc::ESRCH));
102            }
103            Ok(())
104        });
105    }
106    command.spawn()
107}

109/// Whether anything in this network namespace listens on TCP `port`, read
110/// from /proc so as not to use up a gdbserver's one connection.
111fn tcp_port_is_listening(port: u16) -> bool {
112    let wanted = format!(":{port:04X}");
113    ["/proc/net/tcp", "/proc/net/tcp6"].iter().any(|table| {
114        std::fs::read_to_string(table).is_ok_and(|text| {
115            text.lines().skip(1).any(|row| {
116                let fields: Vec<&str> = row.split_whitespace().collect();
117                fields.len() > 3 && fields[1].ends_with(&wanted) && fields[3] == "0A"
118            })
119        })
120    })
121}

130/// The process of `run` that its gdbserver serves: its one traced
131/// descendant, which waits, stopped, for the debugger before it executes the
132/// guest program, already inside the guest's namespaces.
133fn traced_descendant(run: u32) -> Option<i32> {
134    // Each process's parent, from /proc/PID/stat ("PID (COMM) STATE PPID ...").
135    let parents: Vec<(i32, i32)> = std::fs::read_dir("/proc")
136        .ok()?
137        .filter_map(|entry| {
138            let pid: i32 = entry.ok()?.file_name().to_str()?.parse().ok()?;
139            let stat = std::fs::read_to_string(format!("/proc/{pid}/stat")).ok()?;
140            let (_, rest) = stat.rsplit_once(") ")?;
141            Some((pid, rest.split_whitespace().nth(1)?.parse().ok()?))
142        })
143        .collect();
144    let mut family = vec![run as i32];
145    let mut index = 0;
146    while index < family.len() {
147        let parent = family[index];
148        family.extend(
149            parents
150                .iter()
151                .filter(|(_, ppid)| *ppid == parent)
152                .map(|(pid, _)| *pid),
153        );
154        index += 1;
155    }
156    let mut traced = family.into_iter().skip(1).filter(|pid| {
157        std::fs::read_to_string(format!("/proc/{pid}/status")).is_ok_and(|status| {
158            status
159                .lines()
160                .find_map(|line| line.strip_prefix("TracerPid:"))
161                .is_some_and(|tracer| tracer.trim() != "0")
162        })
163    });
164    let first = traced.next()?;
165    traced.next().is_none().then_some(first)
166}

168/// `openat2(dir, path, flags, RESOLVE_IN_ROOT)`: `path` resolved as if `dir`
169/// were the root, so absolute symbolic links and `..` stay inside it.
170fn open_in_root(dir: &OwnedFd, path: &Path, flags: i32) -> std::io::Result<OwnedFd> {
171    let relative = path.strip_prefix("/").unwrap_or(path);
172    let relative = if relative.as_os_str().is_empty() {
173        Path::new(".")
174    } else {
175        relative
176    };
177    let path = CString::new(relative.as_os_str().as_bytes())
178        .map_err(|_| std::io::Error::from_raw_os_error(libc::EINVAL))?;
179    // SAFETY: `open_how` holds only integers, and zero is the value the
180    // kernel requires of every field this call does not set.
181    let mut how: libc::open_how = unsafe { std::mem::zeroed() };
182    how.flags = (flags | libc::O_CLOEXEC) as u64;
183    how.resolve = libc::RESOLVE_IN_ROOT | libc::RESOLVE_NO_MAGICLINKS;
184    // SAFETY: `path` is NUL-terminated, and `how` is an `open_how` of the
185    // size passed. glibc has no wrapper for openat2.
186    let fd = unsafe {
187        libc::syscall(
188            libc::SYS_openat2,
189            dir.as_raw_fd(),
190            path.as_ptr(),
191            &how as *const libc::open_how,
192            std::mem::size_of::<libc::open_how>(),
193        )
194    };
195    if fd < 0 {
196        return Err(std::io::Error::last_os_error());
197    }
198    // SAFETY: openat2 returned a new descriptor that nothing else owns.
199    Ok(unsafe { OwnedFd::from_raw_fd(fd as i32) })
200}

202/// `path` with every symbolic link resolved as the guest whose root is `root`
203/// sees it, the way exec resolves it: absolute links lead from that root.
204fn canonical_in_root(root: &OwnedFd, path: &Path) -> std::io::Result<PathBuf> {
205    let mut pending: std::collections::VecDeque<std::ffi::OsString> = path
206        .components()
207        .map(|component| component.as_os_str().to_owned())
208        .collect();
209    let mut resolved = PathBuf::from("/");
210    let mut links = 0;
211    while let Some(component) = pending.pop_front() {
212        match component.as_bytes() {
213            b"/" | b"." => continue,
214            b".." => {
215                resolved.pop();
216                continue;
217            }
218            _ => {}
219        }
220        let candidate = resolved.join(&component);
221        let fd = open_in_root(root, &candidate, libc::O_PATH | libc::O_NOFOLLOW)?;
222        let metadata = std::fs::File::from(fd.try_clone()?).metadata()?;
223        if !metadata.file_type().is_symlink() {
224            resolved = candidate;
225            continue;
226        }
227        links += 1;
228        if links > 40 {
229            return Err(std::io::Error::from_raw_os_error(libc::ELOOP));
230        }
231        let mut target = vec![0u8; libc::PATH_MAX as usize];
232        // SAFETY: an empty path reads the link `fd` itself names, which an
233        // O_PATH | O_NOFOLLOW descriptor allows, into `target`'s bytes.
234        let length = unsafe {
235            libc::readlinkat(
236                fd.as_raw_fd(),
237                c"".as_ptr(),
238                target.as_mut_ptr().cast(),
239                target.len(),
240            )
241        };
242        if length < 0 {
243            return Err(std::io::Error::last_os_error());
244        }
245        target.truncate(length as usize);
246        let target = PathBuf::from(std::ffi::OsString::from_vec(target));
247        if target.is_absolute() {
248            resolved = PathBuf::from("/");
249        }
250        for component in target.components().rev() {
251            pending.push_front(component.as_os_str().to_owned());
252        }
253    }
254    Ok(resolved)
255}

257/// Where the guest finds its program: a path it names (`Named`), or the
258/// candidates of a `PATH` search, in order (`Search`), of which the guest runs
259/// the first that is a regular file with an execute bit.
260#[derive(Debug)]
261pub enum GuestProgram {
262    Named(PathBuf),
263    Search(Vec<PathBuf>),
264}
265
266/// Whether `path`, followed as the guest sees it, is a regular file with an
267/// execute bit: the test the run's own `PATH` search applies.
268fn executable_in_root(root: &OwnedFd, path: &Path) -> bool {
269    canonical_in_root(root, path)
270        .and_then(|canonical| open_in_root(root, &canonical, libc::O_PATH))
271        .and_then(|fd| std::fs::File::from(fd).metadata())
272        .is_ok_and(|metadata| metadata.is_file() && metadata.mode() & 0o111 != 0)
273}

275/// The host file a debugger must load for the program the guest of process
276/// `guest_pid` is about to execute, `guest_program`, with `to_host` mapping a
277/// guest path without symbolic links to the host file the run's mounts make
278/// it. Both steps happen in the guest's own view, through its root: the
279/// `PATH` search, so a mount that makes an earlier candidate executable
280/// selects it there as it does for the guest, and the symbolic links of the
281/// selected path. The answer is then checked: the file the guest opens there
282/// must be that same host file (device and inode), readable, and an ELF
283/// executable, since a debugger can load nothing else. Anything else is
284/// refused with the reason, before any hint is given.
285pub fn identify_debuggee(
286    guest_pid: i32,
287    guest_program: &GuestProgram,
288    to_host: impl Fn(&Path) -> Option<PathBuf>,
289) -> Result<PathBuf, Error> {
290    let root = std::fs::File::open(format!("/proc/{guest_pid}/root"))
291        .map(OwnedFd::from)
292        .with_context(|| format!("Failed to open the root of the guest's process {guest_pid}"))?;
293    let guest_program = match guest_program {
294        GuestProgram::Named(path) => path.as_path(),
295        GuestProgram::Search(candidates) => candidates
296            .iter()
297            .find(|candidate| executable_in_root(&root, candidate))
298            .ok_or_else(|| {
299                anyhow::anyhow!(
300                    "--dap-listen finds no executable among the guest's PATH candidates {:?}",
301                    candidates
302                )
303            })?,
304    };
305    let canonical = canonical_in_root(&root, guest_program).with_context(|| {
306        format!(
307            "--dap-listen cannot follow {} through the guest's file system",
308            guest_program.display()
309        )
310    })?;
311    let host = to_host(&canonical).ok_or_else(|| {
312        anyhow::anyhow!(
313            "--dap-listen cannot tell the debugger which host file is the guest's {} ({}): \
314             no mount of this run maps it to one",
315            guest_program.display(),
316            canonical.display()
317        )
318    })?;
319    let guest_file = match open_in_root(&root, &canonical, libc::O_RDONLY) {
320        Ok(fd) => std::fs::File::from(fd),
321        Err(error) if error.raw_os_error() == Some(libc::EACCES) => anyhow::bail!(
322            "--dap-listen cannot read {}, so a debugger could not load it either; \
323             make it readable",
324            guest_program.display()
325        ),
326        Err(error) => {
327            return Err(error).with_context(|| {
328                format!(
329                    "--dap-listen cannot open the guest's {}",
330                    guest_program.display()
331                )
332            });
333        }
334    };
335    let mut magic = [0u8; 4];
336    let read = std::io::Read::read(&mut (&guest_file), &mut magic).unwrap_or(0);
337    if magic.starts_with(b"#!") {
338        anyhow::bail!(
339            "--dap-listen debugs an executable, and the guest's {} is a #! script; to debug \
340             its interpreter, run the interpreter with the script as its argument, for \
341             example hermit run --dap-listen=ADDRESS -- /bin/sh {}",
342            guest_program.display(),
343            guest_program.display()
344        );
345    }
346    anyhow::ensure!(
347        read == magic.len() && &magic == b"\x7fELF",
348        "--dap-listen debugs an ELF executable, and the guest's {} is not one",
349        guest_program.display()
350    );
351    let guest_metadata = guest_file
352        .metadata()
353        .context("Failed to stat the guest's program")?;
354    let host_metadata =
355        std::fs::metadata(&host).with_context(|| format!("Failed to stat {}", host.display()))?;
356    anyhow::ensure!(
357        (guest_metadata.dev(), guest_metadata.ino()) == (host_metadata.dev(), host_metadata.ino()),
358        "--dap-listen cannot tell the debugger which host file the guest runs: the guest's {} \
359         ({}) is not the host's {}",
360        guest_program.display(),
361        canonical.display(),
362        host.display()
363    );
364    Ok(host)
365}

367/// Supervises `run`, a `hermit run --gdbserver` on `gdbserver_port`. Once its
368/// gdbserver listens and its guest process is waiting, names the host file the
369/// debugger must load with `identify` (given that process), then starts
370/// `hermit-dap` on `listener` and prints the attach hint. Returns the run's
371/// status when it ends. If the adapter ends, for any reason, while the
372/// gdbserver still waits for its one connection, no debugger can ever attach,
373/// so this ends the run and fails; so does a refusal of `identify`, or a
374/// missing `hermit-dap`. An adapter whose debugger connected and later left is
375/// <https://github.com/rrnewton/hermit/issues/4313>.
376pub fn supervise(
377    mut run: Child,
378    listener: TcpListener,
379    gdbserver_port: u16,
380    identify: impl Fn(i32) -> Result<PathBuf, Error>,
381    gdb: Option<&Path>,
382) -> Result<reverie::process::ExitStatus, Error> {
383    let mut listener = Some(listener);
384    let mut adapter: Option<ListeningAdapter> = None;
385    let mut connected = false;
386    loop {
387        if let Some(status) = run.try_wait().context("Failed to poll the run")? {
388            // Gives a connected client its grace period to finish.
389            drop(adapter);
390            return Ok(status.into());
391        }
392        let listening = tcp_port_is_listening(gdbserver_port);
393        match adapter.as_mut() {
394            None if listening => {
395                let Some(guest) = traced_descendant(run.id()) else {
396                    std::thread::sleep(POLL_INTERVAL);
397                    continue;
398                };
399                let started = hermit_dap_beside_hermit().and_then(|path| {
400                    let program = identify(guest)?;
401                    // Said in one write before hermit-dap shares this stderr,
402                    // so its own lines never land in the middle of this one.
403                    let line = format!(
404                        "hermit run: starting hermit-dap to serve this run to one DAP client; {}\n",
405                        attach_hint(&program, gdbserver_port)
406                    );
407                    let _ = std::io::Write::write_all(&mut std::io::stderr(), line.as_bytes());
408                    let listener = listener
409                        .take()
410                        .context("the DAP listener was already used")?;
411                    ListeningAdapter::spawn(&path, listener, gdb)
412                });
413                match started {
414                    Ok(started) => adapter = Some(started),
415                    Err(error) => {
416                        end_run(&mut run);
417                        return Err(error);
418                    }
419                }
420            }
421            Some(started) if !connected => {
422                if !listening {
423                    // The gdbserver took its one connection.
424                    connected = true;
425                } else if let Some(status) = started
426                    .child
427                    .try_wait()
428                    .context("Failed to poll hermit-dap")?
429                {
430                    end_run(&mut run);
431                    anyhow::bail!(
432                        "hermit-dap ended ({status}) before a debugger connected to this run's \
433                         gdbserver on port {gdbserver_port}, so none can attach; ended the run \
434                         rather than leave it waiting"
435                    );
436                }
437            }
438            _ => {}
439        }
440        std::thread::sleep(POLL_INTERVAL);
441    }
442}

444/// Ends `run` with SIGTERM, then SIGKILL after [`RUN_STOP_GRACE`], and reaps it.
445fn end_run(run: &mut Child) {
446    // SAFETY: kill has no pointer arguments.
447    unsafe {
448        libc::kill(run.id() as libc::pid_t, libc::SIGTERM);
449    }
450    let deadline = Instant::now() + RUN_STOP_GRACE;
451    while Instant::now() < deadline {
452        if let Ok(Some(_)) = run.try_wait() {
453            return;
454        }
455        std::thread::sleep(POLL_INTERVAL);
456    }
457    let _ = run.kill();
458    let _ = run.wait();
459}

461/// A `hermit-dap --listen-fd` serving a `hermit run --gdbserver`. It runs
462/// outside the guest's namespaces, as the GDB client of a plain
463/// `--gdbserver` run does, and is stopped when this value is dropped: at once
464/// if no client ever connected, or after [`ADAPTER_EXIT_GRACE`] if its
465/// session has not ended by then. It also dies with this process.
466#[derive(Debug)]
467pub struct ListeningAdapter {
468    child: Child,
469}
470
471impl ListeningAdapter {
472    /// Starts `adapter` on `listener`, which it inherits as [`LISTEN_FD`],
473    /// running `gdb` if given.
474    pub fn spawn(adapter: &Path, listener: TcpListener, gdb: Option<&Path>) -> Result<Self, Error> {
475        let listener_fd = listener.as_raw_fd();
476        let mut command = Command::new(adapter);
477        command
478            .arg("--listen-fd")
479            .arg(LISTEN_FD.to_string())
480            .stdin(Stdio::null())
481            .stdout(Stdio::null());
482        if let Some(gdb) = gdb {
483            command.arg("--gdb").arg(gdb);
484        }
485        // SAFETY: only async-signal-safe calls (fcntl, dup2) between fork and
486        // exec; they touch no memory of the parent.
487        unsafe {
488            command.pre_exec(move || {
489                // dup2 clears close-on-exec on the copy; a socket already at
490                // LISTEN_FD needs it cleared by hand.
491                if listener_fd == LISTEN_FD {
492                    if libc::fcntl(LISTEN_FD, libc::F_SETFD, 0) == -1 {
493                        return Err(std::io::Error::last_os_error());
494                    }
495                } else if libc::dup2(listener_fd, LISTEN_FD) == -1 {
496                    return Err(std::io::Error::last_os_error());
497                }
498                Ok(())
499            });
500        }
501        let child = spawn_owned(&mut command)
502            .with_context(|| format!("Failed to start {}", adapter.display()))?;
503        // The adapter owns the socket now; this process's copy closes here.
504        drop(listener);
505        Ok(Self { child })
506    }
507
508    /// Whether the adapter is still waiting for its client. `hermit-dap`
509    /// becomes GDB, by `exec`, once a client connects.
510    fn still_waiting_for_a_client(&self) -> bool {
511        std::fs::read_to_string(format!("/proc/{}/comm", self.child.id()))
512            .is_ok_and(|comm| comm.trim_end() == "hermit-dap")
513    }
514}
515
516impl Drop for ListeningAdapter {
517    fn drop(&mut self) {
518        let grace = if self.still_waiting_for_a_client() {
519            Duration::ZERO
520        } else {
521            ADAPTER_EXIT_GRACE
522        };
523        let deadline = Instant::now() + grace;
524        while Instant::now() < deadline {
525            match self.child.try_wait() {
526                Ok(Some(_)) | Err(_) => return,
527                Ok(None) => std::thread::sleep(Duration::from_millis(50)),
528            }
529        }
530        let _ = self.child.kill();
531        let _ = self.child.wait();
532    }
533}
