19pub(super) const RUN_TIMEOUT_UNWIND_GRACE: Duration = Duration::from_secs(10);
20
21/// Test-only: hold the post-expiry path open past the grace so the SIGALRM
22/// fallback is the thing that ends the run.
23///
24/// ⚠️ THIS EXISTS BECAUSE THE FALLBACK COULD NOT BE MADE TO FIRE ANY OTHER WAY,
25/// AND AN UNEXERCISED SAFETY PATH IS THE FAILURE MODE THIS PROJECT KEEPS
26/// FINDING. Measured 2026-08-26 at this commit: the primary path fired at
27/// exactly the bound for a userspace spinner, a guest blocked reading a pipe
28/// with no writer, a guest that `SIGSTOP`s itself, an eight-thread guest
29/// ignoring `SIGTERM`, and a multi-process guest ignoring `SIGTERM` -- five
30/// shapes, five clean unwinds, no wedge. That is a good result for the primary
31/// path and it leaves the fallback with zero executions, which is exactly the
32/// mechanism-that-has-never-run shape.
33///
34/// ⚠️ WHAT THIS DOES AND DOES NOT REPRODUCE, stated precisely rather than
35/// implied. It reproduces the CONDITION the fallback is specified against --
36/// the post-expiry path not completing within `RUN_TIMEOUT_UNWIND_GRACE` -- and
37/// it exercises the real alarm, the real inherited-mask handling, the real
38/// handler, the real message and the real `_exit`. It does NOT reproduce any
39/// particular upstream CAUSE of a slow unwind, because none is known; the delay
40/// is here, after the drop, rather than inside a wedged destructor. A future
41/// reader must not read a passing fallback test as evidence that some specific
42/// teardown hang is handled.
43///
44/// Deliberately keyed off an internal environment variable rather than a
45/// `cfg(test)` gate: the fallback lives in the shipped binary and must be
46/// exercised there, not in a differently-compiled one.
47pub(super) fn stall_the_unwind_if_asked() {
48    const STALL_ENV: &str = "HERMIT_INTERNAL_RUN_TIMEOUT_STALL_UNWIND";
49    if std::env::var_os(STALL_ENV).as_deref() != Some(std::ffi::OsStr::new("1")) {
50        return;
51    }
52    // Comfortably past the grace, so the alarm -- not this sleep -- ends the
53    // process. If the fallback is broken this returns and the caller sees an
54    // ordinary timeout, which is what makes the test able to fail.
55    std::thread::sleep(RUN_TIMEOUT_UNWIND_GRACE + Duration::from_secs(5));
56}
57
58static RUN_TIMEOUT_MESSAGE: AtomicPtr<u8> = AtomicPtr::new(std::ptr::null_mut());
59static RUN_TIMEOUT_MESSAGE_LEN: AtomicUsize = AtomicUsize::new(0);
60
61/// The hard fallback: fires only if the unwind above did not finish in time.
62///
63/// Identical in construction to `record_start.rs`'s `recording_timeout_handler`
64/// -- non-blocking stderr so a full pipe cannot wedge the handler, then
65/// `_exit` -- and identical in exit code, because "a deadline fired" is one
66/// meaning and 124 already carries it for GNU `timeout`, for `safehermit`'s wall
67/// bound, and for `hermit record`'s own deadline. Reusing it here adds no new
68/// collision; inventing a fourth number for the same event would.
69extern "C" fn run_timeout_fallback_handler(_signal: libc::c_int) {
70    let len = RUN_TIMEOUT_MESSAGE_LEN.load(Ordering::Acquire);
71    let message = RUN_TIMEOUT_MESSAGE.load(Ordering::Acquire);
72    if !message.is_null() && len != 0 {
73        // SAFETY: the message is leaked before the timer is armed, and
74        // fcntl(2), write(2) and _exit(2) are async-signal-safe.
75        unsafe {
76            let flags = libc::fcntl(libc::STDERR_FILENO, libc::F_GETFL);
77            if flags != -1 {
78                libc::fcntl(libc::STDERR_FILENO, libc::F_SETFL, flags | libc::O_NONBLOCK);
79            }
80            libc::write(libc::STDERR_FILENO, message.cast(), len);
81        }
82    }
83    // Exiting the namespace init tears down the container and its tracees.
84    // SAFETY: _exit(2) is async-signal-safe and runs no Rust destructors --
85    // which is precisely why this is the fallback and not the primary path.
86    unsafe { libc::_exit(HERMIT_DEADLINE_EXIT) }
87}

89pub(super) struct RunTimeoutFallback {
90    previous_handler: SigAction,
91    reblock_sigalrm: bool,
92}
93
94impl RunTimeoutFallback {
95    pub(super) fn arm(after: Duration) -> Result<Self, Error> {
96        let seconds: libc::c_uint = after
97            .as_secs()
98            .try_into()
99            .map_err(|_| Error::msg("--timeout exceeds the platform alarm limit"))?;
100        let message = Box::leak(
101            format!(
102                "HERMIT_RUN_TIMEOUT_FALLBACK: the --timeout unwind did not complete within {} seconds; \
103                 the container was terminated without a clean teardown\n",
104                RUN_TIMEOUT_UNWIND_GRACE.as_secs()
105            )
106            .into_boxed_str(),
107        );
108        RUN_TIMEOUT_MESSAGE.store(message.as_mut_ptr(), Ordering::Release);
109        RUN_TIMEOUT_MESSAGE_LEN.store(message.len(), Ordering::Release);
110
111        let action = SigAction::new(
112            SigHandler::Handler(run_timeout_fallback_handler),
113            SaFlags::SA_RESETHAND,
114            SigSet::empty(),
115        );
116        // SAFETY: the handler uses only async-signal-safe operations and stays
117        // installed until this guard disarms it.
118        let previous_handler = unsafe { sigaction(Signal::SIGALRM, &action) }?;
119
120        // A blocked SIGALRM stays pending forever and the handler never runs,
121        // silently disabling the fallback. `record_start.rs` learned this too.
122        let mut alarm = SigSet::empty();
123        alarm.add(Signal::SIGALRM);
124        let reblock_sigalrm = SigSet::thread_get_mask()
125            .map(|mask| mask.contains(Signal::SIGALRM))
126            .unwrap_or(false);
127        if reblock_sigalrm {
128            let _ = alarm.thread_unblock();
129        }
130
131        // SAFETY: `seconds` fits c_uint.
132        unsafe { libc::alarm(seconds) };
133        Ok(Self {
134            previous_handler,
135            reblock_sigalrm,
136        })
137    }
138}
139
140impl Drop for RunTimeoutFallback {
141    fn drop(&mut self) {
142        // SAFETY: disarm the alarm before restoring the inherited handler.
143        unsafe {
144            libc::alarm(0);
145            let _ = sigaction(Signal::SIGALRM, &self.previous_handler);
146        }
147        if self.reblock_sigalrm {
148            let mut alarm = SigSet::empty();
149            alarm.add(Signal::SIGALRM);
150            let _ = alarm.thread_block();
151        }
152        RUN_TIMEOUT_MESSAGE_LEN.store(0, Ordering::Release);
153        RUN_TIMEOUT_MESSAGE.store(std::ptr::null_mut(), Ordering::Release);
154    }
155}
