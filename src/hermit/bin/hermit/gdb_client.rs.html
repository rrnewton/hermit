7//! Watch an owned GDB in an exec-started supervisor outside the guest namespace.
8//!
9//! The parent remains single-threaded while constructing/running the raw-clone
10//! container. Only final-scope Drop may start a helper-reaper thread: the two
11//! callers in replay and record_start return directly from that final replay or
12//! its setup error. This is a CLI lifetime contract, not a generic fork-safe
13//! background-process abstraction. A future caller must preserve that ordering.
14

35/// Existing diagnostic; a successful probe is still unauthenticated and may
36/// reach a stranger. This change does not fix that separate reporting limitation.
37pub const CLIENT_EXITED_BEFORE_CONNECTING: &str = "the gdb client hermit spawned exited before it finished connecting to the \
38     gdbserver, so the replay had no debugger to serve";
39
40type ReapOwner = Arc<Mutex<Option<Child>>>;
41// Only exceptional cleanup refusals enter this process-lifetime owner store.
42// Keeping the Child is not a successful reap; the diagnostic says so explicitly.
43static UNCONFIRMED_REAPS: Mutex<Vec<ReapOwner>> = Mutex::new(Vec::new());
44
45struct OwnerIdentity {
46    pid: libc::pid_t,
47    namespace: File,
48    pidfd: OwnedFd,
49}
50
51impl OwnerIdentity {
52    fn capture() -> Result<Self, Error> {
53        // SAFETY: getpid and pidfd_open have no pointer arguments.
54        let pid = unsafe { libc::getpid() };
55        let raw = unsafe { libc::syscall(libc::SYS_pidfd_open, pid, 0) };
56        if raw < 0 {
57            return Err(io::Error::last_os_error()).context("open GDB watch parent pidfd");
58        }
59        // SAFETY: pidfd_open returned a new owned descriptor.
60        let pidfd = unsafe { OwnedFd::from_raw_fd(raw as i32) };
61        Ok(Self {
62            pid,
63            namespace: File::open("/proc/thread-self/ns/pid")
64                .context("retain GDB watch parent PID namespace")?,
65            pidfd,
66        })
67    }
68
69    fn is_current(&self) -> Result<bool, Error> {
70        // Raw clone copies this entire object. A foreign copy must not send
71        // Done, wait the parent's child, or start its own reaper thread.
72        if unsafe { libc::getpid() } != self.pid || helper::parent_exited(self.pidfd.as_raw_fd())? {
73            return Ok(false);
74        }
75        let current = std::fs::metadata("/proc/thread-self/ns/pid")
76            .context("identify GDB watch caller PID namespace")?;
77        let original = self.namespace.metadata()?;
78        Ok(current.dev() == original.dev() && current.ino() == original.ino())
79    }
80}

82fn retain_unconfirmed(owner: ReapOwner, error: impl std::fmt::Display) {
83    eprintln!(
84        "GDB helper reap is unconfirmed: {error}; retaining its child owner until process exit"
85    );
86    UNCONFIRMED_REAPS
87        .lock()
88        .unwrap_or_else(std::sync::PoisonError::into_inner)
89        .push(owner);
90}

92fn reap_after_final_scope(child: Child, force_spawn_refusal: bool) {
93    let owner = Arc::new(Mutex::new(Some(child)));
94    let reaper_owner = Arc::clone(&owner);
95    let reap = move || {
96        let mut child = reaper_owner
97            .lock()
98            .unwrap_or_else(std::sync::PoisonError::into_inner)
99            .take()
100            .expect("one helper reap owner");
101        // Never hold the owner lock across wait, and never use waitpid(-1).
102        if let Err(error) = child.wait() {
103            *reaper_owner
104                .lock()
105                .unwrap_or_else(std::sync::PoisonError::into_inner) = Some(child);
106            retain_unconfirmed(reaper_owner, error);
107        }
108    };
109    let started = if force_spawn_refusal {
110        // Test seam models Builder refusing and destroying its consumed closure.
111        drop(reap);
112        Err(io::Error::from_raw_os_error(libc::EAGAIN))
113    } else {
114        std::thread::Builder::new()
115            .name("gdb-helper-reaper".into())
116            .spawn(reap)
117    };
118    if let Err(error) = started {
119        retain_unconfirmed(owner, error);
120    }
121}

123/// Owns the helper and its private IPC, not the GDB Child (the helper owns that).
124pub struct GdbClientWatch {
125    owner: OwnerIdentity,
126    helper: Option<Child>,
127    stream: Option<UnixStream>,
128    statuses: helper::StatusReader,
129    done_sent: bool,
130    finished: bool,
131    client_exited_early: bool,
132    #[cfg(test)]
133    client_pid: u32,
134    #[cfg(test)]
135    helper_control_fd: i32,
136    #[cfg(test)]
137    client_exited_before_container_finished: bool,
138    #[cfg(test)]
139    force_reaper_spawn_refusal: bool,
140}
141
142impl GdbClientWatch {
143    /// Start the helper before any container clone, then require confirmation
144    /// that it owns the actual GDB. Both CLI command builders use arguments and
145    /// inherited stdio; see ClientCommand's internal configuration contract.
146    pub fn spawn(command: Command, port: u16) -> Result<Self, Error> {
147        let owner = OwnerIdentity::capture()?;
148        let (stream, helper_stream) =
149            UnixStream::pair().context("create GDB helper control socket")?;
150        let control = helper_stream.as_raw_fd();
151        let parent = owner.pidfd.as_raw_fd();
152        let mut launcher = helper::helper_command(control, parent);
153        // The only pre_exec work is async-signal-safe fcntl on two owned FDs.
154        // Neither descriptor is made inheritable in the original parent.
155        unsafe {
156            launcher.pre_exec(move || {
157                for fd in [control, parent] {
158                    if libc::fcntl(fd, libc::F_SETFD, 0) < 0 {
159                        return Err(io::Error::last_os_error());
160                    }
161                }
162                Ok(())
163            });
164        }
165        let child = launcher.spawn().context("start exec-owned GDB helper")?;
166        drop(helper_stream);
167        let mut watch = Self {
168            owner,
169            helper: Some(child),
170            stream: Some(stream),
171            statuses: helper::StatusReader::default(),
172            done_sent: false,
173            finished: false,
174            client_exited_early: false,
175            #[cfg(test)]
176            client_pid: 0,
177            #[cfg(test)]
178            helper_control_fd: control,
179            #[cfg(test)]
180            client_exited_before_container_finished: false,
181            #[cfg(test)]
182            force_reaper_spawn_refusal: false,
183        };
184        helper::ClientCommand::new(command, port).send(watch.stream.as_mut().unwrap())?;
185        let status = watch
186            .statuses
187            .next(watch.stream.as_mut().unwrap())?
188            .context("missing GDB spawn status")?;
189        match status.kind {
190            helper::READY if status.value > 0 => {
191                #[cfg(test)]
192                {
193                    watch.client_pid = status.value;
194                }
195                Ok(watch)
196            }
197            helper::SPAWN_FAILED => Err(io::Error::from_raw_os_error(status.value as i32))
198                .context("Failed to run gdb command. Please make sure it is in your $PATH."),
199            _ => anyhow::bail!("unexpected GDB helper startup status: {status:?}"),
200        }
201    }
202
203    fn signal_done(&mut self) -> Result<(), Error> {
204        if self.done_sent {
205            return Ok(());
206        }
207        let stream = self
208            .stream
209            .as_ref()
210            .context("GDB helper control socket is closed")?;
211        let byte = [helper::DONE];
212        loop {
213            // One byte after the acknowledged command cannot fill our send
214            // queue. DONTWAIT also keeps every Drop error path nonblocking.
215            let sent = unsafe {
216                libc::send(
217                    stream.as_raw_fd(),
218                    byte.as_ptr().cast(),
219                    1,
220                    libc::MSG_NOSIGNAL | libc::MSG_DONTWAIT,
221                )
222            };
223            if sent == 1 {
224                self.done_sent = true;
225                return Ok(());
226            }
227            let error = io::Error::last_os_error();
228            if error.kind() == io::ErrorKind::Interrupted {
229                continue;
230            }
231            return Err(error).context("notify GDB helper of container completion");
232        }
233    }
234
235    fn accept_status(&mut self, status: helper::Status) -> Result<bool, Error> {
236        match status.kind {
237            helper::EXITED_BEFORE_DONE if status.value == 0 => {
238                #[cfg(test)]
239                {
240                    self.client_exited_before_container_finished = true;
241                }
242            }
243            helper::RELEASE_CONNECTED if status.value == 0 => self.client_exited_early = true,
244            helper::FINISHED if status.value <= 1 => {
245                anyhow::ensure!(
246                    self.client_exited_early == (status.value != 0),
247                    "inconsistent final GDB helper status"
248                );
249                return Ok(true);
250            }
251            helper::FAILED => anyhow::bail!(
252                "GDB helper failed: {}",
253                io::Error::from_raw_os_error(status.value as i32)
254            ),
255            _ => anyhow::bail!("unexpected GDB helper status: {status:?}"),
256        }
257        Ok(false)
258    }
259
260    /// Publish completion, wait for GDB and reap its helper. Interactive GDB is
261    /// deliberately allowed to outlive the container; this wait is unchanged.
262    /// Protocol failures are errors, never a silent 'not early' report.
263    pub fn finish(&mut self) -> Result<bool, Error> {
264        anyhow::ensure!(
265            self.owner.is_current()?,
266            "GDB watch used outside its owning process"
267        );
268        if self.finished {
269            return Ok(self.client_exited_early);
270        }
271        self.signal_done()?;
272        let observed: Result<bool, Error> = (|| {
273            self.stream.as_mut().unwrap().set_nonblocking(false)?;
274            loop {
275                let status = self
276                    .statuses
277                    .next(self.stream.as_mut().unwrap())?
278                    .context("missing final GDB helper status")?;
279                if self.accept_status(status)? {
280                    return Ok(self.client_exited_early);
281                }
282            }
283        })();
284        // Reap even when the status stream was malformed or truncated.
285        let reaped = self
286            .helper
287            .as_mut()
288            .context("GDB helper already consumed")?
289            .wait()
290            .context("reap GDB helper");
291        match reaped {
292            Ok(status) => {
293                self.helper.take();
294                self.finished = status.success() && observed.is_ok();
295                if !status.success() {
296                    return Err(observed
297                        .err()
298                        .unwrap_or_else(|| anyhow::anyhow!("GDB helper exited with {status}")));
299                }
300                observed
301            }
302            Err(error) => match observed {
303                Err(observation) => {
304                    Err(observation.context(format!("reap GDB helper failed: {error:#}")))
305                }
306                Ok(_) => Err(error),
307            },
308        }
309    }
310
311    #[cfg(test)]
312    fn observed_client_exit(&mut self) -> Result<bool, Error> {
313        self.stream.as_mut().unwrap().set_nonblocking(true)?;
314        while let Some(status) = self.statuses.next(self.stream.as_mut().unwrap())? {
315            anyhow::ensure!(
316                !self.accept_status(status)?,
317                "helper finished before container completion"
318            );
319        }
320        Ok(self.client_exited_before_container_finished)
321    }
322}

324impl Drop for GdbClientWatch {
325    /// Only the original process owns cleanup. Raw-cloned copies close their
326    /// local aliases without sending Done or touching the parent's child.
327    fn drop(&mut self) {
328        match self.owner.is_current() {
329            Ok(false) => return,
330            Err(error) => {
331                if let Some(child) = self.helper.take() {
332                    retain_unconfirmed(Arc::new(Mutex::new(Some(child))), error);
333                }
334                return;
335            }
336            Ok(true) => {}
337        }
338        let Some(mut child) = self.helper.take() else {
339            return;
340        };
341        if let Err(error) = self.signal_done() {
342            eprintln!("GDB helper completion notification failed: {error:#}");
343        }
344        self.stream.take();
345        match child.try_wait() {
346            Ok(Some(_)) => {}
347            Ok(None) => {
348                #[cfg(test)]
349                let force_refusal = self.force_reaper_spawn_refusal;
350                #[cfg(not(test))]
351                let force_refusal = false;
352                reap_after_final_scope(child, force_refusal);
353            }
354            Err(error) => retain_unconfirmed(Arc::new(Mutex::new(Some(child))), error),
355        }
356    }
357}

389    fn helper_owns_gdb_without_leaking_its_control_descriptors() {
390        let mut command = Command::new("/bin/sleep");
391        command.arg("30");
392        let mut watch = GdbClientWatch::spawn(command, 1234).unwrap();
393        let helper_pid = watch.helper.as_ref().unwrap().id();
394        let client_pid = watch.client_pid;
395        struct EndClient(u32);
396        impl Drop for EndClient {
397            fn drop(&mut self) {
398                // SAFETY: this is the live, owned stand-in announced by READY.
399                unsafe {
400                    libc::kill(self.0 as i32, libc::SIGTERM);
401                }
402            }
403        }
404        let rescue = EndClient(client_pid);
405        // Pin the stand-in before checking its parent, so this pidfd names the
406        // helper's only child even if the PID is reused later.
407        // SAFETY: pidfd_open has no pointer arguments.
408        let raw = unsafe { libc::syscall(libc::SYS_pidfd_open, client_pid as libc::pid_t, 0) };
409        assert!(
410            raw >= 0,
411            "pin the GDB stand-in: {}",
412            io::Error::last_os_error()
413        );
414        // SAFETY: pidfd_open returned a new owned descriptor.
415        let client = unsafe { OwnedFd::from_raw_fd(raw as i32) };
416        let exited = || helper::parent_exited(client.as_raw_fd()).expect("poll the stand-in pidfd");
417        let status = std::fs::read_to_string(format!("/proc/{client_pid}/status")).unwrap();
418        let parent = status
419            .lines()
420            .find_map(|line| line.strip_prefix("PPid:"))
421            .unwrap();
422        assert_eq!(parent.trim().parse::<u32>().unwrap(), helper_pid);
423
424        let private = [watch.helper_control_fd, watch.owner.pidfd.as_raw_fd()].map(|fd| {
425            let metadata = std::fs::metadata(format!("/proc/{helper_pid}/fd/{fd}")).unwrap();
426            (metadata.dev(), metadata.ino())
427        });
428        for entry in std::fs::read_dir(format!("/proc/{client_pid}/fd")).unwrap() {
429            // The stand-in's loader still opens and closes its own descriptors
430            // after READY, so an entry may vanish before its stat. That alone
431            // does not show it was never inherited; see the check below.
432            let metadata = match std::fs::metadata(entry.unwrap().path()) {
433                Ok(metadata) => metadata,
434                Err(error) if error.kind() == io::ErrorKind::NotFound => continue,
435                Err(error) => panic!("stat GDB child descriptor: {error}"),
436            };
437            assert!(
438                !private.contains(&(metadata.dev(), metadata.ino())),
439                "the GDB child inherited a private watcher descriptor"
440            );
441        }
442        // While it runs, the stand-in closes only descriptors it opened itself.
443        // Its exit closes the rest: GNU sleep's atexit handler closes 1 and 2,
444        // exit_files() then drops the whole table, and only after that does
445        // the pidfd report the exit. So the skip is sound only if the same
446        // stand-in still has descriptors 0 to 2 and has not exited.
447        let stdio =
448            (0..3).all(|fd| std::fs::metadata(format!("/proc/{client_pid}/fd/{fd}")).is_ok());
449        assert!(
450            stdio && !exited(),
451            "the GDB stand-in exited during the descriptor scan, so a vanished entry may have been inherited"
452        );
453        drop(rescue);
454        watch.finish().expect("GDB helper finish failed");
455        // Control: finish() returns only after the helper reaps the stand-in,
456        // so the same pidfd must now report the exit the check above rejects.
457        assert!(exited(), "the stand-in pidfd did not report its exit");
458    }

531    /// `finish()` must not sit out a grace period it no longer needs.
532    ///
533    /// ⚠️ THE GRACE SLEEP IS ON HERMIT'S OWN RETURN PATH, because `finish()` sets
534    /// `container_done` and then JOINS this thread. A flat `for _ in
535    /// 0..RELEASE_GRACE_TICKS { sleep }` cannot notice the container finishing, so
536    /// hermit waited out the remainder of a pause whose only purpose is to slow
537    /// down contacting a stranger. MEASURED on this test, entering the sleep and
538    /// then calling `finish()`:
539    ///
540    ///     without the `done` check .... 381.7ms
541    ///     with it ..................... 0.52ms
542    ///
543    /// The bound is one `RELEASE_RETRY_INTERVAL` (20ms) plus scheduling, so 100ms
544    /// is generous while still an order of magnitude below the regression.
545    ///
546    /// ⚠️ THIS IS NOT THE `done` CHECK THAT WAS REMOVED FROM THIS LOOP. That one
547    /// read `done` to decide whether the early-exit report was EARNED, which it
548    /// cannot do -- `done` is correlation, not authentication. This one decides
549    /// only when to stop sleeping, and no report depends on it.

552        let listener = TcpListener::bind(("127.0.0.1", 0)).expect("failed to bind");
553        let port = listener.local_addr().expect("no local addr").port();
554        thread::spawn(move || {

556                match stream {
557                    Ok(stream) => drop(stream),

560            }

563        let client = std::process::Command::new("/bin/true");
564        let mut watch =
565            GdbClientWatch::spawn(client, port).expect("failed to spawn the stand-in client");
566        // Let the client exit, so the watcher reaches the release loop.
567        thread::sleep(Duration::from_millis(200));

569        // grace sleep, short enough that the sleep is still running.
570        thread::sleep(Duration::from_millis(120));

573        let _ = watch.finish().expect("GDB helper finish failed");

584    /// Connecting to a listener that is not ours must not end the watch.
585    ///
586    /// ⚠️ THIS IS THE HANG, RESTORED BY A SUCCESSFUL CONNECT, and it is live on
587    /// `main` without the grace window: the release port is guessable
588    /// (`16384 + tid % 1024`, or 1234 for `replay`), so an unrelated process can
589    /// own it in exactly the window the release loop exists for. A watcher that
590    /// concludes on the first successful connect then stops watching while our
591    /// own gdbserver is still blocked in `accept()` with nobody coming.
592    ///
593    /// ⚠️ THE DISCRIMINATOR IS THE RETRY, NOT THE FLAG, and that is deliberate.
594    /// An earlier version of this fix tried to tell "we released ours" from "a
595    /// stranger accepted" by reading the socket for a prompt EOF. Measured, that
596    /// probe cannot do it: our probe reads BEFORE it disconnects, so a gdbstub
597    /// that has just accepted has no reason to close, and the genuine-release and
598    /// already-served cases BOTH read as not-accepted. Asserting a retry count
599    /// asks only what this loop actually controls.
600    ///
601    /// ⚠️ AND THE NAME NOW SAYS "STILL LATCHES THE REPORT", BECAUSE IT DOES. This
602    /// was called `a_connect_to_a_stranger_is_not_treated_as_a_release` while
603    /// discarding `watch.finish()` and never reading the flag — so the test named
604    /// for the defect did not assert it, and `assert!(!watch.finish())` fails
605    /// here. A stranger IS still treated as a release for reporting purposes.
606    /// Naming the test for the property it has, and asserting the gap it does
607    /// not close, is the difference between coverage and the appearance of it.
608    ///
609    /// Bounded, and asserts a count rather than blocking, so a regression is a
610    /// named red and not a wedged runner.
611    #[test]
612    fn a_stranger_does_not_end_the_release_attempts_but_still_latches_the_report() {

832        let port = listener.local_addr().expect("no local addr").port();
