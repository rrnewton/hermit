9//! Discovery of OCI images already present in the local containers/storage
10//! image store, via the `podman` CLI.
11//!
12//! # Why the CLI and not a library
13//!
14//! `containers/image` and `containers/storage` — the components that own the
15//! image store — are Go libraries. Linking them into this Rust binary means cgo
16//! plus a compiled-in pin of one store implementation, while the on-disk format
17//! is versioned and can differ per install (libpod state may be sqlite while
18//! containers/storage metadata is still JSON). The store's *documented* surface
19//! is the `podman` CLI, so that is what this module binds to. Parsing
20//! `overlay-images/images.json` or `db.sql` directly would bind hermit to an
21//! internal format and is deliberately not done here.
22//!
23//! # Relationship to [`crate::app::image`]
24//!
25//! `buildah` (used by [`crate::app::image::materialize_rootfs`]) and `podman` share
26//! the same graph root, so `--image` already reads this same store. This module
27//! adds the piece that was missing: turning a user-supplied *reference* into a
28//! stable image *identity* before anything is cached against it, plus the
29//! listing and inspection surface behind `hermit oci`.
30

39/// Program used to talk to the local image store.
40const PODMAN: &str = "podman";
41
42/// A resolved local image: the reference the user gave, plus the identity the
43/// store assigned it.
44#[derive(Debug, Clone, PartialEq, Eq)]
45pub(crate) struct ResolvedImage {
46    /// The reference as the user wrote it (tag, short id, digest, ...).
47    pub reference: String,
48    /// The store's canonical 64-hex image ID. This — not [`Self::reference`] —
49    /// is the identity anything may be cached against.
50    pub id: String,
51    /// The image's manifest digest (`sha256:...`), when the store records one.
52    pub digest: Option<String>,
53}

55/// A row of `hermit oci ls`.
56#[derive(Debug, Clone, Deserialize)]
57pub(crate) struct ImageSummary {
58    #[serde(rename = "Id")]
59    pub id: String,
60    #[serde(rename = "Names", default)]
61    pub names: Option<Vec<String>>,
62    #[serde(rename = "Size", default)]
63    pub size: u64,
64}
65
66impl ImageSummary {
67    /// The best human-readable name for this image, or `<none>` when the image
68    /// is untagged (an intermediate build layer, typically).
69    pub fn display_name(&self) -> String {
70        self.names
71            .as_ref()
72            .and_then(|names| names.first().cloned())
73            .unwrap_or_else(|| "<none>".to_string())
74    }
75}

77/// A handle to the local image store.
78pub(crate) struct PodmanStore {
79    /// The store's graph root, as podman reports it. Recorded so error messages
80    /// and `hermit oci inspect` can say *which* store was consulted.
81    graph_root: String,
82}
83
84/// Total WALL budget for waiting out a locked podman database, shared by
85/// every podman invocation in this binary. Concurrent validates share one
86/// containers/storage store; its database lock is held for milliseconds,
87/// so contention is transient — but an unbounded or silent retry would
88/// hide a wedged store, so the wait is bounded and counted.
89pub(crate) const PODMAN_LOCK_WAIT_BUDGET: std::time::Duration = std::time::Duration::from_secs(120);
90
91/// Whether podman's stderr reports the shared storage database lock.
92pub(crate) fn podman_database_locked(stderr: &str) -> bool {
93    stderr.contains("database locked")
94}
95
96/// Wait once for the podman database lock, counting the attempt out loud.
97/// `attempt` is 1-based. Returns false when the budget is exhausted, in
98/// which case the caller surfaces the lock failure (never retries past
99/// the bound, never silently).
100pub(crate) fn wait_for_podman_lock(attempt: u32, started: std::time::Instant) -> bool {
101    let elapsed = started.elapsed();
102    if elapsed >= PODMAN_LOCK_WAIT_BUDGET {
103        return false;
104    }
105    let delay =
106        (std::time::Duration::from_millis(250) * attempt).min(std::time::Duration::from_secs(5));
107    eprintln!(
108        "hermit: podman database locked; lock wait attempt {attempt} \
109         (elapsed {:.1}s of {:.0}s budget)",
110        elapsed.as_secs_f64(),
111        PODMAN_LOCK_WAIT_BUDGET.as_secs_f64(),
112    );
113    std::thread::sleep(delay);
114    true
115}

117/// Run a podman command and capture stdout, mapping a nonzero exit to an error
118/// that includes podman's own stderr (which is where its diagnostics go).
119fn podman_output(args: &[&str]) -> Result<String, Error> {
120    let started = std::time::Instant::now();
121    let mut attempt = 0;
122    loop {
123        let output = Command::new(PODMAN)
124            .args(args)
125            .stdin(Stdio::null())
126            .output()
127            .with_context(|| {
128                format!(
129                    "Failed to run `{PODMAN} {}`. `hermit oci` needs a rootless \
130                     `{PODMAN}` on PATH.",
131                    args.join(" ")
132                )
133            })?;
134        if output.status.success() {
135            return Ok(String::from_utf8_lossy(&output.stdout).trim().to_string());
136        }
137        let stderr = String::from_utf8_lossy(&output.stderr);
138        attempt += 1;
139        if !podman_database_locked(&stderr) || !wait_for_podman_lock(attempt, started) {
140            return Err(Error::msg(format!(
141                "`{PODMAN} {}` failed ({}): {}",
142                args.join(" "),
143                output.status,
144                stderr.trim()
145            )));
146        }
147    }
148}

150impl PodmanStore {
151    /// Detect a usable local image store. Returns `Ok(None)` when podman is
152    /// absent or reports no store, so callers can fall back rather than fail.
153    pub fn probe() -> Result<Option<Self>, Error> {
154        match podman_output(&["info", "--format", "{{.Store.GraphRoot}}"]) {
155            Ok(graph_root) if !graph_root.is_empty() => Ok(Some(Self { graph_root })),
156            // A podman that exists but cannot report a store is not usable, and
157            // neither is a missing podman. Both are "no store here".
158            _ => Ok(None),
159        }
160    }
161
162    pub fn graph_root(&self) -> &str {
163        &self.graph_root
164    }
165
166    /// List the images in the store. `all` includes untagged intermediates.
167    pub fn list(&self, all: bool) -> Result<Vec<ImageSummary>, Error> {
168        let mut args = vec!["images", "--format", "json"];
169        if all {
170            args.push("--all");
171        }
172        let stdout = podman_output(&args)?;
173        if stdout.is_empty() {
174            return Ok(Vec::new());
175        }
176        serde_json::from_str(&stdout)
177            .context("Failed to parse `podman images --format json` output")
178    }
179
180    /// Resolve a reference to the store's canonical image identity **without
181    /// touching the network**.
182    ///
183    /// `podman pull --policy=never` is the right primitive here: it accepts
184    /// every reference form (tag, short id, full id, `name@sha256:...`),
185    /// returns the canonical 64-hex ID, and is guaranteed not to contact a
186    /// registry. That last property is what makes this safe to call on the
187    /// normal `hermit oci run` path — resolution never silently downloads.
188    pub fn resolve(&self, reference: &str) -> Result<ResolvedImage, Error> {
189        let id = podman_output(&["pull", "--policy=never", reference]).map_err(|e| {
190            Error::msg(format!(
191                "No local image matches {reference:?} in the store at {}. \
192                 Use `hermit oci ls` to see what is available, or \
193                 `hermit oci pull {reference}` to download it.\n  ({e})",
194                self.graph_root
195            ))
196        })?;
197        self.identify(reference, &id)
198    }
199
200    /// Download `reference` into the store, then resolve it. This is the only
201    /// method here that contacts a registry.
202    pub fn pull(&self, reference: &str) -> Result<ResolvedImage, Error> {
203        let id = podman_output(&["pull", reference]).map_err(|e| {
204            Error::msg(format!(
205                "Failed to pull {reference:?}. If the registry is reachable only \
206                 through a proxy, run hermit under that proxy.\n  ({e})"
207            ))
208        })?;
209        self.identify(reference, &id)
210    }
211
212    /// Turn a store-reported ID into a [`ResolvedImage`], attaching the
213    /// manifest digest when the store records one.
214    fn identify(&self, reference: &str, id: &str) -> Result<ResolvedImage, Error> {
215        let id = normalize_image_id(id).ok_or_else(|| {
216            Error::msg(format!(
217                "podman returned an unrecognized image id for {reference:?}: {id:?}"
218            ))
219        })?;
220        // An image built locally has no manifest digest until it is pushed or
221        // pulled, so a missing digest is normal and must not be an error.
222        let digest = podman_output(&["image", "inspect", &id, "--format", "{{.Digest}}"])
223            .ok()
224            .filter(|d| d.starts_with("sha256:"));
225        Ok(ResolvedImage {
226            reference: reference.to_string(),
227            id,
228            digest,
229        })
230    }
231
232    /// The image's declared `Config.Env` and `Config.WorkingDir`.
233    pub fn config(&self, image: &ResolvedImage) -> Result<ImageRunConfig, Error> {
234        let stdout = podman_output(&[
235            "image",
236            "inspect",
237            &image.id,
238            "--format",
239            "{{json .Config}}",
240        ])?;
241        let raw: RawConfig = serde_json::from_str(&stdout)
242            .context("Failed to parse `podman image inspect` config output")?;
243        Ok(ImageRunConfig {
244            env: raw
245                .env
246                .unwrap_or_default()
247                .iter()
248                .filter_map(|entry| entry.split_once('='))
249                .map(|(k, v)| (k.to_string(), v.to_string()))
250                .collect(),
251            workdir: raw.working_dir.filter(|w| !w.is_empty() && w != "/"),
252            cmd: raw.cmd.unwrap_or_default(),
253            entrypoint: raw.entrypoint.unwrap_or_default(),
254        })
255    }
256
257    /// The overlay directories backing this image's filesystem, in the order
258    /// an `overlay` mount wants them (topmost first).
259    ///
260    /// This is what makes a future copy-free materialization possible: the
261    /// merged rootfs can be mounted directly from the store's own layers inside
262    /// hermit's user namespace, instead of being copied out.
263    pub fn overlay_dirs(&self, image: &ResolvedImage) -> Result<Vec<String>, Error> {
264        let stdout = podman_output(&[
265            "image",
266            "inspect",
267            &image.id,
268            "--format",
269            "{{json .GraphDriver}}",
270        ])?;
271        let graph: GraphDriver = serde_json::from_str(&stdout)
272            .context("Failed to parse `podman image inspect` GraphDriver output")?;
273        if graph.name != "overlay" {
274            return Err(Error::msg(format!(
275                "image {} uses the {:?} storage driver; only `overlay` exposes \
276                 mountable layer directories",
277                image.id, graph.name
278            )));
279        }
280        Ok(overlay_layer_order(
281            graph.data.get("UpperDir").map(String::as_str),
282            graph.data.get("LowerDir").map(String::as_str),
283        ))
284    }
285}

296#[derive(Deserialize)]
297struct RawConfig {
298    #[serde(rename = "Env")]
299    env: Option<Vec<String>>,

302    #[serde(rename = "Cmd")]

308#[derive(Deserialize)]
309struct GraphDriver {
310    #[serde(rename = "Name")]
311    name: String,
312    #[serde(rename = "Data", default)]
313    data: BTreeMap<String, String>,
314}

316/// Assemble the overlay layer list, topmost first.
317///
318/// podman reports `UpperDir` (the image's own top layer) separately from the
319/// colon-separated `LowerDir` chain, and `LowerDir` is absent entirely for a
320/// single-layer image. An `overlay` mount wants all of them as `lowerdir` in
321/// topmost-first order, because hermit supplies its own writable upper layer.
322fn overlay_layer_order(upper: Option<&str>, lower: Option<&str>) -> Vec<String> {
323    let mut dirs = Vec::new();
324    if let Some(upper) = upper.map(str::trim).filter(|u| !u.is_empty()) {
325        dirs.push(upper.to_string());
326    }
327    if let Some(lower) = lower {
328        dirs.extend(
329            lower
330                .split(':')
331                .map(str::trim)
332                .filter(|d| !d.is_empty())
333                .map(str::to_string),
334        );
335    }
336    dirs
337}

339/// Accept the forms podman prints for an image id and reduce them to bare
340/// 64-hex. `podman pull` prints the bare id; some paths print `sha256:<id>`.
341fn normalize_image_id(raw: &str) -> Option<String> {
342    let candidate = raw.trim();
343    let candidate = candidate.strip_prefix("sha256:").unwrap_or(candidate);
344    (candidate.len() == 64 && candidate.chars().all(|c| c.is_ascii_hexdigit()))
345        .then(|| candidate.to_ascii_lowercase())
346}

352    #[test]
353    fn normalize_image_id_accepts_bare_and_prefixed_hex() {
354        let id = "1ee23df5e47b5b7331304af929f5461f9f516f5c166c3031832d2adfd22f323c";
355        assert_eq!(normalize_image_id(id).as_deref(), Some(id));
356        assert_eq!(
357            normalize_image_id(&format!("sha256:{id}")).as_deref(),
358            Some(id)
359        );
360        assert_eq!(
361            normalize_image_id(&format!("  {id}\n")).as_deref(),
362            Some(id)
363        );
364        assert_eq!(
365            normalize_image_id(&id.to_ascii_uppercase()).as_deref(),
366            Some(id),
367            "podman ids are case-insensitive hex; normalize to lowercase"
368        );
369    }

371    // A short id is a *reference*, not an identity: accepting one here would
372    // reintroduce exactly the aliasing that keying on the resolved id removes.
373    #[test]
374    fn normalize_image_id_rejects_short_ids_and_non_hex() {
375        assert_eq!(normalize_image_id("1ee23df5e47b"), None);
376        assert_eq!(normalize_image_id(""), None);
377        assert_eq!(normalize_image_id("not-an-id"), None);
378        // 64 chars but not hex.
379        assert_eq!(normalize_image_id(&"z".repeat(64)), None);
380    }
