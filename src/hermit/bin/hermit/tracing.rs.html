39/// Ceiling on one run's log file.
40///
41/// A run that makes no progress still logs. Measured 2026-08-11 over the 4111
42/// retained `--verify` comparison logs on one host: p50 0.4 MiB, p90 43.5 MiB,
43/// p99 145.5 MiB -- and then a tail to 928.8 GiB, 4357.4 GiB in total. The
44/// largest single file was a KVM guest livelocked on `sched_yield`, logging the
45/// same fifteen line shapes for 11.7 hours; its registers at the 50 GiB and
46/// 900 GiB offsets were identical.
47///
48/// 1 GiB is about 7x p99, so it never truncates a log anyone reads, and it
49/// removes 98.6% of those bytes while touching 0.56% of the files. The
50/// distribution is bimodal enough that any bound from 0.25 to 16 GiB reclaims
51/// over 94%, so this is a safety limit rather than a tuning parameter.
52///
53/// Compression is NOT the control. `/tmp` happens to be zstd-compressed here,
54/// which is why a terabyte of repetitive DETLOG has been survivable, but a
55/// compression ratio scales with how repetitive the runaway happens to be and
56/// silently converts a hard failure into an invisible one.
57pub const DEFAULT_LOG_MAX_BYTES: u64 = 1 << 30;

59/// Written once, in-band, when the bound is reached.
60///
61/// A truncated log MUST say so. Output that simply stops reads as a run that
62/// ENDED, which would be a worse evidence defect than the disk hazard this
63/// bound removes: a reader would draw conclusions from an absence we created.
64///
65/// The surrounding newlines are load-bearing, not cosmetic. The marker must
66/// occupy a LINE OF ITS OWN at the very END of the file, because that is what
67/// [`detcore::logdiff::log_was_truncated`] anchors on to tell a truncated log
68/// apart from a log that merely quotes the text. This literal is deliberately
69/// NOT `detcore::logdiff::TRUNCATION_MARKER` itself: keeping the two texts
70/// independent is what leaves
71/// `the_written_marker_is_the_text_the_comparator_matches` able to fail.
72const TRUNCATION_MARKER: &[u8] =
73    b"\n=== HERMIT LOG TRUNCATED: reached the configured size bound (HERMIT_LOG_MAX_BYTES). \
74Output beyond this point was DISCARDED. The run itself continued and was NOT affected. ===\n";
75
76/// Resolve the configured ceiling; `0` means unlimited.
77///
78/// A malformed value is an ERROR rather than a fallback. Silently substituting
79/// the 1 GiB default would mean `HERMIT_LOG_MAX_BYTES=unlimited`, or any typo
80/// in the value used to disable the bound, quietly re-enabled it -- i.e. the
81/// documented way to turn the bound off is one keystroke away from turning it
82/// back on without saying so.
83pub fn log_max_bytes() -> Result<u64, String> {
84    match std::env::var(LOG_MAX_BYTES_ENV) {
85        Ok(raw) => raw.trim().parse().map_err(|_| {
86            format!(
87                "{LOG_MAX_BYTES_ENV}={raw:?} is not a byte count. Set a non-negative integer \
88                 number of bytes, or 0 to disable the bound."
89            )
90        }),
91        Err(_) => Ok(DEFAULT_LOG_MAX_BYTES),
92    }
93}

95/// A writer that stops after `limit` bytes and says so in-band.
96///
97/// Beyond the bound this reports the bytes as consumed rather than returning a
98/// short count: `tracing_appender` treats a short write as an I/O error, so a
99/// truthful-but-short return would turn a size limit into a logging failure.
100pub struct BoundedWriter<W: Write> {
101    inner: W,
102    remaining: u64,
103    bounded: bool,
104    announced: bool,
105}
106
107impl<W: Write> BoundedWriter<W> {
108    /// `limit == 0` disables the bound entirely.
109    pub fn new(inner: W, limit: u64) -> Self {
110        Self {
111            inner,
112            remaining: limit,
113            bounded: limit != 0,
114            announced: false,
115        }
116    }
117
118    /// Continues a log that already holds `written` bytes under the same
119    /// `limit` (`0`: unbounded): the bound counts those bytes too, so appending
120    /// to a log cannot take it past the ceiling, and appending to a log that
121    /// has already reached it discards the bytes and announces the truncation.
122    pub fn resume(inner: W, limit: u64, written: u64) -> Self {
123        Self {
124            inner,
125            remaining: limit.saturating_sub(written),
126            bounded: limit != 0,
127            announced: false,
128        }
129    }
130}

132impl<W: Write> BoundedWriter<W> {
133    /// Whether bytes have been discarded and the truncation marker written.
134    /// From then on the marker is this log's final line.
135    fn has_truncated(&self) -> bool {
136        self.announced
137    }
138
139    /// Announce truncation the FIRST time bytes are actually discarded.
140    ///
141    /// Deferring this to the next `write` leaves a log that was truncated on
142    /// its final write silent, which is the exact failure the marker exists to
143    /// prevent -- caught end-to-end: a 100-byte bound produced exactly 100
144    /// bytes and no marker, because no further write ever arrived.
145    fn announce_truncation(&mut self) -> io::Result<()> {
146        if !self.announced {
147            self.announced = true;
148            self.inner.write_all(TRUNCATION_MARKER)?;
149            self.inner.flush()?;
150        }
151        Ok(())
152    }
153}
154
155impl<W: Write> Write for BoundedWriter<W> {
156    fn write(&mut self, buf: &[u8]) -> io::Result<usize> {
157        if !self.bounded {
158            return self.inner.write(buf);
159        }
160        if self.remaining == 0 {
161            self.announce_truncation()?;
162            return Ok(buf.len());
163        }
164        let take = buf.len().min(self.remaining as usize);
165        let written = self.inner.write(&buf[..take])?;
166        self.remaining -= written as u64;
167        if written < take {
168            // A genuine short write by the inner writer: report it truthfully
169            // and let the caller retry the remainder.
170            return Ok(written);
171        }
172        if take < buf.len() {
173            // This write is the one that crossed the bound; its tail is being
174            // dropped, so say so now rather than hoping for another write.
175            self.announce_truncation()?;
176        }
177        Ok(buf.len())
178    }
179
180    fn flush(&mut self) -> io::Result<()> {
181        self.inner.flush()
182    }
183}

185/// Parse a `--max-log-bytes` value: a byte count with an optional binary
186/// suffix. `8G` is 8 GiB; `K`, `M`, `G`, `T` are powers of 1024, may be lower
187/// case, and may be spelled `Ki`/`KiB`/`KB` alike -- there is deliberately no
188/// SI reading, so `8G` and `8GiB` cannot mean two different caps.
189///
190/// Zero is refused rather than read as "unlimited": a cap that aborts on the
191/// first log line is never what anyone meant, and the way to run uncapped is to
192/// omit the flag.
193pub fn parse_max_log_bytes(raw: &str) -> Result<u64, String> {
194    let usage = "Give a positive byte count with an optional K/M/G/T suffix \
195                 (powers of 1024), e.g. --max-log-bytes=8G.";
196    let text = raw.trim();
197    let split = text
198        .find(|c: char| !c.is_ascii_digit())
199        .unwrap_or(text.len());
200    let (digits, suffix) = text.split_at(split);
201    if digits.is_empty() {
202        return Err(format!("{raw:?} is not a byte count. {usage}"));
203    }
204    let shift = match suffix.to_ascii_lowercase().as_str() {
205        "" | "b" => 0,
206        "k" | "ki" | "kb" | "kib" => 10,
207        "m" | "mi" | "mb" | "mib" => 20,
208        "g" | "gi" | "gb" | "gib" => 30,
209        "t" | "ti" | "tb" | "tib" => 40,
210        _ => return Err(format!("{raw:?} has an unknown size suffix. {usage}")),
211    };
212    let count: u64 = digits
213        .parse()
214        .map_err(|_| format!("{raw:?} is too large for a byte count. {usage}"))?;
215    let bytes = count
216        .checked_mul(1u64 << shift)
217        .ok_or_else(|| format!("{raw:?} is too large for a byte count. {usage}"))?;
218    if bytes == 0 {
219        return Err(format!(
220            "{raw:?} would abort the run on its first log line. Omit --max-log-bytes to \
221             leave hermit's log output uncapped, or give a positive size such as 8G."
222        ));
223    }
224    Ok(bytes)
225}

227/// Render a byte count the way `--max-log-bytes` accepts it, choosing the
228/// largest suffix that represents it exactly (`8589934592` -> `8G`).
229pub fn format_byte_size(bytes: u64) -> String {
230    for (shift, suffix) in [(40, "T"), (30, "G"), (20, "M"), (10, "K")] {
231        let unit = 1u64 << shift;
232        if bytes != 0 && bytes.is_multiple_of(unit) {
233            return format!("{}{suffix}", bytes / unit);
234        }
235    }
236    bytes.to_string()
237}

239/// The process-shared state of a [`LogBudget`].
240///
241/// The crossing is recorded in `admission`, separate from `spent`, because
242/// `spent` goes DOWN: a write the sink accepted only in part, or refused, is
243/// refunded. Deciding the crossing from the total alone made it reversible --
244/// with a limit of 10, A charges 6, B charges 5 and crosses, A's failed write
245/// refunds 6, and C's 6 crosses a second time. [`CROSSED`] only ever goes from
246/// clear to set.
247#[repr(C)]
248struct BudgetCells {
249    spent: AtomicU64,
250    /// The admission word. Bit 63 is [`CROSSED`]; the low 63 bits count the
251    /// charged writes that were admitted and have not finished yet, in every
252    /// process sharing the budget. See [`admit_write`].
253    admission: AtomicU64,
254}
255
256/// Bit 63 of [`BudgetCells::admission`]: a write has crossed the cap. Once it
257/// is set no write is admitted, so no new log bytes reach any sink.
258const CROSSED: u64 = 1 << 63;
259
260/// How long the crossing writer waits for log writes that were admitted
261/// before the crossing to finish. If they have not finished by then, it omits
262/// the final message: written while another write was still in progress, the
263/// message could be followed by that write's line.
264const LOG_CAP_DRAIN_BOUND: Duration = Duration::from_millis(50);
265
266/// The longest drain bound any caller may set. The crossing process must exit
267/// within one second of the crossing, whatever its sinks do (review of
268/// https://github.com/rrnewton/hermit/pull/3686, round 2, finding 1): this
269/// wait plus the final message's attempts, which never wait, stay far below
270/// that.
271const LOG_CAP_DRAIN_BOUND_MAX: Duration = Duration::from_millis(250);
272const _: () = assert!(LOG_CAP_DRAIN_BOUND.as_millis() <= LOG_CAP_DRAIN_BOUND_MAX.as_millis());
273
274/// How often the crossing writer looks at the admission word while it waits.
275const LOG_CAP_DRAIN_POLL: Duration = Duration::from_millis(1);
276
277/// Admit one charged write: count it as in progress, unless a write has
278/// already crossed the cap. One compare-and-swap, retried only when another
279/// writer changed the word first. The admitted writer must end its admission
280/// exactly once, with [`finish_write`] or [`claim_crossing`].
281///
282/// WHY ADMISSION AND THE CROSSING SHARE ONE WORD. Every change to it is a
283/// read-modify-write of the same location, so they happen in one order. A
284/// write admitted before the crossing's `fetch_or` is counted in the value that
285/// the crossing writer reads afterwards; a write that comes after it sees
286/// [`CROSSED`] and is refused. So once the count reaches zero no admitted
287/// write is left in progress and no new one can start, and the final message
288/// written then is the last line of every sink that charges this budget
289/// (review of https://github.com/rrnewton/hermit/pull/3686, round 3,
290/// finding 5).
291fn admit_write(admission: &AtomicU64) -> bool {
292    let mut current = admission.load(Ordering::Relaxed);
293    loop {
294        if current & CROSSED != 0 {
295            return false;
296        }
297        match admission.compare_exchange_weak(
298            current,
299            current + 1,
300            Ordering::AcqRel,
301            Ordering::Relaxed,
302        ) {
303            Ok(_) => return true,
304            Err(actual) => current = actual,
305        }
306    }
307}

309/// End an admitted write once its bytes have reached the sink, or failed to.
310fn finish_write(admission: &AtomicU64) {
311    admission.fetch_sub(1, Ordering::Release);
312}
313
314/// For an admitted write whose charge went past the limit: set [`CROSSED`],
315/// then end this write's own admission. True only for the first such write,
316/// which is the crossing; every later one finds the bit already set.
317fn claim_crossing(admission: &AtomicU64) -> bool {
318    let before = admission.fetch_or(CROSSED, Ordering::AcqRel);
319    finish_write(admission);
320    before & CROSSED == 0
321}
322
323/// Admitted writes that have not finished.
324fn writes_in_progress(admission: &AtomicU64) -> u64 {
325    admission.load(Ordering::Acquire) & !CROSSED
326}

328/// Whether some process has already claimed the crossing, read the way
329/// [`exit_if_log_cap_already_crossed`] needs: with a read-modify-write that
330/// changes nothing (`fetch_or(0)`) and has acquire AND release ordering, not
331/// with a load. See that function for why the release half matters.
332fn crossing_already_claimed(admission: &AtomicU64) -> bool {
333    admission.fetch_or(0, Ordering::AcqRel) & CROSSED != 0
334}

351/// Exit 123 at once, writing nothing, if `--max-log-bytes` was already crossed.
352///
353/// Called by a process that is about to start a guest, right after it armed
354/// `PR_SET_PDEATHSIG`: the container init in `arm_container_init_guards`, and
355/// the `run --namespace-only` child in its `pre_exec`. It closes the window
356/// between the fork and that `prctl`: if the parent crossed the cap and exited
357/// 123 inside it, the death signal was armed after the parent was gone, so it
358/// will never arrive, and without this check the guest would start with
359/// nothing left to stop it.
360///
361/// WHY A READ-MODIFY-WRITE, AND WHY IT IS ENOUGH. Every cap-caused exit sets
362/// [`CROSSED`] first, with `fetch_or` in [`claim_crossing`], and every write to
363/// the admission word after it is created is a read-modify-write (the CAS in
364/// [`admit_write`], `fetch_sub` in [`finish_write`], `fetch_or` in
365/// [`claim_crossing`]). So this `fetch_or(0, AcqRel)` heads a release sequence
366/// containing every later change to the word. If it reads CROSSED clear, the
367/// crosser's `fetch_or(CROSSED, AcqRel)` comes later in the word's modification
368/// order, reads a value from that release sequence, and so synchronizes with
369/// this read-modify-write. The caller's `prctl` is sequenced before it, so the
370/// kernel's store of this process's death signal happens before everything
371/// the crosser does next, including its `_exit`. The exit of the parent thread
372/// (the crosser itself, or a sibling the group exit kills under `siglock`) runs
373/// `exit_notify`, which reads each child's death signal under `tasklist_lock`
374/// and sends it, so this process is killed. If it reads CROSSED set, it exits
375/// here, before any guest exists. A plain acquire load would not do: a store
376/// (the death signal) followed by a load here, against the crosser's
377/// read-modify-write followed by its read of the death signal, is the
378/// store-buffering shape, in which both sides can miss each other.
379///
380/// It closes only the part of the fork-to-`prctl` window that the cap opens. A
381/// parent that died for another reason inside that window is caught by
382/// `container::exit_if_parent_already_gone`; see `arm_parent_death_signal`.
383pub(crate) fn exit_if_log_cap_already_crossed() {
384    if let Some(budget) = INVOCATION_LOG_BUDGET.get()
385        && crossing_already_claimed(&budget.cells().admission)
386    {
387        // SAFETY: _exit has no preconditions and is async-signal-safe, so it
388        // is also sound in a `pre_exec` callback. Nothing is written: the
389        // crossing process owns the final message.
390        unsafe { libc::_exit(hermit::HERMIT_LOG_CAP_EXIT) }
391    }
392}

394/// Wait until no admitted write is in progress, polling with `nanosleep`, for
395/// at most `bound`. Returns whether none was left; a zero bound looks once.
396///
397/// A WRITER KILLED IN THE MIDDLE OF A WRITE NEVER FINISHES IT. Its count stays
398/// in the word for the rest of the invocation, so every later wait times out
399/// and the final message is omitted. That costs only the message: the exit
400/// status is still 123, and no line can follow a message that was never
401/// written.
402fn wait_for_writes_in_progress(admission: &AtomicU64, bound: Duration) -> bool {
403    let start = Instant::now();
404    loop {
405        if writes_in_progress(admission) == 0 {
406            return true;
407        }
408        let left = bound.saturating_sub(start.elapsed());
409        if left.is_zero() {
410            return false;
411        }
412        let step = left.min(LOG_CAP_DRAIN_POLL);
413        let pause = libc::timespec {
414            tv_sec: 0,
415            tv_nsec: libc::c_long::from(step.subsec_nanos()),
416        };
417        // SAFETY: nanosleep reads one initialized timespec; the remainder
418        // pointer may be null. An interrupted sleep just polls again.
419        unsafe { libc::nanosleep(&pause, std::ptr::null_mut()) };
420    }
421}

442/// The `--max-log-bytes` budget: a limit and a running total of the bytes
443/// hermit has offered to its public log sink.
444///
445/// THE TOTAL IS PROCESS-SHARED. Tracing is initialized after the container fork, and `run --verify` forks
446/// one container per run. A heap counter would be copied at each fork, so every
447/// run would get the full budget again; this MAP_SHARED cell, created in `main`
448/// before any fork, makes the cap a total for the invocation.
449#[derive(Clone, Debug)]
450pub struct LogBudget {
451    limit: u64,
452    cells: Arc<SharedBudgetCells>,
453    /// Rendered up front so the abort path allocates nothing: it can run on
454    /// any thread, with arbitrary locks held.
455    message: Arc<str>,
456    /// [`LOG_CAP_DRAIN_BOUND`], except in tests.
457    drain_bound: Duration,
458}
459
460/// Outcome of charging one write against a [`LogBudget`].
461#[derive(Debug, PartialEq, Eq)]
462enum Charge {
463    /// The write fits and is admitted; perform it, then call
464    /// [`LogBudget::finish`] once.
465    Within,
466    /// This write is the one that crossed the cap; abort the run. Its
467    /// admission has already ended.
468    Crossed,
469    /// A write already crossed the cap and is aborting; drop this one. Nothing
470    /// is left to finish.
471    AlreadyOver,
472}
473
474impl LogBudget {
475    pub fn new(limit: u64) -> io::Result<Self> {
476        // SAFETY: anonymous shared mapping, no fixed address and no backing fd.
477        let address = unsafe {
478            libc::mmap(
479                std::ptr::null_mut(),
480                mem::size_of::<BudgetCells>(),
481                libc::PROT_READ | libc::PROT_WRITE,
482                libc::MAP_SHARED | libc::MAP_ANONYMOUS,
483                -1,
484                0,
485            )
486        };
487        if address == libc::MAP_FAILED {
488            return Err(io::Error::last_os_error());
489        }
490        let address = NonNull::new(address.cast::<BudgetCells>())
491            .expect("mmap returned a non-null non-failure address");
492        // SAFETY: the fresh mapping is writable and suitably page-aligned.
493        unsafe {
494            address.as_ptr().write(BudgetCells {
495                spent: AtomicU64::new(0),
496                admission: AtomicU64::new(0),
497            })
498        };
499        Ok(Self {
500            limit,
501            cells: Arc::new(SharedBudgetCells { address }),
502            message: exceeded_message(limit).into(),
503            drain_bound: LOG_CAP_DRAIN_BOUND,
504        })
505    }
506
507    /// This budget with another drain bound, never above
508    /// [`LOG_CAP_DRAIN_BOUND_MAX`].
509    #[cfg(test)]
510    fn with_drain_bound(mut self, bound: Duration) -> Self {
511        self.drain_bound = bound.min(LOG_CAP_DRAIN_BOUND_MAX);
512        self
513    }
514
515    /// The final message printed when the cap fires.
516    pub fn exceeded_message(&self) -> &str {
517        &self.message
518    }
519
520    fn cells(&self) -> &BudgetCells {
521        // SAFETY: the Arc keeps the initialized mapping live in this process.
522        unsafe { self.cells.address.as_ref() }
523    }
524
525    /// Bytes charged so far, across every process sharing this budget.
526    #[cfg(test)]
527    fn spent(&self) -> u64 {
528        self.cells().spent.load(Ordering::Relaxed)
529    }
530
531    /// Whether `other` charges the same process-shared cells as `self`.
532    #[cfg(test)]
533    pub(crate) fn shares_counter_with(&self, other: &LogBudget) -> bool {
534        self.cells.address == other.cells.address
535    }
536
537    /// Charge `len` bytes: admit the write ([`admit_write`]), then account for
538    /// it. Two compare-exchange loops per write, each retried only when another
539    /// writer raced it: this sits on the hot path of every log line. A write
540    /// that would take the total past the limit claims the one-way [`CROSSED`]
541    /// bit; the single claimant is the crossing, every other writer is already
542    /// over. Refunds lower `spent` but never clear [`CROSSED`], so a crossing
543    /// is final.
544    fn charge(&self, len: u64) -> Charge {
545        if !admit_write(&self.cells().admission) {
546            return Charge::AlreadyOver;
547        }
548        self.account(len)
549    }
550
551    /// The second step of [`LogBudget::charge`], for a write already admitted.
552    ///
553    /// A TOTAL THAT WOULD OVERFLOW IS A CROSSING. `fetch_add` wrapped the stored
554    /// total, and with the largest accepted limit (`u64::MAX`) no sum could
555    /// ever compare above it, so that cap could never fire. The stored total
556    /// saturates instead of wrapping, and an addition with no `u64` result is
557    /// treated as past every limit.
558    fn account(&self, len: u64) -> Charge {
559        let cells = self.cells();
560        let mut before = cells.spent.load(Ordering::Relaxed);
561        while let Err(current) = cells.spent.compare_exchange_weak(
562            before,
563            before.saturating_add(len),
564            Ordering::Relaxed,
565            Ordering::Relaxed,
566        ) {
567            before = current;
568        }
569        match before.checked_add(len) {
570            Some(total) if total <= self.limit => Charge::Within,
571            _ if claim_crossing(&cells.admission) => Charge::Crossed,
572            _ => Charge::AlreadyOver,
573        }
574    }
575
576    /// Return bytes charged for a write that the sink accepted only in part, so
577    /// the caller's retry of the remainder is not counted twice.
578    fn refund(&self, len: u64) {
579        self.cells().spent.fetch_sub(len, Ordering::Relaxed);
580    }
581
582    /// End a write that [`LogBudget::charge`] admitted as [`Charge::Within`].
583    fn finish(&self) {
584        finish_write(&self.cells().admission);
585    }
586
587    /// For the crossing writer: wait, up to the drain bound, until no admitted
588    /// write is in progress in any process sharing this budget. True when the
589    /// final message may be written as the last line.
590    fn writes_drained(&self) -> bool {
591        wait_for_writes_in_progress(&self.cells().admission, self.drain_bound)
592    }
593
594    /// Admitted writes in progress, for tests.
595    #[cfg(test)]
596    fn in_progress(&self) -> u64 {
597        writes_in_progress(&self.cells().admission)
598    }
599}

633/// Counts the bytes offered to hermit's public log sink against a
634/// [`LogBudget`] and ends the process when the budget is exhausted.
635///
636/// This sits OUTSIDE [`BoundedWriter`]: the file bound silently discards
637/// beyond 1 GiB while the run continues, so counting after it would never see a
638/// runaway once the file stopped growing. The incident this exists for
639/// (2026-08-17) was hermit's own stderr at `--log=info` -- ~4.5 TB per process
640/// from the scheduler's per-poll banner -- which had no bound at all.
641///
642/// WHY `_exit` FROM INSIDE A WRITER. The writer runs wherever tracing does: the
643/// container init (PID 1 of the run's PID namespace, whose exit makes the
644/// kernel SIGKILL every guest in it), the outer hermit process (whose exit
645/// ends that init through its parent-death signal), or the non-blocking
646/// appender's worker thread in one of those processes. In each case ending the
647/// process is exactly how hermit's other deliberate stops tear the guest tree
648/// down (`record --record-timeout`, the container-init stop-signal handler).
649/// Where no PID namespace contains the guest -- `--no-namespace`, whose ptrace
650/// tracer binds the guest with PTRACE_O_EXITKILL only after it exists, and
651/// DBT -- the cap is refused before any guest starts
652/// (`RunOpts::log_cap_refusal`).
653/// `_exit` rather than `exit` because this can run on any thread, possibly
654/// while the subscriber's writer lock is held, and there is nothing left worth
655/// flushing: every sink here is unbuffered. The parent maps the status through
656/// `classify_container_result`.
657///
658/// THE FINAL MESSAGE NEVER WAITS FOR A SINK. It is written with
659/// [`write_without_waiting`], not through the inner writer or
660/// `RetryingStderr`: a blocking `write(2)` to a full pipe whose reader has
661/// stopped reading, or to a FIFO log sink, would hold the `_exit` -- and with
662/// it the guest teardown -- for as long as nobody drains the sink.
663///
664/// THE ONE WAIT IS BOUNDED, AND IT IS FOR OTHER WRITERS. Before the message the
665/// crossing writer waits at most [`LOG_CAP_DRAIN_BOUND`] for log writes that
666/// were admitted before the crossing to finish (see [`admit_write`]), so that
667/// none of them can put a line after the message. If one is still in progress
668/// then -- blocked on a full pipe, say -- the message is omitted, and the exit
669/// follows at once either way.
670pub struct CappedWriter<W: Write> {
671    inner: W,
672    budget: Option<LogBudget>,
673    /// Where the final message goes besides stderr: the descriptor under the
674    /// inner writer. `None` when the inner writer is stderr itself (so the
675    /// message is written once) or has no descriptor (an in-memory test sink).
676    final_message_fd: Option<RawFd>,
677    /// Whether the inner writer has already ended its output with its own
678    /// last line. [`BoundedWriter`] writes the truncation marker as the final
679    /// bytes of a truncated log, and the comparator recognizes a truncated log
680    /// only by that marker at end of file, so the final message must not be
681    /// appended after it.
682    inner_has_final_line: fn(&W) -> bool,
683}
684
685impl<W: Write> CappedWriter<W> {
686    /// Wrap a sink with no descriptor of its own; the final message goes to
687    /// stderr only. `None` disables the cap (no counting at all).
688    pub fn new(inner: W, budget: Option<LogBudget>) -> Self {
689        Self {
690            inner,
691            budget,
692            final_message_fd: None,
693            inner_has_final_line: |_| false,
694        }
695    }
696
697    fn exceeded(&mut self, budget: &LogBudget) -> ! {
698        // No write is admitted any more. Wait, for at most the drain bound,
699        // for the writes admitted before the crossing; a line one of them
700        // wrote after the final message would leave the message not last. If
701        // they do not finish in time, omit the message and exit 123 anyway.
702        if budget.writes_drained() {
703            self.write_final_message(budget);
704        }
705        // SAFETY: _exit has no preconditions; see the type-level note for why
706        // the immediate exit is the teardown.
707        unsafe { libc::_exit(hermit::HERMIT_LOG_CAP_EXIT) }
708    }
709
710    fn write_final_message(&mut self, budget: &LogBudget) {
711        let message = budget.exceeded_message().as_bytes();
712        if let Some(fd) = self.final_message_fd {
713            // Best effort: the log should say why it ends. Written to the
714            // descriptor directly, so it lands even when BoundedWriter is
715            // close to its bound (a few hundred bytes past it, once) -- but
716            // NOT once BoundedWriter has truncated: its marker must stay the
717            // final line of a truncated log, because that is the only thing
718            // `detcore::logdiff::log_was_truncated` accepts as truncation.
719            // Stderr still gets an attempt below, also without waiting, so
720            // delivery there is best effort too.
721            if !(self.inner_has_final_line)(&self.inner) {
722                write_without_waiting(fd, message);
723            }
724        }
725        write_without_waiting(libc::STDERR_FILENO, message);
726    }
727}

762/// Set by `main` when `--max-log-bytes` is in force; see [`bound_log_cap_exit`].
763static LOG_CAP_EXIT_BOUND_ENABLED: AtomicBool = AtomicBool::new(false);

765/// class line -- has been attempted. Both paths claim it with a swap, so it is

770/// How long the outer process may keep reporting after it learns that a run

774/// Let [`bound_log_cap_exit`] act in this process. Only `main` calls this, so
775/// unit tests that classify a 123 status never start the exit timer.

804/// `eprintln!` for a diagnostic that can run in a capped run before the cap
805/// or the 123 classification: the warnings and banners the outer process
806/// prints while it validates its arguments and prepares the guest.
807///
808/// While `--max-log-bytes` is in force
809/// ([`hermit::nonwaiting_write::diagnostics_must_not_wait`]) the formatted
810/// line, newline included, is written once with [`write_without_waiting`]
811/// and omitted when stderr cannot take it at once; a line longer than
812/// `PIPE_BUF` is cut there. Otherwise this is exactly the `eprintln!` it
813/// replaces, with the same arguments, so uncapped output is unchanged.
814macro_rules! diagnostic_eprintln {
815    ($($arg:tt)*) => {
816        if hermit::nonwaiting_write::diagnostics_must_not_wait() {
817            let mut line = format!($($arg)*);
818            line.push('\n');
819            hermit::nonwaiting_write::write_without_waiting(
820                libc::STDERR_FILENO,
821                line.as_bytes(),
822            );
823        } else {
824            eprintln!($($arg)*);
825        }
826    };
827}

830/// The process that classifies a container's 123 status must itself exit 123
831/// within a fixed bound, whatever its remaining diagnostics do: between the
832/// classification and `main`'s final report it can still print through
833/// blocking `eprintln!` calls (cleanup notices, verification notes) to a
834/// stderr nobody reads. This starts a thread that, after
835/// [`LOG_CAP_OUTER_GRACE`], attempts the class line once without waiting
836/// (unless the final report already did) and calls `_exit(123)`. A normal exit
837/// before then ends the thread with the process.
838///
839/// Two ways around that timer are closed here. If the thread cannot be
840/// created, nothing would bound the diagnostics that follow, so the process
841/// exits 123 at once. And from here on a panic exits 123 instead of
842/// unwinding to the panic status 101, because the run's class is already
843/// decided: `eprintln!`, for one, panics when stderr's reader has gone. A
844/// failed `analyze` or `bisect` trial does not panic; it returns its error to
845/// `main`, which reports a cap crossing by its class.
846pub(crate) fn bound_log_cap_exit() {
847    bound_log_cap_exit_with(|timer| {
848        std::thread::Builder::new()
849            .name("log-cap-exit".to_string())
850            .spawn(timer)
851            .map(drop)
852    });
853}
854
855/// [`bound_log_cap_exit`] with the timer thread's creation supplied, so a test
856/// can make it fail.
857fn bound_log_cap_exit_with(start_timer: impl FnOnce(Box<dyn FnOnce() + Send>) -> io::Result<()>) {
858    if !LOG_CAP_EXIT_BOUND_ENABLED.load(Ordering::Relaxed)
859        || LOG_CAP_EXIT_SCHEDULED.swap(true, Ordering::Relaxed)
860    {
861        return;
862    }
863    std::panic::set_hook(Box::new(|info| {
864        write_without_waiting(
865            libc::STDERR_FILENO,
866            format!("hermit: panic after the log cap ended the run: {info}\n").as_bytes(),
867        );
868        exit_with_log_cap_status()
869    }));
870    let timer: Box<dyn FnOnce() + Send> = Box::new(|| {
871        std::thread::sleep(LOG_CAP_OUTER_GRACE);
872        exit_with_log_cap_status()
873    });
874    if start_timer(timer).is_err() {
875        exit_with_log_cap_status()
876    }
877}

903impl<W: Write> Write for CappedWriter<W> {
904    fn write(&mut self, buf: &[u8]) -> io::Result<usize> {
905        let Some(budget) = &self.budget else {
906            return self.inner.write(buf);
907        };
908        match budget.charge(buf.len() as u64) {
909            Charge::Within => {}
910            Charge::Crossed => {
911                let budget = budget.clone();
912                self.exceeded(&budget)
913            }
914            // The crossing writer is already ending the process. Report the
915            // bytes consumed (as BoundedWriter does) rather than erroring.
916            Charge::AlreadyOver => return Ok(buf.len()),
917        }
918        // Ends the admission after the write, on every path out of it.
919        let _admitted = AdmittedWrite(budget);
920        let written = self.inner.write(buf).inspect_err(|_| {
921            budget.refund(buf.len() as u64);
922        })?;
923        if written < buf.len() {
924            budget.refund((buf.len() - written) as u64);
925        }
926        Ok(written)
927    }
928
929    fn flush(&mut self) -> io::Result<()> {
930        self.inner.flush()
931    }
932}

1038/// Make the next guest task use the same identity as the asynchronous logger.
1039///
1040/// The non-blocking file subscriber that `--log-file` used until
1041/// <https://github.com/rrnewton/hermit/issues/4427> (`file_subscriber`) started
1042/// a writer thread inside the fresh PID namespace. The namespace init is PID 1,
1043/// that worker consumed PID 2, and the root guest consequently started at
1044/// [`detcore::ROOT_DETPID`] (PID 3). The synchronous file and stderr
1045/// subscribers have no worker, but their guests still need the same identity
1046/// baseline.
1047///
1048/// Creating and joining a dummy host thread to consume PID 2 is not equivalent:
1049/// the thread participates in the parent's signal/timer state and made KVM's
1050/// `setitimer` determinism case die from SIGALRM. A forked process has separate
1051/// signal and interval-timer state, and it exits before any backend starts.
1052/// Outside a PID-namespace init (notably `--no-namespace`) this is a deliberate
1053/// no-op; Hermit must not create a process merely to alter the host namespace.
1054pub(super) fn align_tracing_pid_baseline(backend: Backend) {
1055    // KVM supplies the deterministic root identity explicitly rather than
1056    // deriving it from a Linux task in this namespace. Allocating PID 2 for
1057    // that backend is both unnecessary and harmful: it makes the setitimer
1058    // sandbox die from SIGALRM before a comparison can run.
1059    if backend == Backend::Kvm {
1060        return;
1061    }
1062
1063    align_namespace_pid_baseline();
1064}

1066/// Reserve the normal Linux tracing task slot before an external physical
1067/// runtime starts coordinator tasks in the fresh PID namespace.
1068pub(super) fn align_namespace_pid_baseline() {
1069    // SAFETY: getpid has no preconditions and cannot fail.
1070    if unsafe { libc::getpid() } != 1 {
1071        return;
1072    }
1073
1074    // SAFETY: the child returns through `_exit` immediately and touches no Rust
1075    // allocation, lock, or destructor inherited across fork. POSIX specifies
1076    // that interval timers are not inherited by a fork child.
1077    let child = unsafe { libc::fork() };
1078    assert!(
1079        child >= 0,
1080        "failed to reserve the tracing PID slot: {}",
1081        io::Error::last_os_error()
1082    );
1083    if child == 0 {
1084        unsafe { libc::_exit(0) };
1085    }
1086
1087    let mut status = 0;
1088    loop {
1089        let waited = unsafe { libc::waitpid(child, &mut status, 0) };
1090        if waited == child {
1091            break;
1092        }
1093        let error = io::Error::last_os_error();
1094        if waited < 0 && error.kind() == io::ErrorKind::Interrupted {
1095            continue;
1096        }
1097        panic!("failed to reap tracing PID-slot process {child}: {error}");
1098    }
1099    assert!(
1100        libc::WIFEXITED(status) && libc::WEXITSTATUS(status) == 0,
1101        "tracing PID-slot process {child} exited unexpectedly: {status:#x}"
1102    );
1103    assert_eq!(
1104        child,
1105        detcore::ROOT_DETPID.as_raw() - 1,
1106        "fresh PID namespace did not reserve the expected tracing slot"
1107    );
1108}

1112/// NOTE: Writes to stderr are unbuffered, so this may be slow.
1113pub fn stderr_subscriber(level: Option<LevelFilter>, budget: Option<LogBudget>) -> impl Subscriber {
1114    let level = level.unwrap_or(DEFAULT_TRACE_LEVEL);
1115
1116    let filter = env_filter(level);
1117    tracing_subscriber::fmt()
1118        .with_env_filter(filter)
1119        // NOT `io::stderr`: a guest can set O_NONBLOCK on the inherited fd 2,
1120        // after which a full pipe makes these writes fail with EAGAIN. The fmt
1121        // layer discards the write error, so log lines would vanish with no
1122        // marker at all. `RetryingStderr` waits for the reader instead of
1123        // dropping, and does not alter the flag the guest set. `CappedWriter`
1124        // charges each event against `--max-log-bytes`, when given.
1125        .with_writer(move || CappedWriter::stderr(budget.clone()))
1126        .with_ansi(stderr().is_terminal())
1127        .finish()
1128}

1130/// Where [`emit_before_tracing`] writes: the run's `--log-file`, or stderr.

1136/// Stderr for a run under `--max-log-bytes`: each event is written once,
1137/// without waiting, and dropped when stderr cannot take it at once (see
1138/// [`write_stderr_diagnostic`]).

1141impl Write for NonWaitingStderr {

1143        write_without_waiting(libc::STDERR_FILENO, buf);
1144        Ok(buf.len())
1145    }

1147    fn flush(&mut self) -> io::Result<()> {
1148        Ok(())
1149    }

1152/// Emit tracing events from `emit` in a process that has not installed its
1153/// global subscriber yet (the outer `hermit` process, which installs one only
1154/// inside the container). They go through a scoped, synchronous subscriber
1155/// with the run's level and format, to the run's own destination, so they
1156/// read and filter exactly like the run's later events. Neither subscriber
1157/// starts a thread, and the scope ends before the global one is installed.
1158pub(crate) fn emit_before_tracing(level: Option<LevelFilter>, log: EarlyLog, emit: impl FnOnce()) {
1159    let level = level.unwrap_or(DEFAULT_TRACE_LEVEL);
1160    match log {
1161        EarlyLog::File(writer) => {
1162            tracing::subscriber::with_default(sync_file_subscriber(level, writer), emit)
1163        }
1164        EarlyLog::Stderr(_) if hermit::nonwaiting_write::diagnostics_must_not_wait() => {
1165            let subscriber = tracing_subscriber::fmt()
1166                .with_env_filter(env_filter(level))
1167                .with_writer(|| NonWaitingStderr)
1168                .with_ansi(false)
1169                .finish();
1170            tracing::subscriber::with_default(subscriber, emit)
1171        }
1172        EarlyLog::Stderr(budget) => {
1173            tracing::subscriber::with_default(stderr_subscriber(Some(level), budget), emit)
1174        }
1175    }
1176}

1885    /// The child: writers on stderr, each through its own CappedWriter on one
1886    /// budget, until one of them crosses and ends the process.
1887    fn race_writers_to_the_crossing(drain_bound: Duration) -> ! {
1888        let budget = LogBudget::new(LOG_CAP_STRESS_LIMIT)
1889            .unwrap()
1890            .with_drain_bound(drain_bound);
1891        let start = Arc::new(std::sync::Barrier::new(LOG_CAP_STRESS_WRITERS));
1892        let writers: Vec<_> = (0..LOG_CAP_STRESS_WRITERS)
1893            .map(|writer| {
1894                let budget = budget.clone();
1895                let start = Arc::clone(&start);
1896                std::thread::spawn(move || {
1897                    let mut sink = CappedWriter::stderr(Some(budget));
1898                    start.wait();
1899                    for line in 0u64..=u64::MAX {
1900                        // One write per line, so each line reaches the pipe whole.
1901                        let _ = sink.write_all(format!("writer {writer} line {line}\n").as_bytes());
1902                    }
1903                })
1904            })
1905            .collect();
1906        for writer in writers {
1907            let _ = writer.join();
1908        }
1909        // Unreachable: the crossing write ends the process with exit 123.
1910        // SAFETY: _exit has no preconditions.
1911        unsafe { libc::_exit(1) }
1912    }
