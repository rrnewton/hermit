3//! The discriminator for a container child whose scheduler refuses a replayed
4//! schedule after a late precise timer event and exits it 122 instead of
5//! finishing the run.
6//!
7//! A run that finishes after a late precise timer event is refused by name as
8//! a skid overshoot: the child takes Reverie's late-event count at the end of
9//! the run and reports it through its result channel (`SkidOvershootReport`
10//! in hermit-cli's lib.rs, `FailureKind::SkidOvershoot`), and test harnesses
11//! retry that refusal. A replay (`--replay-preemptions-from`,
12//! `--replay-schedule-from`) that the late event pushes past a recorded
13//! callback checkpoint never gets there: the scheduler refuses the checkpoint
14//! and exits the child 122 at once (`detcore::scheduler::exit_on_replay_refusal`),
15//! and 122 alone is every refusal's status. So just before that exit the
16//! child writes the late-event count here, through the scheduler's replay
17//! refusal hook (`detcore::set_replay_refusal_exit_hook`), and the parent
18//! reads it when it classifies the 122 (`classify_container_result`): the
19//! refusal is then the late event's skid-overshoot refusal, as on a run that
20//! finished.
21//!
22//! The note is a `MAP_SHARED` cell, as the virtual-timeout note is
23//! (`terminal_outcome_note`). A run that replays a schedule arms it
24//! (`RunOpts::launch_in_container`) before it clones the container child,
25//! which inherits the mapping and the hook; a guest loses both at its
26//! `execve`. Every container launch clears a note already mapped
27//! (`owned_container::run`), so a note always belongs to the latest child.
28
29use std::io;
30use std::ptr::NonNull;
31use std::sync::OnceLock;
32use std::sync::atomic::AtomicU64;
33use std::sync::atomic::Ordering;
34
35use hermit::Context;
36use hermit::Error;
37
38/// One shared note: the late precise timer events a child recorded before its
39/// replay refusal, or zero for none.
40struct Note {
41    late_events: NonNull<AtomicU64>,
42}
43
44// SAFETY: an initialized atomic in a MAP_SHARED mapping that is never
45// unmapped; the atomic supplies synchronization.
46unsafe impl Send for Note {}
47unsafe impl Sync for Note {}
48
49impl Note {
50    fn map() -> io::Result<Self> {
51        // SAFETY: anonymous shared mapping, no fixed address and no backing fd.
52        let address = unsafe {
53            libc::mmap(
54                std::ptr::null_mut(),
55                std::mem::size_of::<AtomicU64>(),
56                libc::PROT_READ | libc::PROT_WRITE,
57                libc::MAP_SHARED | libc::MAP_ANONYMOUS,
58                -1,
59                0,
60            )
61        };
62        if address == libc::MAP_FAILED {
63            return Err(io::Error::last_os_error());
64        }
65        let late_events = NonNull::new(address.cast::<AtomicU64>())
66            .expect("mmap returned a non-null non-failure address");
67        // SAFETY: the fresh mapping is writable and suitably page-aligned.
68        unsafe { late_events.as_ptr().write(AtomicU64::new(0)) };
69        Ok(Self { late_events })
70    }
71
72    fn cell(&self) -> &AtomicU64 {
73        // SAFETY: initialized in `map` and never unmapped.
74        unsafe { self.late_events.as_ref() }
75    }
76
77    fn record(&self, late_events: u64) {
78        self.cell().store(late_events, Ordering::Release);
79    }
80
81    fn take(&self) -> Option<u64> {
82        match self.cell().swap(0, Ordering::Acquire) {
83            0 => None,
84            late_events => Some(late_events),
85        }
86    }
87}
88
89static NOTE: OnceLock<Note> = OnceLock::new();
90
91/// The scheduler's replay refusal hook in a container child: leave the count
92/// for the parent. An atomic store, so it is safe wherever the scheduler
93/// exits from.
94fn record_in_shared_note(late_events: u64) {
95    if let Some(note) = NOTE.get() {
96        note.record(late_events);
97    }
98}
99
100/// Before a container child of a run that replays a schedule is cloned: map
101/// the note once and install the hook that writes it.
102pub(super) fn arm() -> Result<(), Error> {
103    if NOTE.get().is_none() {
104        let note = Note::map().context("mapping the replay refusal note")?;
105        let _ = NOTE.set(note);
106        detcore::set_replay_refusal_exit_hook(Some(record_in_shared_note));
107    }
108    clear();
109    Ok(())
110}
111
112/// Before any container child is cloned: forget what an earlier child left,
113/// so a refusal it never reported cannot be read as this child's. Maps
114/// nothing.
115pub(super) fn clear() {
116    if let Some(note) = NOTE.get() {
117        note.take();
118    }
119}
120
121/// The late precise timer events the latest container child noted before its
122/// replay refusal, if any; reading it clears it. `None` when no child was
123/// armed in this process.
124pub(super) fn take() -> Option<u64> {
125    NOTE.get()?.take()
126}

128#[cfg(test)]
129mod tests {
130    use super::*;
131
132    /// A count written by a forked child just before it exits 122, as a
133    /// scheduler refusing a replay after a late timer event exits, reaches the
134    /// parent once. A private note keeps this away from the process's own,
135    /// which other tests' classifications read.
136    #[test]
137    fn a_late_event_count_noted_by_an_exiting_child_reaches_its_parent_once() {
138        let note = Note::map().unwrap();
139        // SAFETY: the child does only an atomic store and `_exit`, both
140        // async-signal-safe, so forking a multi-threaded test process is sound.
141        let child = unsafe { libc::fork() };
142        assert!(child >= 0, "fork: {}", io::Error::last_os_error());
143        if child == 0 {
144            note.record(3);
145            // SAFETY: async-signal-safe; runs no destructors in the child.
146            unsafe { libc::_exit(detcore_model::HERMIT_POLICY_REFUSAL_EXIT) };
147        }
148        let mut status = 0;
149        // SAFETY: waits for the child forked above.
150        assert_eq!(unsafe { libc::waitpid(child, &mut status, 0) }, child);
151        assert!(libc::WIFEXITED(status));
152        assert_eq!(
153            libc::WEXITSTATUS(status),
154            detcore_model::HERMIT_POLICY_REFUSAL_EXIT
155        );
156        assert_eq!(note.take(), Some(3));
157        assert_eq!(note.take(), None, "read once");
158    }
159}
