9//! Widely-shared type definitions.
10
11use serde::Deserialize;
12use serde::Serialize;
13
14/// Child population addressed by a wait syscall whose exit readiness Detcore can model.
15#[derive(Clone, Copy, Debug, Eq, PartialEq, Hash, Serialize, Deserialize)]
16pub enum ChildWaitSelector {
17    /// One direct child process.
18    Exact(DetPid),
19    /// Any direct child process.
20    Any,
21    /// Any direct child process currently in this process group.
22    ProcessGroup(DetPid),
23}
24
25/// Linux's clone-child filtering for terminal wait events.
26#[derive(Clone, Copy, Debug, Eq, PartialEq, Hash, Serialize, Deserialize)]
27pub enum ChildWaitExitClass {
28    /// Children whose clone exit signal is SIGCHLD (the normal wait population).
29    Sigchld,
30    /// Children whose clone exit signal is zero or a signal other than SIGCHLD.
31    Clone,
32    /// Both normal and clone children (`__WALL`).
33    Any,
34}
35
36/// Scheduler-owned selection criteria for a terminal child wait.
37#[derive(Clone, Copy, Debug, Eq, PartialEq, Hash, Serialize, Deserialize)]
38pub struct ChildWaitSpec {
39    /// PID or process-group population selected by the syscall.
40    pub selector: ChildWaitSelector,
41    /// Restrict to children owned by this exact task (`__WNOTHREAD`).
42    pub owner: Option<DetTid>,
43    /// Filter normal versus clone children.
44    pub exit_class: ChildWaitExitClass,
45}
46
47/// Scheduler-owned lifecycle state for an exact child-process wait.
48#[derive(Clone, Copy, Debug, Eq, PartialEq, Serialize, Deserialize)]
49pub enum ExactChildWaitState {
50    /// The requested process is not a known direct child of the caller.
51    Unknown,
52    /// At least one thread in the child process remains logically live.
53    Running,
54    /// Detcore observed logical exit, but this backend cannot report physical waitability.
55    LogicallyExited,
56    /// The backend has not yet reported the child's final kernel exit status.
57    PhysicalExitPending,
58    /// The backend has reported the child's final kernel exit status.
59    PhysicallyExited,
60}
