8//! Native descriptor publication policy. These classes describe an actual
9//! application attempt, never evidence that a matched Tool handler ran it.
10
11use reverie::syscalls::Errno;
12use reverie::syscalls::Syscall;
13use reverie::syscalls::SyscallInfo;
14
15/// The membership evidence required after a physically issued application call.
16#[derive(Clone, Copy, Debug, Eq, PartialEq)]
17pub(super) enum AcquisitionClass {
18    /// Kernel construction creates fresh descriptions, including partial
19    /// publication before an output error. No existing OFD state is imported.
20    Fresh,
21    /// One installed descriptor is the successful native return value. Errors
22    /// from these syscall implementations precede publication of that result.
23    ReturnedFd,
24    /// Returned payload/message counts do not enumerate installed descriptors.
25    /// A native error can follow ancillary installation and output faults.
26    Received,
27    /// A supported ioctl can install descriptors. Its command is not an OFD
28    /// identity proof, and its generic return/error is not a membership list.
29    IoctlImport,
30}
31
32impl AcquisitionClass {
33    /// This answers only whether the syscall's ordinary descriptor publication
34    /// is absent. It makes no claim about arbitrary driver-internal effects.
35    /// In particular, Err(EFAULT) from recvmsg is NOT a no-install proof:
36    /// Linux net/socket.c ____sys_recvmsg copies the address/flags/controllen
37    /// after sock_recvmsg has already delivered SCM_RIGHTS.
38    pub(super) fn native_error_proves_no_install(self, result: Result<i64, Errno>) -> bool {
39        self == Self::ReturnedFd && result.is_err()
40    }
41
42    pub(super) fn needs_membership(self) -> bool {
43        self != Self::Fresh
44    }
45}
46
47/// Only native ABI calls reach this classifier. Unknown driver operations are
48/// not certified safe by returning None: they retain the separately accepted
49/// S5 scope and their direct-reference policy. In particular this is not an
50/// all-ioctl permanent taint.
51pub(super) fn classify(call: Syscall) -> Option<AcquisitionClass> {
52    use AcquisitionClass::*;
53    let (_, args) = call.into_parts();
54    match call.name() {
55        // The kernel publishes files allocated by these operations, not the
56        // listening socket's description or whichever FD a prior probe saw.
57        "socket" | "socketpair" | "pipe" | "pipe2" | "accept" | "accept4" => Some(Fresh),
58        // Linux fs/open.c do_sys_openat2 and fs/file.c dup/f_dupfd/ksys_dup3:
59        // the descriptor is returned only after publication; ordinary error
60        // paths do not publish it. pidfd_getfd uses receive_fd with no user
61        // output pointer and returns that same published FD.
62        "open" | "openat" | "openat2" | "creat" | "open_by_handle_at" | "dup" | "dup3"
63        | "pidfd_getfd" | "memfd_create" | "eventfd" | "eventfd2" | "timerfd_create"
64        | "inotify_init" | "inotify_init1" | "epoll_create" | "epoll_create1" | "userfaultfd"
65        | "pidfd_open" => Some(ReturnedFd),
66        // dup2(old, old) only validates that existing slot; it does not acquire
67        // or replace a description. Its actual native result is unchanged.
68        "dup2" if args.arg0 as i32 != args.arg1 as i32 => Some(ReturnedFd),
69        "fcntl" if matches!(args.arg1 as i32, libc::F_DUPFD | libc::F_DUPFD_CLOEXEC) => {
70            Some(ReturnedFd)
71        }
72        // An existing signalfd is reconfigured without installing another FD.
73        "signalfd" | "signalfd4" if (args.arg0 as i32) < 0 => Some(ReturnedFd),
74        // Negative FDs cannot reach socket receive. Zero vlen cannot deliver
75        // any message. Zero *payload bytes* from recvmsg is not such a proof.
76        "recvmsg" if (args.arg0 as i32) >= 0 => Some(Received),
77        "recvmmsg" if (args.arg0 as i32) >= 0 && args.arg2 as u32 != 0 => Some(Received),
78        "ioctl" if (args.arg0 as i32) >= 0 && descriptor_ioctl(args.arg1 as u32) => {
79            Some(IoctlImport)
80        }
81        _ => None,
82    }
83}
84
85fn descriptor_ioctl(command: u32) -> bool {
86    // Full native ioctl command, not the low command byte. These are Linux
87    // binder.h BINDER_WRITE_READ, asm-generic/ioctls.h TIOCGPTPEER, and
88    // nsfs.h NS_GET_USERNS / NS_GET_PARENT and the extensible NEXT/PREV
89    // namespace commands. Actual object classification and
90    // complete installed-membership proof remain required at the native stop.
91    matches!(command, 0xc030_6201 | 0x5441 | 0xb701 | 0xb702)
92        || (matches!(command & !0x3fff_0000, 0x8000_b70b | 0x8000_b70c)
93            && (command >> 16) & 0x3fff >= 16)
94}

96#[cfg(test)]
97mod tests {
98    use reverie::syscalls::SyscallArgs;
99    use reverie::syscalls::Sysno;
100
101    use super::*;
102
103    fn call(number: Sysno, args: [usize; 6]) -> Syscall {
104        Syscall::from_raw(
105            number,
106            SyscallArgs::new(args[0], args[1], args[2], args[3], args[4], args[5]),
107        )
108    }
109
110    #[test]
111    fn native_receive_error_and_zero_payload_do_not_prove_empty_membership() {
112        let receive = classify(call(Sysno::recvmsg, [7, 0, 0, 0, 0, 0])).unwrap();
113        assert_eq!(receive, AcquisitionClass::Received);
114        for result in [Ok(0), Ok(1), Err(Errno::EFAULT), Err(Errno::EAGAIN)] {
115            assert!(!receive.native_error_proves_no_install(result));
116        }
117        assert_eq!(
118            classify(call(Sysno::recvmsg, [usize::MAX, 0, 0, 0, 0, 0])),
119            None
120        );
121        assert_eq!(classify(call(Sysno::recvmmsg, [7, 1, 0, 0, 0, 0])), None);
122        let open = classify(call(Sysno::openat, [0; 6])).unwrap();
123        assert!(open.native_error_proves_no_install(Err(Errno::EFAULT)));
124        assert!(!open.native_error_proves_no_install(Ok(7)));
125    }
126
127    #[test]
128    fn acquisition_commands_do_not_taint_unrelated_ioctl_or_existing_signalfd() {
129        for command in [
130            0xc030_6201,
131            0x5441,
132            0xb701,
133            0xb702,
134            0x8010_b70b,
135            0x8020_b70c,
136        ] {
137            assert_eq!(
138                classify(call(Sysno::ioctl, [7, command, 0, 0, 0, 0])),
139                Some(AcquisitionClass::IoctlImport)
140            );
141        }
142        for command in [
143            libc::FIONBIO as usize,
144            libc::FIONREAD as usize,
145            0x6201,
146            0xc020_6201,
147            0x800f_b70b,
148            0x4010_b70b,
149            0x8010_b70a,
150        ] {
151            assert_eq!(classify(call(Sysno::ioctl, [7, command, 0, 0, 0, 0])), None);
152        }
153        assert_eq!(classify(call(Sysno::signalfd4, [7, 0, 8, 0, 0, 0])), None);
154        assert_eq!(
155            classify(call(Sysno::signalfd4, [usize::MAX, 0, 8, 0, 0, 0])),
156            Some(AcquisitionClass::ReturnedFd)
157        );
158    }
159
160    #[test]
161    fn native_fresh_results_keep_the_kernel_contract_without_listener_snapshot() {
162        for number in [
163            Sysno::socket,
164            Sysno::socketpair,
165            Sysno::pipe2,
166            Sysno::accept4,
167        ] {
168            let class = classify(call(number, [0; 6])).unwrap();
169            assert_eq!(class, AcquisitionClass::Fresh);
170            assert!(!class.needs_membership());
171        }
172        assert!(
173            classify(call(Sysno::openat, [0; 6]))
174                .unwrap()
175                .needs_membership()
176        );
177        assert!(
178            classify(call(Sysno::recvmsg, [7, 0, 0, 0, 0, 0]))
179                .unwrap()
180                .needs_membership()
181        );
182    }
