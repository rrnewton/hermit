9//! Detcore's own copy of each guest's signal dispositions and masks.
10//!
11//! Audit step A5 (ai_docs/2026-10-16-a5-signal-delivery-design.md in the
12//! dev-hermit parent; <https://github.com/rrnewton/hermit/issues/4485>) moves
13//! Linux signal policy into Detcore, so that no Detcore decision depends on
14//! reading the host's `/proc/<pid>/task/<tid>/status` for a guest thread.
15//! That read is the guest's real state only on the ptrace backend.
16//!
17//! This module is the state and its transitions, as Linux performs them. It
18//! holds no guest handle and makes no syscall: callers apply a transition
19//! once the guest operation it describes has committed. Step 1 keeps the
20//! state in shadow mode: it is seeded and compared with the kernel's only
21//! where the kernel's signal state is the guest's own (ptrace, serialized
22//! threads), and no decision reads it yet. Other backends need to report a
23//! first image's inherited state first (design step 3).
24//!
25//! What is kept, and where Linux keeps the same thing:
26//! - [`Dispositions`]: the 64 actions of one `struct sighand_struct`. Every
27//!   task cloned with `CLONE_SIGHAND` shares one table, so a thread state
28//!   holds it behind a shared reference, as it holds the descriptor table
29//!   that `CLONE_FILES` shares. `CLONE_SIGHAND` requires `CLONE_VM`, so the
30//!   tasks sharing a table also share the memory that reference lives in, on
31//!   every backend.
32//! - [`ThreadSignals`]: a task's `blocked` mask, the mask an interrupted
33//!   temporary-mask wait puts back, and the masks its open handler frames
34//!   will restore.
35//!
36//! Pending signals are not here: Detcore's existing queue owners keep them.
37

45/// A kernel signal set: bit `n - 1` is signal `n`, as `/proc` and
46/// `rt_sigprocmask` use it.
47pub(crate) type Sigset = u64;
48
49/// The highest signal number on x86-64 Linux (`_NSIG`).
50pub(crate) const MAX_SIGNAL: i32 = 64;
51
52const SIG_DFL: u64 = 0;
53const SIG_IGN: u64 = 1;
54
55/// `sa_flags` bits the kernel keeps and reports back on x86-64: Linux's
56/// `UAPI_SA_FLAGS` (`include/linux/signal_types.h`). `do_sigaction` clears
57/// every other bit of a new action before storing it, so userspace can detect
58/// unsupported flags, for example `SA_UNSUPPORTED`.
59const UAPI_SA_FLAGS: u64 = sa_flag(libc::SA_NOCLDSTOP)
60    | sa_flag(libc::SA_NOCLDWAIT)
61    | sa_flag(libc::SA_SIGINFO)
62    | sa_flag(libc::SA_ONSTACK)
63    | sa_flag(libc::SA_RESTART)
64    | sa_flag(libc::SA_NODEFER)
65    | sa_flag(libc::SA_RESETHAND)
66    | sa_flag(SA_EXPOSE_TAGBITS)
67    | sa_flag(SA_RESTORER);
68
69/// An `sa_flags` bit as the kernel's `unsigned long` holds it. The libc crate
70/// declares the bits as `c_int`, so `SA_RESETHAND` (0x80000000) is negative
71/// and a plain `as u64` would set the upper 32 bits as well.
72const fn sa_flag(flag: libc::c_int) -> u64 {
73    flag as u32 as u64
74}
75
76/// Not exported by the libc crate for x86-64.
77const SA_EXPOSE_TAGBITS: libc::c_int = 0x0000_0800;
78
79/// Not exported by the libc crate for x86-64 (`asm/signal.h`).
80const SA_RESTORER: libc::c_int = 0x0400_0000;
81
82/// The bit of `signal` in a [`Sigset`], or 0 outside `1..=64`.
83pub(crate) fn sigbit(signal: i32) -> Sigset {
84    if (1..=MAX_SIGNAL).contains(&signal) {
85        1 << (signal - 1)
86    } else {
87        0
88    }
89}
90
91/// `clone3`'s flag to reset caught signals in the child (Linux 5.5); not
92/// exported by the libc crate.
93const CLONE_CLEAR_SIGHAND: u64 = 0x1_0000_0000;
94
95/// The most open handler frames followed per thread. A handler left by
96/// `siglongjmp`, which never returns, leaves its frame open, and a handler
97/// whose return changes no mask can never be recognised as returned; beyond
98/// this many, the copy is given up as unknown rather than guessing which frame
99/// went.
100pub(crate) const MAX_OPEN_FRAMES: usize = 32;
101
102/// SIGKILL and SIGSTOP: never blocked, never caught, never ignored.
103const UNBLOCKABLE: Sigset = (1 << (libc::SIGKILL - 1)) | (1 << (libc::SIGSTOP - 1));
104
105/// One signal's action, as `struct kernel_sigaction` lays it out.
106#[derive(Clone, Copy, Debug, Default, Eq, PartialEq, Serialize, Deserialize)]
107pub(crate) struct Action {
108    /// `SIG_DFL` (0), `SIG_IGN` (1), or the handler's address.
109    pub(crate) handler: u64,
110    /// `sa_flags`, limited to [`UAPI_SA_FLAGS`].
111    pub(crate) flags: u64,
112    /// `sa_restorer`.
113    pub(crate) restorer: u64,
114    /// `sa_mask`, without SIGKILL and SIGSTOP.
115    pub(crate) mask: Sigset,
116}
117
118impl Action {
119    /// The action as `do_sigaction` stores it.
120    pub(crate) fn normalized(self) -> Self {
121        Action {
122            flags: self.flags & UAPI_SA_FLAGS,
123            mask: self.mask & !UNBLOCKABLE,
124            ..self
125        }
126    }
127
128    fn is_caught(&self) -> bool {
129        self.handler != SIG_DFL && self.handler != SIG_IGN
130    }
131}

133/// The actions of one disposition domain, indexed by signal number - 1.
134/// Always 64 entries (a `Vec` only because serde does not derive arrays
135/// that long).
136#[derive(Clone, Debug, Eq, PartialEq, Serialize, Deserialize)]
137#[serde(try_from = "Vec<Action>", into = "Vec<Action>")]
138pub(crate) struct Dispositions {
139    actions: Vec<Action>,
140}
141
142impl From<Dispositions> for Vec<Action> {
143    fn from(dispositions: Dispositions) -> Self {
144        dispositions.actions
145    }
146}
147
148impl TryFrom<Vec<Action>> for Dispositions {
149    type Error = String;
150
151    fn try_from(actions: Vec<Action>) -> Result<Self, Self::Error> {
152        if actions.len() == MAX_SIGNAL as usize {
153            Ok(Dispositions { actions })
154        } else {
155            Err(format!(
156                "a signal action table has {} entries, not {MAX_SIGNAL}",
157                actions.len()
158            ))
159        }
160    }
161}
162
163impl Dispositions {
164    /// A freshly executed image's actions: after `execve` a signal is either
165    /// ignored or at its default, so the ignored set alone describes them.
166    pub(crate) fn after_exec(ignored: Sigset) -> Self {
167        let mut actions = vec![Action::default(); MAX_SIGNAL as usize];
168        for (index, action) in actions.iter_mut().enumerate() {
169            if ignored & (1 << index) & !UNBLOCKABLE != 0 {
170                action.handler = SIG_IGN;
171            }
172        }
173        Dispositions { actions }
174    }
175
176    fn index(signal: i32) -> Result<usize, Errno> {
177        if (1..=MAX_SIGNAL).contains(&signal) {
178            Ok((signal - 1) as usize)
179        } else {
180            Err(Errno::EINVAL)
181        }
182    }
183
184    /// `signal`'s current action.
185    #[cfg(test)]
186    pub(crate) fn action(&self, signal: i32) -> Result<Action, Errno> {
187        Ok(self.actions[Self::index(signal)?])
188    }
189
190    /// `rt_sigaction(signal, new, &old)` after the size check, which the
191    /// caller makes first, as the kernel does. Returns the old action, which
192    /// is what the kernel copies out. SIGKILL and SIGSTOP can be queried but
193    /// not changed.
194    pub(crate) fn sigaction(&mut self, signal: i32, new: Option<Action>) -> Result<Action, Errno> {
195        let index = Self::index(signal)?;
196        if new.is_some() && sigbit(signal) & UNBLOCKABLE != 0 {
197            return Err(Errno::EINVAL);
198        }
199        let old = self.actions[index];
200        if let Some(new) = new {
201            self.actions[index] = new.normalized();
202        }
203        Ok(old)
204    }
205
206    /// The action used to deliver `signal`, chosen when the signal is
207    /// dequeued for delivery. `SA_RESETHAND` puts the handler back to
208    /// `SIG_DFL` at this point, before the frame is built, so a frame that
209    /// cannot be written still leaves the default installed (Linux
210    /// `get_signal`).
211    pub(crate) fn select_for_delivery(&mut self, signal: i32) -> Result<Action, Errno> {
212        let index = Self::index(signal)?;
213        let action = self.actions[index];
214        if action.is_caught() && action.flags & sa_flag(libc::SA_RESETHAND) != 0 {
215            self.actions[index].handler = SIG_DFL;
216        }
217        Ok(action)
218    }
219
220    /// `execve`'s reset (Linux `flush_signal_handlers` with
221    /// `force_default` false): a caught signal goes back to its default, an
222    /// ignored one stays ignored, and every action loses its flags, restorer
223    /// and mask.
224    pub(crate) fn reset_for_exec(&mut self) {
225        for action in &mut self.actions {
226            let handler = if action.handler == SIG_IGN {
227                SIG_IGN
228            } else {
229                SIG_DFL
230            };
231            *action = Action {
232                handler,
233                ..Action::default()
234            };
235        }
236    }
237
238    /// The signals whose action is `SIG_IGN`, as `/proc`'s `SigIgn`.
239    pub(crate) fn ignored(&self) -> Sigset {
240        self.select(|action| action.handler == SIG_IGN)
241    }
242
243    /// The signals with a handler installed, as `/proc`'s `SigCgt`.
244    pub(crate) fn caught(&self) -> Sigset {
245        self.select(Action::is_caught)
246    }
247
248    fn select(&self, keep: impl Fn(&Action) -> bool) -> Sigset {
249        self.actions
250            .iter()
251            .enumerate()
252            .filter(|(_, action)| keep(action))
253            .fold(0, |set, (index, _)| set | (1 << index))
254    }
255}

257/// One thread's masks.
258#[derive(Clone, Debug, Default, Eq, PartialEq, Serialize, Deserialize)]
259pub(crate) struct ThreadSignals {
260    /// The signals blocked now (Linux `task->blocked`).
261    pub(crate) blocked: Sigset,
262    /// The mask to put back once a temporary-mask wait (`rt_sigsuspend`, or
263    /// `ppoll`, `pselect6` or `epoll_pwait` given a mask) that a signal
264    /// interrupted returns to the guest: Linux's `saved_sigmask` while
265    /// `TIF_RESTORE_SIGMASK` is set. A handler delivered first saves it in
266    /// its frame instead.
267    #[serde(default)]
268    pub(crate) saved: Option<Sigset>,
269    /// The temporary mask of the wait this thread is in (`rt_sigsuspend`, or
270    /// `ppoll`, `pselect6`, `epoll_pwait` or `epoll_pwait2` given a mask), as
271    /// Linux copied it at the call's entry. A signal that ends the wait finds
272    /// either this mask installed, if the kernel ran the call, or the thread's
273    /// own mask, if the call had not started.
274    #[serde(default)]
275    pub(crate) wait_mask: Option<Sigset>,
276    /// The `uc_sigmask` of each handler frame entered and not yet known to
277    /// have returned, innermost last. A backend that reports `rt_sigreturn`
278    /// pops one per return ([`ThreadSignals::sigreturn`]); where the kernel
279    /// runs `rt_sigreturn` unseen, a return is recognised when the kernel's
280    /// mask equals one of these ([`ThreadSignals::reconcile_returns`]).
281    #[serde(default)]
282    pub(crate) activations: Vec<Sigset>,
283    /// For each open frame, whether a check saw its handler's mask installed.
284    /// A frame taken as returned before that happened leaves its entry
285    /// unverified: the kernel's mask may have equalled the saved mask because
286    /// the handler had returned, or because the entry was not what the copy
287    /// predicted.
288    #[serde(default)]
289    pub(crate) observed: Vec<bool>,
290    /// A backend report changed this thread's mask, or found the frame
291    /// faulting, since the last check: the next syscall entry compares the
292    /// copy with the kernel even with no handler frame open, so a report
293    /// the kernel did not follow fails closed there rather than at a later
294    /// check.
295    #[serde(default)]
296    pub(crate) verify_at_entry: bool,
297}
298
299impl ThreadSignals {
300    /// A thread whose mask is `blocked`.
301    pub(crate) fn new(blocked: Sigset) -> Self {
302        ThreadSignals {
303            blocked: blocked & !UNBLOCKABLE,
304            saved: None,
305            wait_mask: None,
306            activations: Vec::new(),
307            observed: Vec::new(),
308            verify_at_entry: false,
309        }
310    }
311
312    /// `rt_sigprocmask(how, set, &old)` after the size check. Returns the old
313    /// mask, read before any change, which is what the kernel copies out.
314    /// `how` is checked only when `set` is given.
315    pub(crate) fn sigprocmask(&mut self, how: i32, set: Option<Sigset>) -> Result<Sigset, Errno> {
316        let old = self.blocked;
317        if let Some(set) = set {
318            let set = set & !UNBLOCKABLE;
319            self.blocked = match how {
320                libc::SIG_BLOCK => old | set,
321                libc::SIG_UNBLOCK => old & !set,
322                libc::SIG_SETMASK => set,
323                _ => return Err(Errno::EINVAL),
324            };
325        }
326        Ok(old)
327    }
328
329    /// A temporary-mask wait that a signal interrupted returned with `mask`
330    /// still installed, and the mask from before it to be put back.
331    pub(crate) fn interrupted_under_temporary_mask(&mut self, mask: Sigset) {
332        self.settle();
333        self.saved = Some(self.blocked);
334        self.blocked = mask & !UNBLOCKABLE;
335    }
336
337    /// The thread went on without a handler entering first: a pending
338    /// temporary-mask restore has happened.
339    pub(crate) fn settle(&mut self) {
340        if let Some(saved) = self.saved.take() {
341            self.blocked = saved;
342        }
343    }
344
345    /// A handler for `signal` with `action` was entered, its frame written
346    /// (Linux `signal_delivered`). The handler runs with `sa_mask` added and,
347    /// without `SA_NODEFER`, `signal` itself. Its frame saves the mask
348    /// `rt_sigreturn` restores: a pending temporary-mask restore's mask, which
349    /// the frame now carries, else the current one (Linux `sigmask_to_save`).
350    ///
351    /// Returns false when the thread already has [`MAX_OPEN_FRAMES`] open
352    /// frames: the copy can then no longer follow returns, and the caller must
353    /// treat it as unknown.
354    #[must_use]
355    pub(crate) fn enter_handler(&mut self, signal: i32, action: &Action) -> bool {
356        if self.activations.len() == MAX_OPEN_FRAMES {
357            return false;
358        }
359        let frame = self.saved.take().unwrap_or(self.blocked);
360        let mut blocked = self.blocked | action.mask;
361        if action.flags & sa_flag(libc::SA_NODEFER) == 0 {
362            blocked |= sigbit(signal);
363        }
364        self.blocked = blocked & !UNBLOCKABLE;
365        self.activations.push(frame);
366        self.observed.push(false);
367        true
368    }
369
370    /// A check found this thread's current mask installed while handler
371    /// frames are open: the innermost handler's mask is verified, unless it
372    /// equals the mask that frame saves, in which case the check cannot tell
373    /// the handler from its return and verifies nothing.
374    pub(crate) fn observe_handler_mask(&mut self) {
375        if self.activations.last() != Some(&self.blocked)
376            && let Some(observed) = self.observed.last_mut()
377        {
378            *observed = true;
379        }
380    }
381
382    /// Whether the innermost open frame's handler mask was ever verified.
383    pub(crate) fn innermost_observed(&self) -> bool {
384        self.observed.last().copied().unwrap_or(true)
385    }
386
387    /// `rt_sigreturn` read `mask` from the frame and installed it.
388    pub(crate) fn sigreturn(&mut self, mask: Sigset) {
389        self.activations.pop();
390        self.observed.pop();
391        self.saved = None;
392        self.blocked = mask & !UNBLOCKABLE;
393    }
394
395    /// Accounts for handler returns the backend did not report, given the
396    /// kernel's current mask `observed`. Returns false when `observed` is
397    /// neither the copy's mask nor the saved mask of an open handler frame:
398    /// the copy is then wrong, or a handler rewrote its frame's `uc_sigmask` to
399    /// a mask no open frame saved, which Linux allows and the check reports as
400    /// a mismatch. A match pops that frame and every frame inside it.
401    ///
402    /// Bits in `ignore` are left out of every comparison: signals whose
403    /// kernel state the backend itself changes. A recognised return installs
404    /// the frame's mask as the copy holds it, those bits included.
405    ///
406    /// The match is a mask equality, so an inner handler that rewrites its
407    /// `uc_sigmask` to an outer frame's saved mask looks like that many
408    /// returns; a caller that needs certainty treats a match deeper than the
409    /// innermost frame as ambiguous. Only a backend report of each
410    /// `rt_sigreturn` removes the ambiguity, and where the backend reports
411    /// them the copy does not use this.
412    pub(crate) fn reconcile_returns(&mut self, observed: Sigset, ignore: Sigset) -> bool {
413        let same = |mask: Sigset| mask & !ignore == observed & !ignore;
414        if same(self.blocked) {
415            return true;
416        }
417        match self.activations.iter().rposition(|&mask| same(mask)) {
418            Some(frame) => {
419                self.blocked = self.activations[frame];
420                self.activations.truncate(frame);
421                self.observed.truncate(frame);
422                true
423            }
424            None => false,
425        }
426    }
427}

429/// `signal` was selected for delivery and, for a caught signal, its handler
430/// frame written: the shared `dispositions` and the receiving thread's
431/// `thread` change as Linux changes them. Returns the action used. An
432/// ignored or default action changes no mask; `SA_RESETHAND` has reset the
433/// handler in either case.
434///
435/// `Ok(None)` when the handler frame is one more than the copy follows
436/// ([`ThreadSignals::enter_handler`]).
437pub(crate) fn deliver(
438    dispositions: &mut Dispositions,
439    thread: &mut ThreadSignals,
440    signal: i32,
441) -> Result<Option<Action>, Errno> {
442    let action = dispositions.select_for_delivery(signal)?;
443    if action.is_caught() {
444        // A handler ends the wait; an ignored signal does not.
445        thread.wait_mask = None;
446        if !thread.enter_handler(signal, &action) {
447            return Ok(None);
448        }
449    }
450    Ok(Some(action))
451}

453/// One thread's view of the signal state: its process's (more exactly, its
454/// `CLONE_SIGHAND` group's) actions, shared with every thread of the group,
455/// and its own masks.
456#[derive(Clone, Debug, Serialize, Deserialize)]
457pub(crate) struct SignalShadow {
458    /// Shared by every task created with `CLONE_SIGHAND`. `None` once any
459    /// task sharing it lost its copy: that task's later changes would no
460    /// longer reach it, so it is unknown to every sharer.
461    pub(crate) dispositions: Arc<Mutex<Option<Dispositions>>>,
462    /// This thread's masks.
463    pub(crate) thread: ThreadSignals,
464}
465
466impl SignalShadow {
467    /// The first image's state, from the ignored set and mask it inherited.
468    pub(crate) fn seeded(ignored: Sigset, blocked: Sigset) -> Self {
469        SignalShadow {
470            dispositions: Arc::new(Mutex::new(Some(Dispositions::after_exec(ignored)))),
471            thread: ThreadSignals::new(blocked),
472        }
473    }
474
475    /// The state of a task this thread creates with clone flags `flags`. The
476    /// child starts with this thread's mask. With `CLONE_SIGHAND` it shares
477    /// the actions; otherwise it gets a copy, as Linux's `copy_sighand` makes,
478    /// with caught signals reset under `CLONE_CLEAR_SIGHAND`. `None` when the
479    /// actions are unknown, and for any child created inside a handler, whose
480    /// frames the copy cannot follow.
481    pub(crate) fn for_clone(&self, flags: u64) -> Option<Self> {
482        let dispositions = if flags & libc::CLONE_SIGHAND as u64 != 0 {
483            self.dispositions.lock().unwrap().as_ref()?;
484            Arc::clone(&self.dispositions)
485        } else {
486            let mut copy = self.dispositions.lock().unwrap().clone()?;
487            if flags & CLONE_CLEAR_SIGHAND != 0 {
488                copy.reset_for_exec();
489            }
490            Arc::new(Mutex::new(Some(copy)))
491        };
492        let mut thread = self.thread.clone();
493        thread.settle();
494        // A child created inside a handler may carry on in it and return
495        // through its frame (fork, vfork, or a raw clone with no child stack),
496        // or start on a stack of its own; the clone flags do not say which,
497        // so it gets no copy.
498        if !thread.activations.is_empty() {
499            if flags & libc::CLONE_SIGHAND as u64 != 0 {
500                // The untracked child shares these actions and could change
501                // them unseen.
502                self.forget_dispositions();
503            }
504            return None;
505        }
506        thread.activations.clear();
507        thread.observed.clear();
508        Some(SignalShadow {
509            dispositions,
510            thread,
511        })
512    }
513
514    /// This thread completed an `execve`. Its actions become a private copy
515    /// first (Linux `unshare_sighand`), so a task that shared them keeps its
516    /// own, and are then reset. The mask is kept. Returns false when the
517    /// actions were unknown.
518    pub(crate) fn exec(&mut self) -> bool {
519        let Some(mut own) = self.dispositions.lock().unwrap().clone() else {
520            return false;
521        };
522        own.reset_for_exec();
523        self.dispositions = Arc::new(Mutex::new(Some(own)));
524        self.thread.settle();
525        self.thread.activations.clear();
526        self.thread.observed.clear();
527        true
528    }
529
530    /// Marks the shared actions unknown, for this thread's copy being
531    /// dropped.
532    pub(crate) fn forget_dispositions(&self) {
533        *self.dispositions.lock().unwrap() = None;
534    }
535}
