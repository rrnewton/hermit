34/// Give inherited standard streams identities that do not depend on backend
35/// loader activity observed before the guest reaches its entry point.
36fn deterministic_stdio_inode(fd: RawFd) -> Option<DetInode> {
37    (libc::STDIN_FILENO..=libc::STDERR_FILENO)
38        .contains(&fd)
39        .then_some(DetInode::mint(
40            DET_SPECIAL_INODE_OFFSET.as_raw() + fd as u64,
41        ))
42}
43
44/// Preserve the existing inherited-stdio slot identities, but do not assign one
45/// to an ordinary file or pipe that has replaced that slot. Aliases above fd 2
46/// retain their existing pooled identity; extending the fixed namespace to
47/// aliases requires consistent path-stat behavior too.
48pub(crate) fn deterministic_stdio_inode_for_resource(
49    fd: RawFd,
50    resource: Option<ResourceID>,
51) -> Option<DetInode> {
52    match resource {
53        Some(ResourceID::Device(
54            Device::ContainerStdin | Device::ContainerStdout | Device::ContainerStderr,
55        )) => deterministic_stdio_inode(fd),
56        _ => None,
57    }
58}

60#[cfg(test)]
61mod tests {
62    use super::*;
63
64    #[test]
65    fn stdio_inode_namespace_is_fixed() {
66        assert_eq!(
67            deterministic_stdio_inode(libc::STDIN_FILENO),
68            Some(DetInode::mint(1000))
69        );
70        assert_eq!(
71            deterministic_stdio_inode(libc::STDOUT_FILENO),
72            Some(DetInode::mint(1001))
73        );
74        assert_eq!(
75            deterministic_stdio_inode(libc::STDERR_FILENO),
76            Some(DetInode::mint(1002))
77        );
78        assert_eq!(deterministic_stdio_inode(3), None);
79
80        for resource in [
81            ResourceID::Device(Device::ContainerStdin),
82            ResourceID::Device(Device::ContainerStdout),
83            ResourceID::Device(Device::ContainerStderr),
84        ] {
85            for fd in 0..=2 {
86                assert_eq!(
87                    deterministic_stdio_inode_for_resource(fd, Some(resource.clone())),
88                    Some(DetInode::mint(1000 + fd as u64))
89                );
90            }
91            assert_eq!(
92                deterministic_stdio_inode_for_resource(3, Some(resource)),
93                None
94            );
95        }
96        for fd in 0..=2 {
97            assert_eq!(deterministic_stdio_inode_for_resource(fd, None), None);
98            assert_eq!(
99                deterministic_stdio_inode_for_resource(
100                    fd,
101                    Some(ResourceID::FileContents(DetInode::mint(1000))),
102                ),
103                None
104            );
105        }
106    }
