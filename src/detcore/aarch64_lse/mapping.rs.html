9//! Telling code from data in an aarch64 ELF file's executable sections, by
10//! the file's mapping symbols.
11//!
12//! An executable section can hold data as well as instructions: OpenSSL's
13//! generated assembly, for one, keeps its constant tables in `.text`. Those
14//! words decode as instructions as readily as code does, so a scan that reads
15//! the whole section finds load/store-exclusive "loops" that never execute,
16//! and a rewrite that took one would corrupt the table. The AArch64 ELF ABI
17//! marks the boundaries with mapping symbols in the symbol table: `$x` (or
18//! `$x.<any>`) starts A64 code and `$d` (or `$d.<any>`) starts data, each
19//! running to the next mapping symbol of the same section ("ELF for the Arm
20//! 64-bit Architecture", section 5.5.4, Mapping symbols). A stripped file has
21//! no symbol table and so no mapping symbols, and a section can lack them too.
22//!
23//! Where a section has no mapping symbols, the file's unwind table
24//! (`.eh_frame`) is the evidence instead: each frame description entry (FDE)
25//! covers one function's code, stripped files keep the table for unwinding,
26//! and a constant table has no FDE. An FDE covers its whole function, so a
27//! table or literal pool placed inside the function is inside it too: under
28//! that evidence alone, code is what control reaches from the functions'
29//! entries ([`reachable_code`]), less what the code addresses as data, up
30//! to the next instruction a branch reaches ([`data_extents`]), since a
31//! call is taken to return; and a sequence a literal load reaches is
32//! refused in any case ([`literal_targets`]). Bytes neither marks are not
33//! known to be code.
34//!
35//! These are pure functions of the file's bytes, which the rewriter
36//! ([`super::elf`]) consults before it plans a single sequence.

55/// What tells an ELF file's code from its data: its mapping symbols, and the
56/// code ranges its unwind table's FDEs cover.
57#[derive(Clone, Debug, Default, PartialEq, Eq)]
58pub struct CodeEvidence {
59    /// The mapping symbols, sorted as [`mapping_symbols`] returns them.
60    pub symbols: Vec<MappingSymbol>,
61    /// The local, untyped symbols named like mapping symbols that are neither
62    /// `$x` nor `$d` ([`unsupported_mapping_symbols`]), as (section index,
63    /// name). A section that has one is not proved by its mapping symbols.
64    pub unsupported: Vec<(usize, String)>,
65    /// The address range of each FDE, sorted; `None` without an unwind
66    /// table, or with one this reader cannot decode.
67    pub unwind: Option<Vec<(u64, u64)>>,
68}

79impl CodeEvidence {
80    /// The evidence of an ELF64 little-endian file whose section header
81    /// table is `table` and whose section-name string table is section
82    /// `shstrndx`. `read` returns the file bytes at (offset, size), or
83    /// `None` when it will not (out of the file, or larger than the caller
84    /// reads); a table it does not return is evidence the file lacks.
85    ///
86    /// The unwind table is read only for a loaded image (`e_type` `ET_EXEC`
87    /// or `ET_DYN`): in a relocatable object each FDE's start is a
88    /// relocation's to fill, so the bytes alone would place every function
89    /// near address 0.
90    pub fn read(
91        table: &[u8],
92        shstrndx: usize,
93        e_type: u16,
94        mut read: impl FnMut((u64, u64)) -> Option<Vec<u8>>,
95    ) -> Self {
96        const ET_EXEC: u16 = 2;
97        const ET_DYN: u16 = 3;
98        let (symbols, unsupported) = symbol_table(table)
99            .and_then(|(symtab, strtab)| {
100                let (symtab, strtab) = (read(symtab)?, read(strtab)?);
101                Some((
102                    mapping_symbols(&symtab, &strtab),
103                    unsupported_mapping_symbols(&symtab, &strtab),
104                ))
105            })
106            .unwrap_or_default();
107        let unwind = matches!(e_type, ET_EXEC | ET_DYN)
108            .then(|| section_range(table, shstrndx))
109            .flatten()
110            .and_then(&mut read)
111            .and_then(|names| section_named(table, &names, b".eh_frame"))
112            .and_then(|(range, address)| fde_ranges(&read(range)?, address));
113        CodeEvidence {
114            symbols,
115            unsupported,
116            unwind,
117        }
118    }
119
120    /// The first mapping symbol section `section` holds that is neither `$x`
121    /// nor `$d`, if any.
122    pub fn unsupported_in(&self, section: usize) -> Option<&str> {
123        self.unsupported
124            .iter()
125            .find(|(marked, _)| *marked == section)
126            .map(|(_, name)| name.as_str())
127    }

129    /// What section `section` (at `address`, `size` bytes long) holds: by its
130    /// mapping symbols when it has any ([`code_spans`]), or else by the
131    /// FDEs' ranges within it.
132    pub fn classify(&self, section: usize, address: u64, size: u64) -> SectionCode {
133        let end = address.saturating_add(size);
134        let complement = |covered: &[(u64, u64)]| {
135            let mut gaps = Vec::new();
136            let mut at = address;
137            for (start, stop) in covered {
138                if at < *start {
139                    gaps.push((at, *start));
140                }
141                at = at.max(*stop);
142            }
143            if at < end {
144                gaps.push((at, end));
145            }
146            gaps
147        };
148        if let Some(spans) = code_spans(&self.symbols, section, address, size) {
149            // `$d` ranges are data; only the bytes before the first mark are
150            // unknown.
151            let first = self
152                .symbols
153                .iter()
154                .filter(|symbol| symbol.section == section)
155                .map(|symbol| symbol.address)
156                .find(|mark| (address..end).contains(mark))
157                .unwrap_or(end);
158            return SectionCode {
159                code: spans,
160                unknown: complement(&[(first, end)]),
161                evidence: Some(Evidence::MappingSymbols),
162            };
163        }
164        let Some(unwind) = &self.unwind else {
165            return SectionCode {
166                code: Vec::new(),
167                unknown: complement(&[]),
168                evidence: None,
169            };
170        };
171        let mut code: Vec<(u64, u64)> = Vec::new();
172        for (start, stop) in unwind
173            .iter()
174            .map(|(start, stop)| ((*start).max(address), (*stop).min(end)))
175            .filter(|(start, stop)| start < stop)
176        {
177            match code.last_mut() {
178                Some(last) if start <= last.1 => last.1 = last.1.max(stop),
179                _ => code.push((start, stop)),
180            }
181        }
182        // No FDE is not evidence of data: what no FDE covers is unknown.
183        SectionCode {
184            unknown: complement(&code),
185            code,
186            evidence: Some(Evidence::UnwindTable),
187        }
188    }
189}

205/// The code among `words` (at `base`) that control can reach from
206/// `entries` without leaving `covered` (sorted, disjoint): following each
207/// instruction to the next unless it is an unconditional branch, a return,
208/// an indirect jump or a trap, and each direct branch to its target. Under
209/// the unwind table alone this is what is taken for code: an FDE covers its
210/// whole function, including a constant table the function branches over or
211/// keeps after its return, and those bytes are not reached. Code reached
212/// only through a jump table or an exception landing pad is not either, and
213/// is left alone with them. A call falls through only where `returns`
214/// proves the callee comes back; [`data_extents`] takes back what the code
215/// addresses as data.
216pub fn reachable_code(
217    words: &[u32],
218    base: u64,
219    covered: &[(u64, u64)],
220    entries: &[u64],
221    returns: &dyn Fn(u64) -> bool,
222) -> Vec<(u64, u64)> {
223    let index_of = |addr: u64| -> Option<usize> {
224        if addr < base || !addr.is_multiple_of(4) {
225            return None;
226        }
227        let index = usize::try_from((addr - base) / 4).ok()?;
228        let after = covered.partition_point(|(start, _)| *start <= addr);
229        let inside = after.checked_sub(1).is_some_and(|i| addr < covered[i].1);
230        (index < words.len() && inside).then_some(index)
231    };
232    let mut seen = vec![false; words.len()];
233    let mut pending: Vec<usize> = entries
234        .iter()
235        .filter_map(|entry| index_of(*entry))
236        .collect();
237    while let Some(mut index) = pending.pop() {
238        while !seen[index] {
239            seen[index] = true;
240            let pc = base + 4 * index as u64;
241            let (mut falls_through, target) = flow(words[index], pc);
242            // A call comes back only if the function it calls can.
243            if matches!(decode(words[index]), Insn::Branch { link: true, .. })
244                && target.is_some_and(|target| !returns(target))
245            {
246                falls_through = false;
247            }
248            if let Some(target) = target.and_then(index_of) {
249                pending.push(target);
250            }
251            match falls_through.then(|| index_of(pc + 4)).flatten() {
252                Some(next) => index = next,
253                None => break,
254            }
255        }
256    }
257    let mut spans: Vec<(u64, u64)> = Vec::new();
258    for (index, _) in seen.iter().enumerate().filter(|(_, seen)| **seen) {
259        let addr = base + 4 * index as u64;
260        match spans.last_mut() {
261            Some(last) if last.1 == addr => last.1 = addr + 4,
262            _ => spans.push((addr, addr + 4)),
263        }
264    }
265    spans
266}

268/// Where control goes after `word` at `pc`: whether it can fall through to
269/// the next instruction, and the target of a direct branch it can take. A
270/// conditional branch whose outcome the encoding fixes counts as what it
271/// does: `B.AL` and `B.NV` always branch, and the zero register makes `CBZ`
272/// and `TBZ` always branch and `CBNZ` and `TBNZ` never.
273fn flow(word: u32, pc: u64) -> (bool, Option<u64>) {
274    const ZR: super::insn::Reg = 31;
275    let target = |offset: i64| Some(pc.wrapping_add(offset as u64));
276    match decode(word) {
277        Insn::Branch {
278            link: false,
279            offset,
280        } => (false, target(offset)),
281        Insn::CondBranch {
282            cond: 0b1110 | 0b1111,
283            offset,
284        } => (false, target(offset)),
285        Insn::CompareBranch {
286            rt: ZR,
287            nonzero,
288            offset,
289            ..
290        } => {
291            if nonzero {
292                (true, None)
293            } else {
294                (false, target(offset))
295            }
296        }
297        // TBZ/TBNZ: op (24), Rt (4:0).
298        Insn::TestBranch { offset } if word & 0x1f == u32::from(ZR) => {
299            if (word >> 24) & 1 == 1 {
300                (true, None)
301            } else {
302                (false, target(offset))
303            }
304        }
305        Insn::Branch { link: true, offset }
306        | Insn::CondBranch { offset, .. }
307        | Insn::CompareBranch { offset, .. }
308        | Insn::TestBranch { offset } => (true, target(offset)),
309        // Unconditional branch (register): BLR and its authenticated forms
310        // return; BR, RET, ERET and theirs do not.
311        _ if word & 0xFE00_0000 == 0xD600_0000 => {
312            (matches!((word >> 21) & 0xF, 0b0001 | 0b1001), None)
313        }
314        // UDF, BRK and HLT trap.
315        _ if word & 0xFFFF_0000 == 0
316            || word & 0xFFE0_001F == 0xD420_0000
317            || word & 0xFFE0_001F == 0xD440_0000 =>
318        {
319            (false, None)
320        }
321        _ => (true, None),
322    }
323}

326/// [`CodeEvidence::classify`]. What is in neither list is data (`$d`).

329    /// Known to be code.
330    pub code: Vec<(u64, u64)>,
331    /// Neither known code nor known data: bytes before a section's first
332    /// mapping symbol, bytes no FDE covers, or the whole section when

339/// The file range (offset, size) of section `index` of a section header
340/// table.
341fn section_range(table: &[u8], index: usize) -> Option<(u64, u64)> {
342    let entry = table.as_chunks::<SECTION_HEADER_SIZE>().0.get(index)?;
343    Some((
344        u64::from_le_bytes(entry[24..32].try_into().unwrap()),
345        u64::from_le_bytes(entry[32..40].try_into().unwrap()),
346    ))
347}

349/// The file range and address of the section named `name`, given the
350/// section-name string table's bytes.
351fn section_named(table: &[u8], names: &[u8], name: &[u8]) -> Option<((u64, u64), u64)> {
352    table
353        .as_chunks::<SECTION_HEADER_SIZE>()
354        .0
355        .iter()
356        .find(|entry| {
357            let at = u32::from_le_bytes(entry[0..4].try_into().unwrap()) as usize;
358            names
359                .get(at..)
360                .and_then(|rest| rest.get(..name.len() + 1))
361                .is_some_and(|candidate| {
362                    candidate[..name.len()] == *name && candidate[name.len()] == 0
363                })
364        })
365        .map(|entry| {
366            (
367                (
368                    u64::from_le_bytes(entry[24..32].try_into().unwrap()),
369                    u64::from_le_bytes(entry[32..40].try_into().unwrap()),
370                ),
371                u64::from_le_bytes(entry[16..24].try_into().unwrap()),
372            )
373        })
374}

376/// A cursor over the bytes of an unwind table.
377struct Cursor<'a> {
378    bytes: &'a [u8],
379    at: usize,
380}
381
382impl Cursor<'_> {
383    fn take(&mut self, len: usize) -> Option<&[u8]> {
384        let bytes = self.bytes.get(self.at..self.at.checked_add(len)?)?;
385        self.at += len;
386        Some(bytes)
387    }
388
389    fn u8(&mut self) -> Option<u8> {
390        Some(self.take(1)?[0])
391    }
392
393    fn uleb(&mut self) -> Option<u64> {
394        let mut value = 0u64;
395        for shift in (0..64).step_by(7) {
396            let byte = self.u8()?;
397            value |= u64::from(byte & 0x7f) << shift;
398            if byte & 0x80 == 0 {
399                return Some(value);
400            }
401        }
402        None
403    }
404
405    fn sleb(&mut self) -> Option<i64> {
406        let mut value = 0i64;
407        let mut shift = 0;
408        loop {
409            let byte = self.u8()?;
410            value |= i64::from(byte & 0x7f) << shift;
411            shift += 7;
412            if byte & 0x80 == 0 {
413                if shift < 64 && byte & 0x40 != 0 {
414                    value |= -1i64 << shift;
415                }
416                return Some(value);
417            }
418            if shift >= 64 {
419                return None;
420            }
421        }
422    }
423
424    /// A pointer in `encoding` (DW_EH_PE_*), at address `here` of the
425    /// cursor's position. Only absolute and pc-relative values are read;
426    /// any other application, or an indirect one, is `None`.
427    fn pointer(&mut self, encoding: u8, here: u64) -> Option<u64> {
428        let value = self.raw_pointer(encoding)?;
429        match encoding & 0xf0 {
430            0x00 => Some(value),
431            0x10 => Some(here.wrapping_add(value)),
432            _ => None,
433        }
434    }
435
436    /// A pointer's value as stored in `encoding`'s format, before any
437    /// application: how a pointer this reader does not need (a
438    /// personality routine's, usually indirect) is stepped over.
439    fn raw_pointer(&mut self, encoding: u8) -> Option<u64> {
440        Some(match encoding & 0x0f {
441            0x00 | 0x04 => u64::from_le_bytes(self.take(8)?.try_into().unwrap()),
442            0x01 => self.uleb()?,
443            0x02 => u64::from(u16::from_le_bytes(self.take(2)?.try_into().unwrap())),
444            0x03 => u64::from(u32::from_le_bytes(self.take(4)?.try_into().unwrap())),
445            0x09 => self.sleb()? as u64,
446            0x0a => i16::from_le_bytes(self.take(2)?.try_into().unwrap()) as u64,
447            0x0b => i32::from_le_bytes(self.take(4)?.try_into().unwrap()) as u64,
448            0x0c => u64::from_le_bytes(self.take(8)?.try_into().unwrap()),
449            _ => return None,
450        })
451    }
452}

454/// The FDE pointer encoding of a CIE, given its body after the CIE ID:
455/// `DW_EH_PE_absptr` unless its augmentation names one with `R`. `None` for
456/// a version other than 1 or 3 (the two `.eh_frame` uses), an augmentation
457/// this reader does not know, or a body that ends early.
458fn cie_pointer_encoding(body: &[u8]) -> Option<u8> {
459    let mut cie = Cursor { bytes: body, at: 0 };
460    let version = cie.u8()?;
461    if !matches!(version, 1 | 3) {
462        return None;
463    }
464    let augmentation_start = cie.at;
465    while cie.u8()? != 0 {}
466    let augmentation = body[augmentation_start..cie.at - 1].to_vec();
467    if augmentation.windows(2).any(|pair| pair == b"eh") {
468        cie.take(8)?;
469    }
470    cie.uleb()?; // code alignment
471    cie.sleb()?; // data alignment
472    if version == 1 {
473        cie.u8()?;
474    } else {
475        cie.uleb()?;
476    }
477    if augmentation.first() != Some(&b'z') {
478        return Some(0);
479    }
480    cie.uleb()?; // augmentation data length
481    for letter in &augmentation[1..] {
482        match letter {
483            b'R' => return cie.u8(),
484            b'L' => {
485                cie.u8()?;
486            }
487            b'P' => {
488                let encoding = cie.u8()?;
489                // An aligned pointer would need padding this reader does
490                // not track.
491                if encoding & 0x70 == 0x50 {
492                    return None;
493                }
494                cie.raw_pointer(encoding)?;
495            }
496            b'S' | b'B' => {}
497            _ => return None,
498        }
499    }
500    Some(0)
501}

503/// The address range of each FDE of an unwind table (`.eh_frame`, at
504/// `address`), sorted. `None` when the table cannot be read to its end: a
505/// reader that skipped an FDE it could not decode would take that
506/// function's code for data, which is safe, but one that misread a length
507/// could take data for code.
508pub fn fde_ranges(bytes: &[u8], address: u64) -> Option<Vec<(u64, u64)>> {
509    let mut ranges = Vec::new();
510    // The FDE pointer encoding of each CIE walked so far, by the offset of
511    // its length field: an FDE must name one of these.
512    let mut encodings = std::collections::HashMap::new();
513    let mut at = 0usize;
514    while at + 4 <= bytes.len() {
515        let mut header = Cursor { bytes, at };
516        let length = u32::from_le_bytes(header.take(4)?.try_into().unwrap());
517        if length == 0 {
518            break;
519        }
520        let length = if length == u32::MAX {
521            u64::from_le_bytes(header.take(8)?.try_into().unwrap())
522        } else {
523            u64::from(length)
524        };
525        let end = header.at.checked_add(usize::try_from(length).ok()?)?;
526        if end > bytes.len() {
527            return None;
528        }
529        // Each entry is read within its own length.
530        let mut entry = Cursor {
531            bytes: &bytes[..end],
532            at: header.at,
533        };
534        let id_at = entry.at;
535        let id = u32::from_le_bytes(entry.take(4)?.try_into().unwrap());
536        if id == 0 {
537            encodings.insert(at, cie_pointer_encoding(&bytes[entry.at..end])?);
538        } else {
539            // An FDE: its CIE is `id` bytes before the field.
540            let cie = id_at.checked_sub(usize::try_from(id).ok()?)?;
541            let encoding = *encodings.get(&cie)?;
542            let start = entry.pointer(encoding, address.wrapping_add(entry.at as u64))?;
543            let size = entry.pointer(encoding & 0x0f, 0)?;
544            if size != 0 {
545                ranges.push((start, start.checked_add(size)?));
546            }
547        }
548        at = end;
549    }
550    ranges.sort_unstable();
551    Some(ranges)
552}

554/// The byte ranges `words` (at `base`) load as literals or take the address
555/// of: each `LDR` (literal) target with its load's size, and each `ADR`
556/// target as one word, since what follows it is not known. A literal pool
557/// inside a function is read this way, so a candidate sequence any of these
558/// reach is data the unwind table took for code.
559pub fn literal_targets(words: &[u32], base: u64) -> Vec<(u64, u64)> {
560    literal_references(words, base)
561        .into_iter()
562        .map(|reference| reference.range)
563        .collect()
564}

577/// Keep address-taking distinct from a read of the instruction bytes: only
578/// the former can name a function entry without observing a rewrite there.
579pub(super) fn literal_references(words: &[u32], base: u64) -> Vec<LiteralReference> {
580    words
581        .iter()
582        .enumerate()
583        .filter_map(|(index, word)| {
584            let pc = base.wrapping_add(4 * index as u64);
585            if word & 0x3B00_0000 == 0x1800_0000 {
586                // LDR (literal): opc (31:30), V (26), imm19 (23:5).
587                let size = match (word >> 30, (word >> 26) & 1) {
588                    (0b00, _) | (0b10, 0) => 4,
589                    (0b01, _) => 8,
590                    (0b10, _) => 16,
591                    _ => return None, // PRFM (literal) loads nothing
592                };
593                let offset = ((((word >> 5) & 0x7_FFFF) << 13) as i32 >> 11) as i64;
594                let target = pc.wrapping_add(offset as u64);
595                Some(LiteralReference {
596                    range: (target, target.wrapping_add(size)),
597                    kind: LiteralKind::MemoryRead,
598                })
599            } else if word & 0x9F00_0000 == 0x1000_0000 {
600                // ADR: immlo (30:29), immhi (23:5).
601                let imm = ((word >> 5) & 0x7_FFFF) << 2 | ((word >> 29) & 0b11);
602                let offset = ((imm << 11) as i32 >> 11) as i64;
603                let target = pc.wrapping_add(offset as u64);
604                Some(LiteralReference {
605                    range: (target, target.wrapping_add(4)),
606                    kind: LiteralKind::Address,
607                })
608            } else {
609                None
610            }
611        })
612        .collect()
613}

615/// What an `ADRP` or `ADR` lets the code after it read or write through its
616/// register, for the image-wide reader scan ([`literal_references`] covers
617/// the `LDR` (literal) and the `ADR` itself).
618#[derive(Clone, Copy, Debug, PartialEq, Eq)]
619pub(super) enum BaseReference {
620    /// A load or store through the register whose address and width the
621    /// instruction gives: these bytes.
622    Access(u64, u64),
623    /// An address an `ADD` or `SUB` (immediate) forms from the register,
624    /// taken as `ADR`'s target is: one word, and nothing at an FDE start,
625    /// since forming an address is not a read, and one that names a
626    /// function's entry takes a pointer to the function. What the code then
627    /// does with the address is followed like the `ADR`'s or `ADRP`'s own.
628    Address(u64),
629    /// A use of the address whose extent is not known: a load or store
630    /// through it that the instruction does not bound (an index register, a
631    /// scalable vector access, a form not decoded), the address passed on to
632    /// a register or memory the scan does not follow, or the address still
633    /// held where following stops (a transfer of control, a conditional
634    /// branch, whose taken path is not followed, or the end of the window).
635    /// Any executable byte may be read through it, even where `at` is a
636    /// function's entry: a function pointer passed on can be read as well
637    /// as called. `at` is the address, or with `page` the `ADRP`'s page.
638    Unknown { at: u64, page: bool },
639}
640
641/// What one instruction does with the address a register holds.
642#[derive(Clone, Copy, Debug, PartialEq, Eq)]
643enum Effect {
644    /// It does not read the register.
645    Untouched,
646    /// It reads it without accessing memory or passing it on: a compare.
647    Tested,
648    /// It accesses `low..high` from the address, and then, with `writeback`,
649    /// leaves the address plus that in the register.
650    Accesses {
651        low: i64,
652        high: i64,
653        writeback: Option<i64>,
654    },
655    /// It forms the address plus `offset` in register `into` (`ADD` or
656    /// `SUB` immediate).
657    Forms { offset: i64, into: u32 },
658    /// It accesses memory through the address to an extent it does not
659    /// bound.
660    ReadsUnbounded,
661    /// It passes the address on: a copy, register arithmetic, a value stored,
662    /// an index, or a form not decoded.
663    PassesOn,
664}

666/// The bytes `words` (at `base`) may read or write through each `ADRP`'s
667/// page or `ADR`'s address. Each is followed through the instructions after
668/// it, up to [`BASE_WINDOW`], with what each does with every register
669/// holding an address from it ([`effect`]): an access the instruction bounds
670/// is a [`BaseReference::Access`], an `ADD` or `SUB` (immediate) a
671/// [`BaseReference::Address`] whose register is followed too, and a write
672/// ([`writes_register`]) ends the register's part. Anything else fails
673/// closed: an unbounded access, the address passed on, any use by an
674/// encoding [`decoded`] does not cover (wherever [`names`] says it may name
675/// the register), and every register still holding an address where
676/// following stops (a transfer of control, a conditional branch, the
677/// window's or the range's end) is a [`BaseReference::Unknown`].
678pub(super) fn base_references(words: &[u32], base: u64) -> Vec<BaseReference> {
679    let mut references = Vec::new();
680    for (index, word) in words.iter().enumerate() {
681        let pc = base.wrapping_add(4 * index as u64);
682        let register = word & 0x1F;
683        let imm = ((word >> 5) & 0x7_FFFF) << 2 | ((word >> 29) & 0b11);
684        let (origin, page) = if word & 0x9F00_0000 == 0x9000_0000 {
685            // ADRP: the 4 KiB page of `pc`, plus immhi:immlo pages.
686            let pages = ((imm << 11) as i32 >> 11) as i64;
687            ((pc & !0xFFF).wrapping_add((pages << 12) as u64), true)
688        } else if word & 0x9F00_0000 == 0x1000_0000 {
689            // ADR: `pc` plus immhi:immlo bytes.
690            let offset = ((imm << 11) as i32 >> 11) as i64;
691            (pc.wrapping_add(offset as u64), false)
692        } else {
693            continue;
694        };
695        if register == 31 {
696            continue; // into the zero register: nothing is addressed
697        }
698        let mut found = Vec::new();
699        let mut note = |reference: BaseReference| {
700            if !found.contains(&reference) {
701                found.push(reference);
702            }
703        };
704        let unknown = |at: u64, page: bool| BaseReference::Unknown { at, page };
705        // Each register holding an address from this one: the register, the
706        // address, and whether it is the bare page.
707        let mut bases = vec![(register, origin, page)];
708        for next in words[index + 1..].iter().take(BASE_WINDOW) {
709            if transfers_control(*next) || branches_conditionally(*next) {
710                break;
711            }
712            let mut after = Vec::new();
713            for &(held, at, bare) in &bases {
714                let mut kept = Some((held, at, bare));
715                match effect(*next, held) {
716                    Effect::Untouched | Effect::Tested => {}
717                    Effect::Accesses {
718                        low,
719                        high,
720                        writeback,
721                    } => {
722                        match (at.checked_add_signed(low), at.checked_add_signed(high)) {
723                            (Some(start), Some(end)) => note(BaseReference::Access(start, end)),
724                            _ => note(unknown(at, bare)),
725                        }
726                        if let Some(by) = writeback {
727                            kept = Some((held, at.wrapping_add_signed(by), false));
728                        }
729                    }
730                    Effect::Forms { offset, into } => {
731                        let target = at.wrapping_add_signed(offset);
732                        note(BaseReference::Address(target));
733                        after.push((into, target, false));
734                    }
735                    Effect::ReadsUnbounded => {
736                        note(unknown(at, bare));
737                        kept = None;
738                    }
739                    Effect::PassesOn => note(unknown(at, bare)),
740                }
741                if let Some((held, at, bare)) = kept
742                    && !writes_register(*next, held)
743                {
744                    after.push((held, at, bare));
745                }
746            }
747            // A register formed into replaces what it held.
748            bases.clear();
749            for (held, at, bare) in after.into_iter().rev() {
750                if !bases.iter().any(|(other, _, _)| *other == held) {
751                    bases.push((held, at, bare));
752                }
753            }
754            if bases.is_empty() {
755                break;
756            }
757        }
758        // Following stopped with these still holding an address.
759        for (_, at, bare) in bases {
760            note(unknown(at, bare));
761        }
762        references.extend(found);
763    }
764    references
765}

767/// How many instructions after an `ADRP` or `ADR` [`base_references`]
768/// follows its register. A compiler schedules a completion well after its
769/// `ADRP` (glibc's NSS tables complete one eleven instructions on), so this
770/// looks further than [`addressed_starts`]' eight.
771const BASE_WINDOW: usize = 64;
772
773/// Whether control may not continue to the next instruction: an
774/// unconditional branch (immediate or register, calls and returns among
775/// them) or an exception-generating instruction.
776fn transfers_control(word: u32) -> bool {
777    word & 0x7C00_0000 == 0x1400_0000 // B, BL
778        || word & 0xFE00_0000 == 0xD600_0000 // BR, BLR, RET, ERET, ...
779        || word & 0xFF00_0000 == 0xD400_0000 // SVC, HVC, SMC, BRK, HLT, ...
780}
781
782/// Whether control may go on elsewhere as well as to the next instruction:
783/// every word of the branch, exception and system class (op0 `101x`) that
784/// is neither [`transfers_control`] nor in [`system_space`], so `B.cond`,
785/// `BC.cond`, `CBZ`, `TBZ`, `CB<cc>` and any branch form not named here.
786/// The scan does not follow the taken path.
787fn branches_conditionally(word: u32) -> bool {
788    word & 0x1C00_0000 == 0x1400_0000 && !system_space(word) && !transfers_control(word)
789}
790
791/// Whether `word` is in the system instruction space (hints, barriers,
792/// `MSR`, `MRS`, `SYS` and their pair forms), whose instructions all
793/// continue to the next.
794fn system_space(word: u32) -> bool {
795    word & 0xFF80_0000 == 0xD500_0000
796}

798/// Every general register `word` may name, as a mask: each field a
799/// register can occupy in any encoding (bits 4:0, 9:5, 14:10 and 20:16)
800/// and the registers some families name implicitly ([`implicit`]). An
801/// encoding [`decoded`] does not cover touches a register only through
802/// this mask, and any it names counts as passed on.
803fn names(word: u32) -> u32 {
804    let field = |shift: u32| 1u32 << ((word >> shift) & 0x1F);
805    (field(0) | field(5) | field(10) | field(16) | implicit(word)) & !(1 << 31)
806}
807
808/// The registers some families name without a field of their own: a
809/// register list's successors (for any load or store, the register after
810/// each in bits 4:0 and 20:16, as `CASP`, `CASPT`, `RCWCASP` and the
811/// 128-bit atomics' pairs run; the eight of `LD64B` and `ST64B*`; and the
812/// system pair forms' Xt+1), the slice and predicate index registers of SME
813/// and SVE (w8 to w15), and the pointer authentication forms' fixed
814/// registers (x16 and x17 for `*1716`, x30 for `*Z`, `*SP` and `XPACLRI`,
815/// x15 to x17 for `*171615`, x16 for `CHKFEAT`).
816fn implicit(word: u32) -> u32 {
817    let (rt, rs) = (word & 0x1F, (word >> 16) & 0x1F);
818    let run =
819        |first: u32, count: u32| (0..count).fold(0, |mask, i| mask | 1 << ((first + i) & 0x1F));
820    let mut mask = 0;
821    if word & 0x0A00_0000 == 0x0800_0000 {
822        mask |= run(rt, 2) | run(rs, 2);
823    }
824    if ls64(word) {
825        mask |= run(rt, 8);
826    }
827    if word & 0xFFC0_0000 == 0xD540_0000 {
828        mask |= run(rt, 2); // SYSP, MSRR, MRRS
829    }
830    if word & 0x9E00_0000 == 0x8000_0000 || word & 0x1E00_0000 == 0x0400_0000 {
831        mask |= 0xFF00; // SME, SVE
832    }
833    if word & 0xFFFF_FF1F == 0xD503_211F {
834        mask |= 1 << 16 | 1 << 17;
835    }
836    if word & 0xFFFF_FF1F == 0xD503_231F || word == 0xD503_20FF {
837        mask |= 1 << 30;
838    }
839    if word == 0xD503_251F {
840        mask |= 1 << 16;
841    }
842    if word & 0xFFFF_83FF == 0xDAC1_83FE {
843        mask |= 0b111 << 15;
844    }
845    mask
846}

848/// Whether `word` is `LD64B`, `ST64B`, `ST64BV` or `ST64BV0`, which access
849/// 64 bytes at the base through the eight registers from Xt.
850fn ls64(word: u32) -> bool {
851    word & 0xFFE0_8C00 == 0xF820_8000 && matches!((word >> 12) & 0b111, 1 | 2 | 3 | 5)
852}

854/// Whether `word` is one of the forms [`effect`] decodes in full: the data
855/// processing (immediate) and (register) classes but pointer
856/// authentication, the hints, barriers, `MSR` (immediate and register),
857/// `MRS`, `DC CVAC` and `IC IVAU`, and the load and store families
858/// [`load_store_effect`] names. Every other encoding, SIMD and floating
859/// point, SVE and SME among them, uses a register only as [`names`] says,
860/// and any use passes the address on.
861fn decoded(word: u32) -> bool {
862    decoded_effect(word, 0).is_some()
863}
864
865/// What `word` does with the address `register` holds ([`Effect`]): as
866/// [`decoded_effect`] says for a form it decodes, and otherwise passed on
867/// where the word may name the register at all ([`names`]).
868fn effect(word: u32, register: u32) -> Effect {
869    match decoded_effect(word, register) {
870        Some(effect) => effect,
871        None if names(word) & (1 << register) != 0 => Effect::PassesOn,
872        None => Effect::Untouched,
873    }
874}

876/// [`effect`] for the forms it decodes in full, or `None`, whatever the
877/// register, for any other.
878fn decoded_effect(word: u32, register: u32) -> Option<Effect> {
879    let rn = (word >> 5) & 0x1F == register;
880    let rm = (word >> 16) & 0x1F == register;
881    let ra = (word >> 10) & 0x1F == register;
882    let rt = word & 0x1F == register;
883    if word & 0x0A00_0000 == 0x0800_0000 {
884        return load_store_effect(word, register);
885    }
886    if word & 0x1C00_0000 == 0x1000_0000 {
887        // Data processing (immediate).
888        return Some(match word & 0x1F80_0000 {
889            0x1000_0000 | 0x1080_0000 => Effect::Untouched, // ADR, ADRP
890            // MOVK keeps the rest of its register; MOVZ and MOVN read none.
891            0x1280_0000 if rt && (word >> 29) & 0b11 == 0b11 => Effect::PassesOn,
892            0x1280_0000 => Effect::Untouched,
893            0x1100_0000 | 0x1180_0000 if rn => {
894                // ADD/SUB (immediate): a compare into the zero register, or
895                // a 64-bit address formed in another register.
896                let rd = word & 0x1F;
897                let flags = (word >> 29) & 1 == 1;
898                if flags && rd == 31 {
899                    Effect::Tested
900                } else if word >> 31 == 1 && !flags && rd != 31 && (word >> 23) & 1 == 0 {
901                    let imm = i64::from((word >> 10) & 0xFFF) << (12 * ((word >> 22) & 1));
902                    let offset = if (word >> 30) & 1 == 1 { -imm } else { imm };
903                    Effect::Forms { offset, into: rd }
904                } else {
905                    Effect::PassesOn
906                }
907            }
908            // BFM (`BFI`, `BFXIL`) keeps the bits of its destination it does
909            // not insert, so it reads that register as well.
910            0x1300_0000 if rt && (word >> 29) & 0b11 == 0b01 => Effect::PassesOn,
911            // Logical, bitfield and extract (which reads two) pass it on.
912            0x1380_0000 if rn || rm => Effect::PassesOn,
913            _ if rn => Effect::PassesOn,
914            _ => Effect::Untouched,
915        });
916    }
917    if word & 0x0E00_0000 == 0x0A00_0000 {
918        // Data processing (register), but PAC*, AUT* and XPAC* (one source,
919        // opcode2 00001), which read their destination and can name
920        // registers implicitly. Bits 14:10 name a register only in the
921        // three-source class (`MADD`, `MSUB` and their widening forms);
922        // elsewhere they are a shift, an extension, a condition or an
923        // opcode. A compare into the zero register (ADDS, SUBS, ANDS) or a
924        // conditional compare only tests it.
925        if word & 0x5FFF_0000 == 0x5AC1_0000 {
926            return None;
927        }
928        let three_source = word & 0x1F00_0000 == 0x1B00_0000;
929        if !(rn || rm || (three_source && ra)) {
930            return Some(Effect::Untouched);
931        }
932        let into_zero = word & 0x1F == 31;
933        let adds = word & 0x1F00_0000 == 0x0B00_0000 && (word >> 29) & 1 == 1;
934        let ands = word & 0x1F00_0000 == 0x0A00_0000 && (word >> 29) & 0b11 == 0b11;
935        let ccmp = word & 0x1FE0_0000 == 0x1A40_0000;
936        return Some(if (into_zero && (adds || ands)) || ccmp {
937            Effect::Tested
938        } else {
939            Effect::PassesOn
940        });
941    }
942    if word & 0xFFFF_F01F == 0xD503_201F {
943        // A hint names no register but the implicit ones.
944        return Some(if implicit(word) & (1 << register) != 0 {
945            Effect::PassesOn
946        } else {
947            Effect::Untouched
948        });
949    }
950    if word & 0xFFFF_F01F == 0xD503_301F || word & 0xFFF8_F01F == 0xD500_401F {
951        return Some(Effect::Untouched); // barriers, MSR (immediate)
952    }
953    if word & 0xFFF0_0000 == 0xD530_0000 {
954        return Some(Effect::Untouched); // MRS: writes Xt
955    }
956    if word & 0xFFF0_0000 == 0xD510_0000 {
957        return Some(if rt {
958            Effect::PassesOn
959        } else {
960            Effect::Untouched
961        }); // MSR
962    }
963    if matches!(word & 0xFFFF_FFE0, 0xD50B_7A20 | 0xD50B_7520) {
964        // Cleaning a data cache line or invalidating an instruction cache
965        // line by its address (`DC CVAC`, `IC IVAU`) changes no data.
966        return Some(if rt {
967            Effect::Tested
968        } else {
969            Effect::Untouched
970        });
971    }
972    None
973}

975/// [`decoded_effect`] for the loads and stores (op0 `x1x0`) it decodes:
976/// literal loads; single registers with an unsigned, a 9-bit signed or a
977/// register offset; pairs (`STGP` scaled by 16); exclusive, ordered and
978/// compare-and-swap (`CASP` as its own family); the LSE atomics but the
979/// RCW forms, whose size field is not their width; and `LD64B`/`ST64B*`.
980fn load_store_effect(word: u32, register: u32) -> Option<Effect> {
981    let rn = (word >> 5) & 0x1F == register;
982    let rm = (word >> 16) & 0x1F == register;
983    let ra = (word >> 10) & 0x1F == register;
984    let rt = word & 0x1F == register;
985    let listed = |first: u32, count: u32| (0..count).any(|i| (first + i) & 0x1F == register);
986    let size = word >> 30;
987    let vector = (word >> 26) & 1 == 1;
988    let opc = (word >> 22) & 0b11;
989    // A single-register general-purpose store (opc 00) reads its transfer
990    // register; a vector one names a SIMD register there.
991    let stores_it = rt && !vector && opc == 0;
992    let width = if vector && size == 0 && opc & 0b10 != 0 {
993        16
994    } else {
995        1i64 << size
996    };
997    // The single-register forms allocated: a 128-bit vector access only at
998    // size 00, and no general-purpose opc 11 at size 10 or 11.
999    let single = if vector {
1000        size == 0 || opc & 0b10 == 0
1001    } else {
1002        !(opc == 0b11 && size >= 0b10)
1003    };
1004    if word & 0x3B00_0000 == 0x1800_0000 {
1005        // A literal load has no base; vector opc 11 is unallocated.
1006        return (!vector || size != 0b11).then_some(Effect::Untouched);
1007    }
1008    if word & 0x3B00_0000 == 0x3900_0000 && single {
1009        // Unsigned offset, scaled by the access size.
1010        if stores_it {
1011            return Some(Effect::PassesOn);
1012        }
1013        if !rn {
1014            return Some(Effect::Untouched);
1015        }
1016        let offset = i64::from((word >> 10) & 0xFFF) * width;
1017        return Some(Effect::Accesses {
1018            low: offset,
1019            high: offset + width,
1020            writeback: None,
1021        });
1022    }
1023    if word & 0x3B20_0000 == 0x3800_0000 && single {
1024        // A 9-bit signed offset: unscaled, unprivileged, pre- or post-index.
1025        if stores_it {
1026            return Some(Effect::PassesOn);
1027        }
1028        if !rn {
1029            return Some(Effect::Untouched);
1030        }
1031        let offset = i64::from(((((word >> 12) & 0x1FF) << 23) as i32) >> 23);
1032        return Some(match (word >> 10) & 0b11 {
1033            0b01 => Effect::Accesses {
1034                low: 0,
1035                high: width,
1036                writeback: Some(offset),
1037            },
1038            0b11 => Effect::Accesses {
1039                low: offset,
1040                high: offset + width,
1041                writeback: Some(offset),
1042            },
1043            _ => Effect::Accesses {
1044                low: offset,
1045                high: offset + width,
1046                writeback: None,
1047            },
1048        });
1049    }
1050    if word & 0x3B20_0C00 == 0x3820_0800 && single {
1051        // A register offset: not bounded, and its index is passed on.
1052        return Some(if stores_it || rm {
1053            Effect::PassesOn
1054        } else if rn {
1055            Effect::ReadsUnbounded
1056        } else {
1057            Effect::Untouched
1058        });
1059    }
1060    if word & 0x3B20_0C00 == 0x3820_0000 && !vector {
1061        let o3 = (word >> 15) & 1 == 1;
1062        let operation = (word >> 12) & 0b111;
1063        if ls64(word) {
1064            // ST64B, ST64BV and ST64BV0 store Xt to Xt+7; LD64B loads them.
1065            if operation != 0b101 && listed(word & 0x1F, 8) {
1066                return Some(Effect::PassesOn);
1067            }
1068            return Some(if rn {
1069                Effect::Accesses {
1070                    low: 0,
1071                    high: 64,
1072                    writeback: None,
1073                }
1074            } else {
1075                Effect::Untouched
1076            });
1077        }
1078        // LDADD and the other operations, SWP, and LDAPR (A set, R clear,
1079        // Rs 11111): Xs is stored, Xt receives the old value.
1080        let ldapr = operation == 0b100 && opc == 0b10 && (word >> 16) & 0x1F == 31;
1081        if o3 && operation != 0 && !ldapr {
1082            return None;
1083        }
1084        return Some(if rm {
1085            Effect::PassesOn
1086        } else if rn {
1087            Effect::Accesses {
1088                low: 0,
1089                high: width,
1090                writeback: None,
1091            }
1092        } else {
1093            Effect::Untouched
1094        });
1095    }
1096    if word & 0x3A00_0000 == 0x2800_0000 {
1097        // A pair: a 7-bit signed offset scaled by the element size, or by
1098        // 16 for `STGP`, which stores two 64-bit registers.
1099        let load = (word >> 22) & 1 == 1;
1100        let stgp = !vector && size == 0b01 && !load;
1101        let element: i64 = match (vector, size) {
1102            (true, 0b11) | (false, 0b11) => return None,
1103            (true, size) => 4 << size,
1104            (false, 0b00) => 4,
1105            (false, 0b10) => 8,
1106            (false, _) if stgp => 8,
1107            (false, _) => 4, // LDPSW
1108        };
1109        if !vector && size == 0b01 && (word >> 23) & 0b11 == 0 {
1110            return None; // no non-temporal STGP or LDPSW
1111        }
1112        if !load && !vector && (rt || ra) {
1113            return Some(Effect::PassesOn); // a value stored
1114        }
1115        if !rn {
1116            return Some(Effect::Untouched);
1117        }
1118        let scale = if stgp { 16 } else { element };
1119        let offset = i64::from(((((word >> 15) & 0x7F) << 25) as i32) >> 25) * scale;
1120        let (low, high) = (offset, offset + 2 * element);
1121        return Some(match (word >> 23) & 0b11 {
1122            0b01 => Effect::Accesses {
1123                low: 0,
1124                high: 2 * element,
1125                writeback: Some(offset),
1126            },
1127            0b11 => Effect::Accesses {
1128                low,
1129                high,
1130                writeback: Some(offset),
1131            },
1132            _ => Effect::Accesses {
1133                low,
1134                high,
1135                writeback: None,
1136            },
1137        });
1138    }
1139    if word & 0x3F00_0000 == 0x0800_0000 {
1140        let load = (word >> 22) & 1 == 1;
1141        let pair = (word >> 21) & 1 == 1;
1142        let compare = (word >> 23) & 1 == 1 && pair;
1143        if size < 0b10 && pair && !compare {
1144            // CASP: compares Xs:Xs+1 and stores Xt:Xt+1 at the base, then
1145            // writes the old value to Xs:Xs+1.
1146            if word & 0x7C00 != 0x7C00 {
1147                return None;
1148            }
1149            if listed(word & 0x1F, 2) {
1150                return Some(Effect::PassesOn);
1151            }
1152            return Some(if rn {
1153                Effect::Accesses {
1154                    low: 0,
1155                    high: 2 * (4 << size),
1156                    writeback: None,
1157                }
1158            } else if listed((word >> 16) & 0x1F, 2) {
1159                Effect::Tested
1160            } else {
1161                Effect::Untouched
1162            });
1163        }
1164        // Exclusive, ordered and compare-and-swap: at the base itself. A
1165        // compare-and-swap reads Rs and Rt, a store-exclusive Rt (and Rt2).
1166        if (compare && (rt || rm)) || (!compare && !load && (rt || (pair && ra))) {
1167            return Some(Effect::PassesOn);
1168        }
1169        if !rn {
1170            return Some(Effect::Untouched);
1171        }
1172        return Some(Effect::Accesses {
1173            low: 0,
1174            high: if pair || compare { 16 } else { width },
1175            writeback: None,
1176        });
1177    }
1178    None
1179}

1181/// Whether `word` writes `register`, ending the address it held: a form
1182/// [`decoded`] covers whose destination it is (a data-processing result
1183/// other than a flags-only form's, a load's transfer registers, a
1184/// store-exclusive's or compare-and-swap's status or old value, `MRS`'s
1185/// Xt). Any other encoding counts as not writing it, so the address stays
1186/// followed.
1187fn writes_register(word: u32, register: u32) -> bool {
1188    if !decoded(word) {
1189        return false;
1190    }
1191    let rd = word & 0x1F == register;
1192    let rm = (word >> 16) & 0x1F == register;
1193    let ra = (word >> 10) & 0x1F == register;
1194    let listed = |first: u32, count: u32| (0..count).any(|i| (first + i) & 0x1F == register);
1195    // Data processing forms that write only the flags, whose bits 4:0 are
1196    // not a destination: a conditional compare's are its NZCV immediate,
1197    // `RMIF`'s its mask, `SETF8` and `SETF16`'s fixed.
1198    let flags_only = word & 0x1FE0_0000 == 0x1A40_0000
1199        || word & 0xFFE0_7C10 == 0xBA00_0400
1200        || word & 0xFFFF_BC1F == 0x3A00_080D;
1201    if flags_only {
1202        return false;
1203    }
1204    if word & 0x1C00_0000 == 0x1000_0000 || word & 0x0E00_0000 == 0x0A00_0000 {
1205        return rd;
1206    }
1207    if system_space(word) {
1208        return word & 0xFFF0_0000 == 0xD530_0000 && rd; // MRS
1209    }
1210    let vector = (word >> 26) & 1 == 1;
1211    let opc = (word >> 22) & 0b11;
1212    if word & 0x3B20_0C00 == 0x3820_0000 && !vector {
1213        if ls64(word) {
1214            return match (word >> 12) & 0b111 {
1215                0b101 => listed(word & 0x1F, 8),
1216                0b010 | 0b011 => rm, // the status
1217                _ => false,
1218            };
1219        }
1220        return rd; // the old value
1221    }
1222    let load = if word & 0x3B00_0000 == 0x1800_0000 {
1223        word >> 30 != 0b11 // a literal load, not PRFM
1224    } else if word & 0x3A00_0000 == 0x2800_0000 {
1225        (word >> 22) & 1 == 1 // a pair
1226    } else if word & 0x3F00_0000 == 0x0800_0000 {
1227        let pair = (word >> 21) & 1 == 1;
1228        if (word >> 23) & 1 == 1 && pair {
1229            return rm; // compare-and-swap: the old value
1230        }
1231        if word >> 31 == 0 && pair {
1232            return listed((word >> 16) & 0x1F, 2); // CASP: the old value
1233        }
1234        if (word >> 22) & 1 == 0 {
1235            return rm; // the status
1236        }
1237        true
1238    } else {
1239        // A single register: opc 00 stores; PRFM (size 11, opc 10) loads
1240        // nothing.
1241        opc != 0 && !(word >> 30 == 0b11 && opc == 0b10)
1242    };
1243    // LDP and LDXP's second register.
1244    let pair = word & 0x3A00_0000 == 0x2800_0000
1245        || (word & 0x3F00_0000 == 0x0800_0000 && (word >> 21) & 1 == 1);
1246    !vector && load && (rd || (pair && ra))
1247}

1249/// Where `words` (at `base`) address bytes as data, or may: the target of
1250/// each `LDR` (literal) and `ADR`, and of each `ADRP` with every `ADD`
1251/// (immediate) or unsigned-offset load or store that completes it among the
1252/// eight instructions after it, taking the `ADRP`'s register as its base,
1253/// up to the instruction that writes that register without completing it,
1254/// since the page is gone from there. One `ADRP` can address several tables.
1255/// An `ADRP` nothing completes gives its page, the lowest address it can be
1256/// completing. Each is where a table may start: how far it runs is not in
1257/// the instruction ([`data_extents`]).
1258pub fn addressed_starts(words: &[u32], base: u64) -> Addressed {
1259    // How far after an `ADRP` its completing instruction is looked for.
1260    const PAIRING_WINDOW: usize = 8;
1261    let mut starts = Vec::new();
1262    let mut unresolved = Vec::new();
1263    for (index, word) in words.iter().enumerate() {
1264        let pc = base.wrapping_add(4 * index as u64);
1265        if word & 0x3B00_0000 == 0x1800_0000 && word >> 30 != 0b11 {
1266            // LDR (literal), not PRFM: its target is its literal's start.
1267            let offset = ((((word >> 5) & 0x7_FFFF) << 13) as i32 >> 11) as i64;
1268            starts.push(pc.wrapping_add(offset as u64));
1269        } else if word & 0x9F00_0000 == 0x1000_0000 {
1270            // ADR: immlo (30:29), immhi (23:5).
1271            let imm = ((word >> 5) & 0x7_FFFF) << 2 | ((word >> 29) & 0b11);
1272            let offset = ((imm << 11) as i32 >> 11) as i64;
1273            starts.push(pc.wrapping_add(offset as u64));
1274        } else if word & 0x9F00_0000 == 0x9000_0000 {
1275            // ADRP: the 4 KiB page of `pc`, plus immhi:immlo pages.
1276            let imm = ((word >> 5) & 0x7_FFFF) << 2 | ((word >> 29) & 0b11);
1277            let pages = ((imm << 11) as i32 >> 11) as i64;
1278            let page = (pc & !0xFFF).wrapping_add((pages << 12) as u64);
1279            let register = word & 0x1F;
1280            // Every completion, following the register: an `ADD` into the
1281            // register itself moves the base the later ones are relative
1282            // to, and any other write loses it, so the pairing ends there.
1283            let mut base = page;
1284            let mut completions = 0usize;
1285            for next in words[index + 1..].iter().take(PAIRING_WINDOW) {
1286                let offset = page_offset(*next, register);
1287                if let Some(offset) = offset {
1288                    starts.push(base.wrapping_add(offset));
1289                    completions += 1;
1290                    // `ADD Xn, Xn, #imm`: from here the register holds
1291                    // that address, so a later access is relative to it.
1292                    if *next & 0x1F == register && *next & 0xFF80_0000 == 0x9100_0000 {
1293                        base = base.wrapping_add(offset);
1294                        continue;
1295                    }
1296                }
1297                // A write that does not carry the base ends the
1298                // pairing. What the code addresses through the register
1299                // afterwards is then unknown; taking the whole page for
1300                // data there costs about half of a stripped library's
1301                // rewrites (measured), so it is left as a limit this
1302                // module's documentation states.
1303                if writes(*next, register) {
1304                    break;
1305                }
1306            }
1307            if completions == 0 {
1308                // Nothing in the window completes it, so all that is known
1309                // is the page: the caller takes the functions it overlaps
1310                // for data.
1311                unresolved.push(page);
1312            }
1313        }
1314    }
1315    Addressed { starts, unresolved }
1316}

1330/// The offset into an `ADRP`'s page that `word` adds, if it is an `ADD`
1331/// (immediate, 64-bit) or an unsigned-offset load or store whose base is
1332/// `register`.
1333fn page_offset(word: u32, register: u32) -> Option<u64> {
1334    if (word >> 5) & 0x1F != register {
1335        return None;
1336    }
1337    if word & 0xFF80_0000 == 0x9100_0000 {
1338        // ADD Xd, Xn, #imm12 {, LSL #12}: sh (22), imm12 (21:10).
1339        let imm = u64::from((word >> 10) & 0xFFF);
1340        return Some(if (word >> 22) & 1 == 1 {
1341            imm << 12
1342        } else {
1343            imm
1344        });
1345    }
1346    if word & 0x3B00_0000 == 0x3900_0000 {
1347        // LDR/STR (unsigned immediate): size (31:30), V (26), opc (23:22),
1348        // imm12 (21:10), scaled by the access size; a 128-bit SIMD access
1349        // is size 00 with opc 1x.
1350        let size = word >> 30;
1351        let scale = if (word >> 26) & 1 == 1 && size == 0 && (word >> 23) & 1 == 1 {
1352            4
1353        } else {
1354            size
1355        };
1356        return Some(u64::from((word >> 10) & 0xFFF) << scale);
1357    }
1358    None
1359}

1361/// Whether `word` writes `register`, as far as the shapes an `ADRP` pairing
1362/// must not look past: a `MOV`, an `ADD`/`SUB` or another `ADRP` or `ADR`
1363/// into it, or a load into it. An instruction this does not recognise is
1364/// taken not to write it, which only widens the pairing, and the pairing's
1365/// result is a data address, so a mistaken one costs coverage, never
1366/// safety.
1367fn writes(word: u32, register: u32) -> bool {
1368    if word & 0x1F != register {
1369        return false;
1370    }
1371    // ADR and ADRP; ADD/SUB (immediate and shifted register); MOV (wide
1372    // immediate) and ORR (shifted register), which `MOV` register is; and
1373    // the loads, whose destination is also bits 4:0. A store in those two
1374    // classes reads that register instead (`STR Xt, [SP]` spilling the
1375    // page), and `PRFM` loads nothing, so neither ends a pairing.
1376    let loads = |opc: u32| opc != 0b00 && opc != 0b11;
1377    word & 0x1F00_0000 == 0x1000_0000
1378        || word & 0x1F00_0000 == 0x1100_0000
1379        || word & 0x1F00_0000 == 0x0B00_0000
1380        || word & 0x1F80_0000 == 0x1280_0000
1381        || word & 0x1F00_0000 == 0x0A00_0000
1382        || (word & 0x3B00_0000 == 0x3900_0000 && (word >> 22) & 1 == 1)
1383        || (word & 0x3B00_0000 == 0x1800_0000 && loads(word >> 30))
1384}

1398/// The bytes a function may hold as data, where only the unwind table tells
1399/// code from data: from each address the code addresses as data
1400/// (`starts`, [`addressed_starts`]) inside a function (`functions`, the
1401/// FDEs' ranges, sorted) to the next instruction a branch reaches
1402/// (`branches`, as [`direct_branches`] gives them), or the function's end.
1403/// Control does not run into a table, so the code after one is reached by a
1404/// branch, and only a branch that is not itself inside a table counts: a
1405/// table word can decode as one. An address at a function's entry is a
1406/// pointer to the function, not data. The ranges come back sorted and merged.
1407pub fn data_extents(
1408    starts: &[u64],
1409    functions: &[(u64, u64)],
1410    branches: &[(u64, u64)],
1411) -> Vec<(u64, u64)> {
1412    // FDE ranges may nest: a short range inside a longer one does not end
1413    // the data at its own end, so every containing range contributes an
1414    // extent and the merge below keeps the longest.
1415    let mut extents: Vec<(u64, u64)> = starts
1416        .iter()
1417        .flat_map(|start| {
1418            let start = *start;
1419            let after = functions.partition_point(|(entry, _)| *entry <= start);
1420            functions[..after]
1421                .iter()
1422                .filter(move |(entry, end)| *entry < start && start < *end)
1423                .map(move |(_, end)| (start, *end))
1424        })
1425        .collect();
1426    extents.sort_unstable();
1427    extents.dedup();
1428    // Only a branch whose target lies in an extent can end one, and an
1429    // extent only shrinks, so the rest are dropped once: on a large
1430    // stripped library this is thousands of branches instead of millions,
1431    // round after round, for the same result.
1432    let initial = merge(extents.clone());
1433    let reaches = |addr: u64| {
1434        let after = initial.partition_point(|(start, _)| *start <= addr);
1435        after.checked_sub(1).is_some_and(|i| addr < initial[i].1)
1436    };
1437    let branches: Vec<(u64, u64)> = branches
1438        .iter()
1439        .filter(|(_, target)| reaches(*target))
1440        .copied()
1441        .collect();
1442    // Each extent ends at the first target past its start of a branch
1443    // outside every extent. Extents only shrink, and a branch they no
1444    // longer hold can only shrink them further, so this ends.
1445    loop {
1446        let merged = merge(extents.clone());
1447        let inside = |addr: u64| {
1448            let after = merged.partition_point(|(start, _)| *start <= addr);
1449            after.checked_sub(1).is_some_and(|i| addr < merged[i].1)
1450        };
1451        let mut targets: Vec<u64> = branches
1452            .iter()
1453            .filter(|(at, _)| !inside(*at))
1454            .map(|(_, target)| *target)
1455            .collect();
1456        targets.sort_unstable();
1457        let mut changed = false;
1458        for (start, end) in &mut extents {
1459            let next = targets.partition_point(|target| *target <= *start);
1460            if let Some(target) = targets.get(next).filter(|target| **target < *end) {
1461                *end = *target;
1462                changed = true;
1463            }
1464        }
1465        if !changed {
1466            return merged;
1467        }
1468    }
1469}

1471/// `spans` sorted and with overlapping or touching ones joined.
1472fn merge(mut spans: Vec<(u64, u64)>) -> Vec<(u64, u64)> {
1473    spans.sort_unstable();
1474    let mut out: Vec<(u64, u64)> = Vec::new();
1475    for (start, end) in spans {
1476        match out.last_mut() {
1477            Some(last) if start <= last.1 => last.1 = last.1.max(end),
1478            _ => out.push((start, end)),
1479        }
1480    }
1481    out
1482}

1529/// The local, untyped symbols of a symbol table that mark a section, given
1530/// the string table it names, as (section index, address, name). A mapping
1531/// symbol is one of them ("ELF for the Arm 64-bit Architecture", 5.5.4: they
1532/// are `STT_NOTYPE` and `STB_LOCAL`, and the `$` names are reserved for local
1533/// symbols). A global symbol whose name begins `$x` is an ordinary label:
1534/// taking it for a mapping symbol would let one inside a `$d` table start a
1535/// code range there.
1536fn local_untyped_symbols<'a>(
1537    symtab: &'a [u8],
1538    strtab: &'a [u8],
1539) -> impl Iterator<Item = (usize, u64, &'a [u8])> + 'a {
1540    symtab
1541        .as_chunks::<SYMBOL_SIZE>()
1542        .0
1543        .iter()
1544        .filter_map(move |symbol| {
1545            let name =
1546                usize::try_from(u32::from_le_bytes(symbol[0..4].try_into().unwrap())).ok()?;
1547            let section = u16::from_le_bytes(symbol[6..8].try_into().unwrap());
1548            if symbol[4] & 0xf != STT_NOTYPE
1549                || symbol[4] >> 4 != STB_LOCAL
1550                || section == 0
1551                || section >= SHN_LORESERVE
1552            {
1553                return None;
1554            }
1555            let name = strtab.get(name..)?;
1556            let name = &name[..name.iter().position(|byte| *byte == 0)?];
1557            Some((
1558                usize::from(section),
1559                u64::from_le_bytes(symbol[8..16].try_into().unwrap()),
1560                name,
1561            ))
1562        })
1563}

1598/// The code of section `section`, which starts at `address` and is `size`
1599/// bytes long, as address ranges, given `symbols` sorted as
1600/// [`mapping_symbols`] returns them: each `$x` to the next mapping symbol of
1601/// the section, or to its end. Bytes before the section's first mapping
1602/// symbol are not code, and a mark outside the section is ignored. `None`
1603/// when the section has no mapping symbol, so its code cannot be told from
1604/// its data.
1605pub fn code_spans(
1606    symbols: &[MappingSymbol],
1607    section: usize,
1608    address: u64,
1609    size: u64,
1610) -> Option<Vec<(u64, u64)>> {
1611    let end = address.saturating_add(size);
1612    // A mark outside its own section's bytes says nothing of them.
1613    let marks: Vec<&MappingSymbol> = symbols
1614        .iter()
1615        .filter(|symbol| symbol.section == section && (address..end).contains(&symbol.address))
1616        .collect();
1617    if marks.is_empty() {
1618        return None;
1619    }
1620    let mut spans: Vec<(u64, u64)> = Vec::new();
1621    for (i, mark) in marks.iter().enumerate() {
1622        if !mark.code {
1623            continue;
1624        }
1625        let start = mark.address;
1626        let stop = marks.get(i + 1).map_or(end, |next| next.address);
1627        if start == stop {
1628            continue;
1629        }
1630        // Adjacent `$x` marks continue one span.
1631        match spans.last_mut() {
1632            Some(last) if last.1 == start => last.1 = stop,
1633            _ => spans.push((start, stop)),
1634        }
1635    }
1636    Some(spans)
1637}

1653    /// An unwind table at `address` with one CIE (augmentation "zR",
1654    /// pc-relative signed 4-byte pointers) and an FDE for each function
1655    /// (start, end), ended by a zero terminator.
1656    pub(crate) fn eh_frame(address: u64, functions: &[(u64, u64)]) -> Vec<u8> {
1657        let mut bytes = Vec::new();
1658        // CIE: length 16; ID 0; version 1; "zR"; code alignment 4; data
1659        // alignment -8; return register 30; augmentation data: 0x1b; nops.
1660        bytes.extend(16u32.to_le_bytes());
1661        bytes.extend([
1662            0, 0, 0, 0, 1, b'z', b'R', 0, 0x04, 0x78, 0x1e, 0x01, 0x1b, 0, 0, 0,
1663        ]);
1664        for (start, end) in functions {
1665            let at = bytes.len();
1666            bytes.extend(16u32.to_le_bytes());
1667            // The CIE pointer: from this field back to the CIE at 0.
1668            bytes.extend(((at + 4) as u32).to_le_bytes());
1669            let here = address + (at + 8) as u64;
1670            bytes.extend((start.wrapping_sub(here) as u32).to_le_bytes());
1671            bytes.extend(((end - start) as u32).to_le_bytes());
1672            bytes.extend([0, 0, 0, 0]); // no augmentation data; nops
1673        }
1674        bytes.extend(0u32.to_le_bytes());
1675        bytes
1676    }

1878        }

1880            found.extend([15, 16, 17]);
1881        }

1884    }
