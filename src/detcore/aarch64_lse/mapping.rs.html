9//! Telling code from data in an aarch64 ELF file's executable sections, by
10//! the file's mapping symbols.
11//!
12//! An executable section can hold data as well as instructions: OpenSSL's
13//! generated assembly, for one, keeps its constant tables in `.text`. Those
14//! words decode as instructions as readily as code does, so a scan that reads
15//! the whole section finds load/store-exclusive "loops" that never execute,
16//! and a rewrite that took one would corrupt the table. The AArch64 ELF ABI
17//! marks the boundaries with mapping symbols in the symbol table: `$x` (or
18//! `$x.<any>`) starts A64 code and `$d` (or `$d.<any>`) starts data, each
19//! running to the next mapping symbol of the same section ("ELF for the Arm
20//! 64-bit Architecture", section 5.5.4, Mapping symbols). A stripped file has
21//! no symbol table and so no mapping symbols, and a section can lack them too.
22//!
23//! Where a section has no mapping symbols, the file's unwind table
24//! (`.eh_frame`) is the evidence instead: each frame description entry (FDE)
25//! covers one function's code, stripped files keep the table for unwinding,
26//! and a constant table has no FDE. An FDE covers its whole function, so a
27//! table or literal pool placed inside the function is inside it too: under
28//! that evidence alone, code is what control reaches from the functions'
29//! entries ([`reachable_code`]), less what the code addresses as data, up
30//! to the next instruction a branch reaches ([`data_extents`]), since a
31//! call is taken to return; and a sequence a literal load reaches is
32//! refused in any case ([`literal_targets`]). Bytes neither marks are not
33//! known to be code.
34//!
35//! These are pure functions of the file's bytes, which the rewriter
36//! ([`super::elf`]) consults before it plans a single sequence.

55/// What tells an ELF file's code from its data: its mapping symbols, and the
56/// code ranges its unwind table's FDEs cover.
57#[derive(Clone, Debug, Default, PartialEq, Eq)]
58pub struct CodeEvidence {
59    /// The mapping symbols, sorted as [`mapping_symbols`] returns them.
60    pub symbols: Vec<MappingSymbol>,
61    /// The local, untyped symbols named like mapping symbols that are neither
62    /// `$x` nor `$d` ([`unsupported_mapping_symbols`]), as (section index,
63    /// name). A section that has one is not proved by its mapping symbols.
64    pub unsupported: Vec<(usize, String)>,
65    /// The address range of each FDE, sorted; `None` without an unwind
66    /// table, or with one this reader cannot decode.
67    pub unwind: Option<Vec<(u64, u64)>>,
68}

79impl CodeEvidence {
80    /// The evidence of an ELF64 little-endian file whose section header
81    /// table is `table` and whose section-name string table is section
82    /// `shstrndx`. `read` returns the file bytes at (offset, size), or
83    /// `None` when it will not (out of the file, or larger than the caller
84    /// reads); a table it does not return is evidence the file lacks.
85    ///
86    /// The unwind table is read only for a loaded image (`e_type` `ET_EXEC`
87    /// or `ET_DYN`): in a relocatable object each FDE's start is a
88    /// relocation's to fill, so the bytes alone would place every function
89    /// near address 0.
90    pub fn read(
91        table: &[u8],
92        shstrndx: usize,
93        e_type: u16,
94        mut read: impl FnMut((u64, u64)) -> Option<Vec<u8>>,
95    ) -> Self {
96        const ET_EXEC: u16 = 2;
97        const ET_DYN: u16 = 3;
98        let (symbols, unsupported) = symbol_table(table)
99            .and_then(|(symtab, strtab)| {
100                let (symtab, strtab) = (read(symtab)?, read(strtab)?);
101                Some((
102                    mapping_symbols(&symtab, &strtab),
103                    unsupported_mapping_symbols(&symtab, &strtab),
104                ))
105            })
106            .unwrap_or_default();
107        let unwind = matches!(e_type, ET_EXEC | ET_DYN)
108            .then(|| section_range(table, shstrndx))
109            .flatten()
110            .and_then(&mut read)
111            .and_then(|names| section_named(table, &names, b".eh_frame"))
112            .and_then(|(range, address)| fde_ranges(&read(range)?, address));
113        CodeEvidence {
114            symbols,
115            unsupported,
116            unwind,
117        }
118    }
119
120    /// The first mapping symbol section `section` holds that is neither `$x`
121    /// nor `$d`, if any.
122    pub fn unsupported_in(&self, section: usize) -> Option<&str> {
123        self.unsupported
124            .iter()
125            .find(|(marked, _)| *marked == section)
126            .map(|(_, name)| name.as_str())
127    }

129    /// What section `section` (at `address`, `size` bytes long) holds: by its
130    /// mapping symbols when it has any ([`code_spans`]), or else by the
131    /// FDEs' ranges within it.
132    pub fn classify(&self, section: usize, address: u64, size: u64) -> SectionCode {
133        let end = address.saturating_add(size);
134        let complement = |covered: &[(u64, u64)]| {
135            let mut gaps = Vec::new();
136            let mut at = address;
137            for (start, stop) in covered {
138                if at < *start {
139                    gaps.push((at, *start));
140                }
141                at = at.max(*stop);
142            }
143            if at < end {
144                gaps.push((at, end));
145            }
146            gaps
147        };
148        if let Some(spans) = code_spans(&self.symbols, section, address, size) {
149            // `$d` ranges are data; only the bytes before the first mark are
150            // unknown.
151            let first = self
152                .symbols
153                .iter()
154                .filter(|symbol| symbol.section == section)
155                .map(|symbol| symbol.address)
156                .find(|mark| (address..end).contains(mark))
157                .unwrap_or(end);
158            return SectionCode {
159                code: spans,
160                unknown: complement(&[(first, end)]),
161                evidence: Some(Evidence::MappingSymbols),
162            };
163        }
164        let Some(unwind) = &self.unwind else {
165            return SectionCode {
166                code: Vec::new(),
167                unknown: complement(&[]),
168                evidence: None,
169            };
170        };
171        let mut code: Vec<(u64, u64)> = Vec::new();
172        for (start, stop) in unwind
173            .iter()
174            .map(|(start, stop)| ((*start).max(address), (*stop).min(end)))
175            .filter(|(start, stop)| start < stop)
176        {
177            match code.last_mut() {
178                Some(last) if start <= last.1 => last.1 = last.1.max(stop),
179                _ => code.push((start, stop)),
180            }
181        }
182        // No FDE is not evidence of data: what no FDE covers is unknown.
183        SectionCode {
184            unknown: complement(&code),
185            code,
186            evidence: Some(Evidence::UnwindTable),
187        }
188    }
189}

205/// Which of `functions` (sorted by entry, as the FDEs give them) can
206/// return, by [`can_return`] over and over until the answer stops
207/// growing: a function returns if a `RET` is reachable in it, and a call
208/// it makes comes back only if that callee is already known to return and
209/// `never` does not name it as one that cannot.
210/// Each round can only add functions, so a few settle it; the bound keeps
211/// a deep chain from costing more than it is worth, and stopping early
212/// leaves a function unproven, which is the safe side.
213pub fn returning_functions(
214    words: &[u32],
215    base: u64,
216    functions: &[(u64, u64)],
217    never: &dyn Fn(u64) -> bool,
218) -> std::collections::HashSet<u64> {
219    // Chains deeper than this stay unproven.
220    const ROUNDS: usize = 8;
221    let mut returning: std::collections::HashSet<u64> = std::collections::HashSet::new();
222    for _ in 0..ROUNDS {
223        let known = returning.clone();
224        // A call out of this section is taken to return, the ordinary
225        // case; one to a function here only where it is already proven.
226        let returns = |target: u64| {
227            !never(target)
228                && (known.contains(&target)
229                    || functions
230                        .binary_search_by_key(&target, |(entry, _)| *entry)
231                        .is_err())
232        };
233        let grown: Vec<u64> = functions
234            .iter()
235            .filter(|(entry, _)| !returning.contains(entry))
236            .filter(|function| can_return(words, base, **function, &returns))
237            .map(|(entry, _)| *entry)
238            .collect();
239        if grown.is_empty() {
240            break;
241        }
242        returning.extend(grown);
243    }
244    returning
245}

247/// Whether the function `function` (its entry and end, as an FDE gives
248/// them) can return: walking its own words from its entry, some
249/// instruction it reaches is a `RET X30`, or a direct branch out of it (a
250/// tail call) goes to a function that `returns` says comes back. A `RET`
251/// through another register is not a return to this function's caller, and
252/// an `SVC` proves nothing past it, whatever its immediate, since what the
253/// kernel does with it is not in the file. What this does not check is a
254/// write to `X30` before a `RET X30`, so a function that returns somewhere
255/// else through a changed link register is taken to return to its caller,
256/// and an unresolved `BLR`, whose callee is unknown, is taken to come
257/// back. A function
258/// whose entry is outside `words` is taken to return, which is the
259/// ordinary case for a call out of the section. A function whose every path ends in
260/// a trap or in a syscall that does not come back (an exit or abort
261/// helper) cannot, and a call to it does not fall through.
262pub fn can_return(
263    words: &[u32],
264    base: u64,
265    function: (u64, u64),
266    returns: &dyn Fn(u64) -> bool,
267) -> bool {
268    // `RET X30`, and its authenticated forms `RETAA` and `RETAB`: a
269    // return through another register is not a return to this function's
270    // caller.
271    const RET_X30: [u32; 3] = [0xD65F_03C0, 0xD65F_0BFF, 0xD65F_0FFF];
272    // `SVC #imm`: Linux ignores the immediate, and what the kernel does
273    // with the call is not in the file, so nothing past it is proof.
274    let is_svc = |word: u32| word & 0xFFE0_001F == 0xD400_0001;
275    let (entry, end) = function;
276    let index = |addr: u64| -> Option<usize> {
277        (entry..end)
278            .contains(&addr)
279            .then(|| usize::try_from((addr - base) / 4).ok())
280            .flatten()
281            .filter(|index| *index < words.len())
282    };
283    let Some(first) = index(entry) else {
284        return true;
285    };
286    let last = index(end.saturating_sub(4)).unwrap_or(first);
287    let mut seen = vec![false; last - first + 1];
288    let mut pending = vec![first];
289    while let Some(mut at) = pending.pop() {
290        while !seen[at - first] {
291            seen[at - first] = true;
292            if RET_X30.contains(&words[at]) {
293                return true;
294            }
295            if is_svc(words[at]) {
296                break;
297            }
298            let pc = base + 4 * at as u64;
299            let (falls_through, target) = flow(words[at], pc);
300            // A call comes back only where `returns` already says so, so
301            // the answer only grows as more functions are known to return
302            // ([`returning_functions`]).
303            let call = matches!(decode(words[at]), Insn::Branch { link: true, .. });
304            let comes_back = !call || target.is_some_and(returns);
305            // A tail call, a direct branch out of this function, returns
306            // to this function's caller when the function it jumps to
307            // returns at all.
308            if !call
309                && let Some(target) = target
310                && index(target).is_none()
311                && returns(target)
312            {
313                return true;
314            }
315            if let Some(target) = target.filter(|_| !call).and_then(index) {
316                pending.push(target);
317            }
318            match (falls_through && comes_back)
319                .then(|| index(pc + 4))
320                .flatten()
321            {
322                Some(next) => at = next,
323                None => break,
324            }
325        }
326    }
327    false
328}

330/// The code among `words` (at `base`) that control can reach from
331/// `entries` without leaving `covered` (sorted, disjoint): following each
332/// instruction to the next unless it is an unconditional branch, a return,
333/// an indirect jump or a trap, and each direct branch to its target. Under
334/// the unwind table alone this is what is taken for code: an FDE covers its
335/// whole function, including a constant table the function branches over or
336/// keeps after its return, and those bytes are not reached. Code reached
337/// only through a jump table or an exception landing pad is not either, and
338/// is left alone with them. A call falls through only where `returns`
339/// proves the callee comes back; [`data_extents`] takes back what the code
340/// addresses as data.
341pub fn reachable_code(
342    words: &[u32],
343    base: u64,
344    covered: &[(u64, u64)],
345    entries: &[u64],
346    returns: &dyn Fn(u64) -> bool,
347) -> Vec<(u64, u64)> {
348    let index_of = |addr: u64| -> Option<usize> {
349        if addr < base || !addr.is_multiple_of(4) {
350            return None;
351        }
352        let index = usize::try_from((addr - base) / 4).ok()?;
353        let after = covered.partition_point(|(start, _)| *start <= addr);
354        let inside = after.checked_sub(1).is_some_and(|i| addr < covered[i].1);
355        (index < words.len() && inside).then_some(index)
356    };
357    let mut seen = vec![false; words.len()];
358    let mut pending: Vec<usize> = entries
359        .iter()
360        .filter_map(|entry| index_of(*entry))
361        .collect();
362    while let Some(mut index) = pending.pop() {
363        while !seen[index] {
364            seen[index] = true;
365            let pc = base + 4 * index as u64;
366            let (mut falls_through, target) = flow(words[index], pc);
367            // A call comes back only if the function it calls can.
368            if matches!(decode(words[index]), Insn::Branch { link: true, .. })
369                && target.is_some_and(|target| !returns(target))
370            {
371                falls_through = false;
372            }
373            if let Some(target) = target.and_then(index_of) {
374                pending.push(target);
375            }
376            match falls_through.then(|| index_of(pc + 4)).flatten() {
377                Some(next) => index = next,
378                None => break,
379            }
380        }
381    }
382    let mut spans: Vec<(u64, u64)> = Vec::new();
383    for (index, _) in seen.iter().enumerate().filter(|(_, seen)| **seen) {
384        let addr = base + 4 * index as u64;
385        match spans.last_mut() {
386            Some(last) if last.1 == addr => last.1 = addr + 4,
387            _ => spans.push((addr, addr + 4)),
388        }
389    }
390    spans
391}

393/// Where control goes after `word` at `pc`: whether it can fall through to
394/// the next instruction, and the target of a direct branch it can take. A
395/// conditional branch whose outcome the encoding fixes counts as what it
396/// does: `B.AL` and `B.NV` always branch, and the zero register makes `CBZ`
397/// and `TBZ` always branch and `CBNZ` and `TBNZ` never.
398fn flow(word: u32, pc: u64) -> (bool, Option<u64>) {
399    const ZR: super::insn::Reg = 31;
400    let target = |offset: i64| Some(pc.wrapping_add(offset as u64));
401    match decode(word) {
402        Insn::Branch {
403            link: false,
404            offset,
405        } => (false, target(offset)),
406        Insn::CondBranch {
407            cond: 0b1110 | 0b1111,
408            offset,
409        } => (false, target(offset)),
410        Insn::CompareBranch {
411            rt: ZR,
412            nonzero,
413            offset,
414            ..
415        } => {
416            if nonzero {
417                (true, None)
418            } else {
419                (false, target(offset))
420            }
421        }
422        // TBZ/TBNZ: op (24), Rt (4:0).
423        Insn::TestBranch { offset } if word & 0x1f == u32::from(ZR) => {
424            if (word >> 24) & 1 == 1 {
425                (true, None)
426            } else {
427                (false, target(offset))
428            }
429        }
430        Insn::Branch { link: true, offset }
431        | Insn::CondBranch { offset, .. }
432        | Insn::CompareBranch { offset, .. }
433        | Insn::TestBranch { offset } => (true, target(offset)),
434        // Unconditional branch (register): BLR and its authenticated forms
435        // return; BR, RET, ERET and theirs do not.
436        _ if word & 0xFE00_0000 == 0xD600_0000 => {
437            (matches!((word >> 21) & 0xF, 0b0001 | 0b1001), None)
438        }
439        // UDF, BRK and HLT trap.
440        _ if word & 0xFFFF_0000 == 0
441            || word & 0xFFE0_001F == 0xD420_0000
442            || word & 0xFFE0_001F == 0xD440_0000 =>
443        {
444            (false, None)
445        }
446        _ => (true, None),
447    }
448}

450/// What one executable section holds, as address ranges, by
451/// [`CodeEvidence::classify`]. What is in neither list is data (`$d`).
452#[derive(Clone, Debug, PartialEq, Eq)]
453pub struct SectionCode {
454    /// Known to be code.
455    pub code: Vec<(u64, u64)>,
456    /// Neither known code nor known data: bytes before a section's first
457    /// mapping symbol, bytes no FDE covers, or the whole section when
458    /// nothing speaks for it.
459    pub unknown: Vec<(u64, u64)>,
460    /// What `code` was found by; `None` when nothing speaks for the section.
461    pub evidence: Option<Evidence>,
462}

464/// The file range (offset, size) of section `index` of a section header
465/// table.
466fn section_range(table: &[u8], index: usize) -> Option<(u64, u64)> {
467    let entry = table.as_chunks::<SECTION_HEADER_SIZE>().0.get(index)?;
468    Some((
469        u64::from_le_bytes(entry[24..32].try_into().unwrap()),
470        u64::from_le_bytes(entry[32..40].try_into().unwrap()),
471    ))
472}

474/// The file range and address of the section named `name`, given the
475/// section-name string table's bytes.
476fn section_named(table: &[u8], names: &[u8], name: &[u8]) -> Option<((u64, u64), u64)> {
477    table
478        .as_chunks::<SECTION_HEADER_SIZE>()
479        .0
480        .iter()
481        .find(|entry| {
482            let at = u32::from_le_bytes(entry[0..4].try_into().unwrap()) as usize;
483            names
484                .get(at..)
485                .and_then(|rest| rest.get(..name.len() + 1))
486                .is_some_and(|candidate| {
487                    candidate[..name.len()] == *name && candidate[name.len()] == 0
488                })
489        })
490        .map(|entry| {
491            (
492                (
493                    u64::from_le_bytes(entry[24..32].try_into().unwrap()),
494                    u64::from_le_bytes(entry[32..40].try_into().unwrap()),
495                ),
496                u64::from_le_bytes(entry[16..24].try_into().unwrap()),
497            )
498        })
499}

501/// A cursor over the bytes of an unwind table.
502struct Cursor<'a> {
503    bytes: &'a [u8],
504    at: usize,
505}
506
507impl Cursor<'_> {
508    fn take(&mut self, len: usize) -> Option<&[u8]> {
509        let bytes = self.bytes.get(self.at..self.at.checked_add(len)?)?;
510        self.at += len;
511        Some(bytes)
512    }
513
514    fn u8(&mut self) -> Option<u8> {
515        Some(self.take(1)?[0])
516    }
517
518    fn uleb(&mut self) -> Option<u64> {
519        let mut value = 0u64;
520        for shift in (0..64).step_by(7) {
521            let byte = self.u8()?;
522            value |= u64::from(byte & 0x7f) << shift;
523            if byte & 0x80 == 0 {
524                return Some(value);
525            }
526        }
527        None
528    }
529
530    fn sleb(&mut self) -> Option<i64> {
531        let mut value = 0i64;
532        let mut shift = 0;
533        loop {
534            let byte = self.u8()?;
535            value |= i64::from(byte & 0x7f) << shift;
536            shift += 7;
537            if byte & 0x80 == 0 {
538                if shift < 64 && byte & 0x40 != 0 {
539                    value |= -1i64 << shift;
540                }
541                return Some(value);
542            }
543            if shift >= 64 {
544                return None;
545            }
546        }
547    }
548
549    /// A pointer in `encoding` (DW_EH_PE_*), at address `here` of the
550    /// cursor's position. Only absolute and pc-relative values are read;
551    /// any other application, or an indirect one, is `None`.
552    fn pointer(&mut self, encoding: u8, here: u64) -> Option<u64> {
553        let value = self.raw_pointer(encoding)?;
554        match encoding & 0xf0 {
555            0x00 => Some(value),
556            0x10 => Some(here.wrapping_add(value)),
557            _ => None,
558        }
559    }
560
561    /// A pointer's value as stored in `encoding`'s format, before any
562    /// application: how a pointer this reader does not need (a
563    /// personality routine's, usually indirect) is stepped over.
564    fn raw_pointer(&mut self, encoding: u8) -> Option<u64> {
565        Some(match encoding & 0x0f {
566            0x00 | 0x04 => u64::from_le_bytes(self.take(8)?.try_into().unwrap()),
567            0x01 => self.uleb()?,
568            0x02 => u64::from(u16::from_le_bytes(self.take(2)?.try_into().unwrap())),
569            0x03 => u64::from(u32::from_le_bytes(self.take(4)?.try_into().unwrap())),
570            0x09 => self.sleb()? as u64,
571            0x0a => i16::from_le_bytes(self.take(2)?.try_into().unwrap()) as u64,
572            0x0b => i32::from_le_bytes(self.take(4)?.try_into().unwrap()) as u64,
573            0x0c => u64::from_le_bytes(self.take(8)?.try_into().unwrap()),
574            _ => return None,
575        })
576    }
577}

579/// The FDE pointer encoding of a CIE, given its body after the CIE ID:
580/// `DW_EH_PE_absptr` unless its augmentation names one with `R`. `None` for
581/// a version other than 1 or 3 (the two `.eh_frame` uses), an augmentation
582/// this reader does not know, or a body that ends early.
583fn cie_pointer_encoding(body: &[u8]) -> Option<u8> {
584    let mut cie = Cursor { bytes: body, at: 0 };
585    let version = cie.u8()?;
586    if !matches!(version, 1 | 3) {
587        return None;
588    }
589    let augmentation_start = cie.at;
590    while cie.u8()? != 0 {}
591    let augmentation = body[augmentation_start..cie.at - 1].to_vec();
592    if augmentation.windows(2).any(|pair| pair == b"eh") {
593        cie.take(8)?;
594    }
595    cie.uleb()?; // code alignment
596    cie.sleb()?; // data alignment
597    if version == 1 {
598        cie.u8()?;
599    } else {
600        cie.uleb()?;
601    }
602    if augmentation.first() != Some(&b'z') {
603        return Some(0);
604    }
605    cie.uleb()?; // augmentation data length
606    for letter in &augmentation[1..] {
607        match letter {
608            b'R' => return cie.u8(),
609            b'L' => {
610                cie.u8()?;
611            }
612            b'P' => {
613                let encoding = cie.u8()?;
614                // An aligned pointer would need padding this reader does
615                // not track.
616                if encoding & 0x70 == 0x50 {
617                    return None;
618                }
619                cie.raw_pointer(encoding)?;
620            }
621            b'S' | b'B' => {}
622            _ => return None,
623        }
624    }
625    Some(0)
626}

628/// The address range of each FDE of an unwind table (`.eh_frame`, at
629/// `address`), sorted. `None` when the table cannot be read to its end: a
630/// reader that skipped an FDE it could not decode would take that
631/// function's code for data, which is safe, but one that misread a length
632/// could take data for code.
633pub fn fde_ranges(bytes: &[u8], address: u64) -> Option<Vec<(u64, u64)>> {
634    let mut ranges = Vec::new();
635    // The FDE pointer encoding of each CIE walked so far, by the offset of
636    // its length field: an FDE must name one of these.
637    let mut encodings = std::collections::HashMap::new();
638    let mut at = 0usize;
639    while at + 4 <= bytes.len() {
640        let mut header = Cursor { bytes, at };
641        let length = u32::from_le_bytes(header.take(4)?.try_into().unwrap());
642        if length == 0 {
643            break;
644        }
645        let length = if length == u32::MAX {
646            u64::from_le_bytes(header.take(8)?.try_into().unwrap())
647        } else {
648            u64::from(length)
649        };
650        let end = header.at.checked_add(usize::try_from(length).ok()?)?;
651        if end > bytes.len() {
652            return None;
653        }
654        // Each entry is read within its own length.
655        let mut entry = Cursor {
656            bytes: &bytes[..end],
657            at: header.at,
658        };
659        let id_at = entry.at;
660        let id = u32::from_le_bytes(entry.take(4)?.try_into().unwrap());
661        if id == 0 {
662            encodings.insert(at, cie_pointer_encoding(&bytes[entry.at..end])?);
663        } else {
664            // An FDE: its CIE is `id` bytes before the field.
665            let cie = id_at.checked_sub(usize::try_from(id).ok()?)?;
666            let encoding = *encodings.get(&cie)?;
667            let start = entry.pointer(encoding, address.wrapping_add(entry.at as u64))?;
668            let size = entry.pointer(encoding & 0x0f, 0)?;
669            if size != 0 {
670                ranges.push((start, start.checked_add(size)?));
671            }
672        }
673        at = end;
674    }
675    ranges.sort_unstable();
676    Some(ranges)
677}

679/// The byte ranges `words` (at `base`) load as literals or take the address
680/// of: each `LDR` (literal) target with its load's size, and each `ADR`
681/// target as one word, since what follows it is not known. A literal pool
682/// inside a function is read this way, so a candidate sequence any of these
683/// reach is data the unwind table took for code.
684pub fn literal_targets(words: &[u32], base: u64) -> Vec<(u64, u64)> {
685    literal_references(words, base)
686        .into_iter()
687        .map(|reference| reference.range)
688        .collect()
689}

694    Address,

702/// Keep address-taking distinct from a read of the instruction bytes: only
703/// the former can name a function entry without observing a rewrite there.
704pub(super) fn literal_references(words: &[u32], base: u64) -> Vec<LiteralReference> {
705    words
706        .iter()
707        .enumerate()
708        .filter_map(|(index, word)| {
709            let pc = base.wrapping_add(4 * index as u64);
710            if word & 0x3B00_0000 == 0x1800_0000 {
711                // LDR (literal): opc (31:30), V (26), imm19 (23:5).
712                let size = match (word >> 30, (word >> 26) & 1) {
713                    (0b00, _) | (0b10, 0) => 4,
714                    (0b01, _) => 8,
715                    (0b10, _) => 16,
716                    _ => return None, // PRFM (literal) loads nothing
717                };
718                let offset = ((((word >> 5) & 0x7_FFFF) << 13) as i32 >> 11) as i64;
719                let target = pc.wrapping_add(offset as u64);
720                Some(LiteralReference {
721                    range: (target, target.wrapping_add(size)),
722                    kind: LiteralKind::MemoryRead,
723                })
724            } else if word & 0x9F00_0000 == 0x1000_0000 {
725                // ADR: immlo (30:29), immhi (23:5).
726                let imm = ((word >> 5) & 0x7_FFFF) << 2 | ((word >> 29) & 0b11);
727                let offset = ((imm << 11) as i32 >> 11) as i64;
728                let target = pc.wrapping_add(offset as u64);
729                Some(LiteralReference {
730                    range: (target, target.wrapping_add(4)),
731                    kind: LiteralKind::Address,
732                })
733            } else {
734                None
735            }
736        })
737        .collect()
738}

740/// What an `ADRP` or `ADR` lets the code after it read or write through its
741/// register, for the image-wide reader scan ([`literal_references`] covers
742/// the `LDR` (literal) and the `ADR` itself).
743#[derive(Clone, Copy, Debug, PartialEq, Eq)]
744pub(super) enum BaseReference {
745    /// A load or store through the register whose address and width the
746    /// instruction gives: these bytes.
747    Access(u64, u64),
748    /// An address an `ADD` or `SUB` (immediate) forms from the register,
749    /// taken as `ADR`'s target is: one word, and nothing at an FDE start,
750    /// since forming an address is not a read, and one that names a
751    /// function's entry takes a pointer to the function. What the code then
752    /// does with the address is followed like the `ADR`'s or `ADRP`'s own.
753    Address(u64),
754    /// A use of the address whose extent is not known: a load or store
755    /// through it that the instruction does not bound (an index register, a
756    /// scalable vector access, a form not decoded), the address passed on to
757    /// a register or memory the scan does not follow, or the address still
758    /// held where following stops (a transfer of control, a conditional
759    /// branch, whose taken path is not followed, or the end of the window).
760    /// Any executable byte may be read through it, even where `at` is a
761    /// function's entry: a function pointer passed on can be read as well
762    /// as called. `at` is the address, or with `page` the `ADRP`'s page.
763    Unknown { at: u64, page: bool },
764}
765
766/// What one instruction does with the address a register holds.
767#[derive(Clone, Copy, Debug, PartialEq, Eq)]
768enum Effect {
769    /// It does not read the register.
770    Untouched,
771    /// It reads it without accessing memory or passing it on: a compare.
772    Tested,
773    /// It accesses `low..high` from the address, and then, with `writeback`,
774    /// leaves the address plus that in the register.
775    Accesses {
776        low: i64,
777        high: i64,
778        writeback: Option<i64>,
779    },
780    /// It forms the address plus `offset` in register `into` (`ADD` or
781    /// `SUB` immediate).
782    Forms { offset: i64, into: u32 },
783    /// It accesses memory through the address to an extent it does not
784    /// bound.
785    ReadsUnbounded,
786    /// It passes the address on: a copy, register arithmetic, a value stored,
787    /// an index, or a form not decoded.
788    PassesOn,
789}

791/// The bytes `words` (at `base`) may read or write through each `ADRP`'s
792/// page or `ADR`'s address. Each is followed through the instructions after
793/// it, up to [`BASE_WINDOW`], with what each does with every register
794/// holding an address from it ([`effect`]): an access the instruction bounds
795/// is a [`BaseReference::Access`], an `ADD` or `SUB` (immediate) a
796/// [`BaseReference::Address`] whose register is followed too, and a write
797/// ([`writes_register`]) ends the register's part. Anything else fails
798/// closed: an unbounded access, the address passed on, any use by an
799/// encoding [`decoded`] does not cover (wherever [`names`] says it may name
800/// the register), and every register still holding an address where
801/// following stops (a transfer of control, a conditional branch, the
802/// window's or the range's end) is a [`BaseReference::Unknown`].
803pub(super) fn base_references(words: &[u32], base: u64) -> Vec<BaseReference> {
804    let mut references = Vec::new();
805    for (index, word) in words.iter().enumerate() {
806        let pc = base.wrapping_add(4 * index as u64);
807        let register = word & 0x1F;
808        let imm = ((word >> 5) & 0x7_FFFF) << 2 | ((word >> 29) & 0b11);
809        let (origin, page) = if word & 0x9F00_0000 == 0x9000_0000 {
810            // ADRP: the 4 KiB page of `pc`, plus immhi:immlo pages.
811            let pages = ((imm << 11) as i32 >> 11) as i64;
812            ((pc & !0xFFF).wrapping_add((pages << 12) as u64), true)
813        } else if word & 0x9F00_0000 == 0x1000_0000 {
814            // ADR: `pc` plus immhi:immlo bytes.
815            let offset = ((imm << 11) as i32 >> 11) as i64;
816            (pc.wrapping_add(offset as u64), false)
817        } else {
818            continue;
819        };
820        if register == 31 {
821            continue; // into the zero register: nothing is addressed
822        }
823        let mut found = Vec::new();
824        let mut note = |reference: BaseReference| {
825            if !found.contains(&reference) {
826                found.push(reference);
827            }
828        };
829        let unknown = |at: u64, page: bool| BaseReference::Unknown { at, page };
830        // Each register holding an address from this one: the register, the
831        // address, and whether it is the bare page.
832        let mut bases = vec![(register, origin, page)];
833        for next in words[index + 1..].iter().take(BASE_WINDOW) {
834            if transfers_control(*next) || branches_conditionally(*next) {
835                break;
836            }
837            let mut after = Vec::new();
838            for &(held, at, bare) in &bases {
839                let mut kept = Some((held, at, bare));
840                match effect(*next, held) {
841                    Effect::Untouched | Effect::Tested => {}
842                    Effect::Accesses {
843                        low,
844                        high,
845                        writeback,
846                    } => {
847                        match (at.checked_add_signed(low), at.checked_add_signed(high)) {
848                            (Some(start), Some(end)) => note(BaseReference::Access(start, end)),
849                            _ => note(unknown(at, bare)),
850                        }
851                        if let Some(by) = writeback {
852                            kept = Some((held, at.wrapping_add_signed(by), false));
853                        }
854                    }
855                    Effect::Forms { offset, into } => {
856                        let target = at.wrapping_add_signed(offset);
857                        note(BaseReference::Address(target));
858                        after.push((into, target, false));
859                    }
860                    Effect::ReadsUnbounded => {
861                        note(unknown(at, bare));
862                        kept = None;
863                    }
864                    Effect::PassesOn => note(unknown(at, bare)),
865                }
866                if let Some((held, at, bare)) = kept
867                    && !writes_register(*next, held)
868                {
869                    after.push((held, at, bare));
870                }
871            }
872            // A register formed into replaces what it held.
873            bases.clear();
874            for (held, at, bare) in after.into_iter().rev() {
875                if !bases.iter().any(|(other, _, _)| *other == held) {
876                    bases.push((held, at, bare));
877                }
878            }
879            if bases.is_empty() {
880                break;
881            }
882        }
883        // Following stopped with these still holding an address.
884        for (_, at, bare) in bases {
885            note(unknown(at, bare));
886        }
887        references.extend(found);
888    }
889    references
890}

892/// How many instructions after an `ADRP` or `ADR` [`base_references`]
893/// follows its register. A compiler schedules a completion well after its
894/// `ADRP` (glibc's NSS tables complete one eleven instructions on), so this
895/// looks further than [`addressed_starts`]' eight.
896const BASE_WINDOW: usize = 64;
897
898/// Whether control may not continue to the next instruction: an
899/// unconditional branch (immediate or register, calls and returns among
900/// them) or an exception-generating instruction.
901fn transfers_control(word: u32) -> bool {
902    word & 0x7C00_0000 == 0x1400_0000 // B, BL
903        || word & 0xFE00_0000 == 0xD600_0000 // BR, BLR, RET, ERET, ...
904        || word & 0xFF00_0000 == 0xD400_0000 // SVC, HVC, SMC, BRK, HLT, ...
905}
906
907/// Whether control may go on elsewhere as well as to the next instruction:
908/// every word of the branch, exception and system class (op0 `101x`) that
909/// is neither [`transfers_control`] nor in [`system_space`], so `B.cond`,
910/// `BC.cond`, `CBZ`, `TBZ`, `CB<cc>` and any branch form not named here.
911/// The scan does not follow the taken path.
912fn branches_conditionally(word: u32) -> bool {
913    word & 0x1C00_0000 == 0x1400_0000 && !system_space(word) && !transfers_control(word)
914}
915
916/// Whether `word` is in the system instruction space (hints, barriers,
917/// `MSR`, `MRS`, `SYS` and their pair forms), whose instructions all
918/// continue to the next.
919fn system_space(word: u32) -> bool {
920    word & 0xFF80_0000 == 0xD500_0000
921}

923/// Every general register `word` may name, as a mask: each field a
924/// register can occupy in any encoding (bits 4:0, 9:5, 14:10 and 20:16)
925/// and the registers some families name implicitly ([`implicit`]). An
926/// encoding [`decoded`] does not cover touches a register only through
927/// this mask, and any it names counts as passed on.
928fn names(word: u32) -> u32 {
929    let field = |shift: u32| 1u32 << ((word >> shift) & 0x1F);
930    (field(0) | field(5) | field(10) | field(16) | implicit(word)) & !(1 << 31)
931}
932
933/// The registers some families name without a field of their own: a
934/// register list's successors (for any load or store, the register after
935/// each in bits 4:0 and 20:16, as `CASP`, `CASPT`, `RCWCASP` and the
936/// 128-bit atomics' pairs run; the eight of `LD64B` and `ST64B*`; and the
937/// system pair forms' Xt+1), the slice and predicate index registers of SME
938/// and SVE (w8 to w15), and the pointer authentication forms' fixed
939/// registers (x16 and x17 for `*1716`, x30 for `*Z`, `*SP` and `XPACLRI`,
940/// x15 to x17 for `*171615`, x16 for `CHKFEAT`).
941fn implicit(word: u32) -> u32 {
942    let (rt, rs) = (word & 0x1F, (word >> 16) & 0x1F);
943    let run =
944        |first: u32, count: u32| (0..count).fold(0, |mask, i| mask | 1 << ((first + i) & 0x1F));
945    let mut mask = 0;
946    if word & 0x0A00_0000 == 0x0800_0000 {
947        mask |= run(rt, 2) | run(rs, 2);
948    }
949    if ls64(word) {
950        mask |= run(rt, 8);
951    }
952    if word & 0xFFC0_0000 == 0xD540_0000 {
953        mask |= run(rt, 2); // SYSP, MSRR, MRRS
954    }
955    if word & 0x9E00_0000 == 0x8000_0000 || word & 0x1E00_0000 == 0x0400_0000 {
956        mask |= 0xFF00; // SME, SVE
957    }
958    if word & 0xFFFF_FF1F == 0xD503_211F {
959        mask |= 1 << 16 | 1 << 17;
960    }
961    if word & 0xFFFF_FF1F == 0xD503_231F || word == 0xD503_20FF {
962        mask |= 1 << 30;
963    }
964    if word == 0xD503_251F {
965        mask |= 1 << 16;
966    }
967    if word & 0xFFFF_83FF == 0xDAC1_83FE {
968        mask |= 0b111 << 15;
969    }
970    mask
971}

973/// Whether `word` is `LD64B`, `ST64B`, `ST64BV` or `ST64BV0`, which access
974/// 64 bytes at the base through the eight registers from Xt.
975fn ls64(word: u32) -> bool {
976    word & 0xFFE0_8C00 == 0xF820_8000 && matches!((word >> 12) & 0b111, 1 | 2 | 3 | 5)
977}

979/// Whether `word` is one of the forms [`effect`] decodes in full: the data
980/// processing (immediate) and (register) classes but pointer
981/// authentication, the hints, barriers, `MSR` (immediate and register),
982/// `MRS`, `DC CVAC` and `IC IVAU`, and the load and store families
983/// [`load_store_effect`] names. Every other encoding, SIMD and floating
984/// point, SVE and SME among them, uses a register only as [`names`] says,
985/// and any use passes the address on.
986fn decoded(word: u32) -> bool {
987    decoded_effect(word, 0).is_some()
988}

990/// What `word` does with the address `register` holds ([`Effect`]): as
991/// [`decoded_effect`] says for a form it decodes, and otherwise passed on
992/// where the word may name the register at all ([`names`]).
993fn effect(word: u32, register: u32) -> Effect {
994    match decoded_effect(word, register) {
995        Some(effect) => effect,
996        None if names(word) & (1 << register) != 0 => Effect::PassesOn,
997        None => Effect::Untouched,
998    }
999}

1001/// [`effect`] for the forms it decodes in full, or `None`, whatever the
1002/// register, for any other.
1003fn decoded_effect(word: u32, register: u32) -> Option<Effect> {
1004    let rn = (word >> 5) & 0x1F == register;
1005    let rm = (word >> 16) & 0x1F == register;
1006    let ra = (word >> 10) & 0x1F == register;
1007    let rt = word & 0x1F == register;
1008    if word & 0x0A00_0000 == 0x0800_0000 {
1009        return load_store_effect(word, register);
1010    }
1011    if word & 0x1C00_0000 == 0x1000_0000 {
1012        // Data processing (immediate).
1013        return Some(match word & 0x1F80_0000 {
1014            0x1000_0000 | 0x1080_0000 => Effect::Untouched, // ADR, ADRP
1015            // MOVK keeps the rest of its register; MOVZ and MOVN read none.
1016            0x1280_0000 if rt && (word >> 29) & 0b11 == 0b11 => Effect::PassesOn,
1017            0x1280_0000 => Effect::Untouched,
1018            0x1100_0000 | 0x1180_0000 if rn => {
1019                // ADD/SUB (immediate): a compare into the zero register, or
1020                // a 64-bit address formed in another register.
1021                let rd = word & 0x1F;
1022                let flags = (word >> 29) & 1 == 1;
1023                if flags && rd == 31 {
1024                    Effect::Tested
1025                } else if word >> 31 == 1 && !flags && rd != 31 && (word >> 23) & 1 == 0 {
1026                    let imm = i64::from((word >> 10) & 0xFFF) << (12 * ((word >> 22) & 1));
1027                    let offset = if (word >> 30) & 1 == 1 { -imm } else { imm };
1028                    Effect::Forms { offset, into: rd }
1029                } else {
1030                    Effect::PassesOn
1031                }
1032            }
1033            // BFM (`BFI`, `BFXIL`) keeps the bits of its destination it does
1034            // not insert, so it reads that register as well.
1035            0x1300_0000 if rt && (word >> 29) & 0b11 == 0b01 => Effect::PassesOn,
1036            // Logical, bitfield and extract (which reads two) pass it on.
1037            0x1380_0000 if rn || rm => Effect::PassesOn,
1038            _ if rn => Effect::PassesOn,
1039            _ => Effect::Untouched,
1040        });
1041    }
1042    if word & 0x0E00_0000 == 0x0A00_0000 {
1043        // Data processing (register), but PAC*, AUT* and XPAC* (one source,
1044        // opcode2 00001), which read their destination and can name
1045        // registers implicitly. Bits 14:10 name a register only in the
1046        // three-source class (`MADD`, `MSUB` and their widening forms);
1047        // elsewhere they are a shift, an extension, a condition or an
1048        // opcode. A compare into the zero register (ADDS, SUBS, ANDS) or a
1049        // conditional compare only tests it.
1050        if word & 0x5FFF_0000 == 0x5AC1_0000 {
1051            return None;
1052        }
1053        let three_source = word & 0x1F00_0000 == 0x1B00_0000;
1054        if !(rn || rm || (three_source && ra)) {
1055            return Some(Effect::Untouched);
1056        }
1057        let into_zero = word & 0x1F == 31;
1058        let adds = word & 0x1F00_0000 == 0x0B00_0000 && (word >> 29) & 1 == 1;
1059        let ands = word & 0x1F00_0000 == 0x0A00_0000 && (word >> 29) & 0b11 == 0b11;
1060        let ccmp = word & 0x1FE0_0000 == 0x1A40_0000;
1061        return Some(if (into_zero && (adds || ands)) || ccmp {
1062            Effect::Tested
1063        } else {
1064            Effect::PassesOn
1065        });
1066    }
1067    if word & 0xFFFF_F01F == 0xD503_201F {
1068        // A hint names no register but the implicit ones.
1069        return Some(if implicit(word) & (1 << register) != 0 {
1070            Effect::PassesOn
1071        } else {
1072            Effect::Untouched
1073        });
1074    }
1075    if word & 0xFFFF_F01F == 0xD503_301F || word & 0xFFF8_F01F == 0xD500_401F {
1076        return Some(Effect::Untouched); // barriers, MSR (immediate)
1077    }
1078    if word & 0xFFF0_0000 == 0xD530_0000 {
1079        return Some(Effect::Untouched); // MRS: writes Xt
1080    }
1081    if word & 0xFFF0_0000 == 0xD510_0000 {
1082        return Some(if rt {
1083            Effect::PassesOn
1084        } else {
1085            Effect::Untouched
1086        }); // MSR
1087    }
1088    if matches!(word & 0xFFFF_FFE0, 0xD50B_7A20 | 0xD50B_7520) {
1089        // Cleaning a data cache line or invalidating an instruction cache
1090        // line by its address (`DC CVAC`, `IC IVAU`) changes no data.
1091        return Some(if rt {
1092            Effect::Tested
1093        } else {
1094            Effect::Untouched
1095        });
1096    }
1097    None
1098}

1100/// [`decoded_effect`] for the loads and stores (op0 `x1x0`) it decodes:
1101/// literal loads; single registers with an unsigned, a 9-bit signed or a
1102/// register offset; pairs (`STGP` scaled by 16); exclusive, ordered and
1103/// compare-and-swap (`CASP` as its own family); the LSE atomics but the
1104/// RCW forms, whose size field is not their width; and `LD64B`/`ST64B*`.
1105fn load_store_effect(word: u32, register: u32) -> Option<Effect> {
1106    let rn = (word >> 5) & 0x1F == register;
1107    let rm = (word >> 16) & 0x1F == register;
1108    let ra = (word >> 10) & 0x1F == register;
1109    let rt = word & 0x1F == register;
1110    let listed = |first: u32, count: u32| (0..count).any(|i| (first + i) & 0x1F == register);
1111    let size = word >> 30;
1112    let vector = (word >> 26) & 1 == 1;
1113    let opc = (word >> 22) & 0b11;
1114    // A single-register general-purpose store (opc 00) reads its transfer
1115    // register; a vector one names a SIMD register there.
1116    let stores_it = rt && !vector && opc == 0;
1117    let width = if vector && size == 0 && opc & 0b10 != 0 {
1118        16
1119    } else {
1120        1i64 << size
1121    };
1122    // The single-register forms allocated: a 128-bit vector access only at
1123    // size 00, and no general-purpose opc 11 at size 10 or 11.
1124    let single = if vector {
1125        size == 0 || opc & 0b10 == 0
1126    } else {
1127        !(opc == 0b11 && size >= 0b10)
1128    };
1129    if word & 0x3B00_0000 == 0x1800_0000 {
1130        // A literal load has no base; vector opc 11 is unallocated.
1131        return (!vector || size != 0b11).then_some(Effect::Untouched);
1132    }
1133    if word & 0x3B00_0000 == 0x3900_0000 && single {
1134        // Unsigned offset, scaled by the access size.
1135        if stores_it {
1136            return Some(Effect::PassesOn);
1137        }
1138        if !rn {
1139            return Some(Effect::Untouched);
1140        }
1141        let offset = i64::from((word >> 10) & 0xFFF) * width;
1142        return Some(Effect::Accesses {
1143            low: offset,
1144            high: offset + width,
1145            writeback: None,
1146        });
1147    }
1148    if word & 0x3B20_0000 == 0x3800_0000 && single {
1149        // A 9-bit signed offset: unscaled, unprivileged, pre- or post-index.
1150        if stores_it {
1151            return Some(Effect::PassesOn);
1152        }
1153        if !rn {
1154            return Some(Effect::Untouched);
1155        }
1156        let offset = i64::from(((((word >> 12) & 0x1FF) << 23) as i32) >> 23);
1157        return Some(match (word >> 10) & 0b11 {
1158            0b01 => Effect::Accesses {
1159                low: 0,
1160                high: width,
1161                writeback: Some(offset),
1162            },
1163            0b11 => Effect::Accesses {
1164                low: offset,
1165                high: offset + width,
1166                writeback: Some(offset),
1167            },
1168            _ => Effect::Accesses {
1169                low: offset,
1170                high: offset + width,
1171                writeback: None,
1172            },
1173        });
1174    }
1175    if word & 0x3B20_0C00 == 0x3820_0800 && single {
1176        // A register offset: not bounded, and its index is passed on.
1177        return Some(if stores_it || rm {
1178            Effect::PassesOn
1179        } else if rn {
1180            Effect::ReadsUnbounded
1181        } else {
1182            Effect::Untouched
1183        });
1184    }
1185    if word & 0x3B20_0C00 == 0x3820_0000 && !vector {
1186        let o3 = (word >> 15) & 1 == 1;
1187        let operation = (word >> 12) & 0b111;
1188        if ls64(word) {
1189            // ST64B, ST64BV and ST64BV0 store Xt to Xt+7; LD64B loads them.
1190            if operation != 0b101 && listed(word & 0x1F, 8) {
1191                return Some(Effect::PassesOn);
1192            }
1193            return Some(if rn {
1194                Effect::Accesses {
1195                    low: 0,
1196                    high: 64,
1197                    writeback: None,
1198                }
1199            } else {
1200                Effect::Untouched
1201            });
1202        }
1203        // LDADD and the other operations, SWP, and LDAPR (A set, R clear,
1204        // Rs 11111): Xs is stored, Xt receives the old value.
1205        let ldapr = operation == 0b100 && opc == 0b10 && (word >> 16) & 0x1F == 31;
1206        if o3 && operation != 0 && !ldapr {
1207            return None;
1208        }
1209        return Some(if rm {
1210            Effect::PassesOn
1211        } else if rn {
1212            Effect::Accesses {
1213                low: 0,
1214                high: width,
1215                writeback: None,
1216            }
1217        } else {
1218            Effect::Untouched
1219        });
1220    }
1221    if word & 0x3A00_0000 == 0x2800_0000 {
1222        // A pair: a 7-bit signed offset scaled by the element size, or by
1223        // 16 for `STGP`, which stores two 64-bit registers.
1224        let load = (word >> 22) & 1 == 1;
1225        let stgp = !vector && size == 0b01 && !load;
1226        let element: i64 = match (vector, size) {
1227            (true, 0b11) | (false, 0b11) => return None,
1228            (true, size) => 4 << size,
1229            (false, 0b00) => 4,
1230            (false, 0b10) => 8,
1231            (false, _) if stgp => 8,
1232            (false, _) => 4, // LDPSW
1233        };
1234        if !vector && size == 0b01 && (word >> 23) & 0b11 == 0 {
1235            return None; // no non-temporal STGP or LDPSW
1236        }
1237        if !load && !vector && (rt || ra) {
1238            return Some(Effect::PassesOn); // a value stored
1239        }
1240        if !rn {
1241            return Some(Effect::Untouched);
1242        }
1243        let scale = if stgp { 16 } else { element };
1244        let offset = i64::from(((((word >> 15) & 0x7F) << 25) as i32) >> 25) * scale;
1245        let (low, high) = (offset, offset + 2 * element);
1246        return Some(match (word >> 23) & 0b11 {
1247            0b01 => Effect::Accesses {
1248                low: 0,
1249                high: 2 * element,
1250                writeback: Some(offset),
1251            },
1252            0b11 => Effect::Accesses {
1253                low,
1254                high,
1255                writeback: Some(offset),
1256            },
1257            _ => Effect::Accesses {
1258                low,
1259                high,
1260                writeback: None,
1261            },
1262        });
1263    }
1264    if word & 0x3F00_0000 == 0x0800_0000 {
1265        let load = (word >> 22) & 1 == 1;
1266        let pair = (word >> 21) & 1 == 1;
1267        let compare = (word >> 23) & 1 == 1 && pair;
1268        if size < 0b10 && pair && !compare {
1269            // CASP: compares Xs:Xs+1 and stores Xt:Xt+1 at the base, then
1270            // writes the old value to Xs:Xs+1.
1271            if word & 0x7C00 != 0x7C00 {
1272                return None;
1273            }
1274            if listed(word & 0x1F, 2) {
1275                return Some(Effect::PassesOn);
1276            }
1277            return Some(if rn {
1278                Effect::Accesses {
1279                    low: 0,
1280                    high: 2 * (4 << size),
1281                    writeback: None,
1282                }
1283            } else if listed((word >> 16) & 0x1F, 2) {
1284                Effect::Tested
1285            } else {
1286                Effect::Untouched
1287            });
1288        }
1289        // Exclusive, ordered and compare-and-swap: at the base itself. A
1290        // compare-and-swap reads Rs and Rt, a store-exclusive Rt (and Rt2).
1291        if (compare && (rt || rm)) || (!compare && !load && (rt || (pair && ra))) {
1292            return Some(Effect::PassesOn);
1293        }
1294        if !rn {
1295            return Some(Effect::Untouched);
1296        }
1297        return Some(Effect::Accesses {
1298            low: 0,
1299            high: if pair || compare { 16 } else { width },
1300            writeback: None,
1301        });
1302    }
1303    None
1304}

1306/// Whether `word` writes `register`, ending the address it held: a form
1307/// [`decoded`] covers whose destination it is (a data-processing result
1308/// other than a flags-only form's, a load's transfer registers, a
1309/// store-exclusive's or compare-and-swap's status or old value, `MRS`'s
1310/// Xt). Any other encoding counts as not writing it, so the address stays
1311/// followed.
1312fn writes_register(word: u32, register: u32) -> bool {
1313    if !decoded(word) {
1314        return false;
1315    }
1316    let rd = word & 0x1F == register;
1317    let rm = (word >> 16) & 0x1F == register;
1318    let ra = (word >> 10) & 0x1F == register;
1319    let listed = |first: u32, count: u32| (0..count).any(|i| (first + i) & 0x1F == register);
1320    // Data processing forms that write only the flags, whose bits 4:0 are
1321    // not a destination: a conditional compare's are its NZCV immediate,
1322    // `RMIF`'s its mask, `SETF8` and `SETF16`'s fixed.
1323    let flags_only = word & 0x1FE0_0000 == 0x1A40_0000
1324        || word & 0xFFE0_7C10 == 0xBA00_0400
1325        || word & 0xFFFF_BC1F == 0x3A00_080D;
1326    if flags_only {
1327        return false;
1328    }
1329    if word & 0x1C00_0000 == 0x1000_0000 || word & 0x0E00_0000 == 0x0A00_0000 {
1330        return rd;
1331    }
1332    if system_space(word) {
1333        return word & 0xFFF0_0000 == 0xD530_0000 && rd; // MRS
1334    }
1335    let vector = (word >> 26) & 1 == 1;
1336    let opc = (word >> 22) & 0b11;
1337    if word & 0x3B20_0C00 == 0x3820_0000 && !vector {
1338        if ls64(word) {
1339            return match (word >> 12) & 0b111 {
1340                0b101 => listed(word & 0x1F, 8),
1341                0b010 | 0b011 => rm, // the status
1342                _ => false,
1343            };
1344        }
1345        return rd; // the old value
1346    }
1347    let load = if word & 0x3B00_0000 == 0x1800_0000 {
1348        word >> 30 != 0b11 // a literal load, not PRFM
1349    } else if word & 0x3A00_0000 == 0x2800_0000 {
1350        (word >> 22) & 1 == 1 // a pair
1351    } else if word & 0x3F00_0000 == 0x0800_0000 {
1352        let pair = (word >> 21) & 1 == 1;
1353        if (word >> 23) & 1 == 1 && pair {
1354            return rm; // compare-and-swap: the old value
1355        }
1356        if word >> 31 == 0 && pair {
1357            return listed((word >> 16) & 0x1F, 2); // CASP: the old value
1358        }
1359        if (word >> 22) & 1 == 0 {
1360            return rm; // the status
1361        }
1362        true
1363    } else {
1364        // A single register: opc 00 stores; PRFM (size 11, opc 10) loads
1365        // nothing.
1366        opc != 0 && !(word >> 30 == 0b11 && opc == 0b10)
1367    };
1368    // LDP and LDXP's second register.
1369    let pair = word & 0x3A00_0000 == 0x2800_0000
1370        || (word & 0x3F00_0000 == 0x0800_0000 && (word >> 21) & 1 == 1);
1371    !vector && load && (rd || (pair && ra))
1372}

1374/// Where `words` (at `base`) address bytes as data, or may: the target of
1375/// each `LDR` (literal) and `ADR`, and of each `ADRP` with every `ADD`
1376/// (immediate) or unsigned-offset load or store that completes it among the
1377/// eight instructions after it, taking the `ADRP`'s register as its base,
1378/// up to the instruction that writes that register without completing it,
1379/// since the page is gone from there. One `ADRP` can address several tables.
1380/// An `ADRP` nothing completes gives its page, the lowest address it can be
1381/// completing. Each is where a table may start: how far it runs is not in
1382/// the instruction ([`data_extents`]).
1383pub fn addressed_starts(words: &[u32], base: u64) -> Addressed {
1384    // How far after an `ADRP` its completing instruction is looked for.
1385    const PAIRING_WINDOW: usize = 8;
1386    let mut starts = Vec::new();
1387    let mut unresolved = Vec::new();
1388    for (index, word) in words.iter().enumerate() {
1389        let pc = base.wrapping_add(4 * index as u64);
1390        if word & 0x3B00_0000 == 0x1800_0000 && word >> 30 != 0b11 {
1391            // LDR (literal), not PRFM: its target is its literal's start.
1392            let offset = ((((word >> 5) & 0x7_FFFF) << 13) as i32 >> 11) as i64;
1393            starts.push(pc.wrapping_add(offset as u64));
1394        } else if word & 0x9F00_0000 == 0x1000_0000 {
1395            // ADR: immlo (30:29), immhi (23:5).
1396            let imm = ((word >> 5) & 0x7_FFFF) << 2 | ((word >> 29) & 0b11);
1397            let offset = ((imm << 11) as i32 >> 11) as i64;
1398            starts.push(pc.wrapping_add(offset as u64));
1399        } else if word & 0x9F00_0000 == 0x9000_0000 {
1400            // ADRP: the 4 KiB page of `pc`, plus immhi:immlo pages.
1401            let imm = ((word >> 5) & 0x7_FFFF) << 2 | ((word >> 29) & 0b11);
1402            let pages = ((imm << 11) as i32 >> 11) as i64;
1403            let page = (pc & !0xFFF).wrapping_add((pages << 12) as u64);
1404            let register = word & 0x1F;
1405            // Every completion, following the register: an `ADD` into the
1406            // register itself moves the base the later ones are relative
1407            // to, and any other write loses it, so the pairing ends there.
1408            let mut base = page;
1409            let mut completions = 0usize;
1410            for next in words[index + 1..].iter().take(PAIRING_WINDOW) {
1411                let offset = page_offset(*next, register);
1412                if let Some(offset) = offset {
1413                    starts.push(base.wrapping_add(offset));
1414                    completions += 1;
1415                    // `ADD Xn, Xn, #imm`: from here the register holds
1416                    // that address, so a later access is relative to it.
1417                    if *next & 0x1F == register && *next & 0xFF80_0000 == 0x9100_0000 {
1418                        base = base.wrapping_add(offset);
1419                        continue;
1420                    }
1421                }
1422                // A write that does not carry the base ends the
1423                // pairing. What the code addresses through the register
1424                // afterwards is then unknown; taking the whole page for
1425                // data there costs about half of a stripped library's
1426                // rewrites (measured), so it is left as a limit this
1427                // module's documentation states.
1428                if writes(*next, register) {
1429                    break;
1430                }
1431            }
1432            if completions == 0 {
1433                // Nothing in the window completes it, so all that is known
1434                // is the page: the caller takes the functions it overlaps
1435                // for data.
1436                unresolved.push(page);
1437            }
1438        }
1439    }
1440    Addressed { starts, unresolved }
1441}

1455/// The offset into an `ADRP`'s page that `word` adds, if it is an `ADD`
1456/// (immediate, 64-bit) or an unsigned-offset load or store whose base is
1457/// `register`.
1458fn page_offset(word: u32, register: u32) -> Option<u64> {
1459    if (word >> 5) & 0x1F != register {
1460        return None;
1461    }
1462    if word & 0xFF80_0000 == 0x9100_0000 {
1463        // ADD Xd, Xn, #imm12 {, LSL #12}: sh (22), imm12 (21:10).
1464        let imm = u64::from((word >> 10) & 0xFFF);
1465        return Some(if (word >> 22) & 1 == 1 {
1466            imm << 12
1467        } else {
1468            imm
1469        });
1470    }
1471    if word & 0x3B00_0000 == 0x3900_0000 {
1472        // LDR/STR (unsigned immediate): size (31:30), V (26), opc (23:22),
1473        // imm12 (21:10), scaled by the access size; a 128-bit SIMD access
1474        // is size 00 with opc 1x.
1475        let size = word >> 30;
1476        let scale = if (word >> 26) & 1 == 1 && size == 0 && (word >> 23) & 1 == 1 {
1477            4
1478        } else {
1479            size
1480        };
1481        return Some(u64::from((word >> 10) & 0xFFF) << scale);
1482    }
1483    None
1484}

1486/// Whether `word` writes `register`, as far as the shapes an `ADRP` pairing
1487/// must not look past: a `MOV`, an `ADD`/`SUB` or another `ADRP` or `ADR`
1488/// into it, or a load into it. An instruction this does not recognise is
1489/// taken not to write it, which only widens the pairing, and the pairing's
1490/// result is a data address, so a mistaken one costs coverage, never
1491/// safety.
1492fn writes(word: u32, register: u32) -> bool {
1493    if word & 0x1F != register {
1494        return false;
1495    }
1496    // ADR and ADRP; ADD/SUB (immediate and shifted register); MOV (wide
1497    // immediate) and ORR (shifted register), which `MOV` register is; and
1498    // the loads, whose destination is also bits 4:0. A store in those two
1499    // classes reads that register instead (`STR Xt, [SP]` spilling the
1500    // page), and `PRFM` loads nothing, so neither ends a pairing.
1501    let loads = |opc: u32| opc != 0b00 && opc != 0b11;
1502    word & 0x1F00_0000 == 0x1000_0000
1503        || word & 0x1F00_0000 == 0x1100_0000
1504        || word & 0x1F00_0000 == 0x0B00_0000
1505        || word & 0x1F80_0000 == 0x1280_0000
1506        || word & 0x1F00_0000 == 0x0A00_0000
1507        || (word & 0x3B00_0000 == 0x3900_0000 && (word >> 22) & 1 == 1)
1508        || (word & 0x3B00_0000 == 0x1800_0000 && loads(word >> 30))
1509}

1523/// The bytes a function may hold as data, where only the unwind table tells
1524/// code from data: from each address the code addresses as data
1525/// (`starts`, [`addressed_starts`]) inside a function (`functions`, the
1526/// FDEs' ranges, sorted) to the next instruction a branch reaches
1527/// (`branches`, as [`direct_branches`] gives them), or the function's end.
1528/// Control does not run into a table, so the code after one is reached by a
1529/// branch, and only a branch that is not itself inside a table counts: a
1530/// table word can decode as one. An address at a function's entry is a
1531/// pointer to the function, not data. The ranges come back sorted and merged.
1532pub fn data_extents(
1533    starts: &[u64],
1534    functions: &[(u64, u64)],
1535    branches: &[(u64, u64)],
1536) -> Vec<(u64, u64)> {
1537    // FDE ranges may nest: a short range inside a longer one does not end
1538    // the data at its own end, so every containing range contributes an
1539    // extent and the merge below keeps the longest.
1540    let mut extents: Vec<(u64, u64)> = starts
1541        .iter()
1542        .flat_map(|start| {
1543            let start = *start;
1544            let after = functions.partition_point(|(entry, _)| *entry <= start);
1545            functions[..after]
1546                .iter()
1547                .filter(move |(entry, end)| *entry < start && start < *end)
1548                .map(move |(_, end)| (start, *end))
1549        })
1550        .collect();
1551    extents.sort_unstable();
1552    extents.dedup();
1553    // Only a branch whose target lies in an extent can end one, and an
1554    // extent only shrinks, so the rest are dropped once: on a large
1555    // stripped library this is thousands of branches instead of millions,
1556    // round after round, for the same result.
1557    let initial = merge(extents.clone());
1558    let reaches = |addr: u64| {
1559        let after = initial.partition_point(|(start, _)| *start <= addr);
1560        after.checked_sub(1).is_some_and(|i| addr < initial[i].1)
1561    };
1562    let branches: Vec<(u64, u64)> = branches
1563        .iter()
1564        .filter(|(_, target)| reaches(*target))
1565        .copied()
1566        .collect();
1567    // Each extent ends at the first target past its start of a branch
1568    // outside every extent. Extents only shrink, and a branch they no
1569    // longer hold can only shrink them further, so this ends.
1570    loop {
1571        let merged = merge(extents.clone());
1572        let inside = |addr: u64| {
1573            let after = merged.partition_point(|(start, _)| *start <= addr);
1574            after.checked_sub(1).is_some_and(|i| addr < merged[i].1)
1575        };
1576        let mut targets: Vec<u64> = branches
1577            .iter()
1578            .filter(|(at, _)| !inside(*at))
1579            .map(|(_, target)| *target)
1580            .collect();
1581        targets.sort_unstable();
1582        let mut changed = false;
1583        for (start, end) in &mut extents {
1584            let next = targets.partition_point(|target| *target <= *start);
1585            if let Some(target) = targets.get(next).filter(|target| **target < *end) {
1586                *end = *target;
1587                changed = true;
1588            }
1589        }
1590        if !changed {
1591            return merged;
1592        }
1593    }
1594}

1596/// `spans` sorted and with overlapping or touching ones joined.
1597fn merge(mut spans: Vec<(u64, u64)>) -> Vec<(u64, u64)> {
1598    spans.sort_unstable();
1599    let mut out: Vec<(u64, u64)> = Vec::new();
1600    for (start, end) in spans {
1601        match out.last_mut() {
1602            Some(last) if start <= last.1 => last.1 = last.1.max(end),
1603            _ => out.push((start, end)),
1604        }
1605    }
1606    out
1607}

1632/// Where an ELF64 little-endian file's symbol table and the string table it
1633/// names are, as file ranges (offset, size), from its section header table.
1634/// `None` when it has no symbol table (a stripped file) or the table's
1635/// string table is not one.
1636pub fn symbol_table(table: &[u8]) -> Option<((u64, u64), (u64, u64))> {
1637    let entries = table.as_chunks::<SECTION_HEADER_SIZE>().0;
1638    let kind =
1639        |entry: &[u8; SECTION_HEADER_SIZE]| u32::from_le_bytes(entry[4..8].try_into().unwrap());
1640    let range = |entry: &[u8; SECTION_HEADER_SIZE]| {
1641        (
1642            u64::from_le_bytes(entry[24..32].try_into().unwrap()),
1643            u64::from_le_bytes(entry[32..40].try_into().unwrap()),
1644        )
1645    };
1646    let symtab = entries.iter().find(|entry| kind(entry) == SHT_SYMTAB)?;
1647    let link = u32::from_le_bytes(symtab[40..44].try_into().unwrap());
1648    let strtab = entries

1654/// The local, untyped symbols of a symbol table that mark a section, given
1655/// the string table it names, as (section index, address, name). A mapping
1656/// symbol is one of them ("ELF for the Arm 64-bit Architecture", 5.5.4: they
1657/// are `STT_NOTYPE` and `STB_LOCAL`, and the `$` names are reserved for local
1658/// symbols). A global symbol whose name begins `$x` is an ordinary label:
1659/// taking it for a mapping symbol would let one inside a `$d` table start a
1660/// code range there.
1661fn local_untyped_symbols<'a>(
1662    symtab: &'a [u8],
1663    strtab: &'a [u8],
1664) -> impl Iterator<Item = (usize, u64, &'a [u8])> + 'a {
1665    symtab
1666        .as_chunks::<SYMBOL_SIZE>()
1667        .0
1668        .iter()
1669        .filter_map(move |symbol| {
1670            let name =
1671                usize::try_from(u32::from_le_bytes(symbol[0..4].try_into().unwrap())).ok()?;
1672            let section = u16::from_le_bytes(symbol[6..8].try_into().unwrap());
1673            if symbol[4] & 0xf != STT_NOTYPE
1674                || symbol[4] >> 4 != STB_LOCAL
1675                || section == 0
1676                || section >= SHN_LORESERVE
1677            {
1678                return None;
1679            }
1680            let name = strtab.get(name..)?;
1681            let name = &name[..name.iter().position(|byte| *byte == 0)?];
1682            Some((
1683                usize::from(section),
1684                u64::from_le_bytes(symbol[8..16].try_into().unwrap()),
1685                name,
1686            ))
1687        })
1688}

1690/// The mapping symbols of a symbol table, given the string table it names:
1691/// the local, untyped symbols whose names are `$x` or `$d`, sorted by
1692/// section and address. Where a `$x` and a `$d` mark the same
1693/// address, the `$d` sorts last and so wins: a word that may be data is not
1694/// taken for code.
1695pub fn mapping_symbols(symtab: &[u8], strtab: &[u8]) -> Vec<MappingSymbol> {
1696    let mut symbols: Vec<MappingSymbol> = local_untyped_symbols(symtab, strtab)
1697        .filter_map(|(section, address, name)| {
1698            Some(MappingSymbol {
1699                section,
1700                address,
1701                code: mapping_kind(name)?,
1702            })
1703        })
1704        .collect();
1705    symbols.sort_by_key(|symbol| (symbol.section, symbol.address, !symbol.code));
1706    symbols
1707}

1709/// The local, untyped symbols named like mapping symbols (a `$` name) that
1710/// are neither `$x` nor `$d`, with or without a `.suffix`, as (section index,
1711/// name), in table order: AArch32's `$a` and `$t`, and any other `$` name,
1712/// such as `$xyz` or `$dx`. They are reserved for mapping symbols, so this
1713/// reader cannot say where they put code or data, and a section that has one
1714/// is not proved by its mapping symbols: a `$a` after a `$x` would otherwise
1715/// leave the code range running through it.
1716pub fn unsupported_mapping_symbols(symtab: &[u8], strtab: &[u8]) -> Vec<(usize, String)> {
1717    local_untyped_symbols(symtab, strtab)
1718        .filter(|(_, _, name)| name.first() == Some(&b'$') && mapping_kind(name).is_none())
1719        .map(|(section, _, name)| (section, String::from_utf8_lossy(name).into_owned()))
1720        .collect()
1721}

1723/// The code of section `section`, which starts at `address` and is `size`
1724/// bytes long, as address ranges, given `symbols` sorted as
1725/// [`mapping_symbols`] returns them: each `$x` to the next mapping symbol of
1726/// the section, or to its end. Bytes before the section's first mapping
1727/// symbol are not code, and a mark outside the section is ignored. `None`
1728/// when the section has no mapping symbol, so its code cannot be told from
1729/// its data.
1730pub fn code_spans(
1731    symbols: &[MappingSymbol],
1732    section: usize,
1733    address: u64,
1734    size: u64,
1735) -> Option<Vec<(u64, u64)>> {
1736    let end = address.saturating_add(size);
1737    // A mark outside its own section's bytes says nothing of them.
1738    let marks: Vec<&MappingSymbol> = symbols
1739        .iter()
1740        .filter(|symbol| symbol.section == section && (address..end).contains(&symbol.address))
1741        .collect();
1742    if marks.is_empty() {
1743        return None;
1744    }
1745    let mut spans: Vec<(u64, u64)> = Vec::new();
1746    for (i, mark) in marks.iter().enumerate() {
1747        if !mark.code {
1748            continue;
1749        }
1750        let start = mark.address;
1751        let stop = marks.get(i + 1).map_or(end, |next| next.address);
1752        if start == stop {
1753            continue;
1754        }
1755        // Adjacent `$x` marks continue one span.
1756        match spans.last_mut() {
1757            Some(last) if last.1 == start => last.1 = stop,
1758            _ => spans.push((start, stop)),
1759        }
1760    }
1761    Some(spans)
1762}

1782        let mut bytes = Vec::new();
