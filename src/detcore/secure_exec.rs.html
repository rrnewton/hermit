9//! Whether Linux runs an exec in secure-execution mode for the guest, decided
10//! the same way on every backend from the new image's set-id bits and
11//! Detcore's fixed virtual-root identity, and what Detcore does with that
12//! decision once the new image reports its own `AT_SECURE`.
13//!
14//! Linux decides it in `bprm_fill_uid` and `cap_bprm_creds_from_file`
15//! (fs/exec.c and security/commoncap.c, v6.17): a set-user-ID bit makes the
16//! file's owner the effective user id, a set-group-ID bit with group execute
17//! makes the file's group the effective group id, unless the mount is
18//! `nosuid`, the task has `no_new_privs`, or the user namespace maps neither
19//! the owner nor the group; and the exec is secure when that moves the
20//! effective user or group id away from the caller's real one, which for
21//! Detcore's virtual root is 0. A secure exec clears the parent-death signal
22//! (`begin_new_exec`) and reports `AT_SECURE` 1 (`create_elf_tables`).
23
24use std::path::Path;
25
26/// The facts about an image file that decide its set-id transition.
27#[derive(Clone, Copy, Debug, PartialEq, Eq)]
28pub(crate) struct SetIdImage {
29    /// Its `st_mode`.
30    pub mode: u32,
31    /// Its owner and group, as the guest's user namespace shows them.
32    pub uid: u32,
33    pub gid: u32,
34    /// Whether that namespace maps the owner and the group.
35    pub uid_mapped: bool,
36    pub gid_mapped: bool,
37    /// Whether its mount is `nosuid`.
38    pub nosuid: bool,
39}
40
41impl SetIdImage {
42    /// Whether the file has a set-user-ID or set-group-ID bit at all, the
43    /// first test of `bprm_fill_uid`.
44    pub fn has_set_id_bits(&self) -> bool {
45        self.mode & (libc::S_ISUID | libc::S_ISGID) != 0
46    }
47}
48
49/// The effective user and group ids Linux gives a caller whose real and
50/// effective user and group ids are all 0 (Detcore's fixed virtual root) at
51/// an exec of `image`, with `no_new_privs` as the task has it
52/// (`bprm_fill_uid`, fs/exec.c:1534-1583, v6.17): a set-user-ID bit makes the
53/// file's owner the effective user id, and a set-group-ID bit with group
54/// execute makes the file's group the effective group id, unless the task
55/// has `no_new_privs`, the mount is `nosuid`, or the namespace maps the
56/// owner or the group not at all. The filesystem ids follow
57/// (`cap_bprm_creds_from_file`). Anything but (0, 0) moves an effective id
58/// away from the real one, 0, which is what makes the exec run in
59/// secure-execution mode (security/commoncap.c:989-996). File capabilities
60/// never make such an exec secure: root is already fully capable in its
61/// namespace (`handle_privileged_root`).
62pub(crate) fn effective_ids_for_virtual_root(image: &SetIdImage, no_new_privs: bool) -> (u32, u32) {
63    if no_new_privs || image.nosuid || !image.has_set_id_bits() {
64        return (0, 0);
65    }
66    // An owner or group the namespace does not map voids both bits.
67    if !image.uid_mapped || !image.gid_mapped {
68        return (0, 0);
69    }
70    let setuid = image.mode & libc::S_ISUID != 0;
71    let setgid = image.mode & (libc::S_ISGID | libc::S_IXGRP) == (libc::S_ISGID | libc::S_IXGRP);
72    (
73        if setuid { image.uid } else { 0 },
74        if setgid { image.gid } else { 0 },
75    )
76}

78/// The set-id facts of the image file at `path`, as the tool sees it: the
79/// tool shares the guest's user and mount namespaces wherever Detcore
80/// resolves the image itself (`Detcore::exec_record_root`). `None` where the
81/// file or its mount cannot be read. Raw syscalls: in-guest Detcore must not
82/// call libc's wrappers (`crate::util::raw_syscall`).
83pub(crate) fn set_id_image(path: &Path) -> Option<SetIdImage> {
84    let stat = crate::util::raw_stat(path).ok()?;
85    let nosuid = mount_flags(path)? & libc::ST_NOSUID != 0;
86    let uid_map = crate::util::raw_read_file(Path::new("/proc/self/uid_map")).ok()?;
87    let gid_map = crate::util::raw_read_file(Path::new("/proc/self/gid_map")).ok()?;
88    Some(SetIdImage {
89        mode: stat.st_mode,
90        uid: stat.st_uid,
91        gid: stat.st_gid,
92        uid_mapped: maps(&uid_map, stat.st_uid),
93        gid_mapped: maps(&gid_map, stat.st_gid),
94        nosuid,
95    })
96}

98/// The mount flags (`statfs::f_flags`) of the file system holding `path`, by
99/// a raw `statfs`. libc's `statfs` keeps `f_flags` private, so the kernel's
100/// structure is read as the fifteen words it is on x86_64 and aarch64, where
101/// `f_flags` is the eleventh (include/uapi/asm-generic/statfs.h).
102fn mount_flags(path: &Path) -> Option<u64> {
103    use std::os::unix::ffi::OsStrExt;
104    const WORDS: usize = 15;
105    const FLAGS: usize = 10;
106    let path = std::ffi::CString::new(path.as_os_str().as_bytes()).ok()?;
107    let mut words = [0_u64; WORDS];
108    // SAFETY: the path is NUL-terminated and `words` is writable and as
109    // large as the kernel's structure.
110    crate::util::raw_zero_result(unsafe {
111        crate::util::raw_syscall(
112            libc::SYS_statfs,
113            [path.as_ptr() as u64, words.as_mut_ptr() as u64, 0, 0, 0, 0],
114        )
115    })
116    .ok()?;
117    Some(words[FLAGS])
118}

120/// Whether the id map `map` (the contents of a `uid_map` or `gid_map`) maps
121/// `id` inside its namespace: some `inside outside count` line covers it.
122pub(crate) fn maps(map: &[u8], id: u32) -> bool {
123    String::from_utf8_lossy(map).lines().any(|line| {
124        let fields: Vec<u64> = line
125            .split_whitespace()
126            .filter_map(|field| field.parse().ok())
127            .collect();
128        matches!(fields.as_slice(), [inside, _, count]
129            if (*inside..inside.saturating_add(*count)).contains(&u64::from(id)))
130    })
131}

133/// The `NoNewPrivs:` value of a `/proc/<pid>/status` text, `None` where the
134/// line is missing or malformed.
135pub(crate) fn no_new_privs(status: &[u8]) -> Option<bool> {
136    String::from_utf8_lossy(status)
137        .lines()
138        .find_map(|line| line.strip_prefix("NoNewPrivs:"))
139        .and_then(|value| match value.trim() {
140            "0" => Some(false),
141            "1" => Some(true),
142            _ => None,
143        })
144}

146/// What Detcore does at a new image's first callback, from whether Linux
147/// runs the exec securely (`expected`, decided before the exec; `None` where
148/// it was not decided) and the image's own `AT_SECURE` (`reported`).
149#[derive(Clone, Copy, Debug, PartialEq, Eq)]
150pub(crate) enum SecureExecOutcome {
151    /// An ordinary exec: nothing to do.
152    Ordinary,
153    /// The exec ran in secure-execution mode, as the image reports: clear
154    /// the parent-death signal Detcore holds, as Linux's `begin_new_exec`
155    /// clears the task's.
156    Secure,
157    /// Linux runs the exec securely, but the backend ran it without
158    /// secure-execution mode (`AT_SECURE` 0): the new image's credentials,
159    /// auxiliary vector and dumpability differ from Linux's, so the run is
160    /// refused by name. The parent-death signal is cleared as for `Secure`.
161    Unsupported,
162}
163
164/// [`SecureExecOutcome`] for `expected` and `reported`. A backend that ran
165/// the exec securely is believed even where Detcore did not expect it: the
166/// kernel that performed it decided with the task's real credentials.
167pub(crate) fn outcome(expected: Option<bool>, reported: Option<bool>) -> SecureExecOutcome {
168    match (expected, reported) {
169        (_, Some(true)) => SecureExecOutcome::Secure,
170        (Some(true), _) => SecureExecOutcome::Unsupported,
171        _ => SecureExecOutcome::Ordinary,
172    }
173}

179    fn image(mode: u32, uid: u32, gid: u32) -> SetIdImage {
180        SetIdImage {
181            mode: libc::S_IFREG | mode,
182            uid,
183            gid,
184            uid_mapped: true,
185            gid_mapped: true,
186            nosuid: false,
187        }
188    }

190    /// A set-user-ID image owned by someone else, or a set-group-ID image
191    /// with group execute whose group is not 0, gives the virtual root that
192    /// owner or group as its effective id and makes the exec secure; an image
193    /// owned by root, a set-group-ID bit without group execute, and plain
194    /// images change neither.
195    #[test]
196    fn set_id_bits_that_move_an_effective_id_make_the_exec_secure() {
197        for (mode, uid, gid, ids) in [
198            (0o4755, 1000, 0, (1000, 0)),
199            (0o4755, 0, 0, (0, 0)),
200            (0o2755, 0, 1000, (0, 1000)),
201            (0o2745, 0, 1000, (0, 0)),
202            (0o2755, 0, 0, (0, 0)),
203            (0o6755, 0, 1000, (0, 1000)),
204            (0o6755, 1000, 1000, (1000, 1000)),
205            (0o0755, 1000, 1000, (0, 0)),
206        ] {
207            assert_eq!(
208                effective_ids_for_virtual_root(&image(mode, uid, gid), false),
209                ids,
210                "{mode:o} {uid} {gid}"
211            );
212        }
213    }
214
215    /// `no_new_privs`, a `nosuid` mount and an owner or group the namespace
216    /// does not map each void the set-id bits, as `bprm_fill_uid` does.
217    #[test]
218    fn no_new_privs_nosuid_and_unmapped_ids_void_the_set_id_bits() {
219        let secure = image(0o4755, 1000, 0);
220        assert_eq!(effective_ids_for_virtual_root(&secure, false), (1000, 0));
221        assert_eq!(effective_ids_for_virtual_root(&secure, true), (0, 0));
222        for void in [
223            SetIdImage {
224                nosuid: true,
225                ..secure
226            },
227            SetIdImage {
228                uid_mapped: false,
229                ..secure
230            },
231            SetIdImage {
232                gid_mapped: false,
233                ..secure
234            },
235        ] {
236            assert_eq!(
237                effective_ids_for_virtual_root(&void, false),
238                (0, 0),
239                "{void:?}"
240            );
241        }
242    }

244    /// An id map covers exactly its ranges; a status text yields its
245    /// `NoNewPrivs:` value.
246    #[test]
247    fn id_maps_and_no_new_privs_are_read_as_linux_writes_them() {
248        let single = b"         0     212630          1\n";
249        assert!(maps(single, 0));
250        assert!(!maps(single, 1));
251        assert!(!maps(single, 65534));
252        let wide = b"         0     212630          1\n         1 1879048192      65536\n";
253        assert!(maps(wide, 1000));
254        assert!(maps(wide, 65536));
255        assert!(!maps(wide, 65537));
256        assert!(maps(b"         0          0 4294967295\n", 65534));
257        assert_eq!(no_new_privs(b"Name:\tx\nNoNewPrivs:\t1\n"), Some(true));
258        assert_eq!(no_new_privs(b"NoNewPrivs:\t0\n"), Some(false));
259        assert_eq!(no_new_privs(b"Name:\tx\n"), None);
260    }

262    /// The image's own `AT_SECURE` decides a secure exec, expected or not; an
263    /// exec Linux runs securely that the backend ran without it is
264    /// unsupported; anything else is ordinary.
265    #[test]
266    fn the_outcome_follows_the_image_and_refuses_a_missing_secure_exec() {
267        use SecureExecOutcome::*;
268        for (expected, reported, result) in [
269            (None, None, Ordinary),
270            (None, Some(false), Ordinary),
271            (Some(false), Some(false), Ordinary),
272            (None, Some(true), Secure),
273            (Some(false), Some(true), Secure),
274            (Some(true), Some(true), Secure),
275            (Some(true), Some(false), Unsupported),
276            (Some(true), None, Unsupported),
277        ] {
278            assert_eq!(
279                outcome(expected, reported),
280                result,
281                "{expected:?} {reported:?}"
282            );
283        }
284    }
