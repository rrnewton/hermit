9//! Keys for numbering the procfs entries that the host renumbers.
10//!
11//! Linux numbers most procfs inodes when it builds their dentries: every
12//! per-process entry (`/proc/<pid>`, `/proc/<pid>/fd/<n>`, ...) and every
13//! `/proc/sys` entry takes the next value of one host-wide counter
14//! (`get_next_ino`). An unused dentry can be evicted at any time by memory
15//! pressure on the host, or dropped by a failed lookup, and the next lookup
16//! or `readdir` builds it again under a new number. The inode pool numbers a
17//! host `(device, inode)` it has not seen as a new file, so the guest saw a
18//! number that depended on whether the host had rebuilt the entry since the
19//! guest last saw it: a second `lsof` listing of `/proc/3/fd` after the host
20//! rebuilt its entries got new numbers in one `--verify` run and the old ones
21//! in the other.
22//!
23//! So a procfs sighting also names the entry ([`ProcfsName`]): the key of its
24//! path within procfs ([`path_inode`]), which does not depend on where procfs
25//! is mounted, and the task IDs that path names. The pool numbers an entry by
26//! its host inode first: a host inode it already knows keeps its number, so
27//! aliases (one entry listed in `/proc/<pid>/net` and reached as
28//! `/proc/<pid>/task/<pid>/net/tcp`) share it. Only a host inode the pool has
29//! never seen, reached by a path it has seen, is that path's entry rebuilt,
30//! and takes the path's number. A path key only ever leads to a host key; it
31//! numbers nothing itself. A task ID that a new thread reuses names a new
32//! task, whose entries are new files ([`with_incarnation`]), and an entry a
33//! directory snapshot lists names the tasks that held its IDs when the
34//! snapshot was taken ([`ProcfsName::in_snapshot`]).
35//!
36//! Detcore names an entry only once the guest's mount table says its device
37//! is a procfs mount ([`procfs_mounts`]), and reads its canonical path from a
38//! magic link in the guest thread's `/proc/<tid>` ([`inspect_link`]): the
39//! link of the descriptor or working directory the call described, or, for a
40//! call that named a path, the link of an `O_PATH` descriptor that Detcore
41//! opens with the guest's own path, directory and root, and closes again
42//! ([`read_open`], [`read_close`]). The mount the object was reached through,
43//! which `statx` names, gives the mount point to take off that path and the
44//! directory of procfs it shows, once that mount point, resolved without
45//! following a link, leads to the root of the same mount ([`path_within`],
46//! [`mount_root`]). When it cannot name a procfs entry (the mount table cannot
47//! be read, the guest has no descriptor left or no longer holds the path it
48//! named, the call named a path while the guest's threads are not
49//! sequentialized, an injected call had no effect in every attempt, the
50//! object found is not the one the guest saw, or was reached through a mount
51//! the table does not list at a mount point that holds it) it numbers
52//! the entry by its host inode and records a determinism loss, so the run is
53//! not compared.
54//!
55//! Each of these numbers the entry as Detcore did before:
56//!
57//! - An entry rebuilt before Detcore first saw its path gets a new number,
58//!   which that path then leads to.
59//! - A path lookup crosses every mount on the guest's path again, and the
60//!   check of a mount point every mount above it, so a mount on a FUSE file
61//!   system whose server is a guest Detcore holds could stall it.
62//! - `get_next_ino` wraps at 2^32, so a directory other than a procfs root can
63//!   have inode 1, and has its link count reported as 1.
64//! - Under a backend whose guest thread IDs do not name host tasks, there is
65//!   no `/proc/<tid>` to read, and procfs entries keep their host keys: KVM's
66//!   guest threads are not host tasks, and DBT's thread IDs are its
67//!   scheduler's virtual IDs, in no PID namespace that gives its host tasks
68//!   those IDs.

107/// The inode number a procfs file Detcore renders reports, on every backend:
108/// a function of its path alone, so that no other observation of the run can
109/// change it. It is in a range Detcore's inode pool, which counts up from a
110/// few thousand, never reaches.
111///
112/// A per-process file's path carries the task it belongs to
113/// (`procfs::per_process_name`), and its number is that task's
114/// ([`per_process_inode`]) rather than a hash of its text, so that no two
115/// tasks' files can share one.
116pub(crate) fn rendered_inode(path: &Path) -> u64 {
117    const RENDERED_INODE_BASE: u64 = 0xf000_0000;
118    if let Some(name) = crate::procfs::per_process_name(path) {
119        return per_process_inode(name);
120    }
121    RENDERED_INODE_BASE | (fnv1a(FNV_OFFSET_BASIS, path.as_os_str().as_bytes()) & 0x0fff_ffff)
122}

124/// The most incarnations an inode number of a per-process file tells apart,
125/// for its process and for its thread each: how many tasks of one run may
126/// hold one number ([`per_process_inode`]). The 256th task to hold a number
127/// (the 255th reuse of it) has no inode number of its own; Detcore refuses to
128/// name its per-process files, by name (`Detcore::bind_task_identity`),
129/// rather than give them another task's number.
130pub(crate) const MAX_INCARNATION: u32 = (1 << INCARNATION_BITS) - 1;

132/// Bits of a per-process inode number for each incarnation.
133const INCARNATION_BITS: u32 = 8;
134
135/// Bits of a per-process inode number for a pid or a tid: Linux numbers
136/// tasks below `PID_MAX_LIMIT`, 2^22.
137const PID_BITS: u32 = 22;
138
139/// Bits of a per-process inode number for the file's index among the
140/// per-process files Detcore renders, from 1.
141const INDEX_BITS: u32 = 3;
142
143const _: () = assert!(
144    crate::procfs::ProcfsFile::PER_PROCESS_NAMES.len() < 1 << INDEX_BITS,
145    "every per-process file needs an index in its inode number"
146);
147const _: () = assert!(1 + INDEX_BITS + 2 * PID_BITS + 2 * INCARNATION_BITS == 64);

149/// Whether a per-process file of the task with identity `incarnation` has an
150/// inode number of its own ([`MAX_INCARNATION`]).
151pub(crate) fn incarnation_fits(incarnation: crate::procfs::TaskIncarnation) -> bool {
152    incarnation.process <= MAX_INCARNATION
153        && incarnation
154            .thread
155            .is_none_or(|thread| thread <= MAX_INCARNATION)
156}

158/// The inode number of the per-process file `name`, on every backend: the
159/// fields of its identity laid out in 64 bits, so that two files share a
160/// number only if they are the same file of the same task. Bit 63 is set,
161/// which neither a system-wide rendered file's number (below 2^32, above) nor
162/// any number Detcore's inode pool mints (counting up from a few thousand)
163/// has. Then, from the high bits: the file's index among the per-process
164/// files Detcore renders (3 bits, from 1), the process (22 bits) and the
165/// thread below its `task` (22 bits, 0 for none), and the incarnation of each
166/// (8 bits each, from 1; 0 for no thread). Linux numbers tasks below
167/// `PID_MAX_LIMIT`, 2^22, so 22 bits hold every pid and tid, and a task Linux
168/// could not number has no file (its lookup is ENOENT first). An incarnation
169/// holds up to [`MAX_INCARNATION`], 255: 254 reuses of one number in a run.
170/// A greater one is refused by name before a file is named
171/// (`Detcore::bind_task_identity`); a value outside a field here is a
172/// Detcore defect, and panics rather than wraps into another file's number.
173pub(crate) fn per_process_inode(name: crate::procfs::PerProcessName) -> u64 {
174    let index = crate::procfs::ProcfsFile::PER_PROCESS_NAMES
175        .iter()
176        .position(|candidate| *candidate == name.filename)
177        .expect("a per-process file Detcore renders has an index")
178        + 1;
179    let field = |value: i64, bits: u32, what: &str| -> u64 {
180        assert!(
181            (0..1_i64 << bits).contains(&value),
182            "{what} {value} of {} does not fit the {bits}-bit field of its inode number",
183            name.task.process
184        );
185        value as u64
186    };
187    let index = field(index as i64, INDEX_BITS, "file index");
188    let process = field(name.task.process.into(), PID_BITS, "pid");
189    let thread = field(name.task.thread.unwrap_or(0).into(), PID_BITS, "tid");
190    let incarnation = name.incarnation.unwrap_or(crate::procfs::TaskIncarnation {
191        process: 0,
192        thread: None,
193    });
194    let process_incarnation = field(
195        incarnation.process.into(),
196        INCARNATION_BITS,
197        "pid incarnation",
198    );
199    let thread_incarnation = field(
200        incarnation.thread.unwrap_or(0).into(),
201        INCARNATION_BITS,
202        "tid incarnation",
203    );
204    1 << 63
205        | index << (PID_BITS * 2 + INCARNATION_BITS * 2)
206        | process << (PID_BITS + INCARNATION_BITS * 2)
207        | thread << (INCARNATION_BITS * 2)
208        | process_incarnation << INCARNATION_BITS
209        | thread_incarnation
210}

217/// Continues a 64-bit FNV-1a hash from `hash` over `bytes`.
218fn fnv1a(hash: u64, bytes: &[u8]) -> u64 {
219    const PRIME: u64 = 0x0000_0100_0000_01b3;
220    bytes.iter().fold(hash, |hash, &byte| {
221        (hash ^ u64::from(byte)).wrapping_mul(PRIME)
222    })
223}

238/// Whether a stat result is the procfs of a backend whose `/proc` is its
239/// own: where the tool cannot name guest descriptors through host procfs
240/// ([`tool_can_name_guest_descriptors`], KVM), the backend serves `/proc`
241/// itself, on [`KVM_SYNTHETIC_PROC_DEVICE`] with 4096-byte blocks, and that
242/// device is the proof. Elsewhere the kernel's procfs is proved by naming the
243/// entry (`ProcfsEntry`), not by its stat fields: an anonymous device with
244/// 1024-byte blocks is also devpts, autofs or a FUSE mount that reports those
245/// blocks (Codex re-check of https://github.com/rrnewton/hermit/pull/4369).
246pub(crate) fn is_backend_procfs_device(
247    backend: &BackendCapabilities,
248    device: u64,
249    block_size: i64,
250) -> bool {
251    !tool_can_name_guest_descriptors(backend)
252        && device == KVM_SYNTHETIC_PROC_DEVICE
253        && block_size == 4096
254}

257/// prove it: where guest thread IDs do not name host tasks
258/// ([`guest_tids_name_host_tasks`]), Detcore names no procfs entries, but
259/// where the tool still shares the kernel's procfs with the guest
260/// ([`tool_can_name_guest_descriptors`]: DBT), a file with procfs's stat
261/// fields ([`may_be_procfs`]) may be procfs. Taking it as an ordinary file
262/// turned `/proc`'s size 0 and SEEK_END 0 into 4096 and EINVAL there (Codex
263/// re-check of https://github.com/rrnewton/hermit/pull/4369), so it keeps
264/// what it had before; so does a FUSE directory reporting 1024-byte blocks,
265/// until Detcore names procfs entries there
266/// (https://github.com/rrnewton/hermit/issues/4395).
267pub(crate) fn unprovable_procfs_signature(
268    backend: &BackendCapabilities,
269    physical_tid: Option<i32>,
270    device: u64,
271    block_size: i64,

273    tool_can_name_guest_descriptors(backend)
274        && !guest_tids_name_host_tasks(backend, physical_tid)
275        && may_be_procfs(device, block_size)
276}

316/// The path key `key` of an entry that names task ID `task`, for the
317/// `incarnation`th thread created under that ID: the hash continued over
318/// both, with [`PATH_KEY_BIT`] set.
319pub(crate) fn with_incarnation(key: RawInode, task: i32, incarnation: u32) -> RawInode {
320    let hash = fnv1a(
321        fnv1a(key.ino, &task.to_le_bytes()),

361/// The task ID a path component names, if it is one.
362fn task_id(component: Option<Component<'_>>) -> Option<i32> {
363    match component {
364        Some(Component::Normal(name)) => name
365            .to_str()?
366            .parse::<i32>()
367            .ok()
368            .filter(|id| *id > 0 && name.as_bytes()[0] != b'+'),
369        _ => None,
370    }
371}

373/// The task IDs a path within procfs names: the process of `/<pid>/...`, and
374/// then the thread of `/<pid>/task/<tid>/...`.
375fn tasks_named(within: &Path) -> Vec<i32> {
376    let mut components = within
377        .components()
378        .skip_while(|component| *component == Component::RootDir);
379    let Some(pid) = task_id(components.next()) else {
380        return Vec::new();
381    };
382    let mut tasks = vec![pid];
383    if components.next() == Some(Component::Normal(OsStr::new("task")))
384        && let Some(tid) = task_id(components.next())
385    {
386        tasks.push(tid);
387    }
388    tasks
389}

397    root: PathBuf,
398    mount_point: PathBuf,

401/// The procfs mounts of `device` in the mount namespace of the guest thread
402/// `tid`, by its `/proc/<tid>/mountinfo`, which Linux writes without touching
403/// any mounted file system: `Ok(None)` when `device` is not procfs there, and
404/// an error when the tracer cannot tell, or when the descriptor that read the
405/// table did not close (see [`read_checking_close`]).
406///
407/// The table gives each mount point relative to the guest thread's root,
408/// while a magic link reads as a path relative to the root of the thread
409/// that reads it (the tracer's, or, under SaBRe, the guest's own) when it
410/// leads below that root, and from the root of the mount namespace when it
411/// does not. So each mount point is put below the path the guest thread's
412/// root reads as, from its `/proc/<tid>/root`, which the same thread reads.
413/// That puts a mount point and a link in one frame when the guest's root is
414/// below the reader's, but not always when it is not, so [`path_within`]
415/// uses a mount point only where the reader finds the mount.
416///
417/// Detcore refuses `mount`, `unshare` and `setns` once it starts, so a guest's
418/// procfs mount stays one: the mounts of a device found to be procfs are
419/// remembered with its mount namespace and the guest thread's root, which a
420/// `chroot` changes, by its mount, device and inode, and the path it reads
421/// as. Any other device is looked up again, as it may yet become one. A
422/// rename of a directory above a mount point moves the mount and changes
423/// none of these, so [`path_within`] asks for the table again (`reread`),
424/// which reads it and remembers what it finds, or forgets the device when it
425/// is no longer procfs there. A guest thread that changes the root while
426/// Detcore reads for another thread that shares it, which thread
427/// sequentialization rules out, can leave the two paths in different frames.
428pub(crate) fn procfs_mounts(
429    tid: i32,
430    device: u64,
431    reread: bool,
432) -> io::Result<Option<Arc<[ProcfsMount]>>> {
433    // The mount namespace, the root's mount, device, inode and path, and
434    // the device.
435    type Key = (u64, Option<u64>, u64, u64, PathBuf, u64);
436    static PROCFS_DEVICES: OnceLock<Mutex<HashMap<Key, Arc<[ProcfsMount]>>>> = OnceLock::new();
437    let namespace = std::fs::metadata(format!("/proc/{tid}/ns/mnt"))?;
438    let root_link = PathBuf::from(format!("/proc/{tid}/root"));
439    let root = identify(&root_link, true)?;
440    let root_path = std::fs::read_link(&root_link)?;
441    let key = (
442        namespace.ino(),
443        root.mount_id,
444        root.device,
445        root.inode,
446        root_path.clone(),
447        device,
448    );
449    let known = PROCFS_DEVICES.get_or_init(Default::default);
450    if !reread && let Some(mounts) = known.lock().unwrap().get(&key) {
451        return Ok(Some(mounts.clone()));
452    }
453    let mountinfo = read_checking_close(&format!("/proc/{tid}/mountinfo"))?;
454    let mounts = procfs_mounts_in(&mountinfo, device, &root_path);
455    if mounts.is_empty() {
456        known.lock().unwrap().remove(&key);
457        return Ok(None);
458    }
459    let mounts: Arc<[ProcfsMount]> = mounts.into();
460    known.lock().unwrap().insert(key, mounts.clone());
461    Ok(Some(mounts))
462}

464/// How many times Detcore makes a call of its own to read the guest's mount
465/// table that was interrupted (`EINTR`) before it gives up. Under SaBRe the
466/// call runs in the guest's process, under the guest's seccomp filters, and
467/// a filter can return `EINTR` every time without running it.
468const INTERRUPTED_ATTEMPTS: usize = 8;
469
470/// The contents of the file at `path`, read through a descriptor that must
471/// close. Under SaBRe, Detcore runs in the guest's process, so its
472/// descriptors are in the guest's descriptor table. A `close` that a seccomp
473/// filter refuses leaves the descriptor open there, where the guest sees it
474/// and its later descriptors get other numbers, and `File` ignores that
475/// error when it drops. So a failed close is an error, as it is for the
476/// descriptors Detcore opens in the guest to name a path.
477///
478/// The open and each read are tried again when interrupted, but only as
479/// [`retrying_interrupted`] allows, where `File::open` and `read_to_end`
480/// would try again without end. The close is not: Linux releases the
481/// descriptor before anything that can interrupt it.
482fn read_checking_close(path: &str) -> io::Result<Vec<u8>> {
483    let c_path = c_path(Path::new(path))?;
484    let fd = retrying_interrupted("open", path, || {
485        // SAFETY: `c_path` is NUL-terminated.
486        let fd = unsafe { libc::open(c_path.as_ptr(), libc::O_RDONLY | libc::O_CLOEXEC) };
487        if fd < 0 {
488            Err(io::Error::last_os_error())
489        } else {
490            Ok(fd)
491        }
492    })?;
493    let read = read_to_end(fd, path);
494    // SAFETY: `fd` is the descriptor open returned, and nothing else uses it.
495    if unsafe { libc::close(fd) } != 0 {
496        return Err(io::Error::other(format!(
497            "cannot close descriptor {fd} of {path}: {}",
498            describe_io_error(&io::Error::last_os_error())
499        )));
500    }
501    read
502}

504/// Everything left to read from the descriptor `fd` of `path`, one `read` at
505/// a time, each tried again when interrupted only as [`retrying_interrupted`]
506/// allows.
507fn read_to_end(fd: RawFd, path: &str) -> io::Result<Vec<u8>> {
508    let mut contents = Vec::new();
509    let mut chunk = [0u8; 4096];
510    loop {
511        let count = retrying_interrupted("read", path, || {
512            // SAFETY: `chunk` is writable for its whole length.
513            let count = unsafe { libc::read(fd, chunk.as_mut_ptr().cast(), chunk.len()) };
514            usize::try_from(count).map_err(|_| io::Error::last_os_error())
515        })?;
516        if count == 0 {
517            return Ok(contents);
518        }
519        contents.extend_from_slice(&chunk[..count]);
520    }
521}

523/// The result of `call`, the `name` call on `path`, which is made again
524/// while it fails with `EINTR`, but at most [`INTERRUPTED_ATTEMPTS`] times.
525/// Running out is an error of another kind, so that no caller tries again.
526fn retrying_interrupted<T>(
527    name: &str,
528    path: &str,
529    mut call: impl FnMut() -> io::Result<T>,
530) -> io::Result<T> {
531    for _ in 0..INTERRUPTED_ATTEMPTS {
532        match call() {
533            Err(error) if error.kind() == io::ErrorKind::Interrupted => {}
534            done => return done,
535        }
536    }
537    Err(io::Error::other(format!(
538        "the {name} of {path} was interrupted in each of {INTERRUPTED_ATTEMPTS} attempts"
539    )))
540}

542/// The procfs mounts of `device` a `mountinfo` table lists: lines whose third
543/// field is the device's `major:minor` and whose file system type, the field
544/// after the `-` separator that ends the optional fields, is `proc`. Each
545/// mount point, which the table gives relative to the root of the thread it
546/// describes, is put below `guest_root`, the path that root reads as.
547fn procfs_mounts_in(mountinfo: &[u8], device: u64, guest_root: &Path) -> Vec<ProcfsMount> {
548    let wanted = format!("{}:{}", libc::major(device), libc::minor(device));
549    mountinfo
550        .split(|byte| *byte == b'\n')
551        .filter_map(|line| {
552            let fields: Vec<&[u8]> = line.split(|byte| *byte == b' ').collect();
553            if fields.get(2).copied() != Some(wanted.as_bytes()) {
554                return None;
555            }
556            let separator = 6 + fields.get(6..)?.iter().position(|field| *field == b"-")?;
557            if fields.get(separator + 1).copied() != Some(b"proc".as_slice()) {
558                return None;
559            }
560            let mount_point = unescape(fields[4]);
561            Some(ProcfsMount {
562                id: std::str::from_utf8(fields[0]).ok()?.parse().ok()?,
563                root: unescape(fields[3]),
564                mount_point: guest_root.join(mount_point.strip_prefix("/").unwrap_or(&mount_point)),
565            })
566        })
567        .collect()
568}

570/// A `mountinfo` path with its octal escapes (`\040` for a space) decoded.
571fn unescape(field: &[u8]) -> PathBuf {
572    let mut path = Vec::with_capacity(field.len());
573    let mut index = 0;
574    while index < field.len() {
575        let octal = field
576            .get(index + 1..index + 4)
577            .filter(|digits| field[index] == b'\\' && digits.iter().all(u8::is_ascii_digit))
578            .filter(|digits| digits.iter().all(|digit| *digit < b'8'));
579        match octal {
580            Some(digits) => {
581                path.push(digits.iter().fold(0u8, |value, digit| {
582                    value.wrapping_mul(8).wrapping_add(digit - b'0')
583                }));
584                index += 4;
585            }
586            None => {
587                path.push(field[index]);
588                index += 1;
589            }
590        }
591    }
592    PathBuf::from(OsString::from_vec(path))
593}

595/// The path within procfs of the object at the canonical path `canonical`,
596/// reached through the mount `mount_id`: its path below that mount's mount
597/// point, joined to the mount's root. The mount point must be where the
598/// reader finds the root of that mount: `mount_root` gives the mount whose
599/// root a path leads to, as the thread that read `canonical` resolves it
600/// (see [`mount_root`]).
601///
602/// That check makes the name the object's, whichever table the mount point
603/// came from. Along a path that follows no link, which `mount_root` resolves
604/// the mount point as, the reader reaches the root of a mount only through
605/// its one mount point, and `canonical` reads, from the same root, through
606/// that mount point and then the object's path within the mount. So a prefix of
607/// `canonical` where the reader finds the mount is that mount point, and the
608/// rest is the path within. Without it, a mount point remembered from before
609/// a rename of a directory above it, or one read in another frame than the
610/// link (see [`procfs_mounts`]), can be another prefix of `canonical`, and
611/// give the entry another name with no loss.
612///
613/// When the check fails with `mounts`, `reread` gives the guest's table read
614/// again, which is tried once more. An error is the reason no name was found.
615pub(crate) fn path_within(
616    mounts: &[ProcfsMount],
617    mount_id: u64,
618    canonical: &Path,
619    reread: impl FnOnce() -> io::Result<Option<Arc<[ProcfsMount]>>>,
620    mount_root: impl Fn(&Path) -> io::Result<Option<u64>>,
621) -> Result<PathBuf, String> {
622    if let Ok(within) = within_table(mounts, mount_id, canonical, &mount_root) {
623        return Ok(within);
624    }
625    match reread() {
626        Ok(Some(mounts)) => within_table(&mounts, mount_id, canonical, &mount_root),
627        Ok(None) => Err(format!(
628            "{} was reached through mount {mount_id}, and the guest's mount table, read again, lists no procfs mount of its device",
629            canonical.display()
630        )),
631        Err(error) => Err(format!(
632            "{} was reached through mount {mount_id}, and the guest's mount table cannot be read again: {}",
633            canonical.display(),
634            describe_io_error(&error)
635        )),
636    }
637}

639/// [`path_within`] with the mounts of one table.
640fn within_table(
641    mounts: &[ProcfsMount],
642    mount_id: u64,
643    canonical: &Path,
644    mount_root: &impl Fn(&Path) -> io::Result<Option<u64>>,
645) -> Result<PathBuf, String> {
646    let reached = format!(
647        "{} was reached through mount {mount_id}",
648        canonical.display()
649    );
650    let mount = mounts
651        .iter()
652        .find(|mount| mount.id == mount_id)
653        .ok_or_else(|| {
654            format!("{reached}, which the guest's mount table does not list as a procfs mount")
655        })?;
656    let point = mount.mount_point.display();
657    let rest = canonical.strip_prefix(&mount.mount_point).map_err(|_| {
658        format!("{reached}, which the guest's mount table lists at {point}, not above it")
659    })?;
660    match mount_root(&mount.mount_point) {
661        Ok(Some(found)) if found == mount_id => {}
662        Ok(Some(found)) => {
663            return Err(format!(
664                "{reached}, and {point}, where the guest's mount table lists it, is the root of mount {found}"
665            ));
666        }
667        Ok(None) => {
668            return Err(format!(
669                "{reached}, and {point}, where the guest's mount table lists it, is the root of no mount"
670            ));
671        }
672        Err(error) => {
673            return Err(format!(
674                "{reached}, and cannot tell whether {point}, where the guest's mount table lists it, holds it: {}",
675                describe_io_error(&error)
676            ));
677        }
678    }
679    if rest.as_os_str().is_empty() {
680        Ok(mount.root.clone())
681    } else {
682        Ok(mount.root.join(rest))
683    }
684}

686/// The canonical path of the procfs object behind the magic link `link` (a
687/// `/proc/<tid>/fd/<n>` or `/proc/<tid>/cwd`), which must be inode `inode` on
688/// `device`, and the ID of the mount it was reached through, which places the
689/// path in the guest's mount table (see [`path_within`]). Following such a
690/// link touches only procfs and the object, which is on procfs.
691///
692/// Once a task is reaped, a stat through a link to one of its entries fails,
693/// and the link reads as the entry's path with ` (deleted)` appended, as for
694/// a removed file. The task can be reaped between the two, so such a path is
695/// refused too (see [`refuse_deleted`]).
696pub(crate) fn inspect_link(link: &Path, device: u64, inode: u64) -> Result<(PathBuf, u64), String> {
697    let mount_id = check_link_object(link, device, inode)?
698        .ok_or_else(|| format!("the kernel does not report the mount of {}", link.display()))?;
699    let canonical = std::fs::read_link(link).map_err(|error| {
700        format!(
701            "cannot read the link {}: {}",
702            link.display(),
703            describe_io_error(&error)
704        )
705    })?;
706    Ok((refuse_deleted(link, canonical)?, mount_id))
707}
708
709/// `canonical`, read from the magic link `link`, unless it ends in
710/// ` (deleted)`, as the path of an entry of a reaped task does. No procfs
711/// name ends that way, so such a path names nothing.
712fn refuse_deleted(link: &Path, canonical: PathBuf) -> Result<PathBuf, String> {
713    if canonical.as_os_str().as_bytes().ends_with(b" (deleted)") {
714        return Err(format!(
715            "{} leads to {}, an entry of a task that has exited",
716            link.display(),
717            canonical.display()
718        ));
719    }
720    Ok(canonical)
721}

723/// Whether the object behind the magic link `link`, which must be inode
724/// `inode` on `device`, is on a procfs filesystem, whatever mount it was
725/// reached through.
726pub(crate) fn is_on_procfs(link: &Path, device: u64, inode: u64) -> Result<bool, String> {
727    check_link_object(link, device, inode)?;
728    let path = CString::new(link.as_os_str().as_bytes())
729        .map_err(|_| format!("{} contains a NUL byte", link.display()))?;
730    let mut buf = std::mem::MaybeUninit::<libc::statfs>::uninit();
731    // SAFETY: `path` is NUL-terminated and `buf` is large enough for statfs.
732    if unsafe { libc::statfs(path.as_ptr(), buf.as_mut_ptr()) } != 0 {
733        return Err(format!(
734            "cannot statfs {}: {}",
735            link.display(),
736            describe_io_error(&io::Error::last_os_error())
737        ));
738    }
739    // SAFETY: statfs succeeded, so it filled `buf`.
740    Ok(unsafe { buf.assume_init() }.f_type == libc::PROC_SUPER_MAGIC)
741}

743/// Requires that the magic link `link` lead to inode `inode` on `device`, and
744/// returns the ID of the mount the object was reached through, when the
745/// kernel reports one.
746fn check_link_object(link: &Path, device: u64, inode: u64) -> Result<Option<u64>, String> {
747    let found = identify(link, true).map_err(|error| {
748        format!(
749            "cannot stat {}: {}",
750            link.display(),
751            describe_io_error(&error)
752        )
753    })?;
754    if found.device != device || found.inode != inode {
755        return Err(format!(
756            "{} is not the object the guest saw",
757            link.display()
758        ));
759    }
760    Ok(found.mount_id)
761}

763/// The device, inode and mount of a file, by `statx`.
764struct FileIdentity {
765    device: u64,
766    inode: u64,
767    /// The ID of the mount the file was reached through, the first field of
768    /// its line in `mountinfo`, when the kernel reports it (`STATX_MNT_ID`,
769    /// from Linux 5.8).
770    mount_id: Option<u64>,
771    /// Whether the file is the root of that mount, when the kernel reports it
772    /// (`STATX_ATTR_MOUNT_ROOT`, from Linux 5.8).
773    mount_root: Option<bool>,

776/// The identity of the file at `path`, following a final link when `follow`
777/// is set, as a magic link is followed: to the object it leads to, through
778/// the mount it was reached through.
779fn identify(path: &Path, follow: bool) -> io::Result<FileIdentity> {
780    // `AT_NO_AUTOMOUNT`, which `stat` implies, so that looking does not
781    // trigger an automount at the path's last component.
782    let mut flags = libc::AT_NO_AUTOMOUNT;
783    if !follow {
784        flags |= libc::AT_SYMLINK_NOFOLLOW;
785    }
786    identify_at(libc::AT_FDCWD, &c_path(path)?, flags)
787}

795/// The identity of the file at `path`, looked up from the directory `dirfd`
796/// with the statx flags `flags`.
797fn identify_at(dirfd: RawFd, path: &CStr, flags: i32) -> io::Result<FileIdentity> {
798    let mut buf = std::mem::MaybeUninit::<libc::statx>::zeroed();
799    let mask = libc::STATX_INO | libc::STATX_MNT_ID;
800    // SAFETY: `path` is NUL-terminated and `buf` is large enough for statx.
801    if unsafe { libc::statx(dirfd, path.as_ptr(), flags, mask, buf.as_mut_ptr()) } != 0 {
802        return Err(io::Error::last_os_error());
803    }
804    // SAFETY: statx succeeded, so it filled `buf`.
805    let found = unsafe { buf.assume_init() };
806    if found.stx_mask & libc::STATX_INO == 0 {
807        return Err(io::Error::other("statx reported no inode number"));
808    }
809    let mount_root = libc::STATX_ATTR_MOUNT_ROOT as u64;
810    Ok(FileIdentity {
811        device: libc::makedev(found.stx_dev_major, found.stx_dev_minor),
812        inode: found.stx_ino,
813        mount_id: (found.stx_mask & libc::STATX_MNT_ID != 0).then_some(found.stx_mnt_id),
814        mount_root: (found.stx_attributes_mask & mount_root != 0)
815            .then_some(found.stx_attributes & mount_root != 0),
816    })
817}

819/// The ID of the mount whose root the path `path` leads to, as the calling
820/// thread resolves it, without following a link anywhere in the path: `None`
821/// when it leads to no mount's root, and an error when the path holds a link,
822/// or the kernel does not report mount roots.
823///
824/// The path is resolved with `openat2`'s `RESOLVE_NO_SYMLINKS`, which refuses
825/// every link, a magic link included. A path that follows no link and holds
826/// no `..` reaches the root of a mount only through that mount's one mount
827/// point, which [`path_within`] relies on; a link can lead to the same
828/// directory along another path. The descriptor is opened `O_PATH`, which,
829/// like `stat`, does not trigger an automount at the path's last component
830/// (the lookup of an earlier component can, as any lookup's can), and must
831/// close, as for [`read_checking_close`].
832pub(crate) fn mount_root(path: &Path) -> io::Result<Option<u64>> {
833    let path = c_path(path)?;
834    // SAFETY: `open_how` holds only integers, and zero is the value the
835    // kernel requires of every field this call does not set.
836    let mut how: libc::open_how = unsafe { std::mem::zeroed() };
837    how.flags = (libc::O_PATH | libc::O_CLOEXEC) as u64;
838    how.resolve = libc::RESOLVE_NO_SYMLINKS;
839    // SAFETY: `path` is NUL-terminated, and `how` is an `open_how` of the
840    // size passed. glibc has no wrapper for openat2.
841    let fd = unsafe {
842        libc::syscall(
843            libc::SYS_openat2,
844            libc::AT_FDCWD,
845            path.as_ptr(),
846            &how as *const libc::open_how,
847            std::mem::size_of::<libc::open_how>(),
848        )
849    };
850    if fd < 0 {
851        return Err(io::Error::last_os_error());
852    }
853    let fd = RawFd::try_from(fd)
854        .map_err(|_| io::Error::other(format!("openat2 returned {fd}, which is no descriptor")))?;
855    let found = identify_at(fd, c"", libc::AT_EMPTY_PATH | libc::AT_NO_AUTOMOUNT);
856    // SAFETY: `fd` is the descriptor openat2 opened, and nothing else uses it.
857    if unsafe { libc::close(fd) } != 0 {
858        return Err(io::Error::other(format!(
859            "cannot close descriptor {fd}: {}",
860            describe_io_error(&io::Error::last_os_error())
861        )));
862    }
863    let found = found?;
864    match (found.mount_root, found.mount_id) {
865        (Some(false), _) => Ok(None),
866        (Some(true), Some(id)) => Ok(Some(id)),
867        _ => Err(io::Error::other(
868            "the kernel does not report the root of a mount",
869        )),
870    }
871}

873/// What a system call Detcore injected into a guest did, read from its
874/// result: it ran, and returned this; it had no effect, as the result
875/// reports (`NotRun`), and may be injected again; or it failed, for this
876/// reason.
877#[derive(Debug, PartialEq, Eq)]
878pub(crate) enum Injected<T> {
879    Ran(T),
880    NotRun(String),
881    Failed(String),
882}
883
884/// What an `openat` Detcore injected into the guest to open an `O_PATH`
885/// descriptor did, from its result.
886///
887/// Reverie reports an injection that a signal stopped before it ran as a
888/// restart (`ERESTARTSYS`). An open that returns a restart or `EINTR` opened
889/// no descriptor, whether it ran or not (a signal can interrupt a lookup, as
890/// on FUSE), so it is tried again. Any other result is the call's own: a
891/// descriptor, or the error that refused the open.
892pub(crate) fn read_open(result: Result<i64, Errno>) -> Injected<RawFd> {
893    match result {
894        Ok(fd) => RawFd::try_from(fd).map_or_else(
895            |_| Injected::Failed(format!("openat returned {fd}, which is no descriptor")),
896            Injected::Ran,
897        ),
898        Err(
899            errno @ (Errno::EINTR
900            | Errno::ERESTARTSYS
901            | Errno::ERESTARTNOINTR
902            | Errno::ERESTARTNOHAND
903            | Errno::ERESTART_RESTARTBLOCK),
904        ) => Injected::NotRun(errno.to_string()),
905        Err(errno) => Injected::Failed(format!("cannot open the guest's path again: {errno}")),
906    }
907}
908
909/// What a `close` Detcore injected into the guest to close the `O_PATH`
910/// descriptor it opened did, from its result.
911///
912/// Linux's close releases the descriptor before anything that can fail, and
913/// an `O_PATH` descriptor has nothing to flush, so a close that runs returns
914/// 0, and the descriptor is gone. A restart means that the close did not run,
915/// as for [`read_open`]. Any other error, `EINTR` and `EBADF` among them, is
916/// a seccomp filter the guest inherited refusing the close (or a descriptor
917/// that was not open), which leaves the descriptor open.
918pub(crate) fn read_close(result: Result<i64, Errno>) -> Injected<()> {
919    match result {
920        Ok(0) => Injected::Ran(()),
921        Ok(other) => Injected::Failed(format!(
922            "close of the guest's O_PATH descriptor returned {other}, which close never returns"
923        )),
924        Err(
925            errno @ (Errno::ERESTARTSYS
926            | Errno::ERESTARTNOINTR
927            | Errno::ERESTARTNOHAND
928            | Errno::ERESTART_RESTARTBLOCK),
929        ) => Injected::NotRun(errno.to_string()),
930        Err(errno) => Injected::Failed(format!(
931            "close of the guest's O_PATH descriptor returned {errno:?}"
932        )),
933    }
934}

1065    /// Make every call of the calling thread to one of `syscalls` fail with
1066    /// `errno` without running, as a seccomp filter can. The filter binds this
1067    /// thread and what it starts afterwards, and no other thread.
1068    #[cfg(target_arch = "x86_64")]
1069    fn deny_in_this_thread(syscalls: &[i64], errno: i32) {
1070        const AUDIT_ARCH_X86_64: u32 = 0xc000_003e;
1071        let load = |offset: u32| libc::sock_filter {
1072            code: (libc::BPF_LD | libc::BPF_W | libc::BPF_ABS) as u16,
1073            jt: 0,
1074            jf: 0,
1075            k: offset,
1076        };
1077        let jump_if = |value: u32, jt: u8, jf: u8| libc::sock_filter {
1078            code: (libc::BPF_JMP | libc::BPF_JEQ | libc::BPF_K) as u16,
1079            jt,
1080            jf,
1081            k: value,
1082        };
1083        let ret = |action: u32| libc::sock_filter {
1084            code: (libc::BPF_RET | libc::BPF_K) as u16,
1085            jt: 0,
1086            jf: 0,
1087            k: action,
1088        };
1089        // seccomp_data: nr at 0, arch at 4. Each match jumps over the
1090        // matches after it and the allow, to the denial.
1091        let mut program = vec![
1092            load(4),
1093            jump_if(AUDIT_ARCH_X86_64, 1, 0),
1094            ret(libc::SECCOMP_RET_ALLOW),
1095            load(0),
1096        ];
1097        for (index, syscall) in syscalls.iter().enumerate() {
1098            let over = u8::try_from(syscalls.len() - index).unwrap();
1099            program.push(jump_if(u32::try_from(*syscall).unwrap(), over, 0));
1100        }
1101        program.push(ret(libc::SECCOMP_RET_ALLOW));
1102        program.push(ret(libc::SECCOMP_RET_ERRNO | errno as u32));
1103        let fprog = libc::sock_fprog {
1104            len: program.len() as u16,
1105            filter: program.as_mut_ptr(),
1106        };
1107        assert_eq!(
1108            unsafe { libc::prctl(libc::PR_SET_NO_NEW_PRIVS, 1, 0, 0, 0) },
1109            0
1110        );
1111        let installed = unsafe {
1112            libc::syscall(
1113                libc::SYS_seccomp,
1114                libc::SECCOMP_SET_MODE_FILTER,
1115                0,
1116                &fprog as *const libc::sock_fprog,
1117            )
1118        };
1119        assert_eq!(installed, 0, "seccomp: {}", io::Error::last_os_error());
1120    }
