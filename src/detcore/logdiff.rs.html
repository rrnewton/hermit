41/// Options for calling `log_diff`.
42#[derive(Debug, Parser)]
43pub struct LogDiffOpts {
44    /// UNSAFE: strips numbers and temporary paths before comparison.
45    ///
46    /// This erases timestamps and syscall values that bitwise parity exists to
47    /// compare. Never use this option to make a failing parity diff pass; doing
48    /// so is cheating. It is only for non-parity diagnostic localization.
49    #[clap(long = "unsafe-strip-lines")]
50    pub strip_lines: bool,
51
52    /// Canonicalize host memory addresses before comparison WITHOUT erasing them.
53    ///
54    /// Only addresses a producer has explicitly marked with the
55    /// `<hostaddr 0x...>` wrapper (see [`host_addr`]) are canonicalized; each
56    /// distinct marked address is rewritten to an ordinal placeholder
57    /// `<addr{N}>` assigned by order of first appearance within a single run
58    /// (see `canonicalize_addresses_in_line`). Unlike [`Self::strip_lines`],
59    /// this discards ONLY the host-specific raw pointer value: it preserves
60    /// identity (same address -> same ordinal), ordering (introduction
61    /// sequence), and aliasing (two names for one address collapse to one
62    /// ordinal), and it leaves every other byte -- virtual-time timestamps,
63    /// syscall argument/result values, counts, sizes, flags -- untouched for an
64    /// exact comparison. This is the tier-2 normalization of the `Canonical`
65    /// parity policy: canonicalizing preserves the ability to DETECT a
66    /// difference (allocation-order or aliasing changes still diverge), which
67    /// wholesale stripping throws away.
68    ///
69    /// The marker is REQUIRED (a bare `0x...` literal is left exact) because
70    /// nothing in the compared DETLOG stream can otherwise distinguish a varying
71    /// host pointer from a reproducible hex value -- syscall arguments printed
72    /// `{:#x}` (e.g. `flock` `operation=0x2` vs `0x6`), guest memory ranges,
73    /// content digests, cpuid leaves. A blanket `0x` canonicalization would
74    /// collapse those too, silently erasing real syscall-argument divergence:
75    /// a "softer strip" and exactly the fake-green this policy exists to prevent.
76    #[clap(skip)]
77    pub canonicalize_addresses: bool,
78
79    /// The internal message set to compare.
80    #[clap(skip)]
81    pub comparison: LogComparisonMode,
82    /// Limit the number of differences printed. Set to 0 for no limit.
83    #[clap(long, default_value = "20")]
84    pub limit: u64,
85
86    /// Before comparison, filter out lines which contain this substring.
87    #[clap(long)]
88    pub ignore_lines: Vec<String>,
89
90    /// Show this many completed syscalls before each run-specific divergence point.
91    /// Set to 0 to omit history.
92    #[clap(long, default_value = "0")]
93    pub syscall_history: u64,
94    /// Disable colored console output for line diffs.
95    #[clap(long)]
96    pub no_color: bool,
97
98    /// Do not consider "COMMIT" messages for deterministic checks.
99    #[clap(long)]
100    pub skip_commit: bool,
101
102    /// Do not consider "DETLOG" messages for deterministic checks.
103    #[clap(long)]
104    pub skip_detlog: bool,
105
106    /// Use git diff instead of the internal, basic log comparison.
107    #[clap(long)]
108    pub git_diff: bool,
109
110    /// In case --skip-detlog=false this parameter further filters which
111    /// "DETLOG" messages will be included for deterministic checks
112    #[clap(long, default_values = &["syscall", "syscallresult", "other"])]
113    pub include_detlogs: Vec<DetLogFilter>,
114}

116impl LogDiffOpts {
117    fn is_skip(&self, filter: DetLogFilter) -> bool {
118        !self.include_detlogs.contains(&filter)
119    }
120
121    fn skip_detlog(&self, entry: &str) -> bool {
122        if self.skip_detlog {
123            return true;
124        }
125
126        if is_detlog_syscall(entry) && self.is_skip(DetLogFilter::Syscall) {
127            return true;
128        }
129        if is_detlog_syscall_result(entry) && self.is_skip(DetLogFilter::SyscallResult) {
130            return true;
131        }
132
133        if !is_detlog_syscall(entry)
134            && !is_detlog_syscall_result(entry)
135            && self.is_skip(DetLogFilter::Other)
136        {
137            return true;
138        }
139
140        false
141    }
142
143    fn filter_deterministic<'a>(&self, v: &[(usize, &'a str)]) -> Vec<(usize, &'a str)> {
144        v.iter()
145            .filter_map(|(i, s)| {
146                if (is_detlog(s) && !self.skip_detlog(s) && !is_scheduler_committed_time(s))
147                    || (is_commit(s) && !self.skip_commit && !is_internal_io_poll_commit(s))
148                {
149                    Some((*i, *s))
150                } else {
151                    None
152                }
153            })
154            .collect()
155    }
156}

194/// In fully-deterministic modes, many log lines should be fully determinstic across runs.
195/// But as that is a work-in-progress, this utility strips known-nondeterministic
196/// information from logs.
197///
198/// INPUT: `full_container` bool flag, which specifies whether we should expect that the
199/// log message was generated from a full run of hermit/detcore against a standalone
200/// binary, with all the setup that entails. N.B.: this should be *false* for `spawn_fn_*`
201/// variants which fork from another process, not having a true process tree of their own.
202///
203/// Example input/output:
204///   `Input:  COMMIT turn 3, dettid 231635 using resources Resources { tid: DetPid { inner: 231635 }, resources: {Path("/proc/231635/fd/1"): W} }`
205///   `Output: COMMIT turn <NUM>, dettid <NUM> using resources Resources { tid: DetPid { inner: <NUM> }, resources: {Path("/proc/<pid>/fd/<num>"): W} }`
206///
207/// As you can see this is overkill and smarter strategies would be possible. For example,
208/// ones that remember and post-facto-determinize certain identifiers.
209pub fn strip_log_entry(log: &str) -> String {
210    // Memory addresses, like 0x7fcfb7e7d450
211    //
212    // TODO: use a debug allocator that increases only, never reusing. Also, consider
213    // post-facto processing all of these into new virtual addresses based on the order they're seen.
214    static RE0: LazyLock<Regex> = LazyLock::new(|| Regex::new(r"\b0[xX][A-Fa-f0-9]+\b").unwrap());
215
216    // The basic reason for treating spawn_fn style tests differently is that using
217    // detcore for these forked function calls is really a *partial* version of detcore,
218    // not the full and proper setup.  In contrast, for all command tests, and for `hermit
219    // run` itself, the full contents of the COMMIT line should be deterministic and this
220    // hack should not be needed.
221    // Include common duration suffixes so fractional timing jitter is not left behind.
222    static RE1: LazyLock<Regex> =
223        LazyLock::new(|| Regex::new(r"\b[\d][\d_]*(?:\.[\d][\d_]*)?(?:ns|us|µs|ms)?\b").unwrap()); // This one is terrible overkill.
224
225    // TODO: only strip this information if the config specified to the host /tmp through.
226    // Otherwise we can determinize /tmp access fully.
227    static RE2: LazyLock<Regex> = LazyLock::new(|| Regex::new(r#"/tmp/.*""#).unwrap());
228
229    // TODO: only strip this one if we're allowing through the host /proc or failing to determinize tids/pids:
230    static RE3: LazyLock<Regex> = LazyLock::new(|| Regex::new(r"/proc/[\d]+/").unwrap());
231
232    // TODO: only strip this if we're running a library-based test where we can't
233    // guarantee the starting state of the allocator/etc.
234    static RE4: LazyLock<Regex> = LazyLock::new(|| Regex::new(r"\b[\d][\d_.]*s\b").unwrap());
235
236    let log = RE4.replace_all(log, "<NANOSECONDS>");
237    let log = RE3.replace_all(&log, "/proc/<PID>/");
238    let log = RE0.replace_all(&log, "<ADDR>");
239    let log = RE1.replace_all(&log, "<NUM>");
240    let log = RE2.replace_all(&log, "/tmp/<somewhere>");
241    String::from(log)
242}

244/// Wrap a host memory address so `canonicalize_addresses_in_line` will
245/// canonicalize it. Producers that print a genuinely host-specific pointer
246/// (one that varies run-to-run, e.g. a supervisor-side allocation) should emit
247/// it via this helper -- `<hostaddr 0x7fcfb7e7d450>` -- instead of a bare
248/// `0x...` literal. Only marked addresses are canonicalized, so reproducible hex
249/// (syscall arguments, guest memory ranges, digests) is compared exactly.
250///
251/// Note: nothing in detcore's current DETLOG output prints a varying host
252/// pointer -- guest addresses are determinized and every logged `0x...` value is
253/// reproducible -- so this marker is presently unused in production and exists
254/// so a future host-pointer print opts into canonicalization deliberately rather
255/// than being swept up by a blanket regex.
256pub fn host_addr(addr: usize) -> String {
257    format!("<hostaddr {addr:#x}>")
258}
259
260/// Rewrite each MARKED host memory address (`<hostaddr 0x...>`, see
261/// [`host_addr`]) in `line` to an ordinal placeholder `<addr{N}>`, numbered by
262/// order of first appearance within a single run. `map`/`next` carry the per-run
263/// assignment state threaded across all of that run's lines, so `next` should
264/// start at 1 and the same `map`/`next` must be reused for every line of one run
265/// (and a FRESH pair used for the other run).
266///
267/// This is the tier-2 CANONICALIZE step (as opposed to tier-1 STRIP of the
268/// wall-clock prefix and tier-3 EXACT comparison of everything else). It differs
269/// from [`strip_log_entry`]'s `<ADDR>` erasure in one decisive way: erasure maps
270/// every address to a single token, so two runs that allocate in a DIFFERENT
271/// ORDER, or that ALIAS differently (one address printed twice vs. two distinct
272/// addresses), compare EQUAL -- the exact defect parity exists to catch. An
273/// ordinal assigned by first appearance keeps identity, order, and aliasing, so
274/// those cases still diverge while a pure ASLR-shift (same structure, different
275/// raw values) compares equal.
276///
277/// Only the explicit `<hostaddr ...>` marker is canonicalized. A bare `0x...`
278/// literal is left byte-for-byte for exact comparison: a blanket regex cannot
279/// tell a varying host pointer from a reproducible syscall argument printed
280/// `{:#x}` (`flock` `operation=0x2` vs `0x6`, `membarrier` bitmasks), so
281/// canonicalizing every hex token would erase real syscall-argument divergence.
282/// Decimal values (virtual-time timestamps, counts, sizes) are likewise exact.
283fn canonicalize_addresses_in_line(
284    line: &str,
285    map: &mut HashMap<String, usize>,
286    next: &mut usize,
287) -> String {
288    // Match ONLY the explicit host-address marker emitted by `host_addr`; the
289    // captured group is the raw `0x...` value used as the ordinal key.
290    static RE_HOSTADDR: LazyLock<Regex> =
291        LazyLock::new(|| Regex::new(r"<hostaddr (0[xX][A-Fa-f0-9]+)>").unwrap());
292
293    RE_HOSTADDR
294        .replace_all(line, |caps: &regex::Captures| {
295            let addr = &caps[1];
296            let ord = match map.get(addr) {
297                Some(existing) => *existing,
298                None => {
299                    let assigned = *next;
300                    *next += 1;
301                    map.insert(addr.to_string(), assigned);
302                    assigned
303                }
304            };
305            format!("<addr{ord}>")
306        })
307        .into_owned()
308}

310/// Separate a full, continuous log into discrete (possibly-multiline) log messages,
311/// stripping off the timestamps in the process.  Return lines tagged with their
312/// index number.
313fn extract_log_messages(contents: &str) -> Vec<(usize, &str)> {
314    let ts =
315        Regex::new(r"((Jan|Feb|Mar|Apr|May|Jun|Jul|Aug|Sep|Oct|Nov|Dec) \d\d \d\d:\d\d:\d\d\.\d+|\d+-\d\d-\d\dT\d\d:\d\d:\d\d.\d+Z) +")
316            .unwrap();
317    let tag = Regex::new("^(ERROR|WARN|INFO|DEBUG|TRACE) ").unwrap();
318    let iter = ts
319        .split(contents) // Not aware of a streaming version of this RE split operation.
320        .enumerate()
321        .map(|(i, s)| (i, s.trim()))
322        .filter(|(_, s)| !s.is_empty())
323        .map(|(i, s)| {
324            // Only let through lines that start with one of the expected tags:
325            if !tag.is_match(s) {
326                panic!("Log line without expected tag: {}", s);
327            } else {
328                (i, s)
329            }
330        });
331    iter.collect()
332}

346/// A scheduler COMMIT turn that only grants the `InternalIOPolling` resource, i.e. a
347/// granted retry of a nonblocking poll (poll/epoll_wait/wait4/futex/recv/send...). These
348/// grants are internal bookkeeping of Hermit's blocking-via-polling mechanism: how many
349/// times a thread is re-granted permission to re-attempt a nonblocking syscall before it
350/// stops returning would-block depends on when a concurrent external-IO action (e.g. a
351/// child linker process writing to a pipe) becomes ready on the host, which is wall-clock
352/// dependent and not tied to the (RCB-deterministic) logical schedule. The corresponding
353/// `NONCOMMIT ... polling resource` skips are already excluded from comparison (they are
354/// not tagged `COMMIT`); excluding the matching grant-COMMITs keeps the deterministic
355/// comparison consistent and focused on guest-observable events (the actual syscall
356/// results, still compared via their DETLOG entries).
357///
358/// SaBRe's inherited stdio pipes emit an outer device-resource turn before the inner polling
359/// turn. The scheduler tags that outer turn with `[sabre-internal-pipe-io]`; it is the same
360/// host-timing-sensitive operation and is normalized here as well. A SaBRe task with a loopback
361/// peer similarly tags the strong yield before each zero-timeout poll with
362/// `[sabre-loopback-poll-zero-timeout]`. The scheduler additionally suppresses the per-retry
363/// "advance global time for scheduler turn" DETLOG line for these turns at the source (see
364/// `Scheduler::bump_global_time`), so the two mechanisms together make the deterministic
365/// comparison insensitive to host-timing-dependent polling-loop counts.
366fn is_internal_io_poll_commit(line: &str) -> bool {
367    is_commit(line)
368        && (line.contains("{InternalIOPolling: ")
369            || line.contains(" [sabre-internal-pipe-io]")
370            || line.contains(" [sabre-loopback-poll-zero-timeout]"))
371}
372
373/// The scheduler's per-turn `committed_time` advance bookkeeping. `committed_time` tracks
374/// the global logical clock, which still moves forward when an `InternalIOPolling` retry
375/// (see `is_internal_io_poll_commit`) advances time -- and the number of those retries is
376/// host-timing nondeterministic. That makes the *presence* of this line on a given turn
377/// retry-count sensitive, so we exclude it from the deterministic comparison. No
378/// guest-observable signal is lost: the value is redundant with the (retained,
379/// retry-count-insensitive) "advance global time for scheduler turn" DETLOG line and with
380/// the per-turn committed time echoed on each COMMIT line.
381fn is_scheduler_committed_time(line: &str) -> bool {
382    line.contains("advancing committed_time from ")
383}

447fn write_syscall_context(
448    w: &mut impl std::io::Write,
449    left_index: usize,
450    right_index: usize,
451    left_syscalls: &[(usize, &str)],
452    right_syscalls: &[(usize, &str)],
453    history_count: u64,
454) -> std::io::Result<()> {
455    if history_count == 0 {
456        return Ok(());
457    }
458
459    let left_current = syscall_at_or_before(left_syscalls, left_index);
460    let right_current = syscall_at_or_before(right_syscalls, right_index);
461    if left_current.is_none() && right_current.is_none() {
462        return Ok(());
463    }
464
465    writeln!(w, "Divergent syscall context:")?;
466    for (label, current) in [("run 1", left_current), ("run 2", right_current)] {
467        if let Some((index, syscall)) = current {
468            writeln!(w, "  {label}, log message {index}: {syscall}")?;
469        } else {
470            writeln!(w, "  {label}: <no syscall observed>")?;
471        }
472    }
473
474    let history_limit = usize::try_from(history_count).unwrap_or(usize::MAX);
475    for (label, index, syscalls) in [
476        ("run 1", left_index, left_syscalls),
477        ("run 2", right_index, right_syscalls),
478    ] {
479        let history_boundary =
480            syscall_at_or_before(syscalls, index).map_or(index, |(current_index, _)| current_index);
481        let mut history = syscalls
482            .iter()
483            .rev()
484            .filter(|(syscall_index, entry)| {
485                *syscall_index < history_boundary && is_detlog_syscall_result(entry)
486            })
487            .take(history_limit)
488            .copied()
489            .collect::<Vec<_>>();
490        history.reverse();
491        if !history.is_empty() {
492            writeln!(w, "  Prior completed syscalls for {label}:")?;
493            for (_, syscall) in history {
494                writeln!(w, "    {syscall}")?;
495            }
496        }
497    }
498    writeln!(w)?;
499
500    Ok(())
501}

503/// A comparison of two strings.
504///
505/// Displays comparison result without any formatting
506pub struct Comparison<'a> {
507    left: &'a str,
508    right: &'a str,
509    no_color: bool,
510}
511impl<'a> Comparison<'a> {
512    /// Store two values to be compared in future.
513    ///
514    /// Expensive diffing is deferred until calling `Debug::fmt`.

516        Comparison {
517            left,
518            right,
519            no_color,
520        }
521    }

524    fn fmt(&self, f: &mut Formatter) -> Result {
525        if self.no_color {
526            writeln!(f, "Diff < left / right > :")?;
527            writeln!(f, "<\"{}\"", self.left)?;
528            writeln!(f, ">\"{}\"", self.right)
529        } else {
530            pretty_assertions::Comparison::new(&self.left, &self.right).fmt(f)
531        }
532    }
533}

535/// Returns `true` if a difference is found.
536///
537/// We could use an existing diff library on the entire log, but this provides us more
538/// control over how to present the (stripped/unstripped) differences, and to focus on the
539/// per-line divergence(s), and potentially focus on the first point of divergence.
540//
541// Future TODO:
542//  - report bulk differences, e.g. leftover lines, but with truncation
543//  - report only the first K per-line differences
544//  - detect reorderings and/or switch to larger differences for consecutive multi-line mismatches
545fn diff_vecs(
546    which: &str,
547    v1: &[(usize, &str)],
548    v2: &[(usize, &str)],
549    opts: &LogDiffOpts,
550    w: &mut impl std::io::Write,
551    left_syscalls: &[(usize, &str)],
552    right_syscalls: &[(usize, &str)],
553) -> std::io::Result<bool> {
554    writeln!(w, "  Comparing {which} messages...\n")?;
555    if v1.is_empty() && v2.is_empty() {
556        return Ok(false);
557    }
558
559    // Canonicalize addresses in a full pre-pass, per run, BEFORE the compare
560    // loop: the ordinal-by-first-appearance numbering must reflect the whole
561    // ordered message list, not be truncated when the loop breaks at `limit`.
562    // Each run gets its own fresh map/counter so the two numberings are
563    // independent (an ASLR shift compares equal) yet order- and alias-sensitive.
564    let (canon_left, canon_right): (Vec<String>, Vec<String>) = if opts.canonicalize_addresses {
565        let mut left_map = HashMap::new();
566        let mut left_next = 1usize;
567        let mut right_map = HashMap::new();
568        let mut right_next = 1usize;
569        (
570            v1.iter()
571                .map(|(_, s)| canonicalize_addresses_in_line(s, &mut left_map, &mut left_next))
572                .collect(),
573            v2.iter()
574                .map(|(_, s)| canonicalize_addresses_in_line(s, &mut right_map, &mut right_next))
575                .collect(),
576        )
577    } else {
578        (Vec::new(), Vec::new())
579    };
580
581    let mut diff_count = 0;
582    for (position, ((left_index, left), (right_index, right))) in
583        v1.iter().zip(v2.iter()).enumerate()
584    {
585        let (left_compared, right_compared) = if opts.strip_lines {
586            (strip_log_entry(left), strip_log_entry(right))
587        } else if opts.canonicalize_addresses {
588            (canon_left[position].clone(), canon_right[position].clone())
589        } else {
590            (left.to_string(), right.to_string())
591        };
592        if left_compared == right_compared {
593            continue;
594        }
595
596        if diff_count >= opts.limit && opts.limit != 0 {
597            writeln!(
598                w,
599                "More than {} differences, eliding the rest...",
600                opts.limit
601            )?;
602            break;
603        }
604
605        write!(
606            w,
607            "({which}) Mismatch at log messages {left_index} (run 1) and {right_index} (run 2): {}",
608            Comparison::new(opts.no_color, &left_compared, &right_compared)
609        )?;
610        if opts.strip_lines || opts.canonicalize_addresses {
611            write!(
612                w,
613                "({which}) Original entries before normalization: {}",
614                Comparison::new(opts.no_color, left, right)
615            )?;
616        }
617        write_syscall_context(
618            w,
619            *left_index,
620            *right_index,
621            left_syscalls,
622            right_syscalls,
623            opts.syscall_history,
624        )?;
625
626        diff_count += 1;
627    }
628
629    match v1.len().cmp(&v2.len()) {
630        Ordering::Less => {
631            writeln!(
632                w,
633                "Run 2 contains {} extra messages not matched in run 1. Displaying up to 10:",
634                v2.len() - v1.len()
635            )?;
636            diff_count += 1;
637            let start = v2.len() - std::cmp::min(10, v2.len() - v1.len());
638            for (_, message) in &v2[start..] {
639                writeln!(w, "{message}")?;
640            }
641        }
642        Ordering::Greater => {
643            writeln!(
644                w,
645                "Run 1 contains {} extra messages not matched in run 2. Displaying up to 10:",
646                v1.len() - v2.len()
647            )?;
648            diff_count += 1;
649            let start = v1.len() - std::cmp::min(10, v1.len() - v2.len());
650            for (_, message) in &v1[start..] {
651                writeln!(w, "{message}")?;
652            }
653        }
654        Ordering::Equal => {}
655    }
656
657    Ok(diff_count > 0)
658}

660fn git_diff(
661    which: &str,
662    v1: &[(usize, &str)],
663    v2: &[(usize, &str)],
664    opts: &LogDiffOpts,
665    w: &mut impl std::io::Write,
666    left_syscalls: &[(usize, &str)],
667    right_syscalls: &[(usize, &str)],
668) -> std::io::Result<bool> {
669    writeln!(w, "  Comparing {which} messages...\n")?;
670
671    let mut file1 = NamedTempFile::new()?;
672    let mut file2 = NamedTempFile::new()?;
673
674    for (_, line) in v1 {
675        writeln!(file1, "{line}")?;
676    }
677    for (_, line) in v2 {
678        writeln!(file2, "{line}")?;
679    }
680
681    match Command::new("git")
682        .args(["diff", "--color", "--color-words", "-w"])
683        .arg(file1.path())
684        .arg(file2.path())
685        .status()
686    {
687        Ok(code) => Ok(!code.success()),
688        Err(error) => {
689            eprintln!("Error launching git, falling back to basic diff: {error}");
690            diff_vecs(which, v1, v2, opts, w, left_syscalls, right_syscalls)
691        }
692    }

695/// What a log comparison actually compared, alongside whether it differed.
696///
697/// A bare "no difference found" boolean cannot distinguish *"the two message
698/// streams were compared and matched"* from *"there were no messages to
699/// compare"*. Both selected lists being empty is a NO-RESULT, not a match, so
700/// the counts travel with the verdict and a parity consumer can require nonzero
701/// execution before believing a green.
702#[derive(Debug, Clone, Copy, PartialEq, Eq)]
703pub struct LogDiffSummary {
704    /// True if a substantive difference was found between the two runs.
705    pub diff_found: bool,
706    /// Number of messages actually selected for comparison from the first run.
707    pub compared_left: usize,
708    /// Number of messages actually selected for comparison from the second run.
709    pub compared_right: usize,
710}
711
712impl LogDiffSummary {
713    /// True only when the comparison both ran on a nonempty selection *and*
714    /// found no difference. An empty-vs-empty comparison is never a match.
715    pub fn matched_with_evidence(&self) -> bool {
716        !self.diff_found && self.compared_left > 0 && self.compared_right > 0
717    }
718}

720/// Process log messages from two files.  Log messages look like this:
721///     "Apr 09 06:08:03.100  INFO detcore: [detcore, dtid 2]  finish syscall: close(2) = Ok(0)"
722///
723/// With some complexities:
724///  * Some entries are multi-line (contain newlines).
725///  * Some stripping of nondeterministic information is needed for direct comparability.
726///  * Certain lines are intended to be deterministic/comparable, in their contents,
727///    and others in their *presence* but not their details.
728///
729/// Reports only whether the two files differ. See [`log_diff_detailed`] when the
730/// caller must also know how many messages were actually compared; a bare
731/// `false` here cannot distinguish a match from an empty comparison.
732//
733// TODO: we should replace this with a diff algorithm that can handle insertions while maintaining
734// alignment. There's also no reason we can't output the stripped relevant lines and use a separate
735// diff tool.
736pub fn log_diff(file_a: &Path, file_b: &Path, opts: &LogDiffOpts) -> bool {
737    log_diff_detailed(file_a, file_b, opts).diff_found
738}
739
740/// Like [`log_diff`], but returns the counted comparison evidence rather than a
741/// bare boolean.
742pub fn log_diff_detailed(file_a: &Path, file_b: &Path, opts: &LogDiffOpts) -> LogDiffSummary {
743    // For now the log-diff mode reads both logs fully into memory. This could be
744    // modified in the future for a streaming solution, at least for scrolling through
745    // the identical prefixes of very large logs.
746    let vec_a = std::fs::read(file_a).expect("Could not open first input file.");
747    let vec_b = std::fs::read(file_b).expect("Could not open second input file.");
748    let str_a = String::from_utf8_lossy(&vec_a);
749    let str_b = String::from_utf8_lossy(&vec_b);
750    log_diff_summary_from_strs(str_a, str_b, opts, &mut std::io::stderr())
751        .expect("should write succesfully")
752}

754/// Boolean-only wrapper retained for tests that only ask "did it differ?".
755/// Prefer [`log_diff_summary_from_strs`] where the counts matter.
756#[cfg(test)]
757fn log_diff_from_strs(
758    file_a_str: impl AsRef<str>,
759    file_b_str: impl AsRef<str>,
760    opts: &LogDiffOpts,
761    w: &mut impl std::io::Write,
762) -> std::io::Result<bool> {
763    Ok(log_diff_summary_from_strs(file_a_str, file_b_str, opts, w)?.diff_found)
764}

766fn log_diff_summary_from_strs(
767    file_a_str: impl AsRef<str>,
768    file_b_str: impl AsRef<str>,
769    opts: &LogDiffOpts,
770    w: &mut impl std::io::Write,
771) -> std::io::Result<LogDiffSummary> {
772    let all_a = filter_ignored(
773        extract_log_messages(file_a_str.as_ref()),
774        &opts.ignore_lines,
775    );
776    let all_b = filter_ignored(
777        extract_log_messages(file_b_str.as_ref()),
778        &opts.ignore_lines,
779    );
780
781    writeln!(
782        w,
783        "Logs contain {} | {} messages total",
784        all_a.len(),
785        all_b.len(),
786    )?;
787
788    let detcore_a = filter_detcore(&all_a);
789    let detcore_b = filter_detcore(&all_b);
790    let infos_a = filter_infos(&all_a);
791    let infos_b = filter_infos(&all_b);
792    let detlogs_a = opts.filter_deterministic(&detcore_a);
793    let detlogs_b = opts.filter_deterministic(&detcore_b);
794    let left_syscalls = collect_syscalls(&all_a);
795    let right_syscalls = collect_syscalls(&all_b);
796    writeln!(
797        w,
798        "Logs contain {} | {} detcore-specific messages",
799        detcore_a.len(),
800        detcore_b.len(),
801    )?;
802    writeln!(
803        w,
804        "Logs contain {} | {} INFO messages",
805        infos_a.len(),
806        infos_b.len(),
807    )?;
808    writeln!(
809        w,
810        "Logs contain {} | {} DETLOG & scheduler COMMIT messages",
811        detlogs_a.len(),
812        detlogs_b.len(),
813    )?;
814
815    if opts.strip_lines {
816        writeln!(
817            w,
818            "Normalizing known nondeterministic numerical data before comparison..."
819        )?;
820    } else if opts.canonicalize_addresses {
821        writeln!(
822            w,
823            "Canonicalizing host addresses (ordinal by first appearance); comparing everything else exactly..."
824        )?;
825    }
826
827    let (which, compared_a, compared_b) = match opts.comparison {
828        LogComparisonMode::Deterministic => ("DETLOG", &detlogs_a, &detlogs_b),
829        LogComparisonMode::Info => ("INFO", &infos_a, &infos_b),
830        LogComparisonMode::FullTrace => ("full trace", &all_a, &all_b),
831    };
832
833    let diff_found = if opts.git_diff {
834        git_diff(
835            which,
836            compared_a,
837            compared_b,
838            opts,
839            w,
840            &left_syscalls,
841            &right_syscalls,
842        )?
843    } else {
844        diff_vecs(
845            which,
846            compared_a,
847            compared_b,
848            opts,
849            w,
850            &left_syscalls,
851            &right_syscalls,
852        )?
853    };
854
855    let summary = LogDiffSummary {
856        diff_found,
857        compared_left: compared_a.len(),
858        compared_right: compared_b.len(),
859    };
860
861    if diff_found {
862        writeln!(w, "Done processing logs, differences found.")?;
863    } else if summary.compared_left == 0 && summary.compared_right == 0 {
864        // Say this plainly rather than letting "no differences" stand in for
865        // "nothing was compared": a caller that treats the two alike turns a
866        // no-result into a green.
867        writeln!(
868            w,
869            "Done processing logs, but ZERO {which} messages were selected on either side: \
870             nothing was compared (no-result, not a match)."
871        )?;
872    } else {
873        writeln!(
874            w,
875            "Done processing logs, no substantive differences found ({} | {} {which} messages compared).",
876            summary.compared_left, summary.compared_right,
877        )?;
878    }
879    Ok(summary)
880}
