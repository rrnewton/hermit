27/// Selects the set of log messages compared for determinism.
28#[derive(Debug, Default, Clone, Copy, PartialEq, Eq)]
29pub enum LogComparisonMode {
30    /// Compare deterministic Detcore and scheduler messages.
31    #[default]
32    Deterministic,
33    /// Compare every INFO message exactly, while leaving any captured DEBUG or
34    /// TRACE messages available for diagnostics. This is the observation
35    /// envelope used by the `BitwiseInfoV1` verification policy.
36    Info,
37    /// Compare every captured log message without filtering.
38    FullTrace,
39}

41/// Options for calling `log_diff`.
42#[derive(Debug, Parser, Clone)]
43pub struct LogDiffOpts {
44    /// UNSAFE: strips numbers and temporary paths before comparison.
45    ///
46    /// This erases timestamps and syscall values that bitwise parity exists to
47    /// compare. Never use this option to make a failing parity diff pass; doing
48    /// so is cheating. It is only for non-parity diagnostic localization.
49    #[clap(long = "unsafe-strip-lines")]
50    pub strip_lines: bool,
51
52    /// Canonicalize host memory addresses before comparison WITHOUT erasing them.
53    ///
54    /// Only addresses a producer has explicitly marked with the
55    /// `<hostaddr 0x...>` wrapper (see [`host_addr`]) are canonicalized; each
56    /// distinct marked address is rewritten to an ordinal placeholder
57    /// `<addr{N}>` assigned by order of first appearance within a single run
58    /// (see `canonicalize_addresses_in_line`). Unlike [`Self::strip_lines`],
59    /// this discards ONLY the host-specific raw pointer value: it preserves
60    /// identity (same address -> same ordinal), ordering (introduction
61    /// sequence), and aliasing (two names for one address collapse to one
62    /// ordinal), and it leaves every other byte -- virtual-time timestamps,
63    /// syscall argument/result values, counts, sizes, flags -- untouched for an
64    /// exact comparison. In canonical parity, this preserves the ability to DETECT a
65    /// difference (allocation-order or aliasing changes still diverge), which
66    /// wholesale stripping throws away.
67    ///
68    /// The marker is REQUIRED (a bare `0x...` literal is left exact) because
69    /// nothing in the compared DETLOG stream can otherwise distinguish a varying
70    /// host pointer from a reproducible hex value -- syscall arguments printed
71    /// `{:#x}` (e.g. `flock` `operation=0x2` vs `0x6`), guest memory ranges,
72    /// content digests, cpuid leaves. A blanket `0x` canonicalization would
73    /// collapse those too, silently erasing real syscall-argument divergence:
74    /// a "softer strip" and exactly the fake-green this policy exists to prevent.
75    #[clap(skip)]
76    pub canonicalize_addresses: bool,
77
78    /// The internal message set to compare.
79    #[clap(skip)]
80    pub comparison: LogComparisonMode,
81    /// Limit the number of differences printed. Set to 0 for no limit.
82    #[clap(long, default_value = "20")]
83    pub limit: u64,
84
85    /// Before comparison, filter out lines which contain this substring.
86    #[clap(long)]
87    pub ignore_lines: Vec<String>,
88
89    /// Show this many completed syscalls before each run-specific divergence point.
90    /// Set to 0 to omit history.
91    #[clap(long, default_value = "0")]
92    pub syscall_history: u64,
93    /// Disable colored console output for line diffs.
94    #[clap(long)]
95    pub no_color: bool,
96
97    /// Do not consider "COMMIT" messages for deterministic checks.
98    #[clap(long)]
99    pub skip_commit: bool,
100
101    /// Do not consider "DETLOG" messages for deterministic checks.
102    #[clap(long)]
103    pub skip_detlog: bool,
104
105    /// Use git diff instead of the internal, basic log comparison.
106    #[clap(long)]
107    pub git_diff: bool,
108
109    /// In case --skip-detlog=false this parameter further filters which
110    /// "DETLOG" messages will be included for deterministic checks
111    #[clap(long, default_values = &["syscall", "syscallresult", "other"])]
112    pub include_detlogs: Vec<DetLogFilter>,
113}

115impl LogDiffOpts {
116    fn is_skip(&self, filter: DetLogFilter) -> bool {
117        !self.include_detlogs.contains(&filter)
118    }
119
120    fn skip_detlog(&self, entry: &str) -> bool {
121        if self.skip_detlog {
122            return true;
123        }
124
125        if is_detlog_syscall(entry) && self.is_skip(DetLogFilter::Syscall) {
126            return true;
127        }
128        if is_detlog_syscall_result(entry) && self.is_skip(DetLogFilter::SyscallResult) {
129            return true;
130        }
131
132        if !is_detlog_syscall(entry)
133            && !is_detlog_syscall_result(entry)
134            && self.is_skip(DetLogFilter::Other)
135        {
136            return true;
137        }
138
139        false
140    }
141
142    fn filter_deterministic<'a>(&self, v: &[(usize, &'a str)]) -> Vec<(usize, &'a str)> {
143        v.iter()
144            .filter_map(|(i, s)| {
145                if (is_detlog(s) && !self.skip_detlog(s) && !is_scheduler_committed_time(s))
146                    || (is_commit(s) && !self.skip_commit && !is_internal_io_poll_commit(s))
147                {
148                    Some((*i, *s))
149                } else {
150                    None
151                }
152            })
153            .collect()
154    }
155}

193/// In fully-deterministic modes, many log lines should be fully determinstic across runs.
194/// But as that is a work-in-progress, this utility strips known-nondeterministic
195/// information from logs.
196///
197/// This erasure is deliberately lossy and is NOT a parity claim: it backs the
198/// `Stripped` comparator only (`bitwise_parity: false`). `BitwiseInfoV1`
199/// canonicalizes rather than erases -- see `canonicalize_addresses_in_line`.
200/// Lossy as it is, each pattern must still erase only what it names: erasing a
201/// neighbouring field turns a real divergence into a reported match.
202///
203/// Example input/output:
204///   `Input:  COMMIT turn 3, dettid 231635 using resources Resources { tid: DetPid { inner: 231635 }, resources: {Path("/proc/231635/fd/1"): W} }`
205///   `Output: COMMIT turn <NUM>, dettid <NUM> using resources Resources { tid: DetPid { inner: <NUM> }, resources: {Path("/proc/<pid>/fd/<num>"): W} }`
206///
207/// As you can see this is overkill and smarter strategies would be possible. For example,
208/// ones that remember and post-facto-determinize certain identifiers.
209pub fn strip_log_entry(log: &str) -> String {
210    // Memory addresses, like 0x7fcfb7e7d450
211    //
212    // TODO: use a debug allocator that increases only, never reusing. Also, consider
213    // post-facto processing all of these into new virtual addresses based on the order they're seen.
214    static RE0: LazyLock<Regex> = LazyLock::new(|| Regex::new(r"\b0[xX][A-Fa-f0-9]+\b").unwrap());
215
216    // Every number, plus common duration suffixes so fractional timing jitter is
217    // not left behind. This one is terrible overkill: for `hermit run` itself and
218    // for all command tests the full contents of a COMMIT line should already be
219    // deterministic, so nothing here should need erasing. It is retained for
220    // `spawn_fn_*` variants, which fork from another process and so exercise only
221    // a *partial* detcore setup without a true process tree of their own.
222    //
223    // N.B. RE4 must run BEFORE this pattern, or `800.709_180s` is consumed here as
224    // a bare number and never reaches the duration rule.
225    static RE1: LazyLock<Regex> =
226        LazyLock::new(|| Regex::new(r"\b[\d][\d_]*(?:\.[\d][\d_]*)?(?:ns|us|µs|ms)?\b").unwrap());
227
228    // A quoted /tmp path. `[^"]*` stops at the path's OWN closing quote: a greedy
229    // `.*` here ran to the last quote on the line and erased every field after the
230    // path, so two entries differing only downstream of a /tmp path compared equal.
231    //
232    // TODO: only strip this information if the config specified to the host /tmp through.
233    // Otherwise we can determinize /tmp access fully.
234    static RE2: LazyLock<Regex> = LazyLock::new(|| Regex::new(r#"/tmp/[^"]*""#).unwrap());
235
236    // TODO: only strip this one if we're allowing through the host /proc or failing to determinize tids/pids:
237    static RE3: LazyLock<Regex> = LazyLock::new(|| Regex::new(r"/proc/[\d]+/").unwrap());
238
239    // TODO: only strip this if we're running a library-based test where we can't
240    // guarantee the starting state of the allocator/etc.
241    static RE4: LazyLock<Regex> = LazyLock::new(|| Regex::new(r"\b[\d][\d_.]*s\b").unwrap());
242
243    let log = RE4.replace_all(log, "<NANOSECONDS>");
244    let log = RE3.replace_all(&log, "/proc/<PID>/");
245    let log = RE0.replace_all(&log, "<ADDR>");
246    let log = RE1.replace_all(&log, "<NUM>");
247    let log = RE2.replace_all(&log, "/tmp/<somewhere>\"");
248    String::from(log)
249}

251/// Wrap a host memory address so `canonicalize_addresses_in_line` will
252/// canonicalize it. Producers that print a genuinely host-specific pointer
253/// (one that varies run-to-run, e.g. a supervisor-side allocation) should emit
254/// it via this helper -- `<hostaddr 0x7fcfb7e7d450>` -- instead of a bare
255/// `0x...` literal. Only marked addresses are canonicalized, so reproducible hex
256/// (syscall arguments, guest memory ranges, digests) is compared exactly.
257///
258/// Note: nothing in detcore's current DETLOG output prints a varying host
259/// pointer -- guest addresses are determinized and every logged `0x...` value is
260/// reproducible -- so this marker is presently unused in production and exists
261/// so a future host-pointer print opts into canonicalization deliberately rather
262/// than being swept up by a blanket regex.
263pub fn host_addr(addr: usize) -> String {
264    format!("<hostaddr {addr:#x}>")
265}
266
267/// Rewrite each MARKED host memory address (`<hostaddr 0x...>`, see
268/// [`host_addr`]) in `line` to an ordinal placeholder `<addr{N}>`, numbered by
269/// order of first appearance within a single run. `map`/`next` carry the per-run
270/// assignment state threaded across all of that run's lines, so `next` should
271/// start at 1 and the same `map`/`next` must be reused for every line of one run
272/// (and a FRESH pair used for the other run).
273///
274/// Canonical parity strips the wall-clock prefix, canonicalizes these marked
275/// addresses, and compares everything else exactly. This step differs
276/// from [`strip_log_entry`]'s `<ADDR>` erasure in one decisive way: erasure maps
277/// every address to a single token, so two runs that allocate in a DIFFERENT
278/// ORDER, or that ALIAS differently (one address printed twice vs. two distinct
279/// addresses), compare EQUAL -- the exact defect parity exists to catch. An
280/// ordinal assigned by first appearance keeps identity, order, and aliasing, so
281/// those cases still diverge while a pure ASLR-shift (same structure, different
282/// raw values) compares equal.
283///
284/// Only the explicit `<hostaddr ...>` marker is canonicalized. A bare `0x...`
285/// literal is left byte-for-byte for exact comparison: a blanket regex cannot
286/// tell a varying host pointer from a reproducible syscall argument printed
287/// `{:#x}` (`flock` `operation=0x2` vs `0x6`, `membarrier` bitmasks), so
288/// canonicalizing every hex token would erase real syscall-argument divergence.
289/// Decimal values (virtual-time timestamps, counts, sizes) are likewise exact.
290fn canonicalize_addresses_in_line(
291    line: &str,
292    map: &mut HashMap<String, usize>,
293    next: &mut usize,
294) -> String {
295    // Match ONLY the explicit host-address marker emitted by `host_addr`; the
296    // captured group is the raw `0x...` value used as the ordinal key.
297    static RE_HOSTADDR: LazyLock<Regex> =
298        LazyLock::new(|| Regex::new(r"<hostaddr (0[xX][A-Fa-f0-9]+)>").unwrap());
299
300    RE_HOSTADDR
301        .replace_all(line, |caps: &regex::Captures| {
302            let addr = &caps[1];
303            let ord = match map.get(addr) {
304                Some(existing) => *existing,
305                None => {
306                    let assigned = *next;
307                    *next += 1;
308                    map.insert(addr.to_string(), assigned);
309                    assigned
310                }
311            };
312            format!("<addr{ord}>")
313        })
314        .into_owned()
315}

317fn messages_for_comparison(messages: &[(usize, &str)], opts: &LogDiffOpts) -> Vec<String> {
318    if opts.strip_lines {
319        messages
320            .iter()
321            .map(|(_, message)| strip_log_entry(message))
322            .collect()
323    } else if opts.canonicalize_addresses {
324        let mut addresses = HashMap::new();
325        let mut next_address = 1usize;
326        messages
327            .iter()
328            .map(|(_, message)| {
329                canonicalize_addresses_in_line(message, &mut addresses, &mut next_address)
330            })
331            .collect()
332    } else {
333        messages
334            .iter()
335            .map(|(_, message)| (*message).to_owned())
336            .collect()
337    }
338}

340fn canonical_info_from_str(contents: &str) -> Vec<String> {
341    let info = filter_infos(&extract_log_messages(contents));
342    let opts = LogDiffOpts {
343        canonicalize_addresses: true,
344        comparison: LogComparisonMode::Info,
345        ..Default::default()
346    };
347    messages_for_comparison(&info, &opts)
348}

350/// Print the canonical INFO messages that strict verification would compare for
351/// one captured log.
352///
353/// This removes the real wall-clock prefix and rewrites only explicitly marked
354/// host addresses to first-appearance ordinals. It does not run the lossy
355/// `--unsafe-strip-lines` transformation: scheduler turns, virtual time, syscall
356/// values, counts, flags, and every other substantive byte are preserved.
357pub fn write_canonical_info(file: &Path, writer: &mut impl Write) -> std::io::Result<usize> {
358    let bytes = std::fs::read(file)?;
359    let contents = String::from_utf8_lossy(&bytes);
360    let messages = canonical_info_from_str(&contents);
361    for message in &messages {
362        writeln!(writer, "{message}")?;
363    }
364    Ok(messages.len())
365}

367/// Separate a full, continuous log into discrete (possibly-multiline) log messages,
368/// stripping off the timestamps in the process.  Return lines tagged with their
369/// index number.
370/// The timestamp that begins every log record. A record runs from one match to
371/// the next, so records may span multiple lines -- which is why comparison is
372/// done on records and never on lines or bytes.
373static RECORD_START: LazyLock<Regex> = LazyLock::new(|| {
374    Regex::new(r"((Jan|Feb|Mar|Apr|May|Jun|Jul|Aug|Sep|Oct|Nov|Dec) \d\d \d\d:\d\d:\d\d\.\d+|\d+-\d\d-\d\dT\d\d:\d\d:\d\d.\d+Z) +")
375        .unwrap()
376});
377
378/// Byte offsets at which each record starts.
379fn record_starts(contents: &str) -> Vec<usize> {
380    RECORD_START
381        .find_iter(contents)
382        .map(|m| m.start())
383        .collect()
384}
385
386/// How many records in `contents` are known COMPLETE.
387///
388/// A record is complete only once the *next* record has begun, because nothing
389/// else marks its end. In a log still being written the final record may be
390/// half-flushed, so it is never counted and never compared: a partial write must
391/// not read as a difference. A buffer with one record start has zero complete
392/// records, and that is a real answer, not a failure.
393pub fn complete_record_count(contents: &str) -> usize {
394    record_starts(contents).len().saturating_sub(1)
395}
396
397/// The prefix of `contents` holding exactly its first `n` complete records.
398///
399/// Returns `None` when fewer than `n` complete records are present, so a caller
400/// can tell "the runs agree over n records" apart from "n records have not been
401/// written yet". Those two must never collapse into one answer.
402pub fn take_complete_records(contents: &str, n: usize) -> Option<&str> {
403    let starts = record_starts(contents);
404    if n == 0 {
405        return Some(&contents[..0]);
406    }
407    // Record n-1 ends where record n begins, so n complete records require n+1 starts.
408    starts.get(n).map(|end| &contents[..*end])
409}

411fn extract_log_messages(contents: &str) -> Vec<(usize, &str)> {
412    let ts = &*RECORD_START;
413    let tag = Regex::new("^(ERROR|WARN|INFO|DEBUG|TRACE) ").unwrap();
414    let iter = ts
415        .split(contents) // Not aware of a streaming version of this RE split operation.
416        .enumerate()
417        .map(|(i, s)| (i, s.trim()))
418        .filter(|(_, s)| !s.is_empty())
419        .map(|(i, s)| {
420            // Only let through lines that start with one of the expected tags:
421            if !tag.is_match(s) {
422                panic!("Log line without expected tag: {}", s);
423            } else {
424                (i, s)
425            }
426        });
427    iter.collect()
428}

442/// A scheduler COMMIT turn that only grants the `InternalIOPolling` resource, i.e. a
443/// granted retry of a nonblocking poll (poll/epoll_wait/wait4/futex/recv/send...). These
444/// grants are internal bookkeeping of Hermit's blocking-via-polling mechanism: how many
445/// times a thread is re-granted permission to re-attempt a nonblocking syscall before it
446/// stops returning would-block depends on when a concurrent external-IO action (e.g. a
447/// child linker process writing to a pipe) becomes ready on the host, which is wall-clock
448/// dependent and not tied to the (RCB-deterministic) logical schedule. The corresponding
449/// `NONCOMMIT ... polling resource` skips are already excluded from comparison (they are
450/// not tagged `COMMIT`); excluding the matching grant-COMMITs keeps the deterministic
451/// comparison consistent and focused on guest-observable events (the actual syscall
452/// results, still compared via their DETLOG entries).
453///
454/// SaBRe's inherited stdio pipes emit an outer device-resource turn before the inner polling
455/// turn. The scheduler tags that outer turn with `[sabre-internal-pipe-io]`; it is the same
456/// host-timing-sensitive operation and is normalized here as well. A SaBRe task with a loopback
457/// peer similarly tags the strong yield before each zero-timeout poll with
458/// `[sabre-loopback-poll-zero-timeout]`. The scheduler additionally suppresses the per-retry
459/// "advance global time for scheduler turn" DETLOG line for these turns at the source (see
460/// `Scheduler::bump_global_time`), so the two mechanisms together make the deterministic
461/// comparison insensitive to host-timing-dependent polling-loop counts.
462fn is_internal_io_poll_commit(line: &str) -> bool {
463    is_commit(line)
464        && (line.contains("{InternalIOPolling: ")
465            || line.contains(" [sabre-internal-pipe-io]")
466            || line.contains(" [sabre-loopback-poll-zero-timeout]"))
467}

469/// The scheduler's per-turn `committed_time` advance bookkeeping. `committed_time` tracks
470/// the global logical clock, which still moves forward when an `InternalIOPolling` retry
471/// (see `is_internal_io_poll_commit`) advances time -- and the number of those retries is
472/// host-timing nondeterministic. That makes the *presence* of this line on a given turn
473/// retry-count sensitive, so we exclude it from the deterministic comparison. No
474/// guest-observable signal is lost: the value is redundant with the (retained,
475/// retry-count-insensitive) "advance global time for scheduler turn" DETLOG line and with
476/// the per-turn committed time echoed on each COMMIT line.
477fn is_scheduler_committed_time(line: &str) -> bool {
478    line.contains("advancing committed_time from ")
479}

510/// Text of the INFO message the scheduler logs when it finds the run queue
511/// empty but is not finished yet. `Scheduler::step2_process_blocked` in
512/// [`crate::scheduler`] emits it and returns `SkipTurn`, so the loop goes
513/// around again and still exits later through the ordinary
514/// "run queue empty, exiting sched_loop." message.
515///
516/// Two consequences make this the message worth recording. First, both paths
517/// end with the same final scheduler message, so the *last* scheduler line does
518/// not say which path a run took; only whether this kick appeared does. Second,
519/// whether it appears is decided by host timing — the ptrace supervisor may not
520/// yet have reaped a physical process exit when the check runs — so a pair of
521/// runs of the same guest can differ here while agreeing on every committed
522/// scheduling decision.
523///
524/// Kept as a constant so the string sits beside the code that reads it; grep
525/// for this text to find the producing `info!`.
526const SCHEDULER_EMPTY_QUEUE_KICK: &str = "zero threads left anywhere, fizzling.";

528/// How many of `v` record the scheduler's empty-run-queue kick.
529fn count_empty_queue_kicks(v: &[(usize, &str)]) -> usize {
530    v.iter()
531        .filter(|(_, s)| s.contains(SCHEDULER_EMPTY_QUEUE_KICK))

533}

535/// Resource text of the COMMIT record produced when a guest's runtime reads the
536/// process's own memory map during bootstrap.
537///
538/// This record carries the second known shape of backend self-nondeterminism,
539/// and it is unlike the empty-run-queue kick in a way that matters. Two runs of
540/// one guest can agree on every scheduling decision — same turn, same thread,
541/// same resource — and still commit that turn at *different virtual times*,
542/// because virtual time advances with retired conditional branches and the
543/// number of branches the scan executes depends on the map it reads. So the
544/// evidence is not the presence of a message but the value inside one.
545///
546/// A diverging pair prints both times in its diff. A matching pair kept
547/// nothing, which left the prior question — does this guest perform the read at
548/// all, and therefore can it exhibit the drift — answerable only by collecting
549/// divergences over many runs.
550const RUNTIME_MAPS_READ_RESOURCE: &str = r#"Path("/proc/self/maps")"#;

552/// One run's first memory-map read COMMIT, rendered for the retained line.
553/// `None` is spelled out rather than omitted, so a run that never performed the
554/// read is distinguishable from a run whose value was simply not reported.
555fn describe_maps_commit(first: Option<(u64, Option<u64>)>) -> String {
556    match first {
557        Some((turn, Some(nanoseconds))) => {
558            format!("first at turn {turn}, committed virtual time {nanoseconds}ns")
559        }
560        Some((turn, None)) => format!("first at turn {turn}, committed virtual time unrecorded"),
561        None => "no such record".to_string(),
562    }
563}

565/// COMMIT records in `v` that read the process's own memory map: how many there
566/// are, and the turn and committed virtual time of the first.
567fn maps_read_commits(v: &[(usize, &str)]) -> (usize, Option<(u64, Option<u64>)>) {
568    let mut count = 0;
569    let mut first = None;
570    for (_, message) in v {
571        if !message.contains(RUNTIME_MAPS_READ_RESOURCE) {
572            continue;
573        }
574        let Some(position) = commit_position(message) else {
575            continue;
576        };
577        count += 1;
578        if first.is_none() {
579            first = Some(position);
580        }
581    }
582    (count, first)
583}

607fn first_different_message_indices(
608    left: &[(usize, &str)],
609    right: &[(usize, &str)],
610    opts: &LogDiffOpts,
611) -> Option<(Option<usize>, Option<usize>)> {
612    let compared_left = messages_for_comparison(left, opts);
613    let compared_right = messages_for_comparison(right, opts);
614    let common = compared_left.len().min(compared_right.len());
615
616    if let Some(position) =
617        (0..common).find(|&position| compared_left[position] != compared_right[position])
618    {
619        return Some((Some(left[position].0), Some(right[position].0)));
620    }
621
622    match compared_left.len().cmp(&compared_right.len()) {
623        Ordering::Less => Some((None, Some(right[common].0))),
624        Ordering::Greater => Some((Some(left[common].0), None)),
625        Ordering::Equal => None,
626    }

629fn parse_underscored_u64(value: &str) -> Option<u64> {
630    value.replace('_', "").parse().ok()
631}
632
633fn parse_virtual_nanoseconds(value: &str, unit: Option<&str>) -> Option<u64> {
634    match unit {
635        None | Some("ns") if !value.contains('.') => parse_underscored_u64(value),
636        Some("s") => {
637            let value = value.replace('_', "");
638            let (seconds, fraction) = value.split_once('.').unwrap_or((&value, ""));
639            if fraction.len() > 9 || !fraction.bytes().all(|byte| byte.is_ascii_digit()) {
640                return None;
641            }
642            let seconds = seconds.parse::<u64>().ok()?;
643            let fraction = if fraction.is_empty() {
644                0
645            } else {
646                let digits = fraction.parse::<u64>().ok()?;
647                digits.checked_mul(10_u64.pow((9 - fraction.len()) as u32))?
648            };
649            seconds.checked_mul(1_000_000_000)?.checked_add(fraction)
650        }
651        _ => None,
652    }
653}

655fn commit_position(message: &str) -> Option<(u64, Option<u64>)> {
656    static TURN: LazyLock<Regex> =
657        LazyLock::new(|| Regex::new(r"\bCOMMIT turn ([0-9][0-9_]*)\b").unwrap());
658    static TIME: LazyLock<Regex> = LazyLock::new(|| {
659        Regex::new(r"\b(?:at time|on previously committed) ([0-9][0-9_]*(?:\.[0-9_]+)?)(ns|s)?\b")
660            .unwrap()
661    });
662
663    let turn = parse_underscored_u64(TURN.captures(message)?.get(1)?.as_str())?;
664    let virtual_nanoseconds = TIME.captures(message).and_then(|captures| {
665        parse_virtual_nanoseconds(
666            captures.get(1)?.as_str(),
667            captures.get(2).map(|unit| unit.as_str()),
668        )
669    });
670    Some((turn, virtual_nanoseconds))
671}

695fn write_syscall_context(
696    w: &mut impl std::io::Write,
697    left_index: usize,
698    right_index: usize,
699    left_syscalls: &[(usize, &str)],
700    right_syscalls: &[(usize, &str)],
701    history_count: u64,
702) -> std::io::Result<()> {
703    if history_count == 0 {
704        return Ok(());
705    }
706
707    let left_current = syscall_at_or_before(left_syscalls, left_index);
708    let right_current = syscall_at_or_before(right_syscalls, right_index);
709    if left_current.is_none() && right_current.is_none() {
710        return Ok(());
711    }
712
713    writeln!(w, "Divergent syscall context:")?;
714    for (label, current) in [("run 1", left_current), ("run 2", right_current)] {
715        if let Some((index, syscall)) = current {
716            writeln!(w, "  {label}, log message {index}: {syscall}")?;
717        } else {
718            writeln!(w, "  {label}: <no syscall observed>")?;
719        }
720    }
721
722    let history_limit = usize::try_from(history_count).unwrap_or(usize::MAX);
723    for (label, index, syscalls) in [
724        ("run 1", left_index, left_syscalls),
725        ("run 2", right_index, right_syscalls),
726    ] {
727        let history_boundary =
728            syscall_at_or_before(syscalls, index).map_or(index, |(current_index, _)| current_index);
729        let mut history = syscalls
730            .iter()
731            .rev()
732            .filter(|(syscall_index, entry)| {
733                *syscall_index < history_boundary && is_detlog_syscall_result(entry)
734            })
735            .take(history_limit)
736            .copied()
737            .collect::<Vec<_>>();
738        history.reverse();
739        if !history.is_empty() {
740            writeln!(w, "  Prior completed syscalls for {label}:")?;
741            for (_, syscall) in history {
742                writeln!(w, "    {syscall}")?;
743            }
744        }
745    }
746    writeln!(w)?;
747
748    Ok(())
749}

783/// Returns `true` if a difference is found.
784///
785/// We could use an existing diff library on the entire log, but this provides us more
786/// control over how to present the (stripped/unstripped) differences, and to focus on the
787/// per-line divergence(s), and potentially focus on the first point of divergence.
788//
789// Future TODO:
790//  - report bulk differences, e.g. leftover lines, but with truncation
791//  - report only the first K per-line differences
792//  - detect reorderings and/or switch to larger differences for consecutive multi-line mismatches
793fn diff_vecs(
794    which: &str,
795    v1: &[(usize, &str)],
796    v2: &[(usize, &str)],
797    opts: &LogDiffOpts,
798    w: &mut impl std::io::Write,
799    left_syscalls: &[(usize, &str)],
800    right_syscalls: &[(usize, &str)],
801) -> std::io::Result<bool> {
802    writeln!(w, "  Comparing {which} messages...\n")?;
803    if v1.is_empty() && v2.is_empty() {
804        return Ok(false);
805    }
806
807    // Prepare both complete streams before the compare loop: address ordinals
808    // are assigned by first appearance across the full selected stream, not by
809    // however many differences the caller asks us to print.
810    let compared_left = messages_for_comparison(v1, opts);
811    let compared_right = messages_for_comparison(v2, opts);
812
813    let mut diff_count = 0;
814    for (position, ((left_index, left), (right_index, right))) in
815        v1.iter().zip(v2.iter()).enumerate()
816    {
817        let left_compared = &compared_left[position];
818        let right_compared = &compared_right[position];
819        if left_compared == right_compared {
820            continue;
821        }
822
823        if diff_count >= opts.limit && opts.limit != 0 {
824            writeln!(
825                w,
826                "More than {} differences, eliding the rest...",
827                opts.limit
828            )?;
829            break;
830        }
831
832        write!(
833            w,
834            "({which}) Mismatch at log messages {left_index} (run 1) and {right_index} (run 2): {}",
835            Comparison::new(opts.no_color, left_compared, right_compared)
836        )?;
837        if opts.strip_lines || opts.canonicalize_addresses {
838            write!(
839                w,
840                "({which}) Original entries before normalization: {}",
841                Comparison::new(opts.no_color, left, right)
842            )?;
843        }
844        write_syscall_context(
845            w,
846            *left_index,
847            *right_index,
848            left_syscalls,
849            right_syscalls,
850            opts.syscall_history,
851        )?;
852
853        diff_count += 1;
854    }
855
856    match v1.len().cmp(&v2.len()) {
857        Ordering::Less => {
858            writeln!(
859                w,
860                "Run 2 contains {} extra messages not matched in run 1. Displaying up to 10:",
861                v2.len() - v1.len()
862            )?;
863            diff_count += 1;
864            let start = v2.len() - std::cmp::min(10, v2.len() - v1.len());
865            for (_, message) in &v2[start..] {
866                writeln!(w, "{message}")?;
867            }
868        }
869        Ordering::Greater => {
870            writeln!(
871                w,
872                "Run 1 contains {} extra messages not matched in run 2. Displaying up to 10:",
873                v1.len() - v2.len()
874            )?;
875            diff_count += 1;
876            let start = v1.len() - std::cmp::min(10, v1.len() - v2.len());
877            for (_, message) in &v1[start..] {
878                writeln!(w, "{message}")?;
879            }
880        }
881        Ordering::Equal => {}
882    }
883
884    Ok(diff_count > 0)
885}

887fn git_diff(
888    which: &str,
889    v1: &[(usize, &str)],
890    v2: &[(usize, &str)],
891    opts: &LogDiffOpts,
892    w: &mut impl std::io::Write,
893    left_syscalls: &[(usize, &str)],
894    right_syscalls: &[(usize, &str)],
895) -> std::io::Result<bool> {
896    writeln!(w, "  Comparing {which} messages...\n")?;
897
898    let mut file1 = NamedTempFile::new()?;
899    let mut file2 = NamedTempFile::new()?;
900
901    for (_, line) in v1 {
902        writeln!(file1, "{line}")?;
903    }
904    for (_, line) in v2 {
905        writeln!(file2, "{line}")?;
906    }
907
908    match Command::new("git")
909        .args(["diff", "--color", "--color-words", "-w"])
910        .arg(file1.path())
911        .arg(file2.path())
912        .status()
913    {
914        Ok(code) => Ok(!code.success()),
915        Err(error) => {
916            eprintln!("Error launching git, falling back to basic diff: {error}");
917            diff_vecs(which, v1, v2, opts, w, left_syscalls, right_syscalls)
918        }
919    }
920}

922/// What a log comparison actually compared, alongside whether it differed.
923///
924/// A bare "no difference found" boolean cannot distinguish *"the two message
925/// streams were compared and matched"* from *"there were no messages to
926/// compare"*. Both selected lists being empty is a NO-RESULT, not a match, so
927/// the counts travel with the verdict and a parity consumer can require nonzero
928/// execution before believing a green.
929#[derive(Debug, Clone, Copy, PartialEq, Eq)]
930pub struct LogDiffSummary {
931    /// True if a substantive difference was found between the two runs.
932    pub diff_found: bool,
933    /// Number of messages actually selected for comparison from the first run.
934    pub compared_left: usize,
935    /// Number of messages actually selected for comparison from the second run.
936    pub compared_right: usize,
937    /// Scheduler turn at the first different selected message, when the first
938    /// run has a preceding scheduler COMMIT message that identifies it.
939    pub first_divergent_scheduler_turn: Option<u64>,
940    /// Virtual nanoseconds at that same scheduler COMMIT, when its time is
941    /// present and parseable.
942    pub first_divergent_virtual_nanoseconds: Option<u64>,
943    /// 1-based index of the first record that differs. "Diverged somewhere in
944    /// the first N records" is a bound, not a location, and on a long run the
945    /// two are far apart; this is the location.
946    pub first_divergent_record: Option<usize>,
947}
948
949impl LogDiffSummary {
950    /// True only when the comparison both ran on a nonempty selection *and*
951    /// found no difference. An empty-vs-empty comparison is never a match.
952    pub fn matched_with_evidence(&self) -> bool {
953        !self.diff_found && self.compared_left > 0 && self.compared_right > 0
954    }
955}

957/// Process log messages from two files.  Log messages look like this:
958///     "Apr 09 06:08:03.100  INFO detcore: [detcore, dtid 2]  finish syscall: close(2) = Ok(0)"

964///    and others in their *presence* but not their details.
965///
966/// Reports only whether the two files differ. See [`log_diff_detailed`] when the
967/// caller must also know how many messages were actually compared; a bare
968/// `false` here cannot distinguish a match from an empty comparison.
969//
970// TODO: we should replace this with a diff algorithm that can handle insertions while maintaining
971// alignment. There's also no reason we can't output the stripped relevant lines and use a separate
972// diff tool.
973pub fn log_diff(file_a: &Path, file_b: &Path, opts: &LogDiffOpts) -> bool {
974    log_diff_detailed(file_a, file_b, opts).diff_found
975}

978/// bare boolean.

1024/// A comparison of two logs that may still be growing.
1025///
1026/// The verdict alone is not reportable. "The runs agree" and "the runs agree so
1027/// far as anyone has looked" are different claims, and a caller who cannot tell
1028/// them apart will stop early and conclude the wrong thing. The record counts
1029/// are therefore part of the result, not optional detail alongside it.
1030#[derive(Debug, Clone, Copy, PartialEq, Eq)]
1031pub struct PrefixComparison {
1032    /// The verdict over the compared prefix only.
1033    pub summary: LogDiffSummary,
1034    /// Complete records present in the first log when it was read.
1035    pub records_available_left: usize,
1036    /// Complete records present in the second log when it was read.
1037    pub records_available_right: usize,
1038    /// Records actually compared: the shorter of the two available counts.
1039    pub records_compared: usize,
1040}
1041
1042impl PrefixComparison {
1043    /// True when one log has complete records the other has not reached yet, so
1044    /// the comparison is bounded by reading rather than by the runs agreeing.
1045    pub fn one_side_is_ahead(&self) -> bool {
1046        self.records_available_left != self.records_available_right
1047    }
1048}

1050/// Compare only the records both logs have finished writing.
1051///
1052/// The tail of a log being written may be half-flushed, and a half-flushed
1053/// record is not a difference -- it is an absence. Truncating both inputs to
1054/// their common *complete* prefix is what makes this safe to run against a live
1055/// run: bytes that have not been written yet can never be mistaken for bytes
1056/// that disagree.
1057pub fn compare_complete_prefix(
1058    contents_a: &str,
1059    contents_b: &str,
1060    opts: &LogDiffOpts,
1061    w: &mut impl std::io::Write,
1062) -> std::io::Result<PrefixComparison> {
1063    let records_available_left = complete_record_count(contents_a);
1064    let records_available_right = complete_record_count(contents_b);
1065    let records_compared = records_available_left.min(records_available_right);
1066    // Both are Some: `records_compared` is at most each side's complete count.
1067    let prefix_a = take_complete_records(contents_a, records_compared)
1068        .expect("common prefix never exceeds either side's complete record count");
1069    let prefix_b = take_complete_records(contents_b, records_compared)
1070        .expect("common prefix never exceeds either side's complete record count");
1071    let summary = log_diff_summary_from_strs(prefix_a, prefix_b, opts, w)?;
1072    Ok(PrefixComparison {
1073        summary,
1074        records_available_left,
1075        records_available_right,
1076        records_compared,
1077    })
1078}

1092fn log_diff_summary_from_strs(
1093    file_a_str: impl AsRef<str>,
1094    file_b_str: impl AsRef<str>,
1095    opts: &LogDiffOpts,
1096    w: &mut impl std::io::Write,
1097) -> std::io::Result<LogDiffSummary> {
1098    let all_a = filter_ignored(
1099        extract_log_messages(file_a_str.as_ref()),
1100        &opts.ignore_lines,
1101    );
1102    let all_b = filter_ignored(
1103        extract_log_messages(file_b_str.as_ref()),
1104        &opts.ignore_lines,
1105    );
1106
1107    writeln!(
1108        w,
1109        "Logs contain {} | {} messages total",
1110        all_a.len(),
1111        all_b.len(),
1112    )?;
1113
1114    let detcore_a = filter_detcore(&all_a);
1115    let detcore_b = filter_detcore(&all_b);
1116    let infos_a = filter_infos(&all_a);
1117    let infos_b = filter_infos(&all_b);
1118    let detlogs_a = opts.filter_deterministic(&detcore_a);
1119    let detlogs_b = opts.filter_deterministic(&detcore_b);
1120    let left_syscalls = collect_syscalls(&all_a);
1121    let right_syscalls = collect_syscalls(&all_b);
1122    writeln!(
1123        w,
1124        "Logs contain {} | {} detcore-specific messages",
1125        detcore_a.len(),
1126        detcore_b.len(),
1127    )?;
1128    writeln!(
1129        w,
1130        "Logs contain {} | {} INFO messages",
1131        infos_a.len(),
1132        infos_b.len(),
1133    )?;
1134    writeln!(
1135        w,
1136        "Logs contain {} | {} DETLOG & scheduler COMMIT messages",
1137        detlogs_a.len(),
1138        detlogs_b.len(),
1139    )?;
1140
1141    if opts.strip_lines {
1142        writeln!(
1143            w,
1144            "Normalizing known nondeterministic numerical data before comparison..."
1145        )?;
1146    } else if opts.canonicalize_addresses {
1147        writeln!(
1148            w,
1149            "Canonicalizing host addresses (ordinal by first appearance); comparing everything else exactly..."
1150        )?;
1151    }
1152
1153    let (which, compared_a, compared_b) = match opts.comparison {
1154        LogComparisonMode::Deterministic => ("DETLOG", &detlogs_a, &detlogs_b),
1155        LogComparisonMode::Info => ("INFO", &infos_a, &infos_b),
1156        LogComparisonMode::FullTrace => ("full trace", &all_a, &all_b),
1157    };
1158
1159    let first_different = first_different_message_indices(compared_a, compared_b, opts);
1160    let first_position_candidate = first_different.and_then(|(left_index, right_index)| {
1161        left_index
1162            .and_then(|index| commit_position_at_or_before(&all_a, index))
1163            .or_else(|| right_index.and_then(|index| commit_position_at_or_before(&all_b, index)))
1164    });
1165
1166    let diff_found = if opts.git_diff {
1167        git_diff(
1168            which,
1169            compared_a,
1170            compared_b,
1171            opts,
1172            w,
1173            &left_syscalls,
1174            &right_syscalls,
1175        )?
1176    } else {
1177        diff_vecs(
1178            which,
1179            compared_a,
1180            compared_b,
1181            opts,
1182            w,
1183            &left_syscalls,
1184            &right_syscalls,
1185        )?
1186    };
1187
1188    let summary = LogDiffSummary {
1189        diff_found,
1190        compared_left: compared_a.len(),
1191        compared_right: compared_b.len(),
1192        first_divergent_scheduler_turn: diff_found
1193            .then_some(first_position_candidate)
1194            .flatten()
1195            .map(|(turn, _)| turn),
1196        first_divergent_virtual_nanoseconds: diff_found
1197            .then_some(first_position_candidate)
1198            .flatten()
1199            .and_then(|(_, time)| time),
1200        first_divergent_record: diff_found
1201            .then_some(first_different)
1202            .flatten()
1203            .and_then(|(left_index, right_index)| left_index.or(right_index)),
1204    };
1205
1206    if diff_found {
1207        writeln!(w, "Done processing logs, differences found.")?;
1208    } else if summary.compared_left == 0 && summary.compared_right == 0 {
1209        // Say this plainly rather than letting "no differences" stand in for
1210        // "nothing was compared": a caller that treats the two alike turns a
1211        // no-result into a green.
1212        writeln!(
1213            w,
1214            "Done processing logs, but ZERO {which} messages were selected on either side: \
1215             nothing was compared (no-result, not a match)."
1216        )?;
1217    } else {
1218        writeln!(
1219            w,
1220            "Done processing logs, no substantive differences found ({} | {} {which} messages compared).",
1221            summary.compared_left, summary.compared_right,
1222        )?;
1223        // A matching pair keeps only this summary; its logs are not retained.
1224        // Without the next line there is no record of which shutdown path the
1225        // two runs took, so establishing whether a guest is exposed to the
1226        // empty-run-queue timing race at all needs many repeated runs rather
1227        // than one reading. See [`SCHEDULER_EMPTY_QUEUE_KICK`] for why the count
1228        // is the only thing that distinguishes the paths.
1229        //
1230        // Deliberately emitted here only: a diverging pair already reproduces
1231        // the messages verbatim in its diff, and widening retention past this
1232        // one line is how evidence directories stop being navigable.
1233        writeln!(
1234            w,
1235            "Logs contain {} | {} scheduler empty-run-queue kick messages",
1236            count_empty_queue_kicks(&infos_a),
1237            count_empty_queue_kicks(&infos_b),
1238        )?;
1239        // The other retained record, for the same reason and under the same
1240        // limit: one line, this record only. Here the committed virtual time is
1241        // the evidence rather than the record's presence, so it is reported
1242        // alongside the turn.
1243        //
1244        // BOTH runs' values are printed, not just run 1's, and this is a
1245        // correctness requirement rather than a precaution.
1246        //
1247        // Measured: with `strip_lines` -- the lossy comparator that plain
1248        // `--verify` uses -- known nondeterministic numerical data is normalized
1249        // before comparison, so a pair whose two runs committed the map read at
1250        // *different* virtual times is a MATCHING pair and reaches this branch.
1251        // Quoting one side there would report agreement on precisely the
1252        // quantity this record exists to expose: a drift that `--verify-strict`
1253        // catches and `--verify` does not. The counts above can likewise differ
1254        // on a matching pair -- under the default `Deterministic` mode a kick
1255        // asymmetry is not compared, so `1 | 0` is a pass.
1256        //
1257        // A run with no such record therefore has to read as "no such record"
1258        // rather than being silently represented by the other run's value.
1259        let (maps_left, first_left) = maps_read_commits(&infos_a);
1260        let (maps_right, first_right) = maps_read_commits(&infos_b);
1261        let positions = if first_left.is_none() && first_right.is_none() {
1262            String::new()
1263        } else {
1264            format!(
1265                " (run 1 {}, run 2 {})",
1266                describe_maps_commit(first_left),
1267                describe_maps_commit(first_right),
1268            )
1269        };
1270        writeln!(
1271            w,
1272            "Logs contain {maps_left} | {maps_right} scheduler COMMIT records reading /proc/self/maps{positions}",
1273        )?;
1274    }
1275    Ok(summary)
1276}
