30/// The in-band line a bounded log writer emits when a run's log file reaches
31/// its configured size bound.
32///
33/// This lives beside the comparator, not only beside the writer that emits it,
34/// because the comparator is the consumer that must not ignore it. The
35/// comparison below walks the two message lists in lockstep, so it can only
36/// ever speak for the retained prefix; a pair of logs that were both cut at the
37/// bound would agree on that prefix while the discarded tails were never looked
38/// at. Recognizing the marker is what keeps that from being reported as a
39/// match. `hermit-cli`'s writer emits this exact line and a unit test there
40/// binds the two by running the real writer's output through
41/// [`log_was_truncated`].
42///
43/// This is the COMPLETE sentence, not a prefix of it. An earlier version
44/// matched only the leading `=== HERMIT LOG TRUNCATED:` fragment anywhere in
45/// the text, which made the refusal fire on guest-controlled content: DETLOG
46/// records syscall path arguments verbatim, so a guest that merely touched a
47/// path containing that fragment poisoned its own `--verify` -- including with
48/// the bound already disabled, so the refusal's own remedy was unavailable.
49pub const TRUNCATION_MARKER: &str = "=== HERMIT LOG TRUNCATED: reached the configured size bound \
50     (HERMIT_LOG_MAX_BYTES). Output beyond this point was DISCARDED. The run itself continued and \
51     was NOT affected. ===";
52
53/// Whether `log_text` is the output of a writer that hit its size bound.
54///
55/// The question this answers is "was THIS LOG truncated", which is not the same
56/// question as "does this text mention the marker". The distinction is the
57/// whole point: log text contains guest-controlled bytes, so a predicate that
58/// merely searches for the marker is a predicate a guest can satisfy.
59///
60/// Two anchors make it discriminate, and both are properties of how the marker
61/// is produced rather than of what it says:
62///
63/// 1. **End of file.** `BoundedWriter::announce_truncation` writes the marker
64///    at the moment the bound is crossed and every later write is discarded, so
65///    on a truncated log the marker is the final bytes. Trailing newlines are
66///    ignored; nothing else may follow.
67/// 2. **A whole line.** The marker is emitted preceded by its own newline, so
68///    it occupies a line by itself. A DETLOG line always carries a
69///    `<timestamp> LEVEL target:` prefix and therefore can never equal it. Nor
70///    can a guest forge the line break: DETLOG renders path arguments with
71///    `Debug`, which escapes a newline to a literal backslash-`n`, so guest
72///    bytes cannot start a line at all.
73///
74/// A genuinely truncated log still satisfies both, so this narrows the
75/// predicate to the real condition without weakening it.
76pub fn log_was_truncated(log_text: &str) -> bool {
77    let trimmed = log_text.trim_end_matches(['\n', '\r']);
78    if !trimmed.ends_with(TRUNCATION_MARKER) {
79        return false;
80    }
81    // `ends_with` matched, so this offset is on a character boundary.
82    let marker_start = trimmed.len() - TRUNCATION_MARKER.len();
83    marker_start == 0 || trimmed.as_bytes()[marker_start - 1] == b'\n'
84}

132    /// UNSAFE: strips numbers and temporary paths before comparison.
133    ///
134    /// This erases timestamps and syscall values that bitwise parity exists to
135    /// compare. Never use this option to make a failing parity diff pass; doing
136    /// so is cheating. It is only for non-parity diagnostic localization.
137    #[clap(long = "unsafe-strip-lines")]
138    pub strip_lines: bool,
139
140    /// Canonicalize host memory addresses before comparison WITHOUT erasing them.
141    ///
142    /// Only addresses a producer has explicitly marked with the
143    /// `<hostaddr 0x...>` wrapper (see [`host_addr`]) are canonicalized; each
144    /// distinct marked address is rewritten to an ordinal placeholder
145    /// `<addr{N}>` assigned by order of first appearance within a single run
146    /// (see `canonicalize_addresses_in_line`). Unlike [`Self::strip_lines`],
147    /// this discards ONLY the host-specific raw pointer value: it preserves
148    /// identity (same address -> same ordinal), ordering (introduction
149    /// sequence), and aliasing (two names for one address collapse to one
150    /// ordinal), and it leaves every other byte -- virtual-time timestamps,
151    /// syscall argument/result values, counts, sizes, flags -- untouched for an
152    /// exact comparison. In canonical parity, this preserves the ability to DETECT a
153    /// difference (allocation-order or aliasing changes still diverge), which
154    /// wholesale stripping throws away.
155    ///
156    /// The marker is REQUIRED (a bare `0x...` literal is left exact) because
157    /// nothing in the compared DETLOG stream can otherwise distinguish a varying
158    /// host pointer from a reproducible hex value -- syscall arguments printed

161    /// collapse those too, silently erasing real syscall-argument divergence:

221impl LogDiffOpts {
222    fn is_skip(&self, filter: DetLogFilter) -> bool {
223        !self.include_detlogs.contains(&filter)
224    }
225
226    fn skip_detlog(&self, entry: &LogMessage<'_>) -> bool {
227        if self.skip_detlog {
228            return true;
229        }
230
231        if is_detlog_syscall(entry) && self.is_skip(DetLogFilter::Syscall) {
232            return true;
233        }
234        if is_detlog_syscall_result(entry) && self.is_skip(DetLogFilter::SyscallResult) {
235            return true;
236        }
237
238        if !is_detlog_syscall(entry)
239            && !is_detlog_syscall_result(entry)
240            && self.is_skip(DetLogFilter::Other)
241        {
242            return true;
243        }
244
245        false
246    }
247
248    fn filter_deterministic<'a>(&self, v: &[LogMessage<'a>]) -> Vec<LogMessage<'a>> {
249        v.iter()
250            .filter_map(|message| {
251                if (is_detlog(message)
252                    && !self.skip_detlog(message)
253                    && !is_scheduler_committed_time(message))
254                    || (is_commit(message)
255                        && !self.skip_commit
256                        && !is_internal_io_poll_commit(message))
257                {
258                    Some(*message)
259                } else {
260                    None
261                }
262            })
263            .collect()
264    }
265}

285impl LogComparisonPolicy {
286    fn from_options(options: &LogDiffOpts) -> Self {
287        let normalization = if options.strip_lines {
288            LogNormalization::Stripped
289        } else if options.canonicalize_addresses {
290            LogNormalization::Canonical
291        } else {
292            LogNormalization::Exact
293        };
294        Self {
295            comparison: options.comparison,
296            normalization,
297        }
298    }

301        match (self.comparison, self.normalization) {
302            (LogComparisonMode::Deterministic, LogNormalization::Exact) => "Deterministic",
303            (LogComparisonMode::Deterministic, LogNormalization::Stripped) => "Stripped",
304            (LogComparisonMode::Deterministic, LogNormalization::Canonical) => {
305                "Deterministic with Canonical host-address normalization"
306            }
307            (LogComparisonMode::Info, LogNormalization::Exact) => "Info",
308            (LogComparisonMode::Info, LogNormalization::Stripped) => {
309                "Info with Stripped normalization"
310            }
311            (LogComparisonMode::Info, LogNormalization::Canonical) => "Canonical",
312            (LogComparisonMode::FullTrace, LogNormalization::Exact) => "FullTrace",
313            (LogComparisonMode::FullTrace, LogNormalization::Stripped) => {
314                "FullTrace with Stripped normalization"
315            }
316            (LogComparisonMode::FullTrace, LogNormalization::Canonical) => {
317                "FullTrace with Canonical host-address normalization"
318            }
319        }
320    }

359/// In fully-deterministic modes, many log lines should be fully determinstic across runs.
360/// But as that is a work-in-progress, this utility strips known-nondeterministic
361/// information from logs.
362///
363/// This erasure is deliberately lossy and is NOT a parity claim: it backs the
364/// `Stripped` comparator only (`bitwise_parity: false`). `BitwiseInfoV1`
365/// canonicalizes rather than erases -- see `canonicalize_addresses_in_line`.
366/// Lossy as it is, each pattern must still erase only what it names: erasing a
367/// neighbouring field turns a real divergence into a reported match.
368///
369/// Example input/output:
370///   `Input:  COMMIT turn 3, dettid 231635 using resources Resources { tid: DetPid { inner: 231635 }, resources: {Path("/proc/231635/fd/1"): W} }`
371///   `Output: COMMIT turn <NUM>, dettid <NUM> using resources Resources { tid: DetPid { inner: <NUM> }, resources: {Path("/proc/<pid>/fd/<num>"): W} }`
372///
373/// As you can see this is overkill and smarter strategies would be possible. For example,
374/// ones that remember and post-facto-determinize certain identifiers.
375pub fn strip_log_entry(log: &str) -> String {
376    // Memory addresses, like 0x7fcfb7e7d450
377    //
378    // TODO: use a debug allocator that increases only, never reusing. Also, consider
379    // post-facto processing all of these into new virtual addresses based on the order they're seen.
380    static RE0: LazyLock<Regex> = LazyLock::new(|| Regex::new(r"\b0[xX][A-Fa-f0-9]+\b").unwrap());
381
382    // Every number, plus common duration suffixes so fractional timing jitter is
383    // not left behind. This one is terrible overkill: for `hermit run` itself and
384    // for all command tests the full contents of a COMMIT line should already be
385    // deterministic, so nothing here should need erasing. It is retained for
386    // `spawn_fn_*` variants, which fork from another process and so exercise only
387    // a *partial* detcore setup without a true process tree of their own.
388    //
389    // N.B. RE4 must run BEFORE this pattern, or `800.709_180s` is consumed here as
390    // a bare number and never reaches the duration rule.
391    static RE1: LazyLock<Regex> =
392        LazyLock::new(|| Regex::new(r"\b[\d][\d_]*(?:\.[\d][\d_]*)?(?:ns|us|µs|ms)?\b").unwrap());
393
394    // A quoted /tmp path. `[^"]*` stops at the path's OWN closing quote: a greedy
395    // `.*` here ran to the last quote on the line and erased every field after the
396    // path, so two entries differing only downstream of a /tmp path compared equal.
397    //
398    // TODO: only strip this information if the config specified to the host /tmp through.
399    // Otherwise we can determinize /tmp access fully.
400    static RE2: LazyLock<Regex> = LazyLock::new(|| Regex::new(r#"/tmp/[^"]*""#).unwrap());
401
402    // TODO: only strip this one if we're allowing through the host /proc or failing to determinize tids/pids:
403    static RE3: LazyLock<Regex> = LazyLock::new(|| Regex::new(r"/proc/[\d]+/").unwrap());
404
405    // TODO: only strip this if we're running a library-based test where we can't
406    // guarantee the starting state of the allocator/etc.
407    static RE4: LazyLock<Regex> = LazyLock::new(|| Regex::new(r"\b[\d][\d_.]*s\b").unwrap());
408
409    let log = RE4.replace_all(log, "<NANOSECONDS>");
410    let log = RE3.replace_all(&log, "/proc/<PID>/");
411    let log = RE0.replace_all(&log, "<ADDR>");
412    let log = RE1.replace_all(&log, "<NUM>");
413    let log = RE2.replace_all(&log, "/tmp/<somewhere>\"");
414    String::from(log)
415}

417/// Wrap a host memory address so `canonicalize_addresses_in_line` will
418/// canonicalize it. Producers that print a genuinely host-specific pointer
419/// (one that varies run-to-run, e.g. a supervisor-side allocation) should emit
420/// it via this helper -- `<hostaddr 0x7fcfb7e7d450>` -- instead of a bare
421/// `0x...` literal. Only marked addresses are canonicalized, so reproducible hex
422/// (syscall arguments, guest memory ranges, digests) is compared exactly.
423///
424/// Note: nothing in detcore's current DETLOG output prints a varying host
425/// pointer -- guest addresses are determinized and every logged `0x...` value is
426/// reproducible -- so this marker is presently unused in production and exists
427/// so a future host-pointer print opts into canonicalization deliberately rather
428/// than being swept up by a blanket regex.
429pub fn host_addr(addr: usize) -> String {
430    format!("<hostaddr {addr:#x}>")
431}
432
433/// Rewrite each MARKED host memory address (`<hostaddr 0x...>`, see
434/// [`host_addr`]) in `line` to an ordinal placeholder `<addr{N}>`, numbered by
435/// order of first appearance within a single run. `map`/`next` carry the per-run
436/// assignment state threaded across all of that run's lines, so `next` should
437/// start at 1 and the same `map`/`next` must be reused for every line of one run
438/// (and a FRESH pair used for the other run).
439///
440/// Canonical parity strips the wall-clock prefix, canonicalizes these marked
441/// addresses, and compares everything else exactly. This step differs
442/// from [`strip_log_entry`]'s `<ADDR>` erasure in one decisive way: erasure maps
443/// every address to a single token, so two runs that allocate in a DIFFERENT
444/// ORDER, or that ALIAS differently (one address printed twice vs. two distinct
445/// addresses), compare EQUAL -- the exact defect parity exists to catch. An
446/// ordinal assigned by first appearance keeps identity, order, and aliasing, so
447/// those cases still diverge while a pure ASLR-shift (same structure, different
448/// raw values) compares equal.
449///
450/// Only the explicit `<hostaddr ...>` marker is canonicalized. A bare `0x...`
451/// literal is left byte-for-byte for exact comparison: a blanket regex cannot
452/// tell a varying host pointer from a reproducible syscall argument printed
453/// `{:#x}` (`flock` `operation=0x2` vs `0x6`, `membarrier` bitmasks), so
454/// canonicalizing every hex token would erase real syscall-argument divergence.
455/// Decimal values (virtual-time timestamps, counts, sizes) are likewise exact.
456fn canonicalize_addresses_in_line(
457    line: &str,
458    map: &mut HashMap<String, usize>,
459    next: &mut usize,
460) -> String {
461    // Match ONLY the explicit host-address marker emitted by `host_addr`; the
462    // captured group is the raw `0x...` value used as the ordinal key.
463    static RE_HOSTADDR: LazyLock<Regex> =
464        LazyLock::new(|| Regex::new(r"<hostaddr (0[xX][A-Fa-f0-9]+)>").unwrap());
465
466    RE_HOSTADDR
467        .replace_all(line, |caps: &regex::Captures| {
468            let addr = &caps[1];
469            let ord = match map.get(addr) {
470                Some(existing) => *existing,
471                None => {
472                    let assigned = *next;
473                    *next += 1;
474                    map.insert(addr.to_string(), assigned);
475                    assigned
476                }
477            };
478            format!("<addr{ord}>")
479        })
480        .into_owned()
481}

483fn messages_for_comparison(
484    messages: &[LogMessage<'_>],
485    policy: LogComparisonPolicy,
486) -> Vec<String> {
487    match policy.normalization {
488        LogNormalization::Stripped => messages
489            .iter()
490            .map(|message| strip_log_entry(message.text))
491            .collect(),
492        LogNormalization::Canonical => {
493            let mut addresses = HashMap::new();
494            let mut next_address = 1usize;
495            messages
496                .iter()
497                .map(|message| {
498                    canonicalize_addresses_in_line(message.text, &mut addresses, &mut next_address)
499                })
500                .collect()
501        }
502        LogNormalization::Exact => messages
503            .iter()
504            .map(|message| message.text.to_owned())
505            .collect(),
506    }
507}

543    write_canonical_info_with_filter(file, writer, |_| true)

546/// Print canonical INFO messages after applying a caller-supplied record filter.
547///
548/// Every record is parsed and its level tag validated before `keep_record` is
549/// consulted. Existing callers should use [`write_canonical_info`]; backend
550/// adapters use this form to exclude transport-only records at their boundary.
551/// This low-level hook does not name the predicate, so any product verdict or
552/// JSON report using it must bind the function to a typed, serialized policy.
553pub fn write_canonical_info_with_filter(
554    file: &Path,
555    writer: &mut impl Write,
556    keep_record: impl Fn(&str) -> bool,
557) -> std::io::Result<usize> {
558    let bytes = std::fs::read(file)?;
559    let contents = String::from_utf8_lossy(&bytes);
560    let messages = canonical_info_from_str_with_filter(&contents, keep_record)?;
561    for message in &messages {
562        writeln!(writer, "{message}")?;
563    }
564    Ok(messages.len())
565}

567/// Separate a full, continuous log into discrete (possibly-multiline) log messages,
568/// stripping off the timestamps in the process.  Return lines tagged with their
569/// index number.
570/// The timestamp that begins every log record. A record runs from one match to
571/// the next, so records may span multiple lines -- which is why comparison is
572/// done on records and never on lines or bytes.
573static RECORD_START: LazyLock<Regex> = LazyLock::new(|| {
574    Regex::new(r"((Jan|Feb|Mar|Apr|May|Jun|Jul|Aug|Sep|Oct|Nov|Dec) \d\d \d\d:\d\d:\d\d\.\d+|\d+-\d\d-\d\dT\d\d:\d\d:\d\d.\d+Z) +")
575        .unwrap()
576});

578/// Byte offsets at which each record starts.
579fn record_starts(contents: &str) -> Vec<usize> {
580    RECORD_START
581        .find_iter(contents)
582        .map(|m| m.start())
583        .collect()
584}

586/// How many records in `contents` are known COMPLETE.
587///
588/// A record is complete only once the *next* record has begun, because nothing
589/// else marks its end. In a log still being written the final record may be
590/// half-flushed, so it is never counted and never compared: a partial write must
591/// not read as a difference. A buffer with one record start has zero complete
592/// records, and that is a real answer, not a failure.
593pub fn complete_record_count(contents: &str) -> usize {
594    record_starts(contents).len().saturating_sub(1)
595}

597/// The prefix of `contents` holding exactly its first `n` complete records.
598///
599/// Returns `None` when fewer than `n` complete records are present, so a caller
600/// can tell "the runs agree over n records" apart from "n records have not been
601/// written yet". Those two must never collapse into one answer.
602pub fn take_complete_records(contents: &str, n: usize) -> Option<&str> {
603    let starts = record_starts(contents);
604    if n == 0 {
605        return Some(&contents[..0]);
606    }
607    // Record n-1 ends where record n begins, so n complete records require n+1 starts.
608    starts.get(n).map(|end| &contents[..*end])
609}

611/// Split a log into tagged records, REFUSING rather than panicking on a line
612/// that carries no level tag.
613///
614/// This used to `panic!`, and a panic is the wrong failure mode for a
615/// diagnostic tool -- people reach for `log-diff` when something is ALREADY
616/// broken, and unwinding at them is the least helpful thing it can do. Worse,
617/// the `--json` consumer could not tell a crash from a real verdict: the report
618/// came back `verdict: no_result` with null counts, which reads as "no
619/// comparison was reached" rather than "the tool died".
620///
621/// This is still FAIL-CLOSED -- an unrecognised line refuses the whole
622/// comparison rather than being skipped. Skipping would silently change the
623/// compared surface, which is exactly what `RecordEnvelopePolicy` exists to
624/// make explicit and versioned. Choosing what to exclude is a disclosed policy
625/// decision, not something a parser should do on its own initiative.
626///
627/// The refusal names the offending line, because in practice the cause is a
628/// backend emitting its own untagged diagnostics into the same stream --
629/// measured: DBT writes fourteen `detcore-dbt: ...` startup lines, which is
630/// what makes a ptrace-vs-DBT comparison impossible today.
631#[derive(Debug, Clone, Copy, PartialEq, Eq)]
632struct LogMessage<'a> {
633    index: usize,
634    text: &'a str,
635    event: Option<DetLogEvent>,
636}

638fn extract_log_messages(contents: &str) -> std::io::Result<Vec<LogMessage<'_>>> {
639    let ts = &*RECORD_START;
640    let tag = Regex::new("^(ERROR|WARN|INFO|DEBUG|TRACE) ").unwrap();
641    ts.split(contents) // Not aware of a streaming version of this RE split operation.
642        .enumerate()
643        .map(|(i, s)| (i, s.trim()))
644        .filter(|(_, s)| !s.is_empty())
645        .map(|(i, s)| {
646            // Only let through lines that start with one of the expected tags:
647            if !tag.is_match(s) {
648                return Err(std::io::Error::new(
649                    std::io::ErrorKind::InvalidData,
650                    format!(
651                        "log line {i} has no ERROR/WARN/INFO/DEBUG/TRACE tag, so it cannot be \
652                         placed in the compared record stream: {s}"
653                    ),
654                ));
655            }
656            let (text, record) = DetLogRecord::split(s).map_err(|error| {
657                std::io::Error::new(
658                    std::io::ErrorKind::InvalidData,
659                    format!("log record {i} has an invalid structured DETLOG result: {error}"),
660                )
661            })?;
662            Ok(LogMessage {
663                index: i,
664                text,
665                event: record.map(|record| record.event),
666            })
667        })
668        .collect()
669}

704/// A scheduler COMMIT turn that only grants the `InternalIOPolling` resource, i.e. a
705/// granted retry of a nonblocking poll (poll/epoll_wait/wait4/futex/recv/send...). These
706/// grants are internal bookkeeping of Hermit's blocking-via-polling mechanism: how many
707/// times a thread is re-granted permission to re-attempt a nonblocking syscall before it
708/// stops returning would-block depends on when a concurrent external-IO action (e.g. a
709/// child linker process writing to a pipe) becomes ready on the host, which is wall-clock
710/// dependent and not tied to the (RCB-deterministic) logical schedule. The corresponding
711/// `NONCOMMIT ... polling resource` skips are already excluded from comparison (they are
712/// not tagged `COMMIT`); excluding the matching grant-COMMITs keeps the deterministic
713/// comparison consistent and focused on guest-observable events (the actual syscall
714/// results, still compared via their DETLOG entries).
715///
716/// SaBRe's inherited stdio pipes emit an outer device-resource turn before the inner polling
717/// turn. The scheduler tags that outer turn with `[sabre-internal-pipe-io]`; it is the same
718/// host-timing-sensitive operation and is normalized here as well. A SaBRe task with a loopback
719/// peer similarly tags the strong yield before each zero-timeout poll with
720/// `[sabre-loopback-poll-zero-timeout]`. The scheduler additionally suppresses the per-retry
721/// "advance global time for scheduler turn" DETLOG line for these turns at the source (see
722/// `Scheduler::bump_global_time`), so the two mechanisms together make the deterministic
723/// comparison insensitive to host-timing-dependent polling-loop counts.
724fn is_internal_io_poll_commit(message: &LogMessage<'_>) -> bool {
725    match message.event {
726        Some(DetLogEvent::SchedulerCommit {
727            internal_io_poll, ..
728        }) => internal_io_poll,
729        Some(_) => false,
730        None => {
731            historical_is_commit(message.text)
732                && (message.text.contains("{InternalIOPolling: ")
733                    || message.text.contains(" [sabre-internal-pipe-io]")
734                    || message.text.contains(" [sabre-loopback-poll-zero-timeout]"))
735        }
736    }
737}

739/// The scheduler's per-turn `committed_time` advance bookkeeping. `committed_time` tracks
740/// the global logical clock, which still moves forward when an `InternalIOPolling` retry
741/// (see `is_internal_io_poll_commit`) advances time -- and the number of those retries is
742/// host-timing nondeterministic. That makes the *presence* of this line on a given turn
743/// retry-count sensitive, so we exclude it from the deterministic comparison. No
744/// guest-observable signal is lost: the value is redundant with the (retained,
745/// retry-count-insensitive) "advance global time for scheduler turn" DETLOG line and with
746/// the per-turn committed time echoed on each COMMIT line.
747fn is_scheduler_committed_time(message: &LogMessage<'_>) -> bool {
748    match message.event {
749        Some(DetLogEvent::SchedulerCommittedTime) => true,
750        Some(_) => false,
751        None => message.text.contains("advancing committed_time from "),
752    }
753}

800                std::io::ErrorKind::InvalidData,
801                format!(
802                    "{label} log record {} is missing its structured DETLOG result",

805            ));
806        }
807        if let Some(event) = message.event
808            && !event_matches_human_record(event, message.text)

810            return Err(std::io::Error::new(
811                std::io::ErrorKind::InvalidData,
812                format!(
813                    "{label} log record {} has structured DETLOG kind {:?} that disagrees with its human record",
814                    message.index, event
815                ),
816            ));
817        }
818    }
819    Ok(())
820}

843/// run queue empty but is not finished yet. Current records carry
844/// [`DetLogEvent::SchedulerEmptyQueueKick`]; this spelling remains only so
845/// retained logs stay readable. `Scheduler::step2_process_blocked` in

848/// "run queue empty, exiting sched_loop." message.

873/// Historical resource text of the COMMIT record produced when a guest's
874/// runtime reads the process's own memory map during bootstrap. Current records
875/// carry `runtime_maps_read`; this spelling remains only for retained logs.
876///
877/// This record carries the second known shape of backend self-nondeterminism,
878/// and it is unlike the empty-run-queue kick in a way that matters. Two runs of
879/// one guest can agree on every scheduling decision — same turn, same thread,
880/// same resource — and still commit that turn at *different virtual times*,
881/// because virtual time advances with retired conditional branches and the
882/// number of branches the scan executes depends on the map it reads. So the
883/// evidence is not the presence of a message but the value inside one.
884///
885/// A diverging pair prints both times in its diff. A matching pair kept
886/// nothing, which left the prior question — does this guest perform the read at
887/// all, and therefore can it exhibit the drift — answerable only by collecting
888/// divergences over many runs.
889const RUNTIME_MAPS_READ_RESOURCE: &str = r#"Path("/proc/self/maps")"#;
890
891/// One run's first memory-map read COMMIT, rendered for the retained line.
892/// `None` is spelled out rather than omitted, so a run that never performed the
893/// read is distinguishable from a run whose value was simply not reported.
894fn describe_maps_commit(first: Option<(u64, Option<u64>)>) -> String {
895    match first {
896        Some((turn, Some(nanoseconds))) => {
897            format!("first at turn {turn}, committed virtual time {nanoseconds}ns")
898        }
899        Some((turn, None)) => format!("first at turn {turn}, committed virtual time unrecorded"),
900        None => "no such record".to_string(),
901    }
902}

904/// COMMIT records in `v` that read the process's own memory map: how many there
905/// are, and the turn and committed virtual time of the first.
906fn maps_read_commits(v: &[LogMessage<'_>]) -> (usize, Option<(u64, Option<u64>)>) {
907    let mut count = 0;
908    let mut first = None;
909    for message in v {
910        let reads_runtime_maps = match message.event {
911            Some(DetLogEvent::SchedulerCommit {
912                runtime_maps_read, ..
913            }) => runtime_maps_read,
914            Some(_) => false,
915            None => message.text.contains(RUNTIME_MAPS_READ_RESOURCE),
916        };
917        if !reads_runtime_maps {
918            continue;
919        }
920        let Some(position) = commit_position(message) else {
921            continue;
922        };
923        count += 1;
924        if first.is_none() {
925            first = Some(position);
926        }
927    }
928    (count, first)
929}

974/// Keep the content that identifies a differing event while removing only
975/// values already carried in separate fields beside it.
976///
977/// This is deliberately narrower than [`strip_log_entry`]. Syscall arguments,
978/// resource names, thread identities, payload bytes, and all other numbers stay
979/// exact. Record position is not part of the message and remains a separate
980/// observation.
981fn first_divergent_message(message: &LogMessage<'_>) -> String {
982    static FINISHED_SYSCALL: LazyLock<Regex> =
983        LazyLock::new(|| Regex::new(r"(finish syscall #)[0-9][0-9_]*").unwrap());
984    static COMMIT_TURN: LazyLock<Regex> =
985        LazyLock::new(|| Regex::new(r"(\bCOMMIT turn )[0-9][0-9_]*\b").unwrap());
986    static COMMITTED_TIME: LazyLock<Regex> = LazyLock::new(|| {
987        Regex::new(r"(\b(?:at time|on previously committed) )[0-9][0-9_]*(?:\.[0-9_]+)?(?:ns|s)?\b")
988            .unwrap()
989    });
990
991    let (first_line, continuation) = match message.text.split_once('\n') {
992        Some((first_line, continuation)) => (first_line, Some(continuation)),
993        None => (message.text, None),
994    };
995    let first_line =
996        if is_detlog_syscall_result(message) && finished_syscall_number(message).is_some() {
997            FINISHED_SYSCALL
998                .replace_all(first_line, "${1}<NUM>")
999                .into_owned()
1000        } else {
1001            first_line.to_string()
1002        };
1003    let first_line = if is_commit(message) && commit_position(message).is_some() {
1004        let first_line = COMMIT_TURN.replace_all(&first_line, "${1}<NUM>");
1005        COMMITTED_TIME
1006            .replace_all(&first_line, "${1}<NANOSECONDS>")
1007            .into_owned()
1008    } else {
1009        first_line
1010    };
1011    match continuation {
1012        Some(continuation) => format!("{first_line}\n{continuation}"),
1013        None => first_line,
1014    }
1015}

1033fn parse_underscored_u64(value: &str) -> Option<u64> {
1034    value.replace('_', "").parse().ok()
1035}

1037fn parse_virtual_nanoseconds(value: &str, unit: Option<&str>) -> Option<u64> {
1038    match unit {
1039        None | Some("ns") if !value.contains('.') => parse_underscored_u64(value),
1040        Some("s") => {
1041            let value = value.replace('_', "");
1042            let (seconds, fraction) = value.split_once('.').unwrap_or((&value, ""));
1043            if fraction.len() > 9 || !fraction.bytes().all(|byte| byte.is_ascii_digit()) {
1044                return None;
1045            }
1046            let seconds = seconds.parse::<u64>().ok()?;
1047            let fraction = if fraction.is_empty() {
1048                0
1049            } else {
1050                let digits = fraction.parse::<u64>().ok()?;
1051                digits.checked_mul(10_u64.pow((9 - fraction.len()) as u32))?
1052            };
1053            seconds.checked_mul(1_000_000_000)?.checked_add(fraction)
1054        }
1055        _ => None,
1056    }
1057}

1059fn historical_commit_position(message: &str) -> Option<(u64, Option<u64>)> {
1060    static TURN: LazyLock<Regex> =
1061        LazyLock::new(|| Regex::new(r"\bCOMMIT turn ([0-9][0-9_]*)\b").unwrap());
1062    static TIME: LazyLock<Regex> = LazyLock::new(|| {
1063        Regex::new(r"\b(?:at time|on previously committed) ([0-9][0-9_]*(?:\.[0-9_]+)?)(ns|s)?\b")
1064            .unwrap()
1065    });
1066
1067    let turn = parse_underscored_u64(TURN.captures(message)?.get(1)?.as_str())?;
1068    let virtual_nanoseconds = TIME.captures(message).and_then(|captures| {
1069        parse_virtual_nanoseconds(
1070            captures.get(1)?.as_str(),
1071            captures.get(2).map(|unit| unit.as_str()),
1072        )
1073    });
1074    Some((turn, virtual_nanoseconds))
1075}

1160fn write_syscall_context(
1161    w: &mut impl std::io::Write,
1162    left_index: usize,
1163    right_index: usize,
1164    left_syscalls: &[LogMessage<'_>],
1165    right_syscalls: &[LogMessage<'_>],
1166    labels: &ComparisonSideLabels,
1167    history_count: u64,
1168) -> std::io::Result<()> {
1169    if history_count == 0 {
1170        return Ok(());
1171    }
1172
1173    let left_current = syscall_at_or_before(left_syscalls, left_index);
1174    let right_current = syscall_at_or_before(right_syscalls, right_index);
1175    if left_current.is_none() && right_current.is_none() {
1176        return Ok(());
1177    }
1178
1179    writeln!(w, "Divergent syscall context:")?;
1180    for (label, current) in [
1181        (labels.left.as_str(), left_current),
1182        (labels.right.as_str(), right_current),
1183    ] {
1184        if let Some(syscall) = current {
1185            writeln!(
1186                w,
1187                "  {label}, log message {}: {}",
1188                syscall.index, syscall.text
1189            )?;
1190        } else {
1191            writeln!(w, "  {label}: <no syscall observed>")?;
1192        }
1193    }
1194
1195    let history_limit = usize::try_from(history_count).unwrap_or(usize::MAX);
1196    for (label, index, syscalls) in [
1197        (labels.left.as_str(), left_index, left_syscalls),
1198        (labels.right.as_str(), right_index, right_syscalls),
1199    ] {
1200        let history_boundary =
1201            syscall_at_or_before(syscalls, index).map_or(index, |current| current.index);
1202        let mut history = syscalls
1203            .iter()
1204            .rev()
1205            .filter(|entry| entry.index < history_boundary && is_detlog_syscall_result(entry))
1206            .take(history_limit)
1207            .copied()
1208            .collect::<Vec<_>>();
1209        history.reverse();
1210        if !history.is_empty() {
1211            writeln!(w, "  Prior completed syscalls for {label}:")?;
1212            for syscall in history {
1213                writeln!(w, "    {}", syscall.text)?;
1214            }
1215        }
1216    }
1217    writeln!(w)?;
1218
1219    Ok(())
1220}

1254/// Returns `true` if a difference is found.
1255///
1256/// We could use an existing diff library on the entire log, but this provides us more
1257/// control over how to present the (stripped/unstripped) differences, and to focus on the
1258/// per-line divergence(s), and potentially focus on the first point of divergence.
1259//
1260// Future TODO:
1261//  - report bulk differences, e.g. leftover lines, but with truncation
1262//  - report only the first K per-line differences
1263//  - detect reorderings and/or switch to larger differences for consecutive multi-line mismatches
1264fn diff_vecs(
1265    which: &str,
1266    left: (&[LogMessage<'_>], &[String]),
1267    right: (&[LogMessage<'_>], &[String]),
1268    opts: &LogDiffOpts,
1269    w: &mut impl std::io::Write,
1270    left_syscalls: &[LogMessage<'_>],
1271    right_syscalls: &[LogMessage<'_>],
1272) -> std::io::Result<bool> {
1273    let (v1, compared_left) = left;
1274    let (v2, compared_right) = right;
1275    writeln!(w, "  Comparing {which} messages...\n")?;
1276    if v1.is_empty() && v2.is_empty() {
1277        return Ok(false);
1278    }
1279
1280    let mut diff_count = 0;
1281    for (position, (left, right)) in v1.iter().zip(v2.iter()).enumerate() {
1282        let left_compared = &compared_left[position];
1283        let right_compared = &compared_right[position];
1284        if left_compared == right_compared {
1285            continue;
1286        }
1287
1288        if diff_count >= opts.limit && opts.limit != 0 {
1289            writeln!(
1290                w,
1291                "More than {} differences, eliding the rest...",
1292                opts.limit
1293            )?;
1294            break;
1295        }
1296
1297        write!(
1298            w,
1299            "({which}) Mismatch at log messages {} ({}) and {} ({}): {}",
1300            left.index,
1301            opts.side_labels.left,
1302            right.index,
1303            opts.side_labels.right,
1304            Comparison::new(opts.no_color, left_compared, right_compared)
1305        )?;
1306        if opts.strip_lines || opts.canonicalize_addresses {
1307            write!(
1308                w,
1309                "({which}) Original entries before normalization: {}",
1310                Comparison::new(opts.no_color, left.text, right.text)
1311            )?;
1312        }
1313        write_syscall_context(
1314            w,
1315            left.index,
1316            right.index,
1317            left_syscalls,
1318            right_syscalls,
1319            &opts.side_labels,
1320            opts.syscall_history,
1321        )?;
1322
1323        diff_count += 1;
1324    }
1325
1326    match v1.len().cmp(&v2.len()) {
1327        Ordering::Less => {
1328            writeln!(
1329                w,
1330                "{} contains {} extra messages not matched in {}. Displaying up to 10:",
1331                sentence_case_label(&opts.side_labels.right),
1332                v2.len() - v1.len(),
1333                opts.side_labels.left,
1334            )?;
1335            diff_count += 1;
1336            let start = v2.len() - std::cmp::min(10, v2.len() - v1.len());
1337            for message in &compared_right[start..] {
1338                writeln!(w, "{message}")?;
1339            }
1340        }
1341        Ordering::Greater => {
1342            writeln!(
1343                w,
1344                "{} contains {} extra messages not matched in {}. Displaying up to 10:",
1345                sentence_case_label(&opts.side_labels.left),
1346                v1.len() - v2.len(),
1347                opts.side_labels.right,
1348            )?;
1349            diff_count += 1;
1350            let start = v1.len() - std::cmp::min(10, v1.len() - v2.len());
1351            for message in &compared_left[start..] {
1352                writeln!(w, "{message}")?;
1353            }
1354        }
1355        Ordering::Equal => {}
1356    }
1357
1358    Ok(diff_count > 0)
1359}

1388fn git_diff(
1389    which: &str,
1390    left: (&[LogMessage<'_>], &[String]),
1391    right: (&[LogMessage<'_>], &[String]),
1392    opts: &LogDiffOpts,
1393    w: &mut impl std::io::Write,
1394    left_syscalls: &[LogMessage<'_>],
1395    right_syscalls: &[LogMessage<'_>],
1396) -> std::io::Result<bool> {
1397    let (v1, compared_left) = left;
1398    let (v2, compared_right) = right;
1399    writeln!(w, "  Comparing {which} messages...\n")?;
1400
1401    let mut file1 = NamedTempFile::new()?;
1402    let mut file2 = NamedTempFile::new()?;
1403
1404    write_compared_messages(&mut file1, compared_left)?;
1405    write_compared_messages(&mut file2, compared_right)?;
1406
1407    match Command::new("git")
1408        .args(["diff", "--color", "--color-words", "-w"])
1409        .arg(file1.path())
1410        .arg(file2.path())
1411        .status()
1412    {
1413        Ok(code) => Ok(!code.success()),
1414        Err(error) => {
1415            eprintln!("Error launching git, falling back to basic diff: {error}");
1416            diff_vecs(
1417                which,
1418                (v1, compared_left),
1419                (v2, compared_right),
1420                opts,
1421                w,
1422                left_syscalls,
1423                right_syscalls,
1424            )
1425        }
1426    }
1427}

1429/// What a log comparison actually compared, alongside whether it differed.
1430///
1431/// A bare "no difference found" boolean cannot distinguish *"the two message
1432/// streams were compared and matched"* from *"there were no messages to
1433/// compare"*. Both selected lists being empty is a NO-RESULT, not a match, so
1434/// the counts travel with the verdict and a parity consumer can require nonzero
1435/// execution before believing a green.
1436#[derive(Debug, Clone, PartialEq, Eq)]
1437pub struct LogDiffSummary {
1438    /// True if a substantive difference was found between the two runs.
1439    pub diff_found: bool,
1440    /// Number of messages actually selected for comparison from the first run.
1441    pub compared_left: usize,
1442    /// Number of messages actually selected for comparison from the second run.
1443    pub compared_right: usize,
1444    /// Scheduler turn at the first different selected message, when the first
1445    /// run has a preceding scheduler COMMIT message that identifies it.
1446    pub first_divergent_scheduler_turn: Option<u64>,
1447    /// Virtual nanoseconds at that same scheduler COMMIT, when its time is
1448    /// present and parseable.
1449    pub first_divergent_virtual_nanoseconds: Option<u64>,
1450    /// 1-based index of the first record that differs. "Diverged somewhere in
1451    /// the first N records" is a bound, not a location, and on a long run the
1452    /// two are far apart; this is the location.
1453    pub first_divergent_record: Option<usize>,
1454    /// How many syscalls the guest had COMPLETED when the divergence appeared,
1455    /// as detcore's own `finish syscall #N` counter. The fourth unit: a
1456    /// divergence located at record 108 is easier to act on when you also know
1457    /// the guest was 37 syscalls in.
1458    pub first_divergent_syscall: Option<u64>,
1459    /// The first differing compared message from the left execution, after
1460    /// removing only the syscall number, scheduler turn, and committed virtual
1461    /// time that are already recorded in separate fields.
1462    pub first_divergent_left_message: Option<String>,
1463    /// The corresponding first differing compared message from the right
1464    /// execution, with the same narrowly scoped removals.
1465    pub first_divergent_right_message: Option<String>,
1466    /// True when the comparison was REFUSED rather than performed, so there is
1467    /// no verdict about whether the runs agree.
1468    ///
1469    /// [`Self::diff_found`] is also set, because a refusal must never read as a
1470    /// match on any existing predicate. But the two are not the same fact: a
1471    /// difference is something observed, whereas a refusal is the absence of an
1472    /// observation. A caller that reports the outcome must be able to tell them
1473    /// apart -- otherwise a run that compared nothing is announced as a run
1474    /// that found the two executions differing, which is a claim nothing
1475    /// supports.
1476    ///
1477    /// Set today by exactly one condition: an input log that ends at the
1478    /// bounded writer's truncation marker.
1479    pub refused: bool,
1480}
1481
1482impl LogDiffSummary {
1483    /// True only when the comparison both ran on a nonempty selection *and*
1484    /// found no difference. An empty-vs-empty comparison is never a match.
1485    pub fn matched_with_evidence(&self) -> bool {
1486        !self.diff_found && self.compared_left > 0 && self.compared_right > 0
1487    }
1488}

1591/// A comparison of two logs that may still be growing.
1592///
1593/// The verdict alone is not reportable. "The runs agree" and "the runs agree so
1594/// far as anyone has looked" are different claims, and a caller who cannot tell
1595/// them apart will stop early and conclude the wrong thing. The record counts
1596/// are therefore part of the result, not optional detail alongside it.
1597#[derive(Debug, Clone, PartialEq, Eq)]
1598pub struct PrefixComparison {
1599    /// The verdict over the compared prefix only.
1600    pub summary: LogDiffSummary,
1601    /// Complete records present in the first log when it was read.
1602    pub records_available_left: usize,
1603    /// Complete records present in the second log when it was read.
1604    pub records_available_right: usize,
1605    /// Records actually compared: the shorter of the two available counts.
1606    pub records_compared: usize,
1607}
1608
1609impl PrefixComparison {
1610    /// True when one log has complete records the other has not reached yet, so
1611    /// the comparison is bounded by reading rather than by the runs agreeing.
1612    pub fn one_side_is_ahead(&self) -> bool {
1613        self.records_available_left != self.records_available_right
1614    }
1615}

1617/// Compare only the records both logs have finished writing.
1618///
1619/// The tail of a log being written may be half-flushed, and a half-flushed
1620/// record is not a difference -- it is an absence. Truncating both inputs to
1621/// their common *complete* prefix is what makes this safe to run against a live
1622/// run: bytes that have not been written yet can never be mistaken for bytes
1623/// that disagree.
1624pub fn compare_complete_prefix(
1625    contents_a: &str,
1626    contents_b: &str,
1627    opts: &LogDiffOpts,
1628    w: &mut impl std::io::Write,
1629) -> std::io::Result<PrefixComparison> {

1633/// Compare the complete common prefix after applying a caller-supplied record filter.
1634/// Product callers must bind the predicate to a typed, serialized policy; this
1635/// backend-neutral layer intentionally carries no backend policy identity.
1636pub fn compare_complete_prefix_with_filter(
1637    contents_a: &str,
1638    contents_b: &str,
1639    opts: &LogDiffOpts,
1640    w: &mut impl std::io::Write,
1641    keep_record: impl Fn(&str) -> bool,
1642) -> std::io::Result<PrefixComparison> {
1643    let records_available_left = complete_record_count(contents_a);
1644    let records_available_right = complete_record_count(contents_b);
1645    let records_compared = records_available_left.min(records_available_right);
1646    // Both are Some: `records_compared` is at most each side's complete count.
1647    let prefix_a = take_complete_records(contents_a, records_compared)
1648        .expect("common prefix never exceeds either side's complete record count");
1649    let prefix_b = take_complete_records(contents_b, records_compared)
1650        .expect("common prefix never exceeds either side's complete record count");
1651    let summary = log_diff_summary_from_strs_with_filter(prefix_a, prefix_b, opts, w, keep_record)?;
1652    Ok(PrefixComparison {
1653        summary,
1654        records_available_left,
1655        records_available_right,
1656        records_compared,
1657    })

1682/// Compare two in-memory logs after applying a caller-supplied record filter.
1683///
1684/// Every record is parsed and its level tag validated before `keep_record` is
1685/// consulted, so filtering cannot turn malformed input into a successful
1686/// comparison. Backend-specific transport policy belongs in the backend adapter;
1687/// this function supplies only the abstract comparison hook. Product verdicts
1688/// using it must serialize the typed policy bound to this predicate.
1689pub fn log_diff_summary_from_strs_with_filter(
1690    file_a_str: impl AsRef<str>,
1691    file_b_str: impl AsRef<str>,
1692    opts: &LogDiffOpts,
1693    w: &mut impl std::io::Write,
1694    keep_record: impl Fn(&str) -> bool,
1695) -> std::io::Result<LogDiffSummary> {
1696    // A log that reached its size bound stops early and says so in-band. The
1697    // comparison below walks the two selected lists in lockstep, so it speaks
1698    // only for the retained prefix: two runs both cut at the bound with equal
1699    // retained message counts would agree on that prefix while the discarded
1700    // tails were never compared, and reporting that as "no differences found"
1701    // would assert determinism over a region nothing looked at. Refuse the
1702    // verdict instead, in both directions of the green predicate --
1703    // `diff_found` is set AND the compared counts stay zero, so neither
1704    // `log_diff() == false` nor `matched_with_evidence()` can read as a match.
1705    let truncated_a = log_was_truncated(file_a_str.as_ref());
1706    let truncated_b = log_was_truncated(file_b_str.as_ref());
1707    if truncated_a || truncated_b {
1708        let which_side = match (truncated_a, truncated_b) {
1709            (true, true) => "both logs were",
1710            (true, false) => "the first log was",
1711            (false, true) => "the second log was",
1712            (false, false) => unreachable!("guarded by the condition above"),
1713        };
1714        writeln!(
1715            w,
1716            "REFUSING to compare: {which_side} truncated at the configured size bound \
1717             (the log ends with the bounded writer's truncation marker). The discarded tail was \
1718             never written, so no comparison of these files can establish that the runs agree \
1719             past that point. This is a NO-RESULT, not a difference and not a match. Re-run with \
1720             a larger HERMIT_LOG_MAX_BYTES, or 0 to disable the bound."
1721        )?;
1722        return Ok(LogDiffSummary {
1723            diff_found: true,
1724            compared_left: 0,
1725            compared_right: 0,
1726            first_divergent_scheduler_turn: None,
1727            first_divergent_virtual_nanoseconds: None,
1728            first_divergent_record: None,
1729            first_divergent_syscall: None,
1730            first_divergent_left_message: None,
1731            first_divergent_right_message: None,
1732            refused: true,
1733        });
1734    }
1735
1736    let extracted_a = extract_log_messages(file_a_str.as_ref())?;
1737    let extracted_b = extract_log_messages(file_b_str.as_ref())?;
1738    validate_structured_events(
1739        opts.side_labels.left.as_str(),
1740        &extracted_a,
1741        opts.require_structured_events,
1742    )?;
1743    validate_structured_events(
1744        opts.side_labels.right.as_str(),
1745        &extracted_b,
1746        opts.require_structured_events,
1747    )?;
1748    let all_a = filter_ignored(
1749        extracted_a
1750            .into_iter()
1751            .filter(|record| keep_record(record.text))
1752            .collect(),
1753        &opts.ignore_lines,
1754    );
1755    let all_b = filter_ignored(
1756        extracted_b
1757            .into_iter()
1758            .filter(|record| keep_record(record.text))
1759            .collect(),
1760        &opts.ignore_lines,
1761    );
1762
1763    writeln!(
1764        w,
1765        "Logs contain {} | {} messages total",
1766        all_a.len(),
1767        all_b.len(),
1768    )?;
1769
1770    let detcore_a = filter_detcore(&all_a);
1771    let detcore_b = filter_detcore(&all_b);
1772    let infos_a = filter_infos(&all_a);
1773    let infos_b = filter_infos(&all_b);
1774    let detlogs_a = opts.filter_deterministic(&detcore_a);
1775    let detlogs_b = opts.filter_deterministic(&detcore_b);
1776    let left_syscalls = collect_syscalls(&all_a);
1777    let right_syscalls = collect_syscalls(&all_b);
1778    writeln!(
1779        w,
1780        "Logs contain {} | {} detcore-specific messages",
1781        detcore_a.len(),
1782        detcore_b.len(),
1783    )?;
1784    writeln!(
1785        w,
1786        "Logs contain {} | {} INFO messages",
1787        infos_a.len(),
1788        infos_b.len(),
1789    )?;
1790    writeln!(
1791        w,
1792        "Logs contain {} | {} DETLOG & scheduler COMMIT messages",
1793        detlogs_a.len(),
1794        detlogs_b.len(),
1795    )?;
1796
1797    let policy = LogComparisonPolicy::from_options(opts);
1798
1799    if policy.normalization == LogNormalization::Stripped {
1800        writeln!(
1801            w,
1802            "Normalizing known nondeterministic numerical data before comparison..."
1803        )?;
1804    } else if policy.normalization == LogNormalization::Canonical {
1805        writeln!(
1806            w,
1807            "Canonicalizing host addresses (ordinal by first appearance); comparing everything else exactly..."
1808        )?;
1809    }
1810
1811    let (which, compared_a, compared_b) = match policy.comparison {
1812        LogComparisonMode::Deterministic => ("DETLOG", &detlogs_a, &detlogs_b),
1813        LogComparisonMode::Info => ("INFO", &infos_a, &infos_b),
1814        LogComparisonMode::FullTrace => ("full trace", &all_a, &all_b),
1815    };
1816
1817    // Prepare both complete streams exactly once. Address ordinals are assigned
1818    // by first appearance across the full selected stream, and every consumer
1819    // below (printing, position reporting, and comparison) receives these same
1820    // String values rather than independently rendering the logs.
1821    let prepared_a = messages_for_comparison(compared_a, policy);
1822    let prepared_b = messages_for_comparison(compared_b, policy);
1823
1824    if opts.print_logs {
1825        write_compared_logs(w, policy, &prepared_a, &prepared_b, &opts.side_labels)?;
1826    }
1827
1828    let first_different =
1829        first_different_message_indices(compared_a, &prepared_a, compared_b, &prepared_b);
1830    let first_position_candidate = first_different.and_then(|(left_index, right_index)| {
1831        left_index
1832            .and_then(|index| commit_position_at_or_before(&all_a, index))
1833            .or_else(|| right_index.and_then(|index| commit_position_at_or_before(&all_b, index)))
1834    });
1835
1836    let first_divergent_syscall_candidate = first_different.and_then(|(left, right)| {
1837        left.and_then(|index| finished_syscall_at_or_before(&all_a, index))
1838            .or_else(|| right.and_then(|index| finished_syscall_at_or_before(&all_b, index)))
1839    });
1840    let first_divergent_left_message = first_different
1841        .and_then(|(left, _)| compared_message_at_record(compared_a, &prepared_a, left));
1842    let first_divergent_right_message = first_different
1843        .and_then(|(_, right)| compared_message_at_record(compared_b, &prepared_b, right));
1844
1845    let diff_found = if opts.git_diff {
1846        git_diff(
1847            which,
1848            (compared_a, &prepared_a),
1849            (compared_b, &prepared_b),
1850            opts,
1851            w,
1852            &left_syscalls,
1853            &right_syscalls,
1854        )?
1855    } else {
1856        diff_vecs(
1857            which,
1858            (compared_a, &prepared_a),
1859            (compared_b, &prepared_b),
1860            opts,
1861            w,
1862            &left_syscalls,
1863            &right_syscalls,
1864        )?
1865    };
1866
1867    let summary = LogDiffSummary {
1868        diff_found,
1869        compared_left: compared_a.len(),
1870        compared_right: compared_b.len(),
1871        first_divergent_scheduler_turn: diff_found
1872            .then_some(first_position_candidate)
1873            .flatten()
1874            .map(|(turn, _)| turn),
1875        first_divergent_virtual_nanoseconds: diff_found
1876            .then_some(first_position_candidate)
1877            .flatten()
1878            .and_then(|(_, time)| time),
1879        first_divergent_record: diff_found
1880            .then_some(first_different)
1881            .flatten()
1882            .and_then(|(left_index, right_index)| left_index.or(right_index)),
1883        first_divergent_syscall: diff_found
1884            .then_some(first_divergent_syscall_candidate)
1885            .flatten(),
1886        first_divergent_left_message: diff_found.then_some(first_divergent_left_message).flatten(),
1887        first_divergent_right_message: diff_found
1888            .then_some(first_divergent_right_message)
1889            .flatten(),
1890        // This path compared the logs; only the refusal above declines to.
1891        refused: false,
1892    };
1893
1894    if diff_found {
1895        writeln!(w, "Done processing logs, differences found.")?;
1896    } else if summary.compared_left == 0 && summary.compared_right == 0 {
1897        // Say this plainly rather than letting "no differences" stand in for
1898        // "nothing was compared": a caller that treats the two alike turns a
1899        // no-result into a green.
1900        writeln!(
1901            w,
1902            "Done processing logs, but ZERO {which} messages were selected on either side: \
1903             nothing was compared (no-result, not a match)."
1904        )?;
1905    } else {
1906        writeln!(
1907            w,
1908            "Done processing logs, no substantive differences found ({} | {} {which} messages compared).",
1909            summary.compared_left, summary.compared_right,
1910        )?;
1911        // A matching pair keeps only this summary; its logs are not retained.
1912        // Without the next line there is no record of which shutdown path the
1913        // two runs took, so establishing whether a guest is exposed to the
1914        // empty-run-queue timing race at all needs many repeated runs rather
1915        // than one reading. See [`SCHEDULER_EMPTY_QUEUE_KICK`] for why the count
1916        // is the only thing that distinguishes the paths.
1917        //
1918        // Deliberately emitted here only: a diverging pair already reproduces
1919        // the messages verbatim in its diff, and widening retention past this
1920        // one line is how evidence directories stop being navigable.
1921        writeln!(
1922            w,
1923            "Logs contain {} | {} scheduler empty-run-queue kick messages",
1924            count_empty_queue_kicks(&infos_a),
1925            count_empty_queue_kicks(&infos_b),
1926        )?;
1927        // The other retained record, for the same reason and under the same
1928        // limit: one line, this record only. Here the committed virtual time is
1929        // the evidence rather than the record's presence, so it is reported
1930        // alongside the turn.
1931        //
1932        // BOTH runs' values are printed, not just run 1's, and this is a
1933        // correctness requirement rather than a precaution.
1934        //
1935        // Measured: with `strip_lines` -- the lossy comparator that plain
1936        // `--verify` uses -- known nondeterministic numerical data is normalized
1937        // before comparison, so a pair whose two runs committed the map read at
1938        // *different* virtual times is a MATCHING pair and reaches this branch.
1939        // Quoting one side there would report agreement on precisely the
1940        // quantity this record exists to expose: a drift that `--verify-strict`
1941        // catches and `--verify` does not. The counts above can likewise differ
1942        // on a matching pair -- under the default `Deterministic` mode a kick
1943        // asymmetry is not compared, so `1 | 0` is a pass.
1944        //
1945        // A run with no such record therefore has to read as "no such record"
1946        // rather than being silently represented by the other run's value.
1947        let (maps_left, first_left) = maps_read_commits(&infos_a);
1948        let (maps_right, first_right) = maps_read_commits(&infos_b);
1949        let positions = if first_left.is_none() && first_right.is_none() {
1950            String::new()
1951        } else {
1952            format!(
1953                " ({} {}, {} {})",
1954                opts.side_labels.left,
1955                describe_maps_commit(first_left),
1956                opts.side_labels.right,
1957                describe_maps_commit(first_right),
1958            )
1959        };
1960        writeln!(
1961            w,
1962            "Logs contain {maps_left} | {maps_right} scheduler COMMIT records reading /proc/self/maps{positions}",
1963        )?;
1964    }
1965    Ok(summary)
1966}
