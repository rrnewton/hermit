9//! Detcore's side of the access trace (`--record-access-trace-to`); the
10//! record types are `detcore_model::access_trace`.
11//!
12//! The trace is made in three places:
13//! - The guest thread notes the descriptor slots its syscall changes
14//!   (`TracedCall`), and snapshots the descriptors the call names right
15//!   after its prehook event (`FdInfo`). At the posthook, `accesses`, a
16//!   pure function of `CallFacts`, turns these, the arguments and the
17//!   result into the record's accesses.
18//! - The record rides to the scheduler as an optional payload of the
19//!   prehook, posthook and signal-delivery `TraceSchedEvent` requests the
20//!   thread sends anyway ([`AccessPayload`]).
21//! - The scheduler, which numbers the recorded events, completes each record
22//!   and writes it ([`AccessTraceWriter`]).
23//!
24//! Nothing here sends a scheduler request, charges virtual time or changes
25//! state a scheduling decision reads: the trace observes the run.

116/// The accesses a call that may not reach its posthook is known for at its
117/// prehook: exit ends its thread, exit_group its process, and execve
118/// replaces its process's image.
119pub(crate) fn entry_accesses(sysno: Sysno, pid: DetPid, tid: DetTid) -> Vec<Access> {
120    let access = |target, op| vec![Access::Lifetime { target, op }];
121    match sysno {
122        Sysno::exit => access(Task::Thread(tid), LifetimeOp::Exit),
123        Sysno::exit_group => access(Task::Process(pid), LifetimeOp::Exit),
124        Sysno::execve | Sysno::execveat => access(Task::Process(pid), LifetimeOp::Exec),
125        _ => Vec::new(),
126    }
127}

165pub struct SignalNote {
166    /// Signal number.
167    pub signo: i32,
168    /// `si_code`.
169    pub code: i32,
170    /// `si_addr` of a kernel-generated fault signal.
171    pub addr: Option<u64>,
172    /// `si_pid` of a signal sent by kill, sigqueue, tkill or tgkill.
173    pub sender: Option<SignalSender>,
174    /// Installed disposition sampled at the event, not its eventual outcome.
175    pub observed_disposition: ObservedSignalDisposition,
176}

178/// Read the fields the trace keeps from a raw `siginfo_t`
179/// (`include/uapi/asm-generic/siginfo.h`): `si_signo` at byte 0, `si_code`
180/// at byte 8, and the union at byte 16, which holds `si_addr` for a fault and
181/// `si_pid` for a user-sent signal.
182pub(crate) fn signal_note(
183    signo: i32,
184    siginfo: Option<&[u8]>,
185    observed_disposition: ObservedSignalDisposition,
186) -> SignalNote {
187    let field = |info: &[u8], at: usize, len: usize| info.get(at..at + len).map(<[u8]>::to_vec);
188    let int = |info: &[u8], at: usize| {
189        field(info, at, 4).map(|b| i32::from_ne_bytes(b.try_into().expect("four bytes")))
190    };
191    let Some(info) = siginfo else {
192        return SignalNote {
193            signo,
194            code: 0,
195            addr: None,
196            sender: None,
197            observed_disposition,
198        };
199    };
200    let code = int(info, 8).unwrap_or(0);
201    let fault = matches!(
202        signo,
203        libc::SIGSEGV | libc::SIGBUS | libc::SIGILL | libc::SIGFPE | libc::SIGTRAP
204    );
205    // A positive si_code is the kernel's (`si_fromuser`, kernel/signal.c).
206    let addr = (code > 0 && fault)
207        .then(|| field(info, 16, 8))
208        .flatten()
209        .map(|b| u64::from_ne_bytes(b.try_into().expect("eight bytes")));
210    // Only kill (SI_USER), sigqueue (SI_QUEUE) and tkill/tgkill (SI_TKILL)
211    // put the sender's pid there; a timer (SI_TIMER) puts its timer id.
212    let sender = matches!(code, libc::SI_USER | libc::SI_QUEUE | libc::SI_TKILL)
213        .then(|| int(info, 16))
214        .flatten()
215        .map(|pid| SignalSender {
216            pid: DetPid::from_raw(pid),
217            tid: None,
218        });
219    SignalNote {
220        signo,
221        code,
222        addr,
223        sender,
224        observed_disposition,
225    }
226}

231/// What the trace needs to know of one descriptor.
232#[derive(Debug, Clone, PartialEq, Eq)]
233pub(crate) struct FdInfo {
234    pub(crate) open_file: OpenFileId,
235    pub(crate) ty: FdType,
236    pub(crate) inode: Option<DetInode>,
237    pub(crate) fifo: bool,
238    pub(crate) pidfd_target: Option<DetPid>,
239}
240
241impl FdInfo {
242    pub(crate) fn of(detfd: &DetFd) -> Self {
243        let inode = match detfd.resource() {
244            Some(ResourceID::FileContents(inode)) => Some(inode),
245            _ => None,
246        };
247        let fifo = detfd
248            .stat()
249            .is_some_and(|stat| stat.mode & libc::S_IFMT == libc::S_IFIFO);
250        FdInfo {
251            open_file: detfd.open_file_id(),
252            ty: detfd.ty(),
253            inode,
254            fifo,
255            pidfd_target: detfd.pidfd_target(),
256        }
257    }
258
259    fn channel(&self) -> Option<ChannelKind> {
260        match self.ty {
261            FdType::Pipe => Some(ChannelKind::Pipe),
262            FdType::Socket => Some(ChannelKind::Socket),
263            FdType::Eventfd => Some(ChannelKind::Eventfd),
264            FdType::Regular if self.fifo => Some(ChannelKind::Fifo),
265            _ => None,
266        }
267    }
268
269    fn channel_id(&self) -> Option<ChannelId> {
270        self.fifo
271            .then_some(self.inode.map(ChannelId::Inode))
272            .flatten()
273    }
274}

315impl TracedCall {
316    /// Start a traced call whose named descriptors were `fds`.
317    pub(crate) fn begin(&self, fds: BTreeMap<i32, FdInfo>) {
318        *self.0.lock().expect("traced call mutex poisoned") = Some(CallState {
319            fds,
320            notes: Vec::new(),
321        });
322    }
323
324    /// Whether a traced call is in progress, so a slot change is noted.
325    pub(crate) fn is_active(&self) -> bool {
326        self.0.lock().expect("traced call mutex poisoned").is_some()
327    }
328
329    /// Note a slot change of the call in progress, if there is one.
330    pub(crate) fn note(&self, note: FdNote) {
331        if let Some(state) = self.0.lock().expect("traced call mutex poisoned").as_mut() {
332            state.notes.push(note);
333        }
334    }
335
336    /// End the call in progress and return what it gathered.
337    pub(crate) fn take(&self) -> Option<CallState> {
338        self.0.lock().expect("traced call mutex poisoned").take()
339    }

355/// Where, in guest memory, [`Written`] values are, for a call that succeeded
356/// with `result`: the address and the field to read there.
357pub(crate) fn written_locations(
358    sysno: Sysno,
359    args: [u64; 6],
360    result: i64,
361) -> Vec<(u64, WrittenField)> {
362    if result < 0 {
363        return Vec::new();
364    }
365    match sysno {
366        Sysno::timer_create => vec![(args[2], WrittenField::TimerId)],
367        // aarch64 has only pipe2 (include/uapi/asm-generic/unistd.h).
368        #[cfg(target_arch = "x86_64")]
369        Sysno::pipe => vec![(args[0], WrittenField::FdPair)],
370        Sysno::pipe2 => vec![(args[0], WrittenField::FdPair)],
371        Sysno::socketpair => vec![(args[3], WrittenField::FdPair)],
372        // `si_pid` is the union's first member, at byte 16 of the siginfo.
373        Sysno::waitid if args[2] != 0 => vec![(args[2] + 16, WrittenField::WaitedPid)],
374        Sysno::clone3 => vec![(args[0], WrittenField::Clone3Flags)],
375        _ => Vec::new(),
376    }
377}

379/// Read the [`Written`] values of a call that returned `result`. A value the
380/// guest's memory no longer holds readable is left out.
381pub(crate) fn read_written<M: MemoryAccess>(
382    memory: &M,
383    sysno: Sysno,
384    args: [u64; 6],
385    result: i64,
386) -> Written {
387    let mut written = Written::default();
388    for (addr, field) in written_locations(sysno, args, result) {
389        let int =
390            |at: u64| Addr::<i32>::from_raw(at as usize).and_then(|a| memory.read_value(a).ok());
391        match field {
392            WrittenField::TimerId => written.timer_id = int(addr),
393            WrittenField::WaitedPid => written.waited_pid = int(addr),
394            WrittenField::FdPair => {
395                written.fd_pair = int(addr).zip(int(addr + 4)).map(|(a, b)| [a, b]);
396            }
397            WrittenField::Clone3Flags => {
398                written.clone3_flags =
399                    Addr::<u64>::from_raw(addr as usize).and_then(|a| memory.read_value(a).ok());
400            }
401        }
402    }
403    written
404}

431/// The descriptors a call uses as operands, each with its role. A positional
432/// offset of -1 (preadv2, pwritev2) means the cursor.
433fn operand_fds(sysno: Sysno, args: [u64; 6]) -> Vec<(i32, Role)> {
434    use Sysno::*;
435    let fd = |i: usize| args[i] as i32;
436    let at = |i: usize| (args[i] as i64 >= 0).then_some(args[i]);
437    match sysno {
438        read | readv | recvfrom | recvmsg | recvmmsg => vec![(fd(0), Role::Read)],
439        pread64 | preadv => vec![(fd(0), Role::ReadAt(Some(args[3])))],
440        preadv2 => vec![(fd(0), at(3).map_or(Role::Read, |o| Role::ReadAt(Some(o))))],
441        write | writev | sendto | sendmsg | sendmmsg => vec![(fd(0), Role::Write)],
442        pwrite64 | pwritev => vec![(fd(0), Role::WriteAt(Some(args[3])))],
443        pwritev2 => vec![(fd(0), at(3).map_or(Role::Write, |o| Role::WriteAt(Some(o))))],
444        lseek => vec![(fd(0), Role::Seek)],
445        ftruncate | fallocate => vec![(fd(0), Role::Truncate)],
446        shutdown => vec![(fd(0), Role::Shutdown)],
447        timerfd_settime => vec![(fd(0), Role::TimerSet)],
448        timerfd_gettime => vec![(fd(0), Role::TimerGet)],
449        sendfile => vec![(fd(0), Role::Write), (fd(1), Role::Read)],
450        splice | copy_file_range => vec![(fd(0), Role::ReadAt(None)), (fd(2), Role::WriteAt(None))],
451        tee => vec![(fd(0), Role::Read), (fd(1), Role::Write)],
452        epoll_ctl => vec![(fd(0), Role::Use), (fd(2), Role::Use)],
453        mmap if args[3] & libc::MAP_ANONYMOUS as u64 == 0 => vec![(fd(4), Role::Use)],
454        signalfd4 if fd(0) >= 0 => vec![(fd(0), Role::Use)],
455        fstat | fsync | fdatasync | fchmod | fchown | flock | fcntl | ioctl | getdents64
456        | fadvise64 | fstatfs | fchdir | syncfs | accept | accept4 | connect | bind | listen
457        | getsockname | getpeername | setsockopt | getsockopt | epoll_pwait | epoll_pwait2
458        | inotify_add_watch | inotify_rm_watch | fgetxattr | fsetxattr | flistxattr
459        | fremovexattr | pidfd_send_signal | pidfd_getfd | dup | dup3 => {
460            vec![(fd(0), Role::Use)]
461        }
462        // The legacy calls only x86_64 has. aarch64's generic table
463        // (include/uapi/asm-generic/unistd.h) has getdents64, epoll_pwait and
464        // dup3 above instead, and names sync_file_range sync_file_range2.
465        #[cfg(target_arch = "x86_64")]
466        getdents | sync_file_range | epoll_wait | dup2 => vec![(fd(0), Role::Use)],
467        #[cfg(target_arch = "aarch64")]
468        sync_file_range2 => vec![(fd(0), Role::Use)],
469        _ => Vec::new(),
470    }
471}

473/// Every descriptor a call names, whose state before the call the trace
474/// needs: its operands, the descriptor close releases, and the one dup2 or
475/// dup3 replaces.
476pub(crate) fn named_fds(sysno: Sysno, args: [u64; 6]) -> Vec<i32> {
477    let mut fds: Vec<i32> = operand_fds(sysno, args)
478        .into_iter()
479        .map(|(fd, _)| fd)
480        .collect();
481    match sysno {
482        Sysno::close => fds.push(args[0] as i32),
483        #[cfg(target_arch = "x86_64")]
484        Sysno::dup2 => fds.push(args[1] as i32),
485        Sysno::dup3 => fds.push(args[1] as i32),
486        _ => {}
487    }
488    fds.retain(|fd| *fd >= 0);
489    fds.sort_unstable();
490    fds.dedup();
491    fds
492}

494/// What a call that reads a clock or a timer returns, for its
495/// [`ReturnedBytes`]: the result itself when it is the time (`time`,
496/// `alarm`, `times`), and the guest buffers the kernel or Detcore filled.
497/// Buffers are x86-64 sizes: timespec and timeval 16 bytes, itimerspec and
498/// itimerval 32, timezone 8, tms 32. A remaining-time buffer counts only
499/// when a sleep was interrupted, the one case Linux writes it: do_nanosleep
500/// copies it out (`nanosleep_copyout`, kernel/time/hrtimer.c:2072 and
501/// :2121 at v6.16) and returns -ERESTART_RESTARTBLOCK, which the posthook
502/// sees as is when the call will restart and as -EINTR when a handler ran.
503/// An absolute clock_nanosleep never writes it; an old-value
504/// buffer only when the guest passed one.
505pub(crate) fn time_outputs(
506    sysno: Sysno,
507    args: [u64; 6],
508    result: i64,
509) -> (Option<i64>, Vec<ByteRange>) {
510    use Sysno::*;
511    const TIMESPEC: u64 = 16;
512    const TIMEVAL: u64 = 16;
513    const ITIMER: u64 = 32;
514    const TIMEZONE: u64 = 8;
515    const TMS: u64 = 32;
516    let at = |addr: u64, len: u64| {
517        (addr != 0).then_some(ByteRange {
518            start: addr,
519            end: addr.saturating_add(len),
520        })
521    };
522    let ok = result >= 0;
523    /// Linux's restart code for an interrupted relative sleep
524    /// (include/linux/errno.h).
525    const ERESTART_RESTARTBLOCK: i64 = 516;
526    let interrupted = result == -(libc::EINTR as i64) || result == -ERESTART_RESTARTBLOCK;
527    let buffers: Vec<Option<ByteRange>> = match sysno {
528        clock_gettime | clock_getres if ok => vec![at(args[1], TIMESPEC)],
529        gettimeofday if ok => vec![at(args[0], TIMEVAL), at(args[1], TIMEZONE)],
530        nanosleep if interrupted => vec![at(args[1], TIMESPEC)],
531        clock_nanosleep if interrupted && args[1] & libc::TIMER_ABSTIME as u64 == 0 => {
532            vec![at(args[3], TIMESPEC)]
533        }
534        timerfd_gettime | timer_gettime | getitimer if ok => vec![at(args[1], ITIMER)],
535        timerfd_settime | timer_settime if ok => vec![at(args[3], ITIMER)],
536        setitimer if ok => vec![at(args[2], ITIMER)],
537        times if ok => vec![at(args[0], TMS)],
538        _ => Vec::new(),
539    };
540    // time and alarm are x86_64-only; aarch64's C library builds them on
541    // clock_gettime and setitimer, whose outputs are above.
542    #[cfg(target_arch = "x86_64")]
543    let returns_time = matches!(sysno, time | alarm | times);
544    #[cfg(not(target_arch = "x86_64"))]
545    let returns_time = sysno == times;
546    let value = (ok && returns_time).then_some(result);
547    (value, buffers.into_iter().flatten().collect())
548}

579/// The record's accesses: section 4 of the M5 design applied to one call.
580pub(crate) fn accesses(f: &CallFacts<'_>) -> Vec<Access> {
581    let mut out = Vec::new();
582    let ok = f.result >= 0;
583    let slot = |fd: i32| FdSlot { files: f.files, fd };
584
585    for (fd, role) in operand_fds(f.sysno, f.args) {
586        if fd < 0 {
587            continue;
588        }
589        let info = f.fds.get(&fd);
590        out.push(Access::Fd {
591            slot: slot(fd),
592            open_file: info.map(|info| info.open_file),
593            op: FdOp::Use,
594        });
595        if let Some(info) = info {
596            data_accesses(f, info, role, &mut out);
597        }
598    }
599
600    // A close that released nothing Detcore models (EBADF, a double close)
601    // is still a close of the slot.
602    if f.sysno == Sysno::close {
603        let fd = f.args[0] as i32;
604        let noted = f
605            .notes
606            .iter()
607            .any(|note| note.slot.fd == fd && note.op == FdOp::Close);
608        if !noted {
609            out.push(Access::Fd {
610                slot: slot(fd),
611                open_file: f.fds.get(&fd).map(|info| info.open_file),
612                op: FdOp::Close,
613            });
614        }
615    }
616
617    for note in ordered_notes(f) {
618        out.push(Access::Fd {
619            slot: note.slot,
620            open_file: note.open_file,
621            op: note.op.clone(),
622        });
623        if note.op == FdOp::Close
624            && let Some(info) = &note.released
625            && let Some(channel) = info.channel()
626        {
627            out.push(Access::Channel {
628                open_file: info.open_file,
629                id: info.channel_id(),
630                channel,
631                op: ChannelOp::Close,
632            });
633        }
634    }
635
636    futex_accesses(f, &mut out);
637    if ok {
638        memory_accesses(f, &mut out);
639    }
640    timer_accesses(f, &mut out);
641    lifetime_accesses(f, &mut out);
642    for (op, range) in f.buffers {
643        out.push(Access::Memory {
644            mm: f.mm,
645            range: *range,
646            op: *op,
647        });
648    }
649    out
650}

652/// The notes in record order: a pipe's or socketpair's two opens in the
653/// order of its descriptor array, so the first is `fds[0]`.
654fn ordered_notes<'a>(f: &CallFacts<'a>) -> Vec<&'a FdNote> {
655    let mut notes: Vec<&FdNote> = f.notes.iter().collect();
656    if let Some(pair) = f.written.fd_pair {
657        notes.sort_by_key(|note| pair.iter().position(|fd| *fd == note.slot.fd));
658    }
659    notes
660}

662fn data_accesses(f: &CallFacts<'_>, info: &FdInfo, role: Role, out: &mut Vec<Access>) {
663    if let Some(channel) = info.channel() {
664        let op = match role {
665            Role::Read | Role::ReadAt(_) => ChannelOp::Read,
666            Role::Write | Role::WriteAt(_) => ChannelOp::Write,
667            Role::Shutdown => ChannelOp::Shutdown,
668            _ => return,
669        };
670        out.push(Access::Channel {
671            open_file: info.open_file,
672            id: info.channel_id(),
673            channel,
674            op,
675        });
676        return;
677    }
678    if info.ty == FdType::Timerfd {
679        let op = match role {
680            Role::Read => TimerOp::Read,
681            Role::TimerSet => TimerOp::Set,
682            Role::TimerGet => TimerOp::Get,
683            _ => return,
684        };
685        out.push(Access::Timer {
686            timer: TimerId::Fd {
687                open_file: info.open_file,
688            },
689            op,
690        });
691        return;
692    }
693    if !matches!(info.ty, FdType::Regular | FdType::Memfd) {
694        return;
695    }
696    let cursor = |op| Access::OpenFile {
697        open_file: info.open_file,
698        op,
699    };
700    // A positional range covers what the call moved, or, when it moved
701    // nothing (end of file, an error), the bytes it asked for.
702    let range = |offset: Option<u64>| {
703        offset.map(|start| {
704            let moved = u64::try_from(f.result).unwrap_or(0);
705            let len = if moved > 0 {
706                moved
707            } else if matches!(f.sysno, Sysno::pread64 | Sysno::pwrite64) {
708                f.args[2]
709            } else {
710                0
711            };
712            ByteRange {
713                start,
714                end: start.saturating_add(len),
715            }
716        })
717    };
718    let contents = |range: Option<ByteRange>, op| {
719        info.inode
720            .map(|inode| Access::FileContents { inode, range, op })
721    };
722    match role {
723        Role::Read => {
724            out.push(cursor(CursorOp::Read));
725            out.extend(contents(None, DataOp::Read));
726        }
727        Role::Write => {
728            out.push(cursor(CursorOp::Write));
729            out.extend(contents(None, DataOp::Write));
730        }
731        Role::ReadAt(offset) => out.extend(contents(range(offset), DataOp::Read)),
732        Role::WriteAt(offset) => out.extend(contents(range(offset), DataOp::Write)),
733        Role::Seek => out.push(cursor(CursorOp::Seek)),
734        Role::Truncate => out.extend(contents(None, DataOp::Write)),
735        Role::Use | Role::Shutdown | Role::TimerSet | Role::TimerGet => {}
736    }
737}

739/// `FUTEX_LOCK_PI2`, which the libc crate may not name.
740const FUTEX_LOCK_PI2: i32 = 13;
741
742fn futex_accesses(f: &CallFacts<'_>, out: &mut Vec<Access>) {
743    if f.sysno != Sysno::futex {
744        return;
745    }
746    let op = f.args[1] as i32;
747    let private = op & libc::FUTEX_PRIVATE_FLAG != 0;
748    let cmd = op & !(libc::FUTEX_PRIVATE_FLAG | libc::FUTEX_CLOCK_REALTIME);
749    let (uaddr, uaddr2) = (f.args[0], f.args[4]);
750    let words: &[(u64, FutexOp)] = match cmd {
751        libc::FUTEX_WAIT | libc::FUTEX_WAIT_BITSET | libc::FUTEX_WAIT_REQUEUE_PI => {
752            &[(uaddr, FutexOp::Wait)]
753        }
754        libc::FUTEX_WAKE | libc::FUTEX_WAKE_BITSET => &[(uaddr, FutexOp::Wake)],
755        libc::FUTEX_REQUEUE | libc::FUTEX_CMP_REQUEUE | libc::FUTEX_CMP_REQUEUE_PI => {
756            &[(uaddr, FutexOp::Wake), (uaddr2, FutexOp::RequeueTarget)]
757        }
758        libc::FUTEX_WAKE_OP => &[(uaddr, FutexOp::Wake), (uaddr2, FutexOp::ModifyWake)],
759        libc::FUTEX_LOCK_PI | libc::FUTEX_TRYLOCK_PI | FUTEX_LOCK_PI2 => &[(uaddr, FutexOp::Lock)],
760        libc::FUTEX_UNLOCK_PI => &[(uaddr, FutexOp::Unlock)],
761        _ => &[],
762    };
763    for (addr, op) in words {
764        out.push(Access::Futex {
765            futex: (f.futex_id)(*addr, private),
766            op: *op,
767        });
768    }
769}

771fn memory_accesses(f: &CallFacts<'_>, out: &mut Vec<Access>) {
772    let a = f.args;
773    let mut push = |range: ByteRange, op| {
774        out.push(Access::Memory {
775            mm: f.mm,
776            range,
777            op,
778        })
779    };
780    match f.sysno {
781        Sysno::mmap => push(page_range(f.result as u64, a[1]), MemoryOp::Map),
782        Sysno::munmap => push(page_range(a[0], a[1]), MemoryOp::Unmap),
783        Sysno::mprotect | Sysno::pkey_mprotect => push(page_range(a[0], a[1]), MemoryOp::Protect),
784        Sysno::madvise => push(page_range(a[0], a[1]), MemoryOp::Advise),
785        Sysno::mremap => {
786            push(page_range(a[0], a[1]), MemoryOp::Unmap);
787            push(page_range(f.result as u64, a[2]), MemoryOp::Map);
788        }
789        _ => {}
790    }
791}

793fn timer_accesses(f: &CallFacts<'_>, out: &mut Vec<Access>) {
794    let posix = |id: i32| TimerId::Posix { process: f.pid, id };
795    let interval = |which: i32| TimerId::Interval {
796        process: f.pid,
797        which,
798    };
799    let ok = f.result >= 0;
800    let access = match f.sysno {
801        Sysno::timer_create if ok => f.written.timer_id.map(|id| (posix(id), TimerOp::Create)),
802        Sysno::timer_settime => Some((posix(f.args[0] as i32), TimerOp::Set)),
803        Sysno::timer_gettime | Sysno::timer_getoverrun => {
804            Some((posix(f.args[0] as i32), TimerOp::Get))
805        }
806        Sysno::timer_delete => Some((posix(f.args[0] as i32), TimerOp::Delete)),
807        Sysno::setitimer => Some((interval(f.args[0] as i32), TimerOp::Set)),
808        Sysno::getitimer => Some((interval(f.args[0] as i32), TimerOp::Get)),
809        // x86_64 only; on aarch64 alarm(3) is a setitimer call.
810        #[cfg(target_arch = "x86_64")]
811        Sysno::alarm => Some((interval(libc::ITIMER_REAL), TimerOp::Set)),
812        Sysno::timerfd_create if ok => f
813            .notes
814            .iter()
815            .find(|note| note.slot.fd as i64 == f.result && matches!(note.op, FdOp::Open { .. }))
816            .and_then(|note| note.open_file)
817            .map(|open_file| (TimerId::Fd { open_file }, TimerOp::Create)),
818        _ => None,
819    };
820    if let Some((timer, op)) = access {
821        out.push(Access::Timer { timer, op });
822    }
823}

825fn lifetime_accesses(f: &CallFacts<'_>, out: &mut Vec<Access>) {
826    let a = f.args;
827    let raw = |i: usize| a[i] as i32;
828    let created = |flags: u64| {
829        (f.result > 0).then(|| {
830            let child = DetPid::from_raw(f.result as i32);
831            if flags & libc::CLONE_THREAD as u64 != 0 {
832                Task::Thread(child)
833            } else {
834                Task::Process(child)
835            }
836        })
837    };
838    let signal =
839        |target: Option<Task>, signo: i32| target.map(|t| (t, LifetimeOp::Signal { signo }));
840    let access = match f.sysno {
841        Sysno::clone => created(a[0]).map(|t| (t, LifetimeOp::Create)),
842        Sysno::clone3 => f
843            .written
844            .clone3_flags
845            .and_then(created)
846            .map(|t| (t, LifetimeOp::Create)),
847        // x86_64 only; on aarch64 fork(2) and vfork(2) are clone calls
848        // with SIGCHLD (and CLONE_VM | CLONE_VFORK) as the flags, above.
849        #[cfg(target_arch = "x86_64")]
850        Sysno::fork | Sysno::vfork => created(0).map(|t| (t, LifetimeOp::Create)),
851        Sysno::exit => Some((Task::Thread(f.tid), LifetimeOp::Exit)),
852        Sysno::exit_group => Some((Task::Process(f.pid), LifetimeOp::Exit)),
853        Sysno::execve | Sysno::execveat => Some((Task::Process(f.pid), LifetimeOp::Exec)),
854        Sysno::kill => {
855            let target = match raw(0) {
856                pid if pid > 0 => Some(Task::Process(DetPid::from_raw(pid))),
857                // The caller's own group, which the writer resolves.
858                0 => Some(Task::ProcessGroup(OWN_PROCESS_GROUP)),
859                -1 => None,
860                pid => Some(Task::ProcessGroup(DetPid::from_raw(pid.wrapping_neg()))),
861            };
862            signal(target, raw(1))
863        }
864        Sysno::tkill => signal(Some(Task::Thread(DetPid::from_raw(raw(0)))), raw(1)),
865        Sysno::tgkill | Sysno::rt_tgsigqueueinfo => {
866            signal(Some(Task::Thread(DetPid::from_raw(raw(1)))), raw(2))
867        }
868        Sysno::rt_sigqueueinfo => signal(Some(Task::Process(DetPid::from_raw(raw(0)))), raw(1)),
869        Sysno::pidfd_send_signal => signal(
870            f.fds
871                .get(&raw(0))
872                .and_then(|info| info.pidfd_target)
873                .map(Task::Process),
874            raw(1),
875        ),
876        Sysno::wait4 if f.result > 0 => Some((
877            Task::Process(DetPid::from_raw(f.result as i32)),
878            LifetimeOp::Wait,
879        )),
880        Sysno::waitid if f.result == 0 => f
881            .written
882            .waited_pid
883            .filter(|pid| *pid > 0)
884            .map(|pid| (Task::Process(DetPid::from_raw(pid)), LifetimeOp::Wait)),
885        _ => None,
886    };
887    if let Some((target, op)) = access {
888        out.push(Access::Lifetime { target, op });
889    }
890}

892/// kill(0)'s target as the guest thread writes it: the caller's own process

896/// The bytes returned through guest buffers, from the buffers' contents in
897/// order. `read` reads at most `want` bytes of a buffer and fails as the
898/// guest read would. For a call whose return value is output too (`time`, `alarm`,
899/// `times`), `value`'s eight little-endian bytes come first.
900pub(crate) fn summarize_output<E>(
901    value: Option<i64>,
902    buffers: &[ByteRange],
903    mut read: impl FnMut(u64, usize) -> Result<Vec<u8>, E>,
904) -> Option<ReturnedBytes> {
905    let value = value.map(i64::to_le_bytes);
906    let value_len = value.map_or(0, |bytes| bytes.len() as u64);
907    let len: u64 = value_len + buffers.iter().map(|b| b.end - b.start).sum::<u64>();
908    if len == 0 {
909        return None;
910    }
911    let mut digest = value.map_or(RETURNED_DIGEST_SEED, |bytes| {
912        returned_digest(RETURNED_DIGEST_SEED, &bytes)
913    });
914    let mut left = RETURNED_DIGEST_CAP - value_len;
915    for buffer in buffers {
916        if left == 0 {
917            break;
918        }
919        let want = (buffer.end - buffer.start).min(left);
920        let bytes = read(buffer.start, want as usize).ok()?;
921        digest = returned_digest(digest, &bytes);
922        left -= want;
923    }
924    Some(ReturnedBytes {
925        len,
926        digest,
927        truncated: len > RETURNED_DIGEST_CAP,
928    })
929}

1035    /// Take the payload of the event the scheduler just recorded at `index`.
1036    pub fn observe(
1037        &mut self,
1038        index: EventIndex,
1039        event: &SchedEvent,
1040        payload: AccessPayload,
1041        groups: &dyn ProcessGroups,
1042    ) {
1043        let tid = event.dettid;
1044        match (payload, &event.op) {
1045            (AccessPayload::SyscallEntry(entry), Op::Syscall(_, SyscallPhase::Prehook)) => {
1046                // A call that never reached its posthook (a restarted wait,
1047                // a successful execve) ends when its thread enters the next.
1048                if let Some(unfinished) = self.open.remove(&tid) {
1049                    self.finish_record(unfinished);
1050                }
1051                self.open.insert(
1052                    tid,
1053                    SyscallRecord {
1054                        pre: index,
1055                        post: None,
1056                        tid,
1057                        pid: entry.pid,
1058                        nth: entry.nth,
1059                        sysno: entry.sysno,
1060                        args: entry.args,
1061                        result: None,
1062                        accesses: entry.accesses,
1063                        returned: None,
1064                        out_bytes: None,
1065                    },
1066                );
1067            }
1068            (AccessPayload::SyscallExit(outcome), Op::Syscall(_, SyscallPhase::Posthook)) => {
1069                let Some(mut record) = self.open.remove(&tid) else {
1070                    return;
1071                };
1072                let outcome = *outcome;
1073                record.post = Some(index);
1074                record.result = Some(outcome.result);
1075                record.accesses = resolve_groups(outcome.accesses, record.pid, groups);
1076                record.returned = outcome.returned;
1077                record.out_bytes = outcome.out_bytes;
1078                self.finish_record(record);
1079            }
1080            (AccessPayload::Signal(note), Op::SignalReceived(_)) => {
1081                if index >= self.from {
1082                    self.write(TraceRecord::Signal(SignalRecord {
1083                        event: index,
1084                        tid,
1085                        signo: note.signo,
1086                        code: note.code,
1087                        addr: note.addr,
1088                        sender: note.sender,
1089                        observed_disposition: note.observed_disposition,
1090                    }));
1091                }
1092            }
1093            (payload, op) => warn!(
1094                "[access trace] dropped a payload that does not match its event {:?}: {:?}",
1095                op, payload
1096            ),
1097        }
1098    }

1100    /// Name the channels a finished record created, fill in the channel
1101    /// identities of its accesses, and write it unless it ends before
1102    /// `--access-trace-from`.
1103    fn finish_record(&mut self, mut record: SyscallRecord) {
1104        if creates_a_channel_pair(record.sysno) && record.result == Some(0) {
1105            let opened: Vec<OpenFileId> = record
1106                .accesses
1107                .iter()
1108                .filter_map(|access| match access {
1109                    Access::Fd {
1110                        open_file: Some(open_file),
1111                        op: FdOp::Open { .. },
1112                        ..
1113                    } => Some(*open_file),
1114                    _ => None,
1115                })
1116                .collect();
1117            if let [first, second] = opened[..] {
1118                let id = ChannelId::Pair(first);
1119                self.pairs.insert(first, id);
1120                self.pairs.insert(second, id);
1121            }
1122        }
1123        for access in &mut record.accesses {
1124            if let Access::Channel { open_file, id, .. } = access
1125                && id.is_none()
1126            {
1127                *id = self.pairs.get(open_file).copied();
1128            }
1129        }
1130        if record.post.unwrap_or(record.pre) >= self.from {
1131            self.write(TraceRecord::Syscall(record));
1132        }
1133    }

1135    fn write(&mut self, record: TraceRecord) {
1136        let Some(out) = self.out.as_mut() else {
1137            return;
1138        };
1139        let line = record.to_line();
1140        let cap = if let Some(max) = self.max_records.filter(|max| self.records >= *max) {
1141            Some(format!("--access-trace-max-records={max}"))
1142        } else if let Ok(line) = &line
1143            && self.bytes.saturating_add(line.len() as u64) > self.max_bytes
1144        {
1145            Some(format!(
1146                "--access-trace-max-bytes={}{}",
1147                self.max_bytes,
1148                if self.default_max_bytes {
1149                    " (the default)"
1150                } else {
1151                    ""
1152                }
1153            ))
1154        } else {
1155            None
1156        };
1157        if let Some(cap) = cap {
1158            // A cap: stop here, before the first record past it, and leave
1159            // the end line out. Said once, as out is gone after.
1160            warn!(
1161                "[access trace] {} reached {}; it stops here, without its end line",
1162                self.name, cap
1163            );
1164            if let Err(error) = out.flush() {
1165                warn!("[access trace] flushing {} failed ({})", self.name, error);
1166            }
1167            self.out = None;
1168            self.capped = true;
1169            return;
1170        }
1171        match line.and_then(|line| {
1172            out.write_all(&line)?;
1173            Ok(line.len() as u64)
1174        }) {
1175            Ok(written) => {
1176                self.records += 1;
1177                self.bytes += written;
1178            }
1179            Err(error) => {
1180                warn!(
1181                    "[access trace] writing the trace failed ({}); it ends here, without its end line",
1182                    error
1183                );
1184                self.out = None;
1185            }
1186        }
1187    }

1189    /// Write the calls that never returned and the end line, and flush.
1190    /// `events` is the number of events the run recorded.
1191    pub fn finish(mut self, events: u64) -> io::Result<()> {
1192        for record in std::mem::take(&mut self.open).into_values() {
1193            self.finish_record(record);
1194        }
1195        let records = self.records;
1196        if self.capped {
1197            return Ok(());
1198        }
1199        let Some(mut out) = self.out.take() else {
1200            return Err(io::Error::other(
1201                "an earlier write failed; the trace has no end line",
1202            ));
1203        };
1204        TraceRecord::End(TraceEnd { events, records }).write_line(&mut out)?;
1205        out.flush()
1206    }

1209/// Add one process target per member to each group signal, and resolve the
1210/// caller's own group.
1211fn resolve_groups(
1212    accesses: Vec<Access>,
1213    caller: DetPid,
1214    groups: &dyn ProcessGroups,
1215) -> Vec<Access> {
1216    let mut out = Vec::with_capacity(accesses.len());
1217    for access in accesses {
1218        let Access::Lifetime {
1219            target: Task::ProcessGroup(group),
1220            op,
1221        } = access
1222        else {
1223            out.push(access);
1224            continue;
1225        };
1226        let group = if group == OWN_PROCESS_GROUP {
1227            match groups.group_of(caller) {
1228                Some(group) => group,
1229                None => continue,
1230            }
1231        } else {
1232            group
1233        };
1234        out.push(Access::Lifetime {
1235            target: Task::ProcessGroup(group),
1236            op,
1237        });
1238        for member in groups.members(group) {
1239            out.push(Access::Lifetime {
1240                target: Task::Process(member),
1241                op,
1242            });
1243        }
1244    }
1245    out
1246}
