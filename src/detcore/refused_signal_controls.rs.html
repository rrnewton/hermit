9//! Guest signal controls an in-guest runtime refused before Detcore saw the
10//! call, reported once per run.
11//!
12//! On the in-guest backends the runtime answers a guest's handler-installing
13//! `rt_sigaction`, and an alternate signal stack, with `EPERM` itself
14//! (reverie-inguest `protect_runtime_control`): a guest handler would run
15//! outside the Tool's view. Until A5 step 3c makes guest handlers supported,
16//! each refusal is announced once per run, by name, so that a run whose guest
17//! carries on after the `EPERM` (many do, Rust's standard library among them)
18//! is not silently different from Linux.
19//!
20//! The runtime records each refusal; the runtime crate that hosts Detcore
21//! installs the function that takes them ([`install`]); Detcore takes them at
22//! each syscall entry (`report`) and tells the coordinator, which prints one
23//! WARN per refused signal and one for the alternate stack, per run. No other
24//! backend installs a source, so nothing is reported there.
25//!
26//! A refusal is reported at the process's next syscall Detcore sees, which is
27//! its exit call at the latest when it exits normally. A process that dies
28//! before then reports nothing: one killed from outside, and one that faults
29//! in the signal it asked to handle (SIGSEGV's default action is delivered by
30//! the runtime while it traps instructions and by the kernel otherwise; the
31//! kernel ends a process on SIGBUS, SIGFPE or SIGILL), because neither path
32//! reaches Detcore; a thread's exit hook does not run on those deaths either,
33//! so it is not used.
34//!
35//! The texts of Detcore's own refusals of the in-guest runtime's reserved
36//! signals (`Config::in_guest_reserved_signals`: a send to another process,
37//! or an arming) live here too. Detcore decides those itself, at the call,
38//! and prints each through its guest WARN path (`crate::guest_log`).
39

47/// What a runtime refused since it was last asked: the signals whose new
48/// action it refused, bit `n - 1` for signal `n`, and whether it refused an
49/// alternate signal stack.
50pub type RefusedSignalControls = (u64, bool);
51
52static SOURCE: OnceLock<fn() -> RefusedSignalControls> = OnceLock::new();
53
54/// Installs `source`, which takes the signal controls the runtime refused
55/// since it was last called. The runtime crate calls this once per process
56/// image; a forked child inherits it. Returns false if one was installed
57/// already.
58pub fn install(source: fn() -> RefusedSignalControls) -> bool {
59    SOURCE.set(source).is_ok()
60}
61
62/// Takes what the runtime refused since the last call: nothing where no
63/// source is installed.
64fn take() -> RefusedSignalControls {
65    SOURCE.get().map_or((0, false), |source| source())
66}
67
68/// At a syscall's entry, tells the coordinator what the runtime refused since
69/// the last report (`GlobalRequest::AnnounceRefusedSignalControls`). The
70/// message carries no logical time and is answered before any accounting, so
71/// whether a run reports changes neither its schedule nor its records.
72pub(crate) async fn report<G, T>(guest: &mut G)
73where
74    G: Guest<Detcore<T>>,
75    T: RecordOrReplay,
76{
77    let (actions, alternate_stack) = take();
78    if actions == 0 && !alternate_stack {
79        return;
80    }
81    crate::tool_global::announce_refused_signal_controls(guest, actions, alternate_stack).await;
82}

84/// The WARN for a signal whose new action the runtime refused: a handler, or
85/// any action for a signal the runtime keeps for itself (SIGSYS, and SIGSEGV
86/// while it traps instructions).
87pub(crate) fn handler_refused_warning(signum: i32) -> String {
88    let name = signal_name(signum);
89    format!(
90        "HERMIT_SIGNAL_HANDLER_REFUSED signum={signum} ({name}): the guest asked for a new \
91         action for it (a handler, or any action for a signal the in-guest runtime keeps for \
92         itself) and received EPERM, so the handler was not installed: in-guest signal \
93         handlers are unsupported until A5 step 3c (https://github.com/rrnewton/hermit/issues/4485). \
94         A {name} the guest receives takes the action it had before."
95    )
96}

98/// The WARN for a guest handler that Detcore refused with EPERM because the
99/// run's threads are not sequentialized (A5 step 3c slice 0g, design 16.2):
100/// `caller` is the calling thread. Printed for every refused install, through
101/// Detcore's guest WARN path (`crate::guest_log`), like the reserved-signal
102/// refusals: Detcore decides it itself, at the call.
103pub(crate) fn unsequentialized_handler_refused_warning(signum: i32, caller: i32) -> String {
104    let name = signal_name(signum);
105    format!(
106        "HERMIT_UNSEQUENTIALIZED_HANDLER_REFUSED signum={signum} ({name}) caller={caller}: the \
107         guest asked for a handler and received EPERM, so the handler was not installed: threads \
108         are not sequentialized (--no-sequentialize-threads, or --strace-only, which turns \
109         sequentialization off; for a library caller, Config::sequentialize_threads is false), \
110         and an in-guest guest handler needs Detcore's scheduler to run one guest thread at a \
111         time. A named refusal, not Linux's behaviour (A5 step 3c, \
112         https://github.com/rrnewton/hermit/issues/4485). A {name} the guest receives takes the \
113         action it had before."
114    )
115}

147/// A call that arms a notification signal, as the reserved-signal refusal
148/// names it (`Detcore::refuse_reserved_signal_arming`).
149#[derive(Clone, Copy, Debug, PartialEq, Eq)]
150pub(crate) enum ReservedArming {
151    /// `fcntl(F_SETSIG)`: the descriptor's I/O signal.
152    FcntlSetsig,
153    /// `timer_create` with a signal notification.
154    TimerCreate,
155    /// `prctl(PR_SET_PDEATHSIG)`: the parent-death signal.
156    PrctlPdeathsig,
157}
158
159impl ReservedArming {
160    /// The call and operation, as the WARN names them.
161    pub(crate) fn call(self) -> &'static str {
162        match self {
163            Self::FcntlSetsig => "fcntl(F_SETSIG)",
164            Self::TimerCreate => "timer_create",
165            Self::PrctlPdeathsig => "prctl(PR_SET_PDEATHSIG)",
166        }
167    }
168
169    /// Who would send the signal, and when.
170    fn sender(self) -> &'static str {
171        match self {
172            Self::FcntlSetsig => {
173                "the kernel would send it for the descriptor's I/O events, at moments host timing \
174                 decides"
175            }
176            Self::TimerCreate => "Detcore's virtual timer would deliver it at each expiry",
177            Self::PrctlPdeathsig => {
178                "the kernel would send it when the parent thread exits, at a moment host timing \
179                 decides"
180            }
181        }
182    }
183}

204/// The run-wide stopping refusal (A5 design D49) for a syscall that, under
205/// the in-guest delivery gate, finished with one of the kernel's internal
206/// restart codes other than `ERESTARTSYS` although no wait recorded a signal
207/// ending it (design D40): the in-guest runtime would hand the raw code to
208/// the guest, since only `ERESTARTSYS` is re-run there. `ERESTARTSYS` itself
209/// passes through as that re-run.
210pub(crate) fn restart_code_without_signal_refusal(call: &str, code: &str, caller: i32) -> String {
211    format!(
212        "hermit refused to continue the run (HERMIT_RESTART_CODE_WITHOUT_SIGNAL): {call} in \
213         thread {caller} finished with {code} under the in-guest delivery gate, but no wait \
214         recorded a signal that ended it, so the in-guest runtime would return the kernel's \
215         internal restart code to the guest, which Linux never does \
216         (A5 design D40, https://github.com/rrnewton/hermit/issues/4485)."
217    )
218}

299/// The run-wide stopping refusal (A5 design D44(b) through D49) for a send
300/// of `signum` to thread `target`, blocked in a kernel-run call (a vfork
301/// parent's wait when `vfork`) whose continuation turn has not been granted:
302/// the call runs in the kernel under the in-guest runtime's mask, so the
303/// signal would stay pending and the call would never end for it, where Linux
304/// interrupts it. `call` is the sending call (or "the scheduler" for its own
305/// send), `sender` the sending thread. `unreadable`: the target's dispositions
306/// could not be read, which refuses, fail closed.
307pub(crate) fn kernel_run_signal_refusal(
308    call: &str,
309    signum: i32,
310    sender: Option<i32>,
311    target: i32,
312    vfork: bool,
313    unreadable: bool,
314) -> String {
315    let name = signal_name(signum);
316    let sender = sender.map_or_else(String::new, |sender| format!(" by thread {sender}"));
317    let blocked_in = if vfork {
318        "a vfork parent's wait for its child"
319    } else {
320        "a call the kernel runs outside the schedule"
321    };
322    let why = if unreadable {
323        "its signal dispositions could not be read, so the send is refused, fail closed"
324    } else {
325        "Linux would interrupt the call for it, but the in-guest runtime's mask holds it pending \
326         for as long as the call blocks"
327    };
328    format!(
329        "hermit refused to continue the run (HERMIT_KERNEL_RUN_SIGNAL_REFUSED): signal {signum} \
330         ({name}) sent{sender} through {call} to thread {target}, which is blocked in {blocked_in} \
331         and has not been granted its continuation turn: {why} \
332         (A5 design D44, https://github.com/rrnewton/hermit/issues/4485)."
333    )
334}

336/// Why a stop-signal send was refused (`stop_signal_send_refused_warning`).
337#[derive(Clone, Copy, Debug, PartialEq, Eq)]
338pub(crate) enum StopSendRefusal {
339    /// SIGSTOP, which always stops.
340    Sigstop,
341    /// SIGTSTP, SIGTTIN or SIGTTOU while the target's action is `SIG_DFL`, as
342    /// its `/proc/<tid>/status` `SigIgn`/`SigCgt` say in the sender's turn.
343    DefaultAction,
344    /// The target's `/proc/<tid>/status` could not be read, so whether its
345    /// action is `SIG_DFL` is unknown: refused, fail closed.
346    StatusUnreadable,
347}

349/// The WARN for a send of a stop signal that Detcore refused with EPERM
350/// under the in-guest delivery gate (A5 design D48, IR-4): `call` is the
351/// system call, `sender` the calling thread, `target` the thread or process it
352/// named. A program that ignores the EPERM and then waits for the target with
353/// `waitpid(WUNTRACED)` waits until the run's wall timeout (IR-4(c)).
354pub(crate) fn stop_signal_send_refused_warning(
355    call: &str,
356    signum: i32,
357    sender: i32,
358    target: i32,
359    why: StopSendRefusal,
360) -> String {
361    let name = signal_name(signum);
362    let why = match why {
363        StopSendRefusal::Sigstop => "SIGSTOP always stops its target".to_string(),
364        StopSendRefusal::DefaultAction => format!(
365            "the target's action for {name} is SIG_DFL (its /proc status SigIgn and SigCgt, read in \
366             the sender's turn), which stops it"
367        ),
368        StopSendRefusal::StatusUnreadable => format!(
369            "the target's /proc status could not be read, so whether its action for {name} is \
370             SIG_DFL is unknown, and the send is refused, fail closed"
371        ),
372    };
373    format!(
374        "HERMIT_STOP_SIGNAL_SEND_REFUSED signum={signum} ({name}) call={call} sender={sender} \
375         target={target}: received EPERM and nothing was sent: {why}, and the in-guest backend \
376         cannot stop a guest process on the schedule. A named refusal, not Linux's behaviour \
377         (A5 design D48, https://github.com/rrnewton/hermit/issues/4485)."
378    )
379}

381/// `signum`'s name: `SIGUSR1`, or, for a realtime signal, its kernel number
382/// (C libraries reserve different numbers of them, so `SIGRTMIN+n` would
383/// depend on the guest's).
384fn signal_name(signum: i32) -> String {
385    match nix::sys::signal::Signal::try_from(signum) {
386        Ok(signal) => signal.as_str().to_string(),
387        Err(_) => format!("realtime signal {signum}"),
388    }
389}

391#[cfg(test)]
392mod tests {
393    use super::*;
394
395    #[test]
396    fn the_warnings_name_the_signal_the_errno_and_why() {
397        let usr1 = handler_refused_warning(libc::SIGUSR1);
398        assert!(usr1.starts_with("HERMIT_SIGNAL_HANDLER_REFUSED signum=10 (SIGUSR1): "));
399        let stack = alternate_stack_refused_warning();
400        assert!(stack.starts_with("HERMIT_SIGNAL_ALTSTACK_REFUSED: "));
401        for text in [&usr1, &stack] {
402            assert!(text.contains("received EPERM"), "{text}");
403            assert!(text.contains("was not installed"), "{text}");
404            assert!(
405                text.contains("in-guest signal handlers are unsupported until A5 step 3c"),
406                "{text}"
407            );
408        }
409        assert!(handler_refused_warning(36).contains("(realtime signal 36)"));
410
411        let send = reserved_signal_send_refused_warning("tgkill", libc::SIGSYS, 4, 5);
412        assert!(
413            send.starts_with(
414                "HERMIT_RESERVED_SIGNAL_SEND_REFUSED signum=31 (SIGSYS) call=tgkill sender=4 \
415                 target=process 5: received EPERM"
416            ),
417            "{send}"
418        );
419        let arming =
420            reserved_signal_arming_refused_warning(ReservedArming::FcntlSetsig, libc::SIGSEGV, 4);
421        assert!(
422            arming.starts_with(
423                "HERMIT_RESERVED_SIGNAL_ARMING_REFUSED signum=11 (SIGSEGV) \
424                 call=fcntl(F_SETSIG) caller=4: received EPERM"
425            ),
426            "{arming}"
427        );
428        // Each arming names who would send the signal: host timing for the
429        // kernel's I/O and parent-death signals, never for Detcore's timers.
430        assert!(
431            arming.contains("at moments host timing decides"),
432            "{arming}"
433        );
434        let timer =
435            reserved_signal_arming_refused_warning(ReservedArming::TimerCreate, libc::SIGSYS, 4);
436        assert!(timer.contains("call=timer_create caller=4"), "{timer}");
437        assert!(
438            timer.contains("Detcore's virtual timer would deliver it"),
439            "{timer}"
440        );
441        assert!(!timer.contains("host timing"), "{timer}");
442        let pdeathsig = reserved_signal_arming_refused_warning(
443            ReservedArming::PrctlPdeathsig,
444            libc::SIGTRAP,
445            4,
446        );
447        assert!(
448            pdeathsig.contains("call=prctl(PR_SET_PDEATHSIG) caller=4"),
449            "{pdeathsig}"
450        );
451        assert!(
452            pdeathsig.contains("when the parent thread exits"),
453            "{pdeathsig}"
454        );
455        let unsequentialized = unsequentialized_handler_refused_warning(libc::SIGALRM, 4);
456        assert!(
457            unsequentialized.starts_with(
458                "HERMIT_UNSEQUENTIALIZED_HANDLER_REFUSED signum=14 (SIGALRM) caller=4: the guest \
459                 asked for a handler and received EPERM, so the handler was not installed: \
460                 threads are not sequentialized"
461            ),
462            "{unsequentialized}"
463        );
464        // It names what turns sequentialization off, for the CLI and for a
465        // library caller.
466        for cause in [
467            "--no-sequentialize-threads",
468            "--strace-only",
469            "Config::sequentialize_threads",
470        ] {
471            assert!(unsequentialized.contains(cause), "{unsequentialized}");
472        }
473        for text in [&send, &arming, &timer, &pdeathsig, &unsequentialized] {
474            assert!(
475                text.contains("A named refusal, not Linux's behaviour"),
476                "{text}"
477            );
478            assert!(
479                text.contains("https://github.com/rrnewton/hermit/issues/4485"),
480                "{text}"
481            );
482            for backend in ["liteinst", "LiteInst", "in-guest-trap", "sabre", "SaBRe"] {
483                assert!(!text.contains(backend), "{text}");
484            }
485        }
486        // No backend is named: the text is the same wherever it is printed.
487        for text in [usr1, stack] {
488            for backend in ["liteinst", "LiteInst", "in-guest-trap", "sabre", "SaBRe"] {
489                assert!(!text.contains(backend), "{text}");
490            }
491        }
492    }
