9//! Guest signal controls an in-guest runtime refused before Detcore saw the
10//! call, reported once per run.
11//!
12//! On the in-guest backends the runtime answers a guest's handler-installing
13//! `rt_sigaction`, and an alternate signal stack, with `EPERM` itself
14//! (reverie-inguest `protect_runtime_control`): a guest handler would run
15//! outside the Tool's view. Until A5 step 3c makes guest handlers supported,
16//! each refusal is announced once per run, by name, so that a run whose guest
17//! carries on after the `EPERM` (many do, Rust's standard library among them)
18//! is not silently different from Linux.
19//!
20//! The runtime records each refusal; the runtime crate that hosts Detcore
21//! installs the function that takes them ([`install`]); Detcore takes them at
22//! each syscall entry (`report`) and tells the coordinator, which prints one
23//! WARN per refused signal and one for the alternate stack, per run. No other
24//! backend installs a source, so nothing is reported there.
25//!
26//! A refusal is reported at the process's next syscall Detcore sees, which is
27//! its exit call at the latest when it exits normally. A process that dies
28//! before then reports nothing: one killed from outside, and one that faults
29//! in the signal it asked to handle (SIGSEGV's default action is delivered by
30//! the runtime while it traps instructions and by the kernel otherwise; the
31//! kernel ends a process on SIGBUS, SIGFPE or SIGILL), because neither path
32//! reaches Detcore; a thread's exit hook does not run on those deaths either,
33//! so it is not used.
34//!
35//! The texts of Detcore's own refusals of the in-guest runtime's reserved
36//! signals (`Config::in_guest_reserved_signals`: a send to another process,
37//! or an arming) live here too. Detcore decides those itself, at the call,
38//! and prints each through its guest WARN path (`crate::guest_log`).
39

47/// What a runtime refused since it was last asked: the signals whose new
48/// action it refused, bit `n - 1` for signal `n`, and whether it refused an
49/// alternate signal stack.
50pub type RefusedSignalControls = (u64, bool);
51
52static SOURCE: OnceLock<fn() -> RefusedSignalControls> = OnceLock::new();
53
54/// Installs `source`, which takes the signal controls the runtime refused
55/// since it was last called. The runtime crate calls this once per process
56/// image; a forked child inherits it. Returns false if one was installed
57/// already.
58pub fn install(source: fn() -> RefusedSignalControls) -> bool {
59    SOURCE.set(source).is_ok()
60}
61
62/// Takes what the runtime refused since the last call: nothing where no
63/// source is installed.
64fn take() -> RefusedSignalControls {
65    SOURCE.get().map_or((0, false), |source| source())
66}

68/// At a syscall's entry, tells the coordinator what the runtime refused since
69/// the last report (`GlobalRequest::AnnounceRefusedSignalControls`). The
70/// message carries no logical time and is answered before any accounting, so
71/// whether a run reports changes neither its schedule nor its records.
72pub(crate) async fn report<G, T>(guest: &mut G)
73where
74    G: Guest<Detcore<T>>,
75    T: RecordOrReplay,
76{
77    let (actions, alternate_stack) = take();
78    if actions == 0 && !alternate_stack {
79        return;
80    }
81    crate::tool_global::announce_refused_signal_controls(guest, actions, alternate_stack).await;
82}

147/// A call that arms a notification signal, as the reserved-signal refusal
148/// names it (`Detcore::refuse_reserved_signal_arming`).
149#[derive(Clone, Copy, Debug, PartialEq, Eq)]
150pub(crate) enum ReservedArming {
151    /// `fcntl(F_SETSIG)`: the descriptor's I/O signal.
152    FcntlSetsig,
153    /// `timer_create` with a signal notification.
154    TimerCreate,
155    /// `prctl(PR_SET_PDEATHSIG)`: the parent-death signal.
156    PrctlPdeathsig,
157}
158
159impl ReservedArming {
160    /// The call and operation, as the WARN names them.
161    pub(crate) fn call(self) -> &'static str {
162        match self {
163            Self::FcntlSetsig => "fcntl(F_SETSIG)",
164            Self::TimerCreate => "timer_create",
165            Self::PrctlPdeathsig => "prctl(PR_SET_PDEATHSIG)",
166        }
167    }
168
169    /// Who would send the signal, and when.
170    fn sender(self) -> &'static str {
171        match self {
172            Self::FcntlSetsig => {
173                "the kernel would send it for the descriptor's I/O events, at moments host timing \
174                 decides"
175            }
176            Self::TimerCreate => "Detcore's virtual timer would deliver it at each expiry",
177            Self::PrctlPdeathsig => {
178                "the kernel would send it when the parent thread exits, at a moment host timing \
179                 decides"
180            }
181        }
182    }
183}

425/// of `signum` to thread `target`, blocked in a kernel-run call (a vfork
426/// parent's wait when `vfork`) whose continuation turn has not been granted:
427/// the call runs in the kernel under the in-guest runtime's mask, so the

444    } else {

448        "its signal dispositions could not be read, so the send is refused, fail closed"

454        "hermit refused to continue the run (HERMIT_KERNEL_RUN_SIGNAL_REFUSED): signal {signum} \
455         ({name}) sent{sender} through {call} to thread {target}, which is blocked in {blocked_in} \
456         and has not been granted its continuation turn: {why} \
457         (A5 design D44, https://github.com/rrnewton/hermit/issues/4485)."
458    )

461/// Why a stop-signal send was refused (`stop_signal_send_refused_warning`).
462#[derive(Clone, Copy, Debug, PartialEq, Eq)]
463pub(crate) enum StopSendRefusal {
464    /// SIGSTOP, which always stops.
465    Sigstop,
466    /// SIGTSTP, SIGTTIN or SIGTTOU while the target's action is `SIG_DFL`, as
467    /// its `/proc/<tid>/status` `SigIgn`/`SigCgt` say in the sender's turn.
468    DefaultAction,
469    /// The target's `/proc/<tid>/status` could not be read, so whether its
470    /// action is `SIG_DFL` is unknown: refused, fail closed.
471    StatusUnreadable,

474/// The WARN for a send of a stop signal that Detcore refused with EPERM
475/// under the in-guest delivery gate (A5 design D48, IR-4): `call` is the
476/// system call, `sender` the calling thread, `target` the thread or process it
477/// named. A program that ignores the EPERM and then waits for the target with
478/// `waitpid(WUNTRACED)` waits until the run's wall timeout (IR-4(c)).
479pub(crate) fn stop_signal_send_refused_warning(
480    call: &str,
481    signum: i32,
482    sender: i32,
483    target: i32,
484    why: StopSendRefusal,
485) -> String {
486    let name = signal_name(signum);
487    let why = match why {
488        StopSendRefusal::Sigstop => "SIGSTOP always stops its target".to_string(),
489        StopSendRefusal::DefaultAction => format!(
490            "the target's action for {name} is SIG_DFL (its /proc status SigIgn and SigCgt, read in \
491             the sender's turn), which stops it"
492        ),
493        StopSendRefusal::StatusUnreadable => format!(
494            "the target's /proc status could not be read, so whether its action for {name} is \
495             SIG_DFL is unknown, and the send is refused, fail closed"
496        ),
497    };
498    format!(
499        "HERMIT_STOP_SIGNAL_SEND_REFUSED signum={signum} ({name}) call={call} sender={sender} \
500         target={target}: received EPERM and nothing was sent: {why}, and the in-guest backend \
501         cannot stop a guest process on the schedule. A named refusal, not Linux's behaviour \
502         (A5 design D48, https://github.com/rrnewton/hermit/issues/4485)."
503    )
504}
505
506/// `signum`'s name: `SIGUSR1`, or, for a realtime signal, its kernel number
507/// (C libraries reserve different numbers of them, so `SIGRTMIN+n` would
508/// depend on the guest's).
509fn signal_name(signum: i32) -> String {
510    match nix::sys::signal::Signal::try_from(signum) {
511        Ok(signal) => signal.as_str().to_string(),
512        Err(_) => format!("realtime signal {signum}"),
513    }
514}

532            );
533        }
534        assert!(handler_refused_warning(36).contains("(realtime signal 36)"));
535
536        let send = reserved_signal_send_refused_warning("tgkill", libc::SIGSYS, 4, 5);
537        assert!(
538            send.starts_with(
539                "HERMIT_RESERVED_SIGNAL_SEND_REFUSED signum=31 (SIGSYS) call=tgkill sender=4 \
540                 target=process 5: received EPERM"
541            ),
542            "{send}"
543        );
544        let arming =
545            reserved_signal_arming_refused_warning(ReservedArming::FcntlSetsig, libc::SIGSEGV, 4);
546        assert!(
547            arming.starts_with(
548                "HERMIT_RESERVED_SIGNAL_ARMING_REFUSED signum=11 (SIGSEGV) \
549                 call=fcntl(F_SETSIG) caller=4: received EPERM"
550            ),
551            "{arming}"
552        );
553        // Each arming names who would send the signal: host timing for the
554        // kernel's I/O and parent-death signals, never for Detcore's timers.
555        assert!(
556            arming.contains("at moments host timing decides"),
557            "{arming}"
558        );
559        let timer =
560            reserved_signal_arming_refused_warning(ReservedArming::TimerCreate, libc::SIGSYS, 4);
561        assert!(timer.contains("call=timer_create caller=4"), "{timer}");
562        assert!(
563            timer.contains("Detcore's virtual timer would deliver it"),

566        assert!(!timer.contains("host timing"), "{timer}");
567        let pdeathsig = reserved_signal_arming_refused_warning(
568            ReservedArming::PrctlPdeathsig,
569            libc::SIGTRAP,
570            4,
571        );
572        assert!(
573            pdeathsig.contains("call=prctl(PR_SET_PDEATHSIG) caller=4"),
574            "{pdeathsig}"

576        assert!(
577            pdeathsig.contains("when the parent thread exits"),

590        // library caller.

594            "Config::sequentialize_threads",

597        }
598        for text in [&send, &arming, &timer, &pdeathsig, &unsequentialized] {
599            assert!(
600                text.contains("A named refusal, not Linux's behaviour"),

610        }
