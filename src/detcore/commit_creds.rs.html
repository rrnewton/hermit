9//! Linux's rule for when a credential change clears a task's parent-death
10//! signal, shared by every path through which Detcore's model changes a
11//! task's credentials: a credential setter
12//! (`Detcore::reset_parent_death_on_credential_change`) and an exec
13//! (`Detcore::handle_post_exec`). Linux applies it in `commit_creds`
14//! (kernel/cred.c:406-413, v6.17), which both reach: a setter directly, and
15//! an exec through `begin_new_exec` (fs/exec.c:1263).
16
17use reverie::syscalls::Sysno;
18
19/// The reason Detcore refuses an exec, by name, where it holds a
20/// parent-death signal for the exec'ing task but cannot read the task's
21/// permitted capability set: without the set it cannot apply the capability
22/// part of the rule (`clears_parent_death`), and the signal could stay where
23/// Linux clears it.
24pub(crate) const UNREADABLE_PERMITTED_SET: &str = "Detcore holds a parent-death signal for this task \
25     but cannot read its permitted capability set (capget refused, or no stack scratch below the \
26     red zone), so it cannot apply Linux's rule that an exec growing that set clears the signal \
27     (commit_creds)";
28
29/// The parts of a task's credentials that `commit_creds` compares to decide
30/// whether to clear the parent-death signal.
31#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
32pub(crate) struct CommittedCredentials {
33    pub euid: u32,
34    pub egid: u32,
35    pub fsuid: u32,
36    pub fsgid: u32,
37    /// The permitted capability set, its low 64 bits, which hold every
38    /// capability Linux defines.
39    pub permitted: u64,
40}
41
42impl CommittedCredentials {
43    /// Detcore's fixed virtual root, every id 0, with permitted set
44    /// `permitted`.
45    pub(crate) fn virtual_root(permitted: u64) -> Self {
46        Self {
47            permitted,
48            ..Self::default()
49        }
50    }
51
52    /// These credentials after a successful credential setter `sysno` with
53    /// `args` by a privileged caller, as Linux's setters compute them
54    /// (kernel/sys.c, v6.17): `setuid` and `setgid` set every user or group
55    /// id (lines 680 and 498); `setreuid`, `setregid`, `setresuid` and
56    /// `setresgid` set the effective id where they give one (-1 keeps it),
57    /// and the filesystem id follows the effective id (lines 617, 456, 765
58    /// and 859); `setfsuid` and `setfsgid` set the filesystem id alone (lines
59    /// 926 and 970). An invalid id (-1) changes nothing. The permitted set is
60    /// kept: a setter can drop capabilities but never gain one. Any other
61    /// call (`setgroups`, for one) changes none of these.
62    pub(crate) fn after_setter(self, sysno: Sysno, args: [u64; 6]) -> Self {
63        let id = |at: usize| Some(args[at] as u32).filter(|id| *id != u32::MAX);
64        let mut new = self;
65        match sysno {
66            Sysno::setuid => {
67                if let Some(uid) = id(0) {
68                    new.euid = uid;
69                    new.fsuid = uid;
70                }
71            }
72            Sysno::setgid => {
73                if let Some(gid) = id(0) {
74                    new.egid = gid;
75                    new.fsgid = gid;
76                }
77            }
78            Sysno::setreuid | Sysno::setresuid => {
79                if let Some(euid) = id(1) {
80                    new.euid = euid;
81                }
82                new.fsuid = new.euid;
83            }
84            Sysno::setregid | Sysno::setresgid => {
85                if let Some(egid) = id(1) {
86                    new.egid = egid;
87                }
88                new.fsgid = new.egid;
89            }
90            Sysno::setfsuid => {
91                if let Some(fsuid) = id(0) {
92                    new.fsuid = fsuid;
93                }
94            }
95            Sysno::setfsgid => {
96                if let Some(fsgid) = id(0) {
97                    new.fsgid = fsgid;
98                }
99            }
100            _ => {}
101        }
102        new
103    }
104}

106/// Whether Linux's `commit_creds` clears the parent-death signal when a
107/// task's credentials move from `old` to `new` (kernel/cred.c:406-413,
108/// v6.17): an effective or filesystem user or group id changes, or the
109/// permitted set gains a capability (`!cred_cap_issubset`,
110/// kernel/cred.c:353-375, for two credentials in one user namespace, as a
111/// setter's and an exec's are). A capability dropped, or a real or saved id
112/// moved alone, clears nothing.
113pub(crate) fn clears_parent_death(old: &CommittedCredentials, new: &CommittedCredentials) -> bool {
114    old.euid != new.euid
115        || old.egid != new.egid
116        || old.fsuid != new.fsuid
117        || old.fsgid != new.fsgid
118        || new.permitted & !old.permitted != 0
119}
120
121#[cfg(test)]
122mod tests {
123    use super::*;
124
125    /// Each id `commit_creds` compares clears the signal when it changes, and
126    /// so does a capability the permitted set gains; a dropped capability or
127    /// no change keeps it.
128    #[test]
129    fn commit_creds_clears_for_a_moved_id_or_a_gained_capability() {
130        let old = CommittedCredentials::virtual_root(0b0110);
131        for (new, clears) in [
132            (old, false),
133            (CommittedCredentials { euid: 1, ..old }, true),
134            (CommittedCredentials { egid: 1, ..old }, true),
135            (CommittedCredentials { fsuid: 1, ..old }, true),
136            (CommittedCredentials { fsgid: 1, ..old }, true),
137            (
138                CommittedCredentials {
139                    permitted: 0b0111,
140                    ..old
141                },
142                true,
143            ),
144            (
145                CommittedCredentials {
146                    permitted: 0b0010,
147                    ..old
148                },
149                false,
150            ),
151            (
152                CommittedCredentials {
153                    permitted: 0b1010,
154                    ..old
155                },
156                true,
157            ),
158        ] {
159            assert_eq!(clears_parent_death(&old, &new), clears, "{new:?}");
160        }
161    }
162
163    /// A setter clears the parent-death signal of the virtual root exactly
164    /// where it moves an effective or filesystem id away from 0: a real or
165    /// saved id alone, -1, 0 and `setgroups` keep it.
166    #[test]
167    fn only_a_setter_that_moves_an_effective_or_filesystem_id_clears_the_signal() {
168        let keep = u32::MAX as u64;
169        let root = CommittedCredentials::virtual_root(u64::MAX >> 23);
170        for (sysno, args, clears) in [
171            (Sysno::setuid, [1000, 0, 0], true),
172            (Sysno::setuid, [0, 0, 0], false),
173            (Sysno::setuid, [keep, 0, 0], false),
174            (Sysno::setgid, [1000, 0, 0], true),
175            (Sysno::setfsuid, [1000, 0, 0], true),
176            (Sysno::setfsuid, [keep, 0, 0], false),
177            (Sysno::setfsgid, [1000, 0, 0], true),
178            (Sysno::setreuid, [keep, 1000, 0], true),
179            (Sysno::setreuid, [1000, keep, 0], false),
180            (Sysno::setregid, [keep, 1000, 0], true),
181            (Sysno::setregid, [1000, 0, 0], false),
182            (Sysno::setresuid, [keep, 1000, keep], true),
183            (Sysno::setresuid, [1000, keep, 1000], false),
184            (Sysno::setresgid, [keep, 1000, keep], true),
185            (Sysno::setresgid, [1000, keep, 1000], false),
186            (Sysno::setgroups, [1, 0x1000, 0], false),
187        ] {
188            let args = [args[0], args[1], args[2], 0, 0, 0];
189            let new = root.after_setter(sysno, args);
190            assert_eq!(new.permitted, root.permitted, "{sysno} {args:?}");
191            assert_eq!(clears_parent_death(&root, &new), clears, "{sysno} {args:?}");
192        }
193    }
