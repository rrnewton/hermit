20/// A mapping's length as the kernel records it: rounded up to the host's page.
21fn page_aligned_len(len: usize) -> usize {
22    if len == 0 {
23        return 0;
24    }
25    let page = usize::try_from(crate::page::page_size()).expect("the page size fits in usize");
26    len.checked_add(page - 1)
27        .expect("a successful memory range must fit in the address space")
28        & !(page - 1)
29}
30
31#[derive(Debug, Clone, Copy, Serialize, Deserialize)]
32struct SharedMapping {
33    len: usize,
34    object: SharedMemoryObjectId,
35    object_offset: u64,
36}
37
38impl SharedMapping {
39    fn end(self, start: usize) -> usize {
40        start
41            .checked_add(self.len)
42            .expect("a successful memory mapping must fit in the address space")
43    }
44
45    fn offset_at(self, start: usize, address: usize) -> u64 {
46        self.object_offset
47            .checked_add((address - start) as u64)
48            .expect("a mapped backing-object offset must fit in u64")
49    }
50}

52/// Shared mappings visible in one Linux memory address space.
53#[derive(Debug, Clone, Default, Serialize, Deserialize)]
54pub(crate) struct MemoryMetadata {
55    next_anonymous_sequence: u64,
56    shared_mappings: BTreeMap<usize, SharedMapping>,
57    /// The guest's program break when it was first observed, i.e. the base of
58    /// the brk heap.
59    #[serde(default)]
60    brk_start: Option<u64>,
61    /// The guest's most recently observed program break.
62    #[serde(default)]
63    brk_current: Option<u64>,
64    /// Detcore's model of the program break, where Detcore answers `brk(2)`
65    /// itself ([`BrkModel`]); `None` elsewhere, and where the image's layout
66    /// could not be read.
67    #[serde(default)]
68    pub(crate) brk_model: Option<BrkModel>,
69    /// Why there is no [`BrkModel`] where Detcore answers `brk(2)` itself
70    /// (`Detcore::handle_brk` refuses a brk by this name).
71    #[serde(default)]
72    pub(crate) brk_unmodeled: Option<String>,
73    /// Whether the address space has allocated or applied a protection key
74    /// (`pkey_alloc`, `pkey_mprotect`). Hermit's copies of guest memory do
75    /// not check protection keys, so where one may be in force they cannot
76    /// tell which pages the guest can write (`crate::syscalls::inotify`).
77    /// A fork copies the address space, keys included; an exec replaces it.
78    #[serde(default)]
79    pub(crate) protection_keys: bool,
80    /// aarch64: the file-backed ranges `crate::store_exclusive` has seen in
81    /// this address space, and whether each was scanned for exclusive pairs.
82    /// Not part of the serialized model.
83    #[cfg(target_arch = "aarch64")]
84    #[serde(skip)]
85    pub(crate) exclusive_scan: crate::store_exclusive::ScannedRanges,
86}

88/// Detcore's model of a process's program break: where Linux starts it for
89/// the image the process exec'd (`crate::exec_record::ElfLayout`) and where
90/// it stands now. Detcore keeps one, and answers `brk(2)` from it, only where
91/// the run declares a launcher, whose break the kernel's is
92/// (`Detcore::handle_brk`).
93#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
94pub(crate) struct BrkModel {
95    /// `mm->start_brk`.
96    pub(crate) start: u64,
97    /// `mm->brk`, as `brk(2)` last set it (not page-aligned).
98    pub(crate) current: u64,
99    /// `mm->end_data - mm->start_data`
100    /// (`crate::exec_record::ElfLayout::data_span`).
101    pub(crate) data_span: u64,
102    /// The kernel's break when the model was made, the launcher's: what a
103    /// libc that queried the break before Detcore's first callback holds
104    /// as the guest's (`Detcore::handle_brk`).
105    pub(crate) launcher_break: u64,
106}
107
108impl MemoryMetadata {
109    /// Create an empty address-space model.
110    pub(crate) fn new() -> Self {
111        Self::default()
112    }
113
114    /// Inspect the actual replacement model without erasing old observations.
115    pub(crate) fn is_fresh_for_native_exec(&self) -> bool {
116        self.next_anonymous_sequence == 0
117            && self.shared_mappings.is_empty()
118            && self.brk_start.is_none()
119            && self.brk_current.is_none()
120            && self.brk_model.is_none()
121            && self.brk_unmodeled.is_none()
122            && !self.protection_keys
123    }
124
125    /// Record a program break reported by a `brk(2)` return.
126    ///
127    /// Both `brk(NULL)` (a query) and `brk(addr)` (a request) return the break
128    /// in effect afterwards, so every successful return is a valid observation.
129    /// The first one seen fixes the heap base.
130    pub(crate) fn observe_brk(&mut self, brk: u64) {
131        if brk == 0 {
132            return;
133        }
134        self.brk_start.get_or_insert(brk);
135        self.brk_current = Some(brk);
136    }
137
138    /// The guest's brk heap as a half-open `[start, end)` guest-address range,
139    /// or `None` before the break has moved.
140    ///
141    /// This is the *observed* heap rather than the kernel's `[heap]` label.
142    /// The label only exists when the kernel's own program break is the guest's:
143    /// under a backend that loads the guest itself (DynamoRIO), `mm->start_brk`
144    /// belongs to the loader, the kernel tags nothing, and the guest's heap is
145    /// an ordinary anonymous mapping. Binding to the observed break keeps the
146    /// heap identifiable on every backend.
147    ///
148    /// Where Detcore models the break, the heap is the model's: its start is
149    /// known before any brk returns, which the first observation is not when
150    /// the guest's first brk already grows the break.
151    pub(crate) fn brk_heap_range(&self) -> Option<(u64, u64)> {
152        let (start, current) = match self.brk_model {
153            Some(model) => (model.start, model.current),
154            None => (self.brk_start?, self.brk_current?),
155        };
156        (current > start).then_some((start, current))
157    }
158
159    /// Resolve a futex address, falling back to a private key outside tracked shared mappings.
160    pub(crate) fn futex_id(&self, mm: MmId, address: usize) -> FutexID {
161        let Some((&start, &mapping)) = self.shared_mappings.range(..=address).next_back() else {
162            return FutexID::private(mm, address);
163        };
164        let Some(word_end) = address.checked_add(std::mem::size_of::<u32>()) else {
165            return FutexID::private(mm, address);
166        };
167        if word_end > mapping.end(start) {
168            return FutexID::private(mm, address);
169        }
170
171        FutexID::shared(mapping.object, mapping.offset_at(start, address))
172    }
173
174    /// Record a new anonymous shared mapping.
175    pub(crate) fn map_anonymous(&mut self, mm: MmId, start: usize, len: usize) {
176        let object = SharedMemoryObjectId::Anonymous {
177            origin: mm,
178            sequence: self.next_anonymous_sequence,
179        };
180        self.next_anonymous_sequence = self
181            .next_anonymous_sequence
182            .checked_add(1)
183            .expect("anonymous shared mapping sequence exhausted");
184        self.insert_mapping(start, len, object, 0);
185    }
186
187    /// Record a mapping with a resolved backing object.
188    pub(crate) fn map_object(
189        &mut self,
190        start: usize,
191        len: usize,
192        object: SharedMemoryObjectId,
193        object_offset: u64,
194    ) {
195        self.insert_mapping(start, len, object, object_offset);
196    }
197
198    fn insert_mapping(
199        &mut self,
200        start: usize,
201        len: usize,
202        object: SharedMemoryObjectId,
203        object_offset: u64,
204    ) {
205        let len = page_aligned_len(len);
206        if len == 0 {
207            return;
208        }
209        start
210            .checked_add(len)
211            .expect("a successful memory mapping must fit in the address space");
212        self.unmap(start, len);
213        self.shared_mappings.insert(
214            start,
215            SharedMapping {
216                len,
217                object,
218                object_offset,
219            },
220        );
221    }
222
223    /// Remove a range, retaining any mapped portions on either side.
224    pub(crate) fn unmap(&mut self, start: usize, len: usize) {
225        let len = page_aligned_len(len);
226        if len == 0 {
227            return;
228        }
229        let end = start
230            .checked_add(len)
231            .expect("a successful memory range operation must fit in the address space");
232        let overlapping = self
233            .shared_mappings
234            .range(..end)
235            .filter_map(|(&mapping_start, &mapping)| {
236                (mapping.end(mapping_start) > start).then_some((mapping_start, mapping))
237            })
238            .collect::<Vec<_>>();
239
240        for (mapping_start, mapping) in overlapping {
241            self.shared_mappings.remove(&mapping_start);
242            let mapping_end = mapping.end(mapping_start);
243            if mapping_start < start {
244                self.shared_mappings.insert(
245                    mapping_start,
246                    SharedMapping {
247                        len: start - mapping_start,
248                        ..mapping
249                    },
250                );
251            }
252            if mapping_end > end {
253                self.shared_mappings.insert(
254                    end,
255                    SharedMapping {
256                        len: mapping_end - end,
257                        object_offset: mapping.offset_at(mapping_start, end),
258                        ..mapping
259                    },
260                );
261            }
262        }
263    }
264
265    /// Move or resize a mapping after a successful `mremap`.
266    /// Record a successful `mremap` from `old_start` to `new_start`. The new
267    /// range keeps the source's backing object and offset. The source range
268    /// is dropped unless `keep_source` (`MREMAP_DONTUNMAP`), with which Linux
269    /// leaves the old mapping in place (mm/mremap.c), so both ranges then
270    /// alias the same object.
271    pub(crate) fn remap(
272        &mut self,
273        old_start: usize,
274        old_len: usize,
275        new_start: usize,
276        new_len: usize,
277        keep_source: bool,
278    ) {
279        let old_len = page_aligned_len(old_len);
280        let new_len = page_aligned_len(new_len);
281        let old_end = old_start
282            .checked_add(old_len)
283            .expect("a successful mremap source must fit in the address space");
284        let source = self
285            .shared_mappings
286            .range(..=old_start)
287            .next_back()
288            .and_then(|(&mapping_start, &mapping)| {
289                (mapping.end(mapping_start) >= old_end)
290                    .then_some((mapping.object, mapping.offset_at(mapping_start, old_start)))
291            });
292
293        if !keep_source {
294            self.unmap(old_start, old_len);
295        }
296        self.unmap(new_start, new_len);
297        if let Some((object, object_offset)) = source {
298            self.insert_mapping(new_start, new_len, object, object_offset);
299        }
300    }
301}

307    #[test]
308    fn native_exec_fresh_memory_requires_sequence_mappings_and_brk_to_be_empty() {
309        let owner = crate::types::DetTid::from_raw(42);
310        let mut memory = MemoryMetadata::new();
311        assert!(memory.is_fresh_for_native_exec());
312        memory.map_anonymous(MmId::initial(owner), 0x5000, 0x1000);
313        assert!(!memory.is_fresh_for_native_exec());
314        memory.unmap(0x5000, 0x1000);
315        assert!(memory.shared_mappings.is_empty());
316        assert!(
317            !memory.is_fresh_for_native_exec(),
318            "the old allocator sequence remains"
319        );
320        let mut brk = MemoryMetadata::new();
321        brk.observe_brk(0x405000);
322        assert_eq!(brk.brk_heap_range(), None);
323        assert!(
324            !brk.is_fresh_for_native_exec(),
325            "even an unexpanded old heap has observations"
326        );
327        assert!(MemoryMetadata::new().is_fresh_for_native_exec());
328    }

330    #[test]
331    fn brk_range_tracks_the_observed_program_break() {
332        let mut mm = MemoryMetadata::new();
333        // Nothing observed yet: no heap to report.
334        assert_eq!(mm.brk_heap_range(), None);
335
336        // The first observation fixes the base. A heap that has not grown is
337        // empty, not a zero-length record.
338        mm.observe_brk(0x405000);
339        assert_eq!(mm.brk_heap_range(), None);
340
341        // Growth is reported as [start, current).
342        mm.observe_brk(0x426000);
343        assert_eq!(mm.brk_heap_range(), Some((0x405000, 0x426000)));
344        mm.observe_brk(0x447000);
345        assert_eq!(mm.brk_heap_range(), Some((0x405000, 0x447000)));
346
347        // A repeated brk(NULL) query must not move the base.
348        mm.observe_brk(0x447000);
349        assert_eq!(mm.brk_heap_range(), Some((0x405000, 0x447000)));
350
351        // Shrinking the break is legal and narrows the heap; the base is sticky.
352        mm.observe_brk(0x410000);
353        assert_eq!(mm.brk_heap_range(), Some((0x405000, 0x410000)));
354
355        // Back to the base means an empty heap again, not an inverted range.
356        mm.observe_brk(0x405000);
357        assert_eq!(mm.brk_heap_range(), None);
358    }
359
360    #[test]
361    fn a_modeled_heap_is_the_models_from_the_first_growth() {
362        // A guest whose first brk grows the break: the observations alone
363        // would take its result as the base and report no heap.
364        let mut mm = MemoryMetadata::new();
365        mm.brk_model = Some(BrkModel {
366            start: 0x405000,
367            current: 0x426000,
368            data_span: 0x1000,
369            launcher_break: 0x555555580000,
370        });
371        mm.observe_brk(0x426000);
372        assert_eq!(mm.brk_heap_range(), Some((0x405000, 0x426000)));
373        mm.brk_model = Some(BrkModel {
374            current: 0x405000,
375            ..mm.brk_model.unwrap()
376        });
377        assert_eq!(mm.brk_heap_range(), None);
378    }
379
380    #[test]
381    fn a_zero_break_is_ignored() {
382        // brk returning 0 is not a usable address; recording it would make the
383        // heap base 0 and hash a wild range.
384        let mut mm = MemoryMetadata::new();
385        mm.observe_brk(0);
386        assert_eq!(mm.brk_heap_range(), None);
387        mm.observe_brk(0x405000);
388        mm.observe_brk(0);
389        mm.observe_brk(0x426000);
390        assert_eq!(mm.brk_heap_range(), Some((0x405000, 0x426000)));
391    }

403    #[test]
404    fn file_mappings_alias_by_backing_offset() {
405        let mut mappings = MemoryMetadata::new();
406        let object = SharedMemoryObjectId::File {
407            device: 1,
408            inode: 2,
409        };
410        mappings.map_object(0x1000, 0x1000, object, 0);
411        mappings.map_object(0x4000, 0x1000, object, 0);
412        mappings.map_object(0x8000, 0x1000, object, 0x1000);
413
414        assert_eq!(
415            mappings.futex_id(mm(10), 0x1010),
416            mappings.futex_id(mm(11), 0x4010),
417            "aliases of one file offset must share a futex key"
418        );
419        assert_ne!(
420            mappings.futex_id(mm(10), 0x1010),
421            mappings.futex_id(mm(10), 0x8010),
422            "different file offsets must not alias"
423        );
424        assert!(
425            matches!(mappings.futex_id(mm(10), 0x1ffc), FutexID::Shared { .. }),
426            "mmap lengths must be rounded to the kernel's page boundary"
427        );
428    }

430    #[test]
431    fn forked_anonymous_mapping_retains_identity() {
432        let mut parent = MemoryMetadata::new();
433        parent.map_anonymous(mm(10), 0x1000, 0x1000);
434        let mut child = parent.clone();
435
436        assert_eq!(
437            parent.futex_id(mm(10), 0x1010),
438            child.futex_id(mm(11), 0x1010),

441        child.map_anonymous(mm(11), 0x4000, 0x1000);
442        assert_ne!(

444            child.futex_id(mm(11), 0x4010),

449    #[test]
450    fn unmap_and_remap_preserve_only_live_aliases() {
451        let mut mappings = MemoryMetadata::new();
452        mappings.map_anonymous(mm(10), 0x1000, 0x3000);
453        let original = mappings.futex_id(mm(10), 0x2010);
454
455        mappings.unmap(0x2000, 0x1000);
456        assert_eq!(
457            mappings.futex_id(mm(10), 0x2010),
458            FutexID::private(mm(10), 0x2010),
459            "an unmapped word must no longer resolve through its old object"
460        );
461        assert!(
462            matches!(mappings.futex_id(mm(10), 0x3010), FutexID::Shared { .. }),
463            "the right-hand mapping fragment must retain its shared identity"
464        );
465
466        mappings.map_anonymous(mm(10), 0x5000, 0x1000);
467        let before_remap = mappings.futex_id(mm(10), 0x5010);
468        mappings.remap(0x5000, 0x1000, 0x9000, 0x1000, false);
469        assert_eq!(
470            before_remap,
471            mappings.futex_id(mm(10), 0x9010),
472            "mremap must retain the backing-object offset"
473        );
474        assert_ne!(original, before_remap);
475    }

477    /// `MREMAP_DONTUNMAP` leaves the old mapping in place, so the old and new
478    /// ranges both resolve to the same backing object
479    /// (https://github.com/rrnewton/hermit/pull/4020).
480    #[test]
481    fn a_dontunmap_remap_keeps_both_aliases_of_the_object() {
482        let mut mappings = MemoryMetadata::new();
483        mappings.map_anonymous(mm(10), 0x5000, 0x1000);
484        let before = mappings.futex_id(mm(10), 0x5010);
485        mappings.remap(0x5000, 0x1000, 0x9000, 0x1000, true);
486        assert!(matches!(before, FutexID::Shared { .. }));
487        assert_eq!(mappings.futex_id(mm(10), 0x5010), before);
488        assert_eq!(mappings.futex_id(mm(10), 0x9010), before);
489    }
