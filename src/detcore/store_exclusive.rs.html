9//! Store-exclusive sequences on aarch64: detection only.
10//!
11//! An aarch64 load-exclusive/store-exclusive loop (`LDAXR` .. `STLXR`, then
12//! `CBNZ` back) runs once more whenever the local exclusive monitor is cleared
13//! between its load and its store, and any exception clears it: a host
14//! interrupt, a PMU overflow, a ptrace stop, a single step. Each retry adds
15//! retired branches, which Detcore counts to place preemptions and to advance
16//! virtual time, so a program that runs such loops can be preempted at other
17//! points in two runs, and a recorded schedule may not replay. LSE atomics
18//! (`LDADD`, `CAS`, `SWP`; Armv8.1-A) are single instructions and never retry.
19//!
20//! This module reports and changes nothing a guest sees:
21//!
22//! - The exclusive pairs in each executable file mapping, counted after exec
23//!   and after an `mmap` or `mprotect` that maps `PROT_EXEC` from a file. The
24//!   count is a function of the file's bytes, so it goes to INFO and to the
25//!   run summary's INFO view.
26//! - In a run that records or replays preemptions, the store-exclusives the
27//!   guest's tasks failed (`STREX_FAIL_SPEC`) and, where the core counts
28//!   them, executed (`STREX_SPEC`), counted from the start of the root
29//!   task's first image on every task it creates and read once at the end of
30//!   the run. Host interrupts decide them, so they never reach INFO or the
31//!   INFO view: only the full report, the summary JSON, the preemption
32//!   recording and that run's warnings carry them.
33//!
34//! The decoder and the model are pure and compile everywhere; the rest runs
35//! only on aarch64.
36

45    /// An instruction of the load/store exclusive class.
46    #[derive(Clone, Copy, Debug, PartialEq, Eq)]
47    pub(crate) enum Exclusive {
48        /// `LDXR`, `LDAXR`, `LDXP`, `LDAXP` and their byte and halfword forms.
49        Load,
50        /// `STXR`, `STLXR`, `STXP`, `STLXP` and their byte and halfword forms.
51        Store,
52    }
53
54    /// Bits 29:24 (`001000`) and 23 (`o2`) of the load/store exclusive class.
55    /// `o2 = 1` is load-acquire/store-release (`LDAR`, `STLR`) or `CAS`,
56    /// neither of which can fail.
57    const CLASS_MASK: u32 = 0x3F80_0000;
58    const CLASS: u32 = 0x0800_0000;
59    /// `L`: a load.
60    const LOAD: u32 = 1 << 22;
61    /// `o1`: a register pair.
62    const PAIR: u32 = 1 << 21;
63    /// The high size bit, which every exclusive pair sets. With `o1 = 1` and
64    /// this bit clear the encoding is `CASP` (Armv8.1-A), a single
65    /// instruction.
66    const PAIR_SIZE: u32 = 1 << 31;
67
68    /// Whether `insn` is a load-exclusive, a store-exclusive, or neither.
69    pub(crate) const fn classify(insn: u32) -> Option<Exclusive> {
70        if insn & CLASS_MASK != CLASS || (insn & PAIR != 0 && insn & PAIR_SIZE == 0) {
71            None
72        } else if insn & LOAD != 0 {
73            Some(Exclusive::Load)
74        } else {
75            Some(Exclusive::Store)
76        }
77    }

79    /// A store-exclusive is one of a pair when a load-exclusive is at most
80    /// this many instructions before it.
81    pub(crate) const PAIR_WINDOW: u64 = 16;
82
83    /// `MRS <Xt>, <system register>` with the system register fixed and only
84    /// `Rt` (bits 4:0) free.
85    const MRS_MASK: u32 = !0x1F;
86    /// The virtual counter, read from EL0: `MRS Xt, CNTVCT_EL0` and its
87    /// self-synchronized form `CNTVCTSS_EL0`. Linux lets every task read it
88    /// directly (or, on CPUs with a counter erratum, traps and emulates the
89    /// read in the kernel), so the guest gets the host's count with no stop
90    /// Hermit sees and nothing Detcore can virtualize. It is this
91    /// architecture's `RDTSC`, without `PR_SET_TSC` to trap it.
92    ///
93    /// Not counted: `CNTPCT_EL0`, `CNTPCTSS_EL0` and the timer registers
94    /// (`CNTV_TVAL_EL0` and the like), whose EL0 access Linux disables, so a
95    /// read faults with `SIGILL`, which is visible and deterministic; and
96    /// `CNTFRQ_EL0`, a constant frequency rather than a time.
97    const COUNTER_READS: [u32; 2] = [0xD53B_E040, 0xD53B_E0C0];
98
99    /// Whether `insn` reads the virtual counter from EL0.
100    pub(crate) fn is_counter_read(insn: u32) -> bool {
101        COUNTER_READS.contains(&(insn & MRS_MASK))
102    }

104    /// What one pass over a code range finds: the exclusive pairs, whose
105    /// retries make a branch count depend on host interrupts, and the EL0
106    /// counter reads, which bypass virtual time outright.
107    #[derive(Clone, Debug, Default)]
108    pub(crate) struct CodeScan {
109        /// The file offset of the first word.
110        base: u64,
111        index: u64,
112        last_load: Option<u64>,
113        /// The file offset of each pair's load-exclusive.
114        pair_loads: Vec<u64>,
115        /// The file offset just past each pair's retry branch, the word after
116        /// its store-exclusive: with `pair_loads`, the bytes the pair covers.
117        pair_ends: Vec<u64>,
118        /// The file offset of each counter read.
119        reads: Vec<u64>,
120    }
121
122    impl CodeScan {
123        /// A scan of words that start at file offset `base`.
124        pub(crate) fn at(base: u64) -> Self {
125            CodeScan {
126                base,
127                ..CodeScan::default()
128            }
129        }
130
131        /// Take the next instruction word.
132        pub(crate) fn push(&mut self, insn: u32) {
133            match classify(insn) {
134                Some(Exclusive::Load) => self.last_load = Some(self.index),
135                Some(Exclusive::Store) => {
136                    if let Some(load) = self
137                        .last_load
138                        .filter(|load| self.index - load <= PAIR_WINDOW)
139                    {
140                        self.pair_loads.push(self.base + 4 * load);
141                        self.pair_ends.push(self.base + 4 * (self.index + 2));
142                    }
143                }
144                _ if is_counter_read(insn) => self.reads.push(self.base + 4 * self.index),
145                _ => {}
146            }
147            self.index += 1;
148        }
149
150        /// Take little-endian instruction words, as aarch64 stores code in
151        /// either data endianness. A trailing partial word is ignored, so a
152        /// stream split into several calls must split at multiples of four.
153        pub(crate) fn push_bytes(&mut self, bytes: &[u8]) {
154            for word in bytes.as_chunks::<4>().0 {
155                self.push(u32::from_le_bytes(*word));
156            }
157        }
158
159        /// The pairs counted so far.
160        #[cfg(test)]
161        pub(crate) fn pairs(&self) -> u64 {
162            self.pair_loads.len() as u64
163        }
164
165        /// The EL0 counter reads counted so far.
166        #[cfg(test)]
167        pub(crate) fn counter_reads(&self) -> u64 {
168            self.reads.len() as u64
169        }
170
171        /// What the scan found, by file offset.
172        pub(crate) fn into_loops(self) -> super::model::FileLoops {
173            super::model::FileLoops {
174                pair_loads: self.pair_loads,
175                pair_ends: self.pair_ends,
176                counter_reads: self.reads,
177                unverified: Vec::new(),
178            }
179        }
180    }

182    /// Where an ELF64 little-endian file's section header table is: its
183    /// offset and entry count, from the 64-byte file header. `None` for any
184    /// other file, or one without section headers (or more than fit in
185    /// `e_shnum`), whose mapped bytes are then scanned whole.
186    pub(crate) fn elf_section_table(header: &[u8; 64]) -> Option<(u64, usize)> {
187        const SECTION_HEADER_SIZE: u16 = 64;
188        let half = |at: usize| u16::from_le_bytes([header[at], header[at + 1]]);
189        let offset = u64::from_le_bytes(header[0x28..0x30].try_into().unwrap());
190        let (entry_size, count) = (half(0x3A), half(0x3C));
191        // `\x7fELF`, ELFCLASS64, ELFDATA2LSB.
192        (header[..6] == *b"\x7fELF\x02\x01"
193            && offset != 0
194            && count != 0
195            && entry_size == SECTION_HEADER_SIZE)
196            .then_some((offset, usize::from(count)))
197    }

204        pub(crate) index: usize,

214    /// The sections that hold instructions (`SHF_EXECINSTR`, with file
215    /// contents) in an ELF64 little-endian section header table. Only these
216    /// are scanned, so read-only data that shares the executable segment
217    /// cannot pass for an exclusive pair.
218    pub(crate) fn executable_sections(table: &[u8]) -> Vec<ExecutableSection> {
219        const SHT_NOBITS: u32 = 8;
220        const SHF_EXECINSTR: u64 = 0x4;
221        table
222            .as_chunks::<64>()
223            .0
224            .iter()
225            .enumerate()
226            .filter_map(|(index, entry)| {
227                let word = |at: usize| u64::from_le_bytes(entry[at..at + 8].try_into().unwrap());
228                let kind = u32::from_le_bytes(entry[4..8].try_into().unwrap());
229                (kind != SHT_NOBITS && word(8) & SHF_EXECINSTR != 0 && word(32) != 0).then(|| {
230                    ExecutableSection {
231                        index,
232                        kind,
233                        offset: word(24),
234                        size: word(32),
235                        address: word(16),
236                    }
237                })
238            })
239            .collect()
240    }

242    /// File-offset ranges `[start, end)`, in file order.

245    /// The file ranges (start, end) of `sections` to scan, and of those the
246    /// ones scanned without proof, each sorted and merged. Only `SHT_PROGBITS`
247    /// sections admit `$x` and `$d` proof, as the default rewrite takes them
248    /// ([`CodeEvidence::classify`]): what a section's mapping symbols mark
249    /// `$x` is code, and what they mark `$d` is data and is not scanned, so a
250    /// constant table in `.text` cannot pass for an exclusive pair. The rest
251    /// is scanned and labelled unverified: bytes before a section's first
252    /// mapping symbol, and the whole of a section that no mapping symbol
253    /// marks (whatever its unwind table says), that has another section type,
254    /// or that holds a mapping symbol of another kind
255    /// ([`CodeEvidence::unsupported_in`]). The rewriter leaves those
256    /// alone by default, but this census only names where retries may come
257    /// from, and dropping them would hide a real loop.
258    ///
259    /// A section header is taken as the file states it. One whose offset
260    /// plus size runs past the end of the address space saturates there, so
261    /// its ranges end at `u64::MAX`, beyond the end of any file, where the
262    /// caller's clip to the file's size drops them: a malformed header skips
263    /// the bytes it would place past the file, as a header placing a section
264    /// past the end of the file already does, and never panics.
265    ///
266    /// [`CodeEvidence::classify`]: crate::aarch64_lse::mapping::CodeEvidence::classify
267    /// [`CodeEvidence::unsupported_in`]: crate::aarch64_lse::mapping::CodeEvidence::unsupported_in
268    pub(crate) fn code_file_ranges(
269        sections: &[ExecutableSection],
270        evidence: &crate::aarch64_lse::mapping::CodeEvidence,
271    ) -> (FileRanges, FileRanges) {
272        use crate::aarch64_lse::mapping;
273        let mut ranges = Vec::new();
274        let mut unverified = Vec::new();
275        for section in sections {
276            let end = section.address.saturating_add(section.size);
277            let held = evidence.classify(section.index, section.address, section.size);
278            let (code, unknown) = if section.kind == goblin::elf::section_header::SHT_PROGBITS
279                && held.evidence == Some(mapping::Evidence::MappingSymbols)
280                && evidence.unsupported_in(section.index).is_none()
281            {
282                (held.code, held.unknown)
283            } else {
284                (Vec::new(), vec![(section.address, end)])
285            };
286            // The header's offset, size and address are the file's own: a
287            // malformed one saturates rather than overflows, and the caller
288            // drops what lies past the end of the file.
289            let file = |(start, end): &(u64, u64)| {
290                (
291                    section
292                        .offset
293                        .saturating_add(start.saturating_sub(section.address)),
294                    section
295                        .offset
296                        .saturating_add(end.saturating_sub(section.address)),
297                )
298            };
299            ranges.extend(code.iter().chain(&unknown).map(file));
300            unverified.extend(unknown.iter().map(file));
301        }
302        let merged = |mut spans: Vec<(u64, u64)>| {
303            spans.sort_unstable();
304            let mut out: Vec<(u64, u64)> = Vec::new();
305            for (start, end) in spans {
306                match out.last_mut() {
307                    Some(last) if start <= last.1 => last.1 = last.1.max(end),
308                    _ => out.push((start, end)),
309                }
310            }
311            out
312        };
313        (merged(ranges), merged(unverified))
314    }

316    /// What `ranges` of a file hold, each scanned from its start with its
317    /// bytes from `read`, with the `unverified` ranges beside them, as
318    /// [`code_file_ranges`] returns both. Ranges that touch arrive merged,
319    /// so a pair can take its load from proved code and its store from the
320    /// unproved bytes after it; [`super::model::FileLoops::found_in`]
321    /// labels such a pair unverified by every word it covers.
322    pub(crate) fn scan_ranges<E>(
323        ranges: &[(u64, u64)],
324        unverified: FileRanges,
325        mut read: impl FnMut(u64, u64, &mut CodeScan) -> Result<(), E>,
326    ) -> Result<super::model::FileLoops, E> {
327        let mut loops = super::model::FileLoops {
328            unverified,
329            ..Default::default()
330        };
331        for &(start, end) in ranges {
332            let mut scan = CodeScan::at(start);
333            read(start, end, &mut scan)?;
334            let found = scan.into_loops();
335            loops.pair_loads.extend(found.pair_loads);
336            loops.pair_ends.extend(found.pair_ends);
337            loops.counter_reads.extend(found.counter_reads);
338        }
339        Ok(loops)
340    }

387    /// Parse a kernel CPU list such as `0-3,8,10-11`.
388    pub(crate) fn parse_cpu_list(list: &str) -> Option<Vec<usize>> {
389        let mut cpus = Vec::new();
390        for part in list.trim().split(',').filter(|part| !part.is_empty()) {
391            match part.split_once('-') {
392                Some((first, last)) => {
393                    let (first, last): (usize, usize) = (first.parse().ok()?, last.parse().ok()?);
394                    if first > last {
395                        return None;
396                    }
397                    cpus.extend(first..=last);
398                }
399                None => cpus.push(part.parse().ok()?),
400            }
401        }
402        Some(cpus)
403    }

405    /// The Arm CPU PMUs among the entries `names` of `devices`
406    /// (`/sys/bus/event_source/devices`): each named `arm*` with a `type` and
407    /// a `cpus` file. A host with one kind of core has one; a host with
408    /// several kinds (big.LITTLE) has one per kind, and each counts only on
409    /// its own CPUs. Sorted by name.
410    pub(crate) fn arm_cpu_pmus(
411        devices: &Path,
412        names: impl IntoIterator<Item = String>,
413    ) -> Vec<CpuPmu> {
414        let mut pmus: Vec<CpuPmu> = names
415            .into_iter()
416            .filter(|name| name.starts_with("arm"))
417            .filter_map(|name| {
418                let read = |file| std::fs::read_to_string(devices.join(&name).join(file)).ok();
419                let kind = read("type")?.trim().parse().ok()?;
420                let cpus = parse_cpu_list(&read("cpus")?)?;
421                Some(CpuPmu { name, kind, cpus })
422            })
423            .collect();
424        pmus.sort_by(|a, b| a.name.cmp(&b.name));
425        pmus
426    }

428    /// The PMUs that count on at least one of `cpus`, each with the first
429    /// such CPU.

432            .filter_map(|pmu| {
433                let cpu = pmu.cpus.iter().find(|cpu| cpus.contains(cpu))?;

439    /// One event's reads on each PMU, as (value, time enabled, time running),
440    /// combined: the sum of the values, whether the event was on a counter
441    /// for the whole run, and for how much of it, in thousandths of the time
442    /// enabled (1000 for the whole run); or `None` when it was enabled but
443    /// never on a counter (a core that accepts the event and does not count
444    /// it). A task-bound event on one PMU of several is enabled whenever the
445    /// task runs, but runs only while the task is on that PMU's CPUs, so the
446    /// PMUs' times running add up to the time enabled when nothing was
447    /// multiplexed.
448    pub(crate) fn combine(reads: &[(u64, u64, u64)]) -> Option<(u64, bool, u32)> {
449        let value = reads.iter().map(|read| read.0).sum();
450        let enabled = reads.iter().map(|read| read.1).max().unwrap_or(0);
451        let running: u64 = reads.iter().map(|read| read.2).sum();
452        if enabled > 0 && running == 0 {
453            return None;
454        }
455        let whole = running >= enabled;
456        // Rounded down, so a run that missed any time reads below 1000.
457        let per_mille = if whole {
458            1000
459        } else {
460            u32::try_from(u128::from(running) * 1000 / u128::from(enabled)).unwrap_or(999)
461        };
462        Some((value, whole, per_mille))
463    }

634    /// Where the file bytes `[offset, offset + len)` fall in a mapping of the
635    /// file range `[map_offset, map_offset + map_len)` that starts at guest
636    /// address `map_start`.
637    pub(crate) fn place(
638        offset: u64,
639        len: u64,
640        map_start: u64,
641        map_offset: u64,
642        map_len: u64,
643    ) -> Placement {
644        let map_end = map_offset.saturating_add(map_len);
645        if offset < map_offset || offset >= map_end {
646            Placement::Outside
647        } else if offset.saturating_add(len) > map_end {
648            Placement::CrossesEnd
649        } else {
650            Placement::At(map_start + (offset - map_offset))
651        }

697    /// when the scan that `trigger` made over `range` finds it: only at the
698    /// exec stop, and inside the range a successful `mmap` just made (decided
699    /// per sequence, since the kernel may merge a new mapping with an older

711            Trigger::Mapped => range.is_some_and(|(from, to)| from <= start && end <= to),

719    /// mixture of the two.

724    /// File ranges (start, end) in order with overlapping ones merged, so
725    /// that bytes two section headers both claim are scanned, and counted,
726    /// once. Ranges that only touch stay apart: a pair cannot span two
727    /// sections.
728    pub(crate) fn merge_overlaps(mut ranges: Vec<(u64, u64)>) -> Vec<(u64, u64)> {
729        ranges.sort_unstable();
730        let mut merged: Vec<(u64, u64)> = Vec::with_capacity(ranges.len());
731        for (start, end) in ranges {
732            match merged.last_mut() {
733                Some(last) if start < last.1 => last.1 = last.1.max(end),
734                _ => merged.push((start, end)),
735            }
736        }
737        merged
738    }

751    /// An upper bound on one B-tree node's allocation, for the
752    /// [`ByteCache`] key map and a plan's report maps. A node holds up to 11
753    /// keys, 11 values and (inside the tree) 12 child pointers: for a 16-byte
754    /// key and a 24-byte value that is under 560 bytes. A B-tree never has
755    /// more nodes than entries (no node is empty), so this bounds each entry's
756    /// share however sparsely the nodes are filled.
757    const MAP_NODE_BOUND: usize = 1024;

759    /// What one [`ByteCache`] entry's own allocation costs beyond its bytes
760    /// and its value's heap: the entry with its two reference counts.
761    pub(crate) const fn cached_entry_overhead<V>() -> usize {
762        2 * std::mem::size_of::<usize>() + std::mem::size_of::<Cached<V>>()
763    }

765    /// An upper bound on what a [`ByteCache`] of at most `max_entries`
766    /// retains besides its entries, whatever was inserted, evicted or
767    /// cleared: the age queue, whose capacity never exceeds twice the most
768    /// entries it held (it grows by doubling, and entries are evicted before
769    /// one is added), at most one key-map node per entry, and each key's
770    /// candidate list, allocated for the most candidates.
771    pub(crate) const fn spine_bound<V>(max_entries: usize) -> usize {
772        let slots = if max_entries < 2 {
773            4
774        } else {
775            max_entries.saturating_mul(2)
776        };
777        spine_of::<V>(slots, max_entries)
778    }

780    /// The spine of a [`ByteCache`] whose age queue has room for `slots`
781    /// and whose key map has `keys` keys: the queue's slots, a node per key
782    /// (a bound: there are never more nodes than keys) and each key's
783    /// candidate list. Saturating, so an absurd size cannot wrap.
784    const fn spine_of<V>(slots: usize, keys: usize) -> usize {
785        use std::mem::size_of;
786        let per_key = MAP_NODE_BOUND
787            .saturating_add(MAX_CANDIDATES_PER_KEY.saturating_mul(size_of::<Arc<Cached<V>>>()));
788        slots
789            .saturating_mul(size_of::<(CacheKey, Arc<Cached<V>>)>())
790            .saturating_add(keys.saturating_mul(per_key))
791    }

793    /// The most entries [`ByteCache`] compares bytes with for one key.
794    pub(crate) const MAX_CANDIDATES_PER_KEY: usize = 4;

796    /// The heap a cached plan holds, from the capacities of everything it
797    /// allocated, as [`ByteCache`] charges it.
798    pub(crate) fn plan_charge(plan: &Option<crate::aarch64_lse::elf::FilePlan>) -> usize {
799        use std::mem::size_of;
800
801        use crate::aarch64_lse::elf::FilePatch;
802        // The plan in its shared allocation, with the reference counts.
803        let mut charge =
804            2 * size_of::<usize>() + size_of::<Option<crate::aarch64_lse::elf::FilePlan>>();
805        let Some(plan) = plan else {
806            return charge;
807        };
808        let add = |charge: &mut usize, more: usize| *charge = charge.saturating_add(more);
809        add(
810            &mut charge,
811            plan.patches
812                .capacity()
813                .saturating_mul(size_of::<FilePatch>()),
814        );
815        for patch in &plan.patches {
816            add(&mut charge, patch.original.capacity());
817            add(&mut charge, patch.rewritten.capacity());
818        }
819        let report = &plan.report;
820        add(
821            &mut charge,
822            report
823                .refused_sites
824                .capacity()
825                .saturating_mul(size_of::<(u64, String)>()),
826        );
827        for (_, why) in &report.refused_sites {
828            add(&mut charge, why.capacity());
829        }
830        add(
831            &mut charge,
832            report
833                .skipped_segments
834                .capacity()
835                .saturating_mul(size_of::<String>()),
836        );
837        for why in &report.skipped_segments {
838            add(&mut charge, why.capacity());
839        }
840        add(
841            &mut charge,
842            (report.patched.len() + report.refused.len()).saturating_mul(MAP_NODE_BOUND),
843        );
844        charge
845    }

847    /// Values made from byte strings, found by those bytes: an entry is used
848    /// only for bytes equal to its own, so a value that is a pure function of
849    /// its bytes comes back exactly as making it again would. The key (the
850    /// length and a hash of three sampled 4 KiB pages) only picks the
851    /// candidates, at most [`MAX_CANDIDATES_PER_KEY`] of them. Everything it
852    /// retains (the entries with their bytes and their values' heap, and the
853    /// spine bounded by [`spine_bound`]) stays within the `max_retained` it
854    /// is made with, and the entries within `max_entries`: the oldest go
855    /// before a new one is added.
856    pub(crate) struct ByteCache<V> {
857        by_key: BTreeMap<CacheKey, Vec<Arc<Cached<V>>>>,
858        order: VecDeque<(CacheKey, Arc<Cached<V>>)>,
859        held: usize,
860        peak: usize,
861        max_charge: usize,
862        max_entries: usize,
863    }
864
865    impl<V: Clone> ByteCache<V> {
866        pub(crate) const fn new(max_retained: usize, max_entries: usize) -> Self {
867            ByteCache {
868                by_key: BTreeMap::new(),
869                order: VecDeque::new(),
870                held: 0,
871                peak: 0,
872                // The entries get what the spine cannot take.
873                max_charge: max_retained.saturating_sub(spine_bound::<V>(max_entries)),
874                max_entries,
875            }
876        }
877
878        fn key(bytes: &[u8]) -> CacheKey {
879            let mut hasher = std::hash::DefaultHasher::new();
880            let sample = 4096.min(bytes.len());
881            let middle = (bytes.len() - sample) / 2;
882            for part in [
883                &bytes[..sample],
884                &bytes[middle..middle + sample],
885                &bytes[bytes.len() - sample..],
886            ] {
887                std::hash::Hasher::write(&mut hasher, part);
888            }
889            (bytes.len(), std::hash::Hasher::finish(&hasher))
890        }
891
892        /// The value made from bytes equal to `bytes`, if held.
893        pub(crate) fn get(&self, bytes: &[u8]) -> Option<V> {
894            self.by_key
895                .get(&Self::key(bytes))?
896                .iter()
897                .find(|cached| *cached.bytes == *bytes)
898                .map(|cached| cached.value.clone())
899        }
900
901        /// Hold `value`, made from `bytes`, charged the bytes, `value_charge`
902        /// (its own heap) and the entry's bookkeeping, after the oldest
903        /// entries make room. Not held when its key already has
904        /// [`MAX_CANDIDATES_PER_KEY`] candidates, or when it alone is over
905        /// the bound.
906        pub(crate) fn insert(&mut self, bytes: Vec<u8>, value: V, value_charge: usize) {
907            let key = Self::key(&bytes);
908            let charge = bytes
909                .len()
910                .saturating_add(value_charge)
911                .saturating_add(cached_entry_overhead::<V>());
912            if charge > self.max_charge
913                || self.max_entries == 0
914                || self
915                    .by_key
916                    .get(&key)
917                    .is_some_and(|candidates| candidates.len() >= MAX_CANDIDATES_PER_KEY)
918            {
919                return;
920            }
921            while self.held.saturating_add(charge) > self.max_charge
922                || self.order.len() >= self.max_entries
923            {
924                let Some((key, oldest)) = self.order.pop_front() else {
925                    break;
926                };
927                self.held -= oldest.charge;
928                if let Some(candidates) = self.by_key.get_mut(&key) {
929                    candidates.retain(|candidate| !Arc::ptr_eq(candidate, &oldest));
930                    if candidates.is_empty() {
931                        self.by_key.remove(&key);
932                    }
933                }
934            }
935            let cached = Arc::new(Cached {
936                bytes: bytes.into_boxed_slice(),
937                value,
938                charge,
939            });
940            self.held += charge;
941            self.by_key
942                .entry(key)
943                .or_insert_with(|| Vec::with_capacity(MAX_CANDIDATES_PER_KEY))
944                .push(cached.clone());
945            self.order.push_back((key, cached));
946            self.peak = self.peak.max(self.retained());
947        }
948
949        /// What the cache retains now, by the bounds it is charged with: its
950        /// entries and its spine as allocated (the queue's capacity and a
951        /// node and a candidate list per key). Never more than the
952        /// `max_retained` it was made with.
953        pub(crate) fn retained(&self) -> usize {
954            self.held
955                .saturating_add(spine_of::<V>(self.order.capacity(), self.by_key.len()))
956        }
957
958        /// The most it has retained at once ([`ByteCache::retained`]), its
959        /// spine included, since it was made or cleared.
960        pub(crate) fn peak(&self) -> usize {
961            self.peak
962        }
963
964        /// Drop every entry, the spine's allocations and the peak.
965        pub(crate) fn clear(&mut self) {
966            self.by_key = BTreeMap::new();
967            self.order = VecDeque::new();
968            self.held = 0;
969            self.peak = 0;
970        }
971
972        /// What the entries are charged now, and how many there are.
973        #[cfg(test)]
974        pub(crate) fn held(&self) -> (usize, usize) {
975            (self.held, self.order.len())
976        }
977
978        /// What the entries may be charged: `max_retained` less the spine.
979        #[cfg(test)]
980        pub(crate) fn max_charge(&self) -> usize {
981            self.max_charge
982        }
983    }

985    /// What a file's executable bytes hold, by file offset: the load of each
986    /// exclusive pair and each EL0 counter read. A property of the file, not
987    /// of how a process maps it, so one scan serves every mapping of it.
988    #[derive(Clone, Debug, Default, PartialEq, Eq)]
989    pub(crate) struct FileLoops {
990        /// The file offset of each pair's load-exclusive.
991        pub(crate) pair_loads: Vec<u64>,
992        /// The file offset just past each pair's retry branch, in the order
993        /// of `pair_loads`: a pair covers `[load, end)`.
994        pub(crate) pair_ends: Vec<u64>,
995        /// The file offset of each counter read.
996        pub(crate) counter_reads: Vec<u64>,
997        /// The file ranges scanned whole because nothing told their code
998        /// from their data (`decode::code_file_ranges`).
999        pub(crate) unverified: Vec<(u64, u64)>,
1000    }

1002    impl FileLoops {
1003        /// What a mapping of the file range `[start, end)` holds: each pair
1004        /// belongs to the mapping that holds its load-exclusive, so a loop
1005        /// split across two mappings is counted once, whatever is mapped
1006        /// after it, and whether or not the rewriter would take it.
1007        pub(crate) fn found_in(&self, start: u64, end: u64) -> Found {
1008            let inside = |offsets: &[u64]| {
1009                offsets
1010                    .iter()
1011                    .filter(|offset| start <= **offset && **offset < end)
1012                    .count() as u64
1013            };
1014            // A pair is unverified when any of its words, from its load to
1015            // its retry branch, lies in bytes nothing proves are code: one
1016            // whose store follows proved code into an unmarked section must
1017            // not count as verified.
1018            let unverified: Vec<u64> = self
1019                .pair_loads
1020                .iter()
1021                .zip(&self.pair_ends)
1022                .filter(|&(load, end)| {
1023                    self.unverified
1024                        .iter()
1025                        .any(|(start, stop)| start < end && load < stop)
1026                })
1027                .map(|(load, _)| *load)
1028                .collect();
1029            Found {
1030                pairs: inside(&self.pair_loads),
1031                counter_reads: inside(&self.counter_reads),
1032                unverified_pairs: inside(&unverified),
1033            }
1034        }
1035    }

1045    /// One mapping's in-memory rewrite once its file's plan is known. Every
1046    /// path ends with [`Rewrite::settled`], so a file whose plan left loops
1047    /// no mapping symbol proves is named however the rewrite went. `memory`
1048    /// is the guest's code, `None` when it could not be opened: then no
1049    /// sequence is tried, and every pair counts as not written. `range` is
1050    /// where the mapping starts, and the file offset and length it maps.
1051    /// Returns the rewrite and how many sequences run past the mapping's end,
1052    /// left whole.
1053    pub(crate) fn rewrite_planned<M: GuestCode>(
1054        plan: &FilePlan,
1055        pairs: u64,
1056        memory: Option<&M>,
1057        range: (u64, u64, u64),
1058        page: u64,
1059        new: &dyn Fn(u64, u64) -> bool,
1060        path: &Path,
1061    ) -> (Rewrite, u64) {
1062        let (rewrite, crossing) = match memory {
1063            Some(memory) => try_planned(plan, memory, range, page, new, path),
1064            // A failure before any sequence is tried counts at least one,
1065            // even in a mapping whose scan paired nothing: it may hold a
1066            // split sequence.
1067            None => (
1068                Rewrite {
1069                    failed: pairs.max(1),
1070                    not_written: Some(NotWritten::Memory),
1071                    ..Rewrite::default()
1072                },
1073                0,
1074            ),
1075        };
1076        // Every pair the scan found that was neither rewritten nor failed is
1077        // left: refused by the rewriter, not the file's bytes in memory, or
1078        // old code that may be running; and so is each sequence split across
1079        // this mapping's end.
1080        (rewrite.settled(pairs, &plan.report), crossing)
1081    }

1083    /// Try each of `plan`'s sequences that lies in the mapping `range`
1084    /// describes, through `memory`: write it where the guest's memory still
1085    /// holds the file's bytes and `new` says none of it can have run, within
1086    /// one page. Returns the counts before [`Rewrite::settled`] and the
1087    /// sequences that start here and run into the next mapping.
1088    fn try_planned<M: GuestCode>(
1089        plan: &FilePlan,
1090        memory: &M,
1091        (map_start, map_offset, map_len): (u64, u64, u64),
1092        page: u64,
1093        new: &dyn Fn(u64, u64) -> bool,
1094        path: &Path,
1095    ) -> (Rewrite, u64) {
1096        let mut rewrite = Rewrite::default();
1097        // Sequences that start here and run into the next mapping: left
1098        // whole. The scan counted each pair in the mapping that holds its
1099        // load, so they are among the `pairs` of this one or the next.
1100        let mut crossing = 0;
1101        for patch in &plan.patches {
1102            let address = match place(
1103                patch.offset,
1104                patch.original.len() as u64,
1105                map_start,
1106                map_offset,
1107                map_len,
1108            ) {
1109                Placement::Outside => continue,
1110                Placement::CrossesEnd => {
1111                    crossing += 1;
1112                    continue;
1113                }
1114                Placement::At(address) => address,
1115            };
1116            if crosses_page(address, patch.original.len() as u64, page) {
1117                tracing::debug!(
1118                    "[store-exclusive] the sequence at {address:#x} crosses a page; left, so that \
1119                     no page drop can leave half of it"
1120                );
1121                continue;
1122            }
1123            let mut current = vec![0u8; patch.original.len()];
1124            if let Err(error) = memory.read(address, &mut current) {
1125                tracing::debug!(
1126                    "[store-exclusive] cannot read the guest's code at {address:#x}: {error}"
1127                );
1128                rewrite.failed += 1;
1129                rewrite.not_written.get_or_insert(NotWritten::Memory);
1130                continue;
1131            }
1132            let end = address + patch.original.len() as u64;
1133            match step(
1134                &current,
1135                &patch.original,
1136                &patch.rewritten,
1137                new(address, end),
1138            ) {
1139                Step::Done => {
1140                    rewrite.rewritten += 1;
1141                }
1142                Step::Leave => {
1143                    tracing::debug!(
1144                        "[store-exclusive] the guest's code at {address:#x} is not the file's, \
1145                         or may already have run; left"
1146                    );
1147                }
1148                Step::Write => {
1149                    match write_sequence(memory, address, &patch.original, &patch.rewritten) {
1150                        Written::Rewritten => {
1151                            rewrite.rewritten += 1;
1152                        }
1153                        Written::Failed(why) => {
1154                            tracing::debug!(

1247    /// How writing one sequence ended.
1248    #[derive(Clone, Debug, PartialEq, Eq)]
1249    pub(crate) enum Written {
1250        /// Memory holds the replacement, read back.
1251        Rewritten,
1252        /// Memory holds the original, as before: counted as not written.
1253        Failed(String),
1254        /// Memory holds neither, and the original could not be put back:
1255        /// the guest must not run.
1256        Broken(String),
1257    }
1258
1259    /// Write `rewritten` over `original` at `address`, whole or not at all.
1260    ///
1261    /// The original bytes are written first: that makes Linux copy every
1262    /// page the sequence touches for this process, without changing what any
1263    /// of them holds, so a failure there (even after a prefix) leaves the
1264    /// loop as it was. Then the replacement is written and read back; on any
1265    /// error or other bytes, the original is written back and read back. The
1266    /// result says what memory holds, never what was intended.
1267    pub(crate) fn write_sequence(
1268        code: &impl GuestCode,
1269        address: u64,
1270        original: &[u8],
1271        rewritten: &[u8],
1272    ) -> Written {
1273        if let Err(error) = code.write(address, original) {
1274            return Written::Failed(format!("copying the pages failed: {error}"));
1275        }
1276        let holds = |expected: &[u8]| {
1277            let mut now = vec![0u8; expected.len()];
1278            code.read(address, &mut now).is_ok() && now == expected
1279        };
1280        let problem = match code.write(address, rewritten) {
1281            Ok(()) if holds(rewritten) => return Written::Rewritten,
1282            Ok(()) => "the replacement did not read back".to_owned(),
1283            Err(error) => format!("writing the replacement failed: {error}"),
1284        };
1285        // Read back whatever the restore reports: a write that fails part
1286        // way may still have put every byte back.
1287        let restore = code.write(address, original).err();
1288        if holds(original) {
1289            Written::Failed(problem)
1290        } else if holds(rewritten) {
1291            Written::Rewritten
1292        } else {
1293            Written::Broken(match restore {
1294                Some(error) => format!("{problem}, and writing the original back failed: {error}"),
1295                None => format!("{problem}, and the original did not read back"),
1296            })
1297        }
1298    }

1331        /// executable file mapping it saw (`None` when its maps could not be
1332        /// read), and `mappings` the outcomes of those it scanned. A mapping
1333        /// replaces what it overlaps in that address space, and an
1334        /// executable alias of the same file bytes elsewhere counts as its
1335        /// own, so the tally holds each address space's executable mappings
1336        /// as of its last scan.

1338            &mut self,
1339            space: &(String, u64),
1340            mappings: &[ScannedMapping],
1341            live: Option<&[LiveMapping]>,
1342        ) {
1343            // The executable file mappings the scan saw, by guest address
1344            // range: an earlier outcome for anything else in this address
1345            // space (unmapped, moved, split, no longer executable) is gone.
1346            if let Some(live) = live {
1347                for (path, spaces) in self.found.iter_mut() {
1348                    if let Some(ranges) = spaces.get_mut(space) {
1349                        ranges.retain(|&(address, len), _| {
1350                            live.iter().any(|seen| {
1351                                seen.path == *path && seen.address == address && seen.len == len
1352                            })

1354                    }
1355                }
1356            }
1357            for mapping in mappings {
1358                let ranges = self
1359                    .found
1360                    .entry(mapping.path.clone())
1361                    .or_default()
1362                    .entry(space.clone())
1363                    .or_default();

1365                // range split and rejoined by mprotect, or remapped), and
1366                // nothing else: an alias of the same file bytes elsewhere is
1367                // another mapping, with its own outcome.
1368                let end = mapping.address.saturating_add(mapping.len);
1369                ranges.retain(|&(from, length), _| {
1370                    end <= from || from.saturating_add(length) <= mapping.address
1371                });

1374                    (mapping.found, mapping.rewrite),

1398        /// three counts describe one process and the worst exposure is the

1409                            ranges
1410                                .values()
1411                                .fold(Rewrite::default(), |sum, (_, rewrite)| sum.add(*rewrite))
1412                        })
1413                        .inspect(|rewrite| unmarked |= rewrite.unmarked)
1414                        .max_by_key(|r| (r.left + r.failed, r.rewritten + r.left + r.failed))
1415                        .unwrap_or_default();

1417                    (path.clone(), rewrite)
1418                })
1419                .filter(|(_, rewrite)| !rewrite.is_empty())
1420                .collect()
1421        }

1424        pub(crate) fn fill_rewrite_summary(&self, summary: &mut RunSummary) {
1425            // The files whose loops were left for want of mapping symbols
1426            // are named beside those the overlay left
1427            // (`Config::rewrite_exclusives_unmarked`).
1428            let mut unmarked_files = Vec::new();
1429            summary.exclusive_rewrites = self
1430                .rewrites_per_path()
1431                .into_iter()
1432                .map(|(path, rewrite)| {
1433                    let Rewrite {
1434                        rewritten,
1435                        left,
1436                        failed,
1437                        unmarked,
1438                        not_written,
1439                    } = rewrite;
1440                    if unmarked {
1441                        unmarked_files.push(path.clone());
1442                    }
1443                    (
1444                        path,
1445                        ExclusiveRewrite {
1446                            rewritten,
1447                            left,
1448                            failed,
1449                            not_written: not_written.map(|why| why.to_string()),
1450                        },
1451                    )
1452                })
1453                .collect();
1454            summary.exclusive_rewrite_unmarked.extend(unmarked_files);
1455        }

1457        /// The exclusive loops of each guest path that can still retry, in the
1458        /// address space where the most can: what an in-memory rewrite left
1459        /// or could not write, and every pair where none was attempted. Every
1460        /// path with pairs or a rewrite outcome is considered.
1461        pub(crate) fn remaining_per_path(&self) -> BTreeMap<String, u64> {
1462            self.found
1463                .iter()
1464                .map(|(path, spaces)| {
1465                    let most = spaces
1466                        .values()
1467                        .map(|ranges| {
1468                            let pairs: u64 = ranges.values().map(|(found, _)| found.pairs).sum();
1469                            let rewrite = ranges
1470                                .values()
1471                                .fold(Rewrite::default(), |sum, (_, rewrite)| sum.add(*rewrite));
1472                            if rewrite.is_empty() {
1473                                pairs
1474                            } else {
1475                                rewrite.left + rewrite.failed
1476                            }
1477                        })
1478                        .max()
1479                        .unwrap_or(0);
1480                    (path.clone(), most)
1481                })
1482                .filter(|(_, count)| *count != 0)
1483                .collect()
1484        }
1485
1486        /// One count of each guest path that has a nonzero one: the sum over
1487        /// the ranges one address space mapped, and the largest such sum over
1488        /// address spaces, so that a library many processes map counts once.
1489        /// A path is tallied when its mapping has either kind of finding, so
1490        /// the zero of the other kind is left out here.
1491        fn per_path(&self, of: impl Fn(&(Found, Rewrite)) -> u64) -> BTreeMap<String, u64> {
1492            self.found
1493                .iter()
1494                .map(|(path, spaces)| {
1495                    let most = spaces
1496                        .values()
1497                        .map(|ranges| ranges.values().map(&of).sum::<u64>())
1498                        .max()
1499                        .unwrap_or(0);
1500                    (path.clone(), most)
1501                })
1502                .filter(|(_, count)| *count != 0)
1503                .collect()
1504        }
1505    }

1923    /// Open a user-space counting event (not sampled) of `pmu` on task `pid`,
1924    /// or on the calling thread for 0, inherited by the tasks it creates when
1925    /// `inherit` is set. Naming the PMU's own type binds the event to that
1926    /// PMU, where `PERF_TYPE_RAW` would bind it to the first one that accepts
1927    /// it.
1928    ///
1929    /// The event is flexible, never pinned. Reverie pins the branch counter
1930    /// that drives virtual time and the preemption timer (`attr.pinned`, in
1931    /// its `perf::Builder`), and Linux schedules every pinned group onto the
1932    /// PMU before any flexible one (`ctx_sched_in`, kernel/events/core.c), so
1933    /// these events take only counters the pinned ones leave and cannot
1934    /// displace them; when the PMU is short, they are the ones time-shared,
1935    /// which `TOTAL_TIME_RUNNING` shows and the counts report as not counted
1936    /// for the whole run. A pinned branch counter that could not be scheduled
1937    /// would put its event in error, which Reverie refuses rather than read.
1938    fn open_event(pmu: &CpuPmu, config: u64, pid: i32, inherit: bool) -> std::io::Result<OwnedFd> {
1939        let mut attr = PerfEventAttr {
1940            type_: pmu.kind,
1941            size: std::mem::size_of::<PerfEventAttr>() as u32,
1942            config,
1943            read_format: TOTAL_TIME_ENABLED | TOTAL_TIME_RUNNING,
1944            flags: EXCLUDE_KERNEL | EXCLUDE_HV | if inherit { INHERIT } else { 0 },
1945            ..Default::default()
1946        };
1947        // SAFETY: `attr` is a valid `perf_event_attr` prefix whose `size`
1948        // field gives its length, and it outlives the call.
1949        let fd = unsafe {
1950            libc::syscall(
1951                libc::SYS_perf_event_open,
1952                &mut attr as *mut PerfEventAttr,
1953                pid,
1954                -1,
1955                -1,
1956                PERF_FLAG_FD_CLOEXEC,
1957            )
1958        };
1959        if fd < 0 {
1960            return Err(std::io::Error::last_os_error());
1961        }
1962        // SAFETY: perf_event_open returned a new descriptor nothing else owns.
1963        Ok(unsafe { OwnedFd::from_raw_fd(fd as i32) })
1964    }

1966    /// An event's value, time enabled and time running.
1967    fn read_event(event: &OwnedFd) -> std::io::Result<(u64, u64, u64)> {
1968        let mut values = [0u64; 3];
1969        // SAFETY: reads at most 24 bytes into a buffer of 24.
1970        let read = unsafe { libc::read(event.as_raw_fd(), values.as_mut_ptr().cast(), 24) };
1971        match read {
1972            24 => Ok((values[0], values[1], values[2])),
1973            read if read < 0 => Err(std::io::Error::last_os_error()),
1974            _ => Err(std::io::Error::other("short read of a perf event")),
1975        }

1978    /// The CPUs task `pid` may run on.
1979    fn affinity(pid: i32) -> std::io::Result<Vec<usize>> {
1980        // SAFETY: an all-zero `cpu_set_t` is an empty set.
1981        let mut set: libc::cpu_set_t = unsafe { std::mem::zeroed() };
1982        // SAFETY: `set` is a `cpu_set_t` of the size passed.
1983        if unsafe { libc::sched_getaffinity(pid, std::mem::size_of_val(&set), &mut set) } != 0 {
1984            return Err(std::io::Error::last_os_error());
1985        }
1986        // SAFETY: every CPU number is below `CPU_SETSIZE`.
1987        Ok((0..libc::CPU_SETSIZE as usize)
1988            .filter(|cpu| unsafe { libc::CPU_ISSET(*cpu, &set) })
1989            .collect())
1990    }

1992    /// Run the calling thread on `cpus` only.
1993    fn set_affinity(cpus: &[usize]) -> std::io::Result<()> {
1994        // SAFETY: an all-zero `cpu_set_t` is an empty set.
1995        let mut set: libc::cpu_set_t = unsafe { std::mem::zeroed() };
1996        for &cpu in cpus {
1997            // SAFETY: the caller's CPU numbers came from the kernel's CPU list.
1998            unsafe { libc::CPU_SET(cpu, &mut set) };
1999        }
2000        // SAFETY: `set` is a `cpu_set_t` of the size passed.
2001        if unsafe { libc::sched_setaffinity(0, std::mem::size_of_val(&set), &set) } != 0 {
2002            return Err(std::io::Error::last_os_error());
2003        }
2004        Ok(())
2005    }

2012    /// What a PMU counted of the probe's store-exclusives.
2013    struct Probe {
2014        /// `STREX_FAIL_SPEC`'s increase over [`PROBE_LOOPS`] failing ones.
2015        failed: u64,
2016        /// How many of those the CPU itself reported failed.
2017        observed_failed: u64,
2018        /// `STREX_SPEC`'s increase over all [`PROBE_LOOPS`] x 2, if it opens.
2019        executed: Option<u64>,
2020    }

2024    /// Check what `pmu` counts, on the calling thread of this (tracer)
2025    /// process pinned to `cpu`, one of its CPUs, and given back its own CPUs
2026    /// after: [`PROBE_LOOPS`] store-exclusives that fail (`LDXR`; `CLREX`,
2027    /// which clears the exclusive monitor; `STXR`), then as many exclusive
2028    /// increments that succeed.
2029    ///
2030    /// Not on a new thread: the tracer runs in the guest's PID namespace, so
2031    /// a thread it starts takes the next PID there, and every guest thread
2032    /// started after it would get another ID than in a run that does not
2033    /// probe. Runs that record preemptions (`--record-preemptions-to`) or
2034    /// replay them (`--replay-preemptions-from`) probe; a replay of such a
2035    /// run's schedule (`--replay-schedule-from`) does not, so it would lose
2036    /// its threads at their first clone.
2037    fn probe(pmu: &CpuPmu, cpu: usize) -> Result<Probe, String> {
2038        /// Gives the calling thread back its CPUs when the probe ends, also
2039        /// when it unwinds.
2040        struct Restore(Vec<usize>);
2041        impl Drop for Restore {
2042            fn drop(&mut self) {
2043                if let Err(error) = set_affinity(&self.0) {
2044                    warn!(
2045                        "[store-exclusive] cannot give the tracer thread back its CPUs after \
2046                         the probe, so it stays on one CPU: {error}"
2047                    );
2048                }
2049            }
2050        }
2051        let own = affinity(0)
2052            .map_err(|error| format!("cannot read the tracer thread's CPU affinity: {error}"))?;
2053        let _restore = Restore(own);
2054        probe_on(pmu, cpu)
2055    }

2057    fn probe_on(pmu: &CpuPmu, cpu: usize) -> Result<Probe, String> {
2058        pin(cpu).map_err(|error| {
2059            format!(
2060                "cannot run the store-exclusive probe on CPU {cpu} of PMU {}: {error}",
2061                pmu.name
2062            )
2063        })?;
2064        let failed = open_event(pmu, STREX_FAIL_SPEC, 0, false).map_err(|error| {
2065            format!(
2066                "perf_event_open of STREX_FAIL_SPEC (raw 0x6e) on PMU {} failed: {error}",
2067                pmu.name
2068            )
2069        })?;
2070        let executed = open_event(pmu, STREX_SPEC, 0, false).ok();
2071        let value = |event: &OwnedFd| read_event(event).map(|read| read.0).unwrap_or(0);
2072        let (failed_before, executed_before) = (value(&failed), executed.as_ref().map(value));
2073        let mut word = 0u64;
2074        let mut observed_failed = 0u64;
2075        for _ in 0..PROBE_LOOPS {
2076            let status: u64;
2077            // SAFETY: an exclusive load and store of a local word.
2078            unsafe {
2079                core::arch::asm!(
2080                    "ldxr {value}, [{word}]",
2081                    "clrex",
2082                    "stxr {status:w}, {value}, [{word}]",
2083                    word = in(reg) &raw mut word,
2084                    value = out(reg) _,
2085                    status = out(reg) status,
2086                    options(nostack),
2087                );
2088            }
2089            observed_failed += status & 1;
2090        }
2091        for _ in 0..PROBE_LOOPS {
2092            // SAFETY: an exclusive increment of a local word.
2093            unsafe {
2094                core::arch::asm!(
2095                    "2:",
2096                    "ldxr {value}, [{word}]",
2097                    "add {value}, {value}, #1",
2098                    "stxr {status:w}, {value}, [{word}]",
2099                    "cbnz {status:w}, 2b",
2100                    word = in(reg) &raw mut word,
2101                    value = out(reg) _,
2102                    status = out(reg) _,
2103                    options(nostack),
2104                );
2105            }
2106        }
2107        std::hint::black_box(word);
2108        Ok(Probe {
2109            failed: value(&failed).saturating_sub(failed_before),
2110            observed_failed,
2111            executed: executed
2112                .as_ref()
2113                .zip(executed_before)
2114                .map(|(event, before)| value(event).saturating_sub(before)),
2115        })
2116    }

2389    /// Open the file a mapping maps, for planning its rewrite.
2390    ///
2391    /// `/proc/<tid>/map_files/<start>-<end>` is the mapped object itself, so
2392    /// it is tried first; Linux opens it only for a tracer with
2393    /// `CAP_CHECKPOINT_RESTORE` or `CAP_SYS_ADMIN`. Otherwise the mapping's
2394    /// name is opened inside the guest's root and must lead to the same
2395    /// inode, and, outside btrfs (where a maps line names the superblock's
2396    /// device rather than the subvolume's) and overlayfs (where it names the
2397    /// layer's), the same device. What the open proves is returned with it:
2398    /// a scan accepts any, a rewrite plan only a proven one.

2400        tid: Pid,
2401        path: &Path,
2402        map: &MemoryMap,
2403        key: FileRange,
2404    ) -> std::io::Result<(File, Opened)> {
2405        let (start, end) = map.address;
2406        let exact = format!("/proc/{}/map_files/{start:x}-{end:x}", tid.as_raw());
2407        let (file, exact) = match File::open(&exact) {
2408            Ok(file) => (file, true),
2409            Err(_) => (open_in_root(tid, path)?, false),
2410        };
2411        let metadata = file.metadata()?;
2412        let (major, minor) = key.dev;
2413        let same_device = metadata.dev() == libc::makedev(major as u32, minor as u32);
2414        // SAFETY: an all-zero `statfs` is a valid buffer for fstatfs.
2415        let mut fs: libc::statfs = unsafe { std::mem::zeroed() };
2416        // SAFETY: a valid descriptor and a `statfs` buffer.
2417        // A filesystem that cannot be told counts as unreliable.
2418        let device_unreliable = unsafe { libc::fstatfs(file.as_raw_fd(), &mut fs) } != 0
2419            || [BTRFS_SUPER_MAGIC, OVERLAYFS_SUPER_MAGIC].contains(&(fs.f_type as i64));
2420        let opened = Opened::of(
2421            exact,
2422            metadata.ino() == key.inode,
2423            same_device,
2424            device_unreliable,
2425        )
2426        .ok_or_else(|| {
2427            std::io::Error::other("the name now leads to another file than the mapped one")
2428        })?;

2430    }

2438    /// The largest file planned in memory: the whole file is read, and a
2439    /// larger one is counted as not rewritten rather than risk the
2440    /// allocation.
2441    const MAX_PLANNED_FILE: u64 = 1 << 30;

2443    /// The rewrite plans this Hermit process has made, found by the exact
2444    /// bytes they were made from ([`ByteCache`]): a plan is a pure function of
2445    /// those bytes, so a hit is the plan planning them again would give, and
2446    /// no file identity (device, inode, path, time) or digest is trusted. It
2447    /// retains at most 128 MiB (a process on a small device maps tens of MB of
2448    /// code) and 1024 entries; anything past a bound only costs a re-plan.
2449    /// Each run starts it empty ([`RunTally::new_run`]): Hermit runs one
2450    /// guest at a time in a process (`--verify` runs each in its own,
2451    /// `analyze` its trials one after another). Runs that did overlap in one
2452    /// process would share it, and one's start would empty it under the
2453    /// other: that costs re-planning and the accuracy of the full report's
2454    /// peak, never a plan's bytes or the INFO view.
2455    static PLANS: Mutex<ByteCache<Arc<Option<FilePlan>>>> =
2456        Mutex::new(ByteCache::new(128 << 20, 1024));

2458    /// The rewrite plan of the file a mapping maps, from the mapped object's
2459    /// bytes, read afresh each time. The bytes are used only if the file kept
2460    /// its size, modification and status-change times (the last cannot be
2461    /// set from user space) while they were read; planning them is skipped
2462    /// when bytes equal to them were planned before ([`PLANS`]).
2463    fn plan_for(
2464        tid: Pid,
2465        path: &Path,
2466        map: &MemoryMap,
2467        key: FileRange,
2468    ) -> Result<Arc<Option<FilePlan>>, (NotWritten, String)> {
2469        let unreadable = |error: std::io::Error| (NotWritten::Unreadable, error.to_string());
2470        // A name that now leads to another file than the mapped one is a
2471        // file this rewrite cannot prove, not one it could not read.
2472        let (file, opened) = open_mapped(tid, path, map, key).map_err(|error| {
2473            let why = if error.kind() == std::io::ErrorKind::Other {
2474                NotWritten::Unproven
2475            } else {
2476                NotWritten::Unreadable
2477            };
2478            (why, error.to_string())
2479        })?;
2480        if !opened.may_plan_writes() {
2481            return Err((
2482                NotWritten::Unproven,
2483                "the mapped object cannot be proven without /proc/<pid>/map_files on this \
2484                 filesystem (btrfs or overlayfs, where an inode number is not an identity)"
2485                    .to_owned(),
2486            ));
2487        }
2488        let stamp = file_stamp(&file).map_err(unreadable)?;
2489        let too_large = || {
2490            (
2491                NotWritten::Unreadable,
2492                format!("the file is larger than the {MAX_PLANNED_FILE} bytes planned in memory"),
2493            )
2494        };
2495        if stamp.0 > MAX_PLANNED_FILE {
2496            return Err(too_large());
2497        }
2498        let mut bytes = Vec::new();
2499        bytes
2500            .try_reserve_exact(stamp.0 as usize + 1)
2501            .map_err(|error| (NotWritten::Unreadable, error.to_string()))?;
2502        // Read at most one byte past the bound, so a file that grows while it
2503        // is read can neither pass the bound nor grow the buffer unchecked.
2504        std::io::Read::read_to_end(
2505            &mut std::io::Read::take(&file, MAX_PLANNED_FILE + 1),
2506            &mut bytes,
2507        )
2508        .map_err(unreadable)?;
2509        if bytes.len() as u64 > MAX_PLANNED_FILE {
2510            return Err(too_large());
2511        }
2512        // The bytes read are the file the stamp names only if nothing
2513        // changed it meanwhile.
2514        if file_stamp(&file).map_err(unreadable)? != stamp || bytes.len() as u64 != stamp.0 {
2515            return Err((
2516                NotWritten::Unreadable,
2517                "the file changed while it was read".to_owned(),
2518            ));
2519        }
2520        // The plan is a function of the bytes alone, so the cache keys them.
2521        if let Some(plan) = PLANS.lock().unwrap().get(&bytes) {
2522            return Ok(plan);
2523        }
2524        let plan = Arc::new(
2525            crate::aarch64_lse::elf::plan_bytes(&bytes)
2526                .map_err(|error| (NotWritten::Unreadable, error.to_string()))?,
2527        );
2528        let charge = plan_charge(&plan);
2529        PLANS.lock().unwrap().insert(bytes, plan.clone(), charge);
2530        Ok(plan)
2531    }

2546    /// Rewrite the canonical exclusive loops of one executable file mapping
2547    /// in the guest's memory. `pairs` is what the scan found in it.
2548    ///
2549    /// A sequence is written only where the guest's memory still holds the
2550    /// file's bytes, only where no instruction of it can have run (`new` says
2551    /// so for its address range: made by this exec or this `mmap`), and only
2552    /// within one page. Memory that already holds the replacement counts as
2553    /// rewritten; anything else (relocated text, a debugger's breakpoint, a
2554    /// runtime that rewrote itself, text a page drop put back, or old code
2555    /// that may be running) is left.
2556    ///
2557    /// The write goes through `/proc/<tid>/mem`, which may write a private
2558    /// read-only page the way a debugger does: Linux copies the page for this
2559    /// process alone, so the file and every other process mapping it keep the
2560    /// original, and the kernel keeps the instruction cache coherent with the
2561    /// written page. A shared or writable mapping is left alone. Each
2562    /// sequence is written whole or not at all (`write_sequence`, through
2563    /// [`rewrite_planned`]); if the guest's memory ends up holding neither,
2564    /// the run ends here.
2565    fn rewrite_mapping(
2566        tid: Pid,
2567        path: &Path,
2568        key: FileRange,
2569        map: &MemoryMap,
2570        pairs: u64,
2571        new: &dyn Fn(u64, u64) -> bool,
2572    ) -> (Rewrite, u64) {
2573        let all_left = Rewrite {
2574            left: pairs,
2575            ..Rewrite::default()
2576        };
2577        // A failure before the plan is placed counts at least one, even in a
2578        // mapping whose scan paired nothing: it may hold a split sequence.
2579        let all_failed = Rewrite {
2580            failed: pairs.max(1),
2581            ..Rewrite::default()
2582        };
2583        if map.perms.contains(MMPermissions::SHARED) || map.perms.contains(MMPermissions::WRITE) {
2584            debug!(
2585                "[store-exclusive] {} at {:#x} is shared or writable; its exclusive loops are left",
2586                path.display(),
2587                map.address.0
2588            );
2589            return (all_left, 0);
2590        }
2591        let plan = match plan_for(tid, path, map, key) {
2592            Ok(plan) => plan,
2593            Err((why, error)) => {
2594                debug!(
2595                    "[store-exclusive] cannot read {} to plan its rewrite: {error}",
2596                    path.display()
2597                );
2598                return (
2599                    Rewrite {
2600                        not_written: Some(why),
2601                        ..all_failed
2602                    },
2603                    0,
2604                );
2605            }
2606        };
2607        let Some(plan) = plan.as_ref() else {
2608            return (all_left, 0);
2609        };
2610        let memory = match OpenOptions::new()
2611            .read(true)
2612            .write(true)
2613            .open(format!("/proc/{}/mem", tid.as_raw()))
2614        {
2615            Ok(memory) => Some(ProcMem(memory)),
2616            Err(error) => {
2617                debug!(
2618                    "[store-exclusive] cannot open /proc/{}/mem to rewrite {}: {error}",
2619                    tid.as_raw(),
2620                    path.display()
2621                );
2622                None
2623            }
2624        };
2625        rewrite_planned(
2626            plan,
2627            pairs,
2628            memory.as_ref(),
2629            (map.address.0, key.offset, key.len),
2630            page_size(),
2631            new,
2632            path,
2633        )
2634    }

2646            CString::new(relative.as_os_str().as_bytes()).map_err(std::io::Error::other)?;

2649        struct OpenHow {
2650            flags: u64,
2651            mode: u64,

2677    /// Scan the mapped range of a file of `size` bytes, in its executable
2678    /// ELF sections only, for what [`Found`] names.
2679    fn scan_file(file: &File, size: u64) -> std::io::Result<FileLoops> {
2680        let (ranges, unverified) = code_ranges(file, size)?;
2681        // Bounded like a plan: the whole of a file's code is read, whatever
2682        // a process maps of it, so a huge file counts as not scanned.

2686            )));

2689            read_words(file, start, end, scan)

2693    /// The file ranges that hold instructions, and of those the ones scanned
2694    /// without evidence: the code of the file's executable ELF sections
2695    /// ([`code_file_ranges`]), or the whole file, unverified, when it has no
2696    /// section headers. Scanned whole, whatever a process maps of them.
2697    #[allow(clippy::type_complexity)]
2698    fn code_ranges(file: &File, size: u64) -> std::io::Result<(FileRanges, FileRanges)> {
2699        let mut header = [0u8; 64];
2700        let table = match file.read_exact_at(&mut header, 0) {
2701            Ok(()) => elf_section_table(&header),
2702            Err(error) if error.kind() == std::io::ErrorKind::UnexpectedEof => None,
2703            Err(error) => return Err(error),
2704        };
2705        let Some((offset, count)) = table else {
2706            return Ok((vec![(0, size)], vec![(0, size)]));
2707        };
2708        let mut table = vec![0u8; count * 64];
2709        file.read_exact_at(&mut table, offset)?;
2710        let shstrndx = usize::from(u16::from_le_bytes([header[0x3E], header[0x3F]]));
2711        let e_type = u16::from_le_bytes([header[0x10], header[0x11]]);
2712        // Each table the evidence needs, if it lies in the file and is not
2713        // larger than the scan reads; one that is not read is evidence the
2714        // file lacks, which leaves its sections scanned whole.
2715        let mut budget = MAX_EVIDENCE_TABLES;
2716        let evidence = crate::aarch64_lse::mapping::CodeEvidence::read(
2717            &table,
2718            shstrndx,
2719            e_type,
2720            |(offset, length)| {
2721                let end = offset.checked_add(length)?;
2722                if end > size || length > budget {
2723                    return None;
2724                }
2725                budget -= length;
2726                let mut bytes = vec![0u8; usize::try_from(length).ok()?];
2727                file.read_exact_at(&mut bytes, offset).ok()?;
2728                Some(bytes)
2729            },
2730        );
2731        let clip = |ranges: Vec<(u64, u64)>| {
2732            merge_overlaps(
2733                ranges
2734                    .into_iter()
2735                    .map(|(start, end)| (start, end.min(size)))
2736                    .filter(|(start, end)| start < end)
2737                    .collect(),
2738            )
2739        };
2740        let (ranges, unverified) = code_file_ranges(&executable_sections(&table), &evidence);
2741        Ok((clip(ranges), clip(unverified)))
2742    }

2748    /// Feed `counter` the words of `file` in `start..end`.
2749    fn read_words(
2750        file: &File,
2751        start: u64,
2752        end: u64,
2753        counter: &mut CodeScan,
2754    ) -> std::io::Result<()> {
2755        let mut buffer = vec![0u8; (1 << 20).min((end - start) as usize)];
2756        let mut at = start;
2757        while at < end {
2758            let chunk =
2759                usize::try_from(end - at).map_or(buffer.len(), |left| left.min(buffer.len()));
2760            file.read_exact_at(&mut buffer[..chunk], at)?;
2761            counter.push_bytes(&buffer[..chunk]);
2762            at += chunk as u64;
2763        }
2764        Ok(())
2765    }

3052    /// [`range_to_scan`] for an address space whose ranges are `scan`, on
3053    /// pages of `page` bytes: what a successful `call` that returned `result`
3054    /// makes scan, and why. `in_memory`: the run rewrites in memory.
3055    fn scan_after(
3056        call: &Syscall,
3057        result: i64,
3058        page: u64,
3059        in_memory: bool,
3060        scan: &mut ScannedRanges,
3061    ) -> Option<(Range, Trigger)> {
3062        let span = |start: u64, len: usize| page_span(start, len as u64, page);
3063        // Rewriting in memory, every transition to executable is rescanned,
3064        // so that the counts follow a range made writable and executable
3065        // again (it is recounted, never written); otherwise only a range
3066        // that may hold a mapping not scanned.
3067        let made_executable = |scan: &ScannedRanges, start: Option<u64>, len: usize| {
3068            let range = span(start.unwrap_or(0), len);
3069            let needed = in_memory || scan.needs_scan(range);
3070            needed.then_some((range, Trigger::Protected))
3071        };
3072        // Whether executable file text was in a range at the last scan: a
3073        // call that unmaps, moves, replaces or protects it changes the
3074        // tally, and the rescan that follows (which never writes) lets it
3075        // drop what is gone. Every mode, so that a count-only tally follows
3076        // too; anywhere else, nothing to do.
3077        let changed = |scan: &ScannedRanges, range: Range| {
3078            scan.was_executable(range)
3079                .then_some((range, Trigger::Changed))
3080        };
3081        match call {
3082            Syscall::Mmap(mmap) => {
3083                let range = span(result as u64, mmap.len());
3084                let file = !mmap.flags().contains(MapFlags::MAP_ANONYMOUS) && mmap.fd() >= 0;
3085                if file && mmap.prot().contains(ProtFlags::PROT_EXEC) {
3086                    return Some((range, Trigger::Mapped));
3087                }
3088                if file {
3089                    scan.add_unscanned(range);
3090                }
3091                // A fixed mapping that replaced executable text.
3092                changed(scan, range)
3093            }
3094            // Executable text moved away, or a fixed destination replaced
3095            // some. The rescan of the new range also learns the address
3096            // space's ranges afresh, so a later page drop is judged by where
3097            // the text is now.
3098            Syscall::Mremap(mremap) => {
3099                let old = span(
3100                    mremap.addr().map_or(0, |addr| addr.as_raw() as u64),
3101                    mremap.old_len(),
3102                );
3103                let new = span(result as u64, mremap.new_len());
3104                (scan.was_executable(old) || scan.was_executable(new))
3105                    .then_some((new, Trigger::Changed))
3106            }
3107            Syscall::Munmap(munmap) => changed(
3108                scan,
3109                span(
3110                    munmap.addr().map_or(0, |addr| addr.as_raw() as u64),
3111                    munmap.len(),
3112                ),
3113            ),
3114            Syscall::Mprotect(mprotect) if mprotect.protection().contains(ProtFlags::PROT_EXEC) => {
3115                made_executable(
3116                    scan,
3117                    mprotect.addr().map(|addr| addr.as_raw() as u64),
3118                    mprotect.len(),
3119                )
3120            }
3121            Syscall::PkeyMprotect(mprotect) if mprotect.prot() & libc::PROT_EXEC != 0 => {
3122                made_executable(
3123                    scan,
3124                    mprotect.addr().map(|addr| addr.as_raw() as u64),
3125                    mprotect.len(),
3126                )
3127            }
3128            // Text no longer executable.
3129            Syscall::Mprotect(mprotect) => changed(
3130                scan,
3131                span(
3132                    mprotect.addr().map_or(0, |addr| addr.as_raw() as u64),
3133                    mprotect.len(),
3134                ),
3135            ),
3136            Syscall::PkeyMprotect(mprotect) => changed(
3137                scan,
3138                span(
3139                    mprotect.addr().map_or(0, |addr| addr.as_raw() as u64),
3140                    mprotect.len(),
3141                ),
3142            ),
3143            // Dropping a private file mapping's pages puts the file's bytes
3144            // back, undoing an in-memory rewrite there: the scan recounts
3145            // those loops as left. Only where an executable file mapping
3146            // was: an allocator's frequent drops of anonymous memory need no
3147            // scan. (MADV_REMOVE fails on a private mapping; it is listed so
3148            // that a shared one is recounted too.)
3149            Syscall::Madvise(madvise)
3150                if in_memory
3151                    && matches!(
3152                        madvise.advice(),
3153                        libc::MADV_DONTNEED | libc::MADV_REMOVE | MADV_DONTNEED_LOCKED
3154                    ) =>
3155            {
3156                let start = madvise.addr().map_or(0, |addr| addr.as_raw() as u64);
3157                let range = span(start, madvise.len());
3158                scan.was_executable(range)
3159                    .then_some((range, Trigger::Dropped))
3160            }
3161            _ => None,
3162        }
3163    }

5008    /// How a write to [`FakeCode`] goes wrong.
5009    #[derive(Clone, Copy)]
5010    enum Fault {
5011        /// Writes up to the first page boundary after the start, then fails,
5012        /// as a `/proc/<pid>/mem` write does when a later page fails.
5013        AfterFirstPage,
5014        /// Writes everything, but one byte reads back changed.
5015        ReadsBackChanged,
5016    }

5018    /// Guest code in a buffer of pages, whose `n`th write can be made to fail.
5019    struct FakeCode {
5020        bytes: std::cell::RefCell<Vec<u8>>,
5021        page: usize,
5022        writes: std::cell::Cell<usize>,
5023        faults: Vec<(usize, Fault)>,
5024    }
5025
5026    impl FakeCode {
5027        fn new(bytes: &[u8], faults: Vec<(usize, Fault)>) -> Self {
5028            FakeCode {
5029                bytes: std::cell::RefCell::new(bytes.to_vec()),
5030                page: 8,
5031                writes: std::cell::Cell::new(0),
5032                faults,
5033            }
5034        }
5035    }
5036
5037    impl GuestCode for FakeCode {
5038        fn read(&self, address: u64, buf: &mut [u8]) -> std::io::Result<()> {
5039            let at = address as usize;
5040            buf.copy_from_slice(&self.bytes.borrow()[at..at + buf.len()]);
5041            Ok(())
5042        }
5043
5044        fn write(&self, address: u64, bytes: &[u8]) -> std::io::Result<()> {
5045            let n = self.writes.get();
5046            self.writes.set(n + 1);
5047            let at = address as usize;
5048            let fault = self
5049                .faults
5050                .iter()
5051                .find(|(nth, _)| *nth == n)
5052                .map(|(_, fault)| *fault);
5053            let mut memory = self.bytes.borrow_mut();
5054            match fault {
5055                None => {
5056                    memory[at..at + bytes.len()].copy_from_slice(bytes);
5057                    Ok(())
5058                }
5059                Some(Fault::AfterFirstPage) => {
5060                    let prefix = (at / self.page + 1) * self.page - at;
5061                    memory[at..at + prefix].copy_from_slice(&bytes[..prefix]);
5062                    Err(std::io::Error::from_raw_os_error(libc::ENOMEM))
5063                }
5064                Some(Fault::ReadsBackChanged) => {
5065                    memory[at..at + bytes.len()].copy_from_slice(bytes);
5066                    memory[at] ^= 0xff;
5067                    Ok(())
5068                }
5069            }
5070        }
5071    }

5090        let broken = |written: &Written| matches!(written, Written::Broken(_));
