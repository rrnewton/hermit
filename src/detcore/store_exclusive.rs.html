9//! Store-exclusive sequences on aarch64: detection only.
10//!
11//! An aarch64 load-exclusive/store-exclusive loop (`LDAXR` .. `STLXR`, then
12//! `CBNZ` back) runs once more whenever the local exclusive monitor is cleared
13//! between its load and its store, and any exception clears it: a host
14//! interrupt, a PMU overflow, a ptrace stop, a single step. Each retry adds
15//! retired branches, which Detcore counts to place preemptions and to advance
16//! virtual time, so a program that runs such loops can be preempted at other
17//! points in two runs, and a recorded schedule may not replay. LSE atomics
18//! (`LDADD`, `CAS`, `SWP`; Armv8.1-A) are single instructions and never retry.
19//!
20//! This module reports and changes nothing a guest sees:
21//!
22//! - The exclusive pairs in each executable file mapping, counted after exec
23//!   and after an `mmap` or `mprotect` that maps `PROT_EXEC` from a file. The
24//!   count is a function of the file's bytes, so it goes to INFO and to the
25//!   run summary's INFO view.
26//! - In a run that records or replays preemptions, the store-exclusives the
27//!   guest's tasks failed (`STREX_FAIL_SPEC`) and, where the core counts
28//!   them, executed (`STREX_SPEC`), counted from the start of the root
29//!   task's first image on every task it creates and read once at the end of
30//!   the run. Host interrupts decide them, so they never reach INFO or the
31//!   INFO view: only the full report, the summary JSON, the preemption
32//!   recording and that run's warnings carry them.
33//!
34//! The decoder and the model are pure and compile everywhere; the rest runs
35//! only on aarch64.
36

45    /// An instruction of the load/store exclusive class.
46    #[derive(Clone, Copy, Debug, PartialEq, Eq)]
47    pub(crate) enum Exclusive {
48        /// `LDXR`, `LDAXR`, `LDXP`, `LDAXP` and their byte and halfword forms.
49        Load,
50        /// `STXR`, `STLXR`, `STXP`, `STLXP` and their byte and halfword forms.
51        Store,
52    }
53
54    /// Bits 29:24 (`001000`) and 23 (`o2`) of the load/store exclusive class.
55    /// `o2 = 1` is load-acquire/store-release (`LDAR`, `STLR`) or `CAS`,
56    /// neither of which can fail.
57    const CLASS_MASK: u32 = 0x3F80_0000;
58    const CLASS: u32 = 0x0800_0000;
59    /// `L`: a load.
60    const LOAD: u32 = 1 << 22;
61    /// `o1`: a register pair.
62    const PAIR: u32 = 1 << 21;
63    /// The high size bit, which every exclusive pair sets. With `o1 = 1` and
64    /// this bit clear the encoding is `CASP` (Armv8.1-A), a single
65    /// instruction.
66    const PAIR_SIZE: u32 = 1 << 31;
67
68    /// Whether `insn` is a load-exclusive, a store-exclusive, or neither.
69    pub(crate) const fn classify(insn: u32) -> Option<Exclusive> {
70        if insn & CLASS_MASK != CLASS || (insn & PAIR != 0 && insn & PAIR_SIZE == 0) {
71            None
72        } else if insn & LOAD != 0 {
73            Some(Exclusive::Load)
74        } else {
75            Some(Exclusive::Store)
76        }
77    }
78
79    /// A store-exclusive is one of a pair when a load-exclusive is at most
80    /// this many instructions before it.
81    pub(crate) const PAIR_WINDOW: u64 = 16;
82
83    /// Counts the exclusive pairs in a stream of instruction words.
84    #[derive(Clone, Debug, Default)]
85    pub(crate) struct PairCounter {
86        index: u64,
87        last_load: Option<u64>,
88        pairs: u64,
89    }
90
91    impl PairCounter {
92        /// Take the next instruction word.
93        pub(crate) fn push(&mut self, insn: u32) {
94            match classify(insn) {
95                Some(Exclusive::Load) => self.last_load = Some(self.index),
96                Some(Exclusive::Store)
97                    if self
98                        .last_load
99                        .is_some_and(|load| self.index - load <= PAIR_WINDOW) =>
100                {
101                    self.pairs += 1
102                }
103                _ => {}
104            }
105            self.index += 1;
106        }
107
108        /// Take little-endian instruction words, as aarch64 stores code in
109        /// either data endianness. A trailing partial word is ignored, so a
110        /// stream split into several calls must split at multiples of four.
111        pub(crate) fn push_bytes(&mut self, bytes: &[u8]) {
112            for word in bytes.as_chunks::<4>().0 {
113                self.push(u32::from_le_bytes(*word));
114            }
115        }
116
117        /// The pairs counted so far.
118        pub(crate) fn pairs(&self) -> u64 {
119            self.pairs
120        }
121    }
122
123    /// Where an ELF64 little-endian file's section header table is: its
124    /// offset and entry count, from the 64-byte file header. `None` for any
125    /// other file, or one without section headers (or more than fit in
126    /// `e_shnum`), whose mapped bytes are then scanned whole.
127    pub(crate) fn elf_section_table(header: &[u8; 64]) -> Option<(u64, usize)> {
128        const SECTION_HEADER_SIZE: u16 = 64;
129        let half = |at: usize| u16::from_le_bytes([header[at], header[at + 1]]);
130        let offset = u64::from_le_bytes(header[0x28..0x30].try_into().unwrap());
131        let (entry_size, count) = (half(0x3A), half(0x3C));
132        // `\x7fELF`, ELFCLASS64, ELFDATA2LSB.
133        (header[..6] == *b"\x7fELF\x02\x01"
134            && offset != 0
135            && count != 0
136            && entry_size == SECTION_HEADER_SIZE)
137            .then_some((offset, usize::from(count)))
138    }
139
140    /// The file ranges (offset, size) of the sections that hold instructions
141    /// (`SHF_EXECINSTR`, with file contents) in an ELF64 little-endian
142    /// section header table. Only these are scanned, so read-only data that
143    /// shares the executable segment cannot pass for an exclusive pair.
144    pub(crate) fn executable_sections(table: &[u8]) -> Vec<(u64, u64)> {
145        const SHT_NOBITS: u32 = 8;
146        const SHF_EXECINSTR: u64 = 0x4;
147        table
148            .as_chunks::<64>()
149            .0
150            .iter()
151            .filter_map(|entry| {
152                let word = |at: usize| u64::from_le_bytes(entry[at..at + 8].try_into().unwrap());
153                let kind = u32::from_le_bytes(entry[4..8].try_into().unwrap());
154                (kind != SHT_NOBITS && word(8) & SHF_EXECINSTR != 0 && word(32) != 0)
155                    .then(|| (word(24), word(32)))
156            })
157            .collect()
158    }
159
160    /// The exclusive pairs in `words`.
161    #[cfg(test)]
162    pub(crate) fn count_pairs(words: impl IntoIterator<Item = u32>) -> u64 {
163        let mut counter = PairCounter::default();
164        words.into_iter().for_each(|word| counter.push(word));
165        counter.pairs()
166    }
167}

180    /// An Arm CPU PMU, as `/sys/bus/event_source/devices/<name>` describes
181    /// it.
182    #[derive(Clone, Debug, PartialEq, Eq)]
183    pub(crate) struct CpuPmu {
184        pub(crate) name: String,
185        /// The `perf_event_attr.type` that selects this PMU.
186        pub(crate) kind: u32,
187        /// The CPUs it counts on.
188        pub(crate) cpus: Vec<usize>,
189    }
190
191    /// Parse a kernel CPU list such as `0-3,8,10-11`.
192    pub(crate) fn parse_cpu_list(list: &str) -> Option<Vec<usize>> {
193        let mut cpus = Vec::new();
194        for part in list.trim().split(',').filter(|part| !part.is_empty()) {
195            match part.split_once('-') {
196                Some((first, last)) => {
197                    let (first, last): (usize, usize) = (first.parse().ok()?, last.parse().ok()?);
198                    if first > last {
199                        return None;
200                    }
201                    cpus.extend(first..=last);
202                }
203                None => cpus.push(part.parse().ok()?),
204            }
205        }
206        Some(cpus)
207    }

209    /// The Arm CPU PMUs among the entries `names` of `devices`
210    /// (`/sys/bus/event_source/devices`): each named `arm*` with a `type` and
211    /// a `cpus` file. A host with one kind of core has one; a host with
212    /// several kinds (big.LITTLE) has one per kind, and each counts only on
213    /// its own CPUs. Sorted by name.
214    pub(crate) fn arm_cpu_pmus(
215        devices: &Path,
216        names: impl IntoIterator<Item = String>,
217    ) -> Vec<CpuPmu> {
218        let mut pmus: Vec<CpuPmu> = names
219            .into_iter()
220            .filter(|name| name.starts_with("arm"))
221            .filter_map(|name| {
222                let read = |file| std::fs::read_to_string(devices.join(&name).join(file)).ok();
223                let kind = read("type")?.trim().parse().ok()?;
224                let cpus = parse_cpu_list(&read("cpus")?)?;
225                Some(CpuPmu { name, kind, cpus })
226            })
227            .collect();
228        pmus.sort_by(|a, b| a.name.cmp(&b.name));
229        pmus
230    }
231
232    /// The PMUs that count on at least one of `cpus`, each with the first
233    /// such CPU.
234    pub(crate) fn covering(pmus: &[CpuPmu], cpus: &[usize]) -> Vec<(CpuPmu, usize)> {
235        pmus.iter()
236            .filter_map(|pmu| {
237                let cpu = pmu.cpus.iter().find(|cpu| cpus.contains(cpu))?;
238                Some((pmu.clone(), *cpu))
239            })
240            .collect()
241    }

243    /// One event's reads on each PMU, as (value, time enabled, time running),
244    /// combined: the sum of the values and whether the event was on a counter
245    /// for the whole run, or `None` when it was enabled but never on a
246    /// counter (a core that accepts the event and does not count it). A
247    /// task-bound event on one PMU of several is enabled whenever the task
248    /// runs, but runs only while the task is on that PMU's CPUs, so the PMUs'
249    /// times running add up to the time enabled when nothing was multiplexed.
250    pub(crate) fn combine(reads: &[(u64, u64, u64)]) -> Option<(u64, bool)> {
251        let value = reads.iter().map(|read| read.0).sum();
252        let enabled = reads.iter().map(|read| read.1).max().unwrap_or(0);
253        let running: u64 = reads.iter().map(|read| read.2).sum();
254        if enabled > 0 && running == 0 {
255            return None;
256        }
257        Some((value, running >= enabled))
258    }

260    /// One address space's pairs of one file, by mapped file range (offset,
261    /// length).
262    type RangePairs = BTreeMap<(u64, u64), u64>;
263
264    /// The exclusive pairs a run's scans found, by guest path, address space
265    /// and mapped file range.
266    #[derive(Debug, Default)]
267    pub(crate) struct PairTally {
268        pairs: BTreeMap<String, BTreeMap<(String, u64), RangePairs>>,
269    }
270
271    impl PairTally {
272        /// Take one scan of address space `space`: each mapping's guest path,
273        /// file offset, length and pairs. A mapping replaces the ranges of
274        /// the same file it overlaps, so a mapping split and rejoined by
275        /// mprotect counts once.
276        pub(crate) fn note(&mut self, space: &(String, u64), mappings: &[(String, u64, u64, u64)]) {
277            for (path, offset, len, pairs) in mappings {
278                let ranges = self
279                    .pairs
280                    .entry(path.clone())
281                    .or_default()
282                    .entry(space.clone())
283                    .or_default();
284                let end = offset.saturating_add(*len);
285                ranges.retain(|&(from, length), _| {
286                    end <= from || from.saturating_add(length) <= *offset
287                });
288                ranges.insert((*offset, *len), *pairs);
289            }
290        }
291
292        /// The pairs of each guest path: the sum over the ranges one address
293        /// space mapped, and the largest such sum over address spaces, so
294        /// that a library many processes map counts once.
295        pub(crate) fn per_path(&self) -> BTreeMap<String, u64> {
296            self.pairs
297                .iter()
298                .map(|(path, spaces)| {
299                    let most = spaces
300                        .values()
301                        .map(|ranges| ranges.values().sum::<u64>())
302                        .max()
303                        .unwrap_or(0);
304                    (path.clone(), most)
305                })
306                .collect()
307        }
308    }

415    /// A run that executed at least this many store-exclusives warns even

481        /// The end-of-run warnings of a run that records or replays
482        /// preemptions: a counter the core did not count, and, when any
483        /// store-exclusive failed, or enough executed that one could fail in
484        /// a replay, the libraries with exclusive pairs and how to remove
485        /// them.
486        pub(crate) fn warnings(&mut self, cfg: &Config) -> Vec<String> {
487            if !records_or_replays(cfg) {
488                return Vec::new();
489            }
490            let counts = self.counts().cloned();
491            let mut warnings = self.uncounted.clone();
492            let Some(Ok(counts)) = counts else {
493                return warnings;
494            };
495            if counts.failed == 0
496                && counts
497                    .executed
498                    .is_none_or(|executed| executed < WARN_EXECUTED)
499            {
500                return warnings;
501            }
502            let what = match counts.executed {
503                Some(executed) => format!(
504                    "this run executed {}, and {} failed",
505                    counted(executed, "store-exclusive"),
506                    counts.failed
507                ),
508                None => format!(
509                    "{} failed in this run",
510                    counted(counts.failed, "store-exclusive")
511                ),
512            };
513            let partial = if counts.fully_counted {
514                ""
515            } else {
516                "; not counted for the whole run (multiplexed, or the task ran on CPUs whose PMU \
517                 does not carry these events)"
518            };
519            let libraries: Vec<String> = self
520                .pairs
521                .per_path()
522                .iter()
523                .filter(|(_, pairs)| **pairs > 0)
524                .map(|(path, pairs)| format!("{path} ({pairs})"))
525                .collect();
526            let sites = if libraries.is_empty() {
527                "No executable file mapping has any, so they ran in other code".to_owned()
528            } else {
529                format!("Exclusive pairs are in {}", libraries.join(", "))
530            };
531            warnings.push(format!(
532                "{what}{partial}. A store-exclusive that fails retries its load/store-exclusive \
533                 loop and adds retired branches, so recorded preemptions may not replay. {sites}. \
534                 Build with -march=armv8.1-a or later, or with -moutline-atomics, which takes the \
535                 LSE path on cores with LSE such as Neoverse V2 and Cortex-A78"
536            ));
537            warnings
538        }

596    /// Open a user-space counting event (not sampled) of `pmu` on task `pid`,
597    /// or on the calling thread for 0, inherited by the tasks it creates when
598    /// `inherit` is set. Naming the PMU's own type binds the event to that
599    /// PMU, where `PERF_TYPE_RAW` would bind it to the first one that accepts
600    /// it.
601    fn open_event(pmu: &CpuPmu, config: u64, pid: i32, inherit: bool) -> std::io::Result<OwnedFd> {
602        let mut attr = PerfEventAttr {
603            type_: pmu.kind,
604            size: std::mem::size_of::<PerfEventAttr>() as u32,
605            config,
606            read_format: TOTAL_TIME_ENABLED | TOTAL_TIME_RUNNING,
607            flags: EXCLUDE_KERNEL | EXCLUDE_HV | if inherit { INHERIT } else { 0 },
608            ..Default::default()
609        };
610        // SAFETY: `attr` is a valid `perf_event_attr` prefix whose `size`
611        // field gives its length, and it outlives the call.
612        let fd = unsafe {
613            libc::syscall(
614                libc::SYS_perf_event_open,
615                &mut attr as *mut PerfEventAttr,
616                pid,
617                -1,
618                -1,
619                PERF_FLAG_FD_CLOEXEC,
620            )
621        };
622        if fd < 0 {
623            return Err(std::io::Error::last_os_error());
624        }
625        // SAFETY: perf_event_open returned a new descriptor nothing else owns.
626        Ok(unsafe { OwnedFd::from_raw_fd(fd as i32) })
627    }

629    /// An event's value, time enabled and time running.
630    fn read_event(event: &OwnedFd) -> std::io::Result<(u64, u64, u64)> {
631        let mut values = [0u64; 3];
632        // SAFETY: reads at most 24 bytes into a buffer of 24.
633        let read = unsafe { libc::read(event.as_raw_fd(), values.as_mut_ptr().cast(), 24) };
634        match read {
635            24 => Ok((values[0], values[1], values[2])),
636            read if read < 0 => Err(std::io::Error::last_os_error()),
637            _ => Err(std::io::Error::other("short read of a perf event")),
638        }
639    }

641    /// The CPUs task `pid` may run on.
642    fn affinity(pid: i32) -> std::io::Result<Vec<usize>> {
643        // SAFETY: an all-zero `cpu_set_t` is an empty set.
644        let mut set: libc::cpu_set_t = unsafe { std::mem::zeroed() };
645        // SAFETY: `set` is a `cpu_set_t` of the size passed.
646        if unsafe { libc::sched_getaffinity(pid, std::mem::size_of_val(&set), &mut set) } != 0 {
647            return Err(std::io::Error::last_os_error());
648        }
649        // SAFETY: every CPU number is below `CPU_SETSIZE`.
650        Ok((0..libc::CPU_SETSIZE as usize)
651            .filter(|cpu| unsafe { libc::CPU_ISSET(*cpu, &set) })
652            .collect())
653    }

655    /// Run the calling thread on `cpus` only.
656    fn set_affinity(cpus: &[usize]) -> std::io::Result<()> {
657        // SAFETY: an all-zero `cpu_set_t` is an empty set.
658        let mut set: libc::cpu_set_t = unsafe { std::mem::zeroed() };
659        for &cpu in cpus {
660            // SAFETY: the caller's CPU numbers came from the kernel's CPU list.
661            unsafe { libc::CPU_SET(cpu, &mut set) };
662        }
663        // SAFETY: `set` is a `cpu_set_t` of the size passed.
664        if unsafe { libc::sched_setaffinity(0, std::mem::size_of_val(&set), &set) } != 0 {
665            return Err(std::io::Error::last_os_error());
666        }
667        Ok(())
668    }

675    /// What a PMU counted of the probe's store-exclusives.
676    struct Probe {
677        /// `STREX_FAIL_SPEC`'s increase over [`PROBE_LOOPS`] failing ones.
678        failed: u64,
679        /// How many of those the CPU itself reported failed.
680        observed_failed: u64,
681        /// `STREX_SPEC`'s increase over all [`PROBE_LOOPS`] x 2, if it opens.
682        executed: Option<u64>,
683    }
684
685    const PROBE_LOOPS: u64 = 1000;
686
687    /// Check what `pmu` counts, on the calling thread of this (tracer)
688    /// process pinned to `cpu`, one of its CPUs, and given back its own CPUs
689    /// after: [`PROBE_LOOPS`] store-exclusives that fail (`LDXR`; `CLREX`,
690    /// which clears the exclusive monitor; `STXR`), then as many exclusive
691    /// increments that succeed.
692    ///
693    /// Not on a new thread: the tracer runs in the guest's PID namespace, so
694    /// a thread it starts takes the next PID there, and every guest thread
695    /// started after it would get another ID than in a run that does not
696    /// probe. Runs that record preemptions (`--record-preemptions-to`) or
697    /// replay them (`--replay-preemptions-from`) probe; a replay of such a
698    /// run's schedule (`--replay-schedule-from`) does not, so it would lose
699    /// its threads at their first clone.
700    fn probe(pmu: &CpuPmu, cpu: usize) -> Result<Probe, String> {
701        /// Gives the calling thread back its CPUs when the probe ends, also
702        /// when it unwinds.
703        struct Restore(Vec<usize>);
704        impl Drop for Restore {
705            fn drop(&mut self) {
706                if let Err(error) = set_affinity(&self.0) {
707                    warn!(
708                        "[store-exclusive] cannot give the tracer thread back its CPUs after \
709                         the probe, so it stays on one CPU: {error}"
710                    );
711                }
712            }
713        }
714        let own = affinity(0)
715            .map_err(|error| format!("cannot read the tracer thread's CPU affinity: {error}"))?;
716        let _restore = Restore(own);
717        probe_on(pmu, cpu)
718    }
719
720    fn probe_on(pmu: &CpuPmu, cpu: usize) -> Result<Probe, String> {
721        pin(cpu).map_err(|error| {
722            format!(
723                "cannot run the store-exclusive probe on CPU {cpu} of PMU {}: {error}",
724                pmu.name
725            )
726        })?;
727        let failed = open_event(pmu, STREX_FAIL_SPEC, 0, false).map_err(|error| {
728            format!(
729                "perf_event_open of STREX_FAIL_SPEC (raw 0x6e) on PMU {} failed: {error}",
730                pmu.name
731            )
732        })?;
733        let executed = open_event(pmu, STREX_SPEC, 0, false).ok();
734        let value = |event: &OwnedFd| read_event(event).map(|read| read.0).unwrap_or(0);
735        let (failed_before, executed_before) = (value(&failed), executed.as_ref().map(value));
736        let mut word = 0u64;
737        let mut observed_failed = 0u64;
738        for _ in 0..PROBE_LOOPS {
739            let status: u64;
740            // SAFETY: an exclusive load and store of a local word.
741            unsafe {
742                core::arch::asm!(
743                    "ldxr {value}, [{word}]",
744                    "clrex",
745                    "stxr {status:w}, {value}, [{word}]",
746                    word = in(reg) &raw mut word,
747                    value = out(reg) _,
748                    status = out(reg) status,
749                    options(nostack),
750                );
751            }
752            observed_failed += status & 1;
753        }
754        for _ in 0..PROBE_LOOPS {
755            // SAFETY: an exclusive increment of a local word.
756            unsafe {
757                core::arch::asm!(
758                    "2:",
759                    "ldxr {value}, [{word}]",
760                    "add {value}, {value}, #1",
761                    "stxr {status:w}, {value}, [{word}]",
762                    "cbnz {status:w}, 2b",
763                    word = in(reg) &raw mut word,
764                    value = out(reg) _,
765                    status = out(reg) _,
766                    options(nostack),
767                );
768            }
769        }
770        std::hint::black_box(word);
771        Ok(Probe {
772            failed: value(&failed).saturating_sub(failed_before),
773            observed_failed,
774            executed: executed
775                .as_ref()
776                .zip(executed_before)
777                .map(|(event, before)| value(event).saturating_sub(before)),
778        })
779    }

794    impl Counters {
795        fn open(root: i32) -> Result<Self, String> {
796            let cpus = affinity(root)
797                .map_err(|error| format!("cannot read the guest's CPU affinity: {error}"))?;
798            let mut names = Vec::new();
799            crate::util::find_in_directory(Path::new(DEVICES), |name| {
800                names.push(name.to_string_lossy().into_owned());
801                None::<()>
802            })
803            .map_err(|error| format!("cannot list the PMUs in {DEVICES}: {error}"))?;
804            let pmus = arm_cpu_pmus(Path::new(DEVICES), names);
805            let covering = covering(&pmus, &cpus);
806            if covering.is_empty() {
807                return Err(format!(
808                    "no Arm CPU PMU in {DEVICES} counts on the guest's CPUs"
809                ));
810            }
811            let mut count_executed = true;
812            for (pmu, cpu) in &covering {
813                let probe = probe(pmu, *cpu)?;
814                let executed = probe
815                    .executed
816                    .map_or("does not open".to_owned(), |executed| {
817                        format!("counted {executed} of {}", 2 * PROBE_LOOPS)
818                    });
819                debug!(
820                    "[store-exclusive] probe of PMU {} on CPU {cpu}: STREX_FAIL_SPEC counted {} of \
821                     {PROBE_LOOPS} failing store-exclusives (the CPU reported {} failed); \
822                     STREX_SPEC {executed}",
823                    pmu.name, probe.failed, probe.observed_failed
824                );
825                if probe.failed == 0 {
826                    return Err(format!(
827                        "store-exclusive counting unavailable on this core: STREX_FAIL_SPEC (raw \
828                         0x6e) on PMU {} counted none of {PROBE_LOOPS} failing store-exclusives \
829                         on CPU {cpu}",
830                        pmu.name
831                    ));
832                }
833                count_executed &= probe.executed.is_some_and(|executed| executed > 0);
834            }
835            let open = |pmu: &CpuPmu, config, name| {
836                open_event(pmu, config, root, true).map_err(|error| {
837                    format!(
838                        "perf_event_open of {name} on PMU {} for the guest failed: {error}",
839                        pmu.name
840                    )
841                })
842            };
843            covering
844                .iter()
845                .map(|(pmu, _)| {
846                    Ok(PmuEvents {
847                        failed: open(pmu, STREX_FAIL_SPEC, "STREX_FAIL_SPEC")?,
848                        executed: if count_executed {
849                            Some(open(pmu, STREX_SPEC, "STREX_SPEC")?)
850                        } else {
851                            None
852                        },
853                    })
854                })
855                .collect::<Result<_, String>>()
856                .map(Self)
857        }

859        /// Read the counts. A counter that was enabled but never on a
860        /// counter reports nothing, and adds a warning to `uncounted`.
861        fn read(self, uncounted: &mut Vec<String>) -> Result<StoreExclusiveCounts, String> {
862            let reads = |events: Vec<&OwnedFd>, name: &str| {
863                events
864                    .into_iter()
865                    .map(read_event)
866                    .collect::<std::io::Result<Vec<_>>>()
867                    .map_err(|error| format!("reading {name} failed: {error}"))
868            };
869            let never = |name: &str| {
870                format!(
871                    "{name} was enabled for the run but never on a counter: this core accepts \
872                     the event and does not count it"
873                )
874            };
875            let failed = reads(
876                self.0.iter().map(|pmu| &pmu.failed).collect(),
877                "STREX_FAIL_SPEC",
878            )?;
879            let Some((failed, failed_whole)) = combine(&failed) else {
880                let reason = never("STREX_FAIL_SPEC (raw 0x6e)");
881                uncounted.push(format!("store-exclusives were not counted: {reason}"));
882                return Err(reason);
883            };
884            let executed: Option<Vec<&OwnedFd>> =
885                self.0.iter().map(|pmu| pmu.executed.as_ref()).collect();
886            let executed = match executed {
887                Some(events) => match combine(&reads(events, "STREX_SPEC")?) {
888                    Some(combined) => Some(combined),
889                    None => {
890                        uncounted.push(format!(
891                            "executed store-exclusives are not reported: {}",
892                            never("STREX_SPEC (raw 0x6f)")
893                        ));
894                        None
895                    }
896                },
897                None => None,
898            };
899            Ok(StoreExclusiveCounts {
900                failed,
901                executed: executed.map(|(executed, _)| executed),
902                fully_counted: failed_whole && executed.is_none_or(|(_, whole)| whole),
903            })
904        }
905    }

917        ranges: Option<Vec<(u64, u64, bool)>>,

919        /// the thread that scanned it first.

924        /// Whether `range` may hold a file mapping not scanned yet: the
925        /// ranges are unknown, or it overlaps one recorded unscanned.
926        fn needs_scan(&self, (start, end): (u64, u64)) -> bool {
927            self.ranges.as_ref().is_none_or(|ranges| {
928                ranges
929                    .iter()
930                    .any(|&(from, to, scanned)| !scanned && from < end && start < to)

935        fn add_unscanned(&mut self, (start, end): (u64, u64)) {

937                ranges.push((start, end, false));
938            }
939        }

942        /// counts as scanned: every call that made it executable scanned it.

944            self.ranges = Some(
945                maps.iter()
946                    .map(|map| {
947                        (

949                            map.address.1,

958        /// pedigree of the first thread that scanned it, which no other
959        /// thread of the run has, and its exec generation. A fork's copy of

961        fn space(&mut self, mm: MmId, scanner: impl FnOnce() -> String) -> (String, u64) {
962            let pedigree = match &self.owner {

964                _ => {

983    /// A scan result. A count stands while the file keeps the size and
984    /// modification time it was taken at; a failure stands for good, so that
985    /// each mapping of the file is reported the same way.
986    #[derive(Clone, Copy, Debug)]
987    enum Scan {
988        Counted {
989            size: u64,
990            mtime: (i64, i64),
991            pairs: u64,
992        },
993        Failed,
994    }

996    /// Scan results for this Hermit process.
997    static SCANS: Mutex<BTreeMap<FileRange, Scan>> = Mutex::new(BTreeMap::new());

999    /// The exclusive pairs of one mapping, or `None` where it cannot be
1000    /// scanned.
1001    fn scan_mapping(tid: Pid, path: &Path, key: FileRange) -> Option<u64> {
1002        let cached = SCANS.lock().unwrap().get(&key).copied();
1003        if let Some(Scan::Failed) = cached {
1004            return None;
1005        }
1006        let result = (|| {
1007            let file = open_in_root(tid, path)?;
1008            let metadata = file.metadata()?;
1009            // The maps device is the superblock's, which on btrfs is not the
1010            // subvolume device `stat` gives, so only the inode is compared.
1011            if metadata.ino() != key.inode {
1012                return Err(std::io::Error::other(
1013                    "the name now leads to another file than the mapped one",
1014                ));
1015            }
1016            let (size, mtime) = (metadata.size(), (metadata.mtime(), metadata.mtime_nsec()));
1017            let pairs = match cached {
1018                Some(Scan::Counted {
1019                    size: was,
1020                    mtime: then,
1021                    pairs,
1022                }) if (was, then) == (size, mtime) => pairs,
1023                _ => count_file(&file, size, &key)?,
1024            };
1025            Ok(Scan::Counted { size, mtime, pairs })
1026        })();
1027        let scan = result.unwrap_or_else(|error: std::io::Error| {
1028            debug!(
1029                "[store-exclusive] {} not scanned through /proc/{}/root: {error}",
1030                path.display(),
1031                tid.as_raw()
1032            );
1033            Scan::Failed
1034        });
1035        SCANS.lock().unwrap().insert(key, scan);
1036        match scan {
1037            Scan::Counted { pairs, .. } => Some(pairs),
1038            Scan::Failed => None,
1039        }
1040    }

1042    /// Open `path` as the guest names it: resolved inside the guest's root
1043    /// (`RESOLVE_IN_ROOT`, from an `O_PATH` descriptor of `/proc/<tid>/root`),
1044    /// so that neither a symbolic link nor `..` leads out of it.
1045    fn open_in_root(tid: Pid, path: &Path) -> std::io::Result<File> {
1046        let root = OpenOptions::new()
1047            .read(true)
1048            .custom_flags(libc::O_PATH | libc::O_DIRECTORY)
1049            .open(format!("/proc/{}/root", tid.as_raw()))?;
1050        let relative = path.strip_prefix("/").unwrap_or(path);
1051        let relative =
1052            CString::new(relative.as_os_str().as_bytes()).map_err(std::io::Error::other)?;
1053        /// `struct open_how` (`OPEN_HOW_SIZE_VER0`).
1054        #[repr(C)]
1055        struct OpenHow {
1056            flags: u64,
1057            mode: u64,
1058            resolve: u64,
1059        }
1060        let how = OpenHow {
1061            flags: (libc::O_RDONLY | libc::O_CLOEXEC) as u64,
1062            mode: 0,
1063            resolve: libc::RESOLVE_IN_ROOT,
1064        };
1065        // SAFETY: a valid descriptor, a NUL-terminated path and an `open_how`
1066        // of the size passed, all alive for the call.
1067        let fd = unsafe {
1068            libc::syscall(
1069                libc::SYS_openat2,
1070                root.as_raw_fd(),
1071                relative.as_ptr(),
1072                &how as *const OpenHow,
1073                std::mem::size_of::<OpenHow>(),
1074            )
1075        };
1076        if fd < 0 {
1077            return Err(std::io::Error::last_os_error());
1078        }
1079        // SAFETY: openat2 returned a new descriptor nothing else owns.
1080        Ok(unsafe { File::from_raw_fd(fd as i32) })
1081    }

1083    /// Count the exclusive pairs in the mapped range of a file of `size`
1084    /// bytes, in its executable ELF sections only.
1085    fn count_file(file: &File, size: u64, key: &FileRange) -> std::io::Result<u64> {
1086        // A mapping can extend past the end of its file.
1087        let mapped = (key.offset, size.min(key.offset.saturating_add(key.len)));
1088        let mut pairs = 0;
1089        for (start, end) in code_ranges(file, mapped)? {
1090            // A pair cannot span two sections.
1091            let mut section = PairCounter::default();
1092            read_words(file, start, end, &mut section)?;
1093            pairs += section.pairs();
1094        }
1095        Ok(pairs)
1096    }

1098    /// The parts of `mapped` (a file range) that hold instructions: its
1099    /// intersections with the file's executable ELF sections, or all of it
1100    /// when the file has no section headers.
1101    fn code_ranges(file: &File, mapped: (u64, u64)) -> std::io::Result<Vec<(u64, u64)>> {
1102        let mut header = [0u8; 64];
1103        let table = match file.read_exact_at(&mut header, 0) {
1104            Ok(()) => elf_section_table(&header),
1105            Err(error) if error.kind() == std::io::ErrorKind::UnexpectedEof => None,
1106            Err(error) => return Err(error),
1107        };
1108        let Some((offset, count)) = table else {
1109            return Ok(vec![mapped]);
1110        };
1111        let mut table = vec![0u8; count * 64];
1112        file.read_exact_at(&mut table, offset)?;
1113        Ok(executable_sections(&table)
1114            .into_iter()
1115            .filter_map(|(offset, size)| {
1116                let start = offset.max(mapped.0);
1117                let end = offset.saturating_add(size).min(mapped.1);
1118                (start < end).then_some((start, end))
1119            })
1120            .collect())
1121    }

1123    /// Feed `counter` the words of `file` in `start..end`.
1124    fn read_words(
1125        file: &File,
1126        start: u64,
1127        end: u64,
1128        counter: &mut PairCounter,
1129    ) -> std::io::Result<()> {
1130        let mut buffer = vec![0u8; (1 << 20).min((end - start) as usize)];
1131        let mut at = start;
1132        while at < end {
1133            let chunk =
1134                usize::try_from(end - at).map_or(buffer.len(), |left| left.min(buffer.len()));
1135            file.read_exact_at(&mut buffer[..chunk], at)?;
1136            counter.push_bytes(&buffer[..chunk]);
1137            at += chunk as u64;
1138        }
1139        Ok(())
1140    }

1143    /// them for `None`): log each with pairs, or that could not be scanned,
1144    /// at INFO, and report the pairs to the run's tally.
1145    async fn report_pairs<G, T>(guest: &mut G, range: Option<(u64, u64)>)
1146    where
1147        G: Guest<Detcore<T>>,
1148        T: RecordOrReplay,
1149    {

1151        debug!(
1152            "[detcore, dtid {}] store-exclusive scan of {:x?}",
1153            dettid,
1154            range.map_or(0..u64::MAX, |(start, end)| start..end)
1155        );
1156        let maps = match crate::procmaps::from_pid(guest.tid(), |map| {
1157            map.inode != 0 && matches!(map.pathname, MMapPath::Path(_))
1158        }) {
1159            Ok(maps) => maps,
1160            Err(error) => {
1161                debug!(
1162                    "[store-exclusive] cannot read /proc/{}/maps: {error}",
1163                    guest.tid().as_raw()
1164                );
1165                return;
1166            }
1167        };
1168        let space = {
1169            let thread = guest.thread_state();
1170            let mut metadata = thread.memory_metadata.lock().unwrap();
1171            metadata.exclusive_scan.reseed(&maps);
1172            metadata
1173                .exclusive_scan
1174                .space(thread.mm_id, || thread.pedigree.to_string())
1175        };
1176        let mut mappings = Vec::new();
1177        for map in &maps {
1178            let MMapPath::Path(path) = &map.pathname else {
1179                continue;
1180            };
1181            let (start, end) = map.address;
1182            if !map.perms.contains(MMPermissions::EXECUTE)
1183                || range.is_some_and(|(from, to)| end <= from || to <= start)
1184            {
1185                continue;
1186            }
1187            let key = FileRange {
1188                dev: map.dev,
1189                inode: map.inode,
1190                offset: map.offset,
1191                len: end - start,
1192            };
1193            let shown = path.display().to_string();
1194            match scan_mapping(guest.tid(), path, key) {
1195                Some(0) => {}
1196                Some(pairs) => {
1197                    info!(
1198                        "[detcore, dtid {}] {} in executable mapping {} (offset {:#x}, {:#x} bytes)",
1199                        dettid,
1200                        counted(pairs, "store-exclusive (LL/SC) pair"),
1201                        shown,
1202                        key.offset,
1203                        key.len
1204                    );
1205                    mappings.push((shown, key.offset, key.len, pairs));
1206                }
1207                None => info!(
1208                    "[detcore, dtid {}] store-exclusive (LL/SC) pairs not scanned in executable \
1209                     mapping {} (offset {:#x}, {:#x} bytes)",
1210                    dettid, shown, key.offset, key.len
1211                ),
1212            }
1213        }
1214        if !mappings.is_empty() {
1215            send(guest, Report::Pairs { space, mappings }).await;
1216        }
1217    }

1230    /// After a successful exec, while the thread is stopped at the start of
1231    /// the new image: in the root guest process, open the run's counters if
1232    /// this run counts (only its first exec does), so that they count the
1233    /// guest's own images and no launcher code; then scan every executable
1234    /// file mapping of the image (the program and its interpreter).
1235    pub(crate) async fn report_after_exec<G, T>(guest: &mut G)
1236    where
1237        G: Guest<Detcore<T>>,
1238        T: RecordOrReplay,
1239    {
1240        if counts_store_exclusives(guest.config()) && guest.is_root_process() {
1241            let root = guest.tid().as_raw();
1242            send(guest, Report::Attach { root }).await;
1243        }
1244        if reads_guest_tasks(guest.config()) {
1245            report_pairs(guest, None).await;
1246        }
1247    }

1249    /// After a successful syscall: the address range to scan, for an `mmap`
1250    /// that maps `PROT_EXEC` from a file, or an `mprotect` that adds
1251    /// `PROT_EXEC` where a file mapping may not be scanned yet. Anything
1252    /// else reads no maps: a file mapping made without `PROT_EXEC` is only
1253    /// recorded, and `PROT_EXEC` on anonymous memory (a JIT) is skipped.
1254    pub(crate) fn range_to_scan<G, T>(guest: &G, call: &Syscall, result: i64) -> Option<(u64, u64)>
1255    where
1256        G: Guest<Detcore<T>>,
1257        T: RecordOrReplay,
1258    {
1259        if !reads_guest_tasks(guest.config()) {
1260            return None;
1261        }
1262        let span = |start: u64, len: usize| (start, start.saturating_add(len.max(1) as u64));
1263        match call {
1264            Syscall::Mmap(mmap)
1265                if !mmap.flags().contains(MapFlags::MAP_ANONYMOUS) && mmap.fd() >= 0 =>
1266            {
1267                let range = span(result as u64, mmap.len());
1268                if mmap.prot().contains(ProtFlags::PROT_EXEC) {
1269                    return Some(range);
1270                }
1271                let metadata = &guest.thread_state().memory_metadata;
1272                metadata.lock().unwrap().exclusive_scan.add_unscanned(range);
1273                None
1274            }
1275            Syscall::Mprotect(mprotect) if mprotect.protection().contains(ProtFlags::PROT_EXEC) => {
1276                let start = mprotect.addr().map_or(0, |addr| addr.as_raw() as u64);
1277                let range = span(start, mprotect.len());
1278                let metadata = &guest.thread_state().memory_metadata;
1279                let needed = metadata.lock().unwrap().exclusive_scan.needs_scan(range);
1280                needed.then_some(range)
1281            }
1282            _ => None,
1283        }
1284    }

1286    /// After a [`range_to_scan`]: the executable file mappings in that range.
1287    pub(crate) async fn report_new_mapping<G, T>(guest: &mut G, range: (u64, u64))
1288    where
1289        G: Guest<Detcore<T>>,
1290        T: RecordOrReplay,
1291    {
1292        report_pairs(guest, Some(range)).await;
1293    }
