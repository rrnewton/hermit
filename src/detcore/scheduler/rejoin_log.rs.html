9//! The rejoin log: the scheduler turn at which each backgrounded call
10//! rejoined the run queue during `hermit record`, so that `hermit replay`
11//! readmits it at the same turn.
12//!
13//! A backgrounded call (`BlockingExternalIO`, or an `rt_sigsuspend` outside the
14//! scheduler) runs in the host kernel while guest turns continue. Under record,
15//! `step2c_process_io_blockers` readmits it at the first scheduler pass that
16//! finds it finished, so the turn depends on host timing. Replay serves many
17//! of these calls from the recording, and they finish at a different moment,
18//! so readmitting them "when finished" can change the schedule
19//! (<https://github.com/rrnewton/hermit/pull/3908>). Record therefore logs
20//! each readmission, and replay obeys the log instead of host timing.
21//!
22//! `Scheduler::turn` advances only on a committed or skipped turn, never while
23//! a scheduler pass spins, so it names the same point in both runs as long as
24//! the two schedules agree up to it. It does not advance when the scheduler,
25//! with nothing to run, moves virtual time to the next timer, so each entry
26//! also carries `Scheduler::empty_queue_wakes`, the count of those moves.
27//! Record may readmit a call either before or after such a move within one
28//! turn, and the pair tells replay which.
29//!
30//! The file, `rejoins` in the recording directory, is text: a header line, then
31//! one line per readmission, in order:
32//!
33//! ```text
34//! hermit-rejoins 1
35//! <turn> <wakes> <dettid>:<syscall ordinal> [<dettid>:<syscall ordinal> ...]
36//! ```
37//!
38//! The syscall ordinal is the call's `ExternalOpId::sequence`, so replay can
39//! check that the parked call is the one the recording readmitted. Each line
40//! is written with one `write` as it happens, and a final line without its
41//! newline means the recording was cut short; replay refuses such a log.

55/// Name of the rejoin log within the recording directory.
56pub(crate) const REJOIN_LOG_NAME: &str = "rejoins";

58const HEADER: &str = "hermit-rejoins 1";

60/// One readmission: the calls `step2c_process_io_blockers` readmitted together,
61/// in `DetTid` order, and the point at which it did.
62#[derive(
63    Debug,
64    Clone,
65    PartialEq,
66    Eq,
67    Hash,
68    serde::Serialize,
69    serde::Deserialize
70)]
71pub(crate) struct Rejoin {
72    pub(crate) turn: u64,
73    /// `Scheduler::empty_queue_wakes` at the readmission.
74    pub(crate) wakes: u64,
75    /// Each readmitted thread and the `ExternalOpId::sequence` of its call.
76    pub(crate) calls: Vec<(DetTid, u64)>,
77    /// The timed events the scheduler popped in this turn before the
78    /// readmission, in order, each with the deadline it popped at
79    /// (`Scheduler::note_timed_pop`), in a `hermit run` preemption record. A
80    /// schedule replay readmits at the recorded turn once it has made these
81    /// pops, by whatever step, rather than at the recorded wake count, which
82    /// its lagging virtual time can reach by another step
83    /// (`Scheduler::rejoin_place`). `None` in `hermit record`'s text log,
84    /// which does not carry it, and in records written before the field
85    /// existed, which replay by the wake count as before.
86    #[serde(default, skip_serializing_if = "Option::is_none")]
87    pub(crate) popped: Option<Vec<PoppedEvent>>,
88}

90/// A timed event, as a readmission names one it followed
91/// ([`Rejoin::popped`]): the thread whose timed wait ended, or the timer that
92/// fired.
93#[derive(
94    Debug,
95    Clone,
96    Copy,
97    PartialEq,
98    Eq,
99    PartialOrd,
100    Ord,
101    Hash,
102    serde::Serialize,
103    serde::Deserialize
104)]
105#[serde(rename_all = "snake_case")]
106pub(crate) enum TimedEventId {
107    /// A thread's timed wait: a sleep, or a wait with a timeout.
108    Thread(DetTid),
109    /// A process's `ITIMER_REAL` (`alarm`, `setitimer`).
110    Alarm(DetPid),
111    /// A process's POSIX timer, by its timer ID.
112    Posix(DetPid, i32),
113    /// The `SIGCHLD` of a child's exit, to its parent.
114    ChildExit { child: DetPid, parent: DetPid },
115}
116
117impl From<TimedEvent> for TimedEventId {
118    fn from(event: TimedEvent) -> Self {
119        match event {
120            TimedEvent::ThreadEvt(tid) => TimedEventId::Thread(tid),
121            TimedEvent::SignalEvt(SignalTimerId::Alarm(pid), _, _) => TimedEventId::Alarm(pid),
122            TimedEvent::SignalEvt(SignalTimerId::Posix(pid, timer), _, _) => {
123                TimedEventId::Posix(pid, timer)
124            }
125            TimedEvent::SignalEvt(SignalTimerId::ChildExit { child, parent }, _, _) => {
126                TimedEventId::ChildExit { child, parent }
127            }
128        }
129    }
130}

132/// One pop of a timed event: the event, and the deadline it popped at. A
133/// replay has made it once that event has no pending entry due at or before
134/// that deadline (`Scheduler::pop_made`). A periodic timer fires again under
135/// the same name, and how many of its periods fire depends on virtual time
136/// (a late timer skips the periods it missed), so the deadline, not a count
137/// of pops, tells which firing it was.
138#[derive(
139    Debug,
140    Clone,
141    Copy,
142    PartialEq,
143    Eq,
144    Hash,
145    serde::Serialize,
146    serde::Deserialize
147)]
148pub(crate) struct PoppedEvent {
149    pub(crate) event: TimedEventId,
150    pub(crate) deadline: LogicalTime,
151}
152
153impl Rejoin {
154    /// Where the readmission happened; both parts only grow during a run.
155    pub(crate) fn point(&self) -> (u64, u64) {
156        (self.turn, self.wakes)
157    }
158
159    fn to_line(&self) -> String {
160        let mut line = format!("{} {}", self.turn, self.wakes);
161        for (dtid, sequence) in &self.calls {
162            line.push_str(&format!(" {}:{}", dtid.as_raw(), sequence));
163        }
164        line.push('\n');
165        line
166    }
167
168    fn parse(line: &str) -> Result<Self, String> {
169        let mut words = line.split(' ');
170        let turn = words
171            .next()
172            .and_then(|word| word.parse::<u64>().ok())
173            .ok_or_else(|| format!("bad turn in {line:?}"))?;
174        let wakes = words
175            .next()
176            .and_then(|word| word.parse::<u64>().ok())
177            .ok_or_else(|| format!("bad wake count in {line:?}"))?;
178        let mut calls: Vec<(DetTid, u64)> = Vec::new();
179        for word in words {
180            let call = word
181                .split_once(':')
182                .and_then(|(dtid, sequence)| {
183                    Some((
184                        DetTid::from_raw(dtid.parse::<i32>().ok()?),
185                        sequence.parse::<u64>().ok()?,
186                    ))
187                })
188                .ok_or_else(|| format!("bad call {word:?} in {line:?}"))?;
189            if calls.last().is_some_and(|(last, _)| *last >= call.0) {
190                return Err(format!("threads out of order in {line:?}"));
191            }
192            calls.push(call);
193        }
194        if calls.is_empty() {
195            return Err(format!("no calls in {line:?}"));
196        }
197        Ok(Rejoin {
198            turn,
199            wakes,
200            calls,
201            popped: None,
202        })
203    }
204}

206/// Record side: appends each readmission to the log as it happens.
207#[derive(Debug)]
208pub(crate) struct RejoinWriter {
209    file: File,
210}
211
212impl RejoinWriter {
213    /// Create the log in `dir`, header included, so that a recording with no
214    /// readmissions still has one.
215    pub(crate) fn create(dir: &Path) -> io::Result<Self> {
216        let mut file = File::create(dir.join(REJOIN_LOG_NAME))?;
217        file.write_all(format!("{HEADER}\n").as_bytes())?;
218        Ok(RejoinWriter { file })
219    }
220
221    /// Append one readmission with a single write.
222    pub(crate) fn append(&mut self, rejoin: &Rejoin) -> io::Result<()> {
223        self.file.write_all(rejoin.to_line().as_bytes())
224    }
225}

227/// Whether `rejoins` is a log a replay can follow: its entries must be in
228/// the order the recording made them, since replay walks them once.
229pub(crate) fn ordered(rejoins: &[Rejoin]) -> Result<(), String> {
230    for pair in rejoins.windows(2) {
231        if pair[0].point() > pair[1].point() {
232            return Err(format!(
233                "readmissions out of order: turn {} comes after turn {}",
234                pair[1].turn, pair[0].turn
235            ));
236        }
237    }
238    Ok(())
239}

241/// Replay side: read the log the recording in `dir` wrote.
242pub(crate) fn read(dir: &Path) -> Result<VecDeque<Rejoin>, String> {
243    let path = dir.join(REJOIN_LOG_NAME);
244    let text =
245        std::fs::read_to_string(&path).map_err(|err| format!("{}: {err}", path.display()))?;
246    parse(&text)
247}
248
249fn parse(text: &str) -> Result<VecDeque<Rejoin>, String> {
250    let mut lines = text.split_inclusive('\n');
251    if lines.next() != Some(&format!("{HEADER}\n")) {
252        return Err(format!("the log does not start with {HEADER:?}"));
253    }
254    let mut rejoins: VecDeque<Rejoin> = VecDeque::new();
255    for line in lines {
256        let line = line
257            .strip_suffix('\n')
258            .ok_or_else(|| format!("the last line, {line:?}, is incomplete"))?;
259        let rejoin = Rejoin::parse(line)?;
260        if rejoins
261            .back()
262            .is_some_and(|last| last.point() > rejoin.point())
263        {
264            return Err(format!("readmissions out of order at {line:?}"));
265        }
266        rejoins.push_back(rejoin);
267    }
268    Ok(rejoins)
269}

271/// Replay could not follow the rejoin log, so it would readmit a backgrounded
272/// call at a point the recording did not; the scheduler loop ends the run.
273#[derive(Debug, Clone, PartialEq, Eq)]
274pub(crate) struct RejoinRefusal {
275    pub(crate) reason: String,
276}
277
278impl std::fmt::Display for RejoinRefusal {
279    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
280        write!(
281            f,
282            "hermit replay refused to continue: {}, so a backgrounded call would rejoin \
283             the schedule at a different point than in the recording",
284            self.reason
285        )
286    }
287}
288
289impl std::error::Error for RejoinRefusal {}

291#[cfg(test)]
292mod tests {
293    use super::*;
294
295    fn rejoin(turn: u64, wakes: u64, calls: &[(i32, u64)]) -> Rejoin {
296        Rejoin {
297            turn,
298            wakes,
299            calls: calls
300                .iter()
301                .map(|(dtid, sequence)| (DetTid::from_raw(*dtid), *sequence))
302                .collect(),
303            popped: None,
304        }
305    }
306
307    #[test]
308    fn written_log_reads_back() {
309        let dir = tempfile::tempdir().unwrap();
310        let mut writer = RejoinWriter::create(dir.path()).unwrap();
311        assert_eq!(read(dir.path()).unwrap(), VecDeque::new());
312        let written = vec![
313            rejoin(4, 0, &[(3, 17)]),
314            rejoin(4, 1, &[(3, 19), (5, 2)]),
315            rejoin(6, 1, &[(5, 3)]),
316        ];
317        for entry in &written {
318            writer.append(entry).unwrap();
319        }
320        assert_eq!(read(dir.path()).unwrap(), VecDeque::from(written));
321    }
322
323    #[test]
324    fn malformed_logs_are_refused() {
325        let missing = tempfile::tempdir().unwrap();
326        assert!(read(missing.path()).is_err());
327        for text in [
328            "",
329            "hermit-rejoins 2\n",
330            "hermit-rejoins 1\n7 0 3:4",
331            "hermit-rejoins 1\n7 0\n",
332            "hermit-rejoins 1\n7 3:4\n",
333            "hermit-rejoins 1\n7 0 3\n",
334            "hermit-rejoins 1\n7 0 5:1 3:4\n",
335            "hermit-rejoins 1\n7 0 3:4\n6 0 3:5\n",
336            "hermit-rejoins 1\n7 1 3:4\n7 0 3:5\n",
337        ] {
338            assert!(parse(text).is_err(), "{text:?}");
339        }
340    }
