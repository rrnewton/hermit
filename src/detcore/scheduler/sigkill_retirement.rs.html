8//! Retiring a SIGKILLed process at the kill turn on ptrace
9//! (<https://github.com/rrnewton/hermit/issues/3994>).
10//!
11//! On a backend whose only report of a thread's death is its own
12//! deregistration, a process killed by another guest process used to leave the
13//! scheduler whenever that report arrived, at a moment the host chose. Here the
14//! scheduler's view changes at one committed point instead, the sending turn:
15//!
16//! 1. **Reserved.** Before the physical send, the sender reserves the victims:
17//!    every live thread of the target process except its own, with the
18//!    address-space identity each registered with (`Scheduler::thread_mm`).
19//!    A victim's consuming receipt (its deregistration) that arrives now is
20//!    accounted but held: it retires nothing ([`Scheduler::route_sigkill_receipt`]).
21//! 2. **Committed.** After a successful send, the scheduler retires each victim
22//!    in tid order (`logically_kill_thread`). A victim with a filed request is
23//!    never answered or granted again: the kernel ends the tracee, and the
24//!    backend's exit path ends its task without the tool injecting anything. A
25//!    failed send unwinds the reservation: a held receipt then takes the
26//!    ordinary path, as if no kill had been prepared.
27//! 3. **Physical exit accounted.** Each committed victim's first consuming
28//!    receipt is accounted as usual; one held since state 1 already was.
29//! 4. **Fence released.** Turn selection waits until every committed victim is
30//!    in state 3 ([`Scheduler::sigkill_fence_holds`]), so the physical death is
31//!    settled before the next guest instruction; host timing decides only how
32//!    long the pause lasts.
33//!
34//! A thread's own delivery of a signal whose default action ends its process
35//! kills the process's other threads the same way, so they are retired as
36//! committed victims in that delivery turn
37//! ([`Scheduler::retire_fatal_delivery_victims`]).

52/// A victim of a reserved send: a thread, its registered process, and the
53/// address-space identity it registered with.

92/// Why a send is not reserved, so it keeps the ordinary path.
93#[derive(Debug, Clone, Copy, PartialEq, Eq)]
94pub(crate) enum SigkillRefusal {
95    /// The backend reports deaths some other way, or kills through its own
96    /// path (`note_process_sigkill`).
97    Backend,
98    /// The target is not a live registered process.
99    NoTarget,
100    /// A victim has no recorded address-space identity.
101    UnknownAddressSpace(DetTid),
102    /// A victim is already reserved by an open send.
103    AlreadyReserved(DetTid),
104    /// In record or replay, a victim has a background operation in flight:
105    /// its result may never be recorded, so the send keeps the ordinary path.
106    BackgroundOperation(DetTid),
107    /// A victim process is a PID namespace's init, which Linux protects from
108    /// SIGKILL sent inside its namespace: the kill may be discarded.
109    NamespaceInit(DetPid),
110}

112/// A SIGKILL fence, or a sender's `kill`, waited past
113/// `Scheduler::SIGKILL_FENCE_VALVE` for victims' deaths or exit publications
114/// that never arrived. Continuing would let host timing decide the schedule,
115/// so the run is refused (<https://github.com/rrnewton/hermit/issues/3994>).
116#[derive(Debug, Clone, PartialEq, Eq)]
117pub struct SigkillFenceRefusal {
118    /// Committed victims whose physical exit was not accounted, in tid order.
119    pub unaccounted: Vec<DetTid>,
120    /// Processes whose exit publication to the parent had not arrived.
121    pub unpublished: Vec<DetPid>,
122    /// How long, in host time, the wait had lasted.
123    pub waited: std::time::Duration,
124}

130            "hermit refused to continue the run: a SIGKILL's victims waited {:?} for \

139                .iter()
140                .map(|p| p.as_raw())
141                .collect::<Vec<_>>(),
142        )
143    }

171/// Whether a backend lets the scheduler retire a SIGKILL's victims at the
172/// kill turn: ptrace, where the scheduler's record decides waits and each
173/// victim's death reaches it as a deregistration. Backends that cancel killed
174/// threads' RPCs, need thread-directed process signals, or report process
175/// exits on their own keep their existing paths.
176pub(crate) fn retires_sigkill_at_the_kill_turn(backend: &reverie::BackendCapabilities) -> bool {
177    !(backend.needs_killed_thread_rpc_cancellation
178        || backend.requires_thread_directed_process_signals
179        || backend.process_exits_complete_asynchronously
180        || backend.reports_physical_process_exits)
181}

183/// What a consuming receipt (a deregistration) of `tid` means now.
184#[derive(Debug, Clone, Copy, PartialEq, Eq)]
185pub(crate) enum SigkillReceipt {
186    /// Not a victim: the ordinary path retires it.
187    Ordinary,
188    /// A reserved victim, before the commit: account it, retire nothing.
189    Held,
190    /// A committed victim: already retired; this accounts its physical exit.
191    Committed,
192}

194#[derive(Debug, Default)]
195pub(crate) struct SigkillRetirement {
196    next_token: u64,
197    reservations: BTreeMap<u64, Reservation>,
198    /// Reserved victims, before their send commits or unwinds.
199    reserved: BTreeMap<DetTid, u64>,
200    /// Reserved victims whose consuming receipt arrived before the commit.
201    held_receipts: BTreeSet<DetTid>,
202    /// Committed victims whose physical exit is not accounted yet.
203    unaccounted: BTreeSet<DetTid>,
204    /// Owner-death wake batches from victims' cleanup, keyed by (victim, the
205    /// batch's position among that victim's batches). They run when the
206    /// victims' cohort is complete, in key order
207    /// ([`Scheduler::run_retained_sigkill_wakes`]), never at receipt.
208    retained_wakes: BTreeMap<(DetTid, u64), Vec<(DetTid, FutexID)>>,
209    /// The next batch position of each victim with retained batches.
210    wake_batches: BTreeMap<DetTid, u64>,
211    /// The robust lists a sender recorded before its send for the threads the
212    /// send kills, keyed by thread and address space, with the reservation's
213    /// token, until each thread's exit callback takes its own
214    /// ([`Scheduler::take_sigkill_robust_list`]).
215    robust_lists: HashMap<(DetTid, MmId), (u64, RobustListSnapshot)>,
216    /// Victims whose first consuming receipt was counted (held or committed):
217    /// a later deregistration of the same thread changes nothing. Forgotten
218    /// when a new thread registers on the tid (`forget_committed_exit`).
219    accounted: BTreeSet<DetTid>,
220    /// Victim processes whose exit publication the commit holds for: the
221    /// fence waits for these as well, under the same valve.
222    unpublished: BTreeSet<DetPid>,
223    /// The processes of the send being committed, while the commit retires
224    /// them one by one. A subreaper among them adopts none of the others'
225    /// orphans ([`Scheduler::is_dying_sigkill_victim`]).
226    dying: BTreeSet<DetPid>,
227}

230    /// Reserve the victims of a SIGKILL that `sender` is about to send to
231    /// `target`, before the physical send. Returns the reservation's token.
232    pub(crate) fn reserve_sigkill(
233        &mut self,
234        sender: DetTid,
235        target: DetPid,
236        uid: u32,
237    ) -> Result<SigkillReservation, SigkillRefusal> {
238        self.reserve_sigkill_processes(sender, &[target], uid)
239    }

241    /// Reserve the victims of a `kill(-pgid, SIGKILL)` that `sender` is about
242    /// to send: every live process whose recorded process group is `pgid`.
243    /// Every guest process has the same credentials (the credential setters
244    /// are deterministic no-ops, and ptrace sets `PR_SET_NO_NEW_PRIVS`), so
245    /// Linux signals every member, and so all share `uid`.
246    pub(crate) fn reserve_sigkill_group(
247        &mut self,
248        sender: DetTid,
249        pgid: DetPid,
250        uid: u32,
251    ) -> Result<SigkillReservation, SigkillRefusal> {
252        let mut members: Vec<DetPid> = self
253            .thread_tree
254            .thread_group_leaders
255            .iter()
256            .copied()
257            .filter(|process| self.thread_tree.process_group(*process) == Some(pgid))
258            .filter(|process| self.live_process_threads(*process).next().is_some())
259            .collect();
260        members.sort();
261        self.reserve_sigkill_processes(sender, &members, uid)
262    }
263
264    fn reserve_sigkill_processes(
265        &mut self,
266        sender: DetTid,
267        targets: &[DetPid],
268        uid: u32,
269    ) -> Result<SigkillReservation, SigkillRefusal> {
270        if !retires_sigkill_at_the_kill_turn(&self.backend) {
271            return Err(SigkillRefusal::Backend);
272        }
273        if targets.is_empty() {
274            return Err(SigkillRefusal::NoTarget);
275        }
276        let mut victims = Vec::new();
277        let mut includes_sender = false;
278        for target in targets {
279            if !self.thread_tree.thread_group_leaders.contains(target) {
280                return Err(SigkillRefusal::NoTarget);
281            }
282            if self.thread_tree.is_pid_namespace_init(*target) {
283                crate::detlog::write_loss_notice(&format!(
284                    "a SIGKILL victim, process {target}, is a PID namespace's init, which Linux may protect from it, so the send keeps its host-timed retirement (https://github.com/rrnewton/hermit/issues/3994)"
285                ));
286                return Err(SigkillRefusal::NamespaceInit(*target));
287            }
288            // A prior terminal cause stands: a process whose exit was already
289            // granted keeps that exit's status and notification.
290            if self.thread_tree.granted_exits.contains(target) {
291                continue;
292            }
293            let own = self.registered_process(sender) == Some(*target);
294            includes_sender |= own;
295            match self.kill_victims(sender, *target) {
296                Ok(of_target) => victims.extend(of_target),
297                Err(refusal) => {
298                    match refusal {
299                        // A recording made now could not replay this kill: the
300                        // victim's operation may have a recorded result whose
301                        // rejoin will never be consumed. Replay refuses it
302                        // before starting the guest.
303                        SigkillRefusal::BackgroundOperation(tid) => {
304                            crate::detlog::record_replay_refusal(&format!(
305                                "a SIGKILL victim, thread {tid}, had a background operation in flight during record or replay, so its death is retired at its host-timed deregistration (https://github.com/rrnewton/hermit/issues/3994)"
306                            ))
307                        }
308                        SigkillRefusal::UnknownAddressSpace(tid) => {
309                            crate::detlog::write_loss_notice(&format!(
310                                "a SIGKILL victim, thread {tid}, has no recorded address-space identity, so its death is retired at its host-timed deregistration (https://github.com/rrnewton/hermit/issues/3994)"
311                            ))
312                        }
313                        _ => {}
314                    }
315                    return Err(refusal);
316                }
317            }
318        }
319        let token = self.sigkill.next_token;
320        self.sigkill.next_token += 1;
321        for victim in &victims {
322            self.sigkill.reserved.insert(victim.tid, token);
323        }
324        self.sigkill.reservations.insert(
325            token,
326            Reservation {
327                victims,
328                sender,
329                uid,
330                includes_sender,
331            },
332        );
333        Ok(SigkillReservation {
334            token,
335            includes_sender,
336        })
337    }

339    /// The live threads of `target` other than `sender` that a kill of
340    /// `target` retires, in tid order, each with the address-space identity
341    /// it registered with. The sender's own thread is never a victim: a
342    /// SIGKILL it sends to its own process does not return, and its death is
343    /// covered as an `exit_group` caller's; at a fatal delivery it leaves at
344    /// its own exit report. A thread that cannot be retired at the committed
345    /// turn refuses the whole kill, which keeps the host-timed path.
346    fn kill_victims(
347        &self,
348        sender: DetTid,
349        target: DetPid,
350    ) -> Result<Vec<SigkillVictim>, SigkillRefusal> {
351        // `live_process_threads`, not the bare thread tree: it names the
352        // threads whose recorded leader is `target`, so a thread id that a
353        // thread of `target` once had and another process now reuses (its
354        // leader is that process) is not retired with `target`.
355        let mut threads: Vec<DetTid> = self.live_process_threads(target).collect();
356        if threads.is_empty() {
357            return Err(SigkillRefusal::NoTarget);
358        }
359        threads.retain(|tid| *tid != sender);
360        let mut victims = Vec::with_capacity(threads.len());
361        for tid in threads {
362            if self.sigkill.reserved.contains_key(&tid) {
363                return Err(SigkillRefusal::AlreadyReserved(tid));
364            }
365            // A vfork parent whose child has not reached a release edge is
366            // still in the kernel's vfork wait: its vfork has not returned, so
367            // no result was or will be recorded for it, and the kill replays
368            // as it recorded.
369            let in_unreturned_vfork = self.vfork_barriers.contains_key(&tid)
370                && !self.released_vfork_barriers.contains(&tid);
371            if self.recordreplay_modes
372                && self.blocked.external_io_blockers.contains_key(&tid)
373                && !in_unreturned_vfork
374            {
375                return Err(SigkillRefusal::BackgroundOperation(tid));
376            }
377            let Some(mm) = self.thread_mm(tid) else {
378                return Err(SigkillRefusal::UnknownAddressSpace(tid));
379            };
380            victims.push(SigkillVictim {
381                tid,
382                process: target,
383                mm,
384            });
385        }
386        Ok(victims)
387    }

389    /// Retire, in `sender`'s delivery turn of a signal whose default action
390    /// ends its process, the process's other live threads as committed
391    /// victims (states 2 to 4 above, with no reservation: the kernel kills
392    /// none of them before the delivering thread resumes). Each is retired in
393    /// tid order, its late requests are never answered, its deregistration is
394    /// only accounted, and turn selection waits until every one is accounted.
395    /// The process itself still ends at `sender`'s own exit report, which
396    /// keeps the parent's notification on its existing path. Returns the
397    /// victims; none where the kill would be refused, which keeps the
398    /// host-timed retirement and, when a thread is the reason, names it as a
399    /// determinism loss.
400    pub(crate) fn retire_fatal_delivery_victims(&mut self, sender: DetTid) -> Vec<SigkillVictim> {
401        let Some(process) = self.registered_process(sender) else {
402            return Vec::new();
403        };
404        if !retires_sigkill_at_the_kill_turn(&self.backend) {
405            return Vec::new();
406        }
407        let victims = match self.kill_victims(sender, process) {
408            Ok(victims) => victims,
409            Err(refusal) => {
410                let held = match refusal {
411                    SigkillRefusal::BackgroundOperation(tid) => Some((
412                        tid,
413                        "had a background operation in flight during record or replay",
414                    )),
415                    SigkillRefusal::UnknownAddressSpace(tid) => {
416                        Some((tid, "has no recorded address-space identity"))
417                    }
418                    _ => None,
419                };
420                if let Some((tid, why)) = held {
421                    crate::detlog::write_loss_notice(&format!(
422                        "a fatal signal's delivery ends process {process}, whose thread {tid} {why}, so the process's threads leave at their host-timed exit reports"
423                    ));
424                }
425                return Vec::new();
426            }
427        };
428        for victim in &victims {
429            info!(
430                "[scheduler] retiring dettid {} of process {} at its fatal signal's delivery",
431                victim.tid, victim.process
432            );
433            self.sigkill.unaccounted.insert(victim.tid);
434            self.logically_kill_thread(&victim.tid, &victim.process, victim.mm);
435        }
436        victims
437    }

439    /// Route a consuming receipt of `tid` (its deregistration, after its final
440    /// accounting): see [`SigkillReceipt`].
441    pub(crate) fn route_sigkill_receipt(&mut self, tid: DetTid) -> SigkillReceipt {
442        // The thread's exit callback, which takes its recorded robust list,
443        // came before this receipt: whatever is left was never wanted.
444        self.sigkill
445            .robust_lists
446            .retain(|(owner, _), _| *owner != tid);
447        if self.sigkill.reserved.contains_key(&tid) {
448            self.sigkill.held_receipts.insert(tid);
449            self.sigkill.accounted.insert(tid);
450            SigkillReceipt::Held
451        } else if self.sigkill.unaccounted.remove(&tid) {
452            self.sigkill.accounted.insert(tid);
453            SigkillReceipt::Committed
454        } else {
455            SigkillReceipt::Ordinary
456        }
457    }

459    /// Whether a SIGKILL victim's consuming receipt for `tid` was already
460    /// counted, so this one is a duplicate that must change nothing.
461    pub(crate) fn sigkill_receipt_already_counted(&self, tid: DetTid) -> bool {
462        self.sigkill.accounted.contains(&tid)
463    }

465    /// A new thread registers on `tid`: an earlier victim's counted receipt no
466    /// longer applies to it.
467    pub(crate) fn forget_sigkill_receipt(&mut self, tid: DetTid) {
468        self.sigkill.accounted.remove(&tid);
469    }

471    /// Commit (`sent`) or unwind the reservation `token` at the sending
472    /// boundary. Returns the victims retired by the commit.
473    pub(crate) fn commit_sigkill(&mut self, token: u64, sent: bool) -> Vec<SigkillVictim> {
474        let Some(reservation) = self.sigkill.reservations.remove(&token) else {
475            return Vec::new();
476        };
477        for victim in &reservation.victims {
478            self.sigkill.reserved.remove(&victim.tid);
479        }
480        if !sent {
481            // No thread died, so no robust list is walked.
482            self.sigkill
483                .robust_lists
484                .retain(|_, (recorded_for, _)| *recorded_for != token);
485            // No kill happened: wakes and a receipt that arrived meanwhile
486            // take the ordinary path now.
487            let victims: BTreeSet<DetTid> = reservation.victims.iter().map(|v| v.tid).collect();
488            self.run_retained_wakes_of(|tid| victims.contains(&tid));
489            for victim in &reservation.victims {
490                if self.sigkill.held_receipts.remove(&victim.tid) {
491                    self.logically_kill_thread(&victim.tid, &victim.process, victim.mm);
492                }
493            }
494            return Vec::new();
495        }
496        // A victim that died before the commit (its receipt is held) may have
497        // had its exit reported published already. Retiring its leader
498        // forgets that report, which an Exit grant wants; here the report
499        // must still stand, or the notification below installs a hold that
500        // nothing can release.
501        let published: BTreeSet<DetPid> = reservation
502            .victims
503            .iter()
504            .map(|victim| victim.process)
505            .filter(|process| self.child_exit_publications_completed.contains(process))
506            .collect();
507        // Every victim process is marked dying, and its exit granted, before
508        // any is retired: re-parenting an earlier victim's orphans must pass
509        // over a subreaper that this same send kills, and an adoption of a
510        // victim's orphans happens at this committed turn, not at a
511        // host-timed deregistration, so it is no determinism loss.
512        // A send that includes the sender commits at the sender's own Exit
513        // grant, which already notified the sender's process's parent.
514        let already_notified: BTreeSet<DetPid> = reservation
515            .victims
516            .iter()
517            .map(|victim| victim.process)
518            .filter(|process| self.thread_tree.granted_exits.contains(process))
519            .collect();
520        for victim in &reservation.victims {
521            self.sigkill.dying.insert(victim.process);
522            self.thread_tree.granted_exits.insert(victim.process);
523            // A committed SIGKILL dooms its victim for the child-wait
524            // deadlock check (https://github.com/rrnewton/hermit/issues/3904):
525            // a reserved send does not reach the per-send notification.
526            self.record_committed_sigkill(victim.tid, Some(victim.process));
527        }
528        for victim in &reservation.victims {
529            info!(
530                "[scheduler] retiring dettid {} of SIGKILLed process {} at the kill turn",
531                victim.tid, victim.process
532            );
533            if !self.sigkill.held_receipts.remove(&victim.tid) {
534                self.sigkill.unaccounted.insert(victim.tid);
535            }
536            self.logically_kill_thread(&victim.tid, &victim.process, victim.mm);
537        }
538        self.sigkill.dying.clear();
539        let mut processes: Vec<DetPid> = reservation
540            .victims
541            .iter()
542            .map(|victim| victim.process)
543            .collect();
544        processes.dedup();
545        self.child_exit_publications_completed
546            .extend(published.iter().copied());
547        // The processes of one send die together, and the kernel publishes
548        // their exits to a shared parent in an order the host chooses: they
549        // form one cohort for the child-exit ledger, which decides which of
550        // them the kernel's coalesced SIGCHLD names.
551        let cohort = (processes.len() > 1).then_some(token);
552        for process in processes {
553            if !already_notified.contains(&process) {
554                self.own_killed_child_notification(process, reservation.uid, cohort);
555            }
556        }
557        // A report the notification did not consume is forgotten, as
558        // retiring the leader intended.
559        for process in &published {
560            self.child_exit_publications_completed.remove(process);
561        }
562        reservation.victims
563    }

565    /// The parent's notification of `process`'s death by SIGKILL, owned at the
566    /// kill turn as an Exit grant owns it: when Linux would discard the
567    /// parent's SIGCHLD, a hold until the kernel's publication; otherwise
568    /// Linux's siginfo (`CLD_KILLED`, `SIGKILL`) on the scheduler's copy, and
569    /// the child-exit timer. The first copy delivered stands for the exit
570    /// (`child_exit_sigchld`).
571    fn own_killed_child_notification(&mut self, process: DetPid, uid: u32, cohort: Option<u64>) {
572        let Some(parent) = self.thread_tree.parent_process(&process) else {
573            return;
574        };
575        if !self.should_synthesize_child_exit_signal(parent) {
576            return;
577        }
578        // A child whose exit signal is not SIGCHLD (a raw clone with exit
579        // signal 0 or SIGUSR1) is not notified with SIGCHLD: Linux sends its
580        // own exit signal, if any, and the kernel delivers that itself.
581        if matches!(
582            self.thread_tree.notification_signal(process),
583            Some(signal) if signal != libc::SIGCHLD
584        ) {
585            return;
586        }
587        let notification = if self.backend.reports_child_exit_publication {
588            self.classify_child_exit_notification(process, parent)
589        } else {
590            super::ChildExitNotification::Undecided
591        };
592        match notification {
593            super::ChildExitNotification::None | super::ChildExitNotification::Discard => {
594                if !self.child_exit_publications_completed.remove(&process) {
595                    self.child_exit_publications_pending.insert(process);
596                    self.sigkill.unpublished.insert(process);
597                }
598            }
599            super::ChildExitNotification::Deliver | super::ChildExitNotification::Undecided => {
600                // The kernel sends the parent its own SIGCHLD too, once the
601                // tracer has reaped the victim. The fence waits for that
602                // publication, and the sender waits for it before its `kill`
603                // returns (`CommitSigkill`), so the kernel's copy is pending at
604                // a point the schedule fixes and is delivered first; the
605                // scheduler's copy then sends nothing (`child_exit_sigchld`).
606                if self.backend.reports_child_exit_publication
607                    && !self.child_exit_publications_completed.remove(&process)
608                {
609                    self.child_exit_publications_pending.insert(process);
610                    self.sigkill.unpublished.insert(process);
611                }
612                self.child_exit_sigchld.exit_status.insert(
613                    process,
614                    super::child_exit_sigchld::ChildExitSiginfo {
615                        code: libc::CLD_KILLED,
616                        pid: process.as_raw(),
617                        uid,
618                        status: libc::SIGKILL,
619                    },
620                );
621                let deadline = self.committed_time + super::LogicalTime::from_nanos(1);
622                let parent_thread = if self.models_signal_targets {
623                    self.thread_tree
624                        .process_wait
625                        .get(&process)
626                        .map_or(parent, |metadata| metadata.wait_owner)
627                } else {
628                    parent
629                };
630                if self.backend.reports_kernel_child_exit_signals
631                    && self.child_exit_sigchld.own(parent, process)
632                    && let Some(cohort) = cohort
633                {
634                    self.child_exit_sigchld.join_cohort(process, cohort);
635                }
636                self.blocked.timed_waiters.insert_child_exit(
637                    deadline,
638                    process,
639                    parent,
640                    parent_thread,
641                );
642            }
643        }
644    }

646    /// Retire every thread still registered, in DetTid order, as a committed
647    /// victim of a kill the scheduler does not send itself: the termination
648    /// that unwinds the guest at a terminal outcome (`unwind_terminal_outcome`)
649    /// kills every guest thread, and their deaths reach the scheduler as
650    /// deregistrations in the order the host reports them. Retired here, each death is then
651    /// only accounted ([`Self::route_sigkill_receipt`]), so the log the run
652    /// ends with, and the scheduler's state, do not depend on that order. The
653    /// scheduler grants no turn after this. As for a committed SIGKILL
654    /// ([`Self::commit_sigkill`]), every victim process is marked dying and
655    /// its exit granted before any is retired, so re-parenting passes over a
656    /// subreaper that dies too, and a parent-death signal is held, not sent,
657    /// as at any committed exit (nothing drains it before the guest is gone).
658    ///
659    /// Not retired here: every thread on a backend that does not retire a
660    /// kill's victims at the kill turn, and one with no recorded process or
661    /// address-space identity (a named determinism loss), which keep their
662    /// host-timed deregistrations, and a thread reserved by a send, whose
663    /// deregistration is a held receipt that retires nothing.
664    pub(crate) fn retire_threads_at_unwind(&mut self) -> Vec<DetTid> {
665        if !retires_sigkill_at_the_kill_turn(&self.backend) {
666            return Vec::new();
667        }
668        let mut victims = Vec::new();
669        for &tid in self.next_turns.keys() {
670            if self.sigkill.reserved.contains_key(&tid) {
671                continue;
672            }
673            let (Some(process), Some(mm)) = (self.registered_process(tid), self.thread_mm(tid))
674            else {
675                crate::detlog::write_loss_notice(&format!(
676                    "thread {tid}, killed by the unwind of a terminal outcome, has no recorded process or address-space identity, so its death is retired at its host-timed deregistration"
677                ));
678                continue;
679            };
680            victims.push(SigkillVictim { tid, process, mm });
681        }
682        for victim in &victims {
683            self.sigkill.dying.insert(victim.process);
684            self.thread_tree.granted_exits.insert(victim.process);
685        }
686        for victim in &victims {
687            self.sigkill.unaccounted.insert(victim.tid);
688            self.logically_kill_thread(&victim.tid, &victim.process, victim.mm);
689        }
690        self.sigkill.dying.clear();
691        victims.into_iter().map(|victim| victim.tid).collect()
692    }

752    /// Whether a guest's reserved SIGKILL owns `process`'s deaths: a reserved
753    /// send names one of its threads. Such a send retires every thread it
754    /// names at its kill turn, so a death it caused, reported before that
755    /// turn, is never a signal death to wait for at a deregistration
756    /// (`Scheduler::note_signal_killed_group`); after the turn no thread of
757    /// the process is left to wait for.
758    ///
759    /// A committed send's record in `sigkilled_processes` does not count: a
760    /// send that could not be reserved (`SigkillRefusal`) is recorded there
761    /// after the physical send but retires nothing, and its victims' deaths
762    /// arrive at their deregistrations like any other signal death.
763    pub(crate) fn sigkill_owns_process(&self, process: DetPid) -> bool {
764        self.sigkill.reservations.values().any(|reservation| {
765            reservation
766                .victims
767                .iter()
768                .any(|victim| victim.process == process)
769        })
770    }

787    /// The threads reservation `token`'s send kills, each with the address
788    /// space it registered with, in tid order: its victims, and its sender too
789    /// when the send ends the sender's own process. Empty for a token that is
790    /// not reserved.
791    pub(crate) fn sigkill_dying_threads(&self, token: u64) -> Vec<(DetTid, MmId)> {
792        let Some(reservation) = self.sigkill.reservations.get(&token) else {
793            return Vec::new();
794        };
795        let mut dying: Vec<(DetTid, MmId)> = reservation
796            .victims
797            .iter()
798            .map(|victim| (victim.tid, victim.mm))
799            .collect();
800        if reservation.includes_sender
801            && let Some(mm) = self.thread_mm(reservation.sender)
802        {
803            dying.push((reservation.sender, mm));
804        }
805        dying.sort_by_key(|(tid, _)| *tid);
806        dying
807    }

809    /// Keep the robust lists the sender of reservation `token` recorded before
810    /// its send, for the threads it kills (`Self::sigkill_dying_threads`) whose
811    /// receipt has not come; a list for any other thread is dropped. They are
812    /// kept apart from the reservation, which the commit removes before the
813    /// victims' exit callbacks take them.
814    pub(crate) fn record_sigkill_robust_lists(
815        &mut self,
816        token: u64,
817        lists: Vec<(DetTid, MmId, RobustListSnapshot)>,
818    ) {
819        let dying = self.sigkill_dying_threads(token);
820        for (tid, mm, list) in lists {
821            // A victim whose receipt already came died on its own before the
822            // send, and walked its own list then; nothing would take this.
823            if dying.contains(&(tid, mm)) && !self.sigkill.accounted.contains(&tid) {
824                self.sigkill.robust_lists.insert((tid, mm), (token, list));
825            }
826        }
827    }

829    /// The robust list recorded for thread `tid` of address space `mm` before
830    /// the SIGKILL that killed it, taken by its exit callback.
831    pub(crate) fn take_sigkill_robust_list(
832        &mut self,
833        tid: DetTid,
834        mm: MmId,
835    ) -> Option<RobustListSnapshot> {
836        self.sigkill
837            .robust_lists
838            .remove(&(tid, mm))
839            .map(|(_token, list)| list)
840    }

842    /// Retain a victim's non-empty owner-death wake batch instead of running
843    /// it at receipt, and say whether it was retained. Selecting waiters,
844    /// consuming the fuzz arm's selection randomness and logging the result
845    /// all depend on order, and receipts arrive in host order. A retained
846    /// batch runs when the cohort is complete; the victim's exit path does
847    /// not wait for that.
848    pub(crate) fn retain_sigkill_victim_wakes(
849        &mut self,
850        from: DetTid,
851        wakes: &[(DetTid, FutexID)],
852    ) -> bool {
853        if wakes.is_empty()
854            || !(self.sigkill.reserved.contains_key(&from)
855                || self.sigkill.unaccounted.contains(&from))
856        {
857            return false;
858        }
859        let position = self.sigkill.wake_batches.entry(from).or_default();
860        self.sigkill
861            .retained_wakes
862            .insert((from, *position), wakes.to_vec());
863        *position += 1;
864        true
865    }
866
867    /// Run the retained wake batches of every victim `select` names, in
868    /// (victim, position) order, and log each result.
869    fn run_retained_wakes_of(&mut self, select: impl Fn(DetTid) -> bool) {
870        let keys: Vec<(DetTid, u64)> = self
871            .sigkill
872            .retained_wakes
873            .keys()
874            .filter(|(victim, _)| select(*victim))
875            .copied()
876            .collect();
877        for key in keys {
878            let wakes = self.sigkill.retained_wakes.remove(&key).unwrap();
879            let counts = self.wake_futex_waiters_after_exit(&wakes);
880            for ((owner, futex), count) in wakes.into_iter().zip(counts) {
881                info!(
882                    "[detcore, dtid {}] robust-list owner death woke {} waiter(s) on futex {:?} after physical exit",
883                    owner, count, futex,
884                );
885            }
886        }
887        let retained = &self.sigkill.retained_wakes;
888        self.sigkill
889            .wake_batches
890            .retain(|victim, _| retained.keys().any(|(tid, _)| tid == victim));
891    }
892
893    /// Whether futex wakes a SIGKILL victim's exit left are retained until
894    /// the fence lifts (`run_retained_sigkill_wakes`).
895    pub(crate) fn has_retained_sigkill_wakes(&self) -> bool {
896        !self.sigkill.retained_wakes.is_empty()
897    }
898
899    /// Once no committed victim's physical exit is outstanding, run every
900    /// retained wake batch, before step2 drains anything.
901    pub(crate) fn run_retained_sigkill_wakes(&mut self) {
902        if !self.sigkill_fence_holds() && !self.sigkill.retained_wakes.is_empty() {
903            self.run_retained_wakes_of(|_| true);
904        }
905    }

1290    /// Two victims of one send each deliver an owner-death batch, for
1291    /// different futexes with three surviving waiters each, in either order,
1292    /// with a step2 between the receipts (design revision 4, R2). The waiters
1293    /// woken, the run-queue order and the fuzz arm's next draw must not depend
1294    /// on the order, under FIFO and under the fuzz arm.
1295    ///
1296    /// This verifies the retention protocol, not a SIGKILL producer: on
1297    /// ptrace today a SIGKILL victim stages no owner-death batch (the
1298    /// fatal-signal case of https://github.com/rrnewton/hermit/issues/2082),
1299    /// so the batches are handed to the scheduler as
1300    /// `GlobalState::recv_robust_list_wakes` would hand them.
1301    #[test]
1302    fn victims_owner_death_wakes_do_not_depend_on_receipt_order() {
1303        let mm = MmId::initial(DetTid::from_raw(3));
1304        let futexes = [0x404100, 0x404200].map(|addr| FutexID::private(mm, addr));
1305        let run = |fuzz: bool, reverse: bool| {
1306            let mut scheduler = family();
1307            scheduler.fuzz_futexes = fuzz;
1308            for (futex, waiters) in futexes.iter().zip([[11, 12, 13], [14, 15, 16]]) {
1309                for raw in waiters {
1310                    let tid = DetTid::from_raw(raw);
1311                    super::super::test::register_known_thread_for(&mut scheduler, tid);
1312                    scheduler.sleep_futex_waiter(&tid, *futex, None, u32::MAX, None);
1313                }
1314            }
1315            let token = scheduler
1316                .reserve_sigkill(DetTid::from_raw(2), DetPid::from_raw(3), 0)
1317                .unwrap()
1318                .token;
1319            scheduler.commit_sigkill(token, true);
1320            let mut receipts = [
1321                (DetTid::from_raw(3), futexes[0]),
1322                (DetTid::from_raw(4), futexes[1]),
1323            ];
1324            if reverse {
1325                receipts.reverse();
1326            }
1327            for (victim, futex) in receipts {
1328                // As `GlobalState::recv_robust_list_wakes` does.
1329                if !scheduler.retain_sigkill_victim_wakes(victim, &[(victim, futex)]) {
1330                    scheduler.wake_futex_waiters_after_exit(&[(victim, futex)]);
1331                }
1332                let _ = scheduler.step2_drain_prefix();
1333                assert_eq!(
1334                    scheduler.route_sigkill_receipt(victim),
1335                    SigkillReceipt::Committed
1336                );
1337                let _ = scheduler.step2_drain_prefix();
1338            }
1339            assert!(scheduler.sigkill_fence_holds(), "the exit is unpublished");
1340            scheduler.complete_child_exit_publication(DetPid::from_raw(3));
1341            let _ = scheduler.step2_drain_prefix();
1342            let still_waiting: Vec<Vec<DetTid>> = futexes
1343                .iter()
1344                .map(|futex| {
1345                    scheduler
1346                        .blocked
1347                        .futex_waiters
1348                        .get(futex)
1349                        .map(|waiters| waiters.iter().map(|w| w.dettid).collect())
1350                        .unwrap_or_default()
1351                })
1352                .collect();
1353            let queue: Vec<DetTid> = scheduler.run_queue.tids().copied().collect();
1354            let next_draw: u64 = rand::RngExt::random(&mut scheduler.fuzz_prng);
1355            (still_waiting, queue, next_draw)
1356        };
1357        for fuzz in [false, true] {
1358            let forward = run(fuzz, false);
1359            assert_eq!(forward.0.iter().map(Vec::len).collect::<Vec<_>>(), [2, 2]);
1360            assert_eq!(forward, run(fuzz, true), "fuzz={fuzz}");
1361        }
1362    }
