9//! A fair order for the threads of a run past the end of its replayed
10//! schedule, under `--replay-continue-fairly`
11//! (`Config::replay_continue_fairly`).
12//!
13//! Past the end of a replayed schedule every thread has the default priority,
14//! and threads of one priority take turns in queue order. A committing call
15//! (a write, a futex wait) sends its thread to the back of the queue, while a
16//! thread that only computes or reads the clock keeps the CPU until its
17//! timeslice ends. So beside a spinning thread a thread that makes committing
18//! calls runs one of them per timeslice: a pipeline test whose recording ran
19//! in 24 s reached 5 of its 23 restarts in 600 s there.
20//!
21//! With the option, the threads queued in round-robin order in each band
22//! are ordered by the virtual time each has run since the continuation
23//! began, least first, as Linux's fair scheduler orders runnable tasks by
24//! their virtual run time. A thread's run time is what its own clock has
25//! advanced since the continuation first saw it; sleeps and timed waits do
26//! not advance a thread's clock. A thread seen for the first time, or
27//! runnable again after it was not, starts at no less than the least run time
28//! of the threads that were runnable (the floor), so a new or long-blocked
29//! thread does not keep the CPU until it has caught up with the others; a
30//! thread whose clock the scheduler has not seen yet is placed at the floor.
31//! Its selections are still recorded, and its first reported work is charged
32//! from the clock inherited at creation, whichever pass first sees that clock.
33//! Equal run times go to the thread selected least recently, and a thread
34//! runnable again counts as never selected, so a thread raised to the floor
35//! goes ahead of the thread that holds it.
36//!
37//! That is all it changes. A selected thread still runs until its slice ends
38//! as before (`--max-timeslice`): there is no preemption on wakeup, no weight
39//! and no credit for sleeping. Entries pushed to the front of their band (a
40//! timed wake, an eager readmission) and pollers keep their places, ahead of
41//! the ordered entries. Nothing comes from the recording and nothing is drawn
42//! at random: the order is a function of the threads' clocks, of which
43//! threads are runnable and of the order threads were selected in.
44
45use std::collections::BTreeMap;
46
47use crate::types::DetTid;
48use crate::types::LogicalTime;

50/// A parked thread's reported clock and its deterministic startup baseline.
51#[derive(Debug, Clone, Copy)]
52pub(super) struct Clock {
53    pub(super) current: LogicalTime,
54    pub(super) inherited: LogicalTime,
55}
56
57/// One thread's account.
58#[derive(Debug, Clone, Copy, PartialEq, Eq)]
59struct Account {
60    /// The last reported clock, or None until the first startup RPC arrives.
61    clock: Option<LogicalTime>,
62    /// Virtual nanoseconds the thread has run since the continuation began,
63    /// raised to the floor when it became runnable.
64    run_time: u64,
65    /// Whether the thread was runnable then.
66    runnable: bool,
67    /// When the thread was last selected, in selections since the
68    /// continuation began; 0 if never.
69    last_selected: u64,
70}
71
72/// The run time of every thread the continuation has seen.
73#[derive(Debug, Default)]
74pub(super) struct FairContinuation {
75    accounts: BTreeMap<DetTid, Account>,
76    /// The least run time of a runnable thread, never decreasing.
77    floor: u64,
78    /// Selections since the continuation began.
79    selections: u64,
80}
81
82impl FairContinuation {
83    /// Bring the accounts up to date with `clocks`, one slot per registered
84    /// thread, and return the sort key of each thread in `runnable`
85    /// (sorted by thread): its run time, then when it was last selected.
86    /// Drop accounts only when registration ends, not while waiting for the
87    /// first clock. A new thread without a clock starts at the floor, and
88    /// its first report charges work since its inherited startup clock.
89    ///
90    /// Established runnable threads have published their request's clock, so
91    /// charging its advance before raising a newly runnable thread to the
92    /// floor charges work done before it blocked, not after. Startup admission
93    /// can precede the first request: that pending account keeps its selections
94    /// and uses the inherited baseline when a later snapshot sees the clock.
95    pub(super) fn run_times(
96        &mut self,
97        clocks: &BTreeMap<DetTid, Option<Clock>>,
98        runnable: &[DetTid],
99    ) -> Vec<(DetTid, (u64, u64))> {
100        let floor = self.floor;
101        self.accounts.retain(|tid, _| clocks.contains_key(tid));
102        for (&tid, &clock) in clocks {
103            let is_runnable = runnable.binary_search(&tid).is_ok();
104            let account = self.accounts.entry(tid).or_insert(Account {
105                clock: clock.map(|reported| reported.current),
106                run_time: floor,
107                runnable: false,
108                last_selected: 0,
109            });
110            if let Some(reported) = clock {
111                // A clock never goes back; an exec reload restores the same
112                // one. A delayed first report includes the child's first
113                // turn, which must be charged exactly as with an early RPC.
114                let previous = account.clock.unwrap_or(reported.inherited);
115                let ran = reported
116                    .current
117                    .as_nanos()
118                    .saturating_sub(previous.as_nanos());
119                account.run_time = account.run_time.saturating_add(ran);
120                account.clock = Some(reported.current);
121            }
122            if is_runnable && !account.runnable {
123                account.run_time = account.run_time.max(floor);
124                // A thread back in turn has waited: it goes ahead of the
125                // threads it ties with, whichever way it was woken (a wake
126                // pushed to the front was selected, and so noted, before it
127                // got here).
128                account.last_selected = 0;
129            }
130            account.runnable = is_runnable;
131        }
132        let keys: Vec<(DetTid, (u64, u64))> = runnable
133            .iter()
134            .map(|tid| {
135                let key = self.accounts.get(tid).map_or((floor, 0), |account| {
136                    (account.run_time, account.last_selected)
137                });
138                (*tid, key)
139            })
140            .collect();
141        if let Some(least) = keys.iter().map(|(_, (run_time, _))| *run_time).min() {
142            self.floor = self.floor.max(least);
143        }
144        keys
145    }
146
147    /// Note that `tid` was selected and had a turn, including when its first
148    /// clock was absent from the preceding snapshot. The turn that exhausted
149    /// replay can precede the first account snapshot and is not charged.
150    pub(super) fn note_selected(&mut self, tid: DetTid) {
151        self.selections += 1;
152        if let Some(account) = self.accounts.get_mut(&tid) {
153            account.last_selected = self.selections;
154        }
155    }
156
157    /// A non-leader exec moved `from`'s clock to `to`, the leader's thread
158    /// id; its account goes with it, replacing the old leader's.
159    pub(super) fn reassign(&mut self, from: DetTid, to: DetTid) {
160        if from == to {
161            return;
162        }
163        match self.accounts.remove(&from) {
164            Some(account) => {
165                self.accounts.insert(to, account);
166            }
167            None => {
168                self.accounts.remove(&to);
169            }
170        }
171    }
172}

174#[cfg(test)]
175mod tests {
176    use super::*;
177
178    fn clocks(entries: &[(DetTid, u64)]) -> BTreeMap<DetTid, Option<Clock>> {
179        entries
180            .iter()
181            .map(|&(tid, ns)| {
182                (
183                    tid,
184                    Some(Clock {
185                        current: LogicalTime::from_nanos(ns),
186                        inherited: LogicalTime::ZERO,
187                    }),
188                )
189            })
190            .collect()
191    }
192
193    /// A thread's run time is what its clock advanced since it was first
194    /// seen; the thread that ran less comes first.
195    #[test]
196    fn the_thread_that_ran_less_comes_first() {
197        let (writer, spinner) = (DetTid::from_raw(3), DetTid::from_raw(5));
198        let mut fair = FairContinuation::default();
199        assert_eq!(
200            fair.run_times(
201                &clocks(&[(writer, 1_000), (spinner, 1_000)]),
202                &[writer, spinner]
203            ),
204            vec![(writer, (0, 0)), (spinner, (0, 0))]
205        );
206        // The spinner ran a 200 ms slice, the writer half a millisecond.
207        assert_eq!(
208            fair.run_times(
209                &clocks(&[(writer, 501_000), (spinner, 200_001_000)]),
210                &[writer, spinner]
211            ),
212            vec![(writer, (500_000, 0)), (spinner, (200_000_000, 0))]
213        );
214    }
215
216    /// A thread seen for the first time, or runnable again after it was not,
217    /// starts at the least run time of the threads that were runnable, not
218    /// at nothing; one that ran more keeps its own.
219    #[test]
220    fn a_new_or_woken_thread_starts_at_the_floor() {
221        let (writer, spinner, late) = (
222            DetTid::from_raw(3),
223            DetTid::from_raw(5),
224            DetTid::from_raw(7),
225        );
226        let mut fair = FairContinuation::default();
227        fair.run_times(&clocks(&[(writer, 0), (spinner, 0)]), &[writer, spinner]);
228        fair.run_times(
229            &clocks(&[(writer, 30_000_000), (spinner, 50_000_000)]),
230            &[writer, spinner],
231        );
232        // The writer blocks; the spinner runs on.
233        fair.run_times(
234            &clocks(&[(writer, 30_000_000), (spinner, 90_000_000)]),
235            &[spinner],
236        );
237        // The writer wakes and a new thread starts: both at the floor, the
238        // spinner's 90 ms, not at 30 ms or nothing.
239        assert_eq!(
240            fair.run_times(
241                &clocks(&[(writer, 30_000_000), (spinner, 90_000_000), (late, 5)]),
242                &[writer, spinner, late]
243            ),
244            vec![
245                (writer, (90_000_000, 0)),
246                (spinner, (90_000_000, 0)),
247                (late, (90_000_000, 0))
248            ]
249        );
250        // A registered thread without a clock yet keeps an account at the
251        // floor; accounts of threads whose registration ended are dropped.
252        let pending = DetTid::from_raw(9);
253        let mut registered = clocks(&[(writer, 30_000_001)]);
254        registered.insert(pending, None);
255        assert_eq!(
256            fair.run_times(&registered, &[writer, pending]),
257            vec![(writer, (90_000_001, 0)), (pending, (90_000_000, 0))]
258        );
259        assert_eq!(fair.accounts.len(), 2);
260        assert_eq!(fair.accounts[&pending].clock, None);
261        assert!(!fair.accounts.contains_key(&spinner));
262        assert!(!fair.accounts.contains_key(&late));
263        fair.note_selected(pending);
264        // Reporting the inherited clock contributes no work and keeps that
265        // selection, even though the inherited duration is greater than zero.
266        registered.insert(
267            pending,
268            Some(Clock {
269                current: LogicalTime::from_nanos(1_000),
270                inherited: LogicalTime::from_nanos(1_000),
271            }),
272        );
273        assert_eq!(
274            fair.run_times(&registered, &[writer, pending]),
275            vec![(writer, (90_000_001, 0)), (pending, (90_000_000, 1))]
276        );
277    }

279    /// Equal run times go to the thread selected least recently, a thread
280    /// runnable again counts as never selected, and an exec moves the
281    /// caller's account to the leader's thread id.
282    #[test]
283    fn ties_go_to_the_thread_selected_least_recently() {
284        let (main, writer) = (DetTid::from_raw(3), DetTid::from_raw(5));
285        let mut fair = FairContinuation::default();
286        fair.run_times(&clocks(&[(main, 0), (writer, 0)]), &[main, writer]);
287        fair.note_selected(writer);
288        fair.note_selected(main);
289        assert_eq!(
290            fair.run_times(&clocks(&[(main, 0), (writer, 0)]), &[main, writer]),
291            vec![(main, (0, 2)), (writer, (0, 1))]
292        );
293        // The writer blocks while main runs, then wakes (through the front of
294        // the queue, so it was selected last) and ties with main at the floor:
295        // it goes first.
296        fair.run_times(&clocks(&[(main, 10), (writer, 0)]), &[main]);
297        fair.note_selected(writer);
298        assert_eq!(
299            fair.run_times(&clocks(&[(main, 10), (writer, 0)]), &[main, writer]),
300            vec![(main, (10, 2)), (writer, (10, 0))]
301        );
302        fair.reassign(writer, main);
303        assert_eq!(
304            fair.run_times(&clocks(&[(main, 17)]), &[main]),
305            vec![(main, (27, 0))]
306        );
307    }
