9//! `--chaos-timed-wait-spread`: seeded wake thresholds for timed waits.
10//!
11//! Chaos reorders runnable threads, but a sleeping thread wakes when the
12//! global clock reaches its deadline, and the clock advances only with the
13//! other threads' work and the scheduler's turn charges. The ratio between a
14//! sleep and the work done meanwhile is therefore set by the program, and a
15//! race ordered by that ratio changes order only as far as chaos's reordering
16//! of the other threads moves the clock (rarely, in the tests' guest: one
17//! seed in eight). This module draws, for each timed wait, a factor `f`
18//! log-uniformly from `[1/R, R)`, with `R` at most 1000
19//! (`detcore_model::config::MAX_CHAOS_TIMED_WAIT_SPREAD`), and queues the
20//! wait at the wake threshold `start + f * duration` instead of at its
21//! deadline. A threshold at or past the deadline oversleeps, which Linux
22//! allows. A threshold before the deadline is promised its deadline
23//! (`TimedEvents::insert_spread`): when it comes due, the scheduler advances
24//! the global clock to the deadline before the waiter runs, as if every other
25//! thread had been descheduled for the rest of the wait, so the waiter never
26//! observes less than the duration it asked for.
27

37/// The factor of a thread's `index`-th spread wait: `spread^t` for a `t`
38/// uniform in `[-1, 1)` (so the factor is log-uniform in `[1/spread,
39/// spread)`), a pure function of the scheduler seed, the stable thread id and
40/// the index, as `chaos_per_thread_slowdown_factor` keys a thread's slowdown.
41/// Keying by thread and per-thread index, not by a global draw order, keeps a
42/// wait's factor when other threads' events move.
43///
44/// The factor sets a wake time a replay recomputes from the recorded spread
45/// and seed, so it must come out bit-identical on every host and Rust
46/// version. It therefore uses no transcendental function: Rust documents the
47/// precision of `f64::powf`, `exp` and `ln` as unspecified across platforms
48/// and versions. `t` comes from the generator's integer output, not from
49/// `rand`'s float conversion, and `spread^|t|` is a product of repeated square
50/// roots of `spread` selected by the bits of `|t|`: square root,
51/// multiplication and division are IEEE 754 basic operations, correctly
52/// rounded and so identical everywhere, and they run in a fixed order.
53pub(crate) fn timed_wait_factor(seed: u64, dettid: DetTid, index: u64, spread: f64) -> f64 {
54    if spread <= 1.0 {
55        return 1.0;
56    }
57    const SPREAD_SALT: u64 = 0x7469_6d65_6477_6169; // "timedwai"
58    const INDEX_GOLDEN: u64 = 0xbf58_476d_1ce4_e5b9;
59    let mixed = seed
60        ^ SPREAD_SALT
61        ^ ((dettid.as_raw() as u32 as u64).wrapping_mul(0x9e37_79b9_7f4a_7c15))
62        ^ index.wrapping_mul(INDEX_GOLDEN);
63    // `u` in `0..2^53`, so `t = (2u - 2^53) / 2^53` in `[-1, 1)` and the
64    // magnitude `|t| = m / 2^53` with `m` in `0..=2^53`.
65    let u = Pcg64Mcg::seed_from_u64(mixed).next_u64() >> 11;
66    let (m, shorten) = if u >= 1 << 52 {
67        (2 * u - (1 << 53), false)
68    } else {
69        ((1 << 53) - 2 * u, true)
70    };
71    let magnitude = spread_power(spread, m);
72    if shorten { 1.0 / magnitude } else { magnitude }
73}
74
75/// `spread^(m / 2^53)` for `m` in `0..=2^53`, from IEEE 754 basic operations
76/// only: bit `k` of `m` contributes `spread^(2^(k - 53))`, the `53 - k`-th
77/// repeated square root of `spread`, multiplied in from the lowest bit up.
78fn spread_power(spread: f64, m: u64) -> f64 {
79    // roots[j] = spread^(2^-j), for j in 0..=53.
80    let mut roots = [spread; 54];
81    for j in 1..roots.len() {
82        roots[j] = roots[j - 1].sqrt();
83    }
84    let mut product = 1.0;
85    for k in 0..=53u32 {
86        if m >> k & 1 == 1 {
87            product *= roots[53 - k as usize];
88        }
89    }
90    product
91}

93/// The draws of one run: the configured spread, the seed, and how many spread
94/// waits each thread has filed.
95#[derive(Debug, Clone)]
96pub(crate) struct TimedWaitSpread {
97    spread: f64,
98    seed: u64,
99    filed: BTreeMap<DetTid, u64>,
100}
101
102impl TimedWaitSpread {
103    pub(crate) fn new(spread: f64, seed: u64) -> Self {
104        Self {
105            spread,
106            seed,
107            filed: BTreeMap::new(),
108        }
109    }
110
111    /// The wake threshold of a timed wait `dettid` files at committed time
112    /// `now` with deadline `deadline`, with the factor it was drawn. A wait
113    /// with no finite future deadline (already due, or indefinite) keeps its
114    /// deadline, draws nothing and does not advance the thread's index. The
115    /// threshold is at least one nanosecond after `now` and before the
116    /// indefinite sentinel, so it is always a real, future deadline.
117    pub(crate) fn wake_threshold(
118        &mut self,
119        dettid: DetTid,
120        now: LogicalTime,
121        deadline: LogicalTime,
122    ) -> (LogicalTime, f64) {
123        if deadline <= now || deadline.is_indefinite() || now.as_nanos() >= u64::MAX - 2 {
124            return (deadline, 1.0);
125        }
126        let index = self.filed.entry(dettid).or_default();
127        let factor = timed_wait_factor(self.seed, dettid, *index, self.spread);
128        *index += 1;
129        let duration = (deadline.as_nanos() - now.as_nanos()) as f64;
130        // Clamp in integers: near 2^64 an f64 step is 2048 ns, so a bound in
131        // floating point could round up to the indefinite sentinel.
132        let latest = LogicalTime::MAX.as_nanos() - 1 - now.as_nanos();
133        let scaled = ((duration * factor).round() as u64).clamp(1, latest);
134        (now + LogicalTime::from_nanos(scaled), factor)
135    }
136}

142    /// `timed_wait_factor(7, 5, 0..4, 10.0)`, bit for bit.
143    const PINNED_FACTOR_BITS: [u64; 4] = [
144        0x3fc2_db9b_b60e_f9f3,
145        0x400d_59c7_4804_2813,
146        0x3fe2_934e_2e7d_3e11,

150    fn at(ns: u64) -> LogicalTime {
151        LogicalTime::from_nanos(ns)
152    }

154    #[test]
155    fn factors_are_log_uniform_within_the_spread_and_pure() {
156        let spread = 100.0;
157        let mut below = 0;
158        for raw in 1..=8 {
159            for index in 0..256 {
160                let tid = DetTid::from_raw(raw);
161                let factor = timed_wait_factor(7, tid, index, spread);
162                assert_eq!(factor, timed_wait_factor(7, tid, index, spread));
163                assert!(
164                    (1.0 / spread..=spread).contains(&factor),
165                    "{factor} out of range"
166                );
167                below += usize::from(factor < 1.0);
168            }
169        }
170        // Log-uniform: about half the draws shorten. 2048 draws, so a share
171        // outside 40%-60% would be more than eight standard deviations out.
172        assert!((819..=1229).contains(&below), "{below} of 2048 below one");
173        assert_ne!(
174            timed_wait_factor(7, DetTid::from_raw(1), 0, spread),
175            timed_wait_factor(8, DetTid::from_raw(1), 0, spread),
176            "the seed changes the draw"
177        );
178        assert_eq!(timed_wait_factor(7, DetTid::from_raw(1), 3, 1.0), 1.0);
179    }

181    /// The factor is `spread^t` to within a few ulps of the libm result, its
182    /// ends are exact, and it is computed without libm: these exact bits are
183    /// what every host and Rust version must produce, since a replay
184    /// recomputes them from the record (pinned so a change to the arithmetic,
185    /// the generator or the keying shows up as a failure, not as replays that
186    /// drift by a nanosecond).
187    #[test]
188    fn factors_match_the_power_and_are_pinned_bit_for_bit() {
189        for m in [0u64, 1, 1 << 20, 1 << 52, (1 << 53) - 1, 1 << 53] {
190            let exact = 10f64.powf(m as f64 / (1u64 << 53) as f64);
191            let ours = spread_power(10.0, m);
192            assert!(
193                ((ours - exact) / exact).abs() < 64.0 * f64::EPSILON,
194                "m {m}: {ours} against {exact}"
195            );
196        }
197        assert_eq!(spread_power(10.0, 0), 1.0);
198        assert_eq!(spread_power(10.0, 1 << 53), 10.0);
199        let pinned: Vec<u64> = (0..4)
200            .map(|index| timed_wait_factor(7, DetTid::from_raw(5), index, 10.0).to_bits())
201            .collect();
202        assert_eq!(pinned, PINNED_FACTOR_BITS, "{pinned:#x?}");
203    }

205    #[test]
206    fn thresholds_scale_the_remaining_duration_and_count_per_thread() {
207        let mut spread = TimedWaitSpread::new(10.0, 3);
208        let tid = DetTid::from_raw(5);
209        let other = DetTid::from_raw(6);
210        let (wake, factor) = spread.wake_threshold(tid, at(1_000), at(51_000));
211        assert_eq!(
212            wake,
213            at(1_000 + (50_000.0 * factor).round() as u64),
214            "start + factor * duration"
215        );
216        assert_eq!(factor, timed_wait_factor(3, tid, 0, 10.0));
217        // Another thread's draw does not advance this thread's index.
218        spread.wake_threshold(other, at(1_000), at(2_000));
219        let (_, second) = spread.wake_threshold(tid, at(1_000), at(2_000));
220        assert_eq!(second, timed_wait_factor(3, tid, 1, 10.0));
221        // Due, indefinite and past deadlines draw nothing.
222        assert_eq!(
223            spread.wake_threshold(tid, at(5_000), at(5_000)),
224            (at(5_000), 1.0)
225        );
226        assert_eq!(
227            spread.wake_threshold(tid, at(5_000), LogicalTime::INDEFINITE),
228            (LogicalTime::INDEFINITE, 1.0)
229        );
230        let (_, third) = spread.wake_threshold(tid, at(1_000), at(2_000));
231        assert_eq!(third, timed_wait_factor(3, tid, 2, 10.0));
232    }

234    #[test]
235    fn thresholds_stay_future_and_finite_at_the_extremes() {
236        let mut spread = TimedWaitSpread::new(1_000_000.0, 11);
237        let tid = DetTid::from_raw(9);
238        for _ in 0..64 {
239            let (wake, _) = spread.wake_threshold(tid, at(1_000), at(1_001));
240            assert!(
241                wake > at(1_000),
242                "a one-nanosecond wait stays in the future"
243            );
244            let now = at(u64::MAX - 1_000_000);
245            let (wake, _) = spread.wake_threshold(tid, now, at(u64::MAX - 500_000));
246            assert!(wake > now && !wake.is_indefinite());
247        }
248        // An absolute clock near 1.8e18 ns and a far-future timeout: a
249        // lengthened threshold past the clock's range stops just short of
250        // the indefinite sentinel, so the timeout still fires.
251        let mut spread = TimedWaitSpread::new(1_000.0, 5);
252        let now = at(1_800_000_000_000_000_000);
253        for _ in 0..256 {
254            let (wake, _) = spread.wake_threshold(tid, now, now + at(400_000_000_000_000_000));
255            assert!(wake > now && !wake.is_indefinite(), "{wake:?}");
256        }
257    }
