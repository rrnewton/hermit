19/// Encapsulate the set of threads that are waiting for a specific time in the future.
20///
21/// It's possible (but unlikely) that multiple threads are waiting for the same
22/// nanosecond, and this structure must break that symmetry.
23#[derive(Debug, Clone, Default)]
24pub struct TimedEvents {
25    // Inner btreeset is *always* non-empty:
26    map: BTreeMap<LogicalTime, BTreeSet<TimedEvent>>,
27
28    // Keep one alarm(2)/setitimer(2) event per process and one event per POSIX timer id.
29    signal_timers: BTreeMap<SignalTimerId, SignalTimerState>,
30    // Real timers of a backend that provides process signal control recur only after an
31    // actual shared SIGALRM dequeue.
32    controlled_real_timer_deadlines: BTreeMap<DetPid, LogicalTime>,
33}

37#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
38pub(crate) enum SignalTimerId {
39    Alarm(DetPid),
40    Posix(DetPid, i32),
41    /// A deterministic child-exit `SIGCHLD`, synthesized at the child's
42    /// scheduler-ordered `Exit` grant (`t_exit`) rather than delivered by the
43    /// host-async kernel signal. `child` is the exiting process (a unique
44    /// coalescing key so multiple reaped children never collide); `parent` is
45    /// the process that receives the signal. Unlike `Alarm`/`Posix`, a
46    /// `ChildExit` event is one-shot and is never re-armed or cancelled, so it
47    /// is inserted directly into the timed `map` and bypasses the
48    /// `signal_timers` re-arm bookkeeping (see `insert_child_exit`).
49    ChildExit {
50        child: DetPid,
51        parent: DetPid,
52    },
53}
54
55impl SignalTimerId {
56    /// The process the timed signal is delivered to. For `ChildExit` this is the
57    /// *parent* (the reaper), not the exiting child.
58    pub(super) fn process(self) -> DetPid {
59        match self {
60            Self::Alarm(pid) | Self::Posix(pid, _) => pid,
61            Self::ChildExit { parent, .. } => parent,
62        }
63    }
64}

66#[derive(Debug, Clone, Copy, PartialEq, Eq)]
67struct SignalTimerState {
68    /// The pending deadline while `Active`; the last expiry while `Inactive`.
69    deadline: LogicalTime,
70    interval: LogicalTime,
71    phase: AlarmPhase,
72    target: SignalTarget,
73    sig: TimerSignal,
74}

76/// Whether a periodic `ITIMER_REAL` waits for its next expiry or for the
77/// dequeue of the copy its last expiry sent
78/// (https://github.com/rrnewton/hermit/issues/4139).
79///
80/// Linux's `it_real_fn` returns `HRTIMER_NORESTART`: an expiry stops the
81/// timer, and only a dequeue of a SIGALRM from the process's shared queue
82/// restarts it (`posixtimer_rearm_itimer`), on the original grid. So at most
83/// one timer SIGALRM is ever outstanding, and missed periods are dropped.
84#[derive(Debug, Clone, Copy, PartialEq, Eq)]
85pub(crate) enum AlarmPhase {
86    /// A deadline is queued in the timed map.
87    Active,
88    /// The last expiry fired and the timer waits for a shared-queue dequeue;
89    /// no deadline is queued. `copy` records that a SIGALRM is known to be in
90    /// the process's shared queue.
91    Inactive { copy: bool },
92}
93
94/// The first point after `t` on the grid `last + k * interval` (k >= 1).
95pub(crate) fn grid_after(last: LogicalTime, interval: LogicalTime, t: LogicalTime) -> LogicalTime {
96    let interval_ns = interval.as_nanos().max(1);
97    let last_ns = last.as_nanos();
98    let k = if t.as_nanos() >= last_ns {
99        (t.as_nanos() - last_ns) / interval_ns + 1
100    } else {
101        1
102    };
103    LogicalTime::from_nanos(last_ns.saturating_add(k.saturating_mul(interval_ns)))
104}

106/// What a signal timer does when it expires.
107#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
108pub enum TimerSignal {
109    /// Send this signal.
110    Send(Signal),
111    /// End the run with a named refusal. The POSIX timer notifies with this
112    /// real-time signal, which Detcore does not deliver from a timer; the timer could be
113    /// created and armed as on Linux, and only its expiration is unserviceable
114    /// (<https://github.com/rrnewton/hermit/issues/3893>).
115    RefuseRealTime(i32),
116}
117
118impl From<Signal> for TimerSignal {
119    fn from(signal: Signal) -> Self {
120        TimerSignal::Send(signal)
121    }
122}

133/// Who receives a signal timer's expiration.
134#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
135pub enum SignalTarget {
136    /// The timer's process (`SignalTimerId::process`), preferring this thread
137    /// when it is live: a process-directed signal, which any thread of the
138    /// process that does not block it may take.
139    Process(DetTid),
140    /// Exactly this thread: a `SIGEV_THREAD_ID` POSIX timer, which Linux
141    /// queues on the named thread's private queue and drops once that thread
142    /// has exited (<https://github.com/rrnewton/hermit/issues/4164>).
143    Thread(DetTid),
144}
145
146impl SignalTarget {
147    /// The preferred thread of a process-directed expiration, or the named
148    /// thread of a thread-directed one.
149    pub fn tid(self) -> DetTid {
150        match self {
151            SignalTarget::Process(tid) | SignalTarget::Thread(tid) => tid,
152        }
153    }
154}

156impl fmt::Display for SignalTarget {
157    fn fmt(&self, f: &mut fmt::Formatter) -> fmt::Result {
158        match self {
159            SignalTarget::Process(tid) => write!(f, "{}", tid),
160            SignalTarget::Thread(tid) => write!(f, "thread {}", tid),
161        }
162    }

165/// An event that occurs at a particular time in the execution, typically at an offset in the future.
166#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
167pub enum TimedEvent {
168    // An upcoming timer signal and who receives it.
169    SignalEvt(SignalTimerId, SignalTarget, TimerSignal),

171    /// A timed event on a particular thread (sleep, timeout, etc)

173}

298    fn clear_old_signal_timer(&mut self, id: SignalTimerId, old: Option<SignalTimerState>) {
299        if let Some(state) = old.filter(|state| state.phase == AlarmPhase::Active) {
300            // The `map` entry may already be gone if the alarm fired (was
301            // popped by `pop_if_before`) before being cleared. Clearing an
302            // already-fired alarm is a no-op rather than an invariant break.
303            let Some(set) = self.map.get_mut(&state.deadline) else {
304                return;
305            };
306
307            // Could use a drain_filter here, but it is nightly only:
308            let mut to_remove = None;
309            for evt in set.iter() {
310                if matches!(evt, TimedEvent::SignalEvt(evt_id, _, _) if *evt_id == id) {
311                    assert!(to_remove.is_none());
312                    to_remove = Some(*evt);
313                }
314            }
315            if let Some(evt) = to_remove {
316                assert!(set.remove(&evt));
317            }
318
319            // Preserve the invariant that `map` never holds an empty set, which
320            // `is_empty()` and `iter()` rely on.
321            if set.is_empty() {
322                self.map.remove(&state.deadline);
323            }
324        }
325    }

333    /// Register a one-shot, deterministic child-exit `SIGCHLD` to be delivered to
334    /// `parent` at logical time `ns` (the child's `Exit` grant time plus a tick).
335    /// `parent_tid` is the thread the delivery prefers: the thread that created
336    /// the child where gated waits model interruption, otherwise the leader.
337    /// Inserted directly into the timed `map`,
338    /// deliberately bypassing the `signal_timers` re-arm/cancel bookkeeping used
339    /// by `alarm`/`setitimer`/POSIX timers: a child exit fires exactly once and
340    /// is never re-armed or replaced, and its key (`ChildExit{child,parent}`) is
341    /// unique per exiting child, so it cannot collide with a concurrent
342    /// `Alarm`/`Posix` timer on the same process. If an identical event is
343    /// already queued at `ns` (the same child reported twice), the insert is a
344    /// no-op — the redundant delivery is coalesced, matching Linux `SIGCHLD`.
345    pub fn insert_child_exit(
346        &mut self,
347        ns: LogicalTime,
348        child: DetPid,
349        parent: DetPid,
350        parent_tid: DetTid,
351    ) {
352        let evt = TimedEvent::SignalEvt(
353            SignalTimerId::ChildExit { child, parent },
354            SignalTarget::Process(parent_tid),
355            Signal::SIGCHLD.into(),
356        );
357        // BTreeSet::insert returns false on a duplicate; coalescing it is
358        // intentional (see doc comment) rather than a panic.
359        self.map.entry(ns).or_default().insert(evt);
360    }

427    /// After the pop at `fired_at` of `dp`'s periodic `ITIMER_REAL` (which
428    /// `pop_if_before` re-armed at `fired_at + interval`), replace that
429    /// legacy recurrence: `Some(copy)` makes the timer `Inactive`, waiting for
430    /// a dequeue; `None` forwards it to the first grid point after `now`
431    /// (dropping missed periods). Does nothing unless the timer is still the
432    /// one that fired.
433    pub(crate) fn alarm_after_fire(
434        &mut self,
435        dp: DetPid,
436        fired_at: LogicalTime,
437        inactive_with_copy: Option<bool>,
438        now: LogicalTime,
439    ) {
440        self.signal_timer_after_fire(SignalTimerId::Alarm(dp), fired_at, inactive_with_copy, now);
441    }

443    /// [`Self::alarm_after_fire`] for any periodic signal timer `id`: a POSIX
444    /// timer waits for the dequeue of its signal too
445    /// (<https://github.com/rrnewton/hermit/issues/4262>).
446    pub(crate) fn signal_timer_after_fire(
447        &mut self,
448        id: SignalTimerId,
449        fired_at: LogicalTime,
450        inactive_with_copy: Option<bool>,
451        now: LogicalTime,
452    ) {
453        let Some(state) = self.signal_timers.get(&id).copied() else {
454            return;
455        };
456        if state.phase != AlarmPhase::Active
457            || state.interval == LogicalTime::ZERO
458            || state.deadline != fired_at + state.interval
459        {
460            return;
461        }
462        self.clear_old_signal_timer(id, Some(state));
463        let entry = self.signal_timers.get_mut(&id).unwrap();
464        match inactive_with_copy {
465            Some(copy) => {
466                entry.deadline = fired_at;
467                entry.phase = AlarmPhase::Inactive { copy };
468            }
469            None => {
470                let next = grid_after(fired_at, state.interval, now);
471                entry.deadline = next;
472                self.map
473                    .entry(next)
474                    .or_default()
475                    .insert(TimedEvent::SignalEvt(id, state.target, state.sig));
476            }
477        }
478    }

488    pub(crate) fn signal_timer_rearm_after(
489        &mut self,
490        id: SignalTimerId,

575    /// Return the next event if its target time of occurrence is before the supplied time.
576    /// Being a "pop", this destructively removes the entry.
577    // AUTONOMOUS-BOT-IMPLEMENTED
578    // TODO-HUMAN-REVIEW(#869)
579    pub fn pop_if_before(
580        &mut self,
581        current_time: LogicalTime,
582    ) -> Option<(LogicalTime, TimedEvent)> {
583        let (time_ns, evt) = if let Some(mut entry) = self.map.first_entry() {
584            let time_ns = *entry.key();
585            if time_ns <= current_time {
586                let set = entry.get_mut();
587                let evt = set.pop_first().expect("inner set cannot be empty");
588                if set.is_empty() {
589                    entry.remove();
590                }
591                Some((time_ns, evt))
592            } else {
593                None
594            }
595        } else {
596            None
597        }?;
598
599        if let TimedEvent::SignalEvt(SignalTimerId::Alarm(pid), _, _) = evt
600            && self.controlled_real_timer_deadlines.get(&pid) == Some(&time_ns)
601        {
602            self.controlled_real_timer_deadlines.remove(&pid);
603            return Some((time_ns, evt));
604        }
605        if let TimedEvent::SignalEvt(id, _, _) = evt
606            && self
607                .signal_timers
608                .get(&id)
609                .is_some_and(|state| state.deadline == time_ns)
610        {
611            let state = self.signal_timers.get_mut(&id).unwrap();
612            if state.interval == LogicalTime::ZERO {
613                self.signal_timers.remove(&id);
614            } else {
615                state.deadline = time_ns + state.interval;
616                let next_deadline = state.deadline;
617                self.map.entry(next_deadline).or_default().insert(evt);
618            }
619        }
620        Some((time_ns, evt))
621    }
622
623    /// Pop the next event unconditionally, if available.
624    pub fn pop(&mut self) -> Option<(LogicalTime, TimedEvent)> {
625        self.pop_if_before(LogicalTime::MAX)
626    }

628    /// The target time of the earliest pending event, without removing it.
629    ///
630    /// Callers that decide whether virtual time may be fast-forwarded need to
631    /// inspect the deadline *before* committing to popping it; a
632    /// [`LogicalTime::is_indefinite`] deadline is not a real deadline.
633    /// The period of signal timer `id`, if it is armed and periodic.
634    pub(crate) fn signal_timer_period(&self, id: SignalTimerId) -> Option<LogicalTime> {
635        self.signal_timers
636            .get(&id)
637            .map(|state| state.interval)
638            .filter(|interval| *interval != LogicalTime::ZERO)
639    }
640
641    pub fn next_deadline(&self) -> Option<LogicalTime> {
642        self.map.first_key_value().map(|(time_ns, _)| *time_ns)
643    }

645    /// Are there no timed events waiting?
646    pub fn is_empty(&self) -> bool {
647        // Here we rely on the invariant that there are no entries with empty sets on the RHS:
648        self.map.is_empty()
649    }

651    /// No deadline of any kind, and no alarm, interval or POSIX timer in any
652    /// phase (an inactive alarm is reconciled at every quiesced boundary):
653    /// nothing here can act at a scheduler pass.
654    pub fn holds_nothing(&self) -> bool {
655        self.map.is_empty()
656            && self.signal_timers.is_empty()
657            && self.controlled_real_timer_deadlines.is_empty()
658    }
659
660    /// Remove a specific thread from the set of those waiting on time to elapse.
661    pub fn remove(&mut self, dettid: DetTid) {
662        let mut to_remove: Option<LogicalTime> = None;
663        let mut already_removed = false;
664        for (time_key, set) in self.map.iter_mut() {
665            let removed = set.remove(&TimedEvent::ThreadEvt(dettid));
666            if removed {
667                if already_removed {
668                    panic!(
669                        "invariant violation: multiple entries for dtid {} in TimedEvents",
670                        dettid
671                    );
672                } else {
673                    already_removed = true;
674                }
675            }
676            // Cannot allow empty sets to remain:
677            if set.is_empty() {
678                to_remove = Some(*time_key);
679            }
680        }
681        if let Some(time) = to_remove {
682            let _ = self.map.remove(&time);
683        }
684    }
685
686    /// Iterate over the entries in the TimedEvents collection
687    pub fn iter(&self) -> impl Iterator<Item = (LogicalTime, TimedEvent)> + '_ {
688        self.map
689            .iter()
690            .flat_map(|(key, set)| set.iter().map(|dtid| (*key, *dtid)))
691    }
