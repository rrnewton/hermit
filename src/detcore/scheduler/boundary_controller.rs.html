26    /// Look ahead only for the scheduler's authenticated live incarnation.
27    /// This does not consume a record, choose an owner, or commit a turn.
28    pub(crate) fn boundary_budget_for_live_owner(
29        &self,
30        tid: DetTid,
31    ) -> anyhow::Result<Option<crate::slice_boundary::grant::Budget>> {
32        let Some(replayer) = &self.boundary_replayer else {
33            return Ok(None);
34        };
35        let mm = self.thread_mm(tid).ok_or_else(|| {
36            anyhow::anyhow!(
37                "slice-boundary protocol refusal: grant has no registered address space"
38            )
39        })?;
40        let owner = self
41            .boundary_owners
42            .as_ref()
43            .ok_or_else(|| {
44                anyhow::anyhow!("slice-boundary protocol refusal: replay has no lifecycle registry")
45            })?
46            .owner(tid, mm)?;
47        Ok(replayer.budget_for(owner))
48    }
49
50    pub(super) fn prepare_boundary_grant(&mut self, tid: DetTid) -> anyhow::Result<()> {
51        if let Some(budget) = self.boundary_budget_for_live_owner(tid)? {
52            let mm = self
53                .thread_mm(tid)
54                .expect("lookahead authenticated this address space");
55            self.boundary_budgets.queue(mm, budget)?;
56        }
57        Ok(())
58    }

60    /// Consume the annotation before publishing its unchanged resource map.
61    /// Neither a missing owner nor an invalid record can file that request.
62    pub(crate) fn accept_slice_boundary(
63        &mut self,
64        tid: DetTid,
65        mm: MmId,
66        boundary: &Boundary,
67    ) -> anyhow::Result<super::MaybePrintStack> {
68        let owners = self.boundary_owners.as_ref().ok_or_else(|| {
69            anyhow::anyhow!("slice-boundary protocol refusal: annotation outside its protocol")
70        })?;
71        owners.validate_boundary(tid, mm, boundary)?;
72        let recording = match &self.preemption_writer {
73            Some(writer) if writer.records_slice_boundaries() => true,
74            Some(_) => {
75                anyhow::bail!("slice-boundary protocol refusal: annotation has a legacy writer")
76            }
77            None => false,
78        };
79        if !recording && self.boundary_replayer.is_none() {
80            anyhow::bail!(
81                "slice-boundary protocol refusal: annotation has no recorder or replay cursor"
82            );
83        }
84        let next_count = if recording {
85            let next = self.recorded_event_count.checked_add(1).ok_or_else(|| {
86                anyhow::anyhow!("slice-boundary protocol refusal: stored index overflow")
87            })?;
88            if self
89                .preemption_writer
90                .as_ref()
91                .unwrap()
92                .next_boundary_index()
93                .and_then(|index| u64::try_from(index).ok())
94                != Some(self.recorded_event_count)
95            {
96                anyhow::bail!("slice-boundary protocol refusal: inconsistent recording index");
97            }
98            if self.boundary_replayer.is_none()
99                && let Some(iter) = &mut self.stacktrace_events
100                && let Some((requested, event, _)) = iter.peek()
101                && *requested == self.recorded_event_count
102                && event.is_some()
103            {
104                anyhow::bail!(
105                    "slice-boundary protocol refusal: ordinary event selector names a marker"
106                );
107            }
108            Some(next)
109        } else {
110            None
111        };
112        let mut stack = None;
113        if self.boundary_replayer.is_some() {
114            let consumed = self.consume_boundary_record(&Record::TimesliceEnd(boundary.clone()))?;
115            assert!(
116                consumed.keep_running,
117                "a boundary must use its original resource handoff"
118            );
119            stack = consumed.print_stack;
120        }
121        if recording {
122            let index = self
123                .preemption_writer
124                .as_mut()
125                .unwrap()
126                .insert_slice_boundary(boundary.clone())
127                .map_err(anyhow::Error::msg)?;
128            assert_eq!(index as u64, self.recorded_event_count);
129            self.recorded_event_count = next_count.unwrap();
130            if self.boundary_replayer.is_none()
131                && let Some(iter) = &mut self.stacktrace_events
132                && let Some((requested, event, path)) = iter.peek()
133                && *requested == index as u64
134            {
135                assert!(
136                    event.is_none(),
137                    "event selector was validated before consumption"
138                );
139                stack = Some(path.clone());
140                iter.next();
141            }
142        }
143        self.boundary_owners
144            .as_mut()
145            .unwrap()
146            .commit_boundary(boundary);
147        Ok(stack)
148    }

150    pub(crate) fn has_schedule_replay(&self) -> bool {
151        assert!(
152            self.replayer.is_none() || self.boundary_replayer.is_none(),
153            "two replay cursors cannot own the same schedule"
154        );
155        self.replayer.is_some() || self.boundary_replayer.is_some()
156    }
157
158    pub(super) fn replay_has_records_left(&self) -> bool {
159        assert!(self.replayer.is_none() || self.boundary_replayer.is_none());
160        match &self.boundary_replayer {
161            Some(replayer) => replayer.has_records_left(),
162            None => self
163                .replayer
164                .as_ref()
165                .is_some_and(Replayer::has_events_left),
166        }
167    }
168
169    pub(super) fn replay_next_resume(&self) -> Option<(DetTid, Option<LogicalTime>)> {
170        assert!(self.replayer.is_none() || self.boundary_replayer.is_none());
171        match &self.boundary_replayer {
172            // Exec uses the current stored cursor, with no reconstructed
173            // logical-duration heuristic and no cursor/clock/turn mutation.
174            Some(replayer) => replayer.next_thread().map(|tid| (tid, None)),
175            None => self.replayer.as_ref().and_then(Replayer::next_resume),
176        }
177    }
178
179    pub(super) fn consume_boundary_record(
180        &mut self,
181        observed: &Record,
182    ) -> anyhow::Result<ConsumeResult> {
183        assert!(self.replayer.is_none());
184        let replayer = self
185            .boundary_replayer
186            .as_mut()
187            .ok_or_else(|| anyhow::anyhow!("no slice-boundary replay cursor"))?;
188        let index = replayer.stored_index();
189        let action = replayer.observe(observed)?;
190        let complete = replayer.stored_index() != index;
191        // A partial branch has not reached its stored endpoint. A later event
192        // that happens to look identical cannot steal a requested stacktrace.
193        let print_stack = if complete {
194            self.boundary_stacktrace_events.remove(&index)
195        } else {
196            None
197        };
198        self.release_replay_priorities_if_exhausted();
199        let keep_running = match action {
200            ReplayAction::Continue(None) => true,
201            ReplayAction::ContextSwitch(now, next, None) => {
202                let current = match observed {
203                    Record::Event(event) => event.dettid,
204                    Record::TimesliceEnd(boundary) => boundary.identity.owner.tid,
205                };
206                assert_ne!(
207                    current, next,
208                    "same-owner annotations must not rotate the queue"
209                );
210                self.requeue_with_new_priority(current, REPLAY_DEFERRED_PRIORITY);
211                self.requeue_with_new_priority(next, REPLAY_FOREGROUND_PRIORITY);
212                // The expectation rides the existing grant. No old leader's
213                // legacy duration may leak into this new-format continuation.
214                self.timeslices.remove(&next);
215                !now
216            }
217            _ => unreachable!("the boundary cursor cannot resync or infer a logical duration"),
218        };
219        Ok(ConsumeResult {
220            keep_running,
221            print_stack,
222            event_ix: index as u64,
223            timeslice_remaining: None,
224        })
225    }

649                scheduler.register_thread_for_test(thread);

651                scheduler.register_preemption_thread(thread, priority, vfork);
652            }
653            scheduler.readmit_ready_thread(peer);
654            let before = (scheduler.turn, scheduler.committed_time);
655            scheduler.consume_schedevent(&branch);
656            assert_eq!(
657                scheduler.boundary_replayer.as_ref().unwrap().stored_index(),
658                1
659            );
660            assert!(!scheduler.replay_priorities_released);
661            assert_eq!(scheduler.priorities[&tid], REPLAY_FOREGROUND_PRIORITY);
662            assert_eq!(scheduler.priorities[&peer], REPLAY_DEFERRED_PRIORITY);
663            if exhaust {
664                scheduler.consume_boundary_record(&last).unwrap();
665                let late_vfork = DetTid::from_raw(7);
666                scheduler.register_thread_for_test(late_vfork);
667                assert!(
668                    !scheduler.replay_registration_priority(late_vfork, REPLAY_DEFERRED_PRIORITY)
669                );
670                scheduler.register_preemption_thread(late_vfork, REPLAY_DEFERRED_PRIORITY, true);
671                assert_eq!(scheduler.priorities[&late_vfork], DEFAULT_PRIORITY);
672                assert_eq!(
673                    scheduler
674                        .preemption_writer
675                        .as_ref()
676                        .unwrap()
677                        .current_priority(late_vfork),
678                    DEFAULT_PRIORITY - 1
679                );
680            }
681            assert_eq!(
682                scheduler.boundary_replayer.as_ref().unwrap().stored_index(),
683                if exhaust { 2 } else { 1 }
684            );
685            assert_eq!(scheduler.replay_has_records_left(), !exhaust);
686            assert_eq!(scheduler.replay_priorities_released, exhaust);
687            let controls = if exhaust {
688                [DEFAULT_PRIORITY, DEFAULT_PRIORITY]
689            } else {
690                [REPLAY_FOREGROUND_PRIORITY, REPLAY_DEFERRED_PRIORITY]
691            };
692            assert_eq!(scheduler.priorities[&tid], controls[0]);
693            assert_eq!(scheduler.run_queue.queued_priority(peer), Some(controls[1]));
694            assert_eq!((scheduler.turn, scheduler.committed_time), before);
695            let exported: serde_json::Value =
696                serde_json::from_str(&scheduler.preemption_writer.take().unwrap().into_string())
697                    .unwrap();
698            let rows = &exported["slice_boundary_recording"]["preemptions"]["per_thread"];
699            assert_eq!(rows["3"]["final_prio"], DEFAULT_PRIORITY);
700            assert_eq!(rows["4"]["final_prio"], DEFAULT_PRIORITY - 1);
701            if exhaust {
702                assert_eq!(rows["7"]["final_prio"], DEFAULT_PRIORITY - 1);
703            }
704            for thread in ["3", "4"] {
705                assert_eq!(rows[thread]["prio_changes"], serde_json::json!([]));
706            }
707        }
708    }
