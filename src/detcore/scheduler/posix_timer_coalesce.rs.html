9//! Linux's coalescing of POSIX interval timers, with overruns, on the traced
10//! path (<https://github.com/rrnewton/hermit/issues/4262>).
11//!
12//! Linux queues a POSIX timer's expiry as the timer's own preallocated entry,
13//! beside any ordinary copy of the same standard signal, and stops a periodic
14//! timer until that entry is dequeued (`posix_timer_fn` returns
15//! `HRTIMER_NORESTART`). The dequeue re-arms it on its grid
16//! (`hrtimer_forward_now`) and reports the missed periods as `si_overrun`; an
17//! entry whose timer was set again since it was queued is dropped inside the
18//! dequeue. Before this, the traced path re-armed every expiry at the old
19//! deadline plus one interval, so a slice that crossed N periods fired N
20//! signals in a burst, and each send merged into the one pending copy.
21//!
22//! In an observable (process, signal) pair (`signal_observable`) the
23//! scheduler keeps the entries Linux would hold (`signal_lists`), sends only
24//! the front of each list, and at each dequeue it sees (a reported delivery
25//! stop, a take, an unreported pass) pops the front, re-arms its timer and
26//! sends the next one (promotion). Every other pair takes the fallback: each
27//! expiry is sent at its pop with the overrun counted then, and the timer
28//! forwards to its next grid point.
29//!
30//! Record and replay keep the legacy recurrence (`coalesces_alarms`).
31

80/// What a reported delivery stop does with the signal it dequeued
81/// (`Scheduler::timer_signal_dequeue`).

83pub enum DeliveryAnswer {
84    /// Deliver the physical record as it is.
85    AsIs,
86    /// Deliver it with this siginfo (`Guest::set_signal_info`).

152/// The overrun a captured entry's delivery at `at` reports.
153fn captured_overrun(captured: &CapturedEntry, at: LogicalTime) -> i32 {
154    captured
155        .grid
156        .map_or(0, |(last, interval)| overrun_at(last, interval, at))
157}

159/// The overrun Linux reports for a periodic timer last expired at `last`
160/// with `interval`, dequeued at `t`: `hrtimer_forward_now` moves the expiry
161/// past `t` by k intervals, and `it_overrun_last` is k - 1, clamped to
162/// `INT_MAX` (`timer_overrun_to_int`).
163pub(crate) fn overrun_at(last: LogicalTime, interval: LogicalTime, t: LogicalTime) -> i32 {
164    let interval_ns = interval.as_nanos().max(1);
165    // k - 1, the whole intervals elapsed since `last`.
166    let missed = t.as_nanos().saturating_sub(last.as_nanos()) / interval_ns;
167    i32::try_from(missed).unwrap_or(i32::MAX)
168}

191        self.posix_timer_overruns.insert((process, timer), 0);

193        if (1..32).contains(&signo) {
194            self.discarded_at_send.note_timer_history(process, signo);
195        }
196    }

236    /// Whether the scheduler sees every dequeue of `signal` in `process` at a
237    /// committed point (4262 design §3.4). Sticky once false; the first
238    /// false answer is a transition (`lose_timer_observability`).
239    pub(crate) fn signal_observable(&mut self, process: DetPid, signal: i32) -> bool {
240        if !self.coalesces_alarms()
241            || !(1..32).contains(&signal)
242            || EXCLUDED_SIGNALS.contains(&signal)
243        {
244            return false;
245        }
246        if self.signal_unobservable.contains(&(process, signal)) {
247            return false;
248        }
249        let bit = kernel_signal_bit(signal);
250        let threads = self.process_signal_targets(process);
251        let leader = DetTid::from_raw(process.as_raw());
252        let host_timed = threads
253            .iter()
254            .any(|thread| self.host_timed_signals(*thread) & bit != 0);
255        let child_exit = self
256            .thread_tree
257            .unreaped_children_of(process)
258            .into_iter()
259            .filter_map(|child| self.thread_tree.notification_signal(child))
260            .any(|exit_signal| exit_signal == signal);
261        let real_timer =
262            signal == libc::SIGALRM && self.blocked.timed_waiters.alarm_phase(process).is_some();
263        let observable = !self.background_call_processes.contains(&process)
264            && !host_timed
265            && !child_exit
266            && !real_timer
267            && threads.contains(&leader);
268        if !observable {
269            self.lose_timer_observability(process, signal);
270        }
271        observable
272    }

274    /// `signal` of `process` stops being observable (4262 design §4.11). A
275    /// list holding a single valid entry that is its own physical record (or
276    /// a `Plain` one that is not a `Proxy`), with no current park of the
277    /// signal, falls back: the lists go, and the physical record is delivered
278    /// as it stands. Anything else is refused by name, before any producer
279    /// the scheduler cannot see runs.
280    pub(crate) fn lose_timer_observability(&mut self, process: DetPid, signal: i32) {
281        if !self.signal_unobservable.insert((process, signal)) {
282            return;
283        }
284        debug!(
285            "[dpid {}] signal {}: timer entries are not observable from here on; expiries take the fallback.",
286            process, signal
287        );
288        for queue in self.process_queues(process) {
289            let entries = self.discarded_at_send.lists.entries(queue, signal);
290            let Some((front, class, _)) = self.discarded_at_send.lists.front(queue, signal) else {
291                continue;
292            };
293            let single_own = entries.len() == 1
294                && class == FrontClass::Own
295                && match front {
296                    Entry::Plain { .. } => true,
297                    Entry::Timer(timer) => self.timer_entry_valid(&timer),
298                };
299            if !single_own {
300                refuse(format_args!(
301                    "hermit: signal {signal} of process {process} stopped being observable (a call outside the schedule, a host-timed source, or a child's exit signal) while its pending entries in {queue:?} were {entries:?} with a {class:?} front: Linux would deliver them one by one, which the fallback cannot, so the run is refused (https://github.com/rrnewton/hermit/issues/4262)"
302                ));
303            }
304            self.discarded_at_send.lists.take_all(queue, signal);
305        }
306        // A periodic timer that waited for the dequeue of the entry the lists
307        // held goes onto the fallback's recurrence now: no dequeue the
308        // scheduler sees will re-arm it (the Claude review of
309        // https://github.com/rrnewton/hermit/pull/4491, P1-1).
310        if let Ok(sig) = Signal::try_from(signal) {
311            let now = self.committed_time;
312            for id in self
313                .blocked
314                .timed_waiters
315                .inactive_posix_timers(process, sig)
316            {
317                let next = self.blocked.timed_waiters.signal_timer_rearm_after(id, now);
318                debug!(
319                    "[dpid {}] {:?} waited for its entry's dequeue; on the fallback, re-armed at {:?}.",
320                    process, id, next
321                );
322            }
323        }
324        if self
325            .discarded_at_send
326            .parked_timers(process, signal)
327            .iter()
328            .any(|producer| self.posix_timer_expiry_current(producer))
329        {
330            refuse(format_args!(
331                "hermit: signal {signal} of process {process} stopped being observable while a POSIX timer's ignored expiry of it is parked: Linux re-queues it when the signal stops being ignored, which the fallback cannot follow, so the run is refused (https://github.com/rrnewton/hermit/issues/4262)"
332            ));
333        }
334    }

336    /// Under record and replay, a POSIX timer copy was sent into `queue` as
337    /// `signal` (`replayed_timer_copies`); one sent while a signalfd covers
338    /// the signal may be read at once.
339    pub(crate) fn note_replayed_timer_copy(&mut self, queue: QueueId, signal: i32) {
340        if !self.recordreplay_modes || !self.models_signal_targets {
341            return;
342        }
343        self.replayed_timer_copies.insert((queue, signal));
344        let bit = kernel_signal_bit(signal);
345        if self.signalfd_masks.values().any(|mask| mask & bit != 0) {
346            self.replay_unproven_copies.insert((queue, signal));
347        }
348    }

350    /// Under record and replay, a signalfd now covers `mask` in `process`
351    /// (`None`: any process, a signalfd's description being shareable): the
352    /// timer copies of those signals may be read.
353    pub(crate) fn note_replay_take_cover(&mut self, process: Option<DetPid>, mask: u64) {
354        if !self.recordreplay_modes {
355            return;
356        }
357        let covered: Vec<(QueueId, i32)> = self
358            .replayed_timer_copies
359            .iter()
360            .copied()
361            .filter(|(queue, signal)| {
362                let owner = match queue {
363                    QueueId::Process(owner) => *owner,
364                    QueueId::Thread(thread) => self.sigchld_process(*thread),
365                };
366                mask & kernel_signal_bit(*signal) != 0 && process.is_none_or(|p| p == owner)
367            })
368            .collect();
369        self.replay_unproven_copies.extend(covered);
370    }

372    /// Under record and replay, an exec by `caller` that survives as
373    /// `leader` destroyed every other thread of the group (`siblings`, the
374    /// former leader among them) with its own queue, and moved the caller's
375    /// queue to the leader's tid: the replay records move with it, so that
376    /// `replay_exec_refusal` finds them (the Codex re-review of
377    /// https://github.com/rrnewton/hermit/pull/4491, P1).
378    pub(crate) fn transfer_replay_copies(
379        &mut self,
380        caller: DetTid,
381        leader: DetTid,
382        siblings: &[DetTid],
383    ) {
384        let destroyed =
385            |queue: &QueueId| matches!(queue, QueueId::Thread(t) if siblings.contains(t));
386        let rekey = |(queue, signal): (QueueId, i32)| match queue {
387            QueueId::Thread(t) if t == caller => (QueueId::Thread(leader), signal),
388            other => (other, signal),
389        };
390        for set in [
391            &mut self.replayed_timer_copies,
392            &mut self.replay_unproven_copies,
393        ] {
394            *set = std::mem::take(set)
395                .into_iter()
396                .filter(|(queue, _)| !destroyed(queue))
397                .map(rekey)
398                .collect();
399        }
400    }

402    /// Under record and replay, an exec of `process` whose survivor is
403    /// `survivor`, with a signal pending that a timer copy may have been
404    /// taken for unobserved: a replayed take dequeues nothing, so replay would
405    /// flush another record than the recording did. Refused by name, as
406    /// before https://github.com/rrnewton/hermit/issues/4262 (the Claude
407    /// review of https://github.com/rrnewton/hermit/pull/4491, P1-2). The
408    /// process's records go either way.
409    pub(crate) fn replay_exec_refusal(
410        &mut self,
411        survivor: DetTid,
412        process: DetPid,
413    ) -> Option<String> {
414        if !self.recordreplay_modes {
415            return None;
416        }
417        let (private, shared) = (QueueId::Thread(survivor), QueueId::Process(process));
418        let unproven: Vec<(QueueId, i32)> = self
419            .replay_unproven_copies
420            .iter()
421            .copied()
422            .filter(|(queue, _)| *queue == private || *queue == shared)
423            .collect();
424        self.replayed_timer_copies
425            .retain(|(queue, _)| *queue != private && *queue != shared);
426        self.replay_unproven_copies
427            .retain(|(queue, _)| *queue != private && *queue != shared);
428        if unproven.is_empty() {
429            return None;
430        }
431        let state = match self.read_sleeper_signal_state(survivor) {
432            Ok(state) => state,
433            Err(failure) => {
434                return Some(format!(
435                    "the pending signals of thread {survivor} could not be read ({failure:?})"
436                ));
437            }
438        };
439        unproven.into_iter().find_map(|(queue, signal)| {
440            let pending = match queue {
441                QueueId::Thread(_) => state.thread_pending,
442                QueueId::Process(_) => state.shared_pending,
443            };
444            (pending & kernel_signal_bit(signal) != 0).then(|| {
445                format!(
446                    "signal {signal} is pending in {queue:?}, and under record and replay a POSIX timer's copy of it may have been read by a signalfd, which a replayed read does not dequeue, so the exec would replay another flush"
447                )
448            })
449        })
450    }

452    /// Under record and replay, a reported delivery of `signal` to `thread`
453    /// left its queues as read (`thread_holds`, `process_holds`): a timer
454    /// copy no longer there was taken where the schedule sees it.
455    pub(crate) fn note_replayed_copy_taken(
456        &mut self,
457        thread: DetTid,
458        signal: i32,
459        thread_holds: bool,
460        process_holds: bool,
461    ) {
462        if !self.recordreplay_modes {
463            return;
464        }
465        let process = self.sigchld_process(thread);
466        for (queue, holds) in [
467            (QueueId::Thread(thread), thread_holds),
468            (QueueId::Process(process), process_holds),
469        ] {
470            if !holds {
471                self.replayed_timer_copies.remove(&(queue, signal));
472                self.replay_unproven_copies.remove(&(queue, signal));
473            }
474        }
475    }

477    /// The processes with timer-signal entries or parked expiries.
478    pub(crate) fn processes_with_timer_entries(&self) -> Vec<DetPid> {
479        let mut processes: Vec<DetPid> = self
480            .discarded_at_send
481            .lists
482            .listed()
483            .into_iter()
484            .map(|(queue, _)| match queue {
485                QueueId::Process(process) => process,
486                QueueId::Thread(thread) => self.sigchld_process(thread),

495    /// `child` was created: its exit signal, posted to its wait parent when it
496    /// exits, ends that pair's observability (`signal_observable`).
497    pub(crate) fn child_ends_timer_observability(&mut self, child: DetPid) {
498        // A forked child copies its creator's actions.
499        if let Some(parent) = self.thread_tree.parent_process(&child) {
500            let inherited: Vec<_> = self
501                .handler_masks
502                .iter()
503                .filter(|((owner, _), _)| *owner == parent)
504                .map(|((_, signal), action)| (*signal, *action))
505                .collect();
506            for (signal, action) in inherited {
507                self.handler_masks.insert((child, signal), action);
508            }
509        }
510        if !self.coalesces_alarms() {
511            return;
512        }
513        let (Some(parent), Some(signal)) = (
514            self.thread_tree.parent_process(&child),
515            self.thread_tree.notification_signal(child),
516        ) else {
517            return;
518        };
519        if (1..32).contains(&signal) {
520            self.lose_timer_observability(parent, signal);
521        }
522    }

537    /// Fire the expiry at `deadline` of POSIX timer `id`, notifying `target`
538    /// with `signal`, which `pop_if_before` has just popped (and, if
539    /// periodic, re-armed one interval later). Replaces that recurrence.
540    pub(super) fn fire_posix_timer(
541        &mut self,
542        deadline: LogicalTime,
543        id: SignalTimerId,
544        target: SignalTarget,
545        signal: Signal,
546    ) {
547        let SignalTimerId::Posix(process, timer) = id else {
548            unreachable!("fire_posix_timer takes a POSIX timer");
549        };
550        let signo = signal as libc::c_int;
551        if !self.models_signal_targets {
552            // Elsewhere the scheduler cannot single out a thread's queue.
553            match target {
554                SignalTarget::Thread(thread) => self.fire_thread_timer(id, thread, signal),
555                SignalTarget::Process(tid) => {
556                    let producer = self.posix_timer_producer(id);
557                    self.fire_process_timer(process, tid, signal, producer, None);
558                }
559            }
560            return;
561        }
562        let now = self.committed_time;
563        if !self.coalesces_alarms() || !self.signal_observable(process, signo) {
564            self.fire_posix_timer_fallback(deadline, id, target, signal, now);
565            return;
566        }
567        if let SignalTarget::Thread(thread) = target
568            && !self.process_signal_targets(process).contains(&thread)
569        {
570            // Its recurrence stands: each period pops and is dropped
571            // (disclosure 5).
572            info!(
573                "[dpid {}] Timer expired after its thread {} exited; dropped.",
574                process, thread
575            );
576            return;
577        }
578        let periodic = self.blocked.timed_waiters.signal_timer_period(id).is_some();
579        // A periodic timer waits for the dequeue of its entry, or of its
580        // parked expiry's re-queued entry (`HRTIMER_NORESTART`).
581        if periodic {
582            self.blocked
583                .timed_waiters
584                .signal_timer_after_fire(id, deadline, Some(true), now);
585        }
586        let entry = self.timer_entry_now(process, timer);
587        let (queue, named, preferred) = match target {
588            SignalTarget::Thread(thread) => (QueueId::Thread(thread), thread, thread),
589            SignalTarget::Process(tid) => (
590                QueueId::Process(process),
591                DetTid::from_raw(process.as_raw()),
592                tid,
593            ),
594        };
595        // Every expiry first refreshes the timer's queued entry (Linux sets
596        // `it_sigqueue_seq` and `it_sig_periodic` before ALREADY_PENDING).
597        let was_deliverable = self.list_deliverable(queue, signo);
598        if self.discarded_at_send.lists.refresh_timer(entry) {
599            info!(
600                "[dpid {}] POSIX timer {} expired with its entry still queued; refreshed.",
601                process, timer
602            );
603            if !was_deliverable && self.list_deliverable(queue, signo) {
604                self.wake_for_entry(queue, signal, preferred);
605            }
606            return;
607        }
608        if self.discarded_when_sent_to(named, signo, None) {
609            // Ignored: a periodic timer parks (4351's record keeps it, and the
610            // copy, which carries `SI_TIMER`); a one-shot is dropped.
611            let producer = periodic.then(|| self.posix_timer_producer(id)).flatten();
612            if producer.is_none() {
613                self.discarded_at_send.forget_park(process, timer);
614            }
615            let info = self.timer_signal_info(signo, &entry, 0);
616            self.send_timer_record(
617                queue,
618                signal,
619                preferred,
620                info,
621                SendFate::Classify(producer.map(SignalProducer::Posix)),
622            );
623            self.discarded_at_send.note_discarded_si_timer(
624                queue,
625                signo,
626                super::signal_lists::SI_TIMER,
627            );
628            return;
629        }
630        // A parked timer whose signal is now blocked goes back to the queue.
631        self.discarded_at_send.forget_park(process, timer);
632        let standing = self
633            .discarded_at_send
634            .is_discarded(queue, signo)
635            .then(|| self.discarded_at_send.discarded_is_si_timer(queue, signo));
636        match self
637            .discarded_at_send
638            .lists
639            .append_timer(queue, signo, entry, standing)
640        {
641            TimerAppend::SendFront | TimerAppend::ProxyFront => {
642                let info = self.timer_signal_info(signo, &entry, 0);
643                self.send_timer_record(queue, signal, preferred, info, SendFate::Kept);
644            }
645            TimerAppend::Behind => {
646                info!(
647                    "[dpid {}] POSIX timer {} expired behind another entry of signal {}; queued.",
648                    process, timer, signal
649                );
650                if !was_deliverable {
651                    self.wake_for_entry(queue, signal, preferred);
652                }
653            }
654            TimerAppend::AlreadyPending => unreachable!("refreshed above"),
655        }
656        debug_assert!(self.discarded_at_send.lists.invariants_hold());
657    }

659    /// The fallback (4262 design §4.12): send the expiry at the pop, with
660    /// the overrun counted then, and forward the timer to its first grid
661    /// point after the pop.
662    fn fire_posix_timer_fallback(
663        &mut self,
664        deadline: LogicalTime,
665        id: SignalTimerId,
666        target: SignalTarget,
667        signal: Signal,
668        now: LogicalTime,
669    ) {
670        let SignalTimerId::Posix(process, timer) = id else {
671            return;
672        };
673        let overrun = match self.blocked.timed_waiters.signal_timer_period(id) {
674            Some(interval) if self.coalesces_alarms() => {
675                self.blocked
676                    .timed_waiters
677                    .signal_timer_after_fire(id, deadline, None, now);
678                overrun_at(deadline, interval, now)
679            }
680            _ => 0,
681        };
682        if self.coalesces_alarms() {
683            self.posix_timer_overruns.insert((process, timer), overrun);
684        }
685        let entry = self.timer_entry_now(process, timer);
686        let info = self.timer_signal_info(signal as libc::c_int, &entry, overrun);
687        match target {
688            SignalTarget::Thread(thread) => self.fire_thread_timer(id, thread, signal),
689            SignalTarget::Process(tid) => {
690                let producer = self.posix_timer_producer(id);
691                self.fire_process_timer(process, tid, signal, producer, Some(info));
692            }
693        }
694    }

712    /// Send a timer entry's physical record `info` into `queue` as `signal`:
713    /// a shared-queue send goes through `fire_process_alarm`, the one body
714    /// that keeps the shared-queue bookkeeping, preferring `preferred`; a
715    /// private one through `send_guest_signal`'s `rt_tgsigqueueinfo`.
716    fn send_timer_record(
717        &mut self,
718        queue: QueueId,
719        signal: Signal,
720        preferred: DetTid,
721        info: [u8; 128],
722        fate: SendFate,
723    ) -> bool {
724        match queue {
725            QueueId::Process(process) => {
726                self.guest_send_landed = None;

729                // gives a copy-less Inactive ITIMER_REAL its copy
730                // (https://github.com/rrnewton/hermit/issues/4139).
731                if sent && signal == Signal::SIGALRM && self.guest_send_landed != Some(false) {
732                    self.note_guest_sigalrm(

734                        super::itimer_coalesce::GuestSigalrmTarget::Process(process),

736                }

738            }
739            QueueId::Thread(thread) => {
740                let process = self.sigchld_process(thread);
741                #[cfg(test)]
742                {
743                    self.host_signal_attempts += 1;
744                }
745                self.send_guest_signal(thread, signal, Some(process), Some(info), thread, fate);

793    /// At a reported delivery stop of `thread`, which dequeued `signal` at its
794    /// logical time `at`: identify the entry it took and answer what to
795    /// deliver (4262 design rev 8 A', §4.6 step 1). The signal is the one the
796    /// stop reports; the queue comes from the model, private first. With no
797    /// list for the signal in either queue, nothing is identified, and 4351's
798    /// record decides as before.
799    ///
800    /// `at` is `None` for an unreported pass (4262 design §4.10), whose hook
801    /// cannot replace the siginfo and has no thread time: a valid entry
802    /// standing on its own record is delivered with that record (overrun 0,
803    /// disclosed) and its timer reconciled at the committed time, and one
804    /// standing on another sender's record is refused by name.
805    pub(crate) fn timer_signal_dequeue(
806        &mut self,
807        thread: DetTid,
808        signal: i32,
809        at: Option<LogicalTime>,
810    ) -> DeliveryAnswer {
811        if let Some(unfinished) = self.pending_timer_dequeues.remove(&thread) {
812            self.retire_unfinished_dequeue(unfinished);
813        }
814        if !self.coalesces_alarms() || !(1..32).contains(&signal) {
815            return DeliveryAnswer::AsIs;
816        }
817        let process = self.sigchld_process(thread);
818        let Some(queue) = self.identify_listed_queue(thread, process, signal) else {
819            return DeliveryAnswer::AsIs;
820        };
821        let (front, class, _) = self
822            .discarded_at_send
823            .lists
824            .front(queue, signal)
825            .expect("an identified queue has a list");
826        let reported = at.is_some();
827        let at = at.unwrap_or(self.committed_time);
828        let stale = matches!(front, Entry::Timer(entry) if !self.timer_entry_valid(&entry));
829        if !reported && class == FrontClass::Proxy && !stale {
830            refuse(format_args!(
831                "hermit: thread {thread} took signal {signal} from {queue:?} where the backend passes it on unreported, so its siginfo cannot be replaced, and the pending record is another sender's standing for {front:?}: the guest would see that sender's siginfo, so the run is refused (https://github.com/rrnewton/hermit/issues/4262)"
832            ));
833        }
834        let (answer, reconcile) = match front {
835            Entry::Plain { siginfo, .. } => match class {
836                FrontClass::Own => (DeliveryAnswer::AsIs, None),
837                FrontClass::Proxy => (DeliveryAnswer::Rewrite(siginfo), None),
838            },
839            Entry::Timer(_) if stale => (DeliveryAnswer::Suppress, None),
840            Entry::Timer(_) if reported => (DeliveryAnswer::TimerEntry, None),
841            Entry::Timer(entry) => {
842                let overrun = self.dequeue_overrun(&entry, at);
843                (DeliveryAnswer::AsIs, Some((entry.timer, at, overrun)))
844            }
845        };
846        let awaiting_info = match (answer, front) {
847            (DeliveryAnswer::TimerEntry, Entry::Timer(entry)) => Some(self.capture_entry(entry)),
848            _ => None,
849        };
850        self.pending_timer_dequeues.insert(
851            thread,
852            PendingDequeue {
853                queue,
854                signal,
855                delivery: true,
856                reconcile,
857                awaiting_info,
858                early: None,
859            },
860        );
861        answer
862    }

864    /// The siginfo of the valid timer entry `thread`'s delivery stop took
865    /// (`DeliveryAnswer::TimerEntry`), asked for once the delivery's turn is
866    /// granted, at the thread's time `at` then: Linux counts the overrun as it
867    /// dequeues, just before the handler runs, so it is counted where the
868    /// handler's time begins. Counted at the stop instead, it trailed the
869    /// handler's own clock by the delivery's turn: the T1 storm row measured
870    /// 197 against the 198 grid points the handler's clock showed.
871    pub(crate) fn timer_delivery_info(
872        &mut self,
873        thread: DetTid,
874        signal: i32,
875        at: LogicalTime,
876    ) -> Option<KernelSiginfo> {
877        let pending = self.pending_timer_dequeues.get(&thread)?;
878        let captured = pending.awaiting_info.filter(|_| pending.signal == signal)?;
879        let at = at.max(self.committed_time);
880        let overrun = captured_overrun(&captured, at);
881        let pending = self.pending_timer_dequeues.get_mut(&thread)?;
882        pending.awaiting_info = None;
883        pending.reconcile = Some((captured.entry.timer, at, overrun));
884        Some(timer_siginfo(
885            signal,
886            captured.entry.timer,
887            overrun,
888            captured.value,
889        ))
890    }

896            Some((last, interval, AlarmPhase::Inactive { .. }))

898            {

901            _ => None,

914    /// A take of `signal` by `thread` at its time `at` (4262 design §4.4,
915    /// rev 8 A'): the returned signal is authoritative. With no list for it
916    /// in either queue, 4351's record decides as before (`take_from`).
917    /// Otherwise the entry is identified from the model, private first,
918    /// checked against the queues read after the take, where read (the
919    /// identified queue's bit clear, the other queue's as its model says),
920    /// popped, its timer re-armed, and the next entry promoted before the
921    /// caller takes again.
922    pub(crate) fn timer_signal_take(
923        &mut self,
924        thread: DetTid,
925        signal: i32,
926        at: LogicalTime,
927        after: Option<(u64, u64)>,
928    ) -> TakeOutcome {
929        // The take is in the caller's turn, granted at the committed time;
930        // the caller's own clock may not have caught up with it yet (a sleep
931        // it just woke from), and the overrun must not trail it.
932        let at = at.max(self.committed_time);
933        let process = self.sigchld_process(thread);
934        let listed = if self.coalesces_alarms() && (1..32).contains(&signal) {
935            self.identify_listed_queue(thread, process, signal)
936        } else {
937            None
938        };
939        let Some(queue) = listed else {
940            return TakeOutcome {
941                skip: self.discarded_at_send.take_from(thread, process, signal),
942                rewrite: None,
943            };
944        };
945        if let Some((thread_pending, shared_pending)) = after {
946            let bit = kernel_signal_bit(signal);
947            // A take dequeues in the caller's own turn: no send can follow
948            // the kernel's dequeue before this check.
949            self.check_taken_queues(
950                thread,
951                queue,
952                signal,
953                false,
954                thread_pending & bit != 0,
955                shared_pending & bit != 0,
956            );
957        }
958        let (front, class, _) = self
959            .discarded_at_send
960            .lists
961            .front(queue, signal)
962            .expect("an identified queue has a list");
963        let (outcome, reconcile) = match front {
964            Entry::Plain { siginfo, .. } => (
965                TakeOutcome {
966                    skip: false,
967                    rewrite: (class == FrontClass::Proxy).then_some(siginfo),
968                },
969                None,
970            ),
971            Entry::Timer(entry) if !self.timer_entry_valid(&entry) => (
972                TakeOutcome {
973                    skip: true,
974                    rewrite: None,
975                },
976                None,
977            ),
978            Entry::Timer(entry) => {
979                let overrun = self.dequeue_overrun(&entry, at);
980                let info = self.timer_signal_info(signal, &entry, overrun);
981                (
982                    TakeOutcome {
983                        skip: false,
984                        rewrite: Some(KernelSiginfo::from_user(&info)),
985                    },
986                    Some((entry.timer, at, overrun)),
987                )
988            }
989        };
990        self.discarded_at_send.take(queue, signal);
991        self.pop_and_promote(
992            PendingDequeue {
993                queue,
994                signal,
995                delivery: false,
996                reconcile,
997                awaiting_info: None,
998                early: None,
999            },
1000            thread,
1001        );
1002        outcome
1003    }

1005    /// A take for the signals `set` by `thread` at `at` failed copying out
1006    /// the record it dequeued, which the kernel does not name; afterwards the
1007    /// queues held `thread_pending` and `shared_pending` (rev 7 A3, rev 8 A',
1008    /// N8-3). A listed (queue, signal) of the set whose bit is now clear is
1009    /// the one taken: nothing else consumes in the caller's turn. With none,
1010    /// the take was of an unlisted signal, which 4351's record decides
1011    /// (`take_faulted`). Two such cannot be told apart, and refuse by name.
1012    /// A stale entry is popped and the take repeats, as Linux would have
1013    /// skipped it; a valid one is popped, re-armed and promoted, and the
1014    /// fault stands. Returns whether to take again.
1015    pub(crate) fn timer_signal_take_faulted(
1016        &mut self,
1017        thread: DetTid,
1018        set: u64,
1019        thread_pending: u64,
1020        shared_pending: u64,
1021        at: LogicalTime,
1022    ) -> bool {
1023        let at = at.max(self.committed_time);
1024        let process = self.sigchld_process(thread);
1025        let (private, shared) = (QueueId::Thread(thread), QueueId::Process(process));
1026        let mut taken = Vec::new();
1027        if self.coalesces_alarms() {
1028            for (queue, pending) in [(private, thread_pending), (shared, shared_pending)] {
1029                for signal in 1..32 {
1030                    let bit = kernel_signal_bit(signal);
1031                    if set & bit != 0
1032                        && pending & bit == 0
1033                        && self.discarded_at_send.lists.is_listed(queue, signal)
1034                    {
1035                        taken.push((queue, signal));
1036                    }
1037                }
1038            }
1039        }
1040        let (queue, signal) = match taken.as_slice() {
1041            [] => {
1042                return self.discarded_at_send.take_faulted(
1043                    thread,
1044                    process,
1045                    set,
1046                    thread_pending,
1047                    shared_pending,
1048                );
1049            }
1050            [one] => *one,
1051            _ => refuse(format_args!(
1052                "hermit: thread {thread}'s rt_sigtimedwait failed copying out a signal it dequeued, and the entries of {taken:?} all left their queues, so which one Linux took cannot be told; the run is refused (https://github.com/rrnewton/hermit/issues/4262)"
1053            )),
1054        };
1055        let (front, _, _) = self
1056            .discarded_at_send
1057            .lists
1058            .front(queue, signal)
1059            .expect("a listed queue has a front");
1060        let (skip, reconcile) = match front {
1061            Entry::Timer(entry) if !self.timer_entry_valid(&entry) => (true, None),
1062            Entry::Timer(entry) => {
1063                let overrun = self.dequeue_overrun(&entry, at);
1064                (false, Some((entry.timer, at, overrun)))
1065            }
1066            Entry::Plain { .. } => (false, None),
1067        };
1068        self.discarded_at_send.take(queue, signal);
1069        self.pop_and_promote(
1070            PendingDequeue {
1071                queue,
1072                signal,
1073                delivery: false,
1074                reconcile,
1075                awaiting_info: None,
1076                early: None,
1077            },
1078            thread,
1079        );
1080        skip
1081    }

1083    /// The listed queue a dequeue of `signal` by `thread` took from, private
1084    /// first (Linux's `dequeue_signal`), or `None` when neither queue has a
1085    /// list for it (the no-list branch) or the private queue's copy is a
1086    /// discarded one with no list (4351 decides).
1087    fn identify_listed_queue(
1088        &self,
1089        thread: DetTid,
1090        process: DetPid,
1091        signal: i32,
1092    ) -> Option<QueueId> {
1093        let (private, shared) = (QueueId::Thread(thread), QueueId::Process(process));
1094        let lists = &self.discarded_at_send.lists;
1095        if !lists.is_listed(private, signal) && !lists.is_listed(shared, signal) {
1096            return None;
1097        }
1098        let queue = if lists.is_listed(private, signal)
1099            || self.discarded_at_send.is_discarded(private, signal)

1101            private
1102        } else {

1104        };

1108    /// The overrun a valid entry's delivery at `at` reports: for a periodic
1109    /// timer waiting for this dequeue, the missed periods since its last
1110    /// expiry; 0 for a one-shot.
1111    fn dequeue_overrun(&self, entry: &TimerEntry, at: LogicalTime) -> i32 {
1112        let id = SignalTimerId::Posix(entry.process, entry.timer);
1113        match self.blocked.timed_waiters.signal_timer_state(id) {
1114            Some((last, interval, AlarmPhase::Inactive { .. }))
1115                if entry.periodic_at_send && interval != LogicalTime::ZERO =>
1116            {
1117                overrun_at(last, interval, at)
1118            }
1119            _ => 0,
1120        }

1123    /// After a reported stop's hooks read `thread`'s private queue
1124    /// (`thread_holds`) and its process's shared one (`process_holds`): pop
1125    /// the entry the stop took, re-arm its timer, and send the next entry
1126    /// (4262 design §4.6 steps 2 and 3). A copy resumed blocked is requeued
1127    /// by the kernel, so nothing pops. The read must agree with the model
1128    /// (rev 9 A''): the identified queue's bit is clear, since the promotion
1129    /// has not been sent yet, and the other queue's bit is set exactly when
1130    /// its list or its discarded bit says a copy is there.
1131    /// Returns the dequeue to pop (`pop_and_promote`), once 4351's record has
1132    /// retired its bits.
1133    pub(crate) fn finish_timer_signal_dequeue(
1134        &mut self,
1135        thread: DetTid,
1136        signal: i32,
1137        how: Consumption,
1138        thread_holds: bool,
1139        process_holds: bool,
1140    ) -> Option<PendingDequeue> {
1141        let mut pending = self.pending_timer_dequeues.remove(&thread)?;
1142        // A delivery resumed blocked requeues its copy (`ptrace_signal`'s
1143        // `send_signal_locked`). After an early pop, while the joined send's
1144        // copy is still pending in the identified queue, `legacy_queue` drops
1145        // that requeue, and the kernel holds exactly the joined send, as the
1146        // list does; natively the blocked signal would have stayed pending and
1147        // coalesced the same way. The early-popped front is then retired below
1148        // with the other unfinished dequeues (the Claude re-check of the
1149        // in-flight-dequeue fix, R4). If another thread's stop already took
1150        // the joined copy (a shared queue), the kernel queues the requeue,
1151        // which the list cannot hold, so the run is refused (B1).
1152        if pending.signal == signal
1153            && pending.early.is_some()
1154            && how == (Consumption::KeptResume { blocked: true })
1155            && !(if pending.queue == QueueId::Thread(thread) {
1156                thread_holds
1157            } else {
1158                process_holds
1159            })
1160        {
1161            refuse(format_args!(
1162                "hermit: thread {thread}'s delivery of signal {signal} was resumed blocked after a send had taken the place of its front in {:?}, and another thread had already taken that send's copy (a delivery or an `rt_sigtimedwait`): the kernel queues the requeued copy, which the model of pending entries cannot hold, so the run is refused (https://github.com/rrnewton/hermit/issues/4262)",
1163                pending.queue
1164            ));
1165        }
1166        if pending.signal != signal || how == (Consumption::KeptResume { blocked: true }) {
1167            self.retire_unfinished_dequeue(pending);
1168            return None;
1169        }
1170        // A delivery that never asked for its siginfo (it ended the thread
1171        // group first) still re-arms its timer, at the committed time.
1172        if let Some(captured) = pending.awaiting_info.take() {
1173            let at = self.committed_time;
1174            pending.reconcile = Some((captured.entry.timer, at, captured_overrun(&captured, at)));
1175        }
1176        // After an early pop, a send's copy is the identified queue's new
1177        // front, physically there (`pop_in_flight_front`), unless another
1178        // thread's delivery stop has since dequeued it and waits for its own
1179        // turn (this pending dequeue is already removed, so any left for the
1180        // same queue and signal is another's).
1181        let identified_expected = pending.early.is_some()
1182            && self
1183                .discarded_at_send
1184                .lists
1185                .is_listed(pending.queue, signal)
1186            && !self.pending_timer_dequeues.values().any(|other| {
1187                // Another stop whose own front was popped early took a
1188                // copy before this send joined, not the joined copy.
1189                other.queue == pending.queue && other.signal == signal && other.early.is_none()
1190            });
1191        self.check_taken_queues(
1192            thread,
1193            pending.queue,
1194            signal,
1195            identified_expected,
1196            thread_holds,
1197            process_holds,
1198        );
1199        Some(pending)
1200    }

1202    /// The check of a dequeue of `signal` from `queue` by `thread` against
1203    /// its queues read after it, before any promotion (rev 9 A''): the
1204    /// identified queue no longer holds the signal, and the other one holds
1205    /// it exactly when its list or its discarded bit says so. A
1206    /// contradiction is a producer or a consumer the schedule does not see,
1207    /// and refuses by name.
1208    fn check_taken_queues(
1209        &self,
1210        thread: DetTid,
1211        queue: QueueId,
1212        signal: i32,
1213        identified_expected: bool,
1214        thread_holds: bool,
1215        process_holds: bool,
1216    ) {
1217        let process = self.sigchld_process(thread);
1218        let (private, shared) = (QueueId::Thread(thread), QueueId::Process(process));
1219        let (other, identified_holds, other_holds) = if queue == private {
1220            (shared, thread_holds, process_holds)
1221        } else {
1222            (private, process_holds, thread_holds)
1223        };
1224        let other_expected = self.discarded_at_send.lists.is_listed(other, signal)
1225            || self.discarded_at_send.is_discarded(other, signal);
1226        if identified_holds != identified_expected || other_holds != other_expected {
1227            refuse(format_args!(
1228                "hermit: thread {thread} took signal {signal} from {queue:?}, and afterwards that queue {} the signal and {other:?} {} it, where the model of its pending entries expects it {} in {queue:?} and {} in {other:?}: a {} the schedule does not see, so the run is refused (https://github.com/rrnewton/hermit/issues/4262)",
1229                if identified_holds {
1230                    "still held"
1231                } else {
1232                    "no longer held"
1233                },
1234                if other_holds { "held" } else { "did not hold" },
1235                if identified_expected {
1236                    "held"
1237                } else {
1238                    "absent"
1239                },
1240                if other_expected { "held" } else { "absent" },
1241                // A copy the model lacks came from a producer it did not see;
1242                // a copy it expected that is gone, from a consumer.
1243                if (identified_holds && !identified_expected) || (other_holds && !other_expected) {
1244                    "producer"
1245                } else {
1246                    "consumer"
1247                },
1248            ));
1249        }
1250    }

1252    /// A kept send of `signal` into `queue` found the kernel's queue without
1253    /// the signal (`present` false) while its list still has a front: the
1254    /// kernel dequeued that front at a delivery stop whose hooks have not run
1255    /// yet (a `PendingDequeue` for it). Linux queues this send, so the front
1256    /// is popped now, kept in the pending dequeue for the stop to finish, and
1257    /// the send joins the empty list (A1).
1258    ///
1259    /// Without this, a thread that `tgkill`s a signal again while another
1260    /// thread's delivery of it waits for its turn left the new copy out of
1261    /// the model, and the stop refused the run (cargo's jobserver helper
1262    /// shutdown; the regression of
1263    /// https://github.com/rrnewton/hermit/pull/4491). An entry behind the
1264    /// in-flight front means Linux, whose bit stays set, drops this send
1265    /// while the kernel kept it: no list can say that, so it is refused.
1266    ///
1267    /// The joining send's `standing` (`note_send_listed`) is read from the
1268    /// discarded bit, which still describes the in-flight copy until the
1269    /// stop finishes: after an early pop of a `Proxy` front standing on a
1270    /// discarded copy, the send joins as `Proxy` with that record's
1271    /// `SI_TIMER` flag. A later delivery rewrites it from its stored bytes,
1272    /// and an exec re-send merges into the kept copy, so the record the guest
1273    /// sees is the send's own (the Claude review of the in-flight-dequeue
1274    /// fix, F6).
1275    pub(crate) fn pop_in_flight_front(&mut self, queue: QueueId, signal: i32, present: bool) {
1276        if present || !self.discarded_at_send.lists.is_listed(queue, signal) {
1277            return;
1278        }
1279        let Some((&thread, _)) = self.pending_timer_dequeues.iter().find(|(_, pending)| {
1280            pending.queue == queue && pending.signal == signal && pending.early.is_none()
1281        }) else {
1282            return;
1283        };
1284        let Some(popped) = self.discarded_at_send.lists.pop(queue, signal) else {
1285            return;
1286        };
1287        if popped.next.is_some() {
1288            refuse(format_args!(
1289                "hermit: a send of signal {signal} found {queue:?} without it while thread {thread}'s delivery of its front waited for its turn, and the front had an entry behind it: Linux drops that send, but the kernel queued it, which the model of pending entries cannot hold, so the run is refused (https://github.com/rrnewton/hermit/issues/4262)"
1290            ));
1291        }
1292        debug!(
1293            "[dtid {}] signal {} in {:?}: the front a delivery took is popped early; a send joins the list.",
1294            thread, signal, queue
1295        );
1296        if let Some(pending) = self.pending_timer_dequeues.get_mut(&thread) {
1297            pending.early = Some(popped);
1298        }
1299    }

1301    /// Thread `thread` exited with a dequeue pending: a front popped early
1302    /// for it (`pop_in_flight_front`) is retired as its thread's entries are,
1303    /// so a periodic timer that waited for its dequeue re-arms.
1304    pub(crate) fn forget_pending_dequeue(&mut self, thread: DetTid) {
1305        if let Some(unfinished) = self.pending_timer_dequeues.remove(&thread) {
1306            self.retire_unfinished_dequeue(unfinished);
1307        }
1308    }

1310    /// A pending dequeue dropped without finishing (its thread exited, a new
1311    /// stop replaced it, its finish did not match it, or its delivery was
1312    /// resumed blocked): a front it holds
1313    /// after an early pop (`pop_in_flight_front`) no longer has an owner, so
1314    /// it is retired as a dead thread's entries are, and a periodic timer
1315    /// that waited for its dequeue re-arms instead of stopping (the Claude
1316    /// and Codex reviews of the in-flight-dequeue fix, F5 and P2). Before an
1317    /// early pop the front is still in its list, where a later dequeue or the
1318    /// queue's retirement reconciles it.
1319    fn retire_unfinished_dequeue(&mut self, unfinished: PendingDequeue) {
1320        if let Some(popped) = unfinished.early {
1321            self.note_private_entries_retired(vec![popped.entry]);
1322        }
1323    }

1325    /// Pop `pending`'s front, reconcile its timer, and promote the next entry
1326    /// with `receiver` the thread that took it.
1327    pub(crate) fn pop_and_promote(&mut self, pending: PendingDequeue, receiver: DetTid) {
1328        let Some(popped) = pending.early.or_else(|| {
1329            self.discarded_at_send
1330                .lists
1331                .pop(pending.queue, pending.signal)
1332        }) else {
1333            return;
1334        };
1335        if let (Entry::Timer(entry), Some((timer, at, overrun))) = (popped.entry, pending.reconcile)
1336        {
1337            debug_assert_eq!(entry.timer, timer);
1338            let id = SignalTimerId::Posix(entry.process, timer);
1339            if entry.periodic_at_send
1340                && let Some(next) = self.blocked.timed_waiters.signal_timer_rearm_after(id, at)
1341            {
1342                self.posix_timer_overruns
1343                    .insert((entry.process, timer), overrun);
1344                debug!(
1345                    "[dpid {}] POSIX timer {} delivered with overrun {}; re-armed at {}.",
1346                    entry.process, timer, overrun, next
1347                );
1348            }
1349        }
1350        debug_assert!(self.discarded_at_send.lists.invariants_hold());
1351        if let Some(Entry::Timer(next)) = popped.next {
1352            // At a delivery the kernel sets up the handler frame before the
1353            // receiver can take anything else: its mask then decides the
1354            // promotion's target (§4.5, P3-3).
1355            let process = self.sigchld_process(receiver);
1356            self.delivery_mask_override = match self.handler_masks.get(&(process, pending.signal)) {
1357                Some(&(mask, nodefer)) if pending.delivery => {
1358                    let own = self.thread_signal_mask(receiver).unwrap_or(0);
1359                    let deferred = if nodefer {
1360                        0
1361                    } else {
1362                        kernel_signal_bit(pending.signal)
1363                    };
1364                    Some((receiver, own | mask | deferred))
1365                }
1366                _ => None,
1367            };
1368            self.promote(pending.queue, pending.signal, next, receiver);
1369            self.delivery_mask_override = None;
1370        }
1371    }

1373    /// Send the new front `next` of `queue`'s list for `signal` (promotion,
1374    /// 4262 design §4.5): kept, as Linux queued it at its own expiry.
1375    fn promote(&mut self, queue: QueueId, signal: i32, next: TimerEntry, receiver: DetTid) {
1376        let Ok(sig) = Signal::try_from(signal) else {
1377            return;
1378        };
1379        info!(
1380            "[dtid {}] signal {} dequeued from {:?}; promoting POSIX timer {}'s entry.",
1381            receiver, sig, queue, next.timer
1382        );
1383        let info = self.timer_signal_info(signal, &next, 0);
1384        self.send_timer_record(queue, sig, receiver, info, SendFate::Kept);
1385    }

1387    /// Thread `thread` exited, taking its private lists with it
1388    /// (`entries`, N9-5): a periodic timer that waited for the dequeue of one
1389    /// of them re-arms now, so its next expiry finds the thread gone and is
1390    /// dropped, as Linux drops it, with the timer still armed.
1391    pub(crate) fn note_private_entries_retired(&mut self, entries: Vec<Entry>) {
1392        let now = self.committed_time;
1393        for entry in entries {
1394            // Only an entry of the timer's current arming re-arms it: a
1395            // `timer_settime` since the entry was queued started a newer
1396            // generation, which waits for its own entry's dequeue (the Codex
1397            // re-check of the in-flight-dequeue fix, P2-1).
1398            if let Entry::Timer(entry) = entry
1399                && entry.periodic_at_send
1400                && self.timer_entry_valid(&entry)
1401            {
1402                self.blocked.timed_waiters.signal_timer_rearm_after(
1403                    SignalTimerId::Posix(entry.process, entry.timer),
1404                    now,
1405                );
1406            }
1407        }
1408    }

1418    /// `timer_gettime`'s remaining time of `process`'s timer `timer` at `now`
1419    /// (4262 design §5), or `None` where the caller's own answer stands
1420    /// (record and replay, or a timer the scheduler does not know). An Active
1421    /// timer past its deadline, whose pop waits for a turn boundary, reads
1422    /// 1 ns (`common_timer_get` of an armed hrtimer); one
1423    /// waiting for the dequeue of its entry reads the time to its next grid
1424    /// point, a full interval on an exact hit (`timer_forward`); a popped
1425    /// one-shot or a disarmed timer reads 0.
1426    pub(crate) fn posix_timer_remaining(
1427        &self,
1428        process: DetPid,
1429        timer: i32,
1430        now: LogicalTime,
1431    ) -> Option<LogicalTime> {
1432        // A timer the scheduler never registered (a real-time one in a
1433        // permissive run) keeps the caller's own answer.
1434        if !self.coalesces_alarms() || !self.posix_timer_live(process, timer) {
1435            return None;
1436        }
1437        let id = SignalTimerId::Posix(process, timer);
1438        let remaining = match self.blocked.timed_waiters.signal_timer_state(id) {
1439            None => 0,
1440            Some((deadline, _, AlarmPhase::Active)) => {
1441                deadline.as_nanos().saturating_sub(now.as_nanos()).max(1)
1442            }
1443            Some((last, interval, AlarmPhase::Inactive { .. })) => {
1444                grid_after(last, interval, now).as_nanos() - now.as_nanos()
1445            }
1446        };
1447        Some(LogicalTime::from_nanos(remaining))
1448    }
