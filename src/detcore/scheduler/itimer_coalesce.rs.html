9//! Linux's coalescing of a periodic `ITIMER_REAL` on the traced path
10//! (<https://github.com/rrnewton/hermit/issues/4139>).
11//!
12//! Linux's `it_real_fn` sends SIGALRM to the thread-group leader and returns
13//! `HRTIMER_NORESTART`. Only a dequeue of a SIGALRM from the process's shared
14//! queue, whatever sent it, restarts the timer (`posixtimer_rearm_itimer`,
15//! `hrtimer_forward_now` on the original grid). So at most one timer SIGALRM is
16//! outstanding, missed periods are dropped, and `getitimer` reads 0 while the
17//! timer waits. Before this, the traced path re-armed every expiry at the old
18//! deadline plus one interval, so a slice that crossed N periods fired N
19//! SIGALRMs at its end, one per scheduler pass.
20//!
21//! - **Observable processes.** The scheduler re-arms at a dequeue only where it
22//!   can see the dequeue at a committed point: no thread of the process has run
23//!   outside the schedule (`background_call_processes`), no host-timed source
24//!   can post SIGALRM to it (`host_timed_signals`, the unreaped children's exit
25//!   signals), and its leader is live. Those conditions are sticky once false
26//!   (`alarm_unobservable`). Every other process forwards each expiry to the
27//!   first grid point after the pop: missed periods are still dropped, but the
28//!   timer does not wait for the dequeue.
29//! - **The pop.** An observable process's expiry that its leader ignores at
30//!   generation (`sig_ignored`: SIG_IGN, or SIG_DFL in a namespace init, and
31//!   not blocked by the leader) sends nothing and leaves the timer `Inactive`
32//!   with no copy, as Linux discards it. Otherwise the scheduler sends the
33//!   signal and the timer becomes `Inactive` with a copy.
34//! - **Reconcile (O3).** At each quiesced boundary, before the next pass pops
35//!   or grants, an `Inactive` timer with a copy whose process's shared queue no
36//!   longer holds SIGALRM re-arms at the first grid point after the committed
37//!   time. Between two quiesced boundaries only the granted thread of an
38//!   observable process runs, so that read is a function of the schedule.
39//!
40//! Record and replay keep the legacy recurrence and return values
41//! (`recordreplay_modes`), so recordings keep their event stream.
42//!
43//! A run with an installed process signal control keeps its `ITIMER_REAL` in
44//! `real_timer.rs` instead: the backend publishes the copy and acknowledges
45//! each real dequeue. Its timer restarts at these same reconcile points (O1,
46//! O2, O3) from the acknowledged dequeue rather than from a read of the
47//! shared queue (<https://github.com/rrnewton/hermit/issues/4929>).

64/// Which processes a guest's committed SIGALRM send could have queued a
65/// shared-queue copy for (rule 4).
66#[derive(Clone, Copy, Debug, Eq, PartialEq, Serialize, Deserialize)]
67pub enum GuestSigalrmTarget {
68    /// `kill(pid > 0)`, `rt_sigqueueinfo(tgid)`.
69    Process(DetPid),
70    /// `kill(0)` (the sender's group, `None`) or `kill(-pgrp)`.
71    Group(Option<DetPid>),
72    /// `kill(-1)`.
73    All,
74    /// A send whose targets Detcore cannot classify (a nested PID namespace's
75    /// pid, a pidfd): every process it might reach stops being observable.
76    Unclassified,
77}

79/// What a read of a process's shared queue says about its ITIMER_REAL copy.
80#[derive(Clone, Copy, Debug, Eq, PartialEq)]
81enum SharedCopy {
82    Dequeued,
83    Pending,
84    ProcessGone,
85    Unreadable,
86}

93impl Scheduler {
94    /// Whether this scheduler coalesces periodic `ITIMER_REAL` expiries at all:
95    /// only on the traced path, which models signal targets, outside record and
96    /// replay, and not where a backend owns process signal control (that one
97    /// keeps its own timer, `real_timer.rs`, which restarts at the same
98    /// reconcile points: `reconcile_alarm`, `reconcile_inactive_alarms`).
99    pub(super) fn coalesces_alarms(&self) -> bool {
100        self.models_signal_targets && !self.recordreplay_modes && !self.signal_control_installed()
101    }

103    /// Whether the current process state permits identifying a shared-queue
104    /// SIGALRM dequeue. A metadata query must not commit a recurrence-policy
105    /// change during a transient state, such as an exited leader before exec.
106    pub(super) fn alarm_observable_now(&self, process: DetPid) -> bool {
107        if !self.coalesces_alarms() {
108            return false;
109        }
110        if self.alarm_unobservable.contains(&process) {
111            return false;
112        }
113        let bit = sigalrm_bit();
114        let threads = self.process_signal_targets(process);
115        let leader = DetTid::from_raw(process.as_raw());
116        let host_timed = threads
117            .iter()
118            .any(|thread| self.host_timed_signals(*thread) & bit != 0);
119        let foreign_exit = self
120            .thread_tree
121            .unreaped_children_of(process)
122            .into_iter()
123            .filter_map(|child| self.thread_tree.notification_signal(child))
124            .any(|signal| signal == libc::SIGALRM);
125        !self.background_call_processes.contains(&process)
126            && !host_timed
127            && !foreign_exit
128            && threads.contains(&leader)
129    }

131    /// The recurrence owner commits loss of observability at its existing
132    /// timer boundaries. Once committed, it remains sticky.
133    pub(super) fn alarm_observable(&mut self, process: DetPid) -> bool {
134        if !self.coalesces_alarms() || self.alarm_unobservable.contains(&process) {
135            return false;
136        }
137        let observable = self.alarm_observable_now(process);
138        if !observable {
139            debug!(
140                "[dpid {}] ITIMER_REAL dequeues are not observable from here on; expiries forward at the pop.",
141                process
142            );
143            self.alarm_unobservable.insert(process);
144        }
145        observable
146    }

148    /// Whether `process`'s leader ignores SIGALRM at generation, as Linux's
149    /// `sig_ignored(leader)` decides: the disposition is SIG_IGN, or SIG_DFL
150    /// when the process is a guest namespace init (`sig_task_ignored` discards
151    /// that for a `SIGNAL_UNKILLABLE` task), and the leader's committed mask
152    /// does not block it. `None` when the leader's state cannot be read.
153    fn leader_ignores_sigalrm(&self, process: DetPid) -> Option<bool> {
154        let leader = DetTid::from_raw(process.as_raw());
155        let state = self.read_sleeper_signal_state(leader).ok()?;
156        let bit = sigalrm_bit();
157        let mask = self
158            .saved_guest_sigmasks
159            .get(&leader)
160            .copied()
161            .unwrap_or(state.blocked);
162        let sig_ign = state.ignored & bit != 0;
163        let sig_dfl_init = self.thread_tree.is_pid_namespace_init(process)
164            && state.ignored & bit == 0
165            && state.caught & bit == 0;
166        Some((sig_ign || sig_dfl_init) && mask & bit == 0)
167    }

169    /// Fire the expiry at `deadline` of `process`'s `ITIMER_REAL`, which
170    /// `pop_if_before` has just popped (and, if periodic, re-armed one interval
171    /// later), then replace that legacy recurrence.
172    pub(super) fn fire_itimer_real(
173        &mut self,
174        deadline: LogicalTime,
175        process: DetPid,
176        tid: DetTid,
177        signal: Signal,
178    ) {
179        let periodic = self
180            .blocked
181            .timed_waiters
182            .signal_timer_period(SignalTimerId::Alarm(process))
183            .is_some();
184        if !periodic || !self.coalesces_alarms() {
185            self.fire_itimer_real_copy(process, tid, signal);
186            return;
187        }
188        let now = self.committed_time;
189        if self.exec_teardowns.contains_key(&process) {
190            // The leader's mask cannot be read from `/proc` during `de_thread`:
191            // hold the expiry and decide it when the exec barrier ends
192            // (`finish_exec_alarm`), keeping the original grid.
193            self.blocked
194                .timed_waiters
195                .alarm_after_fire(process, deadline, Some(false), now);
196            self.alarm_exec_deferred.insert(process, (tid, signal));
197            return;
198        }
199        if !self.alarm_observable(process) {
200            self.fire_itimer_real_copy(process, tid, signal);
201            self.blocked
202                .timed_waiters
203                .alarm_after_fire(process, deadline, None, now);
204            return;
205        }
206        match self.leader_ignores_sigalrm(process) {
207            Some(true) => {
208                info!(
209                    "[dpid {}] SIGALRM expired while its leader ignores it; discarded at generation, the timer waits for a dequeue.",
210                    process
211                );
212                self.blocked
213                    .timed_waiters
214                    .alarm_after_fire(process, deadline, Some(false), now);
215            }
216            Some(false) => {
217                self.guest_send_landed = None;
218                let sent = self.fire_itimer_real_copy(process, tid, signal);
219                let landed = sent && self.guest_send_landed != Some(false);
220                let copy = landed.then_some(true);
221                self.blocked
222                    .timed_waiters
223                    .alarm_after_fire(process, deadline, copy, now);
224            }
225            None => {
226                self.alarm_unobservable.insert(process);
227                self.fire_itimer_real_copy(process, tid, signal);
228                self.blocked
229                    .timed_waiters
230                    .alarm_after_fire(process, deadline, None, now);
231            }
232        }
233    }

235    /// At the end of `process`'s exec barrier, decide an expiry that popped
236    /// during it, then run the reconcile the barrier deferred. Both use the
237    /// committed time at this point.
238    pub(super) fn finish_exec_alarm(&mut self, process: DetPid) {
239        let t = self.committed_time;
240        if let Some((tid, signal)) = self.alarm_exec_deferred.remove(&process) {
241            let observable = self.alarm_observable(process);
242            let ignored = if observable {
243                self.leader_ignores_sigalrm(process)
244            } else {
245                None
246            };
247            match ignored {
248                Some(true) => {}
249                Some(false) => {
250                    self.guest_send_landed = None;
251                    let sent = self.fire_itimer_real_copy(process, tid, signal);
252                    if sent && self.guest_send_landed != Some(false) {
253                        self.blocked.timed_waiters.alarm_set_copy(process, true);
254                    } else {
255                        self.blocked.timed_waiters.alarm_rearm_after(process, t);
256                    }
257                }
258                None => {
259                    if observable {
260                        self.alarm_unobservable.insert(process);
261                    }
262                    self.fire_itimer_real_copy(process, tid, signal);
263                    self.blocked.timed_waiters.alarm_rearm_after(process, t);
264                }
265            }
266        }
267        self.reconcile_alarm(process, t);
268    }

270    /// `process`'s shared pending set, read through the thread group
271    /// (`/proc/<tgid>/status`, valid while any thread lives).
272    fn read_process_shared_pending(&self, process: DetPid) -> Option<u64> {
273        #[cfg(test)]
274        {
275            let leader = DetTid::from_raw(process.as_raw());
276            self.read_shared_pending(leader)
277        }
278        #[cfg(not(test))]
279        {
280            let status =
281                std::fs::read_to_string(format!("/proc/{}/status", process.as_raw())).ok()?;
282            crate::syscalls::KernelSignalState::parse(&status).map(|state| state.shared_pending)
283        }
284    }

286    /// Reconcile `process`'s `Inactive` `ITIMER_REAL` at the committed time
287    /// `t`: re-arm it when its copy has left the shared queue, or when the
288    /// process stopped being observable (rule 6). Returns the new deadline if
289    /// it re-armed.
290    ///
291    /// With an installed signal control the process's timer is the
292    /// controlled one, which restarts here when its backend has acknowledged
293    /// the copy's dequeue (`reconcile_controlled_real_timer`).
294    pub(crate) fn reconcile_alarm(
295        &mut self,
296        process: DetPid,
297        t: LogicalTime,
298    ) -> Option<LogicalTime> {
299        if self.signal_control_installed() {
300            return self.reconcile_controlled_real_timer(process, t);
301        }
302        let AlarmPhase::Inactive { copy } = self.blocked.timed_waiters.alarm_phase(process)? else {
303            return None;
304        };
305        if self.exec_teardowns.contains_key(&process) {
306            // `/proc/<tgid>` can be briefly unresolvable during `de_thread`;
307            // the barrier's end reconciles.
308            return None;
309        }
310        if !self.alarm_observable(process) {
311            let next = self.blocked.timed_waiters.alarm_rearm_after(process, t);
312            debug!(
313                "[dpid {}] ITIMER_REAL re-armed at {:?}: its process is no longer observable.",
314                process, next
315            );
316            return next;
317        }
318        // Invariant: the queue is read only when a copy is known to be there.
319        // Under ptrace a guest kill() to a leader that ignores SIGALRM leaves a
320        // physical copy native Linux drops; this read must not count it.
321        if !copy {
322            return None;
323        }
324        match self.shared_copy_outcome(process) {
325            SharedCopy::Dequeued => {
326                let next = self.blocked.timed_waiters.alarm_rearm_after(process, t);
327                debug!(
328                    "[dpid {}] SIGALRM left the shared queue; ITIMER_REAL re-armed at {:?}.",
329                    process, next
330                );
331                next
332            }
333            SharedCopy::Pending => None,
334            SharedCopy::ProcessGone => {
335                self.blocked.timed_waiters.remove_alarm(process);
336                None
337            }
338            SharedCopy::Unreadable => exit_on_scheduler_refusal(&format_args!(
339                "HERMIT_SIGALRM_QUEUE_UNREADABLE: the shared signal queue of process {} could not be read while it has a live thread, so whether its ITIMER_REAL copy was dequeued is unknown (https://github.com/rrnewton/hermit/issues/4139)",
340                process
341            )),
342        }
343    }

345    /// What `process`'s shared queue says about its ITIMER_REAL copy. A read
346    /// that fails while the process has a committed-live thread is
347    /// `Unreadable`, which refuses the run; one that fails after its exit
348    /// committed is `ProcessGone`.
349    fn shared_copy_outcome(&self, process: DetPid) -> SharedCopy {
350        match self.read_process_shared_pending(process) {
351            Some(pending) if pending & sigalrm_bit() == 0 => SharedCopy::Dequeued,
352            Some(_) => SharedCopy::Pending,
353            None if self.process_signal_targets(process).is_empty() => SharedCopy::ProcessGone,
354            None => SharedCopy::Unreadable,
355        }
356    }

358    /// Rule 4: a guest's committed SIGALRM send from `sender`. An `Inactive`
359    /// timer with no known copy (its last expiry was discarded at
360    /// generation) now has one, unless the leader still ignores SIGALRM at
361    /// generation, so the dequeue of that copy restarts it, as Linux restarts
362    /// a timer on any shared-queue SIGALRM dequeue.
363    pub(crate) fn note_guest_sigalrm(&mut self, sender: DetTid, target: GuestSigalrmTarget) {
364        if !self.coalesces_alarms() {
365            return;
366        }
367        let inactive = self.blocked.timed_waiters.inactive_alarm_processes();
368        let targets: Vec<DetPid> = match target {
369            GuestSigalrmTarget::Process(process) => vec![process],
370            GuestSigalrmTarget::Group(group) => {
371                let group =
372                    group.or_else(|| self.thread_tree.process_group(self.sigchld_process(sender)));
373                match group {
374                    Some(group) => inactive
375                        .iter()
376                        .copied()
377                        .filter(|process| self.thread_tree.process_group(*process) == Some(group))
378                        .collect(),
379                    None => {
380                        self.alarm_unobservable.extend(inactive.iter().copied());
381                        return;
382                    }
383                }
384            }
385            GuestSigalrmTarget::All => inactive.clone(),
386            GuestSigalrmTarget::Unclassified => {
387                self.alarm_unobservable.extend(inactive.iter().copied());
388                return;
389            }
390        };
391        for process in targets {
392            if self.blocked.timed_waiters.alarm_phase(process)
393                != Some(AlarmPhase::Inactive { copy: false })
394            {
395                continue;
396            }
397            match self.leader_ignores_sigalrm(process) {
398                Some(false) => self.blocked.timed_waiters.alarm_set_copy(process, true),
399                Some(true) => {}
400                None => {
401                    self.alarm_unobservable.insert(process);
402                }
403            }
404        }
405    }

407    /// Rule 5: `caller` changed SIGALRM's disposition. Setting SIG_IGN flushes
408    /// a pending copy from every queue without dequeuing it, so an `Inactive`
409    /// timer's copy is gone and the timer stays inactive (`sig_handler_ignored`;
410    /// SIG_DFL never flushes SIGALRM). Decided from the disposition the kernel
411    /// installed, not the call's return value.
412    pub(crate) fn note_sigalrm_disposition(&mut self, caller: DetTid) {
413        if !self.coalesces_alarms() {
414            return;
415        }
416        let process = self.sigchld_process(caller);
417        if self.blocked.timed_waiters.alarm_phase(process)
418            != Some(AlarmPhase::Inactive { copy: true })
419        {
420            return;
421        }
422        match self.read_sleeper_signal_state(caller) {
423            Ok(state) if state.ignored & sigalrm_bit() != 0 => {
424                self.blocked.timed_waiters.alarm_set_copy(process, false);
425            }
426            Ok(_) => {}
427            Err(_) => {
428                self.alarm_unobservable.insert(process);
429            }
430        }
431    }

433    /// O3: at a quiesced boundary, reconcile every `Inactive` `ITIMER_REAL`,
434    /// or, with an installed signal control, restart every controlled one
435    /// whose copy was dequeued (`reconcile_dequeued_real_timers`).
436    pub(super) fn reconcile_inactive_alarms(&mut self, t: LogicalTime) {
437        if self.signal_control_installed() {
438            self.reconcile_dequeued_real_timers(t);
439            return;
440        }
441        if !self.coalesces_alarms() {
442            return;
443        }
444        for process in self.blocked.timed_waiters.inactive_alarm_processes() {
445            self.reconcile_alarm(process, t);
446        }
447    }

517    /// Pop and dispatch the next timed event at committed time `now`.
518    fn pop_at(scheduler: &mut Scheduler, now: u64) {
519        scheduler.committed_time = at(now);
520        let (time, evt) = scheduler
521            .blocked
522            .timed_waiters
523            .pop_if_before(at(now))
524            .expect("a due expiry");
525        let super::super::timed_waiters::TimedEvent::SignalEvt(id, target, sig) = evt else {
526            panic!("expected a signal timer, got {evt}");
527        };
528        scheduler.dispatch_timed_signal(time, id, target, sig, true, at(now));
529    }

535    /// A pop sends the signal and leaves the timer Inactive with a copy, with
536    /// no deadline queued however many periods the slice crossed; the copy's
537    /// departure from the shared queue re-arms it on the original grid at the
538    /// next quiesced boundary. While the copy is still pending nothing moves.
539    #[test]
540    fn a_pop_waits_for_the_dequeue_and_re_arms_on_the_grid() {
541        let (mut scheduler, process, leader) = one_process();
542        // The slice crossed 5.5 periods before this pop.
543        pop_at(&mut scheduler, 5_500_000);
544        assert_eq!(
545            phase(&scheduler, process),
546            Some(AlarmPhase::Inactive { copy: true })
547        );
548        assert!(
549            scheduler.blocked.timed_waiters.is_empty(),
550            "no backlog of expiries"
551        );
552        // Still pending: no change.
553        set_state(&mut scheduler, leader, sigalrm_bit(), 0, 0);
554        assert_eq!(scheduler.reconcile_alarm(process, at(5_800_000)), None);
555        // Dequeued: re-armed at the first grid point after t.
556        set_state(&mut scheduler, leader, 0, 0, 0);
557        assert_eq!(
558            scheduler.reconcile_alarm(process, at(7_200_000)),
559            Some(at(8_000_000))
560        );
561        assert_eq!(phase(&scheduler, process), Some(AlarmPhase::Active));
562        assert_eq!(
563            scheduler.blocked.timed_waiters.next_deadline(),
564            Some(at(8_000_000))
565        );
566    }
