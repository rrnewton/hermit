9//! Linux's coalescing of a periodic `ITIMER_REAL` on the traced path
10//! (<https://github.com/rrnewton/hermit/issues/4139>).
11//!
12//! Linux's `it_real_fn` sends SIGALRM to the thread-group leader and returns
13//! `HRTIMER_NORESTART`. Only a dequeue of a SIGALRM from the process's shared
14//! queue, whatever sent it, restarts the timer (`posixtimer_rearm_itimer`,
15//! `hrtimer_forward_now` on the original grid). So at most one timer SIGALRM is
16//! outstanding, missed periods are dropped, and `getitimer` reads 0 while the
17//! timer waits. Before this, the traced path re-armed every expiry at the old
18//! deadline plus one interval, so a slice that crossed N periods fired N
19//! SIGALRMs at its end, one per scheduler pass.
20//!
21//! - **Observable processes.** The scheduler re-arms at a dequeue only where it
22//!   can see the dequeue at a committed point: no thread of the process has run
23//!   outside the schedule (`background_call_processes`), no host-timed source
24//!   can post SIGALRM to it (`host_timed_signals`, the unreaped children's exit
25//!   signals), and its leader is live. Those conditions are sticky once false
26//!   (`alarm_unobservable`). Every other process forwards each expiry to the
27//!   first grid point after the pop: missed periods are still dropped, but the
28//!   timer does not wait for the dequeue.
29//! - **The pop.** An observable process's expiry that its leader ignores at
30//!   generation (`sig_ignored`: SIG_IGN, or SIG_DFL in a namespace init, and
31//!   not blocked by the leader) sends nothing and leaves the timer `Inactive`
32//!   with no copy, as Linux discards it. Otherwise the scheduler sends the
33//!   signal and the timer becomes `Inactive` with a copy.
34//! - **Reconcile (O3).** At each quiesced boundary, before the next pass pops
35//!   or grants, an `Inactive` timer with a copy whose process's shared queue no
36//!   longer holds SIGALRM re-arms at the first grid point after the committed
37//!   time. Between two quiesced boundaries only the granted thread of an
38//!   observable process runs, so that read is a function of the schedule.
39//!
40//! Record and replay keep the legacy recurrence and return values
41//! (`recordreplay_modes`), so recordings keep their event stream.
42//!
43//! A run with an installed process signal control keeps its `ITIMER_REAL` in
44//! `real_timer.rs` instead: the backend publishes the copy and acknowledges
45//! each real dequeue. Its timer restarts at these same reconcile points (O1,
46//! O2, O3) from the acknowledged dequeue rather than from a read of the
47//! shared queue (<https://github.com/rrnewton/hermit/issues/4929>).

64/// Which processes a guest's committed SIGALRM send could have queued a
65/// shared-queue copy for (rule 4).
66#[derive(Clone, Copy, Debug, Eq, PartialEq, Serialize, Deserialize)]
67pub enum GuestSigalrmTarget {
68    /// `kill(pid > 0)`, `rt_sigqueueinfo(tgid)`.
69    Process(DetPid),
70    /// `kill(0)` (the sender's group, `None`) or `kill(-pgrp)`.
71    Group(Option<DetPid>),
72    /// `kill(-1)`.
73    All,
74    /// A send whose targets Detcore cannot classify (a nested PID namespace's
75    /// pid, a pidfd): every process it might reach stops being observable.
76    Unclassified,
77}

79/// What a read of a process's shared queue says about its ITIMER_REAL copy.
80#[derive(Clone, Copy, Debug, Eq, PartialEq)]
81enum SharedCopy {
82    Dequeued,
83    Pending,
84    ProcessGone,
85    Unreadable,
86}

93impl Scheduler {
94    /// Whether this scheduler coalesces periodic `ITIMER_REAL` expiries at all:
95    /// only on the traced path, which models signal targets, outside record and
96    /// replay, and not where a backend owns process signal control (that one
97    /// keeps its own timer, `real_timer.rs`, which restarts at the same
98    /// reconcile points: `reconcile_alarm`, `reconcile_inactive_alarms`).
99    pub(super) fn coalesces_alarms(&self) -> bool {
100        self.models_signal_targets && !self.recordreplay_modes && !self.signal_control_installed()
101    }

103    /// Whether the current process state permits identifying a shared-queue
104    /// SIGALRM dequeue. A metadata query must not commit a recurrence-policy
105    /// change during a transient state, such as an exited leader before exec.
106    pub(super) fn alarm_observable_now(&self, process: DetPid) -> bool {
107        if !self.coalesces_alarms() {
108            return false;
109        }
110        if self.alarm_unobservable.contains(&process) {
111            return false;
112        }
113        let bit = sigalrm_bit();
114        let threads = self.process_signal_targets(process);
115        let leader = DetTid::from_raw(process.as_raw());
116        let host_timed = threads
117            .iter()
118            .any(|thread| self.host_timed_signals(*thread) & bit != 0);
119        let foreign_exit = self
120            .thread_tree
121            .unreaped_children_of(process)
122            .into_iter()
123            .filter_map(|child| self.thread_tree.notification_signal(child))
124            .any(|signal| signal == libc::SIGALRM);
125        !self.background_call_processes.contains(&process)
126            && !host_timed
127            && !foreign_exit
128            && threads.contains(&leader)
129    }

131    /// The recurrence owner commits loss of observability at its existing
132    /// timer boundaries. Once committed, it remains sticky.
133    pub(super) fn alarm_observable(&mut self, process: DetPid) -> bool {
134        if !self.coalesces_alarms() || self.alarm_unobservable.contains(&process) {
135            return false;
136        }
137        let observable = self.alarm_observable_now(process);
138        if !observable {
139            debug!(
140                "[dpid {}] ITIMER_REAL dequeues are not observable from here on; expiries forward at the pop.",
141                process
142            );
143            self.alarm_unobservable.insert(process);
144        }
145        observable
146    }

148    /// Whether `process`'s leader ignores SIGALRM at generation, as Linux's
149    /// `sig_ignored(leader)` decides: the disposition is SIG_IGN, or SIG_DFL
150    /// when the process is a guest namespace init (`sig_task_ignored` discards
151    /// that for a `SIGNAL_UNKILLABLE` task), and the leader's committed mask
152    /// does not block it. `None` when the leader's state cannot be read.
153    ///
154    /// The rule is the one every discard at generation is decided by
155    /// (`discarded_at_send::discards_at_generation`,
156    /// <https://github.com/rrnewton/hermit/issues/4929>); the inputs are the
157    /// ones this traced model has always read: the leader's state through
158    /// `read_sleeper_signal_state`, and its mask as Detcore saved it while a
159    /// private mask is installed.
160    fn leader_ignores_sigalrm(&self, process: DetPid) -> Option<bool> {
161        let leader = DetTid::from_raw(process.as_raw());
162        let state = self.read_sleeper_signal_state(leader).ok()?;
163        let mask = self
164            .saved_guest_sigmasks
165            .get(&leader)
166            .copied()
167            .unwrap_or(state.blocked);
168        Some(super::discarded_at_send::discards_at_generation(
169            libc::SIGALRM,
170            mask,
171            state.ignored,
172            state.caught,
173            self.thread_tree.is_pid_namespace_init(process),
174            false,
175        ))
176    }

178    /// Fire the expiry at `deadline` of `process`'s `ITIMER_REAL`, which
179    /// `pop_if_before` has just popped (and, if periodic, re-armed one interval
180    /// later), then replace that legacy recurrence.
181    pub(super) fn fire_itimer_real(
182        &mut self,
183        deadline: LogicalTime,
184        process: DetPid,
185        tid: DetTid,
186        signal: Signal,
187    ) {
188        let periodic = self
189            .blocked
190            .timed_waiters
191            .signal_timer_period(SignalTimerId::Alarm(process))
192            .is_some();
193        if !periodic || !self.coalesces_alarms() {
194            self.fire_itimer_real_copy(process, tid, signal);
195            return;
196        }
197        let now = self.committed_time;
198        if self.exec_teardowns.contains_key(&process) {
199            // The leader's mask cannot be read from `/proc` during `de_thread`:
200            // hold the expiry and decide it when the exec barrier ends
201            // (`finish_exec_alarm`), keeping the original grid.
202            self.blocked
203                .timed_waiters
204                .alarm_after_fire(process, deadline, Some(false), now);
205            self.alarm_exec_deferred.insert(process, (tid, signal));
206            return;
207        }
208        if !self.alarm_observable(process) {
209            self.fire_itimer_real_copy(process, tid, signal);
210            self.blocked
211                .timed_waiters
212                .alarm_after_fire(process, deadline, None, now);
213            return;
214        }
215        match self.leader_ignores_sigalrm(process) {
216            Some(true) => {
217                info!(
218                    "[dpid {}] SIGALRM expired while its leader ignores it; discarded at generation, the timer waits for a dequeue.",
219                    process
220                );
221                self.blocked
222                    .timed_waiters
223                    .alarm_after_fire(process, deadline, Some(false), now);
224            }
225            Some(false) => {
226                self.guest_send_landed = None;
227                let sent = self.fire_itimer_real_copy(process, tid, signal);
228                let landed = sent && self.guest_send_landed != Some(false);
229                let copy = landed.then_some(true);
230                self.blocked
231                    .timed_waiters
232                    .alarm_after_fire(process, deadline, copy, now);
233            }
234            None => {
235                self.alarm_unobservable.insert(process);
236                self.fire_itimer_real_copy(process, tid, signal);
237                self.blocked
238                    .timed_waiters
239                    .alarm_after_fire(process, deadline, None, now);
240            }
241        }
242    }

244    /// At the end of `process`'s exec barrier, decide an expiry that popped
245    /// during it, then run the reconcile the barrier deferred. Both use the
246    /// committed time at this point.
247    pub(super) fn finish_exec_alarm(&mut self, process: DetPid) {
248        let t = self.committed_time;
249        if let Some((tid, signal)) = self.alarm_exec_deferred.remove(&process) {
250            let observable = self.alarm_observable(process);
251            let ignored = if observable {
252                self.leader_ignores_sigalrm(process)
253            } else {
254                None
255            };
256            match ignored {
257                Some(true) => {}
258                Some(false) => {
259                    self.guest_send_landed = None;
260                    let sent = self.fire_itimer_real_copy(process, tid, signal);
261                    if sent && self.guest_send_landed != Some(false) {
262                        self.blocked.timed_waiters.alarm_set_copy(process, true);
263                    } else {
264                        self.blocked.timed_waiters.alarm_rearm_after(process, t);
265                    }
266                }
267                None => {
268                    if observable {
269                        self.alarm_unobservable.insert(process);
270                    }
271                    self.fire_itimer_real_copy(process, tid, signal);
272                    self.blocked.timed_waiters.alarm_rearm_after(process, t);
273                }
274            }
275        }
276        self.reconcile_alarm(process, t);
277    }

279    /// `process`'s shared pending set, read through the thread group
280    /// (`/proc/<tgid>/status`, valid while any thread lives). Where Detcore
281    /// reads signal state from the backend, any live thread of the process
282    /// reports the process's shared queue (`Scheduler::backend_signal_state`).
283    pub(super) fn read_process_shared_pending(&self, process: DetPid) -> Option<u64> {
284        if crate::syscalls::reads_signal_state_from_backend(&self.backend) {
285            return self
286                .process_signal_targets(process)
287                .into_iter()
288                .find_map(|thread| self.backend_signal_state(thread)?.ok())
289                .map(|state| state.shared_pending);
290        }
291        #[cfg(test)]
292        {
293            let leader = DetTid::from_raw(process.as_raw());
294            self.read_shared_pending(leader)
295        }
296        #[cfg(not(test))]
297        {
298            Self::shared_pending_from_status_file(process)
299        }
300    }

302    /// `process`'s shared pending set from its leader's status file
303    /// (`read_process_shared_pending`'s production read, factored out
304    /// so a test runs this exact read). The file is read as bytes: its
305    /// `Name:` line is the leader's `comm`, which need not be UTF-8,
306    /// and only the ASCII signal lines are parsed from it.
307    fn shared_pending_from_status_file(process: DetPid) -> Option<u64> {
308        let status = std::fs::read(format!("/proc/{}/status", process.as_raw())).ok()?;
309        crate::syscalls::KernelSignalState::parse_bytes(&status).map(|state| state.shared_pending)
310    }

312    /// Reconcile `process`'s `Inactive` `ITIMER_REAL` at the committed
313    /// time `t`: re-arm it when its copy has left the shared queue, or
314    /// when the process stopped being observable (rule 6). Returns the
315    /// new deadline if it re-armed.
316    ///
317    /// With an installed signal control the process's timer is the
318    /// controlled one, which restarts here when its backend has acknowledged
319    /// the copy's dequeue (`reconcile_controlled_real_timer`).
320    pub(crate) fn reconcile_alarm(
321        &mut self,
322        process: DetPid,
323        t: LogicalTime,
324    ) -> Option<LogicalTime> {
325        if self.signal_control_installed() {
326            return self.reconcile_controlled_real_timer(process, t);
327        }
328        let AlarmPhase::Inactive { copy } = self.blocked.timed_waiters.alarm_phase(process)? else {
329            return None;
330        };
331        if self.exec_teardowns.contains_key(&process) {
332            // `/proc/<tgid>` can be briefly unresolvable during `de_thread`;
333            // the barrier's end reconciles.
334            return None;
335        }
336        if !self.alarm_observable(process) {
337            let next = self.blocked.timed_waiters.alarm_rearm_after(process, t);
338            debug!(
339                "[dpid {}] ITIMER_REAL re-armed at {:?}: its process is no longer observable.",
340                process, next
341            );
342            return next;
343        }
344        // Invariant: the queue is read only when a copy is known to be there.
345        // Under ptrace a guest kill() to a leader that ignores SIGALRM leaves a
346        // physical copy native Linux drops; this read must not count it.
347        if !copy {
348            return None;
349        }
350        match self.shared_copy_outcome(process) {
351            SharedCopy::Dequeued => {
352                let next = self.blocked.timed_waiters.alarm_rearm_after(process, t);
353                debug!(
354                    "[dpid {}] SIGALRM left the shared queue; ITIMER_REAL re-armed at {:?}.",
355                    process, next
356                );
357                next
358            }
359            SharedCopy::Pending => None,
360            SharedCopy::ProcessGone => {
361                self.blocked.timed_waiters.remove_alarm(process);
362                None
363            }
364            SharedCopy::Unreadable => exit_on_scheduler_refusal(&format_args!(
365                "HERMIT_SIGALRM_QUEUE_UNREADABLE: the shared signal queue of process {} could not be read while it has a live thread, so whether its ITIMER_REAL copy was dequeued is unknown (https://github.com/rrnewton/hermit/issues/4139)",
366                process
367            )),
368        }
369    }

371    /// What `process`'s shared queue says about its ITIMER_REAL copy. A read
372    /// that fails while the process has a committed-live thread is
373    /// `Unreadable`, which refuses the run; one that fails after its exit
374    /// committed is `ProcessGone`.
375    fn shared_copy_outcome(&self, process: DetPid) -> SharedCopy {
376        match self.read_process_shared_pending(process) {
377            Some(pending) if pending & sigalrm_bit() == 0 => SharedCopy::Dequeued,
378            Some(_) => SharedCopy::Pending,
379            None if self.process_signal_targets(process).is_empty() => SharedCopy::ProcessGone,
380            None => SharedCopy::Unreadable,
381        }
382    }

384    /// Rule 4: a guest's committed SIGALRM send from `sender`. An `Inactive`
385    /// timer with no known copy (its last expiry was discarded at
386    /// generation) now has one, unless the leader still ignores SIGALRM at
387    /// generation, so the dequeue of that copy restarts it, as Linux restarts
388    /// a timer on any shared-queue SIGALRM dequeue.
389    pub(crate) fn note_guest_sigalrm(&mut self, sender: DetTid, target: GuestSigalrmTarget) {
390        if !self.coalesces_alarms() {
391            return;
392        }
393        let inactive = self.blocked.timed_waiters.inactive_alarm_processes();
394        let targets: Vec<DetPid> = match target {
395            GuestSigalrmTarget::Process(process) => vec![process],
396            GuestSigalrmTarget::Group(group) => {
397                let group =
398                    group.or_else(|| self.thread_tree.process_group(self.sigchld_process(sender)));
399                match group {
400                    Some(group) => inactive
401                        .iter()
402                        .copied()
403                        .filter(|process| self.thread_tree.process_group(*process) == Some(group))
404                        .collect(),
405                    None => {
406                        self.alarm_unobservable.extend(inactive.iter().copied());
407                        return;
408                    }
409                }
410            }
411            GuestSigalrmTarget::All => inactive.clone(),
412            GuestSigalrmTarget::Unclassified => {
413                self.alarm_unobservable.extend(inactive.iter().copied());
414                return;
415            }
416        };
417        for process in targets {
418            if self.blocked.timed_waiters.alarm_phase(process)
419                != Some(AlarmPhase::Inactive { copy: false })
420            {
421                continue;
422            }
423            match self.leader_ignores_sigalrm(process) {
424                Some(false) => self.blocked.timed_waiters.alarm_set_copy(process, true),
425                Some(true) => {}
426                None => {
427                    self.alarm_unobservable.insert(process);
428                }
429            }
430        }
431    }

433    /// Rule 5: `caller` changed SIGALRM's disposition. Setting SIG_IGN flushes
434    /// a pending copy from every queue without dequeuing it, so an `Inactive`
435    /// timer's copy is gone and the timer stays inactive (`sig_handler_ignored`;
436    /// SIG_DFL never flushes SIGALRM). Decided from the disposition the kernel
437    /// installed, not the call's return value.
438    pub(crate) fn note_sigalrm_disposition(&mut self, caller: DetTid) {
439        if !self.coalesces_alarms() {
440            return;
441        }
442        let process = self.sigchld_process(caller);
443        if self.blocked.timed_waiters.alarm_phase(process)
444            != Some(AlarmPhase::Inactive { copy: true })
445        {
446            return;
447        }
448        match self.read_sleeper_signal_state(caller) {
449            Ok(state) if state.ignored & sigalrm_bit() != 0 => {
450                self.blocked.timed_waiters.alarm_set_copy(process, false);
451            }
452            Ok(_) => {}
453            Err(_) => {
454                self.alarm_unobservable.insert(process);
455            }
456        }
457    }
458
459    /// O3: at a quiesced boundary, reconcile every `Inactive` `ITIMER_REAL`,
460    /// or, with an installed signal control, restart every controlled one
461    /// whose copy was dequeued (`reconcile_dequeued_real_timers`).
462    pub(super) fn reconcile_inactive_alarms(&mut self, t: LogicalTime) {
463        if self.signal_control_installed() {
464            self.reconcile_dequeued_real_timers(t);
465            return;
466        }
467        if !self.coalesces_alarms() {
468            return;
469        }
470        for process in self.blocked.timed_waiters.inactive_alarm_processes() {
471            self.reconcile_alarm(process, t);
472        }
473    }

491    /// The production status read behind `read_process_shared_pending`
492    /// (`shared_pending_from_status_file`, not the test seam) reads a
493    /// process whose leader named itself with bytes that are not UTF-8:
494    /// the `Name:` line is then not UTF-8, and the shared pending set
495    /// must still be read from the ASCII signal lines after it.
496    #[test]
497    fn the_production_status_read_reads_a_process_whose_leader_name_is_not_utf8() {
498        // The child renames itself, then writes a byte to the pipe:
499        // the parent reads the status only after that byte, so the
500        // rename is guaranteed to have happened first. Without the
501        // handshake the parent could read the inherited ASCII name
502        // and the test would pass even with the strict UTF-8 read
503        // restored, whenever it won that race.
504        let mut fds = [0; 2];
505        // SAFETY: the array is writable and owns both descriptors.
506        assert_eq!(unsafe { libc::pipe(fds.as_mut_ptr()) }, 0);
507        // SAFETY: the child only renames itself, reports that, sets
508        // an alarm and pauses; the parent kills and reaps it.
509        let child = unsafe { libc::fork() };
510        assert!(child >= 0);
511        if child == 0 {
512            let name = b"\xff\xfeprobe\0";
513            let name_arg = name.as_ptr() as libc::c_ulong;
514            unsafe {
515                let renamed = libc::prctl(libc::PR_SET_NAME, name_arg, 0, 0, 0);
516                let report: u8 = if renamed == 0 { b'R' } else { b'F' };
517                let _ = libc::write(fds[1], &report as *const u8 as *const libc::c_void, 1);
518                if renamed != 0 {
519                    libc::_exit(5);
520                }
521                libc::alarm(20);
522                loop {
523                    libc::pause();
524                }
525            }
526        }
527        // SAFETY: this parent closes its copies around the child.
528        let mut report: u8 = 0;
529        let read = unsafe {
530            libc::close(fds[1]);
531            libc::read(fds[0], &mut report as *mut u8 as *mut libc::c_void, 1)
532        };
533        // SAFETY: the child is this test's own, killed before reaping.
534        let pending = if read == 1 && report == b'R' {
535            Scheduler::shared_pending_from_status_file(DetPid::from_raw(child))
536        } else {
537            None
538        };
539        unsafe {
540            libc::kill(child, libc::SIGKILL);
541            libc::waitpid(child, std::ptr::null_mut(), 0);
542            libc::close(fds[0]);
543        }
544        assert!(
545            read == 1 && report == b'R',
546            "the forked child failed to rename itself with PR_SET_NAME"
547        );
548        assert_eq!(pending, Some(0));
549    }

551    /// A traced scheduler with one single-threaded process 100 whose
552    /// periodic 1 ms ITIMER_REAL first expires at 1 ms, and whose leader
553    /// neither blocks nor ignores SIGALRM.
554    fn one_process() -> (Scheduler, DetPid, DetTid) {
555        let mut scheduler = Scheduler::new(
556            &Config {
557                sequentialize_threads: true,
558                ..Config::default()
559            }
560            .with_backend(|backend| {
561                detcore_model::config::set_kernel_signal_facts(backend, true);
562                backend.delivers_preemption_timer_events = false;
563                backend.runs_guest_code_between_injections = false;
564            }),
565        );
566        assert!(scheduler.coalesces_alarms());
567        scheduler.test_guest_sends_land = true;
568        let leader = DetTid::from_raw(100);
569        let process = DetPid::from_raw(100);
570        scheduler.thread_tree.add_child(leader, leader, true);
571        register_known_thread_for(&mut scheduler, leader);
572        set_state(&mut scheduler, leader, 0, 0, 0);
573        scheduler.register_alarm(
574            process,
575            leader,

603    /// Pop and dispatch the next timed event at committed time `now`.
604    fn pop_at(scheduler: &mut Scheduler, now: u64) {
605        scheduler.committed_time = at(now);
606        let (time, evt) = scheduler
607            .blocked
608            .timed_waiters
609            .pop_if_before(at(now))
610            .expect("a due expiry");
611        let super::super::timed_waiters::TimedEvent::SignalEvt(id, target, sig) = evt else {
612            panic!("expected a signal timer, got {evt}");
613        };
614        scheduler.dispatch_timed_signal(time, id, target, sig, true, at(now));
615    }
