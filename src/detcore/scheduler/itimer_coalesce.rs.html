9//! Linux's coalescing of a periodic `ITIMER_REAL` on the traced path
10//! (<https://github.com/rrnewton/hermit/issues/4139>).
11//!
12//! Linux's `it_real_fn` sends SIGALRM to the thread-group leader and returns
13//! `HRTIMER_NORESTART`. Only a dequeue of a SIGALRM from the process's shared
14//! queue, whatever sent it, restarts the timer (`posixtimer_rearm_itimer`,
15//! `hrtimer_forward_now` on the original grid). So at most one timer SIGALRM is
16//! outstanding, missed periods are dropped, and `getitimer` reads 0 while the
17//! timer waits. Before this, the traced path re-armed every expiry at the old
18//! deadline plus one interval, so a slice that crossed N periods fired N
19//! SIGALRMs at its end, one per scheduler pass.
20//!
21//! - **Observable processes.** The scheduler re-arms at a dequeue only where it
22//!   can see the dequeue at a committed point: no thread of the process has run
23//!   outside the schedule (`background_call_processes`), no host-timed source
24//!   can post SIGALRM to it (`host_timed_signals`, the unreaped children's exit
25//!   signals), and its leader is live. Those conditions are sticky once false
26//!   (`alarm_unobservable`). Every other process forwards each expiry to the
27//!   first grid point after the pop: missed periods are still dropped, but the
28//!   timer does not wait for the dequeue.
29//! - **The pop.** An observable process's expiry that its leader ignores at
30//!   generation (`sig_ignored`: SIG_IGN, or SIG_DFL in a namespace init, and
31//!   not blocked by the leader) sends nothing and leaves the timer `Inactive`
32//!   with no copy, as Linux discards it. Otherwise the scheduler sends the
33//!   signal and the timer becomes `Inactive` with a copy.
34//! - **Reconcile (O3).** At each quiesced boundary, before the next pass pops
35//!   or grants, an `Inactive` timer with a copy whose process's shared queue no
36//!   longer holds SIGALRM re-arms at the first grid point after the committed
37//!   time. Between two quiesced boundaries only the granted thread of an
38//!   observable process runs, so that read is a function of the schedule.
39//!
40//! Record and replay keep the legacy recurrence and return values
41//! (`recordreplay_modes`), so recordings keep their event stream.

58/// Which processes a guest's committed SIGALRM send could have queued a
59/// shared-queue copy for (rule 4).
60#[derive(Clone, Copy, Debug, Eq, PartialEq, Serialize, Deserialize)]
61pub enum GuestSigalrmTarget {
62    /// `kill(pid > 0)`, `rt_sigqueueinfo(tgid)`.
63    Process(DetPid),
64    /// `kill(0)` (the sender's group, `None`) or `kill(-pgrp)`.
65    Group(Option<DetPid>),
66    /// `kill(-1)`.
67    All,
68    /// A send whose targets Detcore cannot classify (a nested PID namespace's
69    /// pid, a pidfd): every process it might reach stops being observable.
70    Unclassified,
71}

73/// What a read of a process's shared queue says about its ITIMER_REAL copy.
74#[derive(Clone, Copy, Debug, Eq, PartialEq)]
75enum SharedCopy {
76    Dequeued,
77    Pending,
78    ProcessGone,
79    Unreadable,
80}

82/// The bit of SIGALRM in a kernel signal set.
83fn sigalrm_bit() -> u64 {
84    kernel_signal_bit(libc::SIGALRM)

87impl Scheduler {
88    /// Whether this scheduler coalesces periodic `ITIMER_REAL` expiries at all:
89    /// only on the traced path, which models signal targets, outside record and
90    /// replay, and not where a backend owns process signal control (that one
91    /// has its own dequeue-time re-arm, `real_timer.rs`).
92    pub(super) fn coalesces_alarms(&self) -> bool {
93        self.models_signal_targets && !self.recordreplay_modes && !self.signal_control_installed()
94    }

96    /// Whether the current process state permits identifying a shared-queue
97    /// SIGALRM dequeue. A metadata query must not commit a recurrence-policy
98    /// change during a transient state, such as an exited leader before exec.
99    pub(super) fn alarm_observable_now(&self, process: DetPid) -> bool {
100        if !self.coalesces_alarms() {
101            return false;
102        }
103        if self.alarm_unobservable.contains(&process) {
104            return false;
105        }
106        let bit = sigalrm_bit();
107        let threads = self.process_signal_targets(process);
108        let leader = DetTid::from_raw(process.as_raw());
109        let host_timed = threads
110            .iter()
111            .any(|thread| self.host_timed_signals(*thread) & bit != 0);
112        let foreign_exit = self
113            .thread_tree
114            .unreaped_children_of(process)
115            .into_iter()
116            .filter_map(|child| self.thread_tree.notification_signal(child))
117            .any(|signal| signal == libc::SIGALRM);
118        !self.background_call_processes.contains(&process)
119            && !host_timed
120            && !foreign_exit
121            && threads.contains(&leader)
122    }

124    /// The recurrence owner commits loss of observability at its existing
125    /// timer boundaries. Once committed, it remains sticky.
126    pub(super) fn alarm_observable(&mut self, process: DetPid) -> bool {
127        if !self.coalesces_alarms() || self.alarm_unobservable.contains(&process) {
128            return false;
129        }
130        let observable = self.alarm_observable_now(process);
131        if !observable {
132            debug!(
133                "[dpid {}] ITIMER_REAL dequeues are not observable from here on; expiries forward at the pop.",
134                process
135            );
136            self.alarm_unobservable.insert(process);
137        }
138        observable
139    }

141    /// Whether `process`'s leader ignores SIGALRM at generation, as Linux's
142    /// `sig_ignored(leader)` decides: the disposition is SIG_IGN, or SIG_DFL
143    /// when the process is a guest namespace init (`sig_task_ignored` discards
144    /// that for a `SIGNAL_UNKILLABLE` task), and the leader's committed mask
145    /// does not block it. `None` when the leader's state cannot be read.
146    fn leader_ignores_sigalrm(&self, process: DetPid) -> Option<bool> {
147        let leader = DetTid::from_raw(process.as_raw());
148        let state = self.read_sleeper_signal_state(leader).ok()?;
149        let bit = sigalrm_bit();
150        let mask = self
151            .saved_guest_sigmasks
152            .get(&leader)
153            .copied()
154            .unwrap_or(state.blocked);
155        let sig_ign = state.ignored & bit != 0;
156        let sig_dfl_init = self.thread_tree.is_pid_namespace_init(process)
157            && state.ignored & bit == 0
158            && state.caught & bit == 0;
159        Some((sig_ign || sig_dfl_init) && mask & bit == 0)
160    }

162    /// Fire the expiry at `deadline` of `process`'s `ITIMER_REAL`, which
163    /// `pop_if_before` has just popped (and, if periodic, re-armed one interval
164    /// later), then replace that legacy recurrence.
165    pub(super) fn fire_itimer_real(
166        &mut self,
167        deadline: LogicalTime,
168        process: DetPid,
169        tid: DetTid,
170        signal: Signal,
171    ) {
172        let periodic = self
173            .blocked
174            .timed_waiters
175            .signal_timer_period(SignalTimerId::Alarm(process))
176            .is_some();
177        if !periodic || !self.coalesces_alarms() {
178            self.fire_itimer_real_copy(process, tid, signal);
179            return;
180        }
181        let now = self.committed_time;
182        if self.exec_teardowns.contains_key(&process) {
183            // The leader's mask cannot be read from `/proc` during `de_thread`:
184            // hold the expiry and decide it when the exec barrier ends
185            // (`finish_exec_alarm`), keeping the original grid.
186            self.blocked
187                .timed_waiters
188                .alarm_after_fire(process, deadline, Some(false), now);
189            self.alarm_exec_deferred.insert(process, (tid, signal));
190            return;
191        }
192        if !self.alarm_observable(process) {
193            self.fire_itimer_real_copy(process, tid, signal);
194            self.blocked
195                .timed_waiters
196                .alarm_after_fire(process, deadline, None, now);
197            return;
198        }
199        match self.leader_ignores_sigalrm(process) {
200            Some(true) => {
201                info!(
202                    "[dpid {}] SIGALRM expired while its leader ignores it; discarded at generation, the timer waits for a dequeue.",
203                    process
204                );
205                self.blocked
206                    .timed_waiters
207                    .alarm_after_fire(process, deadline, Some(false), now);
208            }
209            Some(false) => {
210                self.guest_send_landed = None;
211                let sent = self.fire_itimer_real_copy(process, tid, signal);
212                let landed = sent && self.guest_send_landed != Some(false);
213                let copy = landed.then_some(true);
214                self.blocked
215                    .timed_waiters
216                    .alarm_after_fire(process, deadline, copy, now);
217            }
218            None => {
219                self.alarm_unobservable.insert(process);
220                self.fire_itimer_real_copy(process, tid, signal);
221                self.blocked
222                    .timed_waiters
223                    .alarm_after_fire(process, deadline, None, now);
224            }
225        }
226    }

228    /// At the end of `process`'s exec barrier, decide an expiry that popped
229    /// during it, then run the reconcile the barrier deferred. Both use the
230    /// committed time at this point.
231    pub(super) fn finish_exec_alarm(&mut self, process: DetPid) {
232        let t = self.committed_time;
233        if let Some((tid, signal)) = self.alarm_exec_deferred.remove(&process) {
234            let observable = self.alarm_observable(process);
235            let ignored = if observable {
236                self.leader_ignores_sigalrm(process)
237            } else {
238                None
239            };
240            match ignored {
241                Some(true) => {}
242                Some(false) => {
243                    self.guest_send_landed = None;
244                    let sent = self.fire_itimer_real_copy(process, tid, signal);
245                    if sent && self.guest_send_landed != Some(false) {
246                        self.blocked.timed_waiters.alarm_set_copy(process, true);
247                    } else {
248                        self.blocked.timed_waiters.alarm_rearm_after(process, t);
249                    }
250                }
251                None => {
252                    if observable {
253                        self.alarm_unobservable.insert(process);
254                    }
255                    self.fire_itimer_real_copy(process, tid, signal);
256                    self.blocked.timed_waiters.alarm_rearm_after(process, t);
257                }
258            }
259        }
260        self.reconcile_alarm(process, t);
261    }

263    /// `process`'s shared pending set, read through the thread group
264    /// (`/proc/<tgid>/status`, valid while any thread lives).
265    fn read_process_shared_pending(&self, process: DetPid) -> Option<u64> {
266        #[cfg(test)]
267        {
268            let leader = DetTid::from_raw(process.as_raw());
269            self.read_shared_pending(leader)
270        }
271        #[cfg(not(test))]
272        {
273            let status =
274                std::fs::read_to_string(format!("/proc/{}/status", process.as_raw())).ok()?;
275            crate::syscalls::KernelSignalState::parse(&status).map(|state| state.shared_pending)
276        }
277    }

279    /// Reconcile `process`'s `Inactive` `ITIMER_REAL` at the committed time
280    /// `t`: re-arm it when its copy has left the shared queue, or when the
281    /// process stopped being observable (rule 6). Returns the new deadline if
282    /// it re-armed.
283    pub(crate) fn reconcile_alarm(
284        &mut self,
285        process: DetPid,
286        t: LogicalTime,
287    ) -> Option<LogicalTime> {
288        let AlarmPhase::Inactive { copy } = self.blocked.timed_waiters.alarm_phase(process)? else {
289            return None;
290        };
291        if self.exec_teardowns.contains_key(&process) {
292            // `/proc/<tgid>` can be briefly unresolvable during `de_thread`;
293            // the barrier's end reconciles.
294            return None;
295        }
296        if !self.alarm_observable(process) {
297            let next = self.blocked.timed_waiters.alarm_rearm_after(process, t);
298            debug!(
299                "[dpid {}] ITIMER_REAL re-armed at {:?}: its process is no longer observable.",
300                process, next
301            );
302            return next;
303        }
304        // Invariant: the queue is read only when a copy is known to be there.
305        // Under ptrace a guest kill() to a leader that ignores SIGALRM leaves a
306        // physical copy native Linux drops; this read must not count it.
307        if !copy {
308            return None;
309        }
310        match self.shared_copy_outcome(process) {
311            SharedCopy::Dequeued => {
312                let next = self.blocked.timed_waiters.alarm_rearm_after(process, t);
313                debug!(
314                    "[dpid {}] SIGALRM left the shared queue; ITIMER_REAL re-armed at {:?}.",
315                    process, next
316                );
317                next
318            }
319            SharedCopy::Pending => None,
320            SharedCopy::ProcessGone => {
321                self.blocked.timed_waiters.remove_alarm(process);
322                None
323            }
324            SharedCopy::Unreadable => exit_on_scheduler_refusal(&format_args!(
325                "HERMIT_SIGALRM_QUEUE_UNREADABLE: the shared signal queue of process {} could not be read while it has a live thread, so whether its ITIMER_REAL copy was dequeued is unknown (https://github.com/rrnewton/hermit/issues/4139)",
326                process
327            )),
328        }
329    }

331    /// What `process`'s shared queue says about its ITIMER_REAL copy. A read
332    /// that fails while the process has a committed-live thread is
333    /// `Unreadable`, which refuses the run; one that fails after its exit
334    /// committed is `ProcessGone`.
335    fn shared_copy_outcome(&self, process: DetPid) -> SharedCopy {
336        match self.read_process_shared_pending(process) {
337            Some(pending) if pending & sigalrm_bit() == 0 => SharedCopy::Dequeued,
338            Some(_) => SharedCopy::Pending,
339            None if self.process_signal_targets(process).is_empty() => SharedCopy::ProcessGone,
340            None => SharedCopy::Unreadable,
341        }
342    }

344    /// Rule 4: a guest's committed SIGALRM send from `sender`. An `Inactive`
345    /// timer with no known copy (its last expiry was discarded at
346    /// generation) now has one, unless the leader still ignores SIGALRM at
347    /// generation, so the dequeue of that copy restarts it, as Linux restarts
348    /// a timer on any shared-queue SIGALRM dequeue.
349    pub(crate) fn note_guest_sigalrm(&mut self, sender: DetTid, target: GuestSigalrmTarget) {
350        if !self.coalesces_alarms() {
351            return;
352        }
353        let inactive = self.blocked.timed_waiters.inactive_alarm_processes();
354        let targets: Vec<DetPid> = match target {
355            GuestSigalrmTarget::Process(process) => vec![process],
356            GuestSigalrmTarget::Group(group) => {
357                let group =
358                    group.or_else(|| self.thread_tree.process_group(self.sigchld_process(sender)));
359                match group {
360                    Some(group) => inactive
361                        .iter()
362                        .copied()
363                        .filter(|process| self.thread_tree.process_group(*process) == Some(group))
364                        .collect(),
365                    None => {
366                        self.alarm_unobservable.extend(inactive.iter().copied());
367                        return;
368                    }
369                }
370            }
371            GuestSigalrmTarget::All => inactive.clone(),
372            GuestSigalrmTarget::Unclassified => {
373                self.alarm_unobservable.extend(inactive.iter().copied());
374                return;
375            }
376        };
377        for process in targets {
378            if self.blocked.timed_waiters.alarm_phase(process)
379                != Some(AlarmPhase::Inactive { copy: false })
380            {
381                continue;
382            }
383            match self.leader_ignores_sigalrm(process) {
384                Some(false) => self.blocked.timed_waiters.alarm_set_copy(process, true),
385                Some(true) => {}
386                None => {
387                    self.alarm_unobservable.insert(process);
388                }
389            }
390        }
391    }

393    /// Rule 5: `caller` changed SIGALRM's disposition. Setting SIG_IGN flushes
394    /// a pending copy from every queue without dequeuing it, so an `Inactive`
395    /// timer's copy is gone and the timer stays inactive (`sig_handler_ignored`;
396    /// SIG_DFL never flushes SIGALRM). Decided from the disposition the kernel
397    /// installed, not the call's return value.
398    pub(crate) fn note_sigalrm_disposition(&mut self, caller: DetTid) {
399        if !self.coalesces_alarms() {
400            return;
401        }
402        let process = self.sigchld_process(caller);
403        if self.blocked.timed_waiters.alarm_phase(process)
404            != Some(AlarmPhase::Inactive { copy: true })
405        {
406            return;
407        }
408        match self.read_sleeper_signal_state(caller) {
409            Ok(state) if state.ignored & sigalrm_bit() != 0 => {
410                self.blocked.timed_waiters.alarm_set_copy(process, false);
411            }
412            Ok(_) => {}
413            Err(_) => {
414                self.alarm_unobservable.insert(process);
415            }
416        }
417    }
418
419    /// O3: at a quiesced boundary, reconcile every `Inactive` `ITIMER_REAL`.
420    pub(super) fn reconcile_inactive_alarms(&mut self, t: LogicalTime) {
421        if !self.coalesces_alarms() {
422            return;
423        }
424        for process in self.blocked.timed_waiters.inactive_alarm_processes() {
425            self.reconcile_alarm(process, t);
426        }
427    }

445    /// A traced scheduler with one single-threaded process 100 whose
446    /// periodic 1 ms ITIMER_REAL first expires at 1 ms, and whose leader
447    /// neither blocks nor ignores SIGALRM.
448    fn one_process() -> (Scheduler, DetPid, DetTid) {
449        let mut scheduler = Scheduler::new(
450            &Config {
451                sequentialize_threads: true,
452                ..Config::default()
453            }
454            .with_backend(|backend| {
455                backend.supports_blocked_wait_signal_interruption = true;
456                backend.delivers_preemption_timer_events = false;
457                backend.runs_guest_code_between_injections = false;
458            }),
459        );
460        assert!(scheduler.coalesces_alarms());
461        scheduler.test_guest_sends_land = true;
462        let leader = DetTid::from_raw(100);
463        let process = DetPid::from_raw(100);
464        scheduler.thread_tree.add_child(leader, leader, true);
465        register_known_thread_for(&mut scheduler, leader);
466        set_state(&mut scheduler, leader, 0, 0, 0);
467        scheduler.register_alarm(
468            process,
469            leader,
470            at(0),
471            at(PERIOD),
472            at(PERIOD),
473            Signal::SIGALRM,
474        );
475        (scheduler, process, leader)
476    }

497    /// Pop and dispatch the next timed event at committed time `now`.
498    fn pop_at(scheduler: &mut Scheduler, now: u64) {
499        scheduler.committed_time = at(now);
500        let (time, evt) = scheduler
501            .blocked
502            .timed_waiters
503            .pop_if_before(at(now))
504            .expect("a due expiry");
505        let super::super::timed_waiters::TimedEvent::SignalEvt(id, target, sig) = evt else {
506            panic!("expected a signal timer, got {evt}");
507        };
508        scheduler.dispatch_timed_signal(time, id, target, sig, true, at(now));
509    }

515    /// A pop sends the signal and leaves the timer Inactive with a copy, with
516    /// no deadline queued however many periods the slice crossed; the copy's
517    /// departure from the shared queue re-arms it on the original grid at the
518    /// next quiesced boundary. While the copy is still pending nothing moves.
519    #[test]
520    fn a_pop_waits_for_the_dequeue_and_re_arms_on_the_grid() {
521        let (mut scheduler, process, leader) = one_process();
522        // The slice crossed 5.5 periods before this pop.
523        pop_at(&mut scheduler, 5_500_000);
524        assert_eq!(
525            phase(&scheduler, process),
526            Some(AlarmPhase::Inactive { copy: true })
527        );
528        assert!(
529            scheduler.blocked.timed_waiters.is_empty(),
530            "no backlog of expiries"
531        );
532        // Still pending: no change.
533        set_state(&mut scheduler, leader, sigalrm_bit(), 0, 0);
534        assert_eq!(scheduler.reconcile_alarm(process, at(5_800_000)), None);
535        // Dequeued: re-armed at the first grid point after t.
536        set_state(&mut scheduler, leader, 0, 0, 0);
537        assert_eq!(
538            scheduler.reconcile_alarm(process, at(7_200_000)),
539            Some(at(8_000_000))
540        );
541        assert_eq!(phase(&scheduler, process), Some(AlarmPhase::Active));
542        assert_eq!(
543            scheduler.blocked.timed_waiters.next_deadline(),
544            Some(at(8_000_000))
545        );
546    }
