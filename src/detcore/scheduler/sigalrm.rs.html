9//! The scheduler's SIGALRM ledger, for guest SIGALRM handlers on a backend
10//! that runs Detcore inside each guest process (in-guest LiteInst).
11//!
12//! Design: dev-hermit `ai_docs/transient/liteinst-inguest-signal-handlers-design-20261007.md`,
13//! phase 1, step I1a, under <https://github.com/rrnewton/hermit/issues/3520>.
14//!
15//! - A process's runtime publishes whether its SIGALRM disposition is a guest
16//!   handler. While it is, an emulated `alarm`/`setitimer` expiry adds a ledger
17//!   entry instead of sending a physical signal. A second expiry while one is
18//!   pending is absorbed: standard-signal coalescing in the process's one
19//!   shared queue.
20//! - The runtime also publishes each thread's "SIGALRM virtually blocked" bit,
21//!   inside the turn of the call that changes its mask. No expiry commits
22//!   while a thread holds its turn, so the bit is current at every expiry and
23//!   every admission. An entry is *due* when that bit is clear. (Publishing it
24//!   by control message rather than on every request leaves the request wire,
25//!   `Resources`, unchanged.)
26//! - A due entry ends an emulated `pause`: at the pause's own admission, when
27//!   the entry is already due as it is selected, or at the expiry, when the
28//!   pause is already parked. Either way the pause returns EINTR.
29//! - A due entry whose owner is blocked in any other wait records a
30//!   determinism loss: Linux would interrupt that wait, and phase 1 cannot.
31//!   That includes a polling wait (`InternalIOPolling` retried after EAGAIN),
32//!   whose thread stays on the run queue between retries. A running thread, or
33//!   one whose filed request has not blocked, takes the entry at its syscall's
34//!   completion; that delivery is the runtime's (later steps).
35//! - Every change happens at a scheduler commit: an expiry, an admission, or
36//!   the sweep that opens step 2. So what is pending, and what is due, is a
37//!   function of the schedule.
38//!
39//! Nothing publishes a handled SIGALRM until the runtime admits guest
40//! handlers, so this module changes no run today.

60/// The scheduler's record of guest-handled SIGALRMs.
61#[derive(Debug, Default)]
62pub(crate) struct SigalrmLedger {
63    /// Processes whose SIGALRM disposition is a guest handler.
64    handled: BTreeSet<DetPid>,
65    /// Processes with a pending SIGALRM: at most one each.
66    pending: BTreeSet<DetPid>,
67    /// Each thread's "SIGALRM virtually blocked" bit, as its runtime last
68    /// published it.
69    blocked: BTreeMap<DetTid, bool>,
70    /// Threads whose `InternalIOPolling` retry the run queue has deferred as a
71    /// poller. Kept here because the deferral resets the stored request's
72    /// attempt count (`upgrade_polled_to_runnable`); cleared when the request
73    /// is granted (`clear_nextturn`) or substituted (`force_unblock_thread`).
74    polling: BTreeSet<DetTid>,
75    /// Pending entries whose determinism loss is already recorded, so a
76    /// stranded entry is reported once.
77    lost: BTreeSet<DetPid>,
78    /// Whether any guest armed a kernel producer of SIGALRM
79    /// (`SigalrmControl::ArmProducer`). Never cleared: the producer may fire
80    /// at any later time.
81    producer_armed: bool,
82}
83
84impl Scheduler {
85    /// Whether `process` has published a guest handler for SIGALRM.
86    // Called by the runtime's control messages, which arrive with its handler
87    // admission (phase 1 step I3); until then only tests call it.
88    #[cfg_attr(not(test), allow(dead_code))]
89    pub(crate) fn sigalrm_handled(&self, process: DetPid) -> bool {
90        self.sigalrm.handled.contains(&process)
91    }
92
93    /// Whether `process` has a pending SIGALRM entry.
94    // Called by the runtime's control messages, which arrive with its handler
95    // admission (phase 1 step I3); until then only tests call it.
96    #[cfg_attr(not(test), allow(dead_code))]
97    pub(crate) fn sigalrm_pending(&self, process: DetPid) -> bool {
98        self.sigalrm.pending.contains(&process)
99    }
100
101    /// Records `process`'s SIGALRM disposition, published by its runtime inside
102    /// the turn of the call that changed it. A handler that becomes ignored or
103    /// default discards a pending entry, as Linux discards a pending signal that
104    /// becomes ignored. The runtime refuses the change to default while an
105    /// entry is pending, so a default-acted signal is never discarded here.
106    pub(crate) fn set_sigalrm_handled(&mut self, process: DetPid, handled: bool) {
107        if handled {
108            self.sigalrm.handled.insert(process);
109        } else {
110            self.sigalrm.handled.remove(&process);
111            self.sigalrm.pending.remove(&process);
112            self.sigalrm.lost.remove(&process);
113        }
114    }
115
116    /// Decides a SIGALRM control message from `thread`'s Detcore, sent in its
117    /// turn before the call reaches the kernel; returns whether it is refused.
118    ///
119    /// While no process handles SIGALRM, nothing is refused, so runs without a
120    /// guest handler behave as before; an allowed producer arming is recorded.
121    /// Otherwise a sender that does not hold the serial grant is refused: its
122    /// question could race a commit. A send is refused when it reaches a
123    /// handling process (one the caller cannot name could be one), every
124    /// producer arming is refused, and a recurring `ITIMER_REAL` is refused in
125    /// a handling process.
126    pub(crate) fn sigalrm_control(&mut self, thread: DetTid, control: SigalrmControl) -> bool {
127        // The runtime's publications (phase 1 step I3) record what its
128        // runtime already changed inside the caller's turn; they hold whether
129        // or not any process handles SIGALRM yet.
130        let process = self.sigchld_process(thread);
131        match control {
132            SigalrmControl::Publish { handled, blocked } => {
133                self.set_sigalrm_handled(process, handled);
134                self.set_sigalrm_blocked(thread, blocked);
135                return false;
136            }
137            SigalrmControl::PublishBlocked(blocked) => {
138                self.set_sigalrm_blocked(thread, blocked);
139                return false;
140            }
141            SigalrmControl::DeliveryLost => {
142                crate::detlog::write_loss_notice(&format!(
143                    "thread {thread} of process {process} took a pending guest-handled SIGALRM, but the runtime could not deliver it"
144                ));
145                self.sigalrm.lost.insert(process);
146                return false;
147            }
148            SigalrmControl::EntryDeliveryUnfaithful => {
149                crate::detlog::write_loss_notice(&format!(
150                    "thread {thread} of process {process} delivers a guest-handled SIGALRM before a syscall that has not run; its handler cannot see the registers Linux would show it (rcx and r11 as the syscall instruction left them)"
151                ));
152                self.sigalrm.lost.insert(process);
153                return false;
154            }
155            _ => {}
156        }
157        // Its questions, like every other control, are refused to a caller
158        // without the serial grant while any process handles SIGALRM: the
159        // answer could race a commit. (A handling process is sequentialized,
160        // so its own calls hold the grant.)
161        if !self.sigalrm.handled.is_empty() && !self.holds_serial_grant(thread) {
162            return true;
163        }
164        match control {
165            SigalrmControl::InstallHandler => {
166                // Also refused while the caller's process has a recurring
167                // ITIMER_REAL armed: phase 1 does not model one in a handling
168                // process (it is refused once a handler exists, too).
169                let recurring = self
170                    .blocked
171                    .timed_waiters
172                    .alarm_state(process)
173                    .is_some_and(|(_, interval)| interval != crate::types::LogicalTime::ZERO);
174                let refused = self.sigalrm_producer_armed() || recurring;
175                if refused {
176                    info!(
177                        "[dtid {}] SIGALRM handler refused: a guest armed a kernel producer of SIGALRM, or its process has a recurring ITIMER_REAL (signal phase 1).",
178                        thread
179                    );
180                }
181                return refused;
182            }
183            SigalrmControl::HandlerToDefault | SigalrmControl::PendingEntry => {
184                return self.sigalrm.pending.contains(&process);
185            }
186            SigalrmControl::DueEntry => return self.sigalrm_due(thread),
187            SigalrmControl::TakeEntry => {
188                let taken = self.take_sigalrm(process, thread);
189                if taken {
190                    info!(
191                        "[dtid {}] takes its process's pending SIGALRM for delivery to its handler.",
192                        thread
193                    );
194                }
195                return taken;
196            }
197            _ => {}
198        }
199        if self.sigalrm.handled.is_empty() {
200            if control == SigalrmControl::ArmProducer {
201                self.sigalrm.producer_armed = true;
202            }
203            return false;
204        }
205        if !self.holds_serial_grant(thread) {
206            return true;
207        }
208        if control == SigalrmControl::UnadmittedStdioIo {
209            if self.sigalrm_due(thread) {
210                self.record_sigalrm_loss(
211                    DetPid::from_raw(thread.as_raw()),
212                    thread,
213                    "a due SIGALRM found it entering I/O on inherited stdio that may sleep unadmitted",
214                );
215            }
216            return false;
217        }
218        let refused = match control {
219            SigalrmControl::SendTo(target) => {
220                self.sigalrm.handled.contains(&self.sigchld_process(target))
221            }
222            SigalrmControl::SendToUnnamed | SigalrmControl::ArmProducer => true,
223            SigalrmControl::UnadmittedStdioIo
224            | SigalrmControl::InstallHandler
225            | SigalrmControl::HandlerToDefault
226            | SigalrmControl::PendingEntry
227            | SigalrmControl::Publish { .. }
228            | SigalrmControl::PublishBlocked(_)
229            | SigalrmControl::DueEntry
230            | SigalrmControl::TakeEntry
231            | SigalrmControl::DeliveryLost
232            | SigalrmControl::EntryDeliveryUnfaithful => unreachable!("answered above"),
233            SigalrmControl::ArmRecurringTimer => {
234                self.sigalrm.handled.contains(&self.sigchld_process(thread))
235            }
236        };
237        if refused {
238            info!(
239                "[dtid {}] {:?} refused: a guest handles SIGALRM (signal phase 1).",
240                thread, control
241            );
242        }
243        refused
244    }

246    /// Whether any guest armed a kernel producer of SIGALRM; a handler

248    pub(crate) fn sigalrm_producer_armed(&self) -> bool {

253    /// runtime inside the turn of the call that changed its mask (and at

255    pub(crate) fn set_sigalrm_blocked(&mut self, thread: DetTid, blocked: bool) {

259    /// Notes that step 4 deferred `thread`'s polling retry `request` as a
260    /// poller, before the deferral resets its attempt count.
261    pub(crate) fn note_sigalrm_polling_wait(&mut self, thread: DetTid, request: &Resources) {
262        if Self::is_polling_retry(request) {
263            self.sigalrm.polling.insert(thread);
264        }
265    }

267    /// Forgets that `thread` is in a polling wait: its request was granted or
268    /// substituted.
269    pub(crate) fn end_sigalrm_polling_wait(&mut self, thread: DetTid) {
270        self.sigalrm.polling.remove(&thread);
271    }

273    /// A retry of an `InternalIOPolling` operation after EAGAIN.
274    fn is_polling_retry(request: &Resources) -> bool {
275        request.poll_attempt > 0
276            && request
277                .resources
278                .contains_key(&ResourceID::InternalIOPolling)
279    }
280
281    /// Whether `thread` is in a polling wait: its filed request is a retry
282    /// step 4 has not seen yet, or step 4 deferred it as a poller. Between its
283    /// grant and its next request the thread holds its turn, when no expiry
284    /// commits.
285    fn in_polling_wait(&self, thread: DetTid) -> bool {
286        self.sigalrm.polling.contains(&thread)
287            || self
288                .next_turns
289                .get(&thread)
290                .and_then(|turn| turn.req.try_read())
291                .is_some_and(|request| request.is_ok_and(|r| Self::is_polling_retry(&r)))
292    }

294    /// Diverts an expiry of `signal` for `process` into the ledger when the
295    /// process handles SIGALRM. Returns whether it was diverted; otherwise the
296    /// caller sends the signal as before.
297    pub(crate) fn divert_sigalrm(&mut self, process: DetPid, signal: Signal) -> bool {
298        if signal != Signal::SIGALRM || !self.sigalrm.handled.contains(&process) {
299            return false;
300        }
301        if self.sigalrm.pending.insert(process) {
302            info!(
303                "[dpid {}] SIGALRM expired; a guest handler is installed, so it is pending in the ledger.",
304                process
305            );
306        } else {
307            info!(
308                "[dpid {}] SIGALRM expired while one was already pending; coalesced.",
309                process
310            );
311        }
312        self.act_on_pending_sigalrm(process);
313        true
314    }
315
316    /// The sweep that opens step 2: acts on every pending entry, in process
317    /// order. It records the loss of a thread that blocked, in the last pass,
318    /// in a wait admitted with its entry already due. It runs before the timed
319    /// events, so a sleep that blocked and is now expiring is seen blocked.
320    pub(crate) fn step2_sigalrm_ledger(&mut self) {
321        let pending: Vec<DetPid> = self.sigalrm.pending.iter().copied().collect();
322        for process in pending {
323            self.act_on_pending_sigalrm(process);
324        }
325    }

327    /// Admission of a `SleepUntil(INDEFINITE)` request by the selected `thread`:
328    /// when it is an emulated `pause` and its process's entry is due, the
329    /// request is resolved now with `Signaled(SIGALRM)` instead of parking, in
330    /// the same selected turn. Returns whether it was; the caller then grants
331    /// the turn.
332    pub(crate) fn admit_pause_with_due_sigalrm(&mut self, thread: DetTid) -> bool {
333        if !self.sigalrm_due(thread) || !self.filed_pause(thread) {
334            return false;
335        }
336        info!(
337            "[dtid {}] pause admitted with a due SIGALRM; it returns at once.",
338            thread
339        );
340        let wake = Self::sigalrm_wake(thread);
341        if let Some(turn) = self.next_turns.get_mut(&thread) {
342            turn.req = Ivar::full(Ok(wake));
343        }
344        true
345    }

347    /// Takes `process`'s entry for delivery, if it has one and it is due for
348    /// `thread`. The caller has checked that `thread` holds the serial grant.
349    pub(crate) fn take_sigalrm(&mut self, process: DetPid, thread: DetTid) -> bool {
350        if !self.sigalrm.handled.contains(&process)
351            || self.sigalrm.blocked.get(&thread) != Some(&false)
352            || !self.sigalrm.pending.remove(&process)
353        {
354            return false;
355        }
356        self.sigalrm.lost.remove(&process);
357        true
358    }

360    /// Forgets `thread`, and its process's ledger state when the thread was
361    /// its process's leader: Linux drops pending signals at exit.
362    pub(crate) fn retire_sigalrm_thread(&mut self, thread: DetTid, process: DetPid) {
363        self.sigalrm.blocked.remove(&thread);
364        self.sigalrm.polling.remove(&thread);
365        if thread.as_raw() == process.as_raw() {
366            self.sigalrm.handled.remove(&process);
367            self.sigalrm.pending.remove(&process);
368            self.sigalrm.lost.remove(&process);
369        }
370    }

372    /// Wakes, reports or leaves `process`'s pending entry according to where its
373    /// thread is.
374    fn act_on_pending_sigalrm(&mut self, process: DetPid) {
375        let thread = Self::sigalrm_owner(process);
376        if !self.next_turns.contains_key(&thread) {
377            // The process is gone; its pending signal goes with it.
378            self.sigalrm.pending.remove(&process);
379            self.sigalrm.lost.remove(&process);
380            return;
381        }
382        match self.sigalrm.blocked.get(&thread) {
383            // Blocked: the entry stays pending, as in Linux.
384            Some(true) => return,
385            Some(false) => {}
386            // Every request of a process that handles SIGALRM carries the bit.
387            // Without it, eligibility is unknown, so fail closed.
388            None => {
389                self.record_sigalrm_loss(process, thread, "its SIGALRM eligibility is unknown");
390                return;
391            }
392        }
393        if !matches!(self.thread_status(thread), ThreadStatus::NotRunning) {
394            if self.in_polling_wait(thread) {
395                self.record_sigalrm_loss(
396                    process,
397                    thread,
398                    "a due SIGALRM found it in a polling wait that phase 1 cannot interrupt",
399                );
400            }
401            // Otherwise running, or a filed request that has not blocked: a
402            // pause is resolved at its admission, any other call takes the
403            // entry at its completion.
404            return;
405        }
406        if self.filed_pause(thread) {
407            info!("[dtid {}] a pending SIGALRM wakes its pause.", thread);
408            self.force_unblock_thread(thread, Self::sigalrm_wake(thread));
409        } else {
410            self.record_sigalrm_loss(
411                process,
412                thread,
413                "a due SIGALRM found it blocked in a wait that phase 1 cannot interrupt",
414            );
415        }
416    }

418    /// The thread that owns `process`'s entry. In-guest processes are
419    /// single-threaded, so it is the process's leader.
420    fn sigalrm_owner(process: DetPid) -> DetTid {
421        DetTid::from_raw(process.as_raw())
422    }
423
424    /// Whether `thread`'s process holds an entry that is due for it.
425    fn sigalrm_due(&self, thread: DetTid) -> bool {
426        let process = DetPid::from_raw(thread.as_raw());
427        self.sigalrm.handled.contains(&process)
428            && self.sigalrm.pending.contains(&process)
429            && self.sigalrm.blocked.get(&thread) == Some(&false)
430    }
431
432    /// The request a SIGALRM wake substitutes for the thread's filed one; its
433    /// grant answers `Signaled(SIGALRM)` (`unblock_guest`).
434    fn sigalrm_wake(thread: DetTid) -> Resources {
435        let mut wake = Resources::new(thread);
436        wake.insert(
437            ResourceID::InboundSignal(SigWrapper::from(Signal::SIGALRM)),
438            Permission::W,
439        );
440        wake
441    }

443    /// Whether a determinism loss is recorded for `process`'s pending entry.
444    #[cfg(test)]
445    pub(crate) fn sigalrm_loss_recorded(&self, process: DetPid) -> bool {
446        self.sigalrm.lost.contains(&process)
447    }

449    fn record_sigalrm_loss(&mut self, process: DetPid, thread: DetTid, why: &str) {
450        if self.sigalrm.lost.insert(process) {
451            crate::detlog::write_loss_notice(&format!(
452                "thread {thread} of process {process} has a pending guest-handled SIGALRM, but {why}"
453            ));
454        }
455    }

457    /// Whether `thread`'s filed request is an emulated `pause`: a wait with no
458    /// deadline (`SleepUntil(INDEFINITE)`) tagged `PAUSE_FYI` by `handle_pause`.
459    /// Under in-guest LiteInst it is an ordinary request, since the backend
460    /// offers no parked signal sites. A `nanosleep` whose deadline saturates
461    /// files the same resource without the tag; it is not a pause.
462    fn filed_pause(&self, thread: DetTid) -> bool {
463        self.next_turns
464            .get(&thread)
465            .and_then(|turn| turn.req.try_read())
466            .is_some_and(|request| {
467                request.is_ok_and(|resources| {
468                    resources.fyi == PAUSE_FYI
469                        && resources.resources.keys().any(|resource| {
470                        matches!(resource, ResourceID::SleepUntil(deadline) if deadline.is_indefinite())
471                    })
472                })
473            })
474    }
