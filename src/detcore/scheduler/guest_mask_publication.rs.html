9//! A5 step 3c.1e, sub-slice A (Detcore groundwork; dev-hermit
10//! ai_docs/2026-10-16-a5-signal-delivery-design.md, "3c.1e plan of record and
11//! decisions D56-D68", and ai_docs/evidence/a5-3c1e-20261024/plan-3c1e-r4.md
12//! section 1 A with the parts it carries from revisions 3 and 2;
13//! <https://github.com/rrnewton/hermit/issues/4485>): what the scheduler
14//! holds of a guest thread's mask under the in-guest delivery gate
15//! (`Config::in_guest_signal_delivery`), beyond the mask each request
16//! publishes (step 3c.1b, `Scheduler::record_request_guest_mask`):
17//!
18//! - IR-18's tri-state: a published mask is [`PublishedGuestMask::Known`],
19//!   or [`PublishedGuestMask::Unknown`] after a rejoin that returned under
20//!   the guest's own mask (`ContinuationMask::Restored`) but published none,
21//!   on a backend that keeps the guest's mask apart for that thread, so that
22//!   no reader shows the call's earlier, temporary mask as the one in force
23//!   (`Scheduler::keep_mask_for_continuation`); on a backend that keeps none
24//!   (every backend before step 3c.1e B) nothing is published then, and
25//!   readers take `SigBlk`, which is the guest's own mask. The `/proc` status
26//!   reader refuses by name on `Unknown` (`HERMIT_GUEST_MASK_UNKNOWN`); every
27//!   scheduler-internal reader takes `Unknown` as nothing published (the
28//!   kernel's `SigBlk`), as it takes a request that publishes nothing.
29//! - Fork-child seeding: a new task starts with its creator's published own
30//!   mask, never a masked call's temporary mask that the creator still
31//!   publishes, as Linux's child starts with its creator's `blocked`
32//!   (`arch_dup_task_struct`, kernel/fork.c:1086-1091 at v6.12, from
33//!   `dup_task_struct`, :1112; `copy_process` empties only `pending`,
34//!   :2268) ([`Scheduler::seed_child_guest_sigmask`]).
35//! - IR-23(a): a masked wait that must make an RPC before its first probe (the
36//!   publication of expired virtual timerfds, `timerfd::publish_expired`,
37//!   while the process holds an armed virtual timerfd) first publishes its
38//!   temporary mask, with the ignored signals it keeps pending from its entry
39//!   in each queue (D37's held masks), through the non-yielding
40//!   `GlobalRequest::PublishTemporaryMask`
41//!   ([`Scheduler::publish_temporary_mask`]). Linux has installed that mask
42//!   for the call's duration (`set_user_sigmask`, kernel/signal.c:3169 at
43//!   v6.12), so a cross-process `/proc` reader during the RPC sees it. The
44//!   next request the thread files supersedes it; a call that files none
45//!   withdraws it as it returns ([`Scheduler::withdraw_temporary_mask`]),
46//!   which puts back what was published before it.
47//! - The committed-send window record for `HERMIT_IGNORED_REALTIME_INSTANCES`:
48//!   while such a publication is in force, each send the scheduler commits to
49//!   the thread, from its two current commit points (`NotifySignalPending`'s
50//!   handler, which runs only after the guest's injected send succeeded, and
51//!   `send_guest_signal` after its `kill` succeeded), is recorded with its
52//!   queue: a thread-directed send in the private queue, a process-directed
53//!   one in the shared queue ([`Scheduler::note_committed_send_in_window`]).
54//!   The withdrawal answers it, so the call's end (D37's R2) can tell a
55//!   real-time instance sent in the window from one kept from entry, which
56//!   the pending bits cannot (plan revision 3, section 1 A).
57//!
58//! Every function here is a no-op with the gate off, and nothing sets the
59//! gate before step 3c.1f.

67/// A guest mask the scheduler holds for a thread under the in-guest delivery
68/// gate (IR-18's tri-state; absent means nothing published, and readers take
69/// the kernel's `SigBlk`). It travels in a bincode RPC, so it has no
70/// `skip_serializing_if`.
71#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
72pub enum PublishedGuestMask {
73    /// The mask the thread published.
74    Known(u64),
75    /// The thread published none where its mask changed: a rejoin that
76    /// returned under the guest's own mask, on a backend that otherwise keeps
77    /// the guest's mask apart for that thread. A reader must not take the
78    /// kernel's `SigBlk` for it, which after step 3c.1e B is the in-guest
79    /// runtime's mask.
80    Unknown,
81}
82
83/// The answer to a `/proc` status reader's `GlobalRequest::PublishedGuestSigmask`
84/// under the in-guest delivery gate (an appended response variant,
85/// `GlobalResponse::GuestSigmaskForStatus`): the published mask, tri-state,
86/// and the ignored signals a temporary-mask publication keeps pending from
87/// the call's entry in each queue (D37's held masks; zero without one).
88#[derive(Debug, Clone, Copy, Default, PartialEq, Eq, Serialize, Deserialize)]
89pub struct StatusGuestMask {
90    /// What the thread published, if anything.
91    pub mask: Option<PublishedGuestMask>,
92    /// `SigPnd_entry ∩ I ∩ own`: ignored signals kept pending in the private
93    /// queue from the masked call's entry.
94    pub held_private: u64,
95    /// `ShdPnd_entry ∩ I ∩ own`: the same in the shared queue.
96    pub held_shared: u64,
97}
98
99/// The payload of `GlobalRequest::PublishTemporaryMask` (IR-23(a)): a
100/// publication, filed once per masked call before the first expired-timerfd
101/// publication, or its withdrawal as a call that filed no later request
102/// returns. It travels in a bincode RPC, so it has no
103/// `skip_serializing_if`.
104#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
105pub enum TemporaryMaskControl {
106    /// `thread` waits under `mask` (as Linux installs it, without SIGKILL
107    /// and SIGSTOP), keeping the ignored signals `held_private` and
108    /// `held_shared` pending from its entry.
109    Publish {
110        /// The thread, the sender itself.
111        thread: DetTid,
112        /// The temporary mask.
113        mask: u64,
114        /// `SigPnd_entry ∩ I ∩ own`.
115        held_private: u64,
116        /// `ShdPnd_entry ∩ I ∩ own`.
117        held_shared: u64,
118    },
119    /// `thread`'s call returned with its publication still in force: answer
120    /// and close the committed-send window, and put back what was published
121    /// before it, unless `keep_published`.
122    Withdraw {
123        /// The thread, the sender itself.
124        thread: DetTid,
125        /// The call ended interrupted (`-ERESTARTNOHAND` or `-EINTR`): Linux
126        /// does not restore the own mask at the return
127        /// (`restore_saved_sigmask_unless`, fs/select.c:301,
128        /// fs/eventpoll.c:2495 and :2547 at v6.12); the delivery then sets
129        /// `blocked` to the temporary mask with the handler's `sa_mask` and
130        /// the signal (`signal_delivered`, kernel/signal.c:2951-2954), and
131        /// the temporary mask leaves `blocked` only at `rt_sigreturn`, which
132        /// restores the own mask saved in the frame. So the temporary mask
133        /// stays published until the thread's next request; step 3c.1e B/C's
134        /// in-guest delivery must file one before the handler's code runs,
135        /// so that the published mask is the handler's.
136        keep_published: bool,
137    },
138}

140/// The sends the scheduler committed to a thread while its temporary-mask
141/// publication was in force, by queue, as signal bits: the answer to a
142/// withdrawal. It travels in a bincode RPC, so it has no
143/// `skip_serializing_if`.
144#[derive(Debug, Clone, Copy, Default, PartialEq, Eq, Serialize, Deserialize)]
145pub struct CommittedSendWindow {
146    /// Thread-directed sends (the private queue).
147    pub sent_private: u64,
148    /// Process-directed sends (the shared queue).
149    pub sent_shared: u64,
150}
151
152/// One temporary-mask publication in force (`Scheduler::temporary_mask_publications`).
153#[derive(Debug, Clone, Copy, Default, PartialEq, Eq)]
154pub(crate) struct TemporaryMaskPublication {
155    /// What the thread had published before it, put back by a withdrawal.
156    prior: Option<PublishedGuestMask>,
157    /// Whether that was a temporary mask (`Scheduler::temporary_guest_sigmasks`).
158    prior_temporary: bool,
159    /// D37's held masks.
160    held_private: u64,
161    held_shared: u64,
162    /// The committed-send window record.
163    window: CommittedSendWindow,
164}
165
166impl Scheduler {
167    /// What the scheduler holds as `dettid`'s published guest mask, tri-state
168    /// (IR-18): `Known`, `Unknown`, or `None` for nothing published.
169    pub(crate) fn published_guest_mask_state(&self, dettid: DetTid) -> Option<PublishedGuestMask> {
170        match self.saved_guest_sigmasks.get(&dettid) {
171            Some(mask) => Some(PublishedGuestMask::Known(*mask)),
172            None if self.unknown_guest_sigmasks.contains(&dettid) => {
173                Some(PublishedGuestMask::Unknown)
174            }
175            None => None,
176        }
177    }
178
179    /// The answer to a `/proc` status reader under the gate
180    /// ([`StatusGuestMask`]).
181    pub(crate) fn status_guest_mask(&self, dettid: DetTid) -> StatusGuestMask {
182        let held = self.temporary_mask_publications.get(&dettid);
183        StatusGuestMask {
184            mask: self.published_guest_mask_state(dettid),
185            held_private: held.map_or(0, |held| held.held_private),
186            held_shared: held.map_or(0, |held| held.held_shared),
187        }
188    }
189
190    /// Make `dettid`'s published mask `mask` (`None`: nothing), whatever it
191    /// was, `Unknown` included.
192    pub(super) fn put_guest_mask_state(
193        &mut self,
194        dettid: DetTid,
195        mask: Option<PublishedGuestMask>,
196    ) {
197        match mask {
198            Some(PublishedGuestMask::Known(mask)) => {
199                self.saved_guest_sigmasks.insert(dettid, mask);
200                self.unknown_guest_sigmasks.remove(&dettid);
201            }
202            Some(PublishedGuestMask::Unknown) => {
203                self.saved_guest_sigmasks.remove(&dettid);
204                self.unknown_guest_sigmasks.insert(dettid);
205            }
206            None => {
207                self.saved_guest_sigmasks.remove(&dettid);
208                self.unknown_guest_sigmasks.remove(&dettid);
209            }
210        }
211    }
212
213    /// [`Self::put_guest_mask_state`], for tests outside the scheduler.
214    #[cfg(test)]
215    pub(crate) fn put_guest_mask_state_for_test(
216        &mut self,
217        dettid: DetTid,
218        mask: Option<PublishedGuestMask>,
219    ) {
220        self.put_guest_mask_state(dettid, mask);
221    }

223    /// IR-18: `dettid` is readmitted on a rejoin that returned under the
224    /// guest's own mask (`ContinuationMask::Restored`: no temporary mask, and
225    /// not unknown) but published none. The mask its call published, possibly
226    /// a temporary one, no longer holds. Where the backend keeps the guest's
227    /// mask apart from the kernel's for this thread (it has published its own
228    /// mask at a request, `own_mask_publishers`), the scheduler does not know
229    /// the mask Linux restored: `Unknown` (the `/proc` status reader refuses,
230    /// defect-only once step 3c.1e B publishes at every request). Otherwise,
231    /// as on every backend before B (none implements
232    /// `Guest::resume_signal_mask`), the kernel's `SigBlk` is the guest's own
233    /// mask: nothing is published, so readers take it, as for any request
234    /// that publishes nothing. A failed vfork's rejoin (`VforkFailed`) takes
235    /// the same rule. Under the gate only.
236    pub(super) fn restored_rejoin_published_nothing(&mut self, dettid: DetTid) {
237        if !self.in_guest_signal_delivery || !self.next_turns.contains_key(&dettid) {
238            return;
239        }
240        self.temporary_guest_sigmasks.remove(&dettid);
241        let mask = self.unknown_unless_kernel_shows_own(dettid);
242        self.put_guest_mask_state(dettid, mask);
243    }
244
245    /// What to publish for `dettid` when the scheduler does not hold the
246    /// guest's own mask for it: `Unknown` where its backend keeps the guest's
247    /// mask apart from the kernel's (it has published its own mask at a
248    /// request, `own_mask_publishers`), and otherwise nothing, so readers
249    /// take the kernel's `SigBlk`, which is then the guest's own mask.
250    fn unknown_unless_kernel_shows_own(&self, dettid: DetTid) -> Option<PublishedGuestMask> {
251        self.own_mask_publishers
252            .contains(&dettid)
253            .then_some(PublishedGuestMask::Unknown)
254    }
255
256    /// Fork-child seeding, under the gate only: `child`, just registered by
257    /// `parent`'s `clone`, `fork` or `vfork` in `parent`'s turn, starts with
258    /// the guest's own mask as `parent` published it, `Known` or `Unknown`
259    /// (nothing when `parent` published nothing, so `child`'s readers take
260    /// its `SigBlk`, which the kernel copied from `parent`'s). Linux's child
261    /// starts with its creator's `blocked` (`arch_dup_task_struct`,
262    /// kernel/fork.c:1086-1091 at v6.12, from `dup_task_struct`, :1112) and
263    /// an empty `pending` (:2268); its first request then publishes its own.
264    /// A temporary mask `parent` still publishes
265    /// (`Scheduler::temporary_guest_sigmasks`: a masked call's, which Linux
266    /// takes down before the call returns, or, after an interrupted end,
267    /// keeps through the signal's delivery together with the handler's
268    /// `sa_mask` and the signal, `signal_delivered`, kernel/signal.c:2951-2954,
269    /// until `rt_sigreturn`) is not the mask `parent` holds at the `fork`, and
270    /// is not copied: `child` takes what a rejoin that publishes nothing takes
271    /// ([`Self::unknown_unless_kernel_shows_own`]). An ordinary clone first
272    /// publishes the creator's current mask (`tool_global::create_child_thread`),
273    /// which clears the mark, so this applies only where that publishes
274    /// nothing.
275    pub(crate) fn seed_child_guest_sigmask(&mut self, parent: DetTid, child: DetTid) {
276        if !self.in_guest_signal_delivery || parent == child {
277            return;
278        }
279        let inherited = if self.temporary_guest_sigmasks.contains(&parent) {
280            self.unknown_unless_kernel_shows_own(parent)
281        } else {
282            self.published_guest_mask_state(parent)
283        };
284        self.temporary_mask_publications.remove(&child);
285        self.temporary_guest_sigmasks.remove(&child);
286        self.put_guest_mask_state(child, inherited);
287    }

289    /// IR-23(a), the answer to `GlobalRequest::PublishTemporaryMask`, in the
290    /// sender's own turn, with no scheduling decision. Under the gate only;
291    /// `None` with it off or for a thread the scheduler does not know, and
292    /// for a withdrawal with no publication in force (a later request
293    /// superseded it).
294    pub(crate) fn temporary_mask_control(
295        &mut self,
296        sender: DetTid,
297        control: TemporaryMaskControl,
298    ) -> Option<CommittedSendWindow> {
299        match control {
300            TemporaryMaskControl::Publish {
301                thread,
302                mask,
303                held_private,
304                held_shared,
305            } if thread == sender => {
306                self.publish_temporary_mask(thread, mask, held_private, held_shared);
307                None
308            }
309            TemporaryMaskControl::Withdraw {
310                thread,
311                keep_published,
312            } if thread == sender => self.withdraw_temporary_mask(thread, keep_published),
313            _ => None,
314        }
315    }
316
317    /// `thread` waits under the temporary mask `mask` from now, keeping the
318    /// ignored signals `held_private` and `held_shared` pending from its
319    /// entry: the published mask becomes `mask` (readers, the `/proc` status
320    /// reader included, take it in place of what the thread's latest request
321    /// published), the held masks are kept for the status reader, and the
322    /// committed-send window opens. Under the gate only. Returns whether the
323    /// publication was taken.
324    pub(crate) fn publish_temporary_mask(
325        &mut self,
326        thread: DetTid,
327        mask: u64,
328        held_private: u64,
329        held_shared: u64,
330    ) -> bool {
331        if !self.in_guest_signal_delivery || !self.next_turns.contains_key(&thread) {
332            return false;
333        }
334        let (prior, prior_temporary) = match self.temporary_mask_publications.get(&thread) {
335            // A second publication in the same call keeps the first one's
336            // prior (the call files the pair once, so this does not happen).
337            Some(publication) => (publication.prior, publication.prior_temporary),
338            None => (
339                self.published_guest_mask_state(thread),
340                self.temporary_guest_sigmasks.contains(&thread),
341            ),
342        };
343        self.put_guest_mask_state(thread, Some(PublishedGuestMask::Known(mask)));
344        self.temporary_guest_sigmasks.insert(thread);
345        self.temporary_mask_publications.insert(
346            thread,
347            TemporaryMaskPublication {
348                prior,
349                prior_temporary,
350                held_private,
351                held_shared,
352                window: CommittedSendWindow::default(),
353            },
354        );
355        true
356    }
357
358    /// `thread`'s masked call returned with its temporary-mask publication
359    /// still in force (it filed no request since): the committed-send window
360    /// is answered and closed, with the held masks, and what was published
361    /// before it is put back, as Linux restores the thread's own mask as the
362    /// call returns (`restore_saved_sigmask_unless`, fs/select.c:301,
363    /// fs/eventpoll.c:2495 and :2547 at v6.12). On an interrupted end
364    /// (`keep_published`) Linux does not restore it at the return, and the
365    /// delivery keeps the temporary mask in `blocked` with the handler's
366    /// `sa_mask` and the signal until `rt_sigreturn` (`signal_delivered`,
367    /// kernel/signal.c:2951-2954), so it stays published until the thread's
368    /// next request.
369    /// `None` when no publication is in force.
370    pub(crate) fn withdraw_temporary_mask(
371        &mut self,
372        thread: DetTid,
373        keep_published: bool,
374    ) -> Option<CommittedSendWindow> {
375        let publication = self.temporary_mask_publications.remove(&thread)?;
376        if !keep_published {
377            self.put_guest_mask_state(thread, publication.prior);
378            if !publication.prior_temporary {
379                self.temporary_guest_sigmasks.remove(&thread);
380            }
381        }
382        Some(publication.window)
383    }
384
385    /// A later publication by `thread` (a request it filed, or an explicit
386    /// `SetSavedGuestSigmask`) supersedes its temporary-mask publication: the
387    /// held masks and the window go with it, and nothing is put back.
388    pub(super) fn forget_temporary_mask_publication(&mut self, thread: DetTid) {
389        self.temporary_mask_publications.remove(&thread);
390    }

392    /// The committed-send window record: a send of `signal` the scheduler has
393    /// just committed to `target`, thread-directed (`private`) or
394    /// process-directed, from one of its two current commit points
395    /// (`NotifySignalPending`'s handler, after the guest's injected send
396    /// succeeded; `send_guest_signal`, after its `kill` succeeded). Recorded
397    /// only while `target` holds a temporary-mask publication; nothing is
398    /// kept, or even looked up, with the gate off.
399    pub(crate) fn note_committed_send_in_window(
400        &mut self,
401        target: DetTid,
402        signal: i32,
403        private: bool,
404    ) {
405        if !self.in_guest_signal_delivery {
406            return;
407        }
408        let Some(publication) = self.temporary_mask_publications.get_mut(&target) else {
409            return;
410        };
411        let bit = super::kernel_signal_bit(signal);
412        if private {
413            publication.window.sent_private |= bit;
414        } else {
415            publication.window.sent_shared |= bit;
416        }
417    }
