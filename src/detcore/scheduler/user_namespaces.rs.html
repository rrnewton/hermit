9//! The user namespace each guest process lives in, relative to the
10//! container's, as Detcore's record of how each process was made.
11//!
12//! Linux puts a new process in its creator's user namespace, or, for
13//! `clone(CLONE_NEWUSER)` (and `clone3` with that flag), in a new namespace
14//! whose parent is the creator's (`copy_creds`, `create_user_ns`). Linux has
15//! two other ways into a user namespace, `unshare(CLONE_NEWUSER)` and
16//! `setns`, and Detcore refuses both (EPERM,
17//! `syscall_classification::is_mount_ns_admin_refused_syscall`), so the clone
18//! is the only way in, and the record, kept in the creator's clone turn, is
19//! complete. An exec does not change a process's namespace.
20//!
21//! A namespace is named by the number the record gives it when a clone makes
22//! it, never reused; the container's namespace has none. A process's chain is
23//! the namespaces it lives in, outermost first: its creator's chain, plus the
24//! new namespace for `CLONE_NEWUSER`. The record keeps a process's chain until
25//! a later process with the same number replaces it, so a process that has
26//! exited and is not yet reaped still has its chain (Linux keeps its
27//! credentials, and their namespace, until it is released).
28//!
29//! What it answers: whether a task lives in the reader's namespace, below it,
30//! or elsewhere ([`UserNamespaceRelation`]), which Linux's
31//! `ptrace_may_access` turns on for a reader without `CAP_SYS_PTRACE` in its
32//! effective set (`Detcore::io_read_allowed_by_number`). Linux's rule also
33//! asks who owns the namespace just below the reader's on the task's chain
34//! (`cap_capable`'s owner rule, which asks about no other namespace): the
35//! effective user ID of the task that made it. The record keeps, for each
36//! namespace, whether its creator's effective user ID may not have been its
37//! real one ([`UserNamespaces::note_owner_may_differ`]), as the creator
38//! reported it before its clone, noted only once the clone has made the
39//! namespace.
40
41use std::collections::BTreeMap;
42use std::collections::BTreeSet;
43
44use crate::types::DetPid;
45
46/// Where a task's user namespace is, from a reader's.
47#[derive(
48    Debug,
49    Clone,
50    Copy,
51    PartialEq,
52    Eq,
53    serde::Serialize,
54    serde::Deserialize
55)]
56pub enum UserNamespaceRelation {
57    /// The task lives in the reader's own namespace.
58    Same,
59    /// The task lives in a namespace below the reader's: the reader's is an
60    /// ancestor of it.
61    Below,
62    /// Anywhere else: above the reader's, or in another branch.
63    Elsewhere,
64}
65
66/// The record (module documentation).
67#[derive(Debug, Default, Clone)]
68pub struct UserNamespaces {
69    /// Each process below the container's namespace: the namespaces it lives
70    /// in, outermost first. A process with no entry lives in the container's.
71    chains: BTreeMap<DetPid, Vec<u64>>,
72    /// How many namespaces the record has named.
73    named: u64,
74    /// The namespaces a task made whose effective user ID was not its real
75    /// user ID, read as the overflow ID, or could not be read: such a
76    /// namespace's owner may not be the effective user ID of a reader whose
77    /// effective ID is its real one.
78    owners_may_differ: BTreeSet<u64>,
79}
80
81impl UserNamespaces {
82    /// A new process `child` starts in its creator's namespace (`creator`;
83    /// `None` for the run's first process, which lives in the container's).
84    /// A process that reuses an earlier one's number replaces its chain.
85    pub fn inherit(&mut self, child: DetPid, creator: Option<DetPid>) {
86        let chain = creator
87            .and_then(|creator| self.chains.get(&creator))
88            .cloned()
89            .unwrap_or_default();
90        if chain.is_empty() {
91            self.chains.remove(&child);
92        } else {
93            self.chains.insert(child, chain);
94        }
95    }
96
97    /// `process`, which `clone(CLONE_NEWUSER)` just made, lives in a new
98    /// namespace below the one it inherited ([`Self::inherit`]).
99    pub fn enter_new(&mut self, process: DetPid) {
100        self.named += 1;
101        let named = self.named;
102        self.chains.entry(process).or_default().push(named);
103    }
104
105    /// Where `task`'s namespace is from `reader`'s.
106    pub fn relation(&self, reader: DetPid, task: DetPid) -> UserNamespaceRelation {
107        let reader = self.chain(reader);
108        let task = self.chain(task);
109        if reader == task {
110            UserNamespaceRelation::Same
111        } else if task.starts_with(reader) {
112            UserNamespaceRelation::Below
113        } else {
114            UserNamespaceRelation::Elsewhere
115        }
116    }
117
118    /// Record that the namespace `process`'s clone just made (the last of its
119    /// chain, [`Self::enter_new`]) was made by a task whose effective user ID
120    /// was not its real one, read as the overflow ID, or could not be read:
121    /// its owner may not be the real user ID every guest task has. Nothing
122    /// for a process the record has below no namespace of its own.
123    pub fn note_owner_may_differ(&mut self, process: DetPid) {
124        if let Some(&newest) = self.chain(process).last() {
125            self.owners_may_differ.insert(newest);
126        }
127    }
128
129    /// Whether the namespace whose owner Linux's rule asks about for a
130    /// `reader` and a `task` below it (the one just below the reader's on the
131    /// task's chain) may be owned by an effective user ID other than the real
132    /// user ID every guest task has ([`Self::note_owner_may_differ`]). False
133    /// where the task is not below the reader, whose answer turns on no
134    /// owner.
135    pub fn owner_may_differ_below(&self, reader: DetPid, task: DetPid) -> bool {
136        let reader = self.chain(reader);
137        let task = self.chain(task);
138        task.len() > reader.len()
139            && task.starts_with(reader)
140            && self.owners_may_differ.contains(&task[reader.len()])
141    }
142
143    fn chain(&self, process: DetPid) -> &[u64] {
144        self.chains.get(&process).map_or(&[], Vec::as_slice)
145    }
146}

148#[cfg(test)]
149mod tests {
150    use super::*;
151
152    fn pid(raw: i32) -> DetPid {
153        DetPid::from_raw(raw)
154    }
155
156    #[test]
157    fn a_process_lives_where_its_creator_does_unless_its_clone_made_a_namespace() {
158        use UserNamespaceRelation::*;
159        let mut record = UserNamespaces::default();
160        record.inherit(pid(3), None);
161        // A plain fork: the creator's namespace.
162        record.inherit(pid(4), Some(pid(3)));
163        assert_eq!(record.relation(pid(3), pid(4)), Same);
164        // A clone(CLONE_NEWUSER): a namespace below the creator's.
165        record.inherit(pid(5), Some(pid(3)));
166        record.enter_new(pid(5));
167        assert_eq!(record.relation(pid(3), pid(5)), Below);
168        assert_eq!(record.relation(pid(4), pid(5)), Below);
169        assert_eq!(record.relation(pid(5), pid(3)), Elsewhere);
170        // Its child, forked plainly, shares it; one made with
171        // CLONE_NEWUSER is below both.
172        record.inherit(pid(6), Some(pid(5)));
173        assert_eq!(record.relation(pid(5), pid(6)), Same);
174        record.inherit(pid(7), Some(pid(6)));
175        record.enter_new(pid(7));
176        assert_eq!(record.relation(pid(3), pid(7)), Below);
177        assert_eq!(record.relation(pid(6), pid(7)), Below);
178        // Two namespaces made from the same one are apart.
179        record.inherit(pid(8), Some(pid(3)));
180        record.enter_new(pid(8));
181        assert_eq!(record.relation(pid(5), pid(8)), Elsewhere);
182        assert_eq!(record.relation(pid(8), pid(7)), Elsewhere);
183        // A process the record never saw made lives in the container's.
184        assert_eq!(record.relation(pid(3), pid(99)), Same);
185        assert_eq!(record.relation(pid(99), pid(5)), Below);
186    }
187
188    #[test]
189    fn a_reused_number_takes_its_new_creators_namespace() {
190        use UserNamespaceRelation::*;
191        let mut record = UserNamespaces::default();
192        record.inherit(pid(3), None);
193        record.inherit(pid(5), Some(pid(3)));
194        record.enter_new(pid(5));
195        // Process 5 is reaped and a plain fork of 3 takes its number.
196        record.inherit(pid(5), Some(pid(3)));
197        assert_eq!(record.relation(pid(3), pid(5)), Same);
198        // A namespace made later gets a new name: it is not the old one.
199        record.inherit(pid(6), Some(pid(3)));
200        record.enter_new(pid(6));
201        record.inherit(pid(9), Some(pid(3)));
202        record.enter_new(pid(9));
203        assert_eq!(record.relation(pid(6), pid(9)), Elsewhere);
204    }
205
206    #[test]
207    fn an_owner_note_is_kept_for_the_one_namespace_linux_asks_about() {
208        let mut record = UserNamespaces::default();
209        record.inherit(pid(3), None);
210        // 5 makes a namespace below the container's; its creator's owner is
211        // noted as one that may differ.
212        record.inherit(pid(5), Some(pid(3)));
213        record.enter_new(pid(5));
214        record.note_owner_may_differ(pid(5));
215        // 7, below 5's, was made by a creator whose owner is not noted.
216        record.inherit(pid(7), Some(pid(5)));
217        record.enter_new(pid(7));
218        // From 3, Linux asks about 5's namespace for both 5 and 7.
219        assert!(record.owner_may_differ_below(pid(3), pid(5)));
220        assert!(record.owner_may_differ_below(pid(3), pid(7)));
221        // From 5, it asks about 7's, whose owner is not noted.
222        assert!(!record.owner_may_differ_below(pid(5), pid(7)));
223        // A namespace made elsewhere is not touched by 5's note.
224        record.inherit(pid(8), Some(pid(3)));
225        record.enter_new(pid(8));
226        assert!(!record.owner_may_differ_below(pid(3), pid(8)));
227        // Not below: no owner is asked about.
228        assert!(!record.owner_may_differ_below(pid(5), pid(3)));
229        assert!(!record.owner_may_differ_below(pid(5), pid(5)));
230        // A process below no namespace of its own notes nothing.
231        record.note_owner_may_differ(pid(3));
232        assert!(!record.owner_may_differ_below(pid(3), pid(8)));
233    }
