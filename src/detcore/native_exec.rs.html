9//! Inactive, backend-neutral validation of a prepared single-thread exec cut.
10//!
11//! These APIs inspect actual Detcore data. They do not authorize kernel exec,
12//! observe a physical task, install a backend, or run a lifecycle callback.
13//! The caller must own the quiescent dispatch and separately prove the admitted
14//! one-main-thread/childless domain, physical identity and registrations.
15//! In particular, probing an Arc's lock does not establish that no other task
16//! owns it. No ordinary exec path calls this module.
17//!
18//! The transport consumer must bound the payload before allocation/decode,
19//! authenticate its exact bytes and Config, and reject trailing bytes. It must
20//! retain the same live cut through its final physical exec edge. Deserializing
21//! serde-skipped capabilities cannot reconstruct them; the live producer must
22//! validate them before preparing the replacement table.
23

46/// Version of the seven-field Detcore prepared-exec payload.
47pub const DETCORE_EXEC_SNAPSHOT_VERSION: u16 = 2;

49/// Maximum encoded snapshot size enforced by the concrete transport consumer.
50pub const MAX_SNAPSHOT_BYTES: usize = 1024 * 1024;

52/// A data validation refusal, not a backend errno or admission capability.
53#[derive(Clone, Copy, Debug, Eq, PartialEq)]
54pub enum NativeExecCutError {
55    /// The supplied binding or a Tool/state/header identity disagrees.
56    Identity(&'static str),
57    /// The immutable configuration is outside the named snapshot domain.
58    Configuration(&'static str),
59    /// Actual embedded Config bytes disagree with the admitted digest.
60    ConfigDigest,
61    /// Existing Config serialization failed.
62    ConfigEncoding,
63    /// A skipped runtime capability or live operation remains.
64    LiveCapability(&'static str),
65    /// A supposedly quiescent model is locked or poisoned.
66    LockUnavailable(&'static str),
67    /// The replacement state was not prepared as the existing exec path does.
68    PreparedState(&'static str),
69    /// A table slot or its within-table alias identity is inconsistent.
70    Descriptor(&'static str),
71    /// The decoded payload version is unsupported.
72    Version,
73}
74
75impl fmt::Display for NativeExecCutError {
76    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
77        let (category, field) = match *self {
78            Self::Identity(field) => ("identity mismatch", field),
79            Self::Configuration(field) => ("unsupported configuration", field),
80            Self::ConfigDigest => ("Config digest mismatch", "Config"),
81            Self::ConfigEncoding => ("Config encoding failed", "Config"),
82            Self::LiveCapability(field) => ("live runtime capability", field),
83            Self::LockUnavailable(field) => ("model locked or poisoned", field),
84            Self::PreparedState(field) => ("unprepared replacement state", field),
85            Self::Descriptor(field) => ("descriptor mismatch", field),
86            Self::Version => ("unsupported snapshot version", "version"),
87        };
88        write!(f, "native exec cut {category}: {field}")
89    }
90}

94/// Bind the genuine initial physical leader before its first thread-start.
95/// This does not initialize deterministic process identity or run a callback.
96///
97/// # Safety
98/// The caller owns the actual authenticated InitialImage lifecycle and passes
99/// PID/TID read from that same current kernel task through its qualified gate.
100/// No guest work has run; this is the unique freshly constructed initial state.
101pub unsafe fn bind_native_initial_physical_identity(
102    thread: &mut ThreadState<()>,
103    physical_pid: i32,
104    physical_tid: i32,
105) -> Result<(), NativeExecCutError> {
106    if physical_tid <= 0
107        || physical_pid != physical_tid
108        || thread.dettid.as_raw() != physical_tid
109        || thread.physical_tid.is_some()
110        || thread.detpid.is_some()
111        || thread.thread_start_entered
112        || thread.mm_id != MmId::initial(thread.dettid)
113        || try_native_exec_lock(&thread.file_metadata, "initial file_metadata")?
114            .native_exec_files_id()
115            != FilesId::initial(thread.dettid)
116    {
117        return Err(NativeExecCutError::Identity("initial physical state"));
118    }
119    if thread.signal_task_identity.is_some() {
120        return Err(NativeExecCutError::LiveCapability("signal_task_identity"));
121    }
122    thread.physical_tid = Some(physical_tid);
123    Ok(())
124}

126/// Rebind from the actual fresh Guest answer before native post-exec.
127/// The first domain has no admitted signal-control capability; equal None is
128/// supported data, never a fabricated task-lifetime identity.
129pub fn bind_native_fresh_signal_identity(
130    thread: &mut ThreadState<()>,
131    mode: reverie::BackendSignalControlMode,
132    actual: Option<reverie::SignalTaskIdentity>,
133) -> Result<(), NativeExecCutError> {
134    if mode != reverie::BackendSignalControlMode::Unchanged
135        || thread.signal_task_identity.is_some()
136        || actual.is_some()
137    {
138        return Err(NativeExecCutError::LiveCapability("signal_task_identity"));
139    }
140    thread.signal_task_identity = actual;
141    Ok(())
142}

144/// Independently supplied identities of the existing prepared transaction.
145///
146/// This is data, not proof of watched birth, physical state, or childlessness.
147/// The old mm/table identities must come from the live owned transaction,
148/// never from the payload being validated.
149#[derive(Clone, Copy, Debug, Eq, PartialEq)]
150pub struct NativeExecCutBinding {
151    /// Deterministic main thread identity.
152    pub dettid: DetTid,
153    /// Continuing deterministic process identity; equal to dettid here.
154    pub detpid: DetPid,
155    /// Actual continuing physical main-thread ID proved by the backend.
156    pub physical_tid: i32,
157    /// Live old-image memory incarnation.
158    pub old_mm: MmId,
159    /// Live old-image descriptor-table incarnation.
160    pub old_files: FilesId,
161    /// Independently admitted SHA-256 of the unchanged legacy wire Config.
162    pub config_digest: [u8; 32],
163}
164
165impl NativeExecCutBinding {
166    fn validate(&self) -> Result<(), NativeExecCutError> {
167        if self.dettid != self.detpid || self.detpid.as_raw() <= 0 {
168            return Err(NativeExecCutError::Identity("main-thread IDs"));
169        }
170        if self.physical_tid <= 0 {
171            return Err(NativeExecCutError::Identity("physical_tid"));
172        }
173        if self.old_mm.generation() == u64::MAX || self.old_files.generation() == u64::MAX {
174            return Err(NativeExecCutError::Identity("incarnation overflow"));
175        }
176        Ok(())
177    }
178
179    /// Replacement mm produced by the existing same-leader exec transition.
180    pub fn prepared_mm(&self) -> MmId {
181        self.old_mm.for_exec(self.detpid)
182    }
183
184    /// Replacement descriptor-table identity produced by the existing path.
185    pub fn prepared_files(&self) -> FilesId {
186        self.old_files.for_exec(self.dettid)
187    }
188}

190/// Dispatch-owned evidence that the actual old table was inspected.
191///
192/// This deliberately implements neither Serialize/Deserialize nor Clone.
193/// Retaining the old table also lets the prepared cut recheck its skipped
194/// capabilities, including descriptions whose last slot was CLOEXEC.
195/// It is not a global admission or physical-exec authority.
196#[derive(Debug)]
197pub struct NativeExecPreparationReceipt {
198    binding: NativeExecCutBinding,
199    old_table: Arc<Mutex<FileMetadata>>,
200}

202/// A borrowed encoding of the actual already-prepared Tool and main state.
203///
204/// Field order and representation match the owned version. No new state is
205/// constructed and no PRNG, accounting or clock observation is performed.
206/// The receipt borrow keeps its old-table ownership alive during encoding.
207/// The caller must still keep all interior model data quiescent.
208#[derive(Debug, Serialize)]
209pub struct DetcoreExecSnapshotRefV2<'a> {
210    version: u16,
211    dettid: DetTid,
212    detpid: DetPid,
213    prepared_mm: MmId,
214    config_digest: [u8; 32],
215    tool: &'a Detcore<NoopTool>,
216    thread: &'a ThreadState<()>,
217    #[serde(skip)]
218    _receipt: &'a NativeExecPreparationReceipt,
219}
220
221/// Owned data decoded only after exact bounded transport authentication.
222///
223/// Deserialization restores the existing within-table OFD aliases. It cannot
224/// prove that skipped producer capabilities were absent before serialization.
225/// Use `into_validated_parts` before attaching this data to a replacement host.
226#[derive(Debug, Serialize, Deserialize)]
227pub struct DetcoreExecSnapshotV2 {
228    version: u16,
229    dettid: DetTid,
230    detpid: DetPid,
231    prepared_mm: MmId,
232    config_digest: [u8; 32],
233    tool: Detcore<NoopTool>,
234    thread: ThreadState<()>,
235}

237impl DetcoreExecSnapshotV2 {
238    /// Check the decoded header and actual state against independent admission.
239    ///
240    /// The expected Config is independently obtained from the admitted run.
241    /// This leaves every field and capability untouched. Backend incarnation
242    /// rebinding (including signal_task_identity) remains the host's explicit
243    /// responsibility before any restored lifecycle use.
244    pub fn validate(
245        &self,
246        binding: &NativeExecCutBinding,
247        expected_config: &Config,
248    ) -> Result<(), NativeExecCutError> {
249        binding.validate()?;
250        if self.version != DETCORE_EXEC_SNAPSHOT_VERSION {
251            return Err(NativeExecCutError::Version);
252        }
253        if self.dettid != binding.dettid || self.detpid != binding.detpid {
254            return Err(NativeExecCutError::Identity("snapshot IDs"));
255        }
256        if self.prepared_mm != binding.prepared_mm() {
257            return Err(NativeExecCutError::Identity("snapshot prepared_mm"));
258        }
259        // The admitted host Config can retain host-only output paths. Its
260        // omitted semantic fields must nevertheless be checked before wire
261        // equality loses them.
262        validate_native_exec_host_configuration(expected_config)?;
263        if self.config_digest != binding.config_digest
264            || native_exec_config_digest(expected_config)? != binding.config_digest
265        {
266            return Err(NativeExecCutError::ConfigDigest);
267        }
268        validate_tool_and_thread(&self.tool, &self.thread, binding)?;
269        self.thread.validate_prepared_native_exec_cut(binding)
270    }
271
272    /// Consume this payload only after all neutral validation succeeds.
273    pub fn into_validated_parts(
274        self,
275        binding: &NativeExecCutBinding,
276        expected_config: &Config,
277    ) -> Result<(Detcore<NoopTool>, ThreadState<()>), NativeExecCutError> {
278        self.validate(binding, expected_config)?;
279        Ok((self.tool, self.thread))
280    }
281}

283/// Validate actual old state before `FileMetadata::for_exec` drops capabilities.
284///
285/// This is an explicit inactive API; it is not an added ordinary syscall hook.
286pub fn validate_old_native_exec_cut(
287    tool: &Detcore<NoopTool>,
288    thread: &ThreadState<()>,
289    binding: &NativeExecCutBinding,
290) -> Result<NativeExecPreparationReceipt, NativeExecCutError> {
291    binding.validate()?;
292    validate_tool_and_thread(tool, thread, binding)?;
293    if thread.mm_id != binding.old_mm {
294        return Err(NativeExecCutError::Identity("old mm"));
295    }
296    {
297        let table = try_native_exec_lock(&thread.file_metadata, "old file_metadata")?;
298        table.validate_native_exec_cut(binding.old_files, false)?;
299    }
300    thread.validate_old_native_exec_robust_state()?;
301    Ok(NativeExecPreparationReceipt {
302        binding: *binding,
303        old_table: Arc::clone(&thread.file_metadata),
304    })
305}

307/// Borrow the actual prepared state, bound to its live old-table receipt.
308///
309/// Both old and prepared tables are checked without comparing their total Arc
310/// counts: the existing rollback future legitimately owns old-table aliases.
311pub fn capture_prepared_native_exec<'a>(
312    tool: &'a Detcore<NoopTool>,
313    thread: &'a ThreadState<()>,
314    binding: &NativeExecCutBinding,
315    receipt: &'a NativeExecPreparationReceipt,
316) -> Result<DetcoreExecSnapshotRefV2<'a>, NativeExecCutError> {
317    binding.validate()?;
318    if receipt.binding != *binding {
319        return Err(NativeExecCutError::Identity("preparation receipt"));
320    }
321    {
322        let old = try_native_exec_lock(&receipt.old_table, "old file_metadata")?;
323        old.validate_native_exec_cut(binding.old_files, false)?;
324    }
325    if Arc::ptr_eq(&receipt.old_table, &thread.file_metadata) {
326        return Err(NativeExecCutError::PreparedState(
327            "replacement table ownership",
328        ));
329    }
330    validate_tool_and_thread(tool, thread, binding)?;
331    thread.validate_prepared_native_exec_cut(binding)?;
332    Ok(DetcoreExecSnapshotRefV2 {
333        version: DETCORE_EXEC_SNAPSHOT_VERSION,
334        dettid: binding.dettid,
335        detpid: binding.detpid,
336        prepared_mm: binding.prepared_mm(),
337        config_digest: binding.config_digest,
338        tool,
339        thread,
340        _receipt: receipt,
341    })
342}

344/// Compare the final actual prepared state directly against already sealed
345/// bytes, using the unchanged legacy serializer without a temporary encoding.
346/// The caller retains the same exclusive dispatch and old-table receipt.
347/// This checks the complete encoding, including exact length, and never resets
348/// accounting, advances a PRNG, or constructs a replacement model.
349pub fn check_prepared_native_exec_bytes(
350    tool: &Detcore<NoopTool>,
351    thread: &ThreadState<()>,
352    binding: &NativeExecCutBinding,
353    receipt: &NativeExecPreparationReceipt,
354    expected: &[u8],
355) -> Result<(), NativeExecCutError> {
356    if expected.is_empty() || expected.len() > MAX_SNAPSHOT_BYTES {
357        return Err(NativeExecCutError::PreparedState("sealed snapshot bounds"));
358    }
359    let live = capture_prepared_native_exec(tool, thread, binding, receipt)?;
360    compare_encoding(&live, expected)
361}
362
363fn compare_encoding<T: Serialize + ?Sized>(
364    live: &T,
365    expected: &[u8],
366) -> Result<(), NativeExecCutError> {
367    struct CompareWriter<'a> {
368        expected: &'a [u8],
369        position: usize,
370        failed: bool,
371    }
372    impl io::Write for CompareWriter<'_> {
373        fn write(&mut self, bytes: &[u8]) -> io::Result<usize> {
374            let end = self.position.checked_add(bytes.len());
375            if self.failed
376                || end.is_none_or(|end| self.expected.get(self.position..end) != Some(bytes))
377            {
378                self.failed = true;
379                return Err(io::ErrorKind::InvalidData.into());
380            }
381            self.position = end.unwrap();
382            Ok(bytes.len())
383        }
384        fn flush(&mut self) -> io::Result<()> {
385            Ok(())
386        }
387    }
388    let mut writer = CompareWriter {
389        expected,
390        position: 0,
391        failed: false,
392    };
393    let result =
394        bincode::serde::encode_into_std_write(live, &mut writer, bincode::config::legacy());
395    if result.is_err() || writer.failed || writer.position != expected.len() {
396        return Err(NativeExecCutError::PreparedState("sealed snapshot changed"));
397    }
398    Ok(())
399}

401fn validate_tool_and_thread(
402    tool: &Detcore<NoopTool>,
403    thread: &ThreadState<()>,
404    binding: &NativeExecCutBinding,
405) -> Result<(), NativeExecCutError> {
406    if tool.detpid != binding.detpid {
407        return Err(NativeExecCutError::Identity("Tool detpid"));
408    }
409    validate_native_exec_guest_configuration(&tool.cfg)?;
410    if native_exec_config_digest(&tool.cfg)? != binding.config_digest {
411        return Err(NativeExecCutError::ConfigDigest);
412    }
413    thread.validate_native_exec_cut(binding, &tool.cfg)
414}

416/// SHA-256 of the actual unchanged Config in the existing legacy bincode form.
417///
418/// This is a neutral identity operation, not transport framing or admission.
419/// It streams into the existing hash dependency instead of cloning Config.
420pub fn native_exec_config_digest(config: &Config) -> Result<[u8; 32], NativeExecCutError> {
421    struct HashWriter(Sha256);
422    impl io::Write for HashWriter {
423        fn write(&mut self, bytes: &[u8]) -> io::Result<usize> {
424            self.0.update(bytes);
425            Ok(bytes.len())
426        }
427
428        fn flush(&mut self) -> io::Result<()> {
429            Ok(())
430        }
431    }
432    let mut writer = HashWriter(Sha256::new());
433    bincode::serde::encode_into_std_write(config, &mut writer, bincode::config::legacy())
434        .map_err(|_| NativeExecCutError::ConfigEncoding)?;
435    Ok(writer.0.finalize().into())
436}

438/// Check the actual host Config before granting native InitialImage or exec authority.
439///
440/// `host_input_log` and `fatal_core_capture` belong to the continuing host
441/// GlobalState. This policy preserves them without copying them to the guest.
442/// Their presence does not establish that the caller retained that owner; the
443/// launcher must keep it alive through exec and the final physical drain.
444/// Neither this check nor a Config digest establishes runtime admission.
445pub fn validate_native_exec_host_configuration(config: &Config) -> Result<(), NativeExecCutError> {
446    validate_native_exec_common_configuration(config)
447}
448
449/// Check the live guest Config before snapshotting or using a restored Tool.
450///
451/// Serde omits these fields, so a wire digest cannot establish their absence.
452/// The continuing host owns output paths; the guest must not carry a discarded
453/// copy. This function inspects the Config without clearing or rebinding it.
454pub fn validate_native_exec_guest_configuration(config: &Config) -> Result<(), NativeExecCutError> {
455    validate_native_exec_common_configuration(config)?;
456    if config.host_input_log.is_some() {
457        return Err(NativeExecCutError::Configuration("host_input_log"));
458    }
459    if config.fatal_core_capture.is_some() {
460        return Err(NativeExecCutError::Configuration("fatal_core_capture"));
461    }
462    Ok(())
463}
464
465fn validate_native_exec_common_configuration(config: &Config) -> Result<(), NativeExecCutError> {
466    // The guest branches on presence, including Some(empty). Do not silently
467    // turn a resolved program into None by passing it through serde.
468    if config.happens_before.is_some() {
469        return Err(NativeExecCutError::Configuration("happens_before"));
470    }
471    // This is an actual image address, never a transferable Config constant.
472    if config.untraced_code_range.is_some() {
473        return Err(NativeExecCutError::Configuration("untraced_code_range"));
474    }
475    if config.use_rcb_time() || config.max_timeslice.is_some() || config.target_timeslice.is_some()
476    {
477        return Err(NativeExecCutError::Configuration("RCB/timeslices"));
478    }
479    if config.imprecise_timers
480        || config.chaos
481        || config.chaos_target_races
482        || config.chaos_per_thread_slowdown
483    {
484        return Err(NativeExecCutError::Configuration("chaos/imprecise timers"));
485    }
486    if config.record_preemptions
487        || config.record_preemptions_to.is_some()
488        || config.replay_preemptions_from.is_some()
489        || config.replay_schedule_from.is_some()
490        || config.preemption_stacktrace
491        || config.preemption_stacktrace_log_file.is_some()
492        || !config.stacktrace_event.is_empty()
493        || config.stacktrace_signal.is_some()
494    {
495        return Err(NativeExecCutError::Configuration(
496            "preemption/replay/stacktrace",
497        ));
498    }
499    if config.recordreplay_modes || config.replaying || config.replay_data.is_some() {
500        return Err(NativeExecCutError::Configuration("record/replay"));
501    }
502    Ok(())
503}

506    lock: &'a Mutex<T>,
