45const SECCOMP_SET_MODE_STRICT: u32 = 0;
46const SECCOMP_SET_MODE_FILTER: u32 = 1;
47const SECCOMP_GET_ACTION_AVAIL: u32 = 2;

51fn seccomp_result(op: u32, flags: u32, has_args: bool) -> Result<i64, Errno> {
52    if op > SECCOMP_GET_NOTIF_SIZES {
53        return Err(Errno::EINVAL);
54    }
55    if op == SECCOMP_SET_MODE_STRICT && (flags != 0 || has_args) {
56        return Err(Errno::EINVAL);
57    }
58    if op == SECCOMP_SET_MODE_FILTER && flags & !SECCOMP_FILTER_FLAG_TSYNC != 0 {
59        return Err(Errno::EINVAL);
60    }
61    if matches!(
62        op,
63        SECCOMP_SET_MODE_FILTER | SECCOMP_GET_ACTION_AVAIL | SECCOMP_GET_NOTIF_SIZES
64    ) && !has_args
65    {
66        return Err(Errno::EFAULT);
67    }
68
69    // Hermit cannot enforce a guest-installed BPF policy in every backend.
70    // Report the operation as unsupported instead of claiming a filter exists.
71    Err(Errno::EOPNOTSUPP)
72}

74fn is_supported_prctl_option(option: libc::c_int) -> bool {
75    matches!(
76        option,
77        libc::PR_SET_NAME
78            | libc::PR_GET_NAME
79            | libc::PR_SET_THP_DISABLE
80            | libc::PR_GET_THP_DISABLE
81            // AUTONOMOUS-BOT-IMPLEMENTED
82            // TODO-HUMAN-REVIEW(#919)
83            //
84            // Dumpability is per-process state initialized deterministically by
85            // ordinary exec and subsequently controlled only by the guest. Keep
86            // Linux's 0/1 validation and state transitions rather than refusing
87            // applications that explicitly disable or restore core-dump access.
88            | libc::PR_SET_DUMPABLE
89            | libc::PR_GET_DUMPABLE
90            // PR_SET_TIMERSLACK and PR_GET_TIMERSLACK are deliberately absent:
91            // `handle_prctl` emulates them against per-thread Detcore state.
92            // AUTONOMOUS-BOT-IMPLEMENTED
93            // TODO-HUMAN-REVIEW(#802)
94            //
95            // PR_{SET,GET}_KEEPCAPS only read/toggle the calling thread's
96            // "keep capabilities across a UID change" flag. The result is a pure
97            // function of the guest's own prior prctl calls (0/1), never host
98            // state, so passthrough is deterministic and bitwise-identical across
99            // runs. Supporting it lets `setpriv` (and the `date`/privilege-drop
100            // wrappers that call it) run under --strict instead of aborting with
101            // "keep process capabilities failed: Function not implemented".
102            | libc::PR_SET_KEEPCAPS
103            | libc::PR_GET_KEEPCAPS
104            // AUTONOMOUS-BOT-IMPLEMENTED
105            // TODO-HUMAN-REVIEW(#824)
106            //
107            // PR_{SET,GET}_PDEATHSIG only read/toggle the calling thread's
108            // parent-death-signal attribute. PR_SET_PDEATHSIG validates its
109            // signal argument (a valid signal or 0 succeeds, anything else
110            // faults EINVAL) and PR_GET_PDEATHSIG reports the value the guest
111            // previously set. The result is a pure function of the guest's own
112            // prior prctl calls and its argument, never host state, so
113            // passthrough is deterministic and bitwise-identical across runs.
114            // The registered signal only ever fires on parent death, which is a
115            // deterministically scheduled event under Hermit. Supporting it lets
116            // `setpriv --pdeathsig` run under --strict instead of aborting with
117            // "set parent death signal failed: Function not implemented".
118            | libc::PR_SET_PDEATHSIG
119            | libc::PR_GET_PDEATHSIG
120    )
121}

129/// Is `which` one of the Linux `PRIO_*` target selectors for get/setpriority?

132        || which == libc::PRIO_PGRP as i32
133        || which == libc::PRIO_USER as i32

175fn from_str(s: &str) -> [i8; 65] {
176    let mut ret: [i8; 65] = [0; 65];
177    for (i, ch) in s.bytes().take(64).enumerate() {
178        ret[i] = ch as i8;
179    }
180    ret
181}

187fn validate_getrandom_flags(flags: usize) -> Result<(), Errno> {
188    let flags = flags as u32;
189    let random = flags & libc::GRND_RANDOM != 0;
190    let insecure = flags & libc::GRND_INSECURE != 0;
191
192    if flags & !GETRANDOM_ALLOWED_FLAGS != 0 || (random && insecure) {
193        Err(Errno::EINVAL)
194    } else {
195        Ok(())
196    }

199const RANDOM_FILL_CHUNK_BYTES: usize = 4096;
200const RANDOM_DEVICE_BYTE_STRIDE: u8 = 73;
201const RANDOM_DEVICE_FIRST_BYTE: u8 = 41;
202// Linux's import_ubuf clamps getrandom requests to MAX_RW_COUNT on x86_64.
203const GETRANDOM_MAX_BYTES: usize = (i32::MAX as usize) & !4095;

206    requested.min(GETRANDOM_MAX_BYTES)

209fn write_random_chunk(
210    mut memory: impl MemoryAccess,
211    remote_buf: AddrMut<u8>,
212    local_buf: &[u8],
213) -> Result<usize, Errno> {
214    const PTRACE_WORD_SPLIT: usize = std::mem::size_of::<u64>() / 2;
215
216    if local_buf.len() != std::mem::size_of::<u64>() {
217        return memory.write(remote_buf, local_buf);
218    }
219
220    // safeptrace uses PTRACE_POKEDATA for exactly eight bytes, which bypasses guest page
221    // protections. Split that case so getrandom observes the same EFAULT boundary as Linux.
222    let first = memory.write(remote_buf, &local_buf[..PTRACE_WORD_SPLIT])?;
223    if first < PTRACE_WORD_SPLIT {
224        return Ok(first);
225    }
226    let Some(second_buf) = remote_buf
227        .as_raw()
228        .checked_add(PTRACE_WORD_SPLIT)
229        .and_then(AddrMut::<u8>::from_raw)
230    else {
231        return Ok(first);
232    };
233    match memory.write(second_buf, &local_buf[PTRACE_WORD_SPLIT..]) {
234        Ok(second) => Ok(first + second),
235        Err(_) => Ok(first),
236    }
237}

239// AUTONOMOUS-BOT-IMPLEMENTED
240// TODO-HUMAN-REVIEW(PR-1096): Review the backend-independent random-device stream.
241fn canonical_random_device_byte(seed: u64, index: u64) -> u8 {
242    let seed_byte = seed.rotate_right(((index % 8) * 8) as u32) as u8;
243    (index as u8)
244        .wrapping_mul(RANDOM_DEVICE_BYTE_STRIDE)
245        .wrapping_add(RANDOM_DEVICE_FIRST_BYTE)
246        ^ seed_byte
247}

273    /// Preserve thread-local bases while hiding host CPU feature controls.
274    pub async fn handle_arch_prctl<G: Guest<Self>>(
275        &self,
276        guest: &mut G,
277        call: syscalls::ArchPrctl,
278    ) -> Result<i64, Error> {
279        let cpuid_uses_backend_policy =
280            self.cfg.virtualize_cpuid && self.cfg.cpuid_virtualized_by_backend;
281        let cpuid_uses_faulting = self.cfg.virtualize_cpuid && guest.has_cpuid_interception();
282        match call.cmd() {
283            ArchPrctlCmd::ARCH_SET_FS(_)
284            | ArchPrctlCmd::ARCH_SET_GS(_)
285            | ArchPrctlCmd::ARCH_GET_FS(_)
286            | ArchPrctlCmd::ARCH_GET_GS(_) => Ok(guest.inject(call).await?),
287
288            // KVM installs a deterministic CPUID table while leaving the instruction enabled.
289            ArchPrctlCmd::ARCH_GET_CPUID(_) if cpuid_uses_backend_policy => Ok(1),
290            ArchPrctlCmd::ARCH_SET_CPUID(value) if cpuid_uses_backend_policy => {
291                if value == 0 {
292                    Err(Errno::EPERM.into())
293                } else {
294                    Ok(0)
295                }
296            }
297
298            // When Reverie successfully disables native CPUID, Detcore answers its fault from a
299            // fixed table. Preserve that backend state and reject attempts to re-enable CPUID.
300            ArchPrctlCmd::ARCH_GET_CPUID(_) if cpuid_uses_faulting => Ok(0),
301            ArchPrctlCmd::ARCH_SET_CPUID(value) if cpuid_uses_faulting => {
302                if value == 0 {
303                    Ok(0)
304                } else {
305                    Err(Errno::EPERM.into())
306                }
307            }
308            // Reverie cannot faithfully deliver a CPUID fault requested by the tracee. In
309            // explicit host-CPUID mode, expose a fixed enabled control state and reject disable.
310            ArchPrctlCmd::ARCH_GET_CPUID(_) if !self.cfg.virtualize_cpuid => Ok(1),
311            ArchPrctlCmd::ARCH_SET_CPUID(value) if !self.cfg.virtualize_cpuid => {
312                if value == 0 {
313                    Err(Errno::EPERM.into())
314                } else {
315                    Ok(0)
316                }
317            }
318
319            // Ptrace hosts without CPUID-faulting support retain the kernel's honest state.
320            ArchPrctlCmd::ARCH_GET_CPUID(_) | ArchPrctlCmd::ARCH_SET_CPUID(_) => {
321                Ok(guest.inject(call).await?)
322            }
323
324            // Expose a conservative virtual CPU with no optional extended-state permissions.
325            ArchPrctlCmd::Other(
326                ARCH_GET_XCOMP_SUPP | ARCH_GET_XCOMP_PERM | ARCH_GET_XCOMP_GUEST_PERM,
327                addr,
328            ) => self.write_arch_prctl_u64(guest, addr, 0),
329            ArchPrctlCmd::Other(ARCH_REQ_XCOMP_PERM | ARCH_REQ_XCOMP_GUEST_PERM, _) => {
330                Err(Errno::EINVAL.into())
331            }
332
333            // Keep shadow stacks disabled in the virtual policy. Disabling an already-disabled
334            // feature is idempotent; enable/lock/unlock requests cannot be honored.
335            ArchPrctlCmd::Other(ARCH_SHSTK_STATUS, addr) => {
336                self.write_arch_prctl_u64(guest, addr, 0)
337            }
338            ArchPrctlCmd::Other(ARCH_SHSTK_DISABLE, features)
339                if features != 0 && features & !ARCH_SHSTK_VALID_MASK == 0 =>
340            {
341                Ok(0)
342            }
343            ArchPrctlCmd::Other(ARCH_SHSTK_DISABLE, _)
344            | ArchPrctlCmd::Other(ARCH_SHSTK_ENABLE | ARCH_SHSTK_LOCK | ARCH_SHSTK_UNLOCK, _) => {
345                Err(Errno::EINVAL.into())
346            }
347
348            ArchPrctlCmd::Other(_, _) => Err(Errno::EINVAL.into()),
349        }
350    }

353    // TODO-HUMAN-REVIEW(#663)
354    /// Preserve deterministic Ruby thread controls, report the container's fixed
355    /// capability bounding set, and reject options that expose unmodeled process
356    /// or host state.
357    pub async fn handle_prctl<G: Guest<Self>>(
358        &self,
359        guest: &mut G,
360        call: syscalls::Prctl,
361    ) -> Result<i64, Error> {
362        match call.option() {
363            // The capability bounding set is fixed by the container launch policy.
364            libc::PR_CAPBSET_READ => Ok(self.record_or_replay(guest, call).await?),
365            // AUTONOMOUS-BOT-IMPLEMENTED
366            // TODO-HUMAN-REVIEW(PR-2150): Timer slack is shared with the
367            // virtual `/proc/<tid>/timerslack_ns` channel.  Never pass either
368            // setter through to the physical tracee: a large physical slack
369            // changes wake timing on Detcore's remaining host-timed waits.
370            libc::PR_SET_TIMERSLACK => {
371                let requested = call.arg2();
372                let state = guest.thread_state_mut();
373                // Linux treats zero as reset-to-default. Hermit virtualizes the
374                // Linux scheduling policy as SCHED_OTHER, so the kernel's
375                // RT/DL no-op branch is unreachable in the guest model.
376                state.timer_slack_ns = if requested == 0 {
377                    state.default_timer_slack_ns
378                } else {
379                    requested
380                };
381                Ok(0)
382            }
383            libc::PR_GET_TIMERSLACK => Ok(guest.thread_state().timer_slack_ns as i64),
384            option
385                if guest.config().backend_virtualizes_capability_prctls
386                    && is_backend_virtualized_capability_prctl(option) =>
387            {
388                self.passthrough(guest, call.into()).await
389            }
390            option if is_supported_prctl_option(option) => {
391                self.passthrough(guest, call.into()).await
392            }
393            _ => Err(Errno::ENOSYS.into()),
394        }
395    }

464    /// Fill guest memory from the deterministic PRNG owned by the current thread.
465    pub(super) fn fill_random_bytes<G: Guest<Self>>(
466        &self,
467        guest: &mut G,
468        remote_buf: AddrMut<u8>,
469        len: usize,
470        source: &str,
471    ) -> Result<usize, Error> {
472        let mut local_words = [0_u64; RANDOM_FILL_CHUNK_BYTES / std::mem::size_of::<u64>()];
473        let mut hasher = DefaultHasher::new();
474        let mut written = 0;
475
476        while written < len {
477            let remote_chunk = match remote_buf
478                .as_raw()
479                .checked_add(written)
480                .and_then(AddrMut::<u8>::from_raw)
481            {
482                Some(address) => address,
483                None if written == 0 => return Err(Errno::EFAULT.into()),
484                None => break,
485            };
486            let chunk_len = (len - written).min(RANDOM_FILL_CHUNK_BYTES);
487            // safeptrace's 8-byte write fast path currently requires an aligned source buffer.
488            let local_buf = unsafe {
489                std::slice::from_raw_parts_mut(local_words.as_mut_ptr().cast::<u8>(), chunk_len)
490            };
491            guest.thread_state_mut().thread_prng().fill(local_buf);
492            let n = match write_random_chunk(guest.memory(), remote_chunk, local_buf) {
493                Ok(n) => n,
494                Err(_) if written > 0 => break,
495                Err(error) => return Err(error.into()),
496            };
497            if n == 0 {
498                if written == 0 {
499                    return Err(Errno::EFAULT.into());
500                }
501                break;
502            }
503            if cfg!(debug_assertions) {
504                Hash::hash_slice(&local_buf[..n], &mut hasher);
505            }
506            written += n;
507            if n < chunk_len {
508                break;
509            }
510        }
511
512        if cfg!(debug_assertions) {
513            detlog!(
514                "[dtid {}] USER RAND [{}] Filled guest memory with {} random bytes, hash of bytes: {}",
515                guest.thread_state().dettid,
516                source,
517                written,
518                hasher.finish()
519            );
520        }
521        Ok(written)
522    }

524    // AUTONOMOUS-BOT-IMPLEMENTED
525    // TODO-HUMAN-REVIEW(PR-1096): Review the backend-independent random-device stream.
526    /// Fill guest memory from the canonical stream used by every backend's
527    /// `/dev/random` and `/dev/urandom` virtualization.
528    pub(super) fn fill_random_device_bytes<G: Guest<Self>>(
529        &self,
530        guest: &mut G,
531        remote_buf: AddrMut<u8>,
532        len: usize,
533        stream_offset: u64,
534    ) -> Result<usize, Error> {
535        let seed = guest.config().rng_seed();
536        let mut local_words = [0_u64; RANDOM_FILL_CHUNK_BYTES / std::mem::size_of::<u64>()];
537        let mut hasher = DefaultHasher::new();
538        let mut written = 0;
539
540        while written < len {
541            let remote_chunk = match remote_buf
542                .as_raw()
543                .checked_add(written)
544                .and_then(AddrMut::<u8>::from_raw)
545            {
546                Some(address) => address,
547                None if written == 0 => return Err(Errno::EFAULT.into()),
548                None => break,
549            };
550            let chunk_len = (len - written).min(RANDOM_FILL_CHUNK_BYTES);
551            let local_buf = unsafe {
552                std::slice::from_raw_parts_mut(local_words.as_mut_ptr().cast::<u8>(), chunk_len)
553            };
554            for (index, byte) in local_buf.iter_mut().enumerate() {
555                *byte = canonical_random_device_byte(
556                    seed,
557                    stream_offset
558                        .saturating_add(written as u64)
559                        .saturating_add(index as u64),
560                );
561            }
562            let n = match write_random_chunk(guest.memory(), remote_chunk, local_buf) {
563                Ok(n) => n,
564                Err(_) if written > 0 => break,
565                Err(error) => return Err(error.into()),
566            };
567            if n == 0 {
568                if written == 0 {
569                    return Err(Errno::EFAULT.into());
570                }
571                break;
572            }
573            if cfg!(debug_assertions) {
574                Hash::hash_slice(&local_buf[..n], &mut hasher);
575            }
576            written += n;
577            if n < chunk_len {
578                break;
579            }
580        }
581
582        if cfg!(debug_assertions) {
583            detlog!(
584                "[dtid {}] USER RAND [/dev/[u]random] Filled guest memory with {} canonical random bytes at offset {}, hash of bytes: {}",
585                guest.thread_state().dettid,
586                written,
587                stream_offset,
588                hasher.finish()
589            );
590        }
591        Ok(written)
592    }

594    /// uname syscall
595    pub async fn handle_uname<G: Guest<Self>>(
596        &self,
597        guest: &mut G,
598        call: syscalls::Uname,
599    ) -> Result<i64, Error> {
600        let ret = self.record_or_replay(guest, call).await?;
601        if let Some(buf) = call.buf() {
602            let mut un = guest.memory().read_value(buf)?;
603            // Keep this in configured UTC: `Local` initializes libc TLS, which is unavailable
604            // while a DynamoRIO application thread is executing a client callback.
605            let epoch = guest.config().epoch;
606
607            if !guest.config().has_uts_namespace {
608                // FIXME: It should be possible to remove this once all tests
609                // are also using namespaces.
610                un.nodename = from_str(DEFAULT_HOSTNAME);
611                un.domainname = from_str(DEFAULT_HOSTNAME.split('.').next_back().unwrap_or(""));
612            }
613
614            un.release = from_str("5.2.0");
615            un.version = from_str(&format!("#1 SMP {}", epoch.format("%a %b %d %T %Z %Y")));
616            guest.memory().write_value(buf, &un)?;
617        }
618
619        Ok(ret)
620    }

622    /// Fill `getrandom(2)` requests from the current thread's seeded deterministic PRNG.
623    /// Supported blocking/source-selection flags share that always-ready stream; invalid Linux
624    /// flag combinations are rejected before guest memory is touched.
625    pub async fn handle_getrandom<G: Guest<Self>>(
626        &self,
627        guest: &mut G,
628        call: syscalls::Getrandom,
629    ) -> Result<i64, Error> {
630        validate_getrandom_flags(call.flags())?;
631        let len = getrandom_request_len(call.buflen());
632        if len == 0 {
633            return Ok(0);
634        }
635
636        let buf = call.buf().ok_or(Errno::EFAULT)?;
637
638        let n = self.fill_random_bytes(guest, buf, len, "getrandom")?;
639        Ok(n as i64)

661    /// after success Detcore mirrors the guest-visible process-group change so
662    /// group-selecting waits do not consult host `/proc` state.
663    pub async fn handle_setpgid<G: Guest<Self>>(
664        &self,
665        guest: &mut G,
666        call: syscalls::Setpgid,
667    ) -> Result<i64, Error> {
668        let res = guest.inject(call).await?;
669        let caller = guest.thread_state().detpid.expect("detpid unset");
670        let process = if call.pid() == 0 {
671            caller
672        } else {
673            DetPid::from_raw(call.pid())
674        };
675        let group = if call.pgid() == 0 {
676            process
677        } else {
678            DetPid::from_raw(call.pgid())
679        };
680        let _ = set_process_group(guest, process, group).await;
681        Ok(res)
682    }

684    /// membarrier (system call).
685    ///
686    /// `membarrier(2)` issues process-wide memory barriers so that userspace can
687    /// use asymmetric fences (e.g. CPython's QSBR, RCU-style reclamation).
688    /// Detcore serializes all guest threads onto a single logical CPU with a
689    /// total memory order, so any requested barrier is *already* satisfied and
690    /// every command is a deterministic no-op. For `MEMBARRIER_CMD_QUERY` we
691    /// report the set of commands we emulate so the guest stays on this
692    /// controlled path instead of a host-dependent fallback; every other command
693    /// returns success without doing anything.
694    pub async fn handle_membarrier<G: Guest<Self>>(
695        &self,
696        guest: &mut G,
697        call: syscalls::Membarrier,
698    ) -> Result<i64, Error> {
699        // Values from <linux/membarrier.h>.
700        const MEMBARRIER_CMD_QUERY: i32 = 0;
701        const MEMBARRIER_CMD_GLOBAL: i32 = 1 << 0;
702        const MEMBARRIER_CMD_GLOBAL_EXPEDITED: i32 = 1 << 1;
703        const MEMBARRIER_CMD_REGISTER_GLOBAL_EXPEDITED: i32 = 1 << 2;
704        const MEMBARRIER_CMD_PRIVATE_EXPEDITED: i32 = 1 << 3;
705        const MEMBARRIER_CMD_REGISTER_PRIVATE_EXPEDITED: i32 = 1 << 4;
706        const SUPPORTED: i32 = MEMBARRIER_CMD_GLOBAL
707            | MEMBARRIER_CMD_GLOBAL_EXPEDITED
708            | MEMBARRIER_CMD_REGISTER_GLOBAL_EXPEDITED
709            | MEMBARRIER_CMD_PRIVATE_EXPEDITED
710            | MEMBARRIER_CMD_REGISTER_PRIVATE_EXPEDITED;
711
712        let cmd = call.cmd();
713        if cmd == MEMBARRIER_CMD_QUERY {
714            detlog!(
715                "[dtid {}] membarrier(QUERY) => reporting emulated commands {:#x}",
716                guest.thread_state().dettid,
717                SUPPORTED,
718            );
719            Ok(SUPPORTED as i64)
720        } else {
721            detlog!(
722                "[dtid {}] membarrier(cmd={}) no-op (threads are serialized on one CPU)",
723                guest.thread_state().dettid,
724                cmd,
725            );
726            Ok(0)
727        }
728    }

732        &self,
733        guest: &mut G,
734        call: syscalls::Getcpu,

736        // Always set the CPU to 0.

743            guest.memory().write_value(node, &0)?;
744        }
745
746        Ok(0)

749    /// getresuid under Hermit. Detcore presents a fixed virtual-root identity, so
750    /// the real, effective, and saved user IDs are all the constant 0. Under the
751    /// ptrace backend the guest runs inside a CLONE_NEWUSER namespace that already
752    /// maps the host uid to 0; emulating the same constant here makes in-process
753    /// backends (DBT) agree with that golden reference instead of leaking the host
754    /// uid, and the fully emulated result is bitwise-identical across --verify and
755    /// record/replay.
756    // AUTONOMOUS-BOT-IMPLEMENTED
757    // TODO-HUMAN-REVIEW(#1549)

759        &self,

761        call: syscalls::Getresuid,
762    ) -> Result<i64, Error> {
763        if let Some(ruid) = call.ruid() {
764            guest.memory().write_value(ruid, &0)?;
765        }
766        if let Some(euid) = call.euid() {
767            guest.memory().write_value(euid, &0)?;
768        }
769        if let Some(suid) = call.suid() {
770            guest.memory().write_value(suid, &0)?;
771        }
772        Ok(0)
773    }

798    /// get_mempolicy under Hermit. The container exposes a single virtual NUMA
799    /// node, so the effective policy is always the default and every address
800    /// resolves to node 0. The result is fully emulated (never injected), so it
801    /// is bitwise-identical across the two --verify runs and under record/replay,
802    /// removing the host-NUMA-topology dependence a passthrough would introduce.
803    // AUTONOMOUS-BOT-IMPLEMENTED
804    // TODO-HUMAN-REVIEW(#720)
805    pub async fn handle_get_mempolicy<G: Guest<Self>>(
806        &self,
807        guest: &mut G,
808        call: syscalls::GetMempolicy,
809    ) -> Result<i64, Error> {
810        // Report MPOL_DEFAULT (0) for the current policy / node when requested.
811        // The nodemask output is left untouched: reverie exposes it as an
812        // immutable pointer, and MPOL_DEFAULT carries no node set.
813        if let Some(policy) = call.policy() {
814            guest.memory().write_value(policy, &0)?;
815        }
816        Ok(0)
817    }

819    /// move_pages under Hermit. On a single virtual NUMA node nothing can be
820    /// relocated, so report every page as residing on node 0 and succeed. The
821    /// answer is a fixed constant, so it is deterministic across --verify and
822    /// record/replay.
823    // AUTONOMOUS-BOT-IMPLEMENTED
824    // TODO-HUMAN-REVIEW(#720)
825    pub async fn handle_move_pages<G: Guest<Self>>(
826        &self,
827        guest: &mut G,
828        call: syscalls::MovePages,
829    ) -> Result<i64, Error> {
830        // When a status buffer is supplied (either a real move request or a
831        // location query with nodes == NULL), report node 0 for every page.
832        if let Some(status) = call.status() {
833            let count = call.nr_pages() as usize;
834            let zeros = vec![0i32; count];
835            guest.memory().write_values(status, &zeros)?;
836        }
837        Ok(0)
838    }
