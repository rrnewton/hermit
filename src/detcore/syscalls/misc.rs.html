52const SECCOMP_SET_MODE_STRICT: u32 = 0;

54const SECCOMP_GET_ACTION_AVAIL: u32 = 2;

58fn seccomp_result(op: u32, flags: u32, has_args: bool) -> Result<i64, Errno> {
59    if op > SECCOMP_GET_NOTIF_SIZES {
60        return Err(Errno::EINVAL);
61    }
62    if op == SECCOMP_SET_MODE_STRICT && (flags != 0 || has_args) {
63        return Err(Errno::EINVAL);
64    }
65    if op == SECCOMP_SET_MODE_FILTER && flags & !SECCOMP_FILTER_FLAG_TSYNC != 0 {
66        return Err(Errno::EINVAL);
67    }
68    if matches!(
69        op,
70        SECCOMP_SET_MODE_FILTER | SECCOMP_GET_ACTION_AVAIL | SECCOMP_GET_NOTIF_SIZES
71    ) && !has_args
72    {
73        return Err(Errno::EFAULT);
74    }
75
76    // Hermit cannot enforce a guest-installed BPF policy in every backend.
77    // Report the operation as unsupported instead of claiming a filter exists.
78    Err(Errno::EOPNOTSUPP)
79}

81fn is_supported_prctl_option(option: libc::c_int) -> bool {
82    matches!(
83        option,
84        libc::PR_SET_NAME
85            | libc::PR_GET_NAME
86            | libc::PR_SET_THP_DISABLE
87            | libc::PR_GET_THP_DISABLE
88            // AUTONOMOUS-BOT-IMPLEMENTED
89            // TODO-HUMAN-REVIEW(#919)
90            //
91            // Dumpability is per-process state initialized deterministically by
92            // ordinary exec and subsequently controlled only by the guest. Keep
93            // Linux's 0/1 validation and state transitions rather than refusing
94            // applications that explicitly disable or restore core-dump access.
95            | libc::PR_SET_DUMPABLE
96            | libc::PR_GET_DUMPABLE
97            // PR_SET_TIMERSLACK and PR_GET_TIMERSLACK are deliberately absent:
98            // `handle_prctl` emulates them against per-thread Detcore state.
99            // AUTONOMOUS-BOT-IMPLEMENTED
100            // TODO-HUMAN-REVIEW(#802)
101            //
102            // PR_{SET,GET}_KEEPCAPS only read/toggle the calling thread's
103            // "keep capabilities across a UID change" flag. The result is a pure
104            // function of the guest's own prior prctl calls (0/1), never host
105            // state, so passthrough is deterministic and bitwise-identical across
106            // runs. Supporting it lets `setpriv` (and the `date`/privilege-drop
107            // wrappers that call it) run under --strict instead of aborting with
108            // "keep process capabilities failed: Function not implemented".
109            | libc::PR_SET_KEEPCAPS
110            | libc::PR_GET_KEEPCAPS
111            // AUTONOMOUS-BOT-IMPLEMENTED
112            // TODO-HUMAN-REVIEW(#824)
113            //
114            // PR_{SET,GET}_PDEATHSIG only read/toggle the calling thread's
115            // parent-death-signal attribute. PR_SET_PDEATHSIG validates its
116            // signal argument (a valid signal or 0 succeeds, anything else
117            // faults EINVAL) and PR_GET_PDEATHSIG reports the value the guest
118            // previously set. The result is a pure function of the guest's own
119            // prior prctl calls and its argument, never host state, so
120            // passthrough is deterministic and bitwise-identical across runs.
121            // The registered signal only ever fires on parent death, which is a
122            // deterministically scheduled event under Hermit. Supporting it lets
123            // `setpriv --pdeathsig` run under --strict instead of aborting with
124            // "set parent death signal failed: Function not implemented".
125            | libc::PR_SET_PDEATHSIG
126            | libc::PR_GET_PDEATHSIG
127    )
128}

136/// Is `which` one of the Linux `PRIO_*` target selectors for get/setpriority?
137fn is_valid_prio_which(which: i32) -> bool {
138    which == libc::PRIO_PROCESS as i32
139        || which == libc::PRIO_PGRP as i32
140        || which == libc::PRIO_USER as i32
141}

145/// Deterministic raw `getpriority(2)` result under Hermit's inert-nice model.
146///
147/// Returns `20 - nice` (nice 0 -> 20) for any valid target regardless of `who`,
148/// because nice is inert under the virtualized scheduler and `getpriority` never

151    if is_valid_prio_which(which) {
152        Ok(20)
153    } else {
154        Err(Errno::EINVAL)

182fn from_str(s: &str) -> [i8; 65] {
183    let mut ret: [i8; 65] = [0; 65];
184    for (i, ch) in s.bytes().take(64).enumerate() {
185        ret[i] = ch as i8;
186    }
187    ret
188}

193// AUTONOMOUS-BOT-IMPLEMENTED
194// TODO-HUMAN-REVIEW(PR-1096): Review the backend-independent random-device stream.
195fn canonical_random_device_byte(seed: u64, index: u64) -> u8 {
196    let seed_byte = seed.rotate_right(((index % 8) * 8) as u32) as u8;
197    (index as u8)
198        .wrapping_mul(RANDOM_DEVICE_BYTE_STRIDE)
199        .wrapping_add(RANDOM_DEVICE_FIRST_BYTE)
200        ^ seed_byte
201}

226    // TODO-HUMAN-REVIEW(#539): Confirm the virtual arch_prctl control policy.
227    /// Preserve thread-local bases while hiding host CPU feature controls.
228    pub async fn handle_arch_prctl<G: Guest<Self>>(
229        &self,
230        guest: &mut G,
231        call: syscalls::ArchPrctl,
232    ) -> Result<i64, Error> {
233        let cpuid_uses_backend_policy =
234            self.cfg.virtualize_cpuid && self.cfg.cpuid_virtualized_by_backend;
235        let cpuid_uses_faulting = self.cfg.virtualize_cpuid && guest.has_cpuid_interception();
236        match call.cmd() {
237            ArchPrctlCmd::ARCH_SET_FS(_)
238            | ArchPrctlCmd::ARCH_SET_GS(_)
239            | ArchPrctlCmd::ARCH_GET_FS(_)
240            | ArchPrctlCmd::ARCH_GET_GS(_) => Ok(guest.inject(call).await?),
241
242            // KVM installs a deterministic CPUID table while leaving the instruction enabled.
243            ArchPrctlCmd::ARCH_GET_CPUID(_) if cpuid_uses_backend_policy => Ok(1),
244            ArchPrctlCmd::ARCH_SET_CPUID(value) if cpuid_uses_backend_policy => {
245                if value == 0 {
246                    Err(Errno::EPERM.into())
247                } else {
248                    Ok(0)
249                }
250            }
251
252            // When Reverie successfully disables native CPUID, Detcore answers its fault from a
253            // fixed table. Preserve that backend state and reject attempts to re-enable CPUID.
254            ArchPrctlCmd::ARCH_GET_CPUID(_) if cpuid_uses_faulting => Ok(0),
255            ArchPrctlCmd::ARCH_SET_CPUID(value) if cpuid_uses_faulting => {
256                if value == 0 {
257                    Ok(0)
258                } else {
259                    Err(Errno::EPERM.into())
260                }
261            }
262            // Reverie cannot faithfully deliver a CPUID fault requested by the tracee. In
263            // explicit host-CPUID mode, expose a fixed enabled control state and reject disable.
264            ArchPrctlCmd::ARCH_GET_CPUID(_) if !self.cfg.virtualize_cpuid => Ok(1),
265            ArchPrctlCmd::ARCH_SET_CPUID(value) if !self.cfg.virtualize_cpuid => {
266                if value == 0 {
267                    Err(Errno::EPERM.into())
268                } else {
269                    Ok(0)
270                }
271            }
272
273            // Ptrace hosts without CPUID-faulting support retain the kernel's honest state.
274            ArchPrctlCmd::ARCH_GET_CPUID(_) | ArchPrctlCmd::ARCH_SET_CPUID(_) => {
275                Ok(guest.inject(call).await?)
276            }
277
278            // Expose a conservative virtual CPU with no optional extended-state permissions.
279            ArchPrctlCmd::Other(
280                ARCH_GET_XCOMP_SUPP | ARCH_GET_XCOMP_PERM | ARCH_GET_XCOMP_GUEST_PERM,
281                addr,
282            ) => self.write_arch_prctl_u64(guest, addr, 0),
283            ArchPrctlCmd::Other(ARCH_REQ_XCOMP_PERM | ARCH_REQ_XCOMP_GUEST_PERM, _) => {
284                Err(Errno::EINVAL.into())
285            }
286
287            // Keep shadow stacks disabled in the virtual policy. Disabling an already-disabled
288            // feature is idempotent; enable/lock/unlock requests cannot be honored.
289            ArchPrctlCmd::Other(ARCH_SHSTK_STATUS, addr) => {
290                self.write_arch_prctl_u64(guest, addr, 0)
291            }
292            ArchPrctlCmd::Other(ARCH_SHSTK_DISABLE, features)
293                if features != 0 && features & !ARCH_SHSTK_VALID_MASK == 0 =>
294            {
295                Ok(0)
296            }
297            ArchPrctlCmd::Other(ARCH_SHSTK_DISABLE, _)
298            | ArchPrctlCmd::Other(ARCH_SHSTK_ENABLE | ARCH_SHSTK_LOCK | ARCH_SHSTK_UNLOCK, _) => {
299                Err(Errno::EINVAL.into())
300            }
301
302            ArchPrctlCmd::Other(_, _) => Err(Errno::EINVAL.into()),
303        }
304    }

307    // TODO-HUMAN-REVIEW(#663)
308    /// Preserve deterministic Ruby thread controls, report the container's fixed
309    /// capability bounding set, and reject options that expose unmodeled process
310    /// or host state.
311    pub async fn handle_prctl<G: Guest<Self>>(
312        &self,
313        guest: &mut G,
314        call: syscalls::Prctl,
315    ) -> Result<i64, Error> {
316        match call.option() {
317            // The capability bounding set is fixed by the container launch policy.
318            libc::PR_CAPBSET_READ => Ok(self.record_or_replay(guest, call).await?),
319            // AUTONOMOUS-BOT-IMPLEMENTED
320            // TODO-HUMAN-REVIEW(PR-2150): Timer slack is shared with the
321            // virtual `/proc/<tid>/timerslack_ns` channel.  Never pass either
322            // setter through to the physical tracee: a large physical slack
323            // changes wake timing on Detcore's remaining host-timed waits.
324            libc::PR_SET_TIMERSLACK => {
325                let requested = call.arg2();
326                let state = guest.thread_state_mut();
327                // Linux treats zero as reset-to-default. Hermit virtualizes the
328                // Linux scheduling policy as SCHED_OTHER, so the kernel's
329                // RT/DL no-op branch is unreachable in the guest model.
330                state.timer_slack_ns = if requested == 0 {
331                    state.default_timer_slack_ns
332                } else {
333                    requested
334                };
335                Ok(0)
336            }
337            libc::PR_GET_TIMERSLACK => Ok(guest.thread_state().timer_slack_ns as i64),
338            option
339                if guest.config().backend_virtualizes_capability_prctls
340                    && is_backend_virtualized_capability_prctl(option) =>
341            {
342                self.passthrough(guest, call.into()).await
343            }
344            option if is_supported_prctl_option(option) => {
345                self.passthrough(guest, call.into()).await
346            }
347            _ => Err(Errno::ENOSYS.into()),
348        }
349    }

404    pub fn handle_process_madvise(pidfd: usize, flags: usize) -> Result<i64, Error> {
405        if flags != 0 {
406            return Err(Errno::EINVAL.into());
407        }
408
409        // Linux interprets pidfd as an int. Preserve its deterministic invalid-fd
410        // rejection, but never let a valid host pidfd alter another process's memory.
411        if (pidfd as libc::c_int) < 0 {
412            Err(Errno::EBADF.into())
413        } else {
414            Err(Errno::EPERM.into())
415        }
416    }

418    // AUTONOMOUS-BOT-IMPLEMENTED
419    // TODO-HUMAN-REVIEW(PR-1096): Review the backend-independent random-device stream.
420    /// Fill guest memory from the canonical stream used by every backend's
421    /// `/dev/random` and `/dev/urandom` virtualization.
422    pub(super) fn fill_random_device_bytes<G: Guest<Self>>(
423        &self,
424        guest: &mut G,
425        remote_buf: AddrMut<u8>,
426        len: usize,
427        stream_offset: u64,
428    ) -> Result<usize, Error> {
429        let seed = guest.config().rng_seed();
430        let mut local_words = [0_u64; RANDOM_FILL_CHUNK_BYTES / std::mem::size_of::<u64>()];
431        let mut hasher = DefaultHasher::new();
432        let mut written = 0;
433
434        while written < len {
435            let remote_chunk = match remote_buf
436                .as_raw()
437                .checked_add(written)
438                .and_then(AddrMut::<u8>::from_raw)
439            {
440                Some(address) => address,
441                None if written == 0 => return Err(Errno::EFAULT.into()),
442                None => break,
443            };
444            let chunk_len = (len - written).min(RANDOM_FILL_CHUNK_BYTES);
445            let local_buf = unsafe {
446                std::slice::from_raw_parts_mut(local_words.as_mut_ptr().cast::<u8>(), chunk_len)
447            };
448            for (index, byte) in local_buf.iter_mut().enumerate() {
449                *byte = canonical_random_device_byte(
450                    seed,
451                    stream_offset
452                        .saturating_add(written as u64)
453                        .saturating_add(index as u64),
454                );
455            }
456            let n = match write_random_chunk(&mut guest.memory(), remote_chunk, local_buf) {
457                Ok(n) => n,
458                Err(_) if written > 0 => break,
459                Err(error) => return Err(error.into()),
460            };
461            if n == 0 {
462                if written == 0 {
463                    return Err(Errno::EFAULT.into());
464                }
465                break;
466            }
467            if cfg!(debug_assertions) {
468                Hash::hash_slice(&local_buf[..n], &mut hasher);
469            }
470            written += n;
471            if n < chunk_len {
472                break;
473            }
474        }
475
476        if cfg!(debug_assertions) {
477            detlog!(
478                "[dtid {}] USER RAND [/dev/[u]random] Filled guest memory with {} canonical random bytes at offset {}, hash of bytes: {}",
479                guest.thread_state().dettid,
480                written,
481                stream_offset,
482                hasher.finish()
483            );
484        }
485        Ok(written)
486    }

488    /// uname syscall
489    pub async fn handle_uname<G: Guest<Self>>(
490        &self,
491        guest: &mut G,
492        call: syscalls::Uname,
493    ) -> Result<i64, Error> {
494        let ret = self.record_or_replay(guest, call).await?;
495        if let Some(buf) = call.buf() {
496            let mut un = guest.memory().read_value(buf)?;
497            // Keep this in configured UTC: `Local` initializes libc TLS, which is unavailable
498            // while a DynamoRIO application thread is executing a client callback.
499            let epoch = guest.config().epoch;
500
501            if !guest.config().has_uts_namespace {
502                // FIXME: It should be possible to remove this once all tests
503                // are also using namespaces.
504                un.nodename = from_str(DEFAULT_HOSTNAME);
505                un.domainname = from_str(DEFAULT_HOSTNAME.split('.').next_back().unwrap_or(""));
506            }
507
508            un.release = from_str("5.2.0");
509            un.version = from_str(&format!("#1 SMP {}", epoch.format("%a %b %d %T %Z %Y")));
510            guest.memory().write_value(buf, &un)?;
511        }
512
513        Ok(ret)
514    }

548    /// setpgid system call. The kernel remains authoritative for validation;
549    /// after success Detcore mirrors the guest-visible process-group change so
550    /// group-selecting waits do not consult host `/proc` state.
551    pub async fn handle_setpgid<G: Guest<Self>>(
552        &self,
553        guest: &mut G,
554        call: syscalls::Setpgid,
555    ) -> Result<i64, Error> {
556        let res = guest.inject(call).await?;
557        let caller = guest.thread_state().detpid.expect("detpid unset");
558        let process = if call.pid() == 0 {
559            caller
560        } else {
561            DetPid::from_raw(call.pid())
562        };
563        let group = if call.pgid() == 0 {
564            process
565        } else {
566            DetPid::from_raw(call.pgid())
567        };
568        let _ = set_process_group(guest, process, group).await;
569        Ok(res)
570    }

572    /// membarrier (system call).
573    ///
574    /// `membarrier(2)` issues process-wide memory barriers so that userspace can
575    /// use asymmetric fences (e.g. CPython's QSBR, RCU-style reclamation).
576    /// Detcore serializes all guest threads onto a single logical CPU with a
577    /// total memory order, so any requested barrier is *already* satisfied and
578    /// every command is a deterministic no-op. For `MEMBARRIER_CMD_QUERY` we
579    /// report the set of commands we emulate so the guest stays on this
580    /// controlled path instead of a host-dependent fallback; every other command
581    /// returns success without doing anything.
582    pub async fn handle_membarrier<G: Guest<Self>>(
583        &self,
584        guest: &mut G,
585        call: syscalls::Membarrier,
586    ) -> Result<i64, Error> {
587        // Values from <linux/membarrier.h>.
588        const MEMBARRIER_CMD_QUERY: i32 = 0;
589        const MEMBARRIER_CMD_GLOBAL: i32 = 1 << 0;
590        const MEMBARRIER_CMD_GLOBAL_EXPEDITED: i32 = 1 << 1;
591        const MEMBARRIER_CMD_REGISTER_GLOBAL_EXPEDITED: i32 = 1 << 2;
592        const MEMBARRIER_CMD_PRIVATE_EXPEDITED: i32 = 1 << 3;
593        const MEMBARRIER_CMD_REGISTER_PRIVATE_EXPEDITED: i32 = 1 << 4;
594        const SUPPORTED: i32 = MEMBARRIER_CMD_GLOBAL
595            | MEMBARRIER_CMD_GLOBAL_EXPEDITED
596            | MEMBARRIER_CMD_REGISTER_GLOBAL_EXPEDITED
597            | MEMBARRIER_CMD_PRIVATE_EXPEDITED
598            | MEMBARRIER_CMD_REGISTER_PRIVATE_EXPEDITED;
599
600        let cmd = call.cmd();
601        if cmd == MEMBARRIER_CMD_QUERY {
602            detlog!(
603                "[dtid {}] membarrier(QUERY) => reporting emulated commands {:#x}",
604                guest.thread_state().dettid,
605                SUPPORTED,
606            );
607            Ok(SUPPORTED as i64)
608        } else {
609            detlog!(
610                "[dtid {}] membarrier(cmd={}) no-op (threads are serialized on one CPU)",
611                guest.thread_state().dettid,
612                cmd,
613            );
614            Ok(0)
615        }
616    }

618    /// getcpu system call
619    pub async fn handle_getcpu<G: Guest<Self>>(
620        &self,
621        guest: &mut G,
622        call: syscalls::Getcpu,
623    ) -> Result<i64, Error> {
624        // Always set the CPU to 0.
625        if let Some(cpu) = call.cpu() {
626            guest.memory().write_value(cpu, &0)?;
627        }
628
629        // Always set the NUMA node to 0.
630        if let Some(node) = call.node() {
631            guest.memory().write_value(node, &0)?;
632        }
633
634        Ok(0)
635    }

637    /// getresuid under Hermit. Detcore presents a fixed virtual-root identity, so
638    /// the real, effective, and saved user IDs are all the constant 0. Under the
639    /// ptrace backend the guest runs inside a CLONE_NEWUSER namespace that already
640    /// maps the host uid to 0; emulating the same constant here makes in-process
641    /// backends (DBT) agree with that golden reference instead of leaking the host
642    /// uid, and the fully emulated result is bitwise-identical across --verify and
643    /// record/replay.
644    // AUTONOMOUS-BOT-IMPLEMENTED
645    // TODO-HUMAN-REVIEW(#1549)
646    pub async fn handle_getresuid<G: Guest<Self>>(
647        &self,
648        guest: &mut G,
649        call: syscalls::Getresuid,
650    ) -> Result<i64, Error> {
651        if let Some(ruid) = call.ruid() {
652            guest.memory().write_value(ruid, &0)?;
653        }
654        if let Some(euid) = call.euid() {
655            guest.memory().write_value(euid, &0)?;
656        }
657        if let Some(suid) = call.suid() {
658            guest.memory().write_value(suid, &0)?;
659        }
660        Ok(0)
661    }

663    /// getresgid under Hermit. The group-ID counterpart of `handle_getresuid`:
664    /// the real, effective, and saved group IDs are all the fixed virtual-root
665    /// constant 0, matching the ptrace CLONE_NEWUSER identity and deterministic
666    /// across --verify and record/replay.
667    // AUTONOMOUS-BOT-IMPLEMENTED
668    // TODO-HUMAN-REVIEW(#1549)
669    pub async fn handle_getresgid<G: Guest<Self>>(
670        &self,
671        guest: &mut G,
672        call: syscalls::Getresgid,
673    ) -> Result<i64, Error> {
674        if let Some(rgid) = call.rgid() {
675            guest.memory().write_value(rgid, &0)?;
676        }
677        if let Some(egid) = call.egid() {
678            guest.memory().write_value(egid, &0)?;
679        }
680        if let Some(sgid) = call.sgid() {
681            guest.memory().write_value(sgid, &0)?;
682        }
683        Ok(0)
684    }

686    /// get_mempolicy under Hermit. The container exposes a single virtual NUMA
687    /// node, so the effective policy is always the default and every address
688    /// resolves to node 0. The result is fully emulated (never injected), so it
689    /// is bitwise-identical across the two --verify runs and under record/replay,
690    /// removing the host-NUMA-topology dependence a passthrough would introduce.
691    // AUTONOMOUS-BOT-IMPLEMENTED
692    // TODO-HUMAN-REVIEW(#720)
693    pub async fn handle_get_mempolicy<G: Guest<Self>>(
694        &self,
695        guest: &mut G,
696        call: syscalls::GetMempolicy,
697    ) -> Result<i64, Error> {
698        // Report MPOL_DEFAULT (0) for the current policy / node when requested.
699        // The nodemask output is left untouched: reverie exposes it as an
700        // immutable pointer, and MPOL_DEFAULT carries no node set.
701        if let Some(policy) = call.policy() {
702            guest.memory().write_value(policy, &0)?;
703        }
704        Ok(0)
705    }
706
707    /// move_pages under Hermit. On a single virtual NUMA node nothing can be
708    /// relocated, so report every page as residing on node 0 and succeed. The
709    /// answer is a fixed constant, so it is deterministic across --verify and
710    /// record/replay.
711    // AUTONOMOUS-BOT-IMPLEMENTED
712    // TODO-HUMAN-REVIEW(#720)
713    pub async fn handle_move_pages<G: Guest<Self>>(
714        &self,
715        guest: &mut G,
716        call: syscalls::MovePages,
717    ) -> Result<i64, Error> {
718        // When a status buffer is supplied (either a real move request or a
719        // location query with nodes == NULL), report node 0 for every page.
720        if let Some(status) = call.status() {
721            let count = call.nr_pages() as usize;
722            let zeros = vec![0i32; count];
723            guest.memory().write_values(status, &zeros)?;
724        }
725        Ok(0)
726    }
