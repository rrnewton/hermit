66fn seccomp_result(op: u32, flags: u32, has_args: bool) -> Result<i64, Errno> {
67    if op > SECCOMP_GET_NOTIF_SIZES {
68        return Err(Errno::EINVAL);
69    }
70    if op == SECCOMP_SET_MODE_STRICT && (flags != 0 || has_args) {
71        return Err(Errno::EINVAL);
72    }
73    if op == SECCOMP_SET_MODE_FILTER && flags & !SECCOMP_FILTER_FLAG_TSYNC != 0 {
74        return Err(Errno::EINVAL);
75    }
76    if matches!(
77        op,
78        SECCOMP_SET_MODE_FILTER | SECCOMP_GET_ACTION_AVAIL | SECCOMP_GET_NOTIF_SIZES
79    ) && !has_args
80    {
81        return Err(Errno::EFAULT);
82    }
83
84    // Hermit cannot enforce a guest-installed BPF policy in every backend.
85    // Report the operation as unsupported instead of claiming a filter exists.
86    Err(Errno::EOPNOTSUPP)
87}

97/// Whether parent-death signal `signal` is one Detcore sends itself: a
98/// standard signal (1 to 31) other than SIGKILL and SIGSTOP, which the
99/// kernel keeps acting on as before (a SIGKILL'd child is never counted as
100/// stuck, `Scheduler::child_is_stuck`), and SIGALRM, whose scheduler sends go
101/// to the process's alarm ledger.
102fn detcore_sends_parent_death_signal(signal: libc::c_ulong) -> bool {
103    (1..=31).contains(&signal)
104        && ![libc::SIGKILL, libc::SIGSTOP, libc::SIGALRM].contains(&(signal as libc::c_int))
105}

107fn is_supported_prctl_option(option: libc::c_int) -> bool {
108    matches!(
109        option,
110        libc::PR_SET_NAME
111            | libc::PR_GET_NAME
112            | libc::PR_SET_THP_DISABLE
113            | libc::PR_GET_THP_DISABLE
114            // AUTONOMOUS-BOT-IMPLEMENTED
115            // TODO-HUMAN-REVIEW(#919)
116            //
117            // Dumpability is per-process state initialized deterministically by
118            // ordinary exec and subsequently controlled only by the guest. Keep
119            // Linux's 0/1 validation and state transitions rather than refusing
120            // applications that explicitly disable or restore core-dump access.
121            | libc::PR_SET_DUMPABLE
122            | libc::PR_GET_DUMPABLE
123            // PR_SET_TIMERSLACK and PR_GET_TIMERSLACK are deliberately absent:
124            // `handle_prctl` emulates them against per-thread Detcore state.
125            // AUTONOMOUS-BOT-IMPLEMENTED
126            // TODO-HUMAN-REVIEW(#802)
127            //
128            // PR_{SET,GET}_KEEPCAPS only read/toggle the calling thread's
129            // "keep capabilities across a UID change" flag. The result is a pure
130            // function of the guest's own prior prctl calls (0/1), never host
131            // state, so passthrough is deterministic and bitwise-identical across
132            // runs. Supporting it lets `setpriv` (and the `date`/privilege-drop
133            // wrappers that call it) run under --strict instead of aborting with
134            // "keep process capabilities failed: Function not implemented".
135            | libc::PR_SET_KEEPCAPS
136            | libc::PR_GET_KEEPCAPS
137            // AUTONOMOUS-BOT-IMPLEMENTED
138            // TODO-HUMAN-REVIEW(#824)
139            //
140            // PR_{SET,GET}_PDEATHSIG only read/toggle the calling thread's
141            // parent-death-signal attribute. PR_SET_PDEATHSIG validates its
142            // signal argument (a valid signal or 0 succeeds, anything else
143            // faults EINVAL) and PR_GET_PDEATHSIG reports the value the guest
144            // previously set. The result is a pure function of the guest's own
145            // prior prctl calls and its argument, never host state, so
146            // passthrough is deterministic and bitwise-identical across runs.
147            // The registered signal fires when the parent thread exits, and the
148            // kernel posts it at a moment set by host timing, so `handle_prctl`
149            // first records a nonzero PR_SET_PDEATHSIG signal as host-timed for
150            // the caller's process: no gated wait of that process ends for it
151            // (https://github.com/rrnewton/hermit/issues/3146). Supporting it lets
152            // `setpriv --pdeathsig` run under --strict instead of aborting with
153            // "set parent death signal failed: Function not implemented".
154            | libc::PR_SET_PDEATHSIG
155            | libc::PR_GET_PDEATHSIG
156            // AUTONOMOUS-BOT-IMPLEMENTED
157            // TODO-HUMAN-REVIEW(PR-1125): Review KVM capability-control prctl forwarding.
158            //
159            // PR_CAPBSET_DROP and PR_CAP_AMBIENT change and read the calling
160            // thread's capability bounding and ambient sets. They reach the
161            // kernel on every backend (the host's, or KVM's provider), as
162            // PR_CAPBSET_READ and capget already do, so each backend answers as
163            // native Linux does for the same credentials: a drop or a raise
164            // the caller is not entitled to fails with EPERM, an unknown
165            // capability with EINVAL. The result is a function of the
166            // container's initial credentials and the guest's own earlier
167            // calls. They used to reach the kernel on KVM only, and fail with
168            // ENOSYS everywhere else.
169            | libc::PR_CAPBSET_DROP
170            | libc::PR_CAP_AMBIENT
171    )
172}

174/// Whether Detcore can admit `PR_SET_CHILD_SUBREAPER`: threads are
175/// sequentialized, so every exit reaches the scheduler through
176/// `deregister_thread` and re-parents there (`Scheduler::reparent_orphans`);
177/// without that, Detcore's record never learns the orphan moved while it still
178/// decides ECHILD. And on `backend`, its own record of children decides waits
179/// (`tracks_process_children`), the kernel rather than the backend performs
180/// them (`!emulates_child_waits`), and a process's exit reaches Detcore only
181/// after the kernel's `exit_notify` has re-parented its children (neither
182/// asynchronous nor separately reported physical exits), so the record agrees
183/// with the kernel. Today that is ptrace, and e9patch preprocessing run under
184/// ptrace, in a sequentialized run.
185fn models_child_subreapers(sequentialize_threads: bool, backend: &BackendCapabilities) -> bool {
186    sequentialize_threads
187        && backend.tracks_process_children
188        && !backend.emulates_child_waits
189        && !backend.process_exits_complete_asynchronously
190        && !backend.reports_physical_process_exits
191}

239/// Write get_mempolicy's output as Linux's `put_user`/`copy_to_user` do:
240/// through the guest's page protections, so a buffer that cannot take all the
241/// bytes is `EFAULT`. Any other failure is the backend's, not a guest fault,
242/// and is reported as a tool error rather than handed to the guest as an
243/// errno get_mempolicy never returns (Claude re-check of
244/// https://github.com/rrnewton/hermit/pull/4249).
245fn write_get_mempolicy_output<M: MemoryAccess>(
246    memory: &mut M,
247    address: AddrMut<'_, u8>,
248    bytes: &[u8],
249) -> Result<(), Error> {
250    match memory.write_with_user_access(address, bytes) {
251        Ok(written) if written == bytes.len() => Ok(()),
252        Ok(_) | Err(Errno::EFAULT) => Err(Errno::EFAULT.into()),
253        Err(errno) => Err(Error::Tool(anyhow::anyhow!(
254            "get_mempolicy output write failed: {errno}"
255        ))),
256    }
257}

260    let mut ret: [i8; 65] = [0; 65];

262        ret[i] = ch as i8;
263    }
264    ret

270// AUTONOMOUS-BOT-IMPLEMENTED
271// TODO-HUMAN-REVIEW(PR-1096): Review the backend-independent random-device stream.
272fn canonical_random_device_byte(seed: u64, index: u64) -> u8 {
273    let seed_byte = seed.rotate_right(((index % 8) * 8) as u32) as u8;
274    (index as u8)
275        .wrapping_mul(RANDOM_DEVICE_BYTE_STRIDE)
276        .wrapping_add(RANDOM_DEVICE_FIRST_BYTE)
277        ^ seed_byte
278}

280/// Scatter one stream through an already-imported array. Scratch space is
281/// independent of request size; only EFAULT becomes a successful copied prefix.
282fn fill_canonical_random_iovecs(
283    memory: &mut impl MemoryAccess,
284    iovecs: &[crate::iovecs::ImportedIovec],
285    seed: u64,
286    stream_offset: u64,
287    hasher: &mut DefaultHasher,
288) -> Result<usize, Error> {
289    let mut local_words = [0_u64; RANDOM_FILL_CHUNK_BYTES / std::mem::size_of::<u64>()];
290    let mut written = 0_usize;
291    for iov in iovecs {
292        let mut segment_written = 0;
293        while segment_written < iov.len {
294            let remote_chunk = match iov
295                .base
296                .checked_add(segment_written)
297                .and_then(AddrMut::<u8>::from_raw)
298            {
299                Some(address) => address,
300                None if written == 0 => return Err(Errno::EFAULT.into()),
301                None => return Ok(written),
302            };
303            let chunk_len = (iov.len - segment_written).min(RANDOM_FILL_CHUNK_BYTES);
304            let local_buf = unsafe {
305                std::slice::from_raw_parts_mut(local_words.as_mut_ptr().cast::<u8>(), chunk_len)
306            };
307            for (index, byte) in local_buf.iter_mut().enumerate() {
308                *byte = canonical_random_device_byte(
309                    seed,
310                    stream_offset
311                        .saturating_add(written as u64)
312                        .saturating_add(index as u64),
313                );
314            }
315            let n = match write_random_chunk(memory, remote_chunk, local_buf) {
316                Ok(n) => n,
317                Err(Errno::EFAULT) if written > 0 => return Ok(written),
318                Err(error) => return Err(crate::random::copy_error(error)),
319            };
320            if n == 0 && written == 0 {
321                return Err(Errno::EFAULT.into());
322            }
323            if cfg!(debug_assertions) {
324                Hash::hash_slice(&local_buf[..n], hasher);
325            }
326            written += n;
327            segment_written += n;
328            if n < chunk_len {
329                return Ok(written);
330            }
331        }
332    }
333    Ok(written)
334}

365    // AUTONOMOUS-BOT-IMPLEMENTED
366    // TODO-HUMAN-REVIEW(#539): Confirm the virtual arch_prctl control policy.
367    /// Preserve thread-local bases while hiding host CPU feature controls.
368    pub async fn handle_arch_prctl<G: Guest<Self>>(
369        &self,
370        guest: &mut G,
371        call: syscalls::ArchPrctl,
372    ) -> Result<i64, Error> {
373        let cpuid_uses_faulting = self.cfg.virtualize_cpuid && guest.has_cpuid_interception();
374        match call.cmd() {
375            ArchPrctlCmd::ARCH_SET_FS(_)
376            | ArchPrctlCmd::ARCH_SET_GS(_)
377            | ArchPrctlCmd::ARCH_GET_FS(_)
378            | ArchPrctlCmd::ARCH_GET_GS(_) => Ok(guest.inject(call).await?),
379
380            // When Reverie successfully disables native CPUID, Detcore answers its fault from a
381            // fixed table. Preserve that backend state and reject attempts to re-enable CPUID.
382            ArchPrctlCmd::ARCH_GET_CPUID(_) if cpuid_uses_faulting => Ok(0),
383            ArchPrctlCmd::ARCH_SET_CPUID(value) if cpuid_uses_faulting => {
384                if value == 0 {
385                    Ok(0)
386                } else {
387                    Err(Errno::EPERM.into())
388                }
389            }
390            // Reverie cannot faithfully deliver a CPUID fault requested by the tracee. In
391            // explicit host-CPUID mode, expose a fixed enabled control state and reject disable.
392            ArchPrctlCmd::ARCH_GET_CPUID(_) if !self.cfg.virtualize_cpuid => Ok(1),
393            ArchPrctlCmd::ARCH_SET_CPUID(value) if !self.cfg.virtualize_cpuid => {
394                if value == 0 {
395                    Err(Errno::EPERM.into())
396                } else {
397                    Ok(0)
398                }
399            }
400
401            // Ptrace hosts without CPUID-faulting support retain the kernel's honest state.
402            ArchPrctlCmd::ARCH_GET_CPUID(_) | ArchPrctlCmd::ARCH_SET_CPUID(_) => {
403                Ok(guest.inject(call).await?)
404            }
405
406            // Under x86-64-v3 the guest's XGETBV and XSAVE run on the host, so
407            // the xstate queries answer with the host's masks, captured at
408            // startup on a host without dynamic features and bound in the
409            // recording; the requests answer as Linux does on such a host:
410            // EOPNOTSUPP below XFEATURE_MAX and EINVAL from it up
411            // (https://github.com/rrnewton/hermit/issues/4243).
412            ArchPrctlCmd::Other(
413                cmd @ (ARCH_GET_XCOMP_SUPP | ARCH_GET_XCOMP_PERM | ARCH_GET_XCOMP_GUEST_PERM),
414                addr,
415            ) if let Some(host) = self.v3_host_xstate() => {
416                let mask = match cmd {
417                    ARCH_GET_XCOMP_SUPP => host.xcomp_supp,
418                    ARCH_GET_XCOMP_PERM => host.xcomp_perm,
419                    _ => host.xcomp_guest_perm,
420                };
421                self.write_arch_prctl_u64(guest, addr, mask)
422            }
423            ArchPrctlCmd::Other(ARCH_REQ_XCOMP_PERM | ARCH_REQ_XCOMP_GUEST_PERM, index)
424                if let Some(host) = self.v3_host_xstate() =>
425            {
426                if index >= host.xfeature_max as usize {
427                    Err(Errno::EINVAL.into())
428                } else {
429                    Err(Errno::EOPNOTSUPP.into())
430                }
431            }
432            // Expose a conservative virtual CPU with no optional extended-state permissions.
433            ArchPrctlCmd::Other(
434                ARCH_GET_XCOMP_SUPP | ARCH_GET_XCOMP_PERM | ARCH_GET_XCOMP_GUEST_PERM,
435                addr,
436            ) => self.write_arch_prctl_u64(guest, addr, 0),
437            ArchPrctlCmd::Other(ARCH_REQ_XCOMP_PERM | ARCH_REQ_XCOMP_GUEST_PERM, _) => {
438                Err(Errno::EINVAL.into())
439            }
440
441            // Keep shadow stacks disabled in the virtual policy. Disabling an already-disabled
442            // feature is idempotent; enable/lock/unlock requests cannot be honored.
443            ArchPrctlCmd::Other(ARCH_SHSTK_STATUS, addr) => {
444                self.write_arch_prctl_u64(guest, addr, 0)
445            }
446            ArchPrctlCmd::Other(ARCH_SHSTK_DISABLE, features)
447                if features != 0 && features & !ARCH_SHSTK_VALID_MASK == 0 =>
448            {
449                Ok(0)
450            }
451            ArchPrctlCmd::Other(ARCH_SHSTK_DISABLE, _)
452            | ArchPrctlCmd::Other(ARCH_SHSTK_ENABLE | ARCH_SHSTK_LOCK | ARCH_SHSTK_UNLOCK, _) => {
453                Err(Errno::EINVAL.into())
454            }
455
456            ArchPrctlCmd::Other(_, _) => Err(Errno::EINVAL.into()),
457        }
458    }

460    // AUTONOMOUS-BOT-IMPLEMENTED
461    // TODO-HUMAN-REVIEW(#663)
462    /// Preserve deterministic Ruby thread controls, report the container's fixed
463    /// capability bounding set, and reject options that expose unmodeled process
464    /// or host state.
465    pub async fn handle_prctl<G: Guest<Self>>(
466        &self,
467        guest: &mut G,
468        call: syscalls::Prctl,
469    ) -> Result<i64, Error> {
470        match call.option() {
471            // The capability bounding set is fixed by the container launch policy.
472            libc::PR_CAPBSET_READ => Ok(self.record_or_replay(guest, call).await?),
473            // AUTONOMOUS-BOT-IMPLEMENTED
474            // TODO-HUMAN-REVIEW(PR-2150): Timer slack is shared with the
475            // virtual `/proc/<tid>/timerslack_ns` channel.  Never pass either
476            // setter through to the physical tracee: a large physical slack
477            // changes wake timing on Detcore's remaining host-timed waits.
478            libc::PR_SET_TIMERSLACK => {
479                let requested = call.arg2();
480                let state = guest.thread_state_mut();
481                // Linux treats zero as reset-to-default. Hermit virtualizes the
482                // Linux scheduling policy as SCHED_OTHER, so the kernel's
483                // RT/DL no-op branch is unreachable in the guest model.
484                state.timer_slack_ns = if requested == 0 {
485                    state.default_timer_slack_ns
486                } else {
487                    requested
488                };
489                Ok(0)
490            }
491            libc::PR_GET_TIMERSLACK => Ok(guest.thread_state().timer_slack_ns as i64),
492            // The kernel sends a parent-death signal process-directed
493            // (`group_send_sig_info` with `PIDTYPE_TGID`) when the parent thread
494            // exits, at a moment set by host timing. Record it in this turn,
495            // before the call reaches the kernel, for every thread of the
496            // caller's process, and never forget it: a later disarm or `exec`
497            // does not remove it (`Scheduler::record_host_timed_signals`). Linux
498            // accepts 0 (disarm) and 1 to 64; anything else fails with EINVAL in
499            // the kernel and arms nothing.
500            // A parent-death signal Detcore sends itself is held by Detcore,
501            // not the kernel (armed with 0); any other comes from the kernel.
502            libc::PR_GET_PDEATHSIG if emulates_parent_death_signal(guest.config()) => {
503                let held = crate::tool_global::parent_death_signal(guest, None).await;
504                if held == 0 {
505                    return self.passthrough(guest, call.into()).await;
506                }
507                let target =
508                    AddrMut::<libc::c_int>::from_raw(call.arg2() as usize).ok_or(Errno::EFAULT)?;
509                guest
510                    .memory()
511                    .write_value(target, &held)
512                    .map_err(|_| Errno::EFAULT)?;
513                Ok(0)
514            }
515            libc::PR_SET_PDEATHSIG => {
516                let signal = call.arg2();
517                if signal == libc::SIGALRM as libc::c_ulong {
518                    refuse_sigalrm(guest, SigalrmControl::ArmProducer).await?;
519                }
520                // A signal Detcore sends itself when the parent thread leaves
521                // (`Scheduler::deliver_parent_death_signals`,
522                // https://github.com/rrnewton/hermit/issues/4440): the kernel
523                // is armed with 0, so it never posts a copy of its own at a
524                // host-timed moment, and Detcore keeps the value. Any other
525                // value takes the kernel's path below, and disarms Detcore's
526                // only once the kernel has accepted it (an invalid one, which
527                // the kernel refuses with EINVAL, changes nothing).

887    /// Give the backend a `setsid` or `setpgid` the model accepted, so that a
888    /// kernel that keeps process groups stays in step with the model. A
889    /// backend that keeps none says so with a typed refusal
890    /// (`UnsupportedOperation::ProcessGroups`), and the model's answer stands.
891    /// Any other failure, whatever its errno, means the backend's process
892    /// groups disagree with the model's: refused by name rather than answered
893    /// with the backend's errno. An errno alone never stands for a refusal: a
894    /// kernel ENOSYS for another reason would otherwise read as success.
895    async fn mirror_group_change<G: Guest<Self>>(
896        &self,
897        guest: &mut G,
898        call: reverie::syscalls::Syscall,
899    ) -> Result<(), Error> {
900        let result = guest.inject(call).await;
901        // Ask before anything else is injected: the next injection replaces
902        // the backend's record.
903        let refusal = result
904            .is_err()
905            .then(|| guest.take_unsupported_refusal())
906            .flatten();
907        match result {
908            Ok(_) => Ok(()),
909            Err(_)
910                if refusal.is_some_and(|refusal| {
911                    refusal.operation == reverie::UnsupportedOperation::ProcessGroups
912                }) =>
913            {
914                Ok(())
915            }
916            Err(errno) => {
917                let sysno = reverie::syscalls::SyscallInfo::number(&call);
918                tracing::error!(
919                    "[detcore, dtid {}] {} was accepted by Detcore's process-group model but the backend answered {}: its process groups disagree with the model's (slice B3a-4c of https://github.com/rrnewton/hermit/issues/4266)",
920                    guest.thread_state().dettid,
921                    sysno,
922                    errno,
923                );
924                self.refuse_unserviceable_operation(guest, sysno, errno)
925                    .await
926                    .map(|_| ())
927            }
928        }
929    }

1082    /// get_mempolicy under Hermit. The container exposes a single virtual NUMA
1083    /// node, so the effective policy is always the default and every address
1084    /// resolves to node 0. The result is fully emulated (never injected), so it
1085    /// is bitwise-identical across the two --verify runs and under record/replay,
1086    /// removing the host-NUMA-topology dependence a passthrough would introduce.
1087    ///
1088    /// It follows Linux's `kernel_get_mempolicy` and `do_get_mempolicy`
1089    /// (mm/mempolicy.c) for a kernel with one node (`nr_node_ids` 1):
1090    /// - a node mask with `maxnode` below 1 is EINVAL;
1091    /// - an unknown flag (any bit beyond `MPOL_F_NODE`, `MPOL_F_ADDR` and
1092    ///   `MPOL_F_MEMS_ALLOWED`, including those above bit 31 of the raw
1093    ///   argument, which Reverie's typed call drops), `MPOL_F_MEMS_ALLOWED`
1094    ///   with `MPOL_F_NODE` or `MPOL_F_ADDR`, and an address without
1095    ///   `MPOL_F_ADDR` are EINVAL;
1096    /// - `*policy` is 0: `MPOL_DEFAULT`, or node 0 for `MPOL_F_NODE`;
1097    /// - the node mask is written as `copy_nodes_to_user` writes it: node 0
1098    ///   alone for `MPOL_F_MEMS_ALLOWED`, as `/proc/<pid>/status` reports
1099    ///   `Mems_allowed_list` (<https://github.com/rrnewton/hermit/issues/4214>),
1100    ///   and no node for the default policy, in `ALIGN(maxnode - 1, 64) / 8`
1101    ///   bytes (none for a `maxnode` of 1), EINVAL beyond a page;
1102    /// - both are written with the guest's own access rights, so a read-only or
1103    ///   unmapped buffer is EFAULT, as from `put_user` and `copy_to_user`.
1104    ///
1105    /// Limits: `set_mempolicy` and `mbind` succeed without recording a policy,
1106    /// so this always reports the default policy, where Linux reports, say,
1107    /// `MPOL_INTERLEAVE` after `set_mempolicy(MPOL_INTERLEAVE)`. For the same
1108    /// reason `MPOL_F_NODE` without `MPOL_F_ADDR` answers node 0, as Linux does
1109    /// for an interleave policy (libnuma's `numa_get_interleave_node`), where
1110    /// it refuses it under the default policy. `MPOL_F_ADDR` does not check
1111    /// that the address is mapped, where Linux answers EFAULT for an unmapped
1112    /// one.
1113    // AUTONOMOUS-BOT-IMPLEMENTED
1114    // TODO-HUMAN-REVIEW(#720)
1115    pub async fn handle_get_mempolicy<G: Guest<Self>>(
1116        &self,
1117        guest: &mut G,
1118        call: syscalls::GetMempolicy,
1119    ) -> Result<i64, Error> {
1120        const MPOL_F_NODE: u64 = 1;
1121        const MPOL_F_ADDR: u64 = 2;
1122        const MPOL_F_MEMS_ALLOWED: u64 = 4;
1123        const PAGE: u64 = 4096;
1124        let mask = call.nodemask();
1125        if mask.is_some() && call.maxnode() < 1 {
1126            return Err(Errno::EINVAL.into());
1127        }
1128        // The flags argument is an unsigned long; Reverie decodes it as u32.
1129        let flags = guest.regs().await.r8;
1130        if flags & !(MPOL_F_NODE | MPOL_F_ADDR | MPOL_F_MEMS_ALLOWED) != 0 {
1131            return Err(Errno::EINVAL.into());
1132        }
1133        let mems_allowed = flags & MPOL_F_MEMS_ALLOWED != 0;
1134        if mems_allowed {
1135            if flags & (MPOL_F_NODE | MPOL_F_ADDR) != 0 {
1136                return Err(Errno::EINVAL.into());
1137            }
1138        } else if flags & MPOL_F_ADDR == 0 && call.addr().is_some() {
1139            return Err(Errno::EINVAL.into());
1140        }
1141        if let Some(policy) = call.policy() {
1142            let policy = AddrMut::<u8>::from_raw(policy.as_raw()).ok_or(Errno::EFAULT)?;
1143            write_get_mempolicy_output(&mut guest.memory(), policy, &0i32.to_ne_bytes())?;
1144        }
1145        if let Some(mask) = mask {
1146            let bytes = (call.maxnode() - 1).div_ceil(64) * 8;
1147            if bytes > PAGE.max(8) {
1148                return Err(Errno::EINVAL.into());
1149            }
1150            let mut words = vec![0u8; bytes as usize];
1151            if mems_allowed && !words.is_empty() {
1152                words[0] = 1;
1153            }
1154            if !words.is_empty() {
1155                let mask = AddrMut::<u8>::from_raw(mask.as_raw()).ok_or(Errno::EFAULT)?;
1156                write_get_mempolicy_output(&mut guest.memory(), mask, &words)?;
1157            }
1158        }
1159        Ok(0)
1160    }
