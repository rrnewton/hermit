42const SECCOMP_SET_MODE_STRICT: u32 = 0;
43const SECCOMP_SET_MODE_FILTER: u32 = 1;
44const SECCOMP_GET_ACTION_AVAIL: u32 = 2;
45const SECCOMP_GET_NOTIF_SIZES: u32 = 3;
46const SECCOMP_FILTER_FLAG_TSYNC: u32 = 1;

48fn seccomp_result(op: u32, flags: u32, has_args: bool) -> Result<i64, Errno> {
49    if op > SECCOMP_GET_NOTIF_SIZES {
50        return Err(Errno::EINVAL);
51    }
52    if op == SECCOMP_SET_MODE_STRICT && (flags != 0 || has_args) {
53        return Err(Errno::EINVAL);
54    }
55    if op == SECCOMP_SET_MODE_FILTER && flags & !SECCOMP_FILTER_FLAG_TSYNC != 0 {
56        return Err(Errno::EINVAL);
57    }
58    if matches!(
59        op,
60        SECCOMP_SET_MODE_FILTER | SECCOMP_GET_ACTION_AVAIL | SECCOMP_GET_NOTIF_SIZES
61    ) && !has_args
62    {
63        return Err(Errno::EFAULT);
64    }
65
66    // Hermit cannot enforce a guest-installed BPF policy in every backend.
67    // Report the operation as unsupported instead of claiming a filter exists.
68    Err(Errno::EOPNOTSUPP)
69}

71fn is_supported_prctl_option(option: libc::c_int) -> bool {
72    matches!(
73        option,
74        libc::PR_SET_NAME
75            | libc::PR_GET_NAME
76            | libc::PR_SET_THP_DISABLE
77            | libc::PR_GET_THP_DISABLE
78            // AUTONOMOUS-BOT-IMPLEMENTED
79            // TODO-HUMAN-REVIEW(#919)
80            //
81            // Dumpability is per-process state initialized deterministically by
82            // ordinary exec and subsequently controlled only by the guest. Keep
83            // Linux's 0/1 validation and state transitions rather than refusing
84            // applications that explicitly disable or restore core-dump access.
85            | libc::PR_SET_DUMPABLE
86            | libc::PR_GET_DUMPABLE
87            // TODO-HUMAN-REVIEW(PR-924)
88            //
89            // Timer slack is a per-thread kernel control for coalescing timer
90            // wakeups. Detcore virtualizes sleeps and thread scheduling, so
91            // passthrough preserves Linux's set/get behavior without changing
92            // guest ordering or logical time. Recording the result also keeps
93            // the inherited default stable during replay.
94            | libc::PR_SET_TIMERSLACK
95            | libc::PR_GET_TIMERSLACK
96            // AUTONOMOUS-BOT-IMPLEMENTED
97            // TODO-HUMAN-REVIEW(#802)
98            //
99            // PR_{SET,GET}_KEEPCAPS only read/toggle the calling thread's
100            // "keep capabilities across a UID change" flag. The result is a pure
101            // function of the guest's own prior prctl calls (0/1), never host
102            // state, so passthrough is deterministic and bitwise-identical across
103            // runs. Supporting it lets `setpriv` (and the `date`/privilege-drop
104            // wrappers that call it) run under --strict instead of aborting with
105            // "keep process capabilities failed: Function not implemented".
106            | libc::PR_SET_KEEPCAPS
107            | libc::PR_GET_KEEPCAPS
108            // AUTONOMOUS-BOT-IMPLEMENTED
109            // TODO-HUMAN-REVIEW(#824)
110            //
111            // PR_{SET,GET}_PDEATHSIG only read/toggle the calling thread's
112            // parent-death-signal attribute. PR_SET_PDEATHSIG validates its
113            // signal argument (a valid signal or 0 succeeds, anything else
114            // faults EINVAL) and PR_GET_PDEATHSIG reports the value the guest
115            // previously set. The result is a pure function of the guest's own
116            // prior prctl calls and its argument, never host state, so
117            // passthrough is deterministic and bitwise-identical across runs.
118            // The registered signal only ever fires on parent death, which is a
119            // deterministically scheduled event under Hermit. Supporting it lets
120            // `setpriv --pdeathsig` run under --strict instead of aborting with
121            // "set parent death signal failed: Function not implemented".
122            | libc::PR_SET_PDEATHSIG
123            | libc::PR_GET_PDEATHSIG
124    )
125}

133/// Is `which` one of the Linux `PRIO_*` target selectors for get/setpriority?

136        || which == libc::PRIO_PGRP as i32
137        || which == libc::PRIO_USER as i32

179fn from_str(s: &str) -> [i8; 65] {
180    let mut ret: [i8; 65] = [0; 65];
181    for (i, ch) in s.bytes().take(64).enumerate() {
182        ret[i] = ch as i8;
183    }
184    ret
185}

189// AUTONOMOUS-BOT-IMPLEMENTED

191fn validate_getrandom_flags(flags: usize) -> Result<(), Errno> {
192    let flags = flags as u32;
193    let random = flags & libc::GRND_RANDOM != 0;
194    let insecure = flags & libc::GRND_INSECURE != 0;
195
196    if flags & !GETRANDOM_ALLOWED_FLAGS != 0 || (random && insecure) {
197        Err(Errno::EINVAL)
198    } else {
199        Ok(())
200    }
201}

203const RANDOM_FILL_CHUNK_BYTES: usize = 4096;
204const RANDOM_DEVICE_BYTE_STRIDE: u8 = 73;
205const RANDOM_DEVICE_FIRST_BYTE: u8 = 41;
206// Linux's import_ubuf clamps getrandom requests to MAX_RW_COUNT on x86_64.
207const GETRANDOM_MAX_BYTES: usize = (i32::MAX as usize) & !4095;

209fn getrandom_request_len(requested: usize) -> usize {
210    requested.min(GETRANDOM_MAX_BYTES)

213fn write_random_chunk(
214    mut memory: impl MemoryAccess,
215    remote_buf: AddrMut<u8>,
216    local_buf: &[u8],
217) -> Result<usize, Errno> {
218    const PTRACE_WORD_SPLIT: usize = std::mem::size_of::<u64>() / 2;
219
220    if local_buf.len() != std::mem::size_of::<u64>() {
221        return memory.write(remote_buf, local_buf);
222    }
223
224    // safeptrace uses PTRACE_POKEDATA for exactly eight bytes, which bypasses guest page
225    // protections. Split that case so getrandom observes the same EFAULT boundary as Linux.
226    let first = memory.write(remote_buf, &local_buf[..PTRACE_WORD_SPLIT])?;
227    if first < PTRACE_WORD_SPLIT {
228        return Ok(first);
229    }
230    let Some(second_buf) = remote_buf
231        .as_raw()
232        .checked_add(PTRACE_WORD_SPLIT)
233        .and_then(AddrMut::<u8>::from_raw)
234    else {
235        return Ok(first);
236    };
237    match memory.write(second_buf, &local_buf[PTRACE_WORD_SPLIT..]) {
238        Ok(second) => Ok(first + second),
239        Err(_) => Ok(first),
240    }
241}

243// AUTONOMOUS-BOT-IMPLEMENTED
244// TODO-HUMAN-REVIEW(PR-1096): Review the backend-independent random-device stream.
245fn canonical_random_device_byte(seed: u64, index: u64) -> u8 {
246    let seed_byte = seed.rotate_right(((index % 8) * 8) as u32) as u8;
247    (index as u8)
248        .wrapping_mul(RANDOM_DEVICE_BYTE_STRIDE)
249        .wrapping_add(RANDOM_DEVICE_FIRST_BYTE)
250        ^ seed_byte
251}

275    // AUTONOMOUS-BOT-IMPLEMENTED
276    // TODO-HUMAN-REVIEW(#539): Confirm the virtual arch_prctl control policy.
277    /// Preserve thread-local bases while hiding host CPU feature controls.
278    pub async fn handle_arch_prctl<G: Guest<Self>>(
279        &self,
280        guest: &mut G,
281        call: syscalls::ArchPrctl,
282    ) -> Result<i64, Error> {
283        let cpuid_uses_backend_policy =
284            self.cfg.virtualize_cpuid && self.cfg.cpuid_virtualized_by_backend;
285        let cpuid_uses_faulting = self.cfg.virtualize_cpuid && guest.has_cpuid_interception();
286        match call.cmd() {
287            ArchPrctlCmd::ARCH_SET_FS(_)
288            | ArchPrctlCmd::ARCH_SET_GS(_)
289            | ArchPrctlCmd::ARCH_GET_FS(_)
290            | ArchPrctlCmd::ARCH_GET_GS(_) => Ok(guest.inject(call).await?),
291
292            // KVM installs a deterministic CPUID table while leaving the instruction enabled.
293            ArchPrctlCmd::ARCH_GET_CPUID(_) if cpuid_uses_backend_policy => Ok(1),
294            ArchPrctlCmd::ARCH_SET_CPUID(value) if cpuid_uses_backend_policy => {
295                if value == 0 {
296                    Err(Errno::EPERM.into())
297                } else {
298                    Ok(0)
299                }
300            }
301
302            // When Reverie successfully disables native CPUID, Detcore answers its fault from a
303            // fixed table. Preserve that backend state and reject attempts to re-enable CPUID.
304            ArchPrctlCmd::ARCH_GET_CPUID(_) if cpuid_uses_faulting => Ok(0),
305            ArchPrctlCmd::ARCH_SET_CPUID(value) if cpuid_uses_faulting => {
306                if value == 0 {
307                    Ok(0)
308                } else {
309                    Err(Errno::EPERM.into())
310                }
311            }
312            // Reverie cannot faithfully deliver a CPUID fault requested by the tracee. In
313            // explicit host-CPUID mode, expose a fixed enabled control state and reject disable.
314            ArchPrctlCmd::ARCH_GET_CPUID(_) if !self.cfg.virtualize_cpuid => Ok(1),
315            ArchPrctlCmd::ARCH_SET_CPUID(value) if !self.cfg.virtualize_cpuid => {
316                if value == 0 {
317                    Err(Errno::EPERM.into())
318                } else {
319                    Ok(0)
320                }
321            }
322
323            // Ptrace hosts without CPUID-faulting support retain the kernel's honest state.
324            ArchPrctlCmd::ARCH_GET_CPUID(_) | ArchPrctlCmd::ARCH_SET_CPUID(_) => {
325                Ok(guest.inject(call).await?)
326            }
327
328            // Expose a conservative virtual CPU with no optional extended-state permissions.
329            ArchPrctlCmd::Other(
330                ARCH_GET_XCOMP_SUPP | ARCH_GET_XCOMP_PERM | ARCH_GET_XCOMP_GUEST_PERM,
331                addr,
332            ) => self.write_arch_prctl_u64(guest, addr, 0),
333            ArchPrctlCmd::Other(ARCH_REQ_XCOMP_PERM | ARCH_REQ_XCOMP_GUEST_PERM, _) => {
334                Err(Errno::EINVAL.into())
335            }
336
337            // Keep shadow stacks disabled in the virtual policy. Disabling an already-disabled
338            // feature is idempotent; enable/lock/unlock requests cannot be honored.
339            ArchPrctlCmd::Other(ARCH_SHSTK_STATUS, addr) => {
340                self.write_arch_prctl_u64(guest, addr, 0)
341            }
342            ArchPrctlCmd::Other(ARCH_SHSTK_DISABLE, features)
343                if features != 0 && features & !ARCH_SHSTK_VALID_MASK == 0 =>
344            {
345                Ok(0)
346            }
347            ArchPrctlCmd::Other(ARCH_SHSTK_DISABLE, _)
348            | ArchPrctlCmd::Other(ARCH_SHSTK_ENABLE | ARCH_SHSTK_LOCK | ARCH_SHSTK_UNLOCK, _) => {
349                Err(Errno::EINVAL.into())
350            }
351
352            ArchPrctlCmd::Other(_, _) => Err(Errno::EINVAL.into()),
353        }
354    }

358    /// Preserve deterministic Ruby thread controls, report the container's fixed
359    /// capability bounding set, and reject options that expose unmodeled process
360    /// or host state.
361    pub async fn handle_prctl<G: Guest<Self>>(
362        &self,
363        guest: &mut G,
364        call: syscalls::Prctl,
365    ) -> Result<i64, Error> {
366        match call.option() {
367            // The capability bounding set is fixed by the container launch policy.
368            libc::PR_CAPBSET_READ => Ok(self.record_or_replay(guest, call).await?),
369            option
370                if guest.config().backend_virtualizes_capability_prctls
371                    && is_backend_virtualized_capability_prctl(option) =>
372            {
373                self.passthrough(guest, call.into()).await
374            }
375            option if is_supported_prctl_option(option) => {
376                self.passthrough(guest, call.into()).await
377            }
378            _ => Err(Errno::ENOSYS.into()),
379        }
380    }

436        if flags != 0 {
437            return Err(Errno::EINVAL.into());
438        }
439
440        // Linux interprets pidfd as an int. Preserve its deterministic invalid-fd
441        // rejection, but never let a valid host pidfd alter another process's memory.

444        } else {
445            Err(Errno::EPERM.into())
446        }

449    /// Fill guest memory from the deterministic PRNG owned by the current thread.
450    pub(super) fn fill_random_bytes<G: Guest<Self>>(
451        &self,
452        guest: &mut G,
453        remote_buf: AddrMut<u8>,
454        len: usize,
455        source: &str,
456    ) -> Result<usize, Error> {
457        let mut local_words = [0_u64; RANDOM_FILL_CHUNK_BYTES / std::mem::size_of::<u64>()];
458        let mut hasher = DefaultHasher::new();
459        let mut written = 0;
460
461        while written < len {
462            let remote_chunk = match remote_buf
463                .as_raw()
464                .checked_add(written)
465                .and_then(AddrMut::<u8>::from_raw)
466            {
467                Some(address) => address,
468                None if written == 0 => return Err(Errno::EFAULT.into()),
469                None => break,
470            };
471            let chunk_len = (len - written).min(RANDOM_FILL_CHUNK_BYTES);
472            // safeptrace's 8-byte write fast path currently requires an aligned source buffer.
473            let local_buf = unsafe {
474                std::slice::from_raw_parts_mut(local_words.as_mut_ptr().cast::<u8>(), chunk_len)
475            };
476            guest.thread_state_mut().thread_prng().fill(local_buf);
477            let n = match write_random_chunk(guest.memory(), remote_chunk, local_buf) {
478                Ok(n) => n,
479                Err(_) if written > 0 => break,
480                Err(error) => return Err(error.into()),
481            };
482            if n == 0 {
483                if written == 0 {
484                    return Err(Errno::EFAULT.into());
485                }
486                break;
487            }
488            if cfg!(debug_assertions) {
489                Hash::hash_slice(&local_buf[..n], &mut hasher);
490            }
491            written += n;
492            if n < chunk_len {
493                break;
494            }
495        }
496
497        if cfg!(debug_assertions) {
498            detlog!(
499                "[dtid {}] USER RAND [{}] Filled guest memory with {} random bytes, hash of bytes: {}",
500                guest.thread_state().dettid,
501                source,
502                written,
503                hasher.finish()
504            );
505        }
506        Ok(written)
507    }

509    // AUTONOMOUS-BOT-IMPLEMENTED
510    // TODO-HUMAN-REVIEW(PR-1096): Review the backend-independent random-device stream.
511    /// Fill guest memory from the canonical stream used by every backend's
512    /// `/dev/random` and `/dev/urandom` virtualization.
513    pub(super) fn fill_random_device_bytes<G: Guest<Self>>(
514        &self,
515        guest: &mut G,
516        remote_buf: AddrMut<u8>,
517        len: usize,
518        stream_offset: u64,
519    ) -> Result<usize, Error> {
520        let seed = guest.config().rng_seed();
521        let mut local_words = [0_u64; RANDOM_FILL_CHUNK_BYTES / std::mem::size_of::<u64>()];
522        let mut hasher = DefaultHasher::new();
523        let mut written = 0;
524
525        while written < len {
526            let remote_chunk = match remote_buf
527                .as_raw()
528                .checked_add(written)
529                .and_then(AddrMut::<u8>::from_raw)
530            {
531                Some(address) => address,
532                None if written == 0 => return Err(Errno::EFAULT.into()),
533                None => break,
534            };
535            let chunk_len = (len - written).min(RANDOM_FILL_CHUNK_BYTES);
536            let local_buf = unsafe {
537                std::slice::from_raw_parts_mut(local_words.as_mut_ptr().cast::<u8>(), chunk_len)
538            };
539            for (index, byte) in local_buf.iter_mut().enumerate() {
540                *byte = canonical_random_device_byte(
541                    seed,
542                    stream_offset
543                        .saturating_add(written as u64)
544                        .saturating_add(index as u64),
545                );
546            }
547            let n = match write_random_chunk(guest.memory(), remote_chunk, local_buf) {
548                Ok(n) => n,
549                Err(_) if written > 0 => break,
550                Err(error) => return Err(error.into()),
551            };
552            if n == 0 {
553                if written == 0 {
554                    return Err(Errno::EFAULT.into());
555                }
556                break;
557            }
558            if cfg!(debug_assertions) {
559                Hash::hash_slice(&local_buf[..n], &mut hasher);
560            }
561            written += n;
562            if n < chunk_len {
563                break;
564            }
565        }
566
567        if cfg!(debug_assertions) {
568            detlog!(
569                "[dtid {}] USER RAND [/dev/[u]random] Filled guest memory with {} canonical random bytes at offset {}, hash of bytes: {}",
570                guest.thread_state().dettid,
571                written,
572                stream_offset,
573                hasher.finish()
574            );
575        }
576        Ok(written)
577    }

579    /// uname syscall
580    pub async fn handle_uname<G: Guest<Self>>(
581        &self,
582        guest: &mut G,
583        call: syscalls::Uname,
584    ) -> Result<i64, Error> {
585        let ret = self.record_or_replay(guest, call).await?;
586        if let Some(buf) = call.buf() {
587            let mut un = guest.memory().read_value(buf)?;
588            // Keep this in configured UTC: `Local` initializes libc TLS, which is unavailable
589            // while a DynamoRIO application thread is executing a client callback.
590            let epoch = guest.config().epoch;
591
592            if !guest.config().has_uts_namespace {
593                // FIXME: It should be possible to remove this once all tests
594                // are also using namespaces.
595                un.nodename = from_str(DEFAULT_HOSTNAME);
596                un.domainname = from_str(DEFAULT_HOSTNAME.split('.').next_back().unwrap_or(""));
597            }
598
599            un.release = from_str("5.2.0");
600            un.version = from_str(&format!("#1 SMP {}", epoch.format("%a %b %d %T %Z %Y")));
601            guest.memory().write_value(buf, &un)?;
602        }
603
604        Ok(ret)
605    }

608    /// Supported blocking/source-selection flags share that always-ready stream; invalid Linux
609    /// flag combinations are rejected before guest memory is touched.
610    pub async fn handle_getrandom<G: Guest<Self>>(
611        &self,
612        guest: &mut G,
613        call: syscalls::Getrandom,
614    ) -> Result<i64, Error> {
615        validate_getrandom_flags(call.flags())?;
616        let len = getrandom_request_len(call.buflen());
617        if len == 0 {
618            return Ok(0);
619        }
620
621        let buf = call.buf().ok_or(Errno::EFAULT)?;
622
623        let n = self.fill_random_bytes(guest, buf, len, "getrandom")?;
624        Ok(n as i64)
625    }

643    /// membarrier (system call).
644    ///
645    /// `membarrier(2)` issues process-wide memory barriers so that userspace can
646    /// use asymmetric fences (e.g. CPython's QSBR, RCU-style reclamation).
647    /// Detcore serializes all guest threads onto a single logical CPU with a
648    /// total memory order, so any requested barrier is *already* satisfied and
649    /// every command is a deterministic no-op. For `MEMBARRIER_CMD_QUERY` we
650    /// report the set of commands we emulate so the guest stays on this
651    /// controlled path instead of a host-dependent fallback; every other command
652    /// returns success without doing anything.
653    pub async fn handle_membarrier<G: Guest<Self>>(
654        &self,
655        guest: &mut G,
656        call: syscalls::Membarrier,
657    ) -> Result<i64, Error> {
658        // Values from <linux/membarrier.h>.
659        const MEMBARRIER_CMD_QUERY: i32 = 0;
660        const MEMBARRIER_CMD_GLOBAL: i32 = 1 << 0;
661        const MEMBARRIER_CMD_GLOBAL_EXPEDITED: i32 = 1 << 1;
662        const MEMBARRIER_CMD_REGISTER_GLOBAL_EXPEDITED: i32 = 1 << 2;
663        const MEMBARRIER_CMD_PRIVATE_EXPEDITED: i32 = 1 << 3;
664        const MEMBARRIER_CMD_REGISTER_PRIVATE_EXPEDITED: i32 = 1 << 4;
665        const SUPPORTED: i32 = MEMBARRIER_CMD_GLOBAL
666            | MEMBARRIER_CMD_GLOBAL_EXPEDITED
667            | MEMBARRIER_CMD_REGISTER_GLOBAL_EXPEDITED
668            | MEMBARRIER_CMD_PRIVATE_EXPEDITED
669            | MEMBARRIER_CMD_REGISTER_PRIVATE_EXPEDITED;
670
671        let cmd = call.cmd();
672        if cmd == MEMBARRIER_CMD_QUERY {
673            detlog!(
674                "[dtid {}] membarrier(QUERY) => reporting emulated commands {:#x}",
675                guest.thread_state().dettid,
676                SUPPORTED,
677            );
678            Ok(SUPPORTED as i64)
679        } else {
680            detlog!(
681                "[dtid {}] membarrier(cmd={}) no-op (threads are serialized on one CPU)",
682                guest.thread_state().dettid,
683                cmd,
684            );
685            Ok(0)
686        }
687    }

689    /// getcpu system call
690    pub async fn handle_getcpu<G: Guest<Self>>(
691        &self,
692        guest: &mut G,
693        call: syscalls::Getcpu,
694    ) -> Result<i64, Error> {
695        // Always set the CPU to 0.
696        if let Some(cpu) = call.cpu() {
697            guest.memory().write_value(cpu, &0)?;
698        }
699
700        // Always set the NUMA node to 0.
701        if let Some(node) = call.node() {
702            guest.memory().write_value(node, &0)?;
703        }
704
705        Ok(0)
706    }

708    /// getresuid under Hermit. Detcore presents a fixed virtual-root identity, so
709    /// the real, effective, and saved user IDs are all the constant 0. Under the
710    /// ptrace backend the guest runs inside a CLONE_NEWUSER namespace that already
711    /// maps the host uid to 0; emulating the same constant here makes in-process
712    /// backends (DBT) agree with that golden reference instead of leaking the host
713    /// uid, and the fully emulated result is bitwise-identical across --verify and
714    /// record/replay.
715    // AUTONOMOUS-BOT-IMPLEMENTED
716    // TODO-HUMAN-REVIEW(#1549)
717    pub async fn handle_getresuid<G: Guest<Self>>(
718        &self,
719        guest: &mut G,
720        call: syscalls::Getresuid,
721    ) -> Result<i64, Error> {
722        if let Some(ruid) = call.ruid() {
723            guest.memory().write_value(ruid, &0)?;
724        }
725        if let Some(euid) = call.euid() {
726            guest.memory().write_value(euid, &0)?;
727        }
728        if let Some(suid) = call.suid() {
729            guest.memory().write_value(suid, &0)?;
730        }
731        Ok(0)
732    }

736    /// constant 0, matching the ptrace CLONE_NEWUSER identity and deterministic
737    /// across --verify and record/replay.
738    // AUTONOMOUS-BOT-IMPLEMENTED
739    // TODO-HUMAN-REVIEW(#1549)

757    /// get_mempolicy under Hermit. The container exposes a single virtual NUMA
758    /// node, so the effective policy is always the default and every address
759    /// resolves to node 0. The result is fully emulated (never injected), so it
760    /// is bitwise-identical across the two --verify runs and under record/replay,
761    /// removing the host-NUMA-topology dependence a passthrough would introduce.
762    // AUTONOMOUS-BOT-IMPLEMENTED
763    // TODO-HUMAN-REVIEW(#720)
764    pub async fn handle_get_mempolicy<G: Guest<Self>>(
765        &self,
766        guest: &mut G,
767        call: syscalls::GetMempolicy,
768    ) -> Result<i64, Error> {
769        // Report MPOL_DEFAULT (0) for the current policy / node when requested.
770        // The nodemask output is left untouched: reverie exposes it as an
771        // immutable pointer, and MPOL_DEFAULT carries no node set.
772        if let Some(policy) = call.policy() {
773            guest.memory().write_value(policy, &0)?;
774        }
775        Ok(0)
776    }

778    /// move_pages under Hermit. On a single virtual NUMA node nothing can be
779    /// relocated, so report every page as residing on node 0 and succeed. The
780    /// answer is a fixed constant, so it is deterministic across --verify and
781    /// record/replay.
782    // AUTONOMOUS-BOT-IMPLEMENTED
783    // TODO-HUMAN-REVIEW(#720)
784    pub async fn handle_move_pages<G: Guest<Self>>(
785        &self,
786        guest: &mut G,
787        call: syscalls::MovePages,
788    ) -> Result<i64, Error> {
789        // When a status buffer is supplied (either a real move request or a
790        // location query with nodes == NULL), report node 0 for every page.
791        if let Some(status) = call.status() {
792            let count = call.nr_pages() as usize;
793            let zeros = vec![0i32; count];
794            guest.memory().write_values(status, &zeros)?;
795        }
796        Ok(0)
797    }
