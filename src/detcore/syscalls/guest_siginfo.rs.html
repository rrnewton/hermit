9//! A guest's `siginfo_t`, in a form a syscall handler can hold across
10//! `.await`s.
11//!
12//! Since libc 0.2.190 the Linux `siginfo_t` holds raw pointers
13//! (`si_addr`, `si_ptr`, `si_call_addr`, ...) and is neither `Send` nor
14//! `Sync`. Neither is an `AddrMut<siginfo_t>` naming one, because Reverie's
15//! address types are `Send` only when their pointee is. A handler that keeps
16//! either alive across an `.await` makes its future non-`Send`, which the
17//! tool's `handle_syscall_event` must be. Detcore copies these values between
18//! guest memory and the kernel and never dereferences their pointer fields, so
19//! the value is plain data. Read and write it as [`GuestSiginfo`], casting the
20//! guest address with `AddrMut::cast`.
21
22/// A guest `siginfo_t` as plain data; see the module documentation.
23#[derive(Clone, Copy)]
24#[repr(transparent)]
25pub(crate) struct GuestSiginfo(pub(crate) libc::siginfo_t);
26
27// SAFETY: the raw-pointer fields are guest addresses, never dereferenced.
28unsafe impl Send for GuestSiginfo {}
29// SAFETY: as for `Send`; shared access only reads plain fields.
30unsafe impl Sync for GuestSiginfo {}
31
32impl GuestSiginfo {
33    /// The all-zero value: siginfo_t has no portable initializer, and waitid
34    /// leaves all-zero in `infop` when WNOHANG finds no child (POSIX and
35    /// Linux).
36    pub(crate) fn zeroed() -> Self {
37        Self(unsafe { std::mem::zeroed() })
38    }
39
40    /// `si_pid`, as a waitid or SIGCHLD record carries it.
41    pub(crate) fn pid(&self) -> libc::pid_t {
42        unsafe { self.0.si_pid() }
43    }
44
45    /// `si_code`.
46    pub(crate) fn code(&self) -> libc::c_int {
47        self.0.si_code
48    }
49}
50
51// The wrapper keeps a handler future `Send` on every libc only if it is
52// `Send` and `Sync` even where `libc::siginfo_t` is not, an address typed by
53// it is `Send`, and it has `siginfo_t`'s layout so a guest address can be
54// cast to it. Checked at compile time.
55const _: () = {
56    fn assert_send<T: Send>() {}
57    fn assert_send_sync<T: Send + Sync>() {}
58    let _ = assert_send_sync::<GuestSiginfo>;
59    let _ = assert_send::<reverie::syscalls::AddrMut<'static, GuestSiginfo>>;
60    assert!(std::mem::size_of::<GuestSiginfo>() == std::mem::size_of::<libc::siginfo_t>());
61    assert!(std::mem::align_of::<GuestSiginfo>() == std::mem::align_of::<libc::siginfo_t>());
62};
