55/// Whether Linux accepts `clockid` in clock_gettime and clock_getres, as far as
56/// the id alone decides: the fixed clocks 0 to 9 and CLOCK_TAI (11). Linux
57/// answers EINVAL for 10 (CLOCK_SGI_CYCLE, gone since Linux 4.x) and for
58/// every id from 12 up. A negative id names a process's or thread's CPU
59/// clock, or a clock descriptor; whether it is valid depends on the process
60/// or descriptor it names, so it is accepted here as before.
61fn clock_id_is_known(clockid: ClockId) -> bool {
62    match raw_clock_id(clockid) {
63        0..=9 | 11 => true,
64        id => id < 0,
65    }
66}

74/// Linux's answer to `timer_create` for `clockid` alone, which it gives
75/// before it reads the notification (`do_timer_create`,
76/// kernel/time/posix-timers.c:448-455 at v6.12): EINVAL for an id with no
77/// clock (`clockid_to_kclock` and the `posix_clocks` table at :1529-1556: 10,
78/// and 12 up), EOPNOTSUPP for a clock with no timers (CLOCK_MONOTONIC_RAW,
79/// CLOCK_REALTIME_COARSE, CLOCK_MONOTONIC_COARSE, and a clock descriptor: a
80/// negative id with `(id & CLOCKFD_MASK) == CLOCKFD`, which names
81/// `clock_posix_dynamic`, kernel/time/posix-clock.c:334-339, whatever
82/// descriptor it encodes; include/linux/posix-timers_types.h:24-32), and
83/// EINVAL for a negative id whose clock bits name no CPU clock (`(id & 7) ==
84/// 7`: per-thread with CPUCLOCK_WHICH 3, which is CPUCLOCK_MAX, so
85/// `pid_for_clock` refuses it from the bits alone,
86/// kernel/time/posix-cpu-timers.c:63-64, and `posix_cpu_timer_create`
87/// answers EINVAL, :392-395). `None` for the clocks with timers, for the
88/// alarm clocks (whose answer depends on the host) and for the other
89/// negative ids, CPU clocks, whose answer depends on the process or thread
90/// the id names: a CPU-clock id naming a task that does not exist therefore
91/// gets the reserved-signal EPERM where Linux answers EINVAL. Only a
92/// `timer_create` whose sigevent names a reserved signal reads it, right
93/// after the copy and before the notification is checked, as Linux orders
94/// them, so that neither the refusal nor Detcore's notification checks come
95/// before Linux's clock answer. Detcore's virtual timers do not otherwise
96/// check the clock, which https://github.com/rrnewton/hermit/issues/4556
97/// tracks together with the CPU-clock case above.
98fn timer_create_clock_refusal(clockid: ClockId) -> Option<Errno> {
99    use reverie::syscalls::FromToRaw;
100    const CLOCKFD: i32 = 3;
101    const CLOCKFD_MASK: i32 = 4 | 3;
102    if !clock_id_is_known(clockid) {
103        return Some(Errno::EINVAL);
104    }
105    match clockid.into_raw() as i32 {
106        libc::CLOCK_MONOTONIC_RAW | libc::CLOCK_REALTIME_COARSE | libc::CLOCK_MONOTONIC_COARSE => {
107            Some(Errno::EOPNOTSUPP)
108        }
109        id if id < 0 && id & CLOCKFD_MASK == CLOCKFD => Some(Errno::EOPNOTSUPP),
110        // CPUCLOCK_PERTHREAD_MASK | CPUCLOCK_MAX: no CPU clock.
111        id if id < 0 && id & CLOCKFD_MASK == CLOCKFD_MASK => Some(Errno::EINVAL),
112        _ => None,
113    }
114}

116fn time_from_resources(rsrcs: &Resources) -> Option<LogicalTime> {
117    if rsrcs.resources.len() > 1 {
118        panic!(
119            "time_from_resources: multiple resource ids in resource request: {:?}",
120            rsrcs
121        );
122    }
123    for rs in rsrcs.resources.iter() {
124        if let (ResourceID::SleepUntil(tm), _) = rs {
125            return Some(*tm);
126        }
127    }
128    None
129}

131/// Flatten a `timespec` to nanoseconds. Negative fields are not valid for the
132/// timer syscalls we handle; treat them as zero rather than panicking.
133fn timespec_to_ns(ts: libc::timespec) -> u64 {
134    let secs = ts.tv_sec.max(0) as u64;
135    let nsec = ts.tv_nsec.max(0) as u64;
136    secs.saturating_mul(1_000_000_000).saturating_add(nsec)
137}

139/// The CPU time an absolute CPU-clock sleep to `deadline` still has to wait
140/// for when the clock reads `now`: none once the deadline has passed, where
141/// Linux returns at once (`posix_cpu_timer_set` fires an expired timer,
142/// kernel/time/posix-cpu-timers.c:729-735 at v6.16). The caller checked the
143/// request's fields; the deadline saturates.
144fn cpu_sleep_remaining(deadline: &Timespec, now: LogicalTime) -> Duration {
145    let deadline =
146        LogicalTime::ZERO + Duration::new(deadline.tv_sec as u64, deadline.tv_nsec as u32);
147    if deadline > now {
148        deadline.duration_since(now)
149    } else {
150        Duration::ZERO
151    }
152}

194/// Replacing host time in the `tv` of a `gettimeofday` that failed with EFAULT
195/// could not finish. `tv` may still hold host wall-clock time, so this is a
196/// failed run, not a guest errno.
197#[derive(Debug, Clone, Copy, PartialEq, Eq)]
198struct TvRepairFailure {
199    /// The word being stored: `tv_sec` or `tv_usec`.
200    field: &'static str,
201    kind: TvRepairFailureKind,
202}
203
204#[derive(Debug, Clone, Copy, PartialEq, Eq)]
205#[cfg_attr(
206    not(target_arch = "x86_64"),
207    allow(
208        dead_code,
209        reason = "the repair's time(2) probes, which build most of these, are x86_64-only"
210    )
211)]
212enum TvRepairFailureKind {
213    /// The backend could not execute the `time(2)` probe. EFAULT is not a
214    /// failure: it means Linux stopped before storing this word.
215    ProbeFailed(Errno),
216    /// After a store probe returned EFAULT, `time(NULL)`, which Linux cannot
217    /// fail, failed too. Something other than the store, such as a seccomp
218    /// filter, produced the EFAULT, so it says nothing about `tv`.
219    ControlProbeFailed(Errno),
220    /// A word at or after the one whose store probe returned EFAULT no longer
221    /// reads as it did before the call, so the original call stored it and the
222    /// EFAULT did not come from the store.
223    StoppedWordChanged,
224    /// A word at or after the one whose store probe returned EFAULT could not
225    /// be read before or after the call, although every byte of it is mapped.
226    /// The read failure may not be a fault: a seccomp filter can deny the
227    /// read while the store went through, so nothing shows the word unstored.
228    StoppedWordUnreadable,
229    /// The guest's memory map, needed to tell an unmapped word from an
230    /// unreadable one, could not be read or was empty.
231    MapsUnavailable,
232    /// The exact overwrite after a successful probe failed. The probe just
233    /// stored host seconds, so the run cannot safely continue.
234    OverwriteFailed(Errno),
235    /// The one-shot overwrite reported a count other than the whole word.
236    OverwriteCount { expected: usize, reported: usize },
237    /// Adding the field offset to the guest's `timeval` address overflowed.
238    AddressOverflow,
239    /// A recorded EFAULT does not prove that this replay execution performed
240    /// the original stores, so live probing would mix replayed and host state.
241    ReplayMode,
242    /// This architecture has no `time(2)` to probe each word with.
243    #[cfg(not(target_arch = "x86_64"))]
244    NoStoreProbe,
245}

247impl std::fmt::Display for TvRepairFailure {
248    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
249        write!(
250            f,
251            "replacing host time in the {} of a failed gettimeofday: ",
252            self.field
253        )?;
254        match self.kind {
255            TvRepairFailureKind::ProbeFailed(errno) => {
256                write!(f, "time(2) store probe failed: {errno}")
257            }
258            TvRepairFailureKind::ControlProbeFailed(errno) => write!(
259                f,
260                "time(NULL) control probe failed: {errno}, so the store probe's EFAULT was not a store fault"
261            ),
262            TvRepairFailureKind::StoppedWordChanged => f.write_str(
263                "the word changed during the call although its store probe returned EFAULT",
264            ),
265            TvRepairFailureKind::StoppedWordUnreadable => f.write_str(
266                "the word is mapped but could not be read, so its store probe's EFAULT is unconfirmed",
267            ),
268            TvRepairFailureKind::MapsUnavailable => {
269                f.write_str("the guest's memory map could not be read")
270            }
271            TvRepairFailureKind::OverwriteFailed(errno) => {
272                write!(f, "virtual-time overwrite failed: {errno}")
273            }
274            TvRepairFailureKind::OverwriteCount { expected, reported } => write!(
275                f,
276                "virtual-time overwrite reported {reported} of {expected} bytes"
277            ),
278            TvRepairFailureKind::AddressOverflow => f.write_str("field address overflowed"),
279            TvRepairFailureKind::ReplayMode => {
280                f.write_str("cannot probe a recorded or replayed EFAULT")
281            }
282            #[cfg(not(target_arch = "x86_64"))]
283            TvRepairFailureKind::NoStoreProbe => f.write_str(
284                "this architecture has no time(2) to probe which words Linux stored",
285            ),
286        }
287    }
288}

362/// Whether every byte of the `time_t` at `word` lies in one of `ranges`, the
363/// half-open address ranges of the guest's mappings.
364#[cfg(target_arch = "x86_64")]
365fn time_word_is_fully_mapped(mut ranges: Vec<(u64, u64)>, word: u64) -> bool {
366    let Some(end) = word.checked_add(std::mem::size_of::<libc::time_t>() as u64) else {
367        return false;
368    };
369    ranges.sort_unstable();
370    let mut covered = word;
371    for (start, stop) in ranges {
372        if start <= covered && covered < stop {
373            covered = stop;
374            if covered >= end {
375                return true;
376            }
377        }
378    }
379    false
380}

382/// An unreadable word is accepted as unstored only when part of it is
383/// unmapped: Linux's eight-byte `put_user` faults there before storing any
384/// byte. A failed read of a fully mapped word is not evidence of a fault: a
385/// seccomp filter can deny the read while the store succeeds. Ptrace reads use
386/// `FOLL_FORCE`, which bypasses page permissions, but some listed mappings
387/// still refuse the read: file pages beyond end of file, hardware-poisoned
388/// pages, `MADV_GUARD_INSTALL` guard regions, `VM_PFNMAP` mappings without an
389/// `access` operation, missing pages of a userfaultfd region in SIGBUS mode,
390/// and `[vsyscall]` in xonly mode. Backends
391/// that read with `process_vm_readv` also cannot read `PROT_NONE` pages. A
392/// `gettimeofday` whose store faults on any of these ends the run rather than
393/// returning EFAULT. An empty map is refused because a filter that fakes a
394/// successful zero-byte read would otherwise make every word look unmapped.
395#[cfg(target_arch = "x86_64")]
396fn require_unmapped_unreadable_word(
397    field: &'static str,
398    maps: Result<Vec<(u64, u64)>, Error>,
399    word: u64,
400) -> Result<(), Error> {
401    let ranges = match maps {
402        Ok(ranges) if !ranges.is_empty() => ranges,
403        Ok(_) => {
404            error!("gettimeofday tv repair: the guest's memory map is empty");
405            return Err(tv_repair_error(field, TvRepairFailureKind::MapsUnavailable));
406        }
407        Err(err) => {
408            error!("gettimeofday tv repair: reading the guest's memory map: {err}");
409            return Err(tv_repair_error(field, TvRepairFailureKind::MapsUnavailable));
410        }
411    };
412    if time_word_is_fully_mapped(ranges, word) {
413        Err(tv_repair_error(
414            field,
415            TvRepairFailureKind::StoppedWordUnreadable,
416        ))
417    } else {
418        Ok(())
419    }
420}

501/// Replaces the host wall-clock time that a `gettimeofday` failing with EFAULT
502/// may have stored in `tv` with virtual time, in exactly the words Linux
503/// stored.
504///
505/// Linux stores `tv_sec`, then `tv_usec`, each with one eight-byte `put_user`,
506/// and only then copies `tz`; the first fault ends the call with EFAULT. A
507/// store that faults on either page it touches commits nothing, so each word
508/// is stored whole or not at all, and nothing after an unstored word is
509/// attempted. `time(2)` uses the same eight-byte `put_user` store for its
510/// `tloc`, so the repair injects it at each word in kernel order. EFAULT means
511/// the original call stopped at that word too. Success means the probe itself
512/// just stored host seconds in those exact bytes; only then does Hermit replace
513/// the word with its virtual value. Other probe errors and overwrite failures
514/// fail closed because host time may remain.
515///
516/// An EFAULT from the probe is only evidence about `tv` if it came from the
517/// store. A seccomp filter can return EFAULT for `time(2)` without running it,
518/// and would otherwise end the repair while host time remains in a writable
519/// `tv`. Two checks therefore confirm every EFAULT before it is trusted.
520/// First, `time(NULL)` is injected: it stores nothing and cannot fail on
521/// Linux, so any error means the probes are not reaching the native call.
522/// Second, every word from the stopped one onward must still read exactly as
523/// it did before the original call: readable with the same value, or
524/// unreadable both times and not fully mapped, so that the store could only
525/// have faulted. A changed word was stored by that call, and an unreadable
526/// word that is fully mapped may have been, because a failed read is not
527/// itself a fault. Each of these failures ends the run, including for the
528/// mapped but unreadable pages listed at `require_unmapped_unreadable_word`.
529///
530/// These checks are observations Detcore makes with its own syscalls, so they
531/// trust those syscalls' results, as every Detcore handler that reads back a
532/// result does. They catch a filter that makes a syscall fail, whether it is
533/// the guest's probe or Detcore's own read. A seccomp filter installed on
534/// Hermit itself can also make one of Detcore's own syscalls report success
535/// without running it, and that is outside this guarantee. A guest cannot
536/// install a filter: Detcore refuses `seccomp(2)` and `PR_SET_SECCOMP`.
537///
538/// The repair never writes a word the kernel could not store, and it does not
539/// test writability by rewriting a word, because a remote write ignores
540/// protection keys that deny the guest's own stores. The pre-call snapshot is
541/// a read only. The host seconds written by a successful
542/// probe exist transiently until the overwrite. Default thread
543/// sequentialization prevents another guest thread from observing that
544/// interval, but another process sharing the page can observe it; such
545/// external shared state is outside Hermit's determinism guarantee.
546/// Standard record/replay disables time virtualization, so this handler is not
547/// used there. A custom configuration that combines replay data with virtual
548/// time fails closed before probing: a recorded EFAULT does not establish that
549/// this execution performed any stores for a live `time(2)` to reproduce.
550///
551/// Every backend runs this repair. The memory map comes from
552/// `Guest::storable_memory_ranges`, which lists `/proc/<pid>/maps` of the
553/// guest by default and the guest's own page record on a backend whose
554/// `pid()` is not the guest (KVM). A backend that executes `gettimeofday` and
555/// `time(2)` itself, as KVM does, stores them as Linux does: each word with an
556/// all-or-nothing store to a writable page, in kernel order.
557#[cfg(target_arch = "x86_64")]
558async fn overwrite_failed_gettimeofday_tv<'a, G, T>(
559    guest: &mut G,
560    tv_addr: AddrMut<'a, Timeval>,
561    tv: &Timeval,
562    before: &TimevalWordSnapshot,
563) -> Result<(), Error>
564where
565    G: Guest<Detcore<T>>,
566    T: RecordOrReplay,
567{
568    let values = [tv.tv_sec as libc::time_t, tv.tv_usec as libc::time_t];
569    for (index, (field, offset)) in TIMEVAL_WORDS.into_iter().enumerate() {
570        let addr = timeval_word_addr(field, tv_addr, offset)?;
571        let probe = syscalls::Time::new().with_tloc(Some(addr));
572        match classify_time_store_probe(field, guest.inject(probe).await)? {
573            TimeStoreProbe::Stored => {
574                let bytes = values[index].to_ne_bytes();
575                let overwrite = guest
576                    .memory()
577                    .write_with_user_access(addr.cast::<u8>(), &bytes);
578                require_complete_time_word_overwrite(field, bytes.len(), overwrite)?;
579            }
580            TimeStoreProbe::Stopped => {
581                let control = syscalls::Time::new().with_tloc(None);
582                require_native_time_control_probe(field, guest.inject(control).await)?;
583                for (stopped, (field, offset)) in TIMEVAL_WORDS.into_iter().enumerate().skip(index)
584                {
585                    let Some(addr) = stopped_timeval_word_addr(tv_addr, offset) else {
586                        continue;
587                    };
588                    let after = guest.memory().read_value(addr);
589                    match require_unchanged_stopped_word(field, before[stopped], after)? {
590                        StoppedWord::Unchanged => {}
591                        StoppedWord::Unreadable => {
592                            let maps = guest.storable_memory_ranges();
593                            require_unmapped_unreadable_word(field, maps, addr.as_raw() as u64)?;
594                        }
595                    }
596                }
597                break;
598            }
599        }
600    }
601    Ok(())
602}

687        &self,
688        guest: &mut G,
689        call: syscalls::Gettimeofday,
690    ) -> Result<i64, Error> {
691        let time_ns = guest_clock_time(guest).await;
692
693        // What Detcore could read in `tv` before the call; the repair uses it
694        // to confirm which words a failing call left alone.
695        let before = call.tv().map(|tp| snapshot_timeval_words(guest, tp.into()));
696
697        // A call failing with EFAULT may still have stored host wall-clock time
698        // in `tv`, so keep its result until `tv` holds virtual time.
699        let result = self
700            .record_or_replay_preserving_tool_errors(guest, call)
701            .await;
702
703        let tv: Timeval = time_ns.into();
704
705        if let Some(tp) = call.tv() {
706            match (&result, &before) {
707                (Ok(_), _) => guest.memory().write_value(tp, &tv)?,
708                // Linux's gettimeofday fails only with EFAULT, which is taken
709                // to be its own even when a seccomp filter returned it without
710                // running the call. Any other error came from the backend, the
711                // tool, a seccomp filter or a replayed log, and says nothing
712                // about what reached `tv`, so memory is left alone.

714                    require_live_time_store_probe(self.cfg.replay_data.is_some())?;
715                    overwrite_failed_gettimeofday_tv(guest, tp.into(), &tv, before).await?
716                }
717                (Err(_), _) => {}
718            }
719        }
720
721        result

859    /// Helper function to wait a given period, which may either succeed or be interrupted by a signal.
860    /// Return 0 or EINTR respectively.
861    async fn wait_and_return<R: Guest<Self>>(
862        &self,
863        guest: &mut R,
864        mut request: Resources,
865        call: NanosleepFamily,
866    ) -> Result<i64, Error> {
867        let target_time = time_from_resources(&request).expect("a sleepuntil resource request");
868        // Lets the scheduler end the sleep for a cross-task signal that
869        // interrupts it (`scheduler::sleep_signal`).
870        request.fyi(crate::resources::NANOSLEEP_FYI);
871        let mut claimed = false;
872        loop {
873            // A signal already pending ends the sleep before it parks, and one
874            // still pending after a wake ends it; a wake whose signal was taken
875            // back parks it again with its original absolute deadline
876            // (`super::signal::sleep_signal`).
877            if super::signal::sleep_signal(guest, SleepCheck::BeforePark, false).await?
878                == SleepSignal::Pending
879            {
880                break;
881            }
882            request.in_guest_sleep_sigalrm = self.in_guest_sleep_sigalrm_snapshot(guest).await;
883            match crate::tool_global::parked_wait_request(
884                guest,
885                request.clone(),
886                crate::scheduler::parked::ParkedWaitPolicy::NanosleepNoHandlerRestart {
887                    absolute_deadline: target_time,
888                },
889            )
890            .await
891            {
892                ResumeStatus::InGuestFatalAlarm => {
893                    // The default action terminates the guest. In particular,
894                    // do not observe time or write `rem` after this fatal wake.
895                    self.stage_in_guest_fatal_alarm(guest).await;
896                    return Err(Errno::EINTR.into());
897                }
898                // A sleep never asks for `NothingElseCanRun`.
899                ResumeStatus::Normal | ResumeStatus::NothingElseCanRun => return Ok(0),
900                ResumeStatus::Signaled(signals) => {
901                    match super::signal::sleep_signal(guest, SleepCheck::AfterWake(signals), false)
902                        .await?
903                    {
904                        SleepSignal::Pending | SleepSignal::Held => break,
905                        SleepSignal::Claimed => {
906                            claimed = true;
907                            break;
908                        }
909                        SleepSignal::None => trace!(
910                            "nanosleep woken for a signal no longer pending for it; sleeping again until {}",
911                            target_time
912                        ),
913                    }
914                }
915            }
916        }
917        let now = thread_observe_time(guest).await;
918        let delta = remaining_sleep_duration(target_time, now);
919        // Linux never touches remain for TIMER_ABSTIME, even when
920        // a caught signal interrupts the absolute sleep.
921        let addr2 = if call.flags() & libc::TIMER_ABSTIME == 0 {
922            call.rem()
923        } else {
924            None
925        };
926        if let Some(addr2) = addr2 {
927            info!(
928                "[interrupted] sleep till (until {}), woke up {:?} early, writing into nanosleep rem argument.",
929                target_time, delta
930            );
931            let t = Timespec {
932                tv_sec: delta.as_secs() as i64,
933                tv_nsec: delta.subsec_nanos() as i64,
934            };
935            guest.memory().write_value(addr2, &t)?;
936        } else {
937            info!("[interrupted] nanosleep rem argument is null, not writing it.")
938        }
939        if claimed {
940            // A claimed `SIGCHLD` is held for delivery as the thread resumes
941            // (`SleepSignal::Claimed`): `EINTR` if a handler runs, else the
942            // kernel's restart. As Linux's own `nanosleep`, a relative sleep
943            // returns `ERESTART_RESTARTBLOCK` and keeps its deadline for the
944            // `restart_syscall` the kernel runs only when no handler ran
945            // (`helpers::RestartBlock`, `resume_nanosleep`), so a new call,
946            // from a handler or a retry, cannot take it. An absolute one
947            // returns `ERESTARTNOHAND`: re-executed, it reads the same target.
948            if call.flags() & libc::TIMER_ABSTIME != 0 {
949                return Err(reverie::Error::Errno(Errno::ERESTARTNOHAND));
950            }
951            let result = Err(reverie::Error::Errno(Errno::ERESTART_RESTARTBLOCK));
952            crate::syscalls::helpers::keep_restart_block(
953                guest,
954                &result,
955                crate::syscalls::helpers::RestartCall::Nanosleep(call),
956                Some(target_time),
957            )
958            .await;
959            return result;
960        }
961        Err(reverie::Error::Errno(Errno::EINTR))
962    }

964    /// The kernel's `restart_syscall` of a relative `nanosleep` that ended for
965    /// a claimed `SIGCHLD` no handler ran for (`helpers::RestartCall::Nanosleep`).
966    /// Like Linux's restart block, it keeps the deadline it already has and
967    /// does not read the request again, which another thread may have changed.
968    pub(crate) async fn resume_nanosleep<R: Guest<Self>>(
969        &self,
970        guest: &mut R,
971        call: NanosleepFamily,
972        deadline: LogicalTime,
973    ) -> Result<i64, Error> {
974        let request = Self::sleep_request_abs(guest, deadline).await;
975        self.wait_and_return(guest, request, call).await
976    }

978    /// clock_nanosleep and nanosleep
979    pub async fn handle_nanosleep_family<R: Guest<Self>>(
980        &self,
981        guest: &mut R,
982        call: NanosleepFamily,
983    ) -> Result<i64, Error> {
984        if call.flags() > libc::TIMER_ABSTIME {
985            trace!("Unhandled clock_nanosleep flags, letting syscall through...");
986            // Linux reads only TIMER_ABSTIME from the flags, so the kernel runs
987            // this as a relative or absolute sleep and may leave a restart block.
988            return crate::syscalls::helpers::inject_keeping_kernel_restart(
989                guest,
990                Syscall::from(call),
991            )
992            .await;
993        }
994
995        // The CPU clocks are Detcore's own only where it virtualizes time;
996        // otherwise the host's clocks are read and slept on.
997        let clock_class = if self.cfg.virtualize_time {
998            // `clock_thread` has no `nsleep` (kernel/time/posix-cpu-timers.c:1666-1670
999            // at v6.16), so Linux answers EOPNOTSUPP before it reads the
1000            // request (`clock_nanosleep`, kernel/time/posix-timers.c:1385-1386).
1001            // glibc answers EINVAL itself, without the syscall.
1002            if raw_clock_id(call.clockid()) == libc::CLOCK_THREAD_CPUTIME_ID {
1003                return Err(Errno::EOPNOTSUPP.into());
1004            }
1005            cpu_clock::classify(raw_clock_id(call.clockid()))
1006        } else {
1007            ClockIdClass::Other
1008        };
1009
1010        let addr = call.req().ok_or(Errno::EFAULT)?;
1011        let t: Timespec = guest.memory().read_value(addr)?;
1012
1013        // Linux validates the requested interval BEFORE sleeping: nanosleep(2)
1014        // and clock_nanosleep(2) both fail EINVAL when tv_nsec is outside
1015        // [0, 999999999] or tv_sec is negative.
1016        //
1017        // Detcore skipped that check and fed the raw fields through `as u64`.
1018        // `Timespec` stores both as i64, so `tv_sec = -1` wrapped to
1019        // u64::MAX (~1.8e19 seconds) and became `SleepUntil(INDEFINITE)`: the
1020        // only guest thread parked with no deadline, the run queue emptied, and
1021        // `step2d_handle_empty_queue` deliberately never jumps the clock for an
1022        // indefinite waiter (doing so would also wake a `pause(2)`). The
1023        // container then died -- exit 1, "Sandbox container exited
1024        // unexpectedly" -- where Linux returns an errno and keeps running.
1025        //
1026        // A past *absolute* deadline is NOT an error and must still return 0,
1027        // so this rejects only malformed fields, never an early deadline.
1028        if t.tv_sec < 0 || t.tv_nsec < 0 || t.tv_nsec > 999_999_999 {
1029            return Err(Errno::EINVAL.into());
1030        }
1031
1032        let cpu_deadline_remaining = match clock_class {
1033            ClockIdClass::Other => None,
1034            ClockIdClass::NoCpuClock => return Err(Errno::EINVAL.into()),
1035            ClockIdClass::Cpu(clock) => self.cpu_clock_sleep(guest, call, clock, &t).await?,
1036        };
1037        if let Some(remaining) = cpu_deadline_remaining {
1038            return self.sleep_for(guest, call, remaining).await;
1039        }
1040
1041        match call.flags() {
1042            0 => {
1043                // Main's `sleep_for` is this arm's sequentialized path plus
1044                // the kernel run; `sleep_for` keeps the kernel restart
1045                // below, which is this change's non-sequentialized arm.
1046                let time =
1047                    Duration::from_secs(t.tv_sec as u64) + Duration::from_nanos(t.tv_nsec as u64);
1048                self.sleep_for(guest, call, time).await
1049            }
1050            libc::TIMER_ABSTIME => {
1051                let target_time = LogicalTime::from_secs(t.tv_sec as u64)
1052                    + LogicalTime::from_nanos(t.tv_nsec as u64);
1053                if self.cfg.sequentialize_threads {
1054                    if self.cfg.virtualize_time {
1055                        let request = Self::sleep_request_abs(guest, target_time).await;
1056                        trace!(
1057                            "nanosleep setting absolute time {:?} to yield request {:?}",
1058                            target_time, &request
1059                        );
1060                        self.wait_and_return(guest, request, call).await
1061                    } else {
1062                        // TODO T124594597: Record-replay case here, need better ideas to enable proper handling of this case.
1063                        error!(
1064                            "Sequentializing but not virtualizing, so can't rely on passed abs time, especially when replaying a recording, just yelding"
1065                        );
1066                        let request = Self::yield_request(guest);
1067                        self.wait_and_return(guest, request, call).await
1068                    }
1069                } else if self.cfg.virtualize_time {
1070                    trace!(
1071                        "Not sequentializing, but virtualizing so calculating relative time and invoking nanosleep..."
1072                    );
1073                    let relative_ts = Self::relative_time_from_abs_target(guest, target_time).await;
1074                    let mut stack = guest.stack().await;
1075                    let req = stack.push(relative_ts);
1076                    stack.commit()?;
1077                    let modified_call = syscalls::Nanosleep::new().with_req(Some(req));
1078                    // Linux restarts an absolute sleep by running it again, but
1079                    // this relative one keeps the host end the kernel computed,
1080                    // so it restarts from the kernel's restart block. Running
1081                    // the absolute call again would wait the whole interval
1082                    // again: logical time stands still while the thread sleeps
1083                    // in the kernel. The block holds no pointer to the request
1084                    // on the stack, and no `rem`.
1085                    crate::syscalls::helpers::inject_keeping_kernel_restart(guest, modified_call)
1086                        .await
1087                } else {
1088                    trace!(
1089                        "Not sequentializing threads not virtualizing, letting nanosleep through..."
1090                    );
1091                    Ok(guest.inject(Syscall::from(call)).await?)
1092                }
1093            }
1094            _ => unreachable!("Unexpected, unhandled flag value"),
1095        }
1096    }

1098    /// Sleep for `time`: on the virtual clock where threads are
1099    /// sequentialized, otherwise in the kernel on the call's own clock. A
1100    /// relative call goes to the kernel as it is. An absolute CPU-clock call,
1101    /// which `cpu_clock_sleep` made relative to Detcore's reading of the
1102    /// clock, goes as a relative sleep for the rest.
1103    async fn sleep_for<G: Guest<Self>>(
1104        &self,
1105        guest: &mut G,
1106        call: NanosleepFamily,
1107        time: Duration,
1108    ) -> Result<i64, Error> {
1109        if self.cfg.sequentialize_threads {
1110            let request = Self::sleep_request(guest, time).await;
1111            trace!(
1112                "nanosleep adding delta {:?} to yield request {:?}",
1113                time, &request
1114            );
1115            return self.wait_and_return(guest, request, call).await;
1116        }
1117        trace!("Not sequentializing threads, letting nanosleep through...");
1118        if call.flags() & libc::TIMER_ABSTIME == 0 {
1119            // A kernel-run relative sleep keeps the kernel's restart
1120            // block, as in the handlers this change converted.
1121            return crate::syscalls::helpers::inject_keeping_kernel_restart(
1122                guest,
1123                Syscall::from(call),
1124            )
1125            .await;
1126        }
1127        let mut stack = guest.stack().await;
1128        let req = stack.push(Timespec {
1129            tv_sec: time.as_secs() as i64,
1130            tv_nsec: time.subsec_nanos() as i64,
1131        });
1132        stack.commit()?;
1133        let relative = syscalls::ClockNanosleep::new()
1134            .with_clockid(call.clockid())
1135            .with_flags(0)
1136            .with_req(Some(req))
1137            .with_rem(None);
1138        // The converted relative sleep keeps the kernel's restart block
1139        // too: restarting the absolute call would wait the whole interval
1140        // again while logical time stands still.
1141        crate::syscalls::helpers::inject_keeping_kernel_restart(guest, relative).await
1142    }

1184    async fn relative_time_from_abs_target<G: Guest<Self>>(
1185        guest: &mut G,
1186        target_time: LogicalTime,
1187    ) -> Timespec {
1188        let base_time = thread_observe_time(guest).await;
1189
1190        // An absolute deadline already in the past is NOT an error on Linux --
1191        // clock_nanosleep(TIMER_ABSTIME) simply returns 0 without sleeping.
1192        // `LogicalTime`'s `Sub` is a plain subtraction (unlike its `Add` impls,
1193        // which saturate deliberately), so `target_time - base_time` underflows
1194        // for a past deadline: a debug build panics with "attempt to subtract
1195        // with overflow", and a release build wraps to an enormous interval --
1196        // the same effectively-indefinite sleep this handler exists to avoid.
1197        //
1198        // Clamped here rather than by making the shared operator saturate,
1199        // because this is the only subtraction of two `LogicalTime`s in the
1200        // tree and a silently-saturating operator could hide a real underflow
1201        // in some future caller.
1202        let relative_logical = if target_time <= base_time {
1203            LogicalTime::from_nanos(0)
1204        } else {
1205            target_time - base_time
1206        };
1207
1208        Timespec {
1209            tv_sec: relative_logical.as_secs() as i64,
1210            tv_nsec: relative_logical.subsec_nanos() as i64,
1211        }
1212    }

1229    /// timer_create: allocate a per-process POSIX timer and hand back a
1230    /// deterministic id, retaining any scheduler-deliverable signal and, for
1231    /// `SIGEV_THREAD_ID`, the thread it notifies. The notification is checked
1232    /// as Linux's `good_sigevent` does: `SIGEV_SIGNAL` and `SIGEV_THREAD` (which
1233    /// the kernel treats alike; libc implements the thread itself) need a
1234    /// signal from 1 to 64; `SIGEV_THREAD_ID` needs, first, a thread of the
1235    /// caller's own process (<https://github.com/rrnewton/hermit/issues/4164>);
1236    /// any other value is EINVAL.
1237    pub async fn handle_timer_create<G: Guest<Self>>(
1238        &self,
1239        guest: &mut G,
1240        call: syscalls::TimerCreate,
1241    ) -> Result<i64, Error> {
1242        // The kernel writes the new timer id here; a null pointer is EFAULT.
1243        let timerid_ptr = call.timerid().ok_or(Errno::EFAULT)?;
1244        let clockid = call.clockid();
1245        let (signo, thread, value) = if let Some(event_ptr) = call.sevp() {
1246            // Only the plain fields: `sigevent` holds a pointer, which must not
1247            // be kept across the query below. The value is kept as the 64 bits
1248            // Linux copies into `si_value`.
1249            let (notify, signo, named, value) = {
1250                let event: libc::sigevent = guest.memory().read_value(event_ptr)?;
1251                (
1252                    event.sigev_notify,
1253                    event.sigev_signo,
1254                    event.sigev_notify_thread_id,
1255                    event.sigev_value.sival_ptr as usize as u64,
1256                )
1257            };
1258            // A reserved in-guest signal: Linux answers for the clock right
1259            // after copying the sigevent, before it looks at the notification
1260            // (`do_timer_create`), so an unsupported clock is that answer even
1261            // when the notification is invalid too
1262            // (`timer_create_clock_refusal`).
1263            if guest.config().reserves_in_guest_signal(signo.into())
1264                && let Some(errno) = timer_create_clock_refusal(clockid)
1265            {
1266                return Err(errno.into());
1267            }
1268            let thread = match notify {
1269                libc::SIGEV_NONE => None,
1270                libc::SIGEV_SIGNAL | libc::SIGEV_THREAD => None,
1271                reverie::syscalls::linux_abi::SIGEV_THREAD_ID => {
1272                    match posix_timer_thread(guest, named).await {
1273                        PosixTimerThread::Thread(thread) => Some(thread),
1274                        PosixTimerThread::NotInGroup => return Err(Errno::EINVAL.into()),
1275                        PosixTimerThread::NestedNamespace => {
1276                            tracing::error!(
1277                                "[detcore, dtid {}] timer_create with SIGEV_THREAD_ID is not supported in a nested PID namespace: the thread id is a pid of that namespace, which Detcore does not translate (https://github.com/rrnewton/hermit/issues/4155)",
1278                                guest.thread_state().dettid,
1279                            );
1280                            return self
1281                                .refuse_unserviceable_operation(
1282                                    guest,
1283                                    Sysno::timer_create,
1284                                    Errno::ENOSYS,
1285                                )
1286                                .await;
1287                        }
1288                    }
1289                }
1290                _ => return Err(Errno::EINVAL.into()),
1291            };
1292            if notify == libc::SIGEV_NONE {
1293                (None, None, Some(value))
1294            } else if !(1..=64).contains(&signo) {
1295                return Err(Errno::EINVAL.into());
1296            } else {
1297                (Some(signo), thread, Some(value))
1298            }
1299        } else {
1300            (Some(libc::SIGALRM), None, None)
1301        };
1302        // A reserved in-guest signal as the timer's notification signal, for
1303        // each notification that delivers one (`SIGEV_SIGNAL`, `SIGEV_THREAD`,
1304        // `SIGEV_THREAD_ID`), after Linux's answers for the clock and the
1305        // sigevent above. With no `sigevent` the signal is SIGALRM, which is
1306        // never reserved.
1307        if let Some(signo) = signo
1308            && guest.config().reserves_in_guest_signal(signo.into())
1309        {
1310            Self::refuse_reserved_signal_arming(
1311                guest,
1312                crate::refused_signal_controls::ReservedArming::TimerCreate,
1313                signo.into(),
1314            )?;
1315        }
1316        // `None` for a real-time signal (32 to 64): see `handle_timer_settime`.
1317        let signal = signo.and_then(|signo| Signal::try_from(signo).ok());
1318        if signal == Some(Signal::SIGALRM) {
1319            refuse_sigalrm(guest, SigalrmControl::ArmProducer).await?;
1320        }
1321        let on_cpu_clock = matches!(
1322            cpu_clock::classify(raw_clock_id(clockid)),
1323            ClockIdClass::Cpu(_)
1324        );
1325        let id = {
1326            let mut timers = guest.thread_state().posix_timers.lock().unwrap();
1327            timers.create(signo, thread, value, on_cpu_clock)
1328        };
1329        guest
1330            .memory()
1331            .write_value(timerid_ptr, &(id as libc::c_int))?;
1332        // The process has a POSIX timer on this signal from its creation, armed
1333        // or not, which decides how a kept send standing on another sender's
1334        // record is classified (https://github.com/rrnewton/hermit/issues/4262).
1335        if crate::scheduler::discarded_at_send::records(guest.config())
1336            && let Some(signo) = signo
1337            && (1..32).contains(&signo)
1338        {
1339            crate::tool_global::discarded_signal(
1340                guest,
1341                crate::scheduler::discarded_at_send::DiscardedSignalControl::TimerCreated {
1342                    signal: signo,
1343                },
1344            )
1345            .await;
1346        }
1347        detlog!(
1348            "[dtid {}] timer_create(clockid={:?}) => deterministic timer id {}, signal {:?}, thread {:?}",
1349            guest.thread_state().dettid,
1350            clockid,
1351            id,
1352            signal,
1353            thread,
1354        );
1355        Ok(0)
1356    }

1360    /// timer_settime: arm or disarm a timer against the deterministic virtual
1361    /// clock. The old arming is reported through `old_value` when requested.
1362    pub async fn handle_timer_settime<G: Guest<Self>>(
1363        &self,
1364        guest: &mut G,
1365        call: syscalls::TimerSettime,
1366    ) -> Result<i64, Error> {
1367        let id = call.timerid();
1368        let new_ptr = call.new_value().ok_or(Errno::EINVAL)?;
1369        let new: libc::itimerspec = guest.memory().read_value(new_ptr)?;
1370        let interval_ns = timespec_to_ns(new.it_interval);
1371        let value_ns = timespec_to_ns(new.it_value);
1372
1373        // A CPU clock's absolute expiration is a CPU time of the task the
1374        // timer was created for (`posix_cpu_timer_set`,
1375        // kernel/time/posix-cpu-timers.c:622-739 at v6.16), which Detcore's
1376        // timers, counting down on the virtual clock, do not record: refused
1377        // rather than read as a virtual time that has long passed.
1378        if self.cfg.virtualize_time
1379            && value_ns != 0
1380            && call.flags() & libc::TIMER_ABSTIME != 0
1381            && guest
1382                .thread_state()
1383                .posix_timers
1384                .lock()
1385                .unwrap()
1386                .on_cpu_clock(id)
1387        {
1388            return Err(self
1389                .refuse_cpu_clock(
1390                    guest,
1391                    Sysno::timer_settime,
1392                    "an absolute expiration on a CPU-clock timer: Detcore's timers count down on the virtual clock and do not record the task whose CPU time an absolute expiration names",
1393                )
1394                .await);
1395        }
1396
1397        let now = thread_observe_time(guest).await;
1398        let deadline = if value_ns == 0 {
1399            None
1400        } else if call.flags() & libc::TIMER_ABSTIME != 0 {
1401            // Absolute expiration is interpreted against the same virtual clock.
1402            Some(LogicalTime::from_nanos(value_ns))
1403        } else {
1404            Some(now + Duration::from_nanos(value_ns))
1405        };
1406
1407        let (old, signal_number, thread, value) = {
1408            let mut timers = guest.thread_state().posix_timers.lock().unwrap();
1409            let old = timers.settime(id, interval_ns, deadline, now);
1410            let signal = timers.signal(id);
1411            (
1412                old,
1413                signal,
1414                timers.thread(id),
1415                timers.value(id).unwrap_or(0),
1416            )
1417        };
1418        let (mut old_remaining_ns, old_interval_ns) = old.ok_or(Errno::EINVAL)?;
1419        let signal_number = signal_number.ok_or(Errno::EINVAL)?;
1420        // The old value is the timer's remaining time as `timer_gettime`
1421        // reads it now, from the scheduler where it keeps the timer's phase
1422        // (4262 design §5).
1423        if let Some((Some(remaining), _)) = posix_timer_readout(guest, id, now).await {
1424            old_remaining_ns = remaining.as_nanos();
1425        }
1426
1427        // Linux changes the timer, its generation and its overrun before it
1428        // copies the old value out (`do_timer_settime`), and an EFAULT there
1429        // undoes none of it, so the scheduler is told first (the Codex review
1430        // of https://github.com/rrnewton/hermit/pull/4491, P2-2).
1431        if let Some(signo) = signal_number {
1432            if self.posix_timer_expiry_is_registered(signo) {
1433                register_posix_timer(
1434                    guest,
1435                    id,
1436                    deadline,
1437                    LogicalTime::from_nanos(interval_ns),
1438                    PosixTimerNotification {
1439                        sig: SigWrapper(signo),
1440                        thread,
1441                        value,
1442                        deleted: false,
1443                    },
1444                )
1445                .await;
1446            } else if deadline.is_some() {
1447                tracing::error!(
1448                    "[detcore, dtid {}] POSIX timer {} notifies with real-time signal {}, which Detcore does not deliver from a timer; it is armed but will never fire",
1449                    guest.thread_state().dettid,
1450                    id,
1451                    signo,
1452                );
1453            }
1454        }
1455
1456        if let Some(old_ptr) = call.old_value() {
1457            let old_spec = libc::itimerspec {
1458                it_interval: ns_to_timespec(old_interval_ns),
1459                it_value: ns_to_timespec(old_remaining_ns),
1460            };
1461            guest.memory().write_value(old_ptr, &old_spec)?;
1462        }
1463
1464        detlog!(
1465            "[dtid {}] timer_settime(id={}, interval_ns={}, value_ns={}) armed against virtual clock",
1466            guest.thread_state().dettid,
1467            id,
1468            interval_ns,
1469            value_ns,
1470        );
1471        Ok(0)
1472    }
