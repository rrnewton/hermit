51/// Whether Linux accepts `clockid` in clock_gettime and clock_getres, as far as
52/// the id alone decides: the fixed clocks 0 to 9 and CLOCK_TAI (11). Linux
53/// answers EINVAL for 10 (CLOCK_SGI_CYCLE, gone since Linux 4.x) and for
54/// every id from 12 up. A negative id names a process's or thread's CPU
55/// clock, or a clock descriptor; whether it is valid depends on the process
56/// or descriptor it names, so it is accepted here as before.
57fn clock_id_is_known(clockid: ClockId) -> bool {
58    use reverie::syscalls::FromToRaw;
59    match clockid.into_raw() as i32 {
60        0..=9 | 11 => true,
61        id => id < 0,
62    }
63}

65/// Linux's answer to `timer_create` for `clockid` alone, which it gives
66/// before it reads the notification (`do_timer_create`,
67/// kernel/time/posix-timers.c:448-455 at v6.12): EINVAL for an id with no
68/// clock (`clockid_to_kclock` and the `posix_clocks` table at :1529-1556: 10,
69/// and 12 up), EOPNOTSUPP for a clock with no timers (CLOCK_MONOTONIC_RAW,
70/// CLOCK_REALTIME_COARSE, CLOCK_MONOTONIC_COARSE, and a clock descriptor: a
71/// negative id with `(id & CLOCKFD_MASK) == CLOCKFD`, which names
72/// `clock_posix_dynamic`, kernel/time/posix-clock.c:334-339, whatever
73/// descriptor it encodes; include/linux/posix-timers_types.h:24-32), and
74/// EINVAL for a negative id whose clock bits name no CPU clock (`(id & 7) ==
75/// 7`: per-thread with CPUCLOCK_WHICH 3, which is CPUCLOCK_MAX, so
76/// `pid_for_clock` refuses it from the bits alone,
77/// kernel/time/posix-cpu-timers.c:63-64, and `posix_cpu_timer_create`
78/// answers EINVAL, :392-395). `None` for the clocks with timers, for the
79/// alarm clocks (whose answer depends on the host) and for the other
80/// negative ids, CPU clocks, whose answer depends on the process or thread
81/// the id names: a CPU-clock id naming a task that does not exist therefore
82/// gets the reserved-signal EPERM where Linux answers EINVAL. Only a
83/// `timer_create` whose sigevent names a reserved signal reads it, right
84/// after the copy and before the notification is checked, as Linux orders
85/// them, so that neither the refusal nor Detcore's notification checks come
86/// before Linux's clock answer. Detcore's virtual timers do not otherwise
87/// check the clock, which https://github.com/rrnewton/hermit/issues/4556
88/// tracks together with the CPU-clock case above.
89fn timer_create_clock_refusal(clockid: ClockId) -> Option<Errno> {
90    use reverie::syscalls::FromToRaw;
91    const CLOCKFD: i32 = 3;
92    const CLOCKFD_MASK: i32 = 4 | 3;
93    if !clock_id_is_known(clockid) {
94        return Some(Errno::EINVAL);
95    }
96    match clockid.into_raw() as i32 {
97        libc::CLOCK_MONOTONIC_RAW | libc::CLOCK_REALTIME_COARSE | libc::CLOCK_MONOTONIC_COARSE => {
98            Some(Errno::EOPNOTSUPP)
99        }
100        id if id < 0 && id & CLOCKFD_MASK == CLOCKFD => Some(Errno::EOPNOTSUPP),
101        // CPUCLOCK_PERTHREAD_MASK | CPUCLOCK_MAX: no CPU clock.
102        id if id < 0 && id & CLOCKFD_MASK == CLOCKFD_MASK => Some(Errno::EINVAL),
103        _ => None,
104    }
105}

108    if rsrcs.resources.len() > 1 {
109        panic!(

122/// Flatten a `timespec` to nanoseconds. Negative fields are not valid for the
123/// timer syscalls we handle; treat them as zero rather than panicking.
124fn timespec_to_ns(ts: libc::timespec) -> u64 {
125    let secs = ts.tv_sec.max(0) as u64;
126    let nsec = ts.tv_nsec.max(0) as u64;
127    secs.saturating_mul(1_000_000_000).saturating_add(nsec)
128}

130/// Inverse of [`timespec_to_ns`].
131fn ns_to_timespec(ns: u64) -> libc::timespec {
132    libc::timespec {
133        tv_sec: (ns / 1_000_000_000) as libc::time_t,
134        tv_nsec: (ns % 1_000_000_000) as libc::c_long,
135    }
136}

170/// Replacing host time in the `tv` of a `gettimeofday` that failed with EFAULT
171/// could not finish. `tv` may still hold host wall-clock time, so this is a
172/// failed run, not a guest errno.
173#[derive(Debug, Clone, Copy, PartialEq, Eq)]
174struct TvRepairFailure {
175    /// The word being stored: `tv_sec` or `tv_usec`.
176    field: &'static str,
177    kind: TvRepairFailureKind,
178}
179
180#[derive(Debug, Clone, Copy, PartialEq, Eq)]
181enum TvRepairFailureKind {
182    /// The backend could not execute the `time(2)` probe. EFAULT is not a
183    /// failure: it means Linux stopped before storing this word.
184    ProbeFailed(Errno),
185    /// After a store probe returned EFAULT, `time(NULL)`, which Linux cannot
186    /// fail, failed too. Something other than the store, such as a seccomp
187    /// filter, produced the EFAULT, so it says nothing about `tv`.
188    ControlProbeFailed(Errno),
189    /// A word at or after the one whose store probe returned EFAULT no longer
190    /// reads as it did before the call, so the original call stored it and the
191    /// EFAULT did not come from the store.
192    StoppedWordChanged,
193    /// A word at or after the one whose store probe returned EFAULT could not
194    /// be read before or after the call, although every byte of it is mapped.
195    /// The read failure may not be a fault: a seccomp filter can deny the
196    /// read while the store went through, so nothing shows the word unstored.
197    StoppedWordUnreadable,
198    /// The guest's memory map, needed to tell an unmapped word from an
199    /// unreadable one, could not be read or was empty.
200    MapsUnavailable,
201    /// The exact overwrite after a successful probe failed. The probe just
202    /// stored host seconds, so the run cannot safely continue.
203    OverwriteFailed(Errno),
204    /// The one-shot overwrite reported a count other than the whole word.
205    OverwriteCount { expected: usize, reported: usize },
206    /// Adding the field offset to the guest's `timeval` address overflowed.
207    AddressOverflow,
208    /// A recorded EFAULT does not prove that this replay execution performed
209    /// the original stores, so live probing would mix replayed and host state.
210    ReplayMode,
211}

213impl std::fmt::Display for TvRepairFailure {
214    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
215        write!(
216            f,
217            "replacing host time in the {} of a failed gettimeofday: ",
218            self.field
219        )?;
220        match self.kind {
221            TvRepairFailureKind::ProbeFailed(errno) => {
222                write!(f, "time(2) store probe failed: {errno}")
223            }
224            TvRepairFailureKind::ControlProbeFailed(errno) => write!(
225                f,
226                "time(NULL) control probe failed: {errno}, so the store probe's EFAULT was not a store fault"
227            ),
228            TvRepairFailureKind::StoppedWordChanged => f.write_str(
229                "the word changed during the call although its store probe returned EFAULT",
230            ),
231            TvRepairFailureKind::StoppedWordUnreadable => f.write_str(
232                "the word is mapped but could not be read, so its store probe's EFAULT is unconfirmed",
233            ),
234            TvRepairFailureKind::MapsUnavailable => {
235                f.write_str("the guest's memory map could not be read")
236            }
237            TvRepairFailureKind::OverwriteFailed(errno) => {
238                write!(f, "virtual-time overwrite failed: {errno}")
239            }
240            TvRepairFailureKind::OverwriteCount { expected, reported } => write!(
241                f,
242                "virtual-time overwrite reported {reported} of {expected} bytes"
243            ),
244            TvRepairFailureKind::AddressOverflow => f.write_str("field address overflowed"),
245            TvRepairFailureKind::ReplayMode => {
246                f.write_str("cannot probe a recorded or replayed EFAULT")
247            }
248        }
249    }
250}

303/// A word Linux did not store keeps both its contents and its readability.
304fn require_unchanged_stopped_word(
305    field: &'static str,
306    before: Result<libc::time_t, Errno>,
307    after: Result<libc::time_t, Errno>,
308) -> Result<StoppedWord, Error> {
309    match (before, after) {
310        (Ok(before), Ok(after)) if before == after => Ok(StoppedWord::Unchanged),

319/// Whether every byte of the `time_t` at `word` lies in one of `ranges`, the
320/// half-open address ranges of the guest's mappings.
321fn time_word_is_fully_mapped(mut ranges: Vec<(u64, u64)>, word: u64) -> bool {
322    let Some(end) = word.checked_add(std::mem::size_of::<libc::time_t>() as u64) else {
323        return false;
324    };
325    ranges.sort_unstable();
326    let mut covered = word;
327    for (start, stop) in ranges {
328        if start <= covered && covered < stop {
329            covered = stop;
330            if covered >= end {
331                return true;
332            }
333        }
334    }
335    false
336}

338/// An unreadable word is accepted as unstored only when part of it is
339/// unmapped: Linux's eight-byte `put_user` faults there before storing any
340/// byte. A failed read of a fully mapped word is not evidence of a fault: a
341/// seccomp filter can deny the read while the store succeeds. Ptrace reads use
342/// `FOLL_FORCE`, which bypasses page permissions, but some listed mappings
343/// still refuse the read: file pages beyond end of file, hardware-poisoned
344/// pages, `MADV_GUARD_INSTALL` guard regions, `VM_PFNMAP` mappings without an
345/// `access` operation, missing pages of a userfaultfd region in SIGBUS mode,
346/// and `[vsyscall]` in xonly mode. Backends
347/// that read with `process_vm_readv` also cannot read `PROT_NONE` pages. A
348/// `gettimeofday` whose store faults on any of these ends the run rather than
349/// returning EFAULT. An empty map is refused because a filter that fakes a
350/// successful zero-byte read would otherwise make every word look unmapped.
351fn require_unmapped_unreadable_word(
352    field: &'static str,
353    maps: Result<Vec<(u64, u64)>, Error>,
354    word: u64,
355) -> Result<(), Error> {
356    let ranges = match maps {
357        Ok(ranges) if !ranges.is_empty() => ranges,
358        Ok(_) => {
359            error!("gettimeofday tv repair: the guest's memory map is empty");
360            return Err(tv_repair_error(field, TvRepairFailureKind::MapsUnavailable));
361        }
362        Err(err) => {
363            error!("gettimeofday tv repair: reading the guest's memory map: {err}");
364            return Err(tv_repair_error(field, TvRepairFailureKind::MapsUnavailable));
365        }
366    };
367    if time_word_is_fully_mapped(ranges, word) {
368        Err(tv_repair_error(
369            field,
370            TvRepairFailureKind::StoppedWordUnreadable,
371        ))
372    } else {
373        Ok(())
374    }
375}

377/// What Detcore's memory reader returned for `tv_sec` and `tv_usec` before a
378/// `gettimeofday`, with the read error where it failed. Under ptrace the read

384    ("tv_sec", std::mem::offset_of!(Timeval, tv_sec)),

454/// Replaces the host wall-clock time that a `gettimeofday` failing with EFAULT
455/// may have stored in `tv` with virtual time, in exactly the words Linux
456/// stored.
457///
458/// Linux stores `tv_sec`, then `tv_usec`, each with one eight-byte `put_user`,
459/// and only then copies `tz`; the first fault ends the call with EFAULT. A
460/// store that faults on either page it touches commits nothing, so each word
461/// is stored whole or not at all, and nothing after an unstored word is
462/// attempted. `time(2)` uses the same eight-byte `put_user` store for its
463/// `tloc`, so the repair injects it at each word in kernel order. EFAULT means
464/// the original call stopped at that word too. Success means the probe itself
465/// just stored host seconds in those exact bytes; only then does Hermit replace
466/// the word with its virtual value. Other probe errors and overwrite failures
467/// fail closed because host time may remain.
468///
469/// An EFAULT from the probe is only evidence about `tv` if it came from the
470/// store. A seccomp filter can return EFAULT for `time(2)` without running it,
471/// and would otherwise end the repair while host time remains in a writable
472/// `tv`. Two checks therefore confirm every EFAULT before it is trusted.
473/// First, `time(NULL)` is injected: it stores nothing and cannot fail on
474/// Linux, so any error means the probes are not reaching the native call.
475/// Second, every word from the stopped one onward must still read exactly as
476/// it did before the original call: readable with the same value, or
477/// unreadable both times and not fully mapped, so that the store could only
478/// have faulted. A changed word was stored by that call, and an unreadable
479/// word that is fully mapped may have been, because a failed read is not
480/// itself a fault. Each of these failures ends the run, including for the
481/// mapped but unreadable pages listed at `require_unmapped_unreadable_word`.
482///
483/// These checks are observations Detcore makes with its own syscalls, so they
484/// trust those syscalls' results, as every Detcore handler that reads back a
485/// result does. They catch a filter that makes a syscall fail, whether it is
486/// the guest's probe or Detcore's own read. A seccomp filter installed on
487/// Hermit itself can also make one of Detcore's own syscalls report success
488/// without running it, and that is outside this guarantee. A guest cannot
489/// install a filter: Detcore refuses `seccomp(2)` and `PR_SET_SECCOMP`.
490///
491/// The repair never writes a word the kernel could not store, and it does not
492/// test writability by rewriting a word, because a remote write ignores
493/// protection keys that deny the guest's own stores. The pre-call snapshot is
494/// a read only. The host seconds written by a successful
495/// probe exist transiently until the overwrite. Default thread
496/// sequentialization prevents another guest thread from observing that
497/// interval, but another process sharing the page can observe it; such
498/// external shared state is outside Hermit's determinism guarantee.
499/// Standard record/replay disables time virtualization, so this handler is not
500/// used there. A custom configuration that combines replay data with virtual
501/// time fails closed before probing: a recorded EFAULT does not establish that
502/// this execution performed any stores for a live `time(2)` to reproduce.
503///
504/// Every backend runs this repair. The memory map comes from
505/// `Guest::storable_memory_ranges`, which lists `/proc/<pid>/maps` of the
506/// guest by default and the guest's own page record on a backend whose
507/// `pid()` is not the guest (KVM). A backend that executes `gettimeofday` and
508/// `time(2)` itself, as KVM does, stores them as Linux does: each word with an
509/// all-or-nothing store to a writable page, in kernel order.
510async fn overwrite_failed_gettimeofday_tv<'a, G, T>(
511    guest: &mut G,
512    tv_addr: AddrMut<'a, Timeval>,
513    tv: &Timeval,
514    before: &TimevalWordSnapshot,
515) -> Result<(), Error>
516where
517    G: Guest<Detcore<T>>,
518    T: RecordOrReplay,
519{
520    let values = [tv.tv_sec as libc::time_t, tv.tv_usec as libc::time_t];
521    for (index, (field, offset)) in TIMEVAL_WORDS.into_iter().enumerate() {
522        let addr = timeval_word_addr(field, tv_addr, offset)?;
523        let probe = syscalls::Time::new().with_tloc(Some(addr));
524        match classify_time_store_probe(field, guest.inject(probe).await)? {
525            TimeStoreProbe::Stored => {
526                let bytes = values[index].to_ne_bytes();
527                let overwrite = guest
528                    .memory()
529                    .write_with_user_access(addr.cast::<u8>(), &bytes);
530                require_complete_time_word_overwrite(field, bytes.len(), overwrite)?;
531            }
532            TimeStoreProbe::Stopped => {
533                let control = syscalls::Time::new().with_tloc(None);
534                require_native_time_control_probe(field, guest.inject(control).await)?;
535                for (stopped, (field, offset)) in TIMEVAL_WORDS.into_iter().enumerate().skip(index)
536                {
537                    let Some(addr) = stopped_timeval_word_addr(tv_addr, offset) else {
538                        continue;
539                    };
540                    let after = guest.memory().read_value(addr);
541                    match require_unchanged_stopped_word(field, before[stopped], after)? {
542                        StoppedWord::Unchanged => {}
543                        StoppedWord::Unreadable => {
544                            let maps = guest.storable_memory_ranges();
545                            require_unmapped_unreadable_word(field, maps, addr.as_raw() as u64)?;
546                        }
547                    }
548                }
549                break;
550            }
551        }
552    }
553    Ok(())
554}

620        &self,
621        guest: &mut G,
622        call: syscalls::Gettimeofday,
623    ) -> Result<i64, Error> {
624        let time_ns = guest_clock_time(guest).await;
625
626        // What Detcore could read in `tv` before the call; the repair uses it
627        // to confirm which words a failing call left alone.
628        let before = call.tv().map(|tp| snapshot_timeval_words(guest, tp.into()));
629
630        // A call failing with EFAULT may still have stored host wall-clock time
631        // in `tv`, so keep its result until `tv` holds virtual time.
632        let result = self
633            .record_or_replay_preserving_tool_errors(guest, call)
634            .await;
635
636        let tv: Timeval = time_ns.into();
637
638        if let Some(tp) = call.tv() {
639            match (&result, &before) {
640                (Ok(_), _) => guest.memory().write_value(tp, &tv)?,
641                // Linux's gettimeofday fails only with EFAULT, which is taken
642                // to be its own even when a seccomp filter returned it without
643                // running the call. Any other error came from the backend, the
644                // tool, a seccomp filter or a replayed log, and says nothing
645                // about what reached `tv`, so memory is left alone.

647                    require_live_time_store_probe(self.cfg.replay_data.is_some())?;
648                    overwrite_failed_gettimeofday_tv(guest, tp.into(), &tv, before).await?

768    /// Helper function to wait a given period, which may either succeed or be interrupted by a signal.
769    /// Return 0 or EINTR respectively.
770    async fn wait_and_return<R: Guest<Self>>(
771        &self,
772        guest: &mut R,
773        mut request: Resources,
774        call: NanosleepFamily,
775    ) -> Result<i64, Error> {
776        let target_time = time_from_resources(&request).expect("a sleepuntil resource request");
777        // Lets the scheduler end the sleep for a cross-task signal that
778        // interrupts it (`scheduler::sleep_signal`).
779        request.fyi(crate::resources::NANOSLEEP_FYI);
780        let mut claimed = false;
781        loop {
782            // A signal already pending ends the sleep before it parks, and one
783            // still pending after a wake ends it; a wake whose signal was taken
784            // back parks it again with its original absolute deadline
785            // (`super::signal::sleep_signal`).
786            if super::signal::sleep_signal(guest, SleepCheck::BeforePark, false).await?
787                == SleepSignal::Pending
788            {
789                break;
790            }
791            request.in_guest_sleep_sigalrm = self.in_guest_sleep_sigalrm_snapshot(guest).await;
792            match crate::tool_global::parked_wait_request(
793                guest,
794                request.clone(),
795                crate::scheduler::parked::ParkedWaitPolicy::NanosleepNoHandlerRestart {
796                    absolute_deadline: target_time,
797                },
798            )
799            .await
800            {
801                ResumeStatus::InGuestFatalAlarm => {
802                    // The default action terminates the guest. In particular,
803                    // do not observe time or write `rem` after this fatal wake.
804                    self.stage_in_guest_fatal_alarm(guest).await;
805                    return Err(Errno::EINTR.into());
806                }
807                // A sleep never asks for `NothingElseCanRun`.
808                ResumeStatus::Normal | ResumeStatus::NothingElseCanRun => return Ok(0),
809                ResumeStatus::Signaled(signals) => {
810                    match super::signal::sleep_signal(guest, SleepCheck::AfterWake(signals), false)
811                        .await?
812                    {
813                        SleepSignal::Pending | SleepSignal::Held => break,
814                        SleepSignal::Claimed => {
815                            claimed = true;
816                            break;
817                        }
818                        SleepSignal::None => trace!(
819                            "nanosleep woken for a signal no longer pending for it; sleeping again until {}",
820                            target_time
821                        ),
822                    }
823                }
824            }
825        }
826        let now = thread_observe_time(guest).await;
827        let delta = remaining_sleep_duration(target_time, now);
828        // Linux never touches remain for TIMER_ABSTIME, even when
829        // a caught signal interrupts the absolute sleep.
830        let addr2 = if call.flags() & libc::TIMER_ABSTIME == 0 {
831            call.rem()
832        } else {
833            None
834        };
835        if let Some(addr2) = addr2 {
836            info!(
837                "[interrupted] sleep till (until {}), woke up {:?} early, writing into nanosleep rem argument.",
838                target_time, delta
839            );
840            let t = Timespec {
841                tv_sec: delta.as_secs() as i64,
842                tv_nsec: delta.subsec_nanos() as i64,
843            };
844            guest.memory().write_value(addr2, &t)?;
845        } else {
846            info!("[interrupted] nanosleep rem argument is null, not writing it.")
847        }
848        if claimed {
849            // A claimed `SIGCHLD` is held for delivery as the thread resumes
850            // (`SleepSignal::Claimed`): `EINTR` if a handler runs, else the
851            // kernel's restart. As Linux's own `nanosleep`, a relative sleep
852            // returns `ERESTART_RESTARTBLOCK` and keeps its deadline for the
853            // `restart_syscall` the kernel runs only when no handler ran
854            // (`helpers::RestartBlock`, `resume_nanosleep`), so a new call,
855            // from a handler or a retry, cannot take it. An absolute one
856            // returns `ERESTARTNOHAND`: re-executed, it reads the same target.
857            if call.flags() & libc::TIMER_ABSTIME != 0 {
858                return Err(reverie::Error::Errno(Errno::ERESTARTNOHAND));
859            }
860            let result = Err(reverie::Error::Errno(Errno::ERESTART_RESTARTBLOCK));
861            crate::syscalls::helpers::keep_restart_block(
862                guest,
863                &result,
864                crate::syscalls::helpers::RestartCall::Nanosleep(call),
865                Some(target_time),
866            )
867            .await;
868            return result;
869        }
870        Err(reverie::Error::Errno(Errno::EINTR))
871    }

875    /// Like Linux's restart block, it keeps the deadline it already has and

877    pub(crate) async fn resume_nanosleep<R: Guest<Self>>(

887    /// clock_nanosleep and nanosleep
888    pub async fn handle_nanosleep_family<R: Guest<Self>>(
889        &self,
890        guest: &mut R,
891        call: NanosleepFamily,
892    ) -> Result<i64, Error> {
893        if call.flags() > libc::TIMER_ABSTIME {
894            trace!("Unhandled clock_nanosleep flags, letting syscall through...");
895            return Ok(guest.inject(Syscall::from(call)).await?);
896        }
897
898        let addr = call.req().ok_or(Errno::EFAULT)?;
899        let t: Timespec = guest.memory().read_value(addr)?;
900
901        // Linux validates the requested interval BEFORE sleeping: nanosleep(2)
902        // and clock_nanosleep(2) both fail EINVAL when tv_nsec is outside
903        // [0, 999999999] or tv_sec is negative.
904        //
905        // Detcore skipped that check and fed the raw fields through `as u64`.
906        // `Timespec` stores both as i64, so `tv_sec = -1` wrapped to
907        // u64::MAX (~1.8e19 seconds) and became `SleepUntil(INDEFINITE)`: the
908        // only guest thread parked with no deadline, the run queue emptied, and
909        // `step2d_handle_empty_queue` deliberately never jumps the clock for an
910        // indefinite waiter (doing so would also wake a `pause(2)`). The
911        // container then died -- exit 1, "Sandbox container exited
912        // unexpectedly" -- where Linux returns an errno and keeps running.
913        //
914        // A past *absolute* deadline is NOT an error and must still return 0,
915        // so this rejects only malformed fields, never an early deadline.
916        if t.tv_sec < 0 || t.tv_nsec < 0 || t.tv_nsec > 999_999_999 {
917            return Err(Errno::EINVAL.into());
918        }
919
920        match call.flags() {
921            0 => {
922                if self.cfg.sequentialize_threads {
923                    let time = Duration::from_secs(t.tv_sec as u64)
924                        + Duration::from_nanos(t.tv_nsec as u64);
925                    let request = Self::sleep_request(guest, time).await;
926                    trace!(
927                        "nanosleep adding delta {:?} to yield request {:?}",
928                        time, &request
929                    );
930                    self.wait_and_return(guest, request, call).await
931                } else {
932                    trace!("Not sequentializing threads, letting nanosleep through...");
933                    Ok(guest.inject(Syscall::from(call)).await?)
934                }
935            }
936            libc::TIMER_ABSTIME => {
937                let target_time = LogicalTime::from_secs(t.tv_sec as u64)
938                    + LogicalTime::from_nanos(t.tv_nsec as u64);
939                if self.cfg.sequentialize_threads {
940                    if self.cfg.virtualize_time {
941                        let request = Self::sleep_request_abs(guest, target_time).await;
942                        trace!(
943                            "nanosleep setting absolute time {:?} to yield request {:?}",
944                            target_time, &request
945                        );
946                        self.wait_and_return(guest, request, call).await
947                    } else {
948                        // TODO T124594597: Record-replay case here, need better ideas to enable proper handling of this case.
949                        error!(
950                            "Sequentializing but not virtualizing, so can't rely on passed abs time, especially when replaying a recording, just yelding"
951                        );
952                        let request = Self::yield_request(guest);
953                        self.wait_and_return(guest, request, call).await
954                    }
955                } else if self.cfg.virtualize_time {
956                    trace!(
957                        "Not sequentializing, but virtualizing so calculating relative time and invoking nanosleep..."
958                    );
959                    let relative_ts = Self::relative_time_from_abs_target(guest, target_time).await;
960                    let mut stack = guest.stack().await;
961                    let req = stack.push(relative_ts);
962                    stack.commit()?;
963                    let modified_call = syscalls::Nanosleep::new().with_req(Some(req));
964                    Ok(guest.inject(modified_call).await?)
965                } else {
966                    trace!(
967                        "Not sequentializing threads not virtualizing, letting nanosleep through..."
968                    );
969                    Ok(guest.inject(Syscall::from(call)).await?)
970                }
971            }
972            _ => unreachable!("Unexpected, unhandled flag value"),
973        }
974    }

976    async fn relative_time_from_abs_target<G: Guest<Self>>(
977        guest: &mut G,
978        target_time: LogicalTime,
979    ) -> Timespec {
980        let base_time = thread_observe_time(guest).await;
981
982        // An absolute deadline already in the past is NOT an error on Linux --
983        // clock_nanosleep(TIMER_ABSTIME) simply returns 0 without sleeping.
984        // `LogicalTime`'s `Sub` is a plain subtraction (unlike its `Add` impls,
985        // which saturate deliberately), so `target_time - base_time` underflows
986        // for a past deadline: a debug build panics with "attempt to subtract
987        // with overflow", and a release build wraps to an enormous interval --
988        // the same effectively-indefinite sleep this handler exists to avoid.
989        //
990        // Clamped here rather than by making the shared operator saturate,
991        // because this is the only subtraction of two `LogicalTime`s in the
992        // tree and a silently-saturating operator could hide a real underflow
993        // in some future caller.
994        let relative_logical = if target_time <= base_time {
995            LogicalTime::from_nanos(0)
996        } else {
997            target_time - base_time
998        };
999
1000        Timespec {
1001            tv_sec: relative_logical.as_secs() as i64,
1002            tv_nsec: relative_logical.subsec_nanos() as i64,
1003        }
1004    }

1021    /// timer_create: allocate a per-process POSIX timer and hand back a
1022    /// deterministic id, retaining any scheduler-deliverable signal and, for
1023    /// `SIGEV_THREAD_ID`, the thread it notifies. The notification is checked
1024    /// as Linux's `good_sigevent` does: `SIGEV_SIGNAL` and `SIGEV_THREAD` (which
1025    /// the kernel treats alike; libc implements the thread itself) need a
1026    /// signal from 1 to 64; `SIGEV_THREAD_ID` needs, first, a thread of the
1027    /// caller's own process (<https://github.com/rrnewton/hermit/issues/4164>);
1028    /// any other value is EINVAL.
1029    pub async fn handle_timer_create<G: Guest<Self>>(
1030        &self,
1031        guest: &mut G,
1032        call: syscalls::TimerCreate,
1033    ) -> Result<i64, Error> {
1034        // The kernel writes the new timer id here; a null pointer is EFAULT.
1035        let timerid_ptr = call.timerid().ok_or(Errno::EFAULT)?;
1036        let clockid = call.clockid();
1037        let (signo, thread, value) = if let Some(event_ptr) = call.sevp() {
1038            // Only the plain fields: `sigevent` holds a pointer, which must not
1039            // be kept across the query below. The value is kept as the 64 bits
1040            // Linux copies into `si_value`.
1041            let (notify, signo, named, value) = {
1042                let event: libc::sigevent = guest.memory().read_value(event_ptr)?;
1043                (
1044                    event.sigev_notify,
1045                    event.sigev_signo,
1046                    event.sigev_notify_thread_id,
1047                    event.sigev_value.sival_ptr as usize as u64,
1048                )
1049            };
1050            // A reserved in-guest signal: Linux answers for the clock right
1051            // after copying the sigevent, before it looks at the notification
1052            // (`do_timer_create`), so an unsupported clock is that answer even
1053            // when the notification is invalid too
1054            // (`timer_create_clock_refusal`).
1055            if guest.config().reserves_in_guest_signal(signo.into())
1056                && let Some(errno) = timer_create_clock_refusal(clockid)
1057            {
1058                return Err(errno.into());
1059            }
1060            let thread = match notify {
1061                libc::SIGEV_NONE => None,
1062                libc::SIGEV_SIGNAL | libc::SIGEV_THREAD => None,
1063                reverie::syscalls::linux_abi::SIGEV_THREAD_ID => {
1064                    match posix_timer_thread(guest, named).await {
1065                        PosixTimerThread::Thread(thread) => Some(thread),
1066                        PosixTimerThread::NotInGroup => return Err(Errno::EINVAL.into()),
1067                        PosixTimerThread::NestedNamespace => {
1068                            tracing::error!(
1069                                "[detcore, dtid {}] timer_create with SIGEV_THREAD_ID is not supported in a nested PID namespace: the thread id is a pid of that namespace, which Detcore does not translate (https://github.com/rrnewton/hermit/issues/4155)",
1070                                guest.thread_state().dettid,
1071                            );
1072                            return self
1073                                .refuse_unserviceable_operation(
1074                                    guest,
1075                                    Sysno::timer_create,
1076                                    Errno::ENOSYS,
1077                                )
1078                                .await;
1079                        }
1080                    }
1081                }
1082                _ => return Err(Errno::EINVAL.into()),
1083            };
1084            if notify == libc::SIGEV_NONE {
1085                (None, None, Some(value))
1086            } else if !(1..=64).contains(&signo) {
1087                return Err(Errno::EINVAL.into());
1088            } else {
1089                (Some(signo), thread, Some(value))
1090            }
1091        } else {
1092            (Some(libc::SIGALRM), None, None)
1093        };
1094        // A reserved in-guest signal as the timer's notification signal, for
1095        // each notification that delivers one (`SIGEV_SIGNAL`, `SIGEV_THREAD`,
1096        // `SIGEV_THREAD_ID`), after Linux's answers for the clock and the
1097        // sigevent above. With no `sigevent` the signal is SIGALRM, which is
1098        // never reserved.
1099        if let Some(signo) = signo
1100            && guest.config().reserves_in_guest_signal(signo.into())
1101        {
1102            Self::refuse_reserved_signal_arming(
1103                guest,
1104                crate::refused_signal_controls::ReservedArming::TimerCreate,
1105                signo.into(),
1106            )?;
1107        }
1108        // `None` for a real-time signal (32 to 64): see `handle_timer_settime`.
1109        let signal = signo.and_then(|signo| Signal::try_from(signo).ok());
1110        if signal == Some(Signal::SIGALRM) {
1111            refuse_sigalrm(guest, SigalrmControl::ArmProducer).await?;
1112        }
1113        let id = {
1114            let mut timers = guest.thread_state().posix_timers.lock().unwrap();
1115            timers.create(signo, thread, value)
1116        };
1117        guest
1118            .memory()
1119            .write_value(timerid_ptr, &(id as libc::c_int))?;
1120        // The process has a POSIX timer on this signal from its creation, armed
1121        // or not, which decides how a kept send standing on another sender's
1122        // record is classified (https://github.com/rrnewton/hermit/issues/4262).
1123        if crate::scheduler::discarded_at_send::records(guest.config())
1124            && let Some(signo) = signo
1125            && (1..32).contains(&signo)
1126        {
1127            crate::tool_global::discarded_signal(
1128                guest,
1129                crate::scheduler::discarded_at_send::DiscardedSignalControl::TimerCreated {
1130                    signal: signo,
1131                },
1132            )
1133            .await;
1134        }
1135        detlog!(
1136            "[dtid {}] timer_create(clockid={:?}) => deterministic timer id {}, signal {:?}, thread {:?}",
1137            guest.thread_state().dettid,
1138            clockid,
1139            id,
1140            signal,
1141            thread,
1142        );
1143        Ok(0)
1144    }

1148    /// timer_settime: arm or disarm a timer against the deterministic virtual
1149    /// clock. The old arming is reported through `old_value` when requested.
1150    pub async fn handle_timer_settime<G: Guest<Self>>(
1151        &self,
1152        guest: &mut G,
1153        call: syscalls::TimerSettime,
1154    ) -> Result<i64, Error> {
1155        let id = call.timerid();
1156        let new_ptr = call.new_value().ok_or(Errno::EINVAL)?;
1157        let new: libc::itimerspec = guest.memory().read_value(new_ptr)?;
1158        let interval_ns = timespec_to_ns(new.it_interval);
1159        let value_ns = timespec_to_ns(new.it_value);
1160
1161        let now = thread_observe_time(guest).await;
1162        let deadline = if value_ns == 0 {
1163            None
1164        } else if call.flags() & libc::TIMER_ABSTIME != 0 {
1165            // Absolute expiration is interpreted against the same virtual clock.
1166            Some(LogicalTime::from_nanos(value_ns))
1167        } else {
1168            Some(now + Duration::from_nanos(value_ns))
1169        };
1170
1171        let (old, signal_number, thread, value) = {
1172            let mut timers = guest.thread_state().posix_timers.lock().unwrap();
1173            let old = timers.settime(id, interval_ns, deadline, now);
1174            let signal = timers.signal(id);
1175            (
1176                old,
1177                signal,
1178                timers.thread(id),
1179                timers.value(id).unwrap_or(0),
1180            )
1181        };
1182        let (mut old_remaining_ns, old_interval_ns) = old.ok_or(Errno::EINVAL)?;
1183        let signal_number = signal_number.ok_or(Errno::EINVAL)?;
1184        // The old value is the timer's remaining time as `timer_gettime`
1185        // reads it now, from the scheduler where it keeps the timer's phase
1186        // (4262 design §5).
1187        if let Some((Some(remaining), _)) = posix_timer_readout(guest, id, now).await {
1188            old_remaining_ns = remaining.as_nanos();
1189        }
1190
1191        // Linux changes the timer, its generation and its overrun before it
1192        // copies the old value out (`do_timer_settime`), and an EFAULT there
1193        // undoes none of it, so the scheduler is told first (the Codex review
1194        // of https://github.com/rrnewton/hermit/pull/4491, P2-2).
1195        if let Some(signo) = signal_number {
1196            if self.posix_timer_expiry_is_registered(signo) {
1197                register_posix_timer(
1198                    guest,
1199                    id,
1200                    deadline,
1201                    LogicalTime::from_nanos(interval_ns),
1202                    PosixTimerNotification {
1203                        sig: SigWrapper(signo),
1204                        thread,
1205                        value,
1206                        deleted: false,
1207                    },
1208                )
1209                .await;
1210            } else if deadline.is_some() {
1211                tracing::error!(
1212                    "[detcore, dtid {}] POSIX timer {} notifies with real-time signal {}, which Detcore does not deliver from a timer; it is armed but will never fire",
1213                    guest.thread_state().dettid,
1214                    id,
1215                    signo,
1216                );
1217            }
1218        }
1219
1220        if let Some(old_ptr) = call.old_value() {
1221            let old_spec = libc::itimerspec {
1222                it_interval: ns_to_timespec(old_interval_ns),
1223                it_value: ns_to_timespec(old_remaining_ns),
1224            };
1225            guest.memory().write_value(old_ptr, &old_spec)?;
1226        }
1227
1228        detlog!(
1229            "[dtid {}] timer_settime(id={}, interval_ns={}, value_ns={}) armed against virtual clock",
1230            guest.thread_state().dettid,
1231            id,
1232            interval_ns,
1233            value_ns,
1234        );
1235        Ok(0)
1236    }

1238    /// timer_gettime: report the time remaining until the next expiration and
1239    /// the reload interval, both computed from the virtual clock.
1240    pub async fn handle_timer_gettime<G: Guest<Self>>(
1241        &self,
1242        guest: &mut G,
1243        call: syscalls::TimerGettime,
1244    ) -> Result<i64, Error> {
1245        let id = call.timerid();
1246        let value_ptr = call.value().ok_or(Errno::EFAULT)?;
1247        let now = thread_observe_time(guest).await;
1248        let cur = {
1249            let timers = guest.thread_state().posix_timers.lock().unwrap();
1250            timers.gettime(id, now)
1251        };
1252        let (mut remaining_ns, interval_ns) = cur.ok_or(Errno::EINVAL)?;
1253        // Where the scheduler coalesces timers, it knows whether this one
1254        // waits for the dequeue of its signal (4262 design §5).
1255        if let Some((Some(remaining), _)) = posix_timer_readout(guest, id, now).await {
1256            remaining_ns = remaining.as_nanos();
1257        }
1258        let spec = libc::itimerspec {
1259            it_interval: ns_to_timespec(interval_ns),
1260            it_value: ns_to_timespec(remaining_ns),
1261        };
1262        guest.memory().write_value(value_ptr, &spec)?;
1263        Ok(0)
1264    }

1266    /// timer_getoverrun: the expirations the timer's last delivered signal
1267    /// missed (`it_overrun_last`), where the scheduler coalesces timers
1268    /// (<https://github.com/rrnewton/hermit/issues/4262>); 0 elsewhere.
1269    pub async fn handle_timer_getoverrun<G: Guest<Self>>(
1270        &self,
1271        guest: &mut G,
1272        call: syscalls::TimerGetoverrun,
1273    ) -> Result<i64, Error> {
1274        let id = call.timerid();
1275        let exists = guest
1276            .thread_state()
1277            .posix_timers
1278            .lock()
1279            .unwrap()
1280            .contains(id);
1281        if !exists {
1282            return Err(Errno::EINVAL.into());
1283        }
1284        // The overrun is the scheduler's record alone, not a function of the
1285        // caller's time, which is not observed here.
1286        Ok(posix_timer_readout(guest, id, LogicalTime::ZERO)
1287            .await
1288            .map_or(0, |(_, overrun)| overrun as i64))
1289    }

1341/// The scheduler's remaining time (`None`: the caller's own answer stands)
1342/// and overrun of the calling process's POSIX timer `id` at `now`, where it
1343/// keeps them (`DiscardedSignalControl::TimerReadout`); `None` elsewhere.
1344async fn posix_timer_readout<G: Guest<Detcore<T>>, T: RecordOrReplay>(
1345    guest: &mut G,
1346    id: i32,
1347    now: LogicalTime,
1348) -> Option<(Option<LogicalTime>, i32)> {
1349    use crate::scheduler::discarded_at_send::DiscardedSignalAnswer;
1350    use crate::scheduler::discarded_at_send::DiscardedSignalControl;
1351    if !crate::scheduler::discarded_at_send::records(guest.config()) {
1352        return None;
1353    }
1354    match crate::tool_global::discarded_signal(
1355        guest,
1356        DiscardedSignalControl::TimerReadout { timer: id, now },
1357    )
1358    .await
1359    {
1360        DiscardedSignalAnswer::TimerReadout { remaining, overrun } => Some((remaining, overrun)),
1361        _ => None,
1362    }
