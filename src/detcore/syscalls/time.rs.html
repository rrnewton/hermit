42fn time_from_resources(rsrcs: &Resources) -> Option<LogicalTime> {
43    if rsrcs.resources.len() > 1 {
44        panic!(
45            "time_from_resources: multiple resource ids in resource request: {:?}",
46            rsrcs
47        );
48    }
49    for rs in rsrcs.resources.iter() {
50        if let (ResourceID::SleepUntil(tm), _) = rs {
51            return Some(*tm);
52        }
53    }
54    None
55}

57/// Flatten a `timespec` to nanoseconds. Negative fields are not valid for the
58/// timer syscalls we handle; treat them as zero rather than panicking.
59fn timespec_to_ns(ts: libc::timespec) -> u64 {
60    let secs = ts.tv_sec.max(0) as u64;
61    let nsec = ts.tv_nsec.max(0) as u64;
62    secs.saturating_mul(1_000_000_000).saturating_add(nsec)
63}
64
65/// Inverse of [`timespec_to_ns`].
66fn ns_to_timespec(ns: u64) -> libc::timespec {
67    libc::timespec {
68        tv_sec: (ns / 1_000_000_000) as libc::time_t,
69        tv_nsec: (ns % 1_000_000_000) as libc::c_long,
70    }
71}

79// AUTONOMOUS-BOT-IMPLEMENTED
80// TODO-HUMAN-REVIEW(PR-857): Host-independent NTP discipline snapshot.
81fn deterministic_timex(now: Timespec) -> libc::timex {
82    // SAFETY: `libc::timex` contains only integer fields and padding; zero is a
83    // valid baseline for the fields not modeled by Hermit.
84    let mut tx: libc::timex = unsafe { std::mem::zeroed() };
85    tx.status = libc::STA_UNSYNC;
86    tx.tick = 10_000;
87    tx.time = libc::timeval {
88        tv_sec: now.tv_sec,
89        tv_usec: now.tv_nsec / 1_000,
90    };
91    tx
92}

105fn remaining_sleep_duration(target: LogicalTime, now: LogicalTime) -> Duration {
106    if target > now {
107        target.duration_since(now)
108    } else {
109        Duration::ZERO
110    }

225        &self,

228    ) -> Result<i64, Error> {
229        // A NULL `res` pointer is valid for clock_getres: the kernel validates
230        // the clockid and returns 0 without storing the resolution. GHC's RTS

235        if let Some(res) = call.res() {
236            // For now we report a constant clock res of 10ms:
237            let clock_res = 10;
238
239            let t = Timespec {
240                tv_sec: 0,
241                tv_nsec: 1000 * clock_res as i64,
242            };
243
244            guest.memory().write_value(res, &t)?;
245        }
246
247        Ok(0)
248    }

293    /// Helper function to wait a given period, which may either succeed or be interrupted by a signal.
294    /// Return 0 or EINTR respectively.
295    async fn wait_and_return<R: Guest<Self>>(
296        guest: &mut R,
297        request: Resources,
298        call: NanosleepFamily,
299    ) -> Result<i64, Error> {
300        let target_time = time_from_resources(&request).expect("a sleepuntil resource request");
301        match resource_request(guest, request).await {
302            ResumeStatus::Normal => Ok(0),
303            ResumeStatus::Signaled(_) => {
304                let now = thread_observe_time(guest).await;
305                let delta = remaining_sleep_duration(target_time, now);
306                let addr2 = call.rem();
307                if let Some(addr2) = addr2 {
308                    info!(
309                        "[interrupted] sleep till (until {}), woke up {:?} early, writing into nanosleep rem argument.",
310                        target_time, delta
311                    );
312                    let t = Timespec {
313                        tv_sec: delta.as_secs() as i64,
314                        tv_nsec: delta.subsec_nanos() as i64,
315                    };
316                    guest.memory().write_value(addr2, &t)?;
317                } else {
318                    info!("[interrupted] nanosleep rem argument is null, not writing it.")
319                }
320                Err(reverie::Error::Errno(Errno::EINTR))
321            }
322        }
323    }

325    /// clock_nanosleep and nanosleep
326    pub async fn handle_nanosleep_family<R: Guest<Self>>(
327        &self,
328        guest: &mut R,
329        call: NanosleepFamily,
330    ) -> Result<i64, Error> {
331        if call.flags() > libc::TIMER_ABSTIME {
332            trace!("Unhandled clock_nanosleep flags, letting syscall through...");
333            return Ok(guest.inject(Syscall::from(call)).await?);
334        }
335
336        // TODO: use 2nd, `rem` argument when providing a way for a signal to interrupt the
337        // logical sleep.
338        let addr = call.req().ok_or(Errno::EFAULT)?;
339        let t: Timespec = guest
340            .memory()
341            .read_value(addr)
342            .expect("should be able to read from memory");
343
344        // Linux validates the requested interval BEFORE sleeping: nanosleep(2)
345        // and clock_nanosleep(2) both fail EINVAL when tv_nsec is outside
346        // [0, 999999999] or tv_sec is negative.
347        //
348        // Detcore skipped that check and fed the raw fields through `as u64`.
349        // `Timespec` stores both as i64, so `tv_sec = -1` wrapped to
350        // u64::MAX (~1.8e19 seconds) and became `SleepUntil(INDEFINITE)`: the
351        // only guest thread parked with no deadline, the run queue emptied, and
352        // `step2d_handle_empty_queue` deliberately never jumps the clock for an
353        // indefinite waiter (doing so would also wake a `pause(2)`). The
354        // container then died -- exit 1, "Sandbox container exited
355        // unexpectedly" -- where Linux returns an errno and keeps running.
356        //
357        // A past *absolute* deadline is NOT an error and must still return 0,
358        // so this rejects only malformed fields, never an early deadline.
359        if t.tv_sec < 0 || t.tv_nsec < 0 || t.tv_nsec > 999_999_999 {
360            return Err(Errno::EINVAL.into());
361        }
362
363        match call.flags() {
364            0 => {
365                if self.cfg.sequentialize_threads {
366                    let time = Duration::from_secs(t.tv_sec as u64)
367                        + Duration::from_nanos(t.tv_nsec as u64);
368                    let request = Self::sleep_request(guest, time).await;
369                    trace!(
370                        "nanosleep adding delta {:?} to yield request {:?}",
371                        time, &request
372                    );
373                    Self::wait_and_return(guest, request, call).await
374                } else {
375                    trace!("Not sequentializing threads, letting nanosleep through...");
376                    Ok(guest.inject(Syscall::from(call)).await?)
377                }
378            }
379            libc::TIMER_ABSTIME => {
380                let target_time = LogicalTime::from_secs(t.tv_sec as u64)
381                    + LogicalTime::from_nanos(t.tv_nsec as u64);
382                if self.cfg.sequentialize_threads {
383                    if self.cfg.virtualize_time {
384                        let request = Self::sleep_request_abs(guest, target_time).await;
385                        trace!(
386                            "nanosleep setting absolute time {:?} to yield request {:?}",
387                            target_time, &request
388                        );
389                        Self::wait_and_return(guest, request, call).await
390                    } else {
391                        // TODO T124594597: Record-replay case here, need better ideas to enable proper handling of this case.
392                        error!(
393                            "Sequentializing but not virtualizing, so can't rely on passed abs time, especially when replaying a recording, just yelding"
394                        );
395                        let request = Self::yield_request(guest);
396                        Self::wait_and_return(guest, request, call).await
397                    }
398                } else if self.cfg.virtualize_time {
399                    trace!(
400                        "Not sequentializing, but virtualizing so calculating relative time and invoking nanosleep..."
401                    );
402                    let relative_ts = Self::relative_time_from_abs_target(guest, target_time).await;
403                    let mut stack = guest.stack().await;
404                    let req = stack.push(relative_ts);
405                    stack.commit()?;
406                    let modified_call = syscalls::Nanosleep::new().with_req(Some(req));
407                    Ok(guest.inject(modified_call).await?)
408                } else {
409                    trace!(
410                        "Not sequentializing threads not virtualizing, letting nanosleep through..."
411                    );
412                    Ok(guest.inject(Syscall::from(call)).await?)
413                }
414            }
415            _ => unreachable!("Unexpected, unhandled flag value"),
416        }
417    }

419    async fn relative_time_from_abs_target<G: Guest<Self>>(
420        guest: &mut G,
421        target_time: LogicalTime,
422    ) -> Timespec {
423        let base_time = thread_observe_time(guest).await;
424
425        // An absolute deadline already in the past is NOT an error on Linux --
426        // clock_nanosleep(TIMER_ABSTIME) simply returns 0 without sleeping.
427        // `LogicalTime`'s `Sub` is a plain subtraction (unlike its `Add` impls,
428        // which saturate deliberately), so `target_time - base_time` underflows
429        // for a past deadline: a debug build panics with "attempt to subtract
430        // with overflow", and a release build wraps to an enormous interval --
431        // the same effectively-indefinite sleep this handler exists to avoid.
432        //
433        // Clamped here rather than by making the shared operator saturate,
434        // because this is the only subtraction of two `LogicalTime`s in the
435        // tree and a silently-saturating operator could hide a real underflow
436        // in some future caller.
437        let relative_logical = if target_time <= base_time {
438            LogicalTime::from_nanos(0)
439        } else {
440            target_time - base_time
441        };
442
443        Timespec {
444            tv_sec: relative_logical.as_secs() as i64,
445            tv_nsec: relative_logical.subsec_nanos() as i64,
446        }
447    }

451    /// timer_create: allocate a per-process POSIX timer and hand back a
452    /// deterministic id, retaining any scheduler-deliverable signal.
453    pub async fn handle_timer_create<G: Guest<Self>>(
454        &self,
455        guest: &mut G,
456        call: syscalls::TimerCreate,
457    ) -> Result<i64, Error> {
458        // The kernel writes the new timer id here; a null pointer is EFAULT.
459        let timerid_ptr = call.timerid().ok_or(Errno::EFAULT)?;
460        let clockid = call.clockid();
461        let signal = if let Some(event_ptr) = call.sevp() {
462            let event: libc::sigevent = guest.memory().read_value(event_ptr)?;
463            match event.sigev_notify {
464                libc::SIGEV_NONE => None,
465                // Linux uses 4 for SIGEV_THREAD_ID. Treat it as process-directed
466                // until Detcore tracks per-timer thread targeting.
467                libc::SIGEV_SIGNAL | 4 => {
468                    if !(1..=64).contains(&event.sigev_signo) {
469                        return Err(Errno::EINVAL.into());
470                    }
471                    Signal::try_from(event.sigev_signo).ok()
472                }
473                _ => return Err(Errno::ENOSYS.into()),
474            }
475        } else {
476            Some(Signal::SIGALRM)
477        };
478        let id = {
479            let mut timers = guest.thread_state().posix_timers.lock().unwrap();
480            timers.create(signal.map(|sig| sig as i32))
481        };
482        guest
483            .memory()
484            .write_value(timerid_ptr, &(id as libc::c_int))?;
485        detlog!(
486            "[dtid {}] timer_create(clockid={:?}) => deterministic timer id {}, signal {:?}",
487            guest.thread_state().dettid,
488            clockid,
489            id,
490            signal,
491        );
492        Ok(0)
493    }

497    /// timer_settime: arm or disarm a timer against the deterministic virtual
498    /// clock. The old arming is reported through `old_value` when requested.
499    pub async fn handle_timer_settime<G: Guest<Self>>(
500        &self,
501        guest: &mut G,
502        call: syscalls::TimerSettime,
503    ) -> Result<i64, Error> {
504        let id = call.timerid();
505        let new_ptr = call.new_value().ok_or(Errno::EINVAL)?;
506        let new: libc::itimerspec = guest.memory().read_value(new_ptr)?;
507        let interval_ns = timespec_to_ns(new.it_interval);
508        let value_ns = timespec_to_ns(new.it_value);
509
510        let now = thread_observe_time(guest).await;
511        let deadline = if value_ns == 0 {
512            None
513        } else if call.flags() & libc::TIMER_ABSTIME != 0 {
514            // Absolute expiration is interpreted against the same virtual clock.
515            Some(LogicalTime::from_nanos(value_ns))
516        } else {
517            Some(now + Duration::from_nanos(value_ns))
518        };
519
520        let (old, signal_number) = {
521            let mut timers = guest.thread_state().posix_timers.lock().unwrap();
522            let old = timers.settime(id, interval_ns, deadline, now);
523            let signal = timers.signal(id);
524            (old, signal)
525        };
526        let (old_remaining_ns, old_interval_ns) = old.ok_or(Errno::EINVAL)?;
527        let signal_number = signal_number.ok_or(Errno::EINVAL)?;
528
529        if let Some(old_ptr) = call.old_value() {
530            let old_spec = libc::itimerspec {
531                it_interval: ns_to_timespec(old_interval_ns),
532                it_value: ns_to_timespec(old_remaining_ns),
533            };
534            guest.memory().write_value(old_ptr, &old_spec)?;
535        }
536
537        if let Some(signal) = signal_number.and_then(|signum| Signal::try_from(signum).ok()) {
538            register_posix_timer(
539                guest,
540                id,
541                deadline,
542                LogicalTime::from_nanos(interval_ns),
543                signal,
544            )
545            .await;
546        }
547
548        detlog!(
549            "[dtid {}] timer_settime(id={}, interval_ns={}, value_ns={}) armed against virtual clock",
550            guest.thread_state().dettid,
551            id,
552            interval_ns,
553            value_ns,
554        );
555        Ok(0)
556    }

577        guest.memory().write_value(value_ptr, &spec)?;

605    pub async fn handle_timer_delete<G: Guest<Self>>(
606        &self,
607        guest: &mut G,
608        call: syscalls::TimerDelete,
609    ) -> Result<i64, Error> {
610        let id = call.timerid();
611        let signal_number = guest
612            .thread_state()
613            .posix_timers
614            .lock()
615            .unwrap()
616            .signal(id)
617            .ok_or(Errno::EINVAL)?;
618        let existed = {
619            let mut timers = guest.thread_state().posix_timers.lock().unwrap();
620            timers.remove(id)
621        };
622        if existed {
623            if let Some(signal) = signal_number.and_then(|signum| Signal::try_from(signum).ok()) {
624                register_posix_timer(guest, id, None, LogicalTime::ZERO, signal).await;
625            }
626            detlog!(
627                "[dtid {}] timer_delete(id={})",
628                guest.thread_state().dettid,
629                id,
630            );
631            Ok(0)
632        } else {
633            Err(Errno::EINVAL.into())
634        }
635    }
636}
