46fn time_from_resources(rsrcs: &Resources) -> Option<LogicalTime> {
47    if rsrcs.resources.len() > 1 {
48        panic!(
49            "time_from_resources: multiple resource ids in resource request: {:?}",
50            rsrcs
51        );
52    }
53    for rs in rsrcs.resources.iter() {
54        if let (ResourceID::SleepUntil(tm), _) = rs {
55            return Some(*tm);
56        }
57    }
58    None
59}

61/// Flatten a `timespec` to nanoseconds. Negative fields are not valid for the
62/// timer syscalls we handle; treat them as zero rather than panicking.
63fn timespec_to_ns(ts: libc::timespec) -> u64 {
64    let secs = ts.tv_sec.max(0) as u64;
65    let nsec = ts.tv_nsec.max(0) as u64;
66    secs.saturating_mul(1_000_000_000).saturating_add(nsec)
67}
68
69/// Inverse of [`timespec_to_ns`].
70fn ns_to_timespec(ns: u64) -> libc::timespec {
71    libc::timespec {
72        tv_sec: (ns / 1_000_000_000) as libc::time_t,
73        tv_nsec: (ns % 1_000_000_000) as libc::c_long,
74    }
75}

85fn deterministic_timex(now: Timespec) -> libc::timex {
86    // SAFETY: `libc::timex` contains only integer fields and padding; zero is a
87    // valid baseline for the fields not modeled by Hermit.
88    let mut tx: libc::timex = unsafe { std::mem::zeroed() };
89    tx.status = libc::STA_UNSYNC;
90    tx.tick = 10_000;
91    tx.time = libc::timeval {
92        tv_sec: now.tv_sec,
93        tv_usec: now.tv_nsec / 1_000,
94    };
95    tx

109/// Replacing host time in the `tv` of a `gettimeofday` that failed with EFAULT
110/// could not finish. `tv` may still hold host wall-clock time, so this is a
111/// failed run, not a guest errno.
112#[derive(Debug, Clone, Copy, PartialEq, Eq)]
113struct TvRepairFailure {
114    /// The word being stored: `tv_sec` or `tv_usec`.
115    field: &'static str,
116    kind: TvRepairFailureKind,
117}
118
119#[derive(Debug, Clone, Copy, PartialEq, Eq)]
120enum TvRepairFailureKind {
121    /// The backend could not execute the `time(2)` probe. EFAULT is not a
122    /// failure: it means Linux stopped before storing this word.
123    ProbeFailed(Errno),
124    /// After a store probe returned EFAULT, `time(NULL)`, which Linux cannot
125    /// fail, failed too. Something other than the store, such as a seccomp
126    /// filter, produced the EFAULT, so it says nothing about `tv`.
127    ControlProbeFailed(Errno),
128    /// A word at or after the one whose store probe returned EFAULT no longer
129    /// reads as it did before the call, so the original call stored it and the
130    /// EFAULT did not come from the store.
131    StoppedWordChanged,
132    /// A word at or after the one whose store probe returned EFAULT could not
133    /// be read before or after the call, although every byte of it is mapped.
134    /// The read failure may not be a fault: a seccomp filter can deny the
135    /// read while the store went through, so nothing shows the word unstored.
136    StoppedWordUnreadable,
137    /// The guest's memory map, needed to tell an unmapped word from an
138    /// unreadable one, could not be read or was empty.
139    MapsUnavailable,
140    /// The exact overwrite after a successful probe failed. The probe just
141    /// stored host seconds, so the run cannot safely continue.
142    OverwriteFailed(Errno),
143    /// The one-shot overwrite reported a count other than the whole word.
144    OverwriteCount { expected: usize, reported: usize },
145    /// Adding the field offset to the guest's `timeval` address overflowed.
146    AddressOverflow,
147    /// A recorded EFAULT does not prove that this replay execution performed
148    /// the original stores, so live probing would mix replayed and host state.
149    ReplayMode,
150}
151
152impl std::fmt::Display for TvRepairFailure {
153    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
154        write!(
155            f,
156            "replacing host time in the {} of a failed gettimeofday: ",
157            self.field
158        )?;
159        match self.kind {
160            TvRepairFailureKind::ProbeFailed(errno) => {
161                write!(f, "time(2) store probe failed: {errno}")
162            }
163            TvRepairFailureKind::ControlProbeFailed(errno) => write!(
164                f,
165                "time(NULL) control probe failed: {errno}, so the store probe's EFAULT was not a store fault"
166            ),
167            TvRepairFailureKind::StoppedWordChanged => f.write_str(
168                "the word changed during the call although its store probe returned EFAULT",
169            ),
170            TvRepairFailureKind::StoppedWordUnreadable => f.write_str(
171                "the word is mapped but could not be read, so its store probe's EFAULT is unconfirmed",
172            ),
173            TvRepairFailureKind::MapsUnavailable => {
174                f.write_str("the guest's memory map could not be read")
175            }
176            TvRepairFailureKind::OverwriteFailed(errno) => {
177                write!(f, "virtual-time overwrite failed: {errno}")
178            }
179            TvRepairFailureKind::OverwriteCount { expected, reported } => write!(
180                f,
181                "virtual-time overwrite reported {reported} of {expected} bytes"
182            ),
183            TvRepairFailureKind::AddressOverflow => f.write_str("field address overflowed"),
184            TvRepairFailureKind::ReplayMode => {
185                f.write_str("cannot probe a recorded or replayed EFAULT")
186            }
187        }
188    }
189}

232/// How a word that Linux did not store reads after the call.
233#[derive(Debug, Clone, Copy, PartialEq, Eq)]
234enum StoppedWord {
235    /// Readable, with the value it held before the call.
236    Unchanged,
237    /// Unreadable both before and after the call. This proves nothing by
238    /// itself; see [`require_unmapped_unreadable_word`].
239    Unreadable,
240}

242/// A word Linux did not store keeps both its contents and its readability.
243fn require_unchanged_stopped_word(
244    field: &'static str,
245    before: Result<libc::time_t, Errno>,
246    after: Result<libc::time_t, Errno>,
247) -> Result<StoppedWord, Error> {
248    match (before, after) {
249        (Ok(before), Ok(after)) if before == after => Ok(StoppedWord::Unchanged),
250        (Err(_), Err(_)) => Ok(StoppedWord::Unreadable),
251        _ => Err(tv_repair_error(
252            field,
253            TvRepairFailureKind::StoppedWordChanged,
254        )),
255    }
256}

258/// Whether every byte of the `time_t` at `word` lies in one of `ranges`, the
259/// half-open address ranges of the guest's mappings.
260fn time_word_is_fully_mapped(mut ranges: Vec<(u64, u64)>, word: u64) -> bool {
261    let Some(end) = word.checked_add(std::mem::size_of::<libc::time_t>() as u64) else {
262        return false;
263    };
264    ranges.sort_unstable();
265    let mut covered = word;
266    for (start, stop) in ranges {
267        if start <= covered && covered < stop {
268            covered = stop;
269            if covered >= end {
270                return true;
271            }
272        }
273    }
274    false
275}

277/// An unreadable word is accepted as unstored only when part of it is
278/// unmapped: Linux's eight-byte `put_user` faults there before storing any
279/// byte. A failed read of a fully mapped word is not evidence of a fault: a
280/// seccomp filter can deny the read while the store succeeds. Ptrace reads use
281/// `FOLL_FORCE`, which bypasses page permissions, but some listed mappings
282/// still refuse the read: file pages beyond end of file, hardware-poisoned
283/// pages, `MADV_GUARD_INSTALL` guard regions, `VM_PFNMAP` mappings without an
284/// `access` operation, missing pages of a userfaultfd region in SIGBUS mode,
285/// and `[vsyscall]` in xonly mode. Backends
286/// that read with `process_vm_readv` also cannot read `PROT_NONE` pages. A
287/// `gettimeofday` whose store faults on any of these ends the run rather than
288/// returning EFAULT. An empty map is refused because a filter that fakes a
289/// successful zero-byte read would otherwise make every word look unmapped.
290fn require_unmapped_unreadable_word(
291    field: &'static str,
292    maps: Result<Vec<(u64, u64)>, Error>,
293    word: u64,
294) -> Result<(), Error> {
295    let ranges = match maps {
296        Ok(ranges) if !ranges.is_empty() => ranges,
297        Ok(_) => {
298            error!("gettimeofday tv repair: the guest's memory map is empty");
299            return Err(tv_repair_error(field, TvRepairFailureKind::MapsUnavailable));
300        }
301        Err(err) => {
302            error!("gettimeofday tv repair: reading the guest's memory map: {err}");
303            return Err(tv_repair_error(field, TvRepairFailureKind::MapsUnavailable));
304        }
305    };
306    if time_word_is_fully_mapped(ranges, word) {
307        Err(tv_repair_error(
308            field,
309            TvRepairFailureKind::StoppedWordUnreadable,
310        ))
311    } else {
312        Ok(())
313    }
314}

316/// What Detcore's memory reader returned for `tv_sec` and `tv_usec` before a
317/// `gettimeofday`, with the read error where it failed. Under ptrace the read
318/// ignores page protection and protection keys; backends that read with
319/// `process_vm_readv` respect `VM_READ`.
320type TimevalWordSnapshot = [Result<libc::time_t, Errno>; 2];
321
322const TIMEVAL_WORDS: [(&str, usize); 2] = [
323    ("tv_sec", std::mem::offset_of!(Timeval, tv_sec)),
324    ("tv_usec", std::mem::offset_of!(Timeval, tv_usec)),
325];
326
327fn snapshot_timeval_words<'a, G, T>(
328    guest: &mut G,
329    tv_addr: AddrMut<'a, Timeval>,
330) -> TimevalWordSnapshot
331where
332    G: Guest<Detcore<T>>,
333    T: RecordOrReplay,
334{
335    TIMEVAL_WORDS.map(|(field, offset)| {
336        let addr = timeval_word_addr(field, tv_addr, offset).map_err(|_| Errno::EFAULT)?;
337        guest.memory().read_value(addr)
338    })
339}

341fn timeval_word_addr<'a>(
342    field: &'static str,
343    tv_addr: AddrMut<'a, Timeval>,
344    offset: usize,
345) -> Result<AddrMut<'a, libc::time_t>, Error> {
346    tv_addr
347        .as_raw()
348        .checked_add(offset)
349        .and_then(AddrMut::<libc::time_t>::from_raw)

393/// Replaces the host wall-clock time that a `gettimeofday` failing with EFAULT
394/// may have stored in `tv` with virtual time, in exactly the words Linux
395/// stored.
396///
397/// Linux stores `tv_sec`, then `tv_usec`, each with one eight-byte `put_user`,
398/// and only then copies `tz`; the first fault ends the call with EFAULT. A
399/// store that faults on either page it touches commits nothing, so each word
400/// is stored whole or not at all, and nothing after an unstored word is
401/// attempted. `time(2)` uses the same eight-byte `put_user` store for its
402/// `tloc`, so the repair injects it at each word in kernel order. EFAULT means
403/// the original call stopped at that word too. Success means the probe itself
404/// just stored host seconds in those exact bytes; only then does Hermit replace
405/// the word with its virtual value. Other probe errors and overwrite failures
406/// fail closed because host time may remain.
407///
408/// An EFAULT from the probe is only evidence about `tv` if it came from the
409/// store. A seccomp filter can return EFAULT for `time(2)` without running it,
410/// and would otherwise end the repair while host time remains in a writable
411/// `tv`. Two checks therefore confirm every EFAULT before it is trusted.
412/// First, `time(NULL)` is injected: it stores nothing and cannot fail on
413/// Linux, so any error means the probes are not reaching the native call.
414/// Second, every word from the stopped one onward must still read exactly as
415/// it did before the original call: readable with the same value, or
416/// unreadable both times and not fully mapped, so that the store could only
417/// have faulted. A changed word was stored by that call, and an unreadable
418/// word that is fully mapped may have been, because a failed read is not
419/// itself a fault. Each of these failures ends the run, including for the
420/// mapped but unreadable pages listed at `require_unmapped_unreadable_word`.
421///
422/// These checks are observations Detcore makes with its own syscalls, so they
423/// trust those syscalls' results, as every Detcore handler that reads back a
424/// result does. They catch a filter that makes a syscall fail, whether it is
425/// the guest's probe or Detcore's own read. A seccomp filter installed on
426/// Hermit itself can also make one of Detcore's own syscalls report success
427/// without running it, and that is outside this guarantee. A guest cannot
428/// install a filter: Detcore refuses `seccomp(2)` and `PR_SET_SECCOMP`.
429///
430/// The repair never writes a word the kernel could not store, and it does not
431/// test writability by rewriting a word, because a remote write ignores
432/// protection keys that deny the guest's own stores. The pre-call snapshot is
433/// a read only. The host seconds written by a successful
434/// probe exist transiently until the overwrite. Default thread
435/// sequentialization prevents another guest thread from observing that
436/// interval, but another process sharing the page can observe it; such
437/// external shared state is outside Hermit's determinism guarantee.
438/// Standard record/replay disables time virtualization, so this handler is not
439/// used there. A custom configuration that combines replay data with virtual
440/// time fails closed before probing: a recorded EFAULT does not establish that
441/// this execution performed any stores for a live `time(2)` to reproduce.
442///
443/// Every backend runs this repair. The memory map comes from
444/// `Guest::storable_memory_ranges`, which lists `/proc/<pid>/maps` of the
445/// guest by default and the guest's own page record on a backend whose
446/// `pid()` is not the guest (KVM). A backend that executes `gettimeofday` and
447/// `time(2)` itself, as KVM does, stores them as Linux does: each word with an
448/// all-or-nothing store to a writable page, in kernel order.
449async fn overwrite_failed_gettimeofday_tv<'a, G, T>(
450    guest: &mut G,
451    tv_addr: AddrMut<'a, Timeval>,
452    tv: &Timeval,
453    before: &TimevalWordSnapshot,
454) -> Result<(), Error>
455where
456    G: Guest<Detcore<T>>,
457    T: RecordOrReplay,
458{
459    let values = [tv.tv_sec as libc::time_t, tv.tv_usec as libc::time_t];
460    for (index, (field, offset)) in TIMEVAL_WORDS.into_iter().enumerate() {
461        let addr = timeval_word_addr(field, tv_addr, offset)?;
462        let probe = syscalls::Time::new().with_tloc(Some(addr));
463        match classify_time_store_probe(field, guest.inject(probe).await)? {
464            TimeStoreProbe::Stored => {
465                let bytes = values[index].to_ne_bytes();
466                let overwrite = guest
467                    .memory()
468                    .write_with_user_access(addr.cast::<u8>(), &bytes);
469                require_complete_time_word_overwrite(field, bytes.len(), overwrite)?;
470            }
471            TimeStoreProbe::Stopped => {
472                let control = syscalls::Time::new().with_tloc(None);
473                require_native_time_control_probe(field, guest.inject(control).await)?;
474                for (stopped, (field, offset)) in TIMEVAL_WORDS.into_iter().enumerate().skip(index)
475                {
476                    let Some(addr) = stopped_timeval_word_addr(tv_addr, offset) else {
477                        continue;
478                    };
479                    let after = guest.memory().read_value(addr);
480                    match require_unchanged_stopped_word(field, before[stopped], after)? {
481                        StoppedWord::Unchanged => {}
482                        StoppedWord::Unreadable => {
483                            let maps = guest.storable_memory_ranges();
484                            require_unmapped_unreadable_word(field, maps, addr.as_raw() as u64)?;
485                        }
486                    }
487                }
488                break;
489            }
490        }
491    }
492    Ok(())
493}

557    /// gettimeofday
558    pub async fn handle_gettimeofday<G: Guest<Self>>(
559        &self,
560        guest: &mut G,
561        call: syscalls::Gettimeofday,
562    ) -> Result<i64, Error> {
563        let time_ns = guest_clock_time(guest).await;
564
565        // What Detcore could read in `tv` before the call; the repair uses it
566        // to confirm which words a failing call left alone.
567        let before = call.tv().map(|tp| snapshot_timeval_words(guest, tp.into()));
568
569        // A call failing with EFAULT may still have stored host wall-clock time
570        // in `tv`, so keep its result until `tv` holds virtual time.
571        let result = self
572            .record_or_replay_preserving_tool_errors(guest, call)
573            .await;
574
575        let tv: Timeval = time_ns.into();
576
577        if let Some(tp) = call.tv() {
578            match (&result, &before) {
579                (Ok(_), _) => guest.memory().write_value(tp, &tv)?,
580                // Linux's gettimeofday fails only with EFAULT, which is taken
581                // to be its own even when a seccomp filter returned it without
582                // running the call. Any other error came from the backend, the
583                // tool, a seccomp filter or a replayed log, and says nothing
584                // about what reached `tv`, so memory is left alone.
585                (Err(Error::Errno(Errno::EFAULT)), Some(before)) => {
586                    require_live_time_store_probe(self.cfg.replay_data.is_some())?;
587                    overwrite_failed_gettimeofday_tv(guest, tp.into(), &tv, before).await?
588                }
589                (Err(_), _) => {}
590            }
591        }
592
593        result
594    }

701    /// Helper function to wait a given period, which may either succeed or be interrupted by a signal.
702    /// Return 0 or EINTR respectively.
703    async fn wait_and_return<R: Guest<Self>>(
704        &self,
705        guest: &mut R,
706        mut request: Resources,
707        call: NanosleepFamily,
708    ) -> Result<i64, Error> {
709        let target_time = time_from_resources(&request).expect("a sleepuntil resource request");
710        // Lets the scheduler end the sleep for a cross-task signal that
711        // interrupts it (`scheduler::sleep_signal`).
712        request.fyi(crate::resources::NANOSLEEP_FYI);
713        loop {
714            // A signal already pending ends the sleep before it parks, and one
715            // still pending after a wake ends it; a wake whose signal was taken
716            // back parks it again with its original absolute deadline
717            // (`super::signal::sleep_signal`).
718            if super::signal::sleep_signal(guest, SleepCheck::BeforePark, false).await?
719                == SleepSignal::Pending
720            {
721                break;
722            }
723            request.in_guest_sleep_sigalrm = self.in_guest_sleep_sigalrm_snapshot(guest).await;
724            match crate::tool_global::parked_wait_request(
725                guest,
726                request.clone(),
727                crate::scheduler::parked::ParkedWaitPolicy::NanosleepNoHandlerRestart {
728                    absolute_deadline: target_time,
729                },
730            )
731            .await
732            {
733                ResumeStatus::InGuestFatalAlarm => {
734                    // The default action terminates the guest. In particular,
735                    // do not observe time or write `rem` after this fatal wake.
736                    self.stage_in_guest_fatal_alarm(guest).await;
737                    return Err(Errno::EINTR.into());
738                }
739                ResumeStatus::Normal => return Ok(0),
740                ResumeStatus::Signaled(signals) => {
741                    match super::signal::sleep_signal(guest, SleepCheck::AfterWake(signals), false)
742                        .await?
743                    {
744                        SleepSignal::Pending | SleepSignal::Held => break,
745                        SleepSignal::None => trace!(
746                            "nanosleep woken for a signal no longer pending for it; sleeping again until {}",
747                            target_time
748                        ),
749                    }
750                }
751            }
752        }
753        let now = thread_observe_time(guest).await;
754        let delta = remaining_sleep_duration(target_time, now);
755        // Linux never touches remain for TIMER_ABSTIME, even when
756        // a caught signal interrupts the absolute sleep.
757        let addr2 = if call.flags() & libc::TIMER_ABSTIME == 0 {
758            call.rem()
759        } else {
760            None
761        };
762        if let Some(addr2) = addr2 {
763            info!(
764                "[interrupted] sleep till (until {}), woke up {:?} early, writing into nanosleep rem argument.",
765                target_time, delta
766            );
767            let t = Timespec {
768                tv_sec: delta.as_secs() as i64,
769                tv_nsec: delta.subsec_nanos() as i64,
770            };
771            guest.memory().write_value(addr2, &t)?;
772        } else {
773            info!("[interrupted] nanosleep rem argument is null, not writing it.")
774        }
775        Err(reverie::Error::Errno(Errno::EINTR))
776    }

778    /// clock_nanosleep and nanosleep
779    pub async fn handle_nanosleep_family<R: Guest<Self>>(
780        &self,
781        guest: &mut R,
782        call: NanosleepFamily,
783    ) -> Result<i64, Error> {
784        if call.flags() > libc::TIMER_ABSTIME {
785            trace!("Unhandled clock_nanosleep flags, letting syscall through...");
786            return Ok(guest.inject(Syscall::from(call)).await?);
787        }
788
789        let addr = call.req().ok_or(Errno::EFAULT)?;
790        let t: Timespec = guest.memory().read_value(addr)?;
791
792        // Linux validates the requested interval BEFORE sleeping: nanosleep(2)
793        // and clock_nanosleep(2) both fail EINVAL when tv_nsec is outside
794        // [0, 999999999] or tv_sec is negative.
795        //
796        // Detcore skipped that check and fed the raw fields through `as u64`.
797        // `Timespec` stores both as i64, so `tv_sec = -1` wrapped to
798        // u64::MAX (~1.8e19 seconds) and became `SleepUntil(INDEFINITE)`: the
799        // only guest thread parked with no deadline, the run queue emptied, and
800        // `step2d_handle_empty_queue` deliberately never jumps the clock for an
801        // indefinite waiter (doing so would also wake a `pause(2)`). The
802        // container then died -- exit 1, "Sandbox container exited
803        // unexpectedly" -- where Linux returns an errno and keeps running.
804        //
805        // A past *absolute* deadline is NOT an error and must still return 0,
806        // so this rejects only malformed fields, never an early deadline.
807        if t.tv_sec < 0 || t.tv_nsec < 0 || t.tv_nsec > 999_999_999 {
808            return Err(Errno::EINVAL.into());
809        }
810
811        match call.flags() {
812            0 => {
813                if self.cfg.sequentialize_threads {
814                    let time = Duration::from_secs(t.tv_sec as u64)
815                        + Duration::from_nanos(t.tv_nsec as u64);
816                    let request = Self::sleep_request(guest, time).await;
817                    trace!(
818                        "nanosleep adding delta {:?} to yield request {:?}",
819                        time, &request
820                    );
821                    self.wait_and_return(guest, request, call).await
822                } else {
823                    trace!("Not sequentializing threads, letting nanosleep through...");
824                    Ok(guest.inject(Syscall::from(call)).await?)
825                }
826            }
827            libc::TIMER_ABSTIME => {
828                let target_time = LogicalTime::from_secs(t.tv_sec as u64)
829                    + LogicalTime::from_nanos(t.tv_nsec as u64);
830                if self.cfg.sequentialize_threads {
831                    if self.cfg.virtualize_time {
832                        let request = Self::sleep_request_abs(guest, target_time).await;
833                        trace!(
834                            "nanosleep setting absolute time {:?} to yield request {:?}",
835                            target_time, &request
836                        );
837                        self.wait_and_return(guest, request, call).await
838                    } else {
839                        // TODO T124594597: Record-replay case here, need better ideas to enable proper handling of this case.
840                        error!(
841                            "Sequentializing but not virtualizing, so can't rely on passed abs time, especially when replaying a recording, just yelding"
842                        );
843                        let request = Self::yield_request(guest);
844                        self.wait_and_return(guest, request, call).await
845                    }
846                } else if self.cfg.virtualize_time {
847                    trace!(
848                        "Not sequentializing, but virtualizing so calculating relative time and invoking nanosleep..."
849                    );
850                    let relative_ts = Self::relative_time_from_abs_target(guest, target_time).await;
851                    let mut stack = guest.stack().await;
852                    let req = stack.push(relative_ts);
853                    stack.commit()?;
854                    let modified_call = syscalls::Nanosleep::new().with_req(Some(req));
855                    Ok(guest.inject(modified_call).await?)
856                } else {
857                    trace!(
858                        "Not sequentializing threads not virtualizing, letting nanosleep through..."
859                    );
860                    Ok(guest.inject(Syscall::from(call)).await?)
861                }
862            }
863            _ => unreachable!("Unexpected, unhandled flag value"),
864        }
865    }

867    async fn relative_time_from_abs_target<G: Guest<Self>>(
868        guest: &mut G,
869        target_time: LogicalTime,
870    ) -> Timespec {
871        let base_time = thread_observe_time(guest).await;
872
873        // An absolute deadline already in the past is NOT an error on Linux --
874        // clock_nanosleep(TIMER_ABSTIME) simply returns 0 without sleeping.
875        // `LogicalTime`'s `Sub` is a plain subtraction (unlike its `Add` impls,
876        // which saturate deliberately), so `target_time - base_time` underflows
877        // for a past deadline: a debug build panics with "attempt to subtract
878        // with overflow", and a release build wraps to an enormous interval --
879        // the same effectively-indefinite sleep this handler exists to avoid.
880        //
881        // Clamped here rather than by making the shared operator saturate,
882        // because this is the only subtraction of two `LogicalTime`s in the
883        // tree and a silently-saturating operator could hide a real underflow
884        // in some future caller.
885        let relative_logical = if target_time <= base_time {
886            LogicalTime::from_nanos(0)
887        } else {
888            target_time - base_time
889        };
890
891        Timespec {
892            tv_sec: relative_logical.as_secs() as i64,
893            tv_nsec: relative_logical.subsec_nanos() as i64,
894        }
895    }

897    /// Whether a POSIX timer notifying with `signo` registers its expirations
898    /// with the scheduler. A standard signal always does. A real-time signal
899    /// (32 to 64; glibc's SIGEV_THREAD timers use one through SIGEV_THREAD_ID)
900    /// cannot be delivered by Detcore, whose signal model carries nix's `Signal`
901    /// (https://github.com/rrnewton/hermit/issues/3893). Creating, arming, reading
902    /// and deleting such a timer still behave as on Linux; only an expiration is
903    /// unserviceable. A fail-closed run registers it anyway, and the scheduler
904    /// refuses the run by name if it ever comes due; a permissive run
905    /// (`--allow-unsupported-syscalls`) leaves it unregistered, so it never fires.
906    fn posix_timer_expiry_is_registered(&self, signo: i32) -> bool {
907        Signal::try_from(signo).is_ok() || self.cfg.panic_on_unsupported_syscalls
908    }

912    /// timer_create: allocate a per-process POSIX timer and hand back a
913    /// deterministic id, retaining any scheduler-deliverable signal.
914    pub async fn handle_timer_create<G: Guest<Self>>(
915        &self,
916        guest: &mut G,
917        call: syscalls::TimerCreate,
918    ) -> Result<i64, Error> {
919        // The kernel writes the new timer id here; a null pointer is EFAULT.
920        let timerid_ptr = call.timerid().ok_or(Errno::EFAULT)?;
921        let clockid = call.clockid();
922        let signo = if let Some(event_ptr) = call.sevp() {
923            let event: libc::sigevent = guest.memory().read_value(event_ptr)?;
924            match event.sigev_notify {
925                libc::SIGEV_NONE => None,
926                // Linux uses 4 for SIGEV_THREAD_ID. Treat it as process-directed
927                // until Detcore tracks per-timer thread targeting.
928                libc::SIGEV_SIGNAL | 4 => {
929                    if !(1..=64).contains(&event.sigev_signo) {
930                        return Err(Errno::EINVAL.into());
931                    }
932                    Some(event.sigev_signo)
933                }
934                _ => return Err(Errno::ENOSYS.into()),
935            }
936        } else {
937            Some(libc::SIGALRM)
938        };
939        // `None` for a real-time signal (32 to 64): see `handle_timer_settime`.
940        let signal = signo.and_then(|signo| Signal::try_from(signo).ok());
941        if signal == Some(Signal::SIGALRM) {
942            refuse_sigalrm(guest, SigalrmControl::ArmProducer).await?;
943        }
944        let id = {
945            let mut timers = guest.thread_state().posix_timers.lock().unwrap();
946            timers.create(signo)
947        };
948        guest
949            .memory()
950            .write_value(timerid_ptr, &(id as libc::c_int))?;
951        detlog!(
952            "[dtid {}] timer_create(clockid={:?}) => deterministic timer id {}, signal {:?}",
953            guest.thread_state().dettid,
954            clockid,
955            id,
956            signal,
957        );
958        Ok(0)
959    }

963    /// timer_settime: arm or disarm a timer against the deterministic virtual
964    /// clock. The old arming is reported through `old_value` when requested.
965    pub async fn handle_timer_settime<G: Guest<Self>>(
966        &self,
967        guest: &mut G,
968        call: syscalls::TimerSettime,
969    ) -> Result<i64, Error> {
970        let id = call.timerid();
971        let new_ptr = call.new_value().ok_or(Errno::EINVAL)?;
972        let new: libc::itimerspec = guest.memory().read_value(new_ptr)?;
973        let interval_ns = timespec_to_ns(new.it_interval);
974        let value_ns = timespec_to_ns(new.it_value);
975
976        let now = thread_observe_time(guest).await;
977        let deadline = if value_ns == 0 {
978            None
979        } else if call.flags() & libc::TIMER_ABSTIME != 0 {
980            // Absolute expiration is interpreted against the same virtual clock.
981            Some(LogicalTime::from_nanos(value_ns))
982        } else {
983            Some(now + Duration::from_nanos(value_ns))
984        };
985
986        let (old, signal_number) = {
987            let mut timers = guest.thread_state().posix_timers.lock().unwrap();
988            let old = timers.settime(id, interval_ns, deadline, now);
989            let signal = timers.signal(id);
990            (old, signal)
991        };
992        let (old_remaining_ns, old_interval_ns) = old.ok_or(Errno::EINVAL)?;
993        let signal_number = signal_number.ok_or(Errno::EINVAL)?;
994
995        if let Some(old_ptr) = call.old_value() {
996            let old_spec = libc::itimerspec {
997                it_interval: ns_to_timespec(old_interval_ns),
998                it_value: ns_to_timespec(old_remaining_ns),
999            };
1000            guest.memory().write_value(old_ptr, &old_spec)?;
1001        }
1002
1003        if let Some(signo) = signal_number {
1004            if self.posix_timer_expiry_is_registered(signo) {
1005                register_posix_timer(
1006                    guest,
1007                    id,
1008                    deadline,
1009                    LogicalTime::from_nanos(interval_ns),
1010                    signo,
1011                )
1012                .await;
1013            } else if deadline.is_some() {
1014                tracing::error!(
1015                    "[detcore, dtid {}] POSIX timer {} notifies with real-time signal {}, which Detcore cannot deliver; it is armed but will never fire",
1016                    guest.thread_state().dettid,
1017                    id,
1018                    signo,
1019                );
1020            }
1021        }
1022
1023        detlog!(
1024            "[dtid {}] timer_settime(id={}, interval_ns={}, value_ns={}) armed against virtual clock",
1025            guest.thread_state().dettid,
1026            id,
1027            interval_ns,
1028            value_ns,
1029        );
1030        Ok(0)
1031    }

1319        #[test]
1320        fn timeval_word_addresses_follow_kernel_order_and_overflow_fails_closed_only_for_probes() {
1321            assert_eq!(TIMEVAL_WORDS, [("tv_sec", 0), ("tv_usec", 8)]);
1322            let tv_addr = AddrMut::<Timeval>::from_raw(0x10_0000).unwrap();
1323            assert_eq!(
1324                timeval_word_addr("tv_sec", tv_addr, std::mem::offset_of!(Timeval, tv_sec))
1325                    .unwrap()
1326                    .as_raw(),
1327                tv_addr.as_raw()
1328            );
1329            assert_eq!(
1330                timeval_word_addr("tv_usec", tv_addr, std::mem::offset_of!(Timeval, tv_usec))
1331                    .unwrap()
1332                    .as_raw(),
1333                tv_addr.as_raw() + 8
1334            );
1335
1336            let overflowing = AddrMut::<Timeval>::from_raw(usize::MAX - 3).unwrap();
1337            let error = timeval_word_addr("tv_usec", overflowing, 8)
1338                .expect_err("overflowing field address must fail the repair");
1339            assert_eq!(
1340                failure(error),
1341                TvRepairFailure {
1342                    field: "tv_usec",
1343                    kind: TvRepairFailureKind::AddressOverflow,
1344                }
1345            );
1346
1347            // Checking a stopped word needs no probe, so a word past the end of
1348            // the address space is simply one no store reached.
1349            for (_, offset) in TIMEVAL_WORDS {
1350                assert_eq!(
1351                    stopped_timeval_word_addr(tv_addr, offset).map(|addr| addr.as_raw()),
1352                    Some(tv_addr.as_raw() + offset)
1353                );
1354            }
1355            // `gettimeofday((void *)-1, NULL)`: `tv_sec` is the last byte of
1356            // the address space and `tv_usec` has no address at all.
1357            let last = AddrMut::<Timeval>::from_raw(usize::MAX).unwrap();
1358            assert_eq!(
1359                stopped_timeval_word_addr(last, 0).map(|addr| addr.as_raw()),
1360                Some(usize::MAX)
1361            );
1362            assert!(stopped_timeval_word_addr(last, 8).is_none());
1363            assert!(stopped_timeval_word_addr(overflowing, 8).is_none());
1364        }
