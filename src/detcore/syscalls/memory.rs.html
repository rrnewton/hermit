22// Added in Linux 6.13 and not yet exposed by the pinned libc crate.
23const MADV_GUARD_INSTALL: i32 = 102;
24const MADV_GUARD_REMOVE: i32 = 103;
25
26#[derive(Debug, Clone, Copy, Eq, PartialEq)]
27enum MadviseAction {
28    ForwardHint,
29    ForwardSemantic,
30    Ignore,
31    Reject(Errno),
32    Unknown,
33}
34
35const fn madvise_action(advice: i32) -> MadviseAction {
36    match advice {
37        // Pure access-pattern and prefetch hints have no required memory-content
38        // side effect. Backends without native madvise support accept them as no-ops.
39        libc::MADV_NORMAL | libc::MADV_RANDOM | libc::MADV_SEQUENTIAL | libc::MADV_WILLNEED => {
40            MadviseAction::ForwardHint
41        }
42
43        // Operations with guest-visible memory, fork, backing-store, dump, or guard
44        // semantics must reach a backend that implements native madvise behavior.
45        libc::MADV_DONTNEED
46        | libc::MADV_DONTFORK
47        | libc::MADV_DOFORK
48        | libc::MADV_DONTDUMP
49        | libc::MADV_DODUMP
50        | libc::MADV_WIPEONFORK
51        | libc::MADV_KEEPONFORK
52        | libc::MADV_DONTNEED_LOCKED
53        | MADV_GUARD_INSTALL
54        | MADV_GUARD_REMOVE => MadviseAction::ForwardSemantic,
55
56        // These are optional reclaim or asynchronous VM-policy controls. Their host
57        // effects depend on memory pressure, KSM, and THP activity. Hermit accepts
58        // them as fixed no-ops after deterministic argument validation; it deliberately
59        // does not reproduce each advice's host- and mapping-specific EINVAL cases.
60        libc::MADV_FREE
61        | libc::MADV_MERGEABLE
62        | libc::MADV_UNMERGEABLE
63        | libc::MADV_HUGEPAGE
64        | libc::MADV_NOHUGEPAGE
65        | libc::MADV_COLD
66        | libc::MADV_PAGEOUT => MadviseAction::Ignore,
67
68        // Hole punching mutates backing storage and every mapping alias. Refuse it
69        // until Detcore can update file resources and replay all affected aliases.
70        libc::MADV_REMOVE => MadviseAction::Reject(Errno::EINVAL),
71
72        // Successful population and collapse promise synchronous, resource-
73        // dependent work. Report the same deterministic error Linux uses when
74        // an advice value is unsupported so callers can take their fallback.
75        libc::MADV_POPULATE_READ
76        | libc::MADV_POPULATE_WRITE
77        | reverie::syscalls::linux_abi::MADV_COLLAPSE => MadviseAction::Reject(Errno::EINVAL),
78
79        // Never let a guest inject host memory failures, even if the container
80        // unexpectedly has enough privilege to make these operations succeed.
81        libc::MADV_HWPOISON | libc::MADV_SOFT_OFFLINE => MadviseAction::Reject(Errno::EPERM),
82
83        _ => MadviseAction::Unknown,
84    }
85}

87/// Whether `advice`, once the backend runs it, may set or clear a flag Linux
88/// keeps on the mappings it covers (`madvise_vma_behavior`, mm/madvise.c in
89/// v6.17), which `smaps` prints (`VmFlags`) and the backend's memory map does
90/// not list (`MemoryMetadata::mapping_flags_changed`): the access-pattern
91/// advice (VM_SEQ_READ, VM_RAND_READ) and the fork and dump advice
92/// (VM_DONTCOPY, VM_DONTDUMP, VM_WIPEONFORK). The huge-page and KSM advice
93/// change flags too, but Detcore answers it itself and never passes it on
94/// ([`MadviseAction::Ignore`]); the rest acts on pages, not on flags.
95const fn advice_changes_mapping_flags(advice: i32) -> bool {
96    matches!(
97        advice,
98        libc::MADV_NORMAL
99            | libc::MADV_RANDOM
100            | libc::MADV_SEQUENTIAL
101            | libc::MADV_DONTFORK
102            | libc::MADV_DOFORK
103            | libc::MADV_DONTDUMP
104            | libc::MADV_DODUMP
105            | libc::MADV_WIPEONFORK
106            | libc::MADV_KEEPONFORK
107    )
108}
109
110/// Whether `advice`, once the backend runs it, may leave a mapping it covers
111/// without the pages it held (`madvise_dontneed_free` and
112/// `madvise_guard_install`, mm/madvise.c in v6.17): `MADV_DONTNEED` and
113/// `MADV_DONTNEED_LOCKED` zap them, `MADV_GUARD_INSTALL` zaps them and
114/// installs guard markers, and `MADV_WIPEONFORK` leaves a later fork's copy
115/// empty. Whether a mapping still holds a resident page is then not a
116/// fact the memory map carries (`MemoryMetadata::pages_discarded`). `MADV_FREE`,
117/// `MADV_COLD` and `MADV_PAGEOUT` would too, but Detcore answers them itself
118/// and never passes them on ([`MadviseAction::Ignore`]).
119const fn advice_discards_pages(advice: i32) -> bool {
120    matches!(
121        advice,
122        libc::MADV_DONTNEED
123            | libc::MADV_DONTNEED_LOCKED
124            | MADV_GUARD_INSTALL
125            | libc::MADV_WIPEONFORK
126    )
127}

129/// Whether a forwarded madvise call that returned `result` may have changed
130/// mapping flags or pages: every outcome but ENOSYS, which says the backend
131/// ran nothing (a backend that does not implement the advice). Linux applies
132/// advice to the mapped part of a range it then reports ENOMEM for, and may
133/// fail with EINVAL at a later mapping after changing earlier ones.
134fn madvise_may_have_run(result: &Result<i64, Errno>) -> bool {
135    !matches!(result, Err(Errno::ENOSYS))
136}

138fn validate_common_args(call: syscalls::Madvise) -> Result<(), Error> {
139    let page = crate::page::page_size() as usize;
140    let start = call.addr().map(AddrMut::as_raw).unwrap_or(0);
141    if !start.is_multiple_of(page) {
142        return Err(Errno::EINVAL.into());
143    }
144    if call.len() == 0 {
145        return Ok(());
146    }
147
148    let end = start.checked_add(call.len()).ok_or(Errno::EINVAL)?;
149    end.checked_add(page - 1).ok_or(Errno::EINVAL)?;
150    Ok(())
151}

153impl<T: RecordOrReplay> Detcore<T> {
154    /// Apply a deterministic policy to madvise(2).
155    ///
156    /// Hints and advice with guest-visible semantics are forwarded to the
157    /// backend, on every backend. A backend that does not implement an advice
158    /// answers ENOSYS with the typed refusal `MemoryAdvice`, which is logged as a
159    /// warning naming the advice; the guest sees the ENOSYS.
160    /// Record/replay accepts pure hints as no-ops and passes guest-semantic advice
161    /// to the recorder and replayer, which record and restore the effects that
162    /// differ because replay replaces file mappings with anonymous mappings.
163    /// Reclaim and asynchronous VM-policy advice receives fixed success without exposing host memory pressure. Resource-
164    /// dependent, backing-store, and hardware-failure operations receive fixed errors.
165    // AUTONOMOUS-BOT-IMPLEMENTED
166    // TODO-HUMAN-REVIEW(#548): Recheck advice policy and record/replay boundaries.
167    pub async fn handle_madvise<G: Guest<Self>>(
168        &self,
169        guest: &mut G,
170        call: syscalls::Madvise,
171    ) -> Result<i64, Error> {
172        let advice = call.advice();
173        let action = madvise_action(advice);
174        validate_common_args(call)?;
175
176        if call.len() == 0 {
177            return match action {
178                MadviseAction::Unknown => Err(Errno::EINVAL.into()),
179                _ => Ok(0),
180            };
181        }
182        if self.cfg.recordreplay_modes && action == MadviseAction::ForwardHint {
183            crate::detlog!(
184                "[dtid {}] madvise hint {} accepted as record/replay no-op",
185                guest.thread_state().dettid,
186                advice,
187            );
188            return Ok(0);
189        }
190
191        match action {
192            MadviseAction::ForwardHint | MadviseAction::ForwardSemantic => {
193                let advised = (
194                    call.addr().map_or(0, |addr| addr.as_raw() as u64),
195                    call.len() as u64,
196                );
197                let result = self.record_or_replay(guest, call).await;
198                // The backend's memory map lists neither the flags this advice
199                // sets nor the pages it discards, so a `smaps`
200                // Detcore renders from that map can no longer derive them for
201                // a listed mapping the advice reached
202                // (`MemoryMetadata::mapping_flags_changed`, `pages_discarded`).
203                let changes_flags = advice_changes_mapping_flags(advice);
204                let discards_pages = advice_discards_pages(advice);
205                if (changes_flags || discards_pages)
206                    && madvise_may_have_run(&result)
207                    && self.renders_address_space_files()
208                    && Self::range_reaches_listed_mapping(guest, advised.0, advised.1)
209                {
210                    if changes_flags {
211                        guest.thread_state().note_mapping_flags_changed();
212                    }
213                    if discards_pages {
214                        guest.thread_state().note_pages_discarded();
215                    }
216                }
217                if result.is_err()
218                    && let Some(refusal) = guest.take_unsupported_refusal().filter(|refusal| {
219                        refusal.operation == reverie::UnsupportedOperation::MemoryAdvice
220                    })
221                {
222                    warn!(
223                        "[dtid {}] madvise advice {} refused by the backend: {}",
224                        guest.thread_state().dettid,
225                        advice,
226                        refusal.diagnostic,
227                    );
228                }
229                Ok(result?)
230            }
231            MadviseAction::Ignore => {
232                crate::detlog!(
233                    "[dtid {}] madvise advice {} accepted as deterministic no-op",
234                    guest.thread_state().dettid,
235                    advice,
236                );
237                Ok(0)
238            }
239            MadviseAction::Reject(errno) => {
240                crate::detlog!(
241                    "[dtid {}] madvise advice {} rejected with {}",
242                    guest.thread_state().dettid,
243                    advice,
244                    errno,
245                );
246                Err(errno.into())
247            }
248            MadviseAction::Unknown => {
249                crate::detlog!(
250                    "[dtid {}] unknown madvise advice {} rejected with EINVAL",
251                    guest.thread_state().dettid,
252                    advice,
253                );
254                Err(Errno::EINVAL.into())
255            }
256        }
257    }
258
259    /// Deterministic `mincore(2)`.
260    ///
261    /// Real page residency reflects host memory pressure and is therefore
262    /// nondeterministic, which is why mincore was previously classified as an
263    /// unsupported syscall. GNU grep (and other glibc consumers) invoke mincore
264    /// under the KVM backend on a code path the ptrace backend does not take, so
265    /// leaving it unsupported aborts the guest under `--strict`.
266    ///
267    /// Inject the call first so the backend preserves Linux pointer and mapping
268    /// validation, then report every mapped page as resident. The residency
269    /// vector is only an advisory hint, so replacing those nondeterministic bits
270    /// with a constant answer remains bitwise-identical across runs.
271    // AUTONOMOUS-BOT-IMPLEMENTED
272    // TODO-HUMAN-REVIEW(#775): Review deterministic mincore residency emulation.
273    pub async fn handle_mincore<G: Guest<Self>>(
274        &self,
275        guest: &mut G,
276        call: syscalls::Mincore,
277    ) -> Result<i64, Error> {
278        // Linux rejects a start address that is not page-aligned with EINVAL.
279        let page = crate::page::page_size() as usize;
280        let start = call.addr().map(AddrMut::as_raw).unwrap_or(0);
281        if !start.is_multiple_of(page) {
282            return Err(Errno::EINVAL.into());
283        }
284        let len = call.len();
285        if len == 0 {
286            return Ok(0);
287        }
288        // Match madvise's overflow guard on the requested range.
289        start.checked_add(len).ok_or(Errno::EINVAL)?;
290        let page_count = len.div_ceil(page);
291        let vec = call.vec().ok_or(Errno::EFAULT)?;
292        guest.inject(syscalls::Syscall::from(call)).await?;
293        let residency = vec![1u8; page_count];
294        guest.memory().write_exact(vec, &residency)?;
295        Ok(0)
296    }
297}

299/// `address` rounded up to a `page`-byte boundary, as `brk(2)` rounds the
300/// break (`PAGE_ALIGN`).
301fn brk_page_align(address: u64, page: u64) -> Option<u64> {
302    Some(address.checked_add(page - 1)? & !(page - 1))
303}

305/// Linux's default `stack_guard_gap` with `page`-byte pages: 256 pages kept
306/// free below a mapping that grows down.
307fn stack_guard_gap(page: u64) -> u64 {
308    256 * page
309}

311/// One mapping of a `/proc/<pid>/smaps` text: its range, and the gap Linux
312/// keeps free below it (`stack_guard_start_gap`): `stack_guard_gap` below a
313/// mapping that grows down (`gd` among its `VmFlags`), a page below a
314/// shadow stack (`ss`), none below any other.
315#[derive(Debug, Clone, Copy, PartialEq, Eq)]
316struct Vma {
317    start: u64,
318    end: u64,
319    start_gap: u64,
320}
321
322/// The mappings of a `/proc/<pid>/smaps` text, in address order, parsed as
323/// they are asked for: nothing is allocated, so where the tool shares the
324/// guest's process its allocator maps nothing between the read and the
325/// injected call that relies on it.
326fn smaps_vmas(smaps: &[u8], page: u64) -> SmapsVmas<'_> {
327    SmapsVmas { rest: smaps, page }
328}
329
330#[derive(Clone)]
331struct SmapsVmas<'a> {
332    rest: &'a [u8],
333    /// The host's page size, which sizes the guard gaps.
334    page: u64,
335}
336
337impl<'a> SmapsVmas<'a> {
338    /// The next line, without consuming it.
339    fn peek(&self) -> Option<&'a [u8]> {
340        (!self.rest.is_empty()).then(|| {
341            self.rest
342                .split(|&byte| byte == b'\n')
343                .next()
344                .unwrap_or_default()
345        })
346    }
347
348    fn advance(&mut self) {
349        self.rest = match self.rest.iter().position(|&byte| byte == b'\n') {
350            Some(newline) => &self.rest[newline + 1..],
351            None => &[],
352        };
353    }
354}

356/// A mapping's range, from the first field of its header line; `None` for
357/// any other line. Bytes, not text: the pathname that follows is whatever
358/// bytes the file was named with.
359fn smaps_range(line: &[u8]) -> Option<(u64, u64)> {
360    let range = line.split(|&byte| byte == b' ').next()?;
361    let (start, end) = std::str::from_utf8(range).ok()?.split_once('-')?;
362    Some((
363        u64::from_str_radix(start, 16).ok()?,
364        u64::from_str_radix(end, 16).ok()?,
365    ))
366}
367
368impl Iterator for SmapsVmas<'_> {
369    type Item = Vma;
370
371    fn next(&mut self) -> Option<Vma> {
372        let (start, end) = loop {
373            let line = self.peek()?;
374            self.advance();
375            if let Some(range) = smaps_range(line) {
376                break range;
377            }
378        };
379        let mut vma = Vma {
380            start,
381            end,
382            start_gap: 0,
383        };
384        // The mapping's fields, up to the next mapping's header.
385        while let Some(line) = self.peek() {
386            if smaps_range(line).is_some() {
387                break;
388            }
389            self.advance();
390            if let Some(flags) = line.strip_prefix(b"VmFlags:") {
391                let mut flags = flags.split(|byte| byte.is_ascii_whitespace());
392                vma.start_gap = if flags.clone().any(|flag| flag == b"gd") {
393                    stack_guard_gap(self.page)
394                } else if flags.any(|flag| flag == b"ss") {
395                    self.page
396                } else {
397                    0
398                };
399            }
400        }
401        Some(vma)
402    }
403}

405/// Whether Linux's `brk` grows a break whose page-aligned end is `old_end`
406/// to `new_end`: no mapping may meet `[old_end, new_end + PAGE)`, nor a
407/// mapping that grows down have its guard gap there
408/// (`vma_find(&vmi, newbrk + PAGE_SIZE + stack_guard_gap)` and
409/// `vm_start_gap`).
410fn brk_may_grow(
411    mut vmas: impl Iterator<Item = Vma>,
412    old_end: u64,
413    new_end: u64,
414    page: u64,
415) -> bool {
416    let Some(limit) = new_end.checked_add(page) else {
417        return false;
418    };
419    let search_end = limit.saturating_add(stack_guard_gap(page));
420    let Some(next) = vmas.find(|vma| vma.end > old_end && vma.start < search_end) else {
421        return true;
422    };
423    // `vm_start_gap`, which gives 0 where the gap would pass it.
424    let gap_start = next.start.saturating_sub(next.start_gap);
425    limit <= gap_start
426}
427
428/// Whether Linux's `brk` shrinks a break whose page-aligned end is `old_end`
429/// to `new_end`: only where a mapping meets `[new_end, old_end)`
430/// (`vma_find(&vmi, oldbrk)` from `newbrk`); otherwise it keeps the old
431/// break.
432fn brk_may_shrink(mut vmas: impl Iterator<Item = Vma>, new_end: u64, old_end: u64) -> bool {
433    vmas.any(|vma| vma.end > new_end && vma.start < old_end)
434}

436// Bounded DEBUG diagnostic: no new procfs read and no formatting between
437// the metadata capture and a mapping operation. Only the first rejected moving
438// request in this process is retained. Fork inherits the diagnostic latch.
439struct BrkDiagnostic<'a> {
440    dettid: i32,
441    model: &'a crate::memory::BrkModel,
442    requested: u64,
443}

445struct BrkDiagnosticHex<'a>(&'a [u8]);

447impl std::fmt::Display for BrkDiagnosticHex<'_> {
448    fn fmt(&self, formatter: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
449        for byte in self.0 {
450            write!(formatter, "{byte:02x}")?;
451        }
452        Ok(())
453    }

456impl BrkDiagnostic<'_> {
457    fn rejected(
458        &self,
459        branch: &'static str,
460        limit: Option<u64>,
461        smaps: Option<&[u8]>,
462        kernel_result: &dyn std::fmt::Debug,
463    ) {
464        use std::sync::atomic::AtomicBool;
465        use std::sync::atomic::Ordering;
466        static CAPTURED: AtomicBool = AtomicBool::new(false);
467        const MAX_BYTES: usize = 256 * 1024;
468        if self.requested == 0
469            || self.requested == self.model.current
470            || !tracing::enabled!(target: "detcore::brk_diagnostic", tracing::Level::DEBUG)
471            || CAPTURED.swap(true, Ordering::Relaxed)
472        {
473            return;
474        }
475        let complete = smaps.is_some_and(|bytes| bytes.len() <= MAX_BYTES);
476        tracing::debug!(
477            target: "detcore::brk_diagnostic",
478            dettid = self.dettid,
479            requested = self.requested,
480            start = self.model.start,
481            current = self.model.current,
482            data_span = self.model.data_span,
483            launcher_break = self.model.launcher_break,
484            ?limit,
485            branch,
486            ?kernel_result,
487            smaps_len = ?smaps.map(<[u8]>::len),
488            complete,
489            "BRK_DENIAL_V1 begin",
490        );
491        if complete {
492            for (index, chunk) in smaps.unwrap().chunks(1024).enumerate() {
493                tracing::debug!(
494                    target: "detcore::brk_diagnostic",
495                    dettid = self.dettid,
496                    index,
497                    bytes = %BrkDiagnosticHex(chunk),
498                    "BRK_DENIAL_V1 chunk",
499                );
500            }
501        }
502        tracing::debug!(
503            target: "detcore::brk_diagnostic",
504            dettid = self.dettid,
505            complete,
506            "BRK_DENIAL_V1 end",
507        );
508    }
509}

511/// The mapping-dependent part of a modeled program-break request. Endpoints
512/// are page-aligned; resource limits and the actual mapping operation stay
513/// with the caller, independently of where the smaps bytes were captured.
514#[derive(
515    Debug,
516    Clone,
517    Copy,
518    PartialEq,
519    Eq,
520    serde::Serialize,
521    serde::Deserialize
522)]
523pub enum BrkMappingCheck {
524    /// Check new pages and Linux's guard gap above the old break.
525    Grow { old_end: u64, new_end: u64 },
526    /// Check that some mapping overlaps the pages being removed.
527    Shrink { new_end: u64, old_end: u64 },
528}
529
530impl BrkMappingCheck {
531    pub(crate) fn allows(self, smaps: &[u8]) -> bool {
532        self.allows_with_page(smaps, crate::page::page_size())
533    }
534
535    /// [`Self::allows`] on a host whose pages are `page` bytes.
536    fn allows_with_page(self, smaps: &[u8], page: u64) -> bool {
537        match self {
538            Self::Grow { old_end, new_end } => {
539                brk_may_grow(smaps_vmas(smaps, page), old_end, new_end, page)
540            }
541            Self::Shrink { new_end, old_end } => {
542                brk_may_shrink(smaps_vmas(smaps, page), new_end, old_end)
543            }
544        }
545    }
546}

548impl<T: RecordOrReplay> Detcore<T> {
549    /// `brk(addr)` in a run that declares a launcher
550    /// (`Detcore::declares_launcher`), where the kernel's program break is the
551    /// launcher's: answered from Detcore's model of the guest's break
552    /// ([`crate::memory::BrkModel`]), as Linux's `brk` (mm/mmap.c) answers
553    /// from `mm->brk`. A request below the start, one past `RLIMIT_DATA`,
554    /// growth that would meet another mapping or its guard gap, or a shrink
555    /// over a range with no mapping leaves the break where it is; the call
556    /// returns the break in effect afterwards, never an error. Growth maps
557    /// exactly the new anonymous zero pages (`MAP_FIXED_NOREPLACE`, so an
558    /// existing mapping is never replaced) and shrinking unmaps them.
559    ///
560    /// Refused by name, where an answer would be a plausible wrong one: a
561    /// brk without a model for the image (with the reason there is none),
562    /// and a request at or above the launcher's break but below the model's
563    /// start. Only a libc that cached the launcher's break before Detcore's
564    /// first callback asks for that, and it would take Linux's answer, the
565    /// break in effect above its request, for a grant at its stale address.
566    pub(crate) async fn handle_brk<G: Guest<Self>>(
567        &self,
568        guest: &mut G,
569        call: syscalls::Syscall,
570    ) -> Result<i64, Error> {
571        let (model, unmodeled) = {
572            let metadata = guest
573                .thread_state()
574                .memory_metadata
575                .lock()
576                .expect("memory metadata mutex poisoned");
577            (metadata.brk_model, metadata.brk_unmodeled.clone())
578        };
579        let Some(model) = model else {
580            let reason = format!(
581                "unsupported: brk where the kernel's program break is the launcher's and Detcore has no model of the guest's: {}",
582                unmodeled
583                    .as_deref()
584                    .unwrap_or("the image was never modeled"),
585            );
586            return self
587                .refuse_launched_exe_because(guest, syscalls::Sysno::brk, reason)
588                .await;
589        };
590        let requested = syscalls::SyscallInfo::into_parts(call).1.arg0 as u64;
591        let granted = match self.granted_brk(guest, &model, requested).await {
592            Ok(granted) => granted,
593            Err(reason) => {
594                return self
595                    .refuse_launched_exe_because(guest, syscalls::Sysno::brk, reason.to_owned())
596                    .await;
597            }
598        };
599        guest
600            .thread_state()
601            .memory_metadata
602            .lock()
603            .expect("memory metadata mutex poisoned")
604            .brk_model = Some(crate::memory::BrkModel {
605            current: granted,
606            ..model
607        });
608        Ok(granted as i64)
609    }

611    /// The break `brk(requested)` leaves in effect, from `model`, mapping or
612    /// unmapping the pages between the old and the new break; an error where
613    /// Detcore could not keep the address space and the model in step (a
614    /// mapping it could neither place nor remove).
615    async fn granted_brk<G: Guest<Self>>(
616        &self,
617        guest: &mut G,
618        model: &crate::memory::BrkModel,
619        requested: u64,
620    ) -> Result<u64, &'static str> {
621        let current = model.current;
622        let diagnostic = BrkDiagnostic {
623            dettid: guest.thread_state().dettid.as_raw(),
624            model,
625            requested,
626        };
627        if requested < model.start {
628            if requested >= model.launcher_break {
629                diagnostic.rejected("below_start_launcher", None, None, &"not_injected");
630                return Err(
631                    "unsupported: brk from the launcher's break, which the guest's libc read before Detcore modeled the guest's",
632                );
633            }
634            diagnostic.rejected("below_start", None, None, &"not_injected");
635            return Ok(current);
636        }
637        let limit = guest
638            .thread_state()
639            .resource_limits
640            .lock()
641            .unwrap()
642            .get(reverie::syscalls::linux_abi::RLIMIT_DATA)
643            .map_or(libc::RLIM64_INFINITY, |limit| limit.current);
644        let used = (requested - model.start).saturating_add(model.data_span);
645        if limit != libc::RLIM64_INFINITY && used > limit {
646            diagnostic.rejected("rlimit_data", Some(limit), None, &"not_injected");
647            return Ok(current);
648        }
649        let page = crate::page::page_size();
650        let (Some(new_end), Some(old_end)) = (
651            brk_page_align(requested, page),
652            brk_page_align(current, page),
653        ) else {
654            diagnostic.rejected("page_overflow", Some(limit), None, &"not_injected");
655            return Ok(current);
656        };
657        if new_end == old_end {
658            return Ok(requested);
659        }
660        // The address space as Linux's brk inspects it, read with raw
661        // syscalls and no allocation between reads where the tool shares the
662        // guest's process; the guest's other threads are held by the
663        // sequentialized schedule.
664        let smaps = self
665            .guest_task_directory(guest)
666            .and_then(|directory| crate::util::raw_read_listing(&directory.join("smaps")).ok());
667        let check = if requested <= current {
668            BrkMappingCheck::Shrink { new_end, old_end }
669        } else {
670            BrkMappingCheck::Grow { old_end, new_end }
671        };
672        let allowed = match smaps.as_deref() {
673            Some(smaps) => Some(check.allows(smaps)),
674            None if guest
675                .thread_state()
676                .changed_root
677                .load(std::sync::atomic::Ordering::SeqCst) =>
678            {
679                crate::tool_global::check_brk_mappings(guest, check)
680            }
681            None => None,
682        }
683        .ok_or_else(|| {
684            diagnostic.rejected("smaps_unavailable", Some(limit), None, &"not_injected");
685            "unsupported: brk where Detcore cannot read the guest's mappings"
686        })?;
687        if !allowed {
688            diagnostic.rejected(
689                if requested <= current {
690                    "shrink_predicate"
691                } else {
692                    "grow_predicate"
693                },
694                Some(limit),
695                smaps.as_deref(),
696                &"not_injected",
697            );
698            return Ok(current);
699        }
700        if requested <= current {
701            // As Linux, which keeps the old break where the unmap fails.
702            let unmapped = guest
703                .inject_with_retry(syscalls::Syscall::Munmap(
704                    syscalls::Munmap::new()
705                        .with_addr(syscalls::Addr::from_raw(new_end as usize))
706                        .with_len((old_end - new_end) as usize),
707                ))
708                .await;
709            if unmapped.is_err() {
710                diagnostic.rejected("munmap_error", Some(limit), smaps.as_deref(), &unmapped);
711            }
712            return Ok(if unmapped.is_ok() { requested } else { current });
713        }
714        let length = new_end - old_end;
715        let mapped = guest
716            .inject_with_retry(syscalls::Syscall::Mmap(
717                syscalls::Mmap::new()
718                    .with_addr(syscalls::Addr::from_raw(old_end as usize))
719                    .with_len(length as usize)
720                    .with_prot(syscalls::ProtFlags::PROT_READ | syscalls::ProtFlags::PROT_WRITE)
721                    .with_flags(
722                        syscalls::MapFlags::MAP_PRIVATE
723                            | syscalls::MapFlags::MAP_ANONYMOUS
724                            | syscalls::MapFlags::MAP_FIXED_NOREPLACE,
725                    )
726                    .with_fd(-1)
727                    .with_offset(0),
728            ))
729            .await;
730        match &mapped {
731            Ok(address) if *address as u64 == old_end => Ok(requested),
732            // A kernel without MAP_FIXED_NOREPLACE treats the address as a
733            // hint and may map elsewhere: give that mapping back, or refuse
734            // where it cannot be.
735            Ok(address) => {
736                let removed = guest
737                    .inject_with_retry(syscalls::Syscall::Munmap(
738                        syscalls::Munmap::new()
739                            .with_addr(syscalls::Addr::from_raw(*address as usize))
740                            .with_len(length as usize),
741                    ))
742                    .await;
743                diagnostic.rejected(
744                    "mmap_wrong_address",
745                    Some(limit),
746                    smaps.as_deref(),
747                    &(&mapped, &removed),
748                );
749                if removed.is_ok() {
750                    Ok(current)
751                } else {
752                    Err(
753                        "unsupported: brk growth the kernel mapped elsewhere and Detcore could not remove",
754                    )
755                }
756            }
757            Err(_) => {
758                diagnostic.rejected("mmap_error", Some(limit), smaps.as_deref(), &mapped);
759                Ok(current)
760            }
761        }
762    }
763}
