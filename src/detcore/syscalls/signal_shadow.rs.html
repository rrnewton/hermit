9//! Keeping each thread's [`SignalShadow`] in step with the guest.
10//!
11//! On unless `Config::signal_state_copy` is off (`--no-signal-state-copy`;
12//! docs/SIGNAL_STATE_CHECK.md).
13//!
14//! Every function here runs after the guest operation it records has
15//! committed, uses only values Detcore already holds for that operation, and
16//! issues no injection and no scheduler request: recording changes nothing the
17//! guest or the schedule can observe, and adds no log line at INFO or above
18//! while the copy matches (one TRACE line per check, for counting). The
19//! scheduler learns the copy's dispositions separately, from
20//! [`publish_dispositions`], a message that carries no logical time.
21//!
22//! Where the backend's kernel signal state is the guest's own
23//! ([`kernel_state_is_the_guests`]), the
24//! copy is checked against the kernel after each recorded call: the old action
25//! or old mask the kernel wrote back, and `SigBlk`, `SigIgn` and `SigCgt`. A
26//! mismatch is logged once, by name, and the thread's copy is dropped: it is
27//! never repaired from the kernel, which would hide the defect the check is
28//! for.

57/// The signal whose kernel state differs from the guest's by design, left out
58/// of every comparison: Reverie's preemption signal is Hermit's, and Detcore
59/// keeps the guest from installing or blocking it (`handle_rt_sigaction`,
60/// `rt_sigprocmask_inner`).
61fn reserved() -> KernelSigset {
62    sigbit(reverie::PERF_EVENT_SIGNAL as i32)
63}
64
65/// The signals Linux force-delivers (`force_sig_info_to_task`): forcing one
66/// that is blocked or ignored, or caught but blocked, unblocks it and resets
67/// its action to the default, for the whole group sharing the actions,
68/// before any ptrace stop. The copy sees none of these:
69/// - SIGTRAP from the ptrace backend's single-step (Reverie characterizes it
70///   in `sigtrap_profile_follows_each_handler_shape`);
71/// - SIGSEGV from the backend's own trapped `rdtsc` or `cpuid`, which it
72///   emulates and suppresses;
73/// - a guest's own fault (SIGSEGV, SIGBUS, SIGILL, SIGFPE), and SIGSYS from a
74///   guest's seccomp `RET_TRAP` filter.
75///
76/// So a difference confined to these signals makes the copy unknown (it is
77/// given up quietly) rather than reported, and never passes as a match.
78fn forced() -> KernelSigset {
79    [
80        libc::SIGTRAP,
81        libc::SIGSEGV,
82        libc::SIGBUS,
83        libc::SIGILL,
84        libc::SIGFPE,
85        libc::SIGSYS,
86    ]
87    .into_iter()
88    .fold(0, |set, signal| set | sigbit(signal))
89}

91/// What comparing one of the copy's sets with the kernel's found.
92#[derive(Clone, Copy, Debug, Eq, PartialEq)]
93enum Compared {
94    Same,
95    /// Only signals Linux force-delivers differ ([`forced`]).
96    ForcedOnly,
97    Different,
98}
99
100/// A forced delivery only ever clears a signal's bit (it unblocks the
101/// signal and resets its action to the default), so a difference is
102/// [`Compared::ForcedOnly`] only when every differing bit is a signal Linux
103/// force-delivers and the kernel holds none the copy lacks. Anything else is
104/// a real difference.
105fn compare(kernel: KernelSigset, copy: KernelSigset) -> Compared {
106    match (kernel ^ copy) & !reserved() {
107        0 => Compared::Same,
108        difference if difference & !forced() == 0 && kernel & !copy & !reserved() == 0 => {
109            Compared::ForcedOnly
110        }
111        _ => Compared::Different,
112    }
113}

115/// Gives the copy up quietly: the kernel's state of a signal Linux
116/// force-delivers differs from it in a way only a forced delivery explains
117/// ([`forced`]).
118fn forced_changed<G, T>(guest: &mut G, at: &str)
119where
120    G: Guest<Detcore<T>>,
121    T: RecordOrReplay,
122{
123    debug!(
124        "[dtid {}] signal-state copy dropped at {at}: a force-delivered signal's kernel state \
125         differs (SIGTRAP from a ptrace single-step, SIGSEGV from a trapped rdtsc or cpuid, \
126         or a fault)",
127        guest.thread_state().dettid
128    );
129    drop_copy(guest);
130}

143/// Whether this backend's kernel signal state for a guest thread is the
144/// guest's own: the guest thread is a host task, and no runtime in the
145/// guest's process installs handlers or masks of its own
146/// (`BackendCapabilities::kernel_signal_state_is_guests`: ptrace, and
147/// e9patch, which runs under ptrace). Threads must also be
148/// serialized: otherwise two threads sharing actions could change them in the
149/// kernel in one order and in the copy in another. Elsewhere the copy is
150/// neither seeded from the kernel nor checked against it.
151fn kernel_state_is_the_guests<G, T>(guest: &G) -> bool
152where
153    G: Guest<Detcore<T>>,
154    T: RecordOrReplay,
155{
156    let config = guest.config();
157    config.signal_state_copy
158        && config.sequentialize_threads
159        && config.backend.kernel_signal_state_is_guests
160        && !host_signal_state_reads_refused()
161}

163/// Applies `change` to this thread's copy: the actions its group shares and
164/// its own masks. Returns `None`, and drops the thread's copy without a new
165/// report, when the shared actions are unknown because another thread
166/// sharing them lost its copy (that loss was reported).
167fn with_copy<G, T, R>(
168    guest: &mut G,
169    change: impl FnOnce(&mut Dispositions, &mut ThreadSignals) -> R,
170) -> Option<R>
171where
172    G: Guest<Detcore<T>>,
173    T: RecordOrReplay,
174{
175    let shadow = guest.thread_state_mut().signal_shadow.as_mut()?;
176    let actions = std::sync::Arc::clone(&shadow.dispositions);
177    let mut actions = actions.lock().unwrap();
178    match actions.as_mut() {
179        Some(dispositions) => Some(change(dispositions, &mut shadow.thread)),
180        None => {
181            drop(actions);
182            debug!(
183                "[dtid {}] signal-state copy dropped: a thread sharing its actions lost its copy",
184                guest.thread_state().dettid
185            );
186            drop_copy(guest);
187            None
188        }
189    }
190}

192/// Gives this thread's copy up quietly, for a call whose effect cannot be
193/// known (`reason` says why).
194pub(crate) fn give_up<G, T>(guest: &mut G, reason: &str)
195where
196    G: Guest<Detcore<T>>,
197    T: RecordOrReplay,
198{
199    if guest.thread_state().signal_shadow.is_some() {
200        debug!(
201            "[dtid {}] signal-state copy dropped: {reason}",
202            guest.thread_state().dettid
203        );
204        drop_copy(guest);
205    }
206}

208/// Drops this thread's copy: threads it creates start without one, and the
209/// actions it shares become unknown, because this thread's later changes to
210/// them would go unrecorded. A copy given up quietly is seeded again at the
211/// thread's next exec, whose ignored set and mask describe the whole state
212/// (nothing of the old image is repaired); [`diverged`] marks a copy that
213/// diverged as never to be seeded again.
214fn drop_copy<G, T>(guest: &mut G)
215where
216    G: Guest<Detcore<T>>,
217    T: RecordOrReplay,
218{
219    drop_thread_copy(guest.thread_state_mut());
220}

222/// [`drop_copy`] given only the thread's state.
223fn drop_thread_copy<T>(thread: &mut crate::tool_local::ThreadState<T>) {
224    if let Some(shadow) = thread.signal_shadow.as_ref() {
225        shadow.forget_dispositions();
226    }
227    thread.signal_shadow = None;
228}

230/// At a syscall's entry, the thread has run guest code since its last event:
231/// a temporary-mask restore no handler took has happened, a signal may have
232/// reached a handler unreported, and an open handler frame may have returned
233/// unseen. Accounts for all three before the call can depend on the mask (a
234/// clone copies it; an exec keeps it). No guest memory is read here: a wait's
235/// temporary mask is recorded by its handler, in the call's own turn
236/// ([`note_wait_mask`]).
237pub(crate) fn at_syscall_entry<G, T>(guest: &mut G)
238where
239    G: Guest<Detcore<T>>,
240    T: RecordOrReplay,
241{
242    take_unreported(guest);
243    let Some(shadow) = guest.thread_state_mut().signal_shadow.as_mut() else {
244        return;
245    };
246    shadow.thread.settle();
247    shadow.thread.wait_mask = None;
248    shadow.thread.wait_ran = None;
249    reconcile_returns(guest, "syscall entry");
250}

252/// Signals the backend passed on to a thread without a report (the ptrace
253/// backend does so for many signals it held during a Tool injection), as
254/// (host pid, host tid, signal). They are recorded where the copy lives, in
255/// this process (`Tool::filter_unreported_signal`), and consumed at that
256/// thread's next operation, whether or not it has a copy. Realtime signals
257/// are shown and recorded like any other
258/// (<https://github.com/rrnewton/reverie/issues/207>).
259#[cfg(not(test))]
260static UNREPORTED: std::sync::Mutex<Vec<(i32, i32, i32)>> = std::sync::Mutex::new(Vec::new());
261
262/// Whether [`UNREPORTED`] may hold an entry, so that threads on a backend
263/// that never notes one (or with none noted) never take its lock.
264#[cfg(not(test))]
265static ANY_UNREPORTED: std::sync::atomic::AtomicBool = std::sync::atomic::AtomicBool::new(false);
266
267// Tests keep their notes per test thread, so concurrent tests cannot see each
268// other's.
269#[cfg(test)]
270thread_local! {
271    static UNREPORTED_NOTES: std::cell::RefCell<Vec<(i32, i32, i32)>> =
272        const { std::cell::RefCell::new(Vec::new()) };
273}

275/// Runs `f` on the notes.
276fn with_notes<R>(f: impl FnOnce(&mut Vec<(i32, i32, i32)>) -> R) -> R {
277    #[cfg(not(test))]
278    {
279        let mut notes = UNREPORTED.lock().unwrap();
280        let result = f(&mut notes);
281        ANY_UNREPORTED.store(!notes.is_empty(), std::sync::atomic::Ordering::SeqCst);
282        result
283    }
284    #[cfg(test)]
285    {
286        UNREPORTED_NOTES.with(|notes| f(&mut notes.borrow_mut()))
287    }
288}

290/// Whether any note may exist, without taking a lock.
291fn any_notes() -> bool {
292    #[cfg(not(test))]
293    {
294        ANY_UNREPORTED.load(std::sync::atomic::Ordering::SeqCst)
295    }
296    #[cfg(test)]
297    {
298        UNREPORTED_NOTES.with(|notes| !notes.borrow().is_empty())
299    }
300}

302/// Records that `signal` reached host thread `tid` of process `pid` without a
303/// report.
304pub(crate) fn note_unreported(pid: i32, tid: i32, signal: i32) {
305    trace!("[tid {tid}] signal-state copy: signal {signal} was passed on unreported");
306    with_notes(|notes| notes.push((pid, tid, signal)));
307}

309/// Whether another live thread has an unreported signal noted: if it shares
310/// this thread's actions, its handler may have changed them (`SA_RESETHAND`)
311/// unseen, until it gives its own copy up. Entries of threads that no longer
312/// exist are removed here.
313fn unreported_elsewhere(tid: i32) -> bool {
314    if !any_notes() {
315        return false;
316    }
317    with_notes(|notes| {
318        notes.retain(|(_, t, _)| {
319            !super::threads::thread_is_gone(Pid::from_raw(*t), Pid::from_raw(*t))
320        });
321        // Any other thread: a CLONE_SIGHAND group need not be one process.
322        notes.iter().any(|(_, t, _)| *t != tid)
323    })
324}

326/// Whether any unreported signal is noted and not yet consumed: a clone now
327/// could copy actions a handler has already changed unseen.
328pub(crate) fn unreported_pending() -> bool {
329    any_notes()
330        && with_notes(|notes| {
331            notes.retain(|(_, t, _)| {
332                !super::threads::thread_is_gone(Pid::from_raw(*t), Pid::from_raw(*t))
333            });
334            !notes.is_empty()
335        })
336}

338/// A signal that reached this thread unreported may have entered a handler
339/// the copy never saw, with the action as it stood at delivery, which the
340/// copy cannot know any more: the copy is given up. This thread's notes are
341/// consumed whether or not it has a copy.
342fn take_unreported<G, T>(guest: &mut G)
343where
344    G: Guest<Detcore<T>>,
345    T: RecordOrReplay,
346{
347    if !any_notes() {
348        return;
349    }
350    let (pid, tid) = (guest.pid().as_raw(), guest.tid().as_raw());
351    let signals: Vec<i32> = with_notes(|notes| {
352        let (mine, others): (Vec<_>, Vec<_>) = notes
353            .drain(..)
354            .partition(|(p, t, _)| *p == pid && *t == tid);
355        *notes = others;
356        mine.into_iter().map(|(_, _, signal)| signal).collect()
357    });
358    if let Some(signal) = signals.first() {
359        give_up(guest, &format!("signal {signal} was passed on unreported"));
360    }
361}

363/// Before a transition that depends on the mask: where the backend reports
364/// `rt_sigreturn`, checks that the kernel holds the mask the copy predicts
365/// (with a handler frame open, or after a report); elsewhere, accounts for
366/// handler returns the kernel ran without a report, from the kernel's
367/// current mask. Does nothing otherwise, or where the kernel's state is not
368/// the guest's.
369fn reconcile_returns<G, T>(guest: &mut G, before: &str)
370where
371    G: Guest<Detcore<T>>,
372    T: RecordOrReplay,
373{
374    let open_frames = guest
375        .thread_state()
376        .signal_shadow
377        .as_ref()
378        .is_some_and(|shadow| !shadow.thread.activations.is_empty());
379    let verify = guest
380        .thread_state()
381        .signal_shadow
382        .as_ref()
383        .is_some_and(|shadow| shadow.thread.verify_at_entry);
384    if !(open_frames || verify) || !kernel_state_is_the_guests(guest) {
385        return;
386    }
387    let Some(kernel) = read_or_give_up(guest, before) else {
388        return;
389    };
390    // Where the backend reports rt_sigreturn, frames close only on its
391    // reports, so nothing is inferred. The kernel must still hold the mask
392    // the copy predicts, for an open handler or after a report; otherwise a
393    // report was missed (the capability names the calls it does not
394    // report), the kernel did not follow one, or the handler's frame was
395    // never written, and the copy fails closed.
396    if observes_sigreturns(guest.config()) {
397        let Some(shadow) = guest.thread_state_mut().signal_shadow.as_mut() else {
398            return;
399        };
400        shadow.thread.verify_at_entry = false;
401        match compare(kernel.blocked, shadow.thread.blocked) {
402            Compared::Same => shadow.thread.observe_handler_mask(),
403            Compared::ForcedOnly => forced_changed(guest, before),
404            Compared::Different => give_up(
405                guest,
406                "the kernel's mask is not the one the copy predicts after a reported return or \
407                 for an open handler frame",
408            ),
409        }
410        return;
411    }
412    let Some(shadow) = guest.thread_state_mut().signal_shadow.as_mut() else {
413        return;
414    };
415    let copy = shadow.thread.blocked;
416    let open = shadow.thread.activations.len();
417    let innermost_observed = shadow.thread.innermost_observed();
418    if shadow
419        .thread
420        .reconcile_returns(kernel.blocked, reserved() | forced())
421    {
422        let closed = open - shadow.thread.activations.len();
423        if closed > 1 {
424            // An outer frame's mask: several returns, or an inner handler
425            // that rewrote its uc_sigmask. Mask equality cannot tell which.
426            debug!(
427                "[dtid {}] signal-state copy dropped at {before}: the kernel's mask matches an \
428                 outer handler frame ({closed} frames), which is ambiguous",
429                guest.thread_state().dettid
430            );
431            drop_copy(guest);
432        } else {
433            if closed == 1 {
434                debug!(
435                    "[dtid {}] signal-state copy: a handler frame taken as returned at {before}",
436                    guest.thread_state().dettid
437                );
438                if !innermost_observed {
439                    // A return, or a handler entry the copy predicted wrongly
440                    // (or a frame the kernel failed to write): mask equality
441                    // cannot tell which.
442                    #[cfg(test)]
443                    UNVERIFIED_ENTRIES.with(|count| count.set(count.get() + 1));
444                    return give_up(
445                        guest,
446                        "a handler frame was taken as returned before any check saw its \
447                         handler's mask",
448                    );
449                }
450            } else if let Some(shadow) = guest.thread_state_mut().signal_shadow.as_mut() {
451                // The kernel holds the mask the copy predicts for the open
452                // handler.
453                shadow.thread.observe_handler_mask();
454            }
455            let blocked = guest
456                .thread_state()
457                .signal_shadow
458                .as_ref()
459                .map(|shadow| shadow.thread.blocked);
460            // The frames were matched with the forced signals left out; the
461            // whole mask must still be the copy's, or differ only as a forced
462            // delivery leaves it.
463            match blocked.map(|blocked| compare(kernel.blocked, blocked)) {
464                Some(Compared::ForcedOnly) => forced_changed(guest, before),
465                Some(Compared::Different) => give_up(
466                    guest,
467                    "the kernel's mask holds a force-delivered signal the copy's mask does not",
468                ),
469                Some(Compared::Same) | None => {}
470            }
471        }
472    } else if open > 0 {
473        // A handler may install any mask by rewriting its frame's uc_sigmask
474        // before an unseen return.
475        give_up(
476            guest,
477            &format!(
478                "at {before} the kernel's mask matches no open handler frame, which a \
479                 handler's rewritten uc_sigmask could explain"
480            ),
481        );
482    } else {
483        diverged(
484            guest,
485            before,
486            &format!(
487                "SigBlk kernel {:#018x} is neither the copy's {:#018x} nor an open handler frame's",
488                kernel.blocked & !reserved(),
489                copy & !reserved()
490            ),
491        );
492    }
493}

495/// Seeds the first image's copy, or applies `execve`'s reset to a later one.
496/// Called from the new image's first callback. The first image's ignored set
497/// and mask are what it inherited from the launcher, read from the kernel
498/// before it runs a guest instruction; where that read is unavailable, the
499/// copy stays unknown. Seeding there needs the backend to report the first
500/// image's inherited state, which the A5 design puts in the `Guest` contract
501/// (its step 3).
502pub(crate) fn at_exec<G, T>(guest: &mut G)
503where
504    G: Guest<Detcore<T>>,
505    T: RecordOrReplay,
506{
507    // A nonleader exec takes the leader's tid, whose entry the scheduler
508    // drops with the old leader, and the copy may not survive the exec: what
509    // the scheduler holds is unknown, so the mask, or its withdrawal, is
510    // published again.
511    guest.thread_state_mut().published_mask = None;
512    if let Some(shadow) = guest.thread_state_mut().signal_shadow.as_mut() {
513        if !shadow.exec() {
514            drop_copy(guest);
515            return;
516        }
517        trace!(
518            "[dtid {}] signal-state copy: reset for execve",
519            guest.thread_state().dettid
520        );
521        check_against_kernel(guest, "execve");
522        return;
523    }
524    // A thread with no copy is seeded at its exec, where the ignored set and
525    // the mask describe the whole state. One whose copy diverged, or whose
526    // creator's did, is never seeded again.
527    if guest.thread_state().signal_shadow_lost || !kernel_state_is_the_guests(guest) {
528        return;
529    }
530    match read_thread_signal_state(guest.pid(), guest.tid()) {
531        Ok(state) => {
532            guest.thread_state_mut().signal_shadow =
533                Some(SignalShadow::seeded(state.ignored, state.blocked));
534            trace!(
535                "[dtid {}] signal-state copy: seeded from the first image",
536                guest.thread_state().dettid
537            );
538        }
539        Err(errno) => debug!(
540            "[dtid {}] signal-state copy not seeded: the kernel state is unreadable ({errno})",
541            guest.thread_state().dettid
542        ),
543    }
544}

546/// Records a committed `rt_sigaction`. `new` is the action the kernel
547/// installed (Detcore's sanitized copy of an action the kernel validated),
548/// `None` for a query. `result` is the call's result: an `EFAULT` after the
549/// new action was read can only come from writing `old_action`, after the new
550/// action is installed.
551pub(super) fn record_rt_sigaction<G, T>(
552    guest: &mut G,
553    signum: i32,
554    new: Option<KernelSigaction>,
555    old_action: Option<AddrMut<'_, libc::sigaction>>,
556    result: &Result<i64, Errno>,
557) where
558    G: Guest<Detcore<T>>,
559    T: RecordOrReplay,
560{
561    let committed = result.is_ok() || (new.is_some() && *result == Err(Errno::EFAULT));
562    if !committed {
563        return;
564    }
565    let Some(predicted) = with_copy(guest, |dispositions, _| {
566        dispositions.sigaction(signum, new.map(Action::from))
567    }) else {
568        return;
569    };
570    let old = match predicted {
571        Ok(old) => old,
572        Err(errno) => {
573            return diverged(
574                guest,
575                "rt_sigaction",
576                &format!(
577                    "the kernel accepted signal {signum}, which the copy refuses with {errno}"
578                ),
579            );
580        }
581    };
582    let written = match (result.is_ok(), old_action) {
583        (true, Some(address)) => match guest
584            .memory()
585            .read_value::<_, KernelSigaction>(address.cast::<KernelSigaction>())
586        {
587            Ok(written) => Some(written),
588            Err(_) => {
589                return give_up(guest, "the old action the kernel wrote cannot be read back");
590            }
591        },
592        _ => None,
593    };
594    // A sibling's unreported handler may have reset this action unseen
595    // (SA_RESETHAND): the old action is not compared while one is noted.
596    if let Some(written) = written
597        && Action::from(written) != old
598        && !unreported_elsewhere(guest.tid().as_raw())
599    {
600        // A forced delivery changes only the handler, to the default. Whether
601        // the signal was blocked when it was forced is not knowable here (see
602        // `check_against_kernel`).
603        if forced() & sigbit(signum) != 0
604            && Action::from(written)
605                == (Action {
606                    handler: libc::SIG_DFL as u64,
607                    ..old
608                })
609            && old.handler != libc::SIG_DFL as u64
610        {
611            return forced_changed(guest, "rt_sigaction");
612        }
613        return diverged(
614            guest,
615            "rt_sigaction",
616            &format!(
617                "old action of signal {signum}: the kernel wrote {:?}, the copy holds {old:?}",
618                Action::from(written)
619            ),
620        );
621    }
622    check_against_kernel(guest, "rt_sigaction");
623}

625/// What Detcore knows of an `rt_sigprocmask` call's set.
626#[derive(Clone, Copy, Debug)]
627pub(super) enum SetInput {
628    /// The call passed no set: a query.
629    None,
630    /// Detcore never injected the call: its own check of the set failed
631    /// first, so the kernel did nothing.
632    NotRun,
633    /// The kernel read this set and accepted it (the `SIG_BLOCK` and
634    /// `SIG_SETMASK` paths validate it before injecting). It is the set
635    /// without Hermit's reserved signal.
636    Validated(KernelSigset),
637    /// The call ran with a set the kernel validated only by running; Detcore
638    /// reads it after a successful call, from memory the kernel just read
639    /// (`None` when the old mask overwrote it, or the read failed).
640    Unvalidated(Option<KernelSigset>),
641}

643/// Records a committed `rt_sigprocmask`. An `EFAULT` is a committed change
644/// only when it can come from writing `oldset` alone.
645pub(super) fn record_rt_sigprocmask<G, T>(
646    guest: &mut G,
647    how: i32,
648    set: SetInput,
649    oldset: Option<AddrMut<'_, libc::sigset_t>>,
650    result: &Result<i64, Errno>,
651) where
652    G: Guest<Detcore<T>>,
653    T: RecordOrReplay,
654{
655    let valid_how = matches!(how, libc::SIG_BLOCK | libc::SIG_UNBLOCK | libc::SIG_SETMASK);
656    let set = match (set, result) {
657        (SetInput::NotRun, _) => return,
658        (SetInput::None, Ok(_)) => None,
659        (SetInput::Validated(set), Ok(_) | Err(Errno::EFAULT)) => Some(set),
660        (SetInput::Unvalidated(Some(set)), Ok(_)) => Some(set),
661        (SetInput::Unvalidated(None), Ok(_)) => {
662            debug!(
663                "[dtid {}] signal-state copy dropped: rt_sigprocmask's set is not known after \
664                 the call (it aliases the old mask, or could not be read)",
665                guest.thread_state().dettid
666            );
667            drop_copy(guest);
668            return;
669        }
670        // The kernel reads the set before checking `how`, and before writing
671        // the old mask: without an old mask, or with an invalid `how`, the
672        // fault was the set's and nothing changed.
673        (SetInput::Unvalidated(_), Err(Errno::EFAULT)) if oldset.is_none() || !valid_how => {
674            return;
675        }
676        (SetInput::Unvalidated(_), Err(Errno::EFAULT)) => {
677            // The set or the old mask: Linux leaves no way to tell which.
678            debug!(
679                "[dtid {}] signal-state copy dropped: rt_sigprocmask's EFAULT may come from \
680                 its set or its old mask",
681                guest.thread_state().dettid
682            );
683            drop_copy(guest);
684            return;
685        }
686        _ => return,
687    };
688    let Some(predicted) = with_copy(guest, |_, thread| thread.sigprocmask(how, set)) else {
689        return;
690    };
691    let old = match predicted {
692        Ok(old) => old,
693        Err(errno) => {
694            return diverged(
695                guest,
696                "rt_sigprocmask",
697                &format!("the kernel accepted how={how}, which the copy refuses with {errno}"),
698            );
699        }
700    };
701    let written = match (result.is_ok(), oldset) {
702        (true, Some(address)) => match guest
703            .memory()
704            .read_value::<_, KernelSigset>(address.cast::<KernelSigset>())
705        {
706            Ok(written) => Some(written),
707            Err(_) => return give_up(guest, "the old mask the kernel wrote cannot be read back"),
708        },
709        _ => None,
710    };
711    if let Some(written) = written {
712        match compare(written, old) {
713            Compared::Same => {}
714            Compared::ForcedOnly => return forced_changed(guest, "rt_sigprocmask"),
715            Compared::Different => {
716                return diverged(
717                    guest,
718                    "rt_sigprocmask",
719                    &format!(
720                        "old mask: the kernel wrote {:#x}, the copy holds {:#x}",
721                        written & !reserved(),
722                        old & !reserved()
723                    ),
724                );
725            }
726        }
727    }
728    check_against_kernel(guest, "rt_sigprocmask");
729}

731/// Records that the Tool approved `signal` for delivery to this thread, so the
732/// backend delivers it now: a caught signal enters its handler with
733/// `sa_mask` added, and `SA_RESETHAND` puts the default back. The handler
734/// mask is checked at the thread's next recorded call, when the frame has been
735/// written. A delivery whose frame cannot be written becomes a SIGSEGV and
736/// leaves the old mask, which the check can take for the frame's return: that
737/// case is not told apart.
738///
739/// Before the copy changes, a caught signal's delivery is checked against the
740/// kernel through the backend (`check_delivery`, A5 step 3.2). Checking adds
741/// only reads and a refusal: the copy changes exactly as it would unchecked,
742/// at the same point, unless a check fails, which drops the copy, or the
743/// backend reports that the kernel will not deliver the signal at this
744/// resume, when nothing is recorded until a later delivery. This must
745/// be the last thing the signal callback does before it returns the signal:
746/// a later injection would void the backend's approval.
747pub(crate) async fn check_and_record_delivery<G, T>(guest: &mut G, signal: i32)
748where
749    G: Guest<Detcore<T>>,
750    T: RecordOrReplay,
751{
752    if check_delivery_to_record(guest, signal).await {
753        record_checked_delivery(guest, signal);
754    }
755}

757/// The check half of [`check_and_record_delivery`]: whether `signal`'s
758/// delivery is to be recorded, after the copy has caught up and
759/// `Guest::deliver_signal` has checked it. A two-phase commit
760/// (`Tool::commit_signal_event`) runs it, checks that it was granted no turn
761/// (its `deliver_signal` must not wait for one), and only then records.
762pub(crate) async fn check_delivery_to_record<G, T>(guest: &mut G, signal: i32) -> bool
763where
764    G: Guest<Detcore<T>>,
765    T: RecordOrReplay,
766{
767    let had_copy = guest.thread_state().signal_shadow.is_some();
768    // At a delivery stop the kernel has not installed the handler mask yet.
769    take_unreported(guest);
770    take_wait_outcome(guest).await;
771    reconcile_returns(guest, "signal delivery");
772    // A signal the kernel does not deliver at this resume is delivered, and
773    // recorded, at a later stop, if at all.
774    let record = check_delivery(guest, signal, had_copy).await;
775    // A two-phase commit's test sees the check made, and can grant a turn in
776    // it, as a `deliver_signal` that waited for one would
777    // (`signal_announcement_tests`).
778    #[cfg(test)]
779    {
780        crate::signal_announcement_tests::note_phase("checked");
781        crate::signal_announcement_tests::grant_during_check_if_asked(guest.thread_state_mut());
782    }
783    record
784}

812/// The copy's half of [`check_and_record_delivery`], after reconciliation:
813/// `signal` is delivered as the copy holds it.
814fn commit_delivery<G, T>(guest: &mut G, signal: i32)
815where
816    G: Guest<Detcore<T>>,
817    T: RecordOrReplay,
818{
819    let Some(delivered) = with_copy(guest, |dispositions, thread| {
820        crate::signal_state::deliver(dispositions, thread, signal)
821    }) else {
822        return;
823    };
824    match delivered {
825        Ok(Some(action)) => {
826            trace!(
827                "[dtid {}] signal-state copy: recorded the delivery of signal {signal}",
828                guest.thread_state().dettid
829            );
830            // A handler whose mask equals an open frame's saved mask cannot
831            // be told from that frame's return: the next check verifies
832            // nothing about this entry.
833            if let Some(shadow) = guest.thread_state().signal_shadow.as_ref()
834                && action.handler > 1
835                && shadow.thread.activations.last() == Some(&shadow.thread.blocked)
836            {
837                debug!(
838                    "[dtid {}] signal-state copy: the handler mask of signal {signal} equals the \
839                     mask its frame saves, so its entry cannot be told from its return",
840                    guest.thread_state().dettid
841                );
842            }
843        }
844        Ok(None) => {
845            debug!(
846                "[dtid {}] signal-state copy dropped: more than {} handler frames open",
847                guest.thread_state().dettid,
848                crate::signal_state::MAX_OPEN_FRAMES
849            );
850            drop_copy(guest);
851        }
852        Err(errno) => diverged(
853            guest,
854            "signal delivery",
855            &format!("the backend delivers signal {signal}, which the copy refuses with {errno}"),
856        ),
857    }
858}

860/// One thread's counts of the deliveries [`check_delivery`] saw (A5 step 3.2):
861/// approved (and of those, approved at a held signal's report), contradicted,
862/// checked with the interruption taken from the registers
863/// (`delivery_interruption`), and not checked, by reason; and the signals the
864/// backend passed on without reporting them, which no check saw, by the
865/// backend's reason ([`reverie::UnreportedSignalReason`]). Logged at TRACE
866/// when the thread exits, below INFO so the logs that runs compare are
867/// unchanged.
868#[derive(Clone, Debug, Default)]
869pub(crate) struct DeliveryCounters {
870    /// The thread the counts were taken under. A non-leader exec gives the
871    /// thread its leader's identity before it exits, so the counts keep the
872    /// identity they were taken under.
873    owner: Option<crate::types::DetTid>,
874    approved: u64,
875    /// Approved deliveries of a signal the backend held during an injection
876    /// and reported after the callback that held it
877    /// ([`reverie::SignalReportPath::Held`]).
878    held: u64,
879    /// Approved deliveries whose interruption was a syscall result Detcore
880    /// finished, so the backend compared `rax` with it.
881    result_checked: u64,
882    mismatched: u64,
883    interruption_from_registers: u64,
884    skipped: std::collections::BTreeMap<&'static str, u64>,
885    /// Signals the backend passed on without a report, by the backend's
886    /// reason: those that could reach a guest handler, as the copy notes them
887    /// (`note_unreported`). Each pass is counted: at that resume the kernel
888    /// delivers the signal, puts it back in the queue when the thread's mask
889    /// blocks it (a later stop can take the same instance again), queues a
890    /// new instance with a kernel siginfo (a syscall stop), or discards it
891    /// (an ignored signal, or a ptrace event stop or group stop). Signals the
892    /// filter dropped are not counted.
893    unreported: std::collections::BTreeMap<&'static str, u64>,
894}
895
896impl DeliveryCounters {
897    fn skip(&mut self, reason: &'static str) {
898        *self.skipped.entry(reason).or_default() += 1;
899    }
900
901    /// Counts a signal the backend passed on unreported for `reason`, taken
902    /// by thread `dettid`.
903    pub(crate) fn count_unreported(
904        &mut self,
905        dettid: crate::types::DetTid,
906        reason: reverie::UnreportedSignalReason,
907    ) {
908        self.owner.get_or_insert(dettid);
909        *self.unreported.entry(reason.name()).or_default() += 1;
910    }
911
912    /// Logs the counts, if any delivery was seen, under the identity they
913    /// were taken under (`dettid` if none was recorded).
914    pub(crate) fn log(&self, dettid: crate::types::DetTid) {
915        if self.approved + self.mismatched == 0
916            && self.skipped.is_empty()
917            && self.unreported.is_empty()
918        {
919            return;
920        }
921        let dettid = self.owner.unwrap_or(dettid);
922        trace!(
923            target: "detcore::signal_delivery",
924            "[dtid {dettid}] signal deliveries: approved={} held={} result_checked={} mismatched={} interruption_from_registers={} skipped={:?} unreported={:?}",
925            self.approved,
926            self.held,
927            self.result_checked,
928            self.mismatched,
929            self.interruption_from_registers,
930            self.skipped,
931            self.unreported
932        );
933    }
934}

965/// `sizeof(struct ucontext)` on x86-64 Linux (`include/uapi/asm-generic/ucontext.h`):
966/// `uc_flags` and `uc_link` (8 bytes each), `stack_t uc_stack` (24),
967/// `struct sigcontext uc_mcontext` (256) and the kernel's 8-byte
968/// `uc_sigmask`. Not libc's `ucontext_t`, which is larger.
969#[cfg(target_arch = "x86_64")]
970const KERNEL_UCONTEXT_SIZE: u64 = 304;

972/// Whether `regs` are the registers Linux leaves when it has just built a
973/// frame for `signal`'s handler at `handler`, at a stop whose `orig_ax` was
974/// `orig_ax`, so that no guest instruction has run since: a chained delivery
975/// at the same exit. Linux v6.12 `x64_setup_rt_frame`
976/// (https://git.kernel.org/pub/scm/linux/kernel/git/stable/linux.git/tree/arch/x86/kernel/signal_64.c?h=v6.12#n202)
977/// sets `rdi` to the signal, `rsi` to the frame's siginfo, `rdx` to its
978/// ucontext, `rip` to the handler and `rsp` to the frame, whose layout is
979/// `struct rt_sigframe { pretcode; uc; info; }` (`arch/x86/include/asm/sigframe.h`).
980/// `get_sigframe` puts the frame 8 bytes below a 16-byte boundary
981/// (https://git.kernel.org/pub/scm/linux/kernel/git/stable/linux.git/tree/arch/x86/kernel/signal.c?h=v6.12#n144).
982/// Neither writes `orig_ax`, while every later entry to the kernel other
983/// than by a syscall sets it to -1
984/// (https://git.kernel.org/pub/scm/linux/kernel/git/stable/linux.git/tree/arch/x86/entry/entry_64.S?h=v6.12#n344).
985///
986/// The handler's address alone was matched again by a handler whose code
987/// returns to its own first instruction (Codex review of A5 step 3.2). Some
988/// stops still match that are not chained: a signal that reaches the
989/// handler's first instruction after the guest came back to it with these
990/// registers unchanged, when the frame's delivery stop had `orig_ax` -1, or
991/// a fault there (which `delivery_interruption` excludes by its siginfo,
992/// `synchronous_with_a_kernel_code`).
993/// `rax`, which Linux sets to 0 too, is not checked: a syscall the tool
994/// injects at the stop leaves its result there (`reverie::Interruption`).
995/// x32 and ia32 frames have another layout and never match.
996#[cfg(target_arch = "x86_64")]
997fn at_frame_entry(regs: &libc::user_regs_struct, handler: u64, signal: i32, orig_ax: i64) -> bool {
998    regs.rip == handler
999        && regs.rdi == signal as u64
1000        && regs.rdx == regs.rsp.wrapping_add(8)
1001        && regs.rsi == regs.rdx.wrapping_add(KERNEL_UCONTEXT_SIZE)
1002        && regs.rsp % 16 == 8
1003        && regs.orig_rax as i64 == orig_ax
1004}

1006/// Whether `signal` is one of Linux's synchronous signals
1007/// (https://git.kernel.org/pub/scm/linux/kernel/git/stable/linux.git/tree/kernel/signal.c?h=v6.12#n196)
1008/// with a positive `si_code`. A fault has one, and needs the thread back in
1009/// user code, so it is never a chained delivery at the same exit; such a
1010/// signal is therefore never taken for one. That loses two genuine chained
1011/// deliveries, which then take the registers' answer, always safe and
1012/// counted: a SIGSEGV for a frame Linux failed to build after an earlier
1013/// one, and a synchronous signal the guest queued with a positive code,
1014/// which Linux lets a thread send only to itself (`do_rt_tgsigqueueinfo`,
1015/// https://git.kernel.org/pub/scm/linux/kernel/git/stable/linux.git/tree/kernel/signal.c?h=v6.12#n4112)
1016/// or a thread-group leader to its own process (`do_rt_sigqueueinfo`,
1017/// https://git.kernel.org/pub/scm/linux/kernel/git/stable/linux.git/tree/kernel/signal.c?h=v6.12#n4065).
1018#[cfg(target_arch = "x86_64")]
1019fn synchronous_with_a_kernel_code(
1020    signal: i32,
1021    info: Option<[u8; reverie::SIGNAL_INFO_SIZE]>,
1022) -> bool {
1023    let synchronous = [
1024        libc::SIGSEGV,
1025        libc::SIGBUS,
1026        libc::SIGILL,
1027        libc::SIGTRAP,
1028        libc::SIGFPE,
1029        libc::SIGSYS,
1030    ];
1031    synchronous.contains(&signal)
1032        && info.is_some_and(|info| i32::from_ne_bytes(info[8..12].try_into().unwrap()) > 0)
1033}

1035/// What the thread was doing at this delivery stop of `signal`, for the
1036/// backend's check of the syscall registers; whether it was taken from the
1037/// registers themselves rather than from Detcore's record; and the stop's
1038/// `orig_ax` (-1 where no register is read).
1039///
1040/// Detcore's record (`ThreadState::delivery_interruption`) is set at each
1041/// guest event: the syscall entered, the result Detcore finished it with, a
1042/// handler frame built at this exit, or no syscall (a preemption or a trapped
1043/// instruction). Between events the guest runs unseen, so the record cannot
1044/// know whether the thread is still where the record left it or has run on in
1045/// user code since. The registers answer that one question, and nothing else:
1046/// - `orig_ax` is -1 once the thread entered the kernel other than by a
1047///   syscall, so a syscall record with `orig_ax` -1 means the thread returned
1048///   to user code since: no syscall.
1049/// - A frame's chained delivery at the same exit stops before any guest
1050///   instruction, with the registers Linux set for the frame and the frame's
1051///   own stop's `orig_ax` (`at_frame_entry`), and is never a synchronous
1052///   signal with a positive `si_code`, as a fault has
1053///   (`synchronous_with_a_kernel_code`). Otherwise, either the
1054///   guest ran since, and with `orig_ax` -1 that is no syscall, or the frame
1055///   was not built (a frame fault, whose SIGSEGV comes at the same exit), and
1056///   the registers' answer is used.
1057/// - A record naming the syscall `orig_ax` names is sent as it stands, so
1058///   `rax` is compared with the result Detcore finished the call with.
1059/// - Where `orig_ax` names a syscall Detcore cannot see (a number outside
1060///   Reverie's table, or a negative one), no callback updated the record, and
1061///   the registers' answer is used.
1062/// - A record that disagrees with a known syscall in `orig_ax` is sent as it
1063///   stands, so the backend reports the contradiction.
1064/// - With no record (a new thread or image), or where Detcore does not see
1065///   every syscall (`--passthru-opt`), the registers' answer is used, and the
1066///   check of the syscall registers verifies nothing.
1067async fn delivery_interruption<G, T>(
1068    guest: &mut G,
1069    signal: i32,
1070) -> (reverie::Interruption, bool, i64)
1071where
1072    G: Guest<Detcore<T>>,
1073    T: RecordOrReplay,
1074{
1075    use reverie::Interruption;
1076    let record = guest.thread_state().delivery_interruption;
1077    // Only x86-64's registers are read here; elsewhere nothing is checked.
1078    #[cfg(not(target_arch = "x86_64"))]
1079    {
1080        let _ = (record, signal);
1081        (Interruption::AfterFrame, true, -1)
1082    }
1083    #[cfg(target_arch = "x86_64")]
1084    {
1085        let regs = guest.regs().await;
1086        let orig = regs.orig_rax as i64;
1087        // Only -1 means no syscall: another negative number is a syscall the
1088        // kernel refused with ENOSYS.
1089        let from_registers = if orig == -1 {
1090            Interruption::NotInSyscall
1091        } else {
1092            Interruption::KernelRun { syscall: orig }
1093        };
1094        if guest.config().passthru_opt {
1095            return (from_registers, true, orig);
1096        }
1097        let state = guest.thread_state();
1098        let (interruption, from_registers) = match record {
1099            Some(Interruption::AfterFrame)
1100                if !synchronous_with_a_kernel_code(signal, guest.signal_info())
1101                    && at_frame_entry(
1102                        &regs,
1103                        state.delivery_frame_handler,
1104                        state.delivery_frame_signal,
1105                        state.delivery_frame_orig_ax,
1106                    ) =>
1107            {
1108                (Interruption::AfterFrame, false)
1109            }
1110            Some(
1111                record @ (Interruption::Return { syscall, .. }
1112                | Interruption::KernelRun { syscall }),
1113            ) if syscall == orig => (record, false),
1114            Some(Interruption::NotInSyscall) if orig == -1 => (Interruption::NotInSyscall, false),
1115            // The record does not describe this stop, and the registers
1116            // decide:
1117            // - there is no record (a new thread or image);
1118            // - the guest ran on since the record (`orig_ax` -1 after a
1119            //   syscall or a frame);
1120            // - still at the syscall's exit without the frame's registers, the
1121            //   frame was not built (a frame fault raises SIGSEGV at the same
1122            //   exit);
1123            // - a synchronous signal with a positive `si_code`, as a fault has,
1124            //   which needs the thread back in user code
1125            //   (`synchronous_with_a_kernel_code`);
1126            // - the thread is in a syscall Detcore cannot see: a number
1127            //   outside Reverie's table, or a negative one the kernel refuses,
1128            //   reaches no Tool callback, so the record still names an earlier
1129            //   event.
1130            None | Some(Interruption::AfterFrame) => (from_registers, true),
1131            _ if orig == -1 || !sees_syscall(orig) => (from_registers, true),
1132            // A known syscall that the record does not name: Detcore missed an
1133            // event it subscribes to. Sent as it stands, so the backend reports
1134            // the contradiction.
1135            Some(record) => (record, false),
1136        };
1137        (interruption, from_registers, orig)
1138    }
1139}

1141/// Checks a caught signal's delivery against the kernel through the backend
1142/// (`Guest::deliver_signal`), from the reconciled copy, before the copy
1143/// records it (A5 step 3.2). Reads only:
1144/// - the backend's approval binds the guest's resume to exactly this signal
1145///   and these masks;
1146/// - a contradiction, where the copy and the kernel are both certain, is a
1147///   Detcore defect: it is reported once, as a divergence, and the copy is
1148///   dropped, so the delivery goes on as the kernel makes it;
1149/// - a point the backend cannot check is counted by reason and changes
1150///   nothing.
1151///
1152/// Not checked, and counted: no copy (told apart from a copy this delivery's
1153/// reconciliation gave up, `had_copy`), the shared actions unknown, a signal
1154/// that is not caught (the kernel ignores it or takes its default action).
1155///
1156/// Returns false only when the backend says the kernel will not deliver the
1157/// signal at this resume, so the copy must not record a delivery:
1158/// [`reverie::SignalDeliveryUnavailable::BlockedAtResume`] (it requeues it)
1159/// and [`reverie::SignalDeliveryUnavailable::NotADeliveryStop`] (it queues
1160/// it anew or drops it).
1161async fn check_delivery<G, T>(guest: &mut G, signal: i32, had_copy: bool) -> bool
1162where
1163    G: Guest<Detcore<T>>,
1164    T: RecordOrReplay,
1165{
1166    if !kernel_state_is_the_guests(guest) {
1167        return true;
1168    }
1169    let dettid = guest.thread_state().dettid;
1170    guest
1171        .thread_state_mut()
1172        .delivery_counters
1173        .owner
1174        .get_or_insert(dettid);
1175    let state = guest.thread_state().signal_shadow.as_ref().map(|shadow| {
1176        let actions = shadow.dispositions.lock().unwrap();
1177        (
1178            actions
1179                .as_ref()
1180                .map(|dispositions| dispositions.action(signal)),
1181            shadow.thread.saved.unwrap_or(shadow.thread.blocked),
1182            shadow.thread.blocked,
1183            shadow.thread.clone(),
1184        )
1185    });
1186    let skip = |guest: &mut G, reason| {
1187        guest.thread_state_mut().delivery_counters.skip(reason);
1188        true
1189    };
1190    let (action, frame_mask, current_mask, mut entered) = match state {
1191        None if had_copy => return skip(guest, "copy given up at this delivery"),
1192        None => return skip(guest, "no copy"),
1193        Some((None, ..)) => return skip(guest, "dispositions unknown"),
1194        Some((Some(Err(_)), ..)) => return skip(guest, "invalid signal"),
1195        Some((Some(Ok(action)), frame, current, thread)) => (action, frame, current, thread),
1196    };
1197    // A force-delivered signal the copy holds blocked or ignored was
1198    // unblocked, and its action reset to the default, by the kernel before
1199    // this stop (`force_sig_info_to_task`), which the copy did not see: it is
1200    // not the copy's to check, and the copy no longer knows the action.
1201    // - Blocked: the kernel never dequeues a signal the thread blocks, so a
1202    //   delivery stop for one the copy blocks shows the forced unblock,
1203    //   whichever instance was dequeued (a pending `raise` of it, say).
1204    // - Ignored: only a signal the kernel generated (`si_code` > 0: a fault
1205    //   or SI_KERNEL) can have been forced; a guest's `kill` of the same
1206    //   number changed nothing. Without a siginfo nothing shows it was forced
1207    //   (and the backend cannot check the delivery either: `NoSignalRecord`).
1208    if forced() & sigbit(signal) != 0 {
1209        if current_mask & sigbit(signal) != 0 {
1210            forced_changed(guest, "signal delivery");
1211            return skip(guest, "forced while blocked");
1212        }
1213        let kernel_generated = guest
1214            .signal_info()
1215            .is_some_and(|info| i32::from_ne_bytes(info[8..12].try_into().unwrap()) > 0);
1216        if kernel_generated && action.is_ignored() {
1217            forced_changed(guest, "signal delivery");
1218            return skip(guest, "forced while ignored");
1219        }
1220    }
1221    if !action.is_caught() {
1222        return skip(guest, "not caught");
1223    }
1224    // The handler's mask as the copy will record it (`enter_handler`), which
1225    // the backend compares with Linux's rule over the same masks and action.
1226    if !entered.enter_handler(signal, &action) {
1227        // One frame more than the copy follows: recording drops the copy.
1228        return skip(guest, "too many frames");
1229    }
1230    let handler_mask = entered.blocked;
1231    // Force-delivered bits are exempt in each of the copy's masks that holds
1232    // them, and only there: a forced delivery the copy did not see can only
1233    // have taken such a bit out of the kernel's masks. Inside a
1234    // temporary-mask wait the two masks differ, and a forced bit in only one
1235    // of them stays checked in the other.
1236    let forced_frame = forced() & frame_mask & !sigbit(signal);
1237    let forced_current = forced() & current_mask & !sigbit(signal);
1238    let (interruption, from_registers, orig_ax) = delivery_interruption(guest, signal).await;
1239    let action = reverie::SignalAction {
1240        handler: action.handler,
1241        flags: action.flags,
1242        restorer: action.restorer,
1243        mask: action.mask,
1244    };
1245    let delivery = reverie::SignalDelivery {
1246        signal,
1247        action,
1248        frame_mask,
1249        current_mask,
1250        handler_mask,
1251        // Reverie's preemption signal, whose kernel state differs from the
1252        // copy's by design; and in each mask, the force-delivered signals
1253        // the copy blocks there: a forced delivery the copy did not see can
1254        // only have unblocked one (`compare`), so one the copy holds
1255        // unblocked stays checked. The delivered signal is always checked.
1256        unchecked_mask: reserved() & !sigbit(signal),
1257        unchecked_frame_mask: forced_frame,
1258        unchecked_current_mask: forced_current,
1259        interruption,
1260    };
1261    let result_checked = matches!(interruption, reverie::Interruption::Return { .. });
1262    let held = guest.signal_report_path() == Some(reverie::SignalReportPath::Held);
1263    let outcome = guest.deliver_signal(delivery).await;
1264    let not_delivered = matches!(
1265        outcome,
1266        reverie::SignalDeliveryOutcome::Unavailable(
1267            reverie::SignalDeliveryUnavailable::BlockedAtResume
1268                | reverie::SignalDeliveryUnavailable::NotADeliveryStop
1269        )
1270    );
1271    if !not_delivered {
1272        // The kernel builds this handler's frame as the guest resumes: a
1273        // further delivery before a guest instruction runs is at the handler's
1274        // first instruction.
1275        let state = guest.thread_state_mut();
1276        state.delivery_interruption = Some(reverie::Interruption::AfterFrame);
1277        state.delivery_frame_handler = action.handler;
1278        state.delivery_frame_signal = signal;
1279        state.delivery_frame_orig_ax = orig_ax;
1280    }
1281    let counters = &mut guest.thread_state_mut().delivery_counters;
1282    counters.interruption_from_registers += u64::from(from_registers);
1283    match outcome {
1284        reverie::SignalDeliveryOutcome::Accepted => {
1285            counters.approved += 1;
1286            counters.held += u64::from(held);
1287            counters.result_checked += u64::from(result_checked);
1288            trace!(
1289                target: "detcore::signal_delivery",
1290                "[dtid {}] signal-state copy: the delivery of signal {signal} is approved, \
1291                 interrupting {interruption:?}",
1292                guest.thread_state().dettid
1293            );
1294        }
1295        reverie::SignalDeliveryOutcome::Mismatch(mismatch) => {
1296            counters.mismatched += 1;
1297            let what = format!(
1298                "the kernel contradicts the delivery of signal {signal} in {:?}: the copy holds \
1299                 {:#x}, the kernel {:#x}",
1300                mismatch.field, mismatch.expected, mismatch.actual
1301            );
1302            if guest.config().signal_delivery_check_fatal {
1303                // A Detcore defect where both sides are certain: a strict run
1304                // ends rather than continue on the kernel's word alone.
1305                let failure = format!(
1306                    "HERMIT_INTERNAL_FAILURE class=signal-delivery-check dtid={} {what}",
1307                    guest.thread_state().dettid
1308                );
1309                tracing::error!("{failure}");
1310                panic!("{failure}");
1311            }
1312            diverged(guest, "signal delivery", &what);
1313        }
1314        reverie::SignalDeliveryOutcome::Unavailable(reason) => {
1315            counters.skip(unavailable_reason(reason));
1316        }
1317        reverie::SignalDeliveryOutcome::Refused(_) => counters.skip("backend refuses"),
1318        // Only a frame-building backend answers these, and no backend that
1319        // keeps a copy builds frames.
1320        reverie::SignalDeliveryOutcome::Installed | reverie::SignalDeliveryOutcome::FrameFault => {
1321            counters.skip("frame-building outcome");
1322        }
1323    }
1324    !not_delivered
1325}

1327/// At a delivery stop inside a temporary-mask wait, Linux holds one of two
1328/// masks: the wait's, when the kernel ran the call (the handler's frame then
1329/// saves the thread's own mask), or the thread's own, when the signal arrived
1330/// before the call started, which depends on host timing (see
1331/// `record_or_replay_rt_sigsuspend`). The kernel's current mask tells which;
1332/// any other mask is left for the checks that follow to report.
1333///
1334/// The masks are compared with the signals Linux can force-deliver left out
1335/// ([`forced`]), since a forced delivery the copy did not see can clear one.
1336/// When the wait's mask and the copy's differ only in such signals, that
1337/// comparison cannot tell them apart, and an exact one alone can mislead: a
1338/// forced clear only takes bits out, and can turn the copy's mask into the
1339/// wait's. Then the copy keeps its own mask if the kernel holds it exactly,
1340/// and takes the wait's if the kernel holds that exactly and either it has a
1341/// bit the copy's lacks (no clear adds one) or the call's raw result says the
1342/// kernel ran the wait ([`kernel_ran_the_wait`]), installing its mask over
1343/// any earlier clear. Any other subset of either mask gives the copy up
1344/// quietly, as a forced delivery's clear; a kernel mask with a bit neither
1345/// mask has is left to the checks that follow to report.
1346async fn take_wait_outcome<G, T>(guest: &mut G)
1347where
1348    G: Guest<Detcore<T>>,
1349    T: RecordOrReplay,
1350{
1351    let Some((wait_mask, saved, blocked)) =
1352        guest.thread_state().signal_shadow.as_ref().map(|shadow| {
1353            (
1354                shadow.thread.wait_mask,
1355                shadow.thread.saved,
1356                shadow.thread.blocked,
1357            )
1358        })
1359    else {
1360        return;
1361    };
1362    if (wait_mask.is_none() && saved.is_none()) || !kernel_state_is_the_guests(guest) {
1363        return;
1364    }
1365    let Some(kernel) = read_or_give_up(guest, "signal delivery") else {
1366        return;
1367    };
1368    let compared = !(reserved() | forced());
1369    let same = |a: KernelSigset, b: KernelSigset| a & compared == b & compared;
1370    let exact = |a: KernelSigset, b: KernelSigset| a & !reserved() == b & !reserved();
1371    // A temporary-mask restore no handler took has happened if the kernel
1372    // already shows the restored mask (the wait returned to the guest).
1373    let settled = match saved {
1374        Some(saved) if same(kernel.blocked, saved) => Some(saved),
1375        _ => None,
1376    };
1377    let current = settled.unwrap_or(blocked);
1378    let forced_only = |wait: KernelSigset| same(wait, current) && !exact(wait, current);
1379    let take_wait = match wait_mask {
1380        Some(wait) if forced_only(wait) && same(kernel.blocked, wait) => {
1381            // A forced delivery only takes bits out of the kernel's mask. So
1382            // the copy's own mask, held exactly, stays; and the wait's,
1383            // held exactly, was installed if it has a bit the copy's lacks,
1384            // and otherwise only if the call's raw result says the kernel
1385            // ran it. Any other subset of either mask can be a forced clear,
1386            // which gives the copy up; a kernel mask with a bit neither has
1387            // no forced delivery explains, and is left to the checks that
1388            // follow to report, as before.
1389            let subset = |a: KernelSigset, b: KernelSigset| a & !b & !reserved() == 0;
1390            if exact(kernel.blocked, current) {
1391                None
1392            } else if exact(kernel.blocked, wait)
1393                && (wait & !current & !reserved() != 0 || kernel_ran_the_wait(guest).await)
1394            {
1395                Some(wait)
1396            } else if subset(kernel.blocked, current) || subset(kernel.blocked, wait) {
1397                forced_changed(guest, "signal delivery");
1398                return;
1399            } else {
1400                None
1401            }
1402        }
1403        Some(wait) if same(kernel.blocked, wait) && !same(current, wait) => Some(wait),
1404        _ => None,
1405    };
1406    let Some(shadow) = guest.thread_state_mut().signal_shadow.as_mut() else {
1407        return;
1408    };
1409    if settled.is_some() {
1410        shadow.thread.settle();
1411    }
1412    if let Some(wait) = take_wait {
1413        shadow.thread.interrupted_under_temporary_mask(wait);
1414    }
1415    // The outcome was chosen with the forced signals left out. A forced
1416    // delivery's clear gives the copy up now; any other difference is left to
1417    // the next check, which reports it.
1418    if compare(kernel.blocked, shadow.thread.blocked) == Compared::ForcedOnly {
1419        forced_changed(guest, "signal delivery");
1420    }
1421}

1423/// Whether the kernel ran the temporary-mask wait this delivery stop ends,
1424/// and so installed its mask: the call's own raw result, before any
1425/// normalization, was not the not-run report ([`note_wait_ran`]; only
1426/// rt_sigsuspend records it today, so other waits never count as run here).
1427/// Two more conditions tie it to this stop: Detcore finished the call
1428/// interrupted (`-EINTR` or `-ERESTARTNOHAND`), and the stop's `orig_ax`
1429/// still names that call, so the thread has not run on.
1430async fn kernel_ran_the_wait<G, T>(guest: &mut G) -> bool
1431where
1432    G: Guest<Detcore<T>>,
1433    T: RecordOrReplay,
1434{
1435    const ERESTARTNOHAND: i64 = 514;
1436    let ran = guest
1437        .thread_state()
1438        .signal_shadow
1439        .as_ref()
1440        .and_then(|shadow| shadow.thread.wait_ran);
1441    if ran != Some(true) {
1442        return false;
1443    }
1444    let Some(reverie::Interruption::Return { syscall, result }) =
1445        guest.thread_state().delivery_interruption
1446    else {
1447        return false;
1448    };
1449    if result != -i64::from(libc::EINTR) && result != -ERESTARTNOHAND {
1450        return false;
1451    }
1452    #[cfg(target_arch = "x86_64")]
1453    {
1454        guest.regs().await.orig_rax as i64 == syscall
1455    }
1456    #[cfg(not(target_arch = "x86_64"))]
1457    {
1458        let _ = syscall;
1459        false
1460    }
1461}

1561/// The publication this thread has to make for its process
1562/// ([`publish_dispositions`]), with the next generation, taken under the
1563/// record's lock with the copy read under the same lock: the copy's
1564/// dispositions, or `None` to withdraw them, because it has no copy or the
1565/// shared actions became unknown. Nothing when the scheduler already holds
1566/// that for the process, and where no copy is kept.
1567fn next_publication<G, T>(guest: &G) -> Option<(u64, Option<CopiedDispositions>)>
1568where
1569    G: Guest<Detcore<T>>,
1570    T: RecordOrReplay,
1571{
1572    if !kernel_state_is_the_guests(guest) {
1573        return None;
1574    }
1575    let mut record = guest.thread_state().published_dispositions.lock().unwrap();
1576    let current = table_dispositions(guest.thread_state());
1577    if current == record.dispositions {
1578        return None;
1579    }
1580    record.generation += 1;
1581    Some((record.generation, current))
1582}

1584/// Applies the scheduler's `answer` to publication `generation` of `current`,
1585/// and returns whether to publish again at once.
1586/// - Taken, and no later answer was applied: the record takes its
1587///   dispositions.
1588/// - Not taken: the scheduler holds a generation at least as late, from a
1589///   later publication of this process or from an entry this process never
1590///   made. The record carries on after it, keeping its dispositions, and the
1591///   thread publishes again at once, before its request can let another
1592///   thread read the entry.
1593/// - Refused (`None`, a thread the scheduler does not know): nothing changes,
1594///   and the thread publishes again at its next request.
1595fn settle_publication(
1596    record: &mut PublishedDispositions,
1597    generation: u64,
1598    current: Option<CopiedDispositions>,
1599    answer: Option<PublicationAnswer>,
1600) -> bool {
1601    let Some(answer) = answer else {
1602        return false;
1603    };
1604    if answer.taken {
1605        if generation > record.accepted {
1606            record.accepted = generation;
1607            record.dispositions = current;
1608        }
1609        return false;
1610    }
1611    if answer.held > record.generation {
1612        record.generation = answer.held;
1613    }
1614    true
1615}

1617/// Publishes this thread's copied dispositions to the scheduler as its
1618/// process's (A5 step 2b), where they differ from what the scheduler holds
1619/// for it (`ThreadState::published_dispositions`), so that a decision about another thread reads them in
1620/// place of the kernel's `/proc` view (`Scheduler::published_dispositions_for`).
1621///
1622/// Called before each of the thread's requests that can let another thread
1623/// run (`tool_global::send_and_update_time`). Every change to the actions is
1624/// recorded in the operations of the thread that makes it, which publishes it
1625/// before its next request, so the scheduler holds the new actions before any
1626/// other thread can read them. Threads that do not hold the turn publish too
1627/// (a thread rejoining the schedule after a host call, a new thread's start,
1628/// a control message at a signal stop), so publications can arrive out of
1629/// order: each carries a generation ([`PublishedDispositions`]) and the
1630/// scheduler keeps the latest. A thread the scheduler has not registered yet
1631/// is refused, and publishes again at its next request
1632/// ([`settle_publication`]).
1633///
1634/// Every thread sharing the actions holds them behind one reference, so a
1635/// thread publishes them as they are now, whoever changed them. (After an
1636/// exec, the old image's other threads keep the old actions until the kernel
1637/// has killed them and their teardown runs; they make no request in
1638/// between.) The scheduler keys them by process, which is right because the
1639/// actions of a process are never shared with another one: [`before_clone`]
1640/// makes them unknown first.
1641///
1642/// That the scheduler's entry is never older than the copy rests on two
1643/// facts: where a copy is kept (ptrace), a publication is answered inline,
1644/// within the sender's own call, so it cannot be delayed past its sender;
1645/// and backends whose messages could be delayed keep no copy. The
1646/// generations, the scheduler's refusal of retired incarnations, and the
1647/// immediate republication after an answer that did not take it, guard
1648/// against an entry that should not be there all the same.
1649pub(crate) async fn publish_dispositions<G, T>(guest: &mut G)
1650where
1651    G: Guest<Detcore<T>>,
1652    T: RecordOrReplay,
1653{
1654    // A publication that was not taken is made again once, with a
1655    // generation past the one the scheduler holds, so it is taken.
1656    for _ in 0..2 {
1657        let Some((generation, current)) = next_publication(guest) else {
1658            return;
1659        };
1660        let answer =
1661            crate::tool_global::publish_signal_dispositions(guest, generation, current).await;
1662        trace!(
1663            "[dtid {}] signal-state copy: published its process's dispositions {:?} (generation {}, answer: {:?})",
1664            guest.thread_state().dettid,
1665            current,
1666            generation,
1667            answer
1668        );
1669        let mut record = guest.thread_state().published_dispositions.lock().unwrap();
1670        if !settle_publication(&mut record, generation, current, answer) {
1671            return;
1672        }
1673    }
1674}

1678/// unreported signal is noted, whose handler may have changed the actions

1681pub(crate) fn published_dispositions_decide(signal: i32) -> bool {

1685/// The mask this thread publishes to the scheduler (A5 step 2b,
1686/// [`publish_mask`]): the copy's `blocked`, where that is the mask the kernel
1687/// holds for the thread whenever it is stopped at a request. `None` (nothing
1688/// to publish) without a copy, and while it may not be:
1689/// - inside a temporary-mask wait (`wait_mask`), whose mask the kernel may
1690///   have installed, or with an interrupted wait's mask still to put back
1691///   (`saved`);
1692/// - before the next syscall entry verifies a mask a backend report set
1693///   (`verify_at_entry`);
1694/// - with a handler frame open where its return would go unseen
1695///   (`observes_returns` false), or whose entry no check has seen yet (the
1696///   kernel keeps the old mask when it cannot write the frame).
1697///
1698/// A mask Detcore installs privately is not this one either. While one is
1699/// installed the scheduler reads no published mask: discard at send, poll
1700/// admission and real-time timer fate take the guest's mask from
1701/// `Scheduler::saved_guest_sigmasks`, and the choice of a process-directed
1702/// signal's thread and the first SIGCHLD taker read the kernel's.
1703fn mask_to_publish<T>(
1704    thread: &crate::tool_local::ThreadState<T>,
1705    observes_returns: bool,
1706) -> Option<KernelSigset> {
1707    let masks = &thread.signal_shadow.as_ref()?.thread;
1708    let uncertain = masks.wait_mask.is_some()
1709        || masks.saved.is_some()
1710        || masks.verify_at_entry
1711        || (!masks.activations.is_empty() && !(observes_returns && masks.innermost_observed()));
1712    (!uncertain).then_some(masks.blocked)
1713}

1715/// Publishes this thread's mask from Detcore's copy to the scheduler
1716/// ([`mask_to_publish`]), or withdraws it, where that differs from what it
1717/// last had taken or what the scheduler holds is unknown
1718/// (`ThreadState::published_mask`), so that a decision about
1719/// this thread reads it in place of the kernel's `SigBlk`
1720/// (`Scheduler::published_mask_for`). Called with [`publish_dispositions`],
1721/// before each request that can let another thread run. A mask is the
1722/// thread's own and only the thread changes or publishes it, so its
1723/// publications cannot overtake each other. A thread the scheduler has not
1724/// registered yet is refused, and publishes again at its next request.
1725pub(crate) async fn publish_mask<G, T>(guest: &mut G)
1726where
1727    G: Guest<Detcore<T>>,
1728    T: RecordOrReplay,
1729{
1730    if !kernel_state_is_the_guests(guest) {
1731        return;
1732    }
1733    let current = mask_to_publish(guest.thread_state(), observes_sigreturns(guest.config()));
1734    if guest.thread_state().published_mask == Some(current) {
1735        return;
1736    }
1737    let taken = crate::tool_global::publish_signal_mask(guest, current).await;
1738    trace!(
1739        "[dtid {}] signal-state copy: published its mask {:?} (taken: {})",
1740        guest.thread_state().dettid,
1741        current.map(|mask| format!("{mask:#x}")),
1742        taken
1743    );
1744    if taken {
1745        guest.thread_state_mut().published_mask = Some(current);
1746    }
1747}

1750/// actions from another process (`CLONE_SIGHAND` without `CLONE_THREAD`)

1752/// dispositions per process, would not see it for this one. The actions
1753/// become unknown to every thread sharing them, and are withdrawn before the
1754/// child exists, also for a clone that then fails. Such a child gets no copy

1777/// (`Tool::observes_sigreturns`): where the copy is kept (serialized
1778/// threads), the backend can report them, no gdbserver is attached (the

1800/// `sizeof(struct rt_sigframe)` on x86-64: the return address (8), the
1801/// `ucontext` (304) and the `siginfo` (128).
1802const RT_SIGFRAME_SIZE: u64 = 8 + 304 + 128;
1803
1804/// The end of the x86-64 user address space under 4-level paging
1805/// (`TASK_SIZE_MAX`). Under 5-level paging it is higher.
1806const USER_ADDRESS_LIMIT_4_LEVEL: u64 = 0x7fff_ffff_f000;
1807
1808/// Records a guest `rt_sigreturn` the backend reported before it runs
1809/// (`Tool::handle_sigreturn`). x86-64 `rt_sigreturn` installs the frame's
1810/// mask once `access_ok` accepts the whole frame at `sp - 8` and the frame's
1811/// `uc_sigmask` and `uc_flags` read without a fault, before restoring the
1812/// registers can fail (`arch/x86/kernel/signal_64.c`): the innermost handler
1813/// frame then closes and the mask is installed. A frame that fails either
1814/// (an overflowing range, or an unmapped word) changes no mask; the kernel
1815/// then sends SIGSEGV. The copy gives itself up quietly where it cannot tell
1816/// what Linux does: a frame that reaches past 4-level paging's user limit,
1817/// which the paging mode accepts or not, and a protected frame, whose
1818/// protection decides whether the kernel's read faults. For every report
1819/// that keeps the copy, the next syscall entry compares the copy with the
1820/// kernel ([`reconcile_returns`]), so a report the kernel did not follow (a
1821/// protection key, say) fails closed there.
1822pub(crate) fn record_sigreturn_report<T>(
1823    thread: &mut crate::tool_local::ThreadState<T>,
1824    report: reverie::SigreturnReport,
1825) {
1826    let dettid = thread.dettid;
1827    if thread.signal_shadow.is_none() {
1828        return;
1829    }
1830    let frame_end = report
1831        .stack_pointer()
1832        .checked_sub(8)
1833        .and_then(|frame| frame.checked_add(RT_SIGFRAME_SIZE));
1834    let installed = match (report.frame(), frame_end) {
1835        (reverie::SigreturnFrame::Readable { mask }, Some(end))
1836            if end <= USER_ADDRESS_LIMIT_4_LEVEL =>
1837        {
1838            Some(mask)
1839        }
1840        (reverie::SigreturnFrame::Readable { .. }, Some(_)) => {
1841            debug!(
1842                "[dtid {dettid}] signal-state copy dropped: rt_sigreturn's frame reaches past \
1843                 the 4-level user address limit"
1844            );
1845            return drop_thread_copy(thread);
1846        }
1847        (reverie::SigreturnFrame::Protected, Some(_)) => {
1848            debug!(
1849                "[dtid {dettid}] signal-state copy dropped: rt_sigreturn's frame is mapped but \
1850                 protected from the guest"
1851            );
1852            return drop_thread_copy(thread);
1853        }
1854        _ => None,
1855    };
1856    let Some(shadow) = thread.signal_shadow.as_mut() else {
1857        return;
1858    };
1859    shadow.thread.verify_at_entry = true;
1860    match installed {
1861        Some(mask) => {
1862            shadow.thread.sigreturn(mask & !reserved());
1863            #[cfg(test)]
1864            REPORTED_RETURNS.with(|count| count.set(count.get() + 1));
1865            trace!(
1866                "[dtid {dettid}] signal-state copy: rt_sigreturn reported, installing {:#x}",
1867                shadow.thread.blocked
1868            );
1869        }
1870        None => {
1871            trace!("[dtid {dettid}] signal-state copy: rt_sigreturn's frame faults, no mask change")
1872        }
1873    }
1874}

1876/// For a backend that shows `rt_sigreturn` as a syscall, which none does
1877/// today; the ptrace backend reports it through [`record_sigreturn_report`].
1878///
1879/// Records an `rt_sigreturn` at its entry, before it runs: Linux installs the
1880/// frame's `uc_sigmask` as soon as it can read it, before restoring the
1881/// registers can fail (`arch/x86/kernel/signal_64.c`). A frame whose mask
1882/// cannot be read changes no mask; the kernel then sends SIGSEGV.
1883pub(crate) async fn record_rt_sigreturn<G, T>(guest: &mut G)
1884where
1885    G: Guest<Detcore<T>>,
1886    T: RecordOrReplay,
1887{
1888    if guest.thread_state().signal_shadow.is_none() {
1889        return;
1890    }
1891    let sp = crate::arch::stack_pointer(&guest.regs().await) as usize;
1892    let restored = Addr::<u64>::from_raw(sp.wrapping_add(super::signal::UC_SIGMASK_OFFSET))
1893        .and_then(|address| guest.memory().read_value(address).ok());
1894    if let Some(mask) = restored {
1895        with_copy(guest, |_, thread| thread.sigreturn(mask & !reserved()));
1896    }
1897}

1899/// Compares the copy with the kernel's `/proc` view of this thread, where that
1900/// view is the guest's own.
1901fn check_against_kernel<G, T>(guest: &mut G, after: &str)
1902where
1903    G: Guest<Detcore<T>>,
1904    T: RecordOrReplay,
1905{
1906    take_unreported(guest);
1907    if !kernel_state_is_the_guests(guest) || guest.thread_state().signal_shadow.is_none() {
1908        return;
1909    }
1910    let Some(kernel) = read_or_give_up(guest, after) else {
1911        return;
1912    };
1913    let Some(copy) = with_copy(guest, |dispositions, thread| {
1914        (
1915            thread.blocked,
1916            dispositions.ignored(),
1917            dispositions.caught(),
1918        )
1919    }) else {
1920        return;
1921    };
1922    let (blocked, ignored, caught) = copy;
1923    // A sibling's unreported handler may have changed the shared actions
1924    // unseen; that sibling gives the copy up at its own next operation.
1925    let shared_unsure = unreported_elsewhere(guest.tid().as_raw());
1926    // Linux resets a caught action only for a signal that was blocked when it
1927    // was forced. That mask is not this check's: the guest may have blocked or
1928    // unblocked the signal since, unseen by the reset, and a sibling sharing
1929    // the actions may have taken it. So a caught bit the kernel cleared counts
1930    // as forced on its direction alone (`compare`).
1931    let fields = [
1932        ("SigBlk", kernel.blocked, blocked, true),
1933        ("SigIgn", kernel.ignored, ignored, !shared_unsure),
1934        ("SigCgt", kernel.caught, caught, !shared_unsure),
1935    ];
1936    let mut forced_only = false;
1937    let mut mismatches = Vec::new();
1938    for (field, kernel, copy, checked) in fields {
1939        if !checked {
1940            continue;
1941        }
1942        match compare(kernel, copy) {
1943            Compared::Same => {}
1944            Compared::ForcedOnly => forced_only = true,
1945            Compared::Different => mismatches.push(format!(
1946                "{field} kernel {:#018x} copy {:#018x}",
1947                kernel & !reserved(),
1948                copy & !reserved()
1949            )),
1950        }
1951    }
1952    if !mismatches.is_empty() {
1953        diverged(guest, after, &mismatches.join(", "));
1954    } else if forced_only {
1955        forced_changed(guest, after);
1956    } else if shared_unsure {
1957        trace!(
1958            "[dtid {}] signal-state copy matches the kernel's mask at {after} (actions not \
1959             compared: a sibling's signal was passed on unreported)",
1960            guest.thread_state().dettid
1961        );
1962    } else {
1963        if let Some(shadow) = guest.thread_state_mut().signal_shadow.as_mut() {
1964            shadow.thread.observe_handler_mask();
1965        }
1966        #[cfg(test)]
1967        CHECKS_MATCHED.with(|count| count.set(count.get() + 1));
1968        trace!(
1969            "[dtid {}] signal-state copy matches the kernel at {after}",
1970            guest.thread_state().dettid
1971        );
1972    }
1973}
