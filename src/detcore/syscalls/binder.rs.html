9//! Descriptors a guest receives in a binder transaction.
10//!
11//! A binder transaction can carry file descriptors: a `BINDER_TYPE_FD`
12//! object, or a `BINDER_TYPE_FDA` array inside a buffer object. Linux opens
13//! each one in the receiving process when that process reads the transaction
14//! with `ioctl(BINDER_WRITE_READ)`, and writes the new numbers into the
15//! transaction's buffer in the receiver's binder mapping
16//! (`binder_apply_fd_fixups`, always with `O_CLOEXEC`). No call that names
17//! them as new descriptors ever reaches Detcore, as with `SCM_RIGHTS`
18//! (`crate::syscalls::scm_rights`): on a backend where the tool keeps its own
19//! descriptor table (ptrace, KVM), every later call through one of them that
20//! consults that table failed with `EBADF`.
21//!
22//! So after a `BINDER_WRITE_READ` that read something, Detcore walks the
23//! return commands Linux wrote (`read_buffer[..read_consumed]`), and for
24//! each delivered transaction (`BR_TRANSACTION`, `BR_TRANSACTION_SEC_CTX`,
25//! `BR_REPLY`) reads its offsets array and objects from the receiver's
26//! mapping and names the descriptors they now hold. They are then recorded as
27//! received `SCM_RIGHTS` descriptors are
28//! (`Detcore::record_received_descriptors`), from the kernel's own facts.
29//!
30//! Layouts are the 64-bit ABI of `include/uapi/linux/android/binder.h`, the
31//! only one an arm64 or x86-64 kernel builds (`binder_uintptr_t` and
32//! `binder_size_t` are 64 bits). Nothing here changes what the guest sees:
33//! the call has already returned, and its memory is only read. A buffer that
34//! cannot be read or parsed names no descriptor, and says why.
35

43/// `BINDER_WRITE_READ`: `_IOWR('b', 1, struct binder_write_read)`, whose
44/// struct is six 64-bit words.
45pub(crate) const BINDER_WRITE_READ: u64 = 0xc030_6201;
46
47/// `_IOR('r', 2, struct binder_transaction_data)`.
48const BR_TRANSACTION: u32 = 0x8040_7202;
49/// `_IOR('r', 2, struct binder_transaction_data_secctx)`.
50const BR_TRANSACTION_SEC_CTX: u32 = 0x8048_7202;
51/// `_IOR('r', 3, struct binder_transaction_data)`.
52const BR_REPLY: u32 = 0x8040_7203;
53
54/// `TF_STATUS_CODE`: the transaction's data is a 32-bit status. Linux does
55/// not read it when it installs descriptors; the tests use it.
56#[cfg(test)]
57const TF_STATUS_CODE: u32 = 0x08;
58
59/// `B_PACK_CHARS('p', 't', '*', B_TYPE_LARGE)`, a buffer object: a
60/// descriptor array's parent, which the tests build.
61#[cfg(test)]
62const BINDER_TYPE_PTR: u32 = 0x7074_2a85;
63
64/// `B_PACK_CHARS('f', 'd', '*', B_TYPE_LARGE)`.
65const BINDER_TYPE_FD: u32 = 0x6664_2a85;
66/// `B_PACK_CHARS('f', 'd', 'a', B_TYPE_LARGE)`.
67const BINDER_TYPE_FDA: u32 = 0x6664_6185;

69/// The most return-command bytes read from one call: Linux writes at most
70/// the caller's `read_size`, and libbinder reads 256 bytes at a time.
71const MAX_COMMAND_BYTES: usize = 64 * 1024;
72/// The most objects read from one transaction's offsets array.
73const MAX_OBJECTS: usize = 4096;
74
75/// Whether `request`, an `ioctl` request word, is `BINDER_WRITE_READ`.
76/// Only the low 32 bits are the request: the kernel's `cmd` is an
77/// `unsigned int`.
78pub(crate) fn is_write_read(request: u64) -> bool {
79    request & 0xffff_ffff == BINDER_WRITE_READ
80}

82/// The `read_consumed` of the `struct binder_write_read` at `argument`,
83/// read before the call: the bytes of `read_buffer` an earlier call wrote,
84/// which this one leaves alone. Zero where it cannot be read, so the whole
85/// buffer is parsed, as before.
86pub(crate) fn consumed_before<M: MemoryAccess>(memory: &M, argument: usize) -> u64 {
87    Addr::<[u64; 6]>::from_raw(argument)
88        .and_then(|address| memory.read_value(address).ok())
89        .map_or(0, |words: [u64; 6]| words[4])
90}

92/// The descriptors a successful `BINDER_WRITE_READ`, whose
93/// `struct binder_write_read` is at `argument`, made Linux install in the
94/// caller, read from guest memory after the call. `already_consumed` is the
95/// `read_consumed` the caller passed in, read before the call: Linux writes
96/// this call's commands after it.
97pub(crate) fn delivered<M: MemoryAccess>(
98    memory: &M,
99    argument: usize,
100    already_consumed: u64,
101) -> Delivered {
102    let Some(address) = Addr::<[u64; 6]>::from_raw(argument) else {
103        return refused("its binder_write_read is at address 0");
104    };
105    let Ok(words) = memory.read_value(address) else {
106        return refused("its binder_write_read cannot be read");
107    };
108    // write_size, write_consumed, write_buffer, read_size, read_consumed,
109    // read_buffer.
110    let [_, _, _, read_size, read_consumed, read_buffer] = words;
111    if read_consumed == 0 || read_consumed == already_consumed {
112        return Delivered::default();
113    }
114    if read_consumed > read_size || already_consumed > read_consumed {
115        return refused("its read_consumed is not within its read_size");
116    }
117    // Linux writes this call's commands after the read_consumed the caller
118    // passed in, leaving any earlier ones alone: those are an older call's,
119    // and their buffers may already be freed.
120    let newly_consumed = read_consumed - already_consumed;
121    if newly_consumed > MAX_COMMAND_BYTES as u64 {
122        return refused("its newly returned commands exceed Detcore's read limit");
123    }
124    let length = newly_consumed as usize;
125    let Some(address) = Addr::<u8>::from_raw((read_buffer + already_consumed) as usize) else {
126        return refused("its read_buffer is at address 0");
127    };
128    let mut commands = vec![0; length];
129    if memory.read_exact(address, &mut commands).is_err() {
130        return refused("its return commands cannot be read");
131    }
132    match installed_descriptors(memory, &commands) {
133        Ok(fds) => Delivered {
134            // One transaction: its descriptors arrive as one message.
135            messages: vec![crate::syscalls::scm_rights::DeliveredMessage {
136                count: fds.len(),
137                truncated: false,
138            }],
139            fds,
140            refused: None,
141        },
142        Err(why) => refused(why),
143    }
144}
145
146fn refused(why: &'static str) -> Delivered {
147    Delivered {
148        fds: Vec::new(),
149        messages: Vec::new(),
150        refused: Some(why),
151    }
152}

154/// `_IOC_SIZE(cmd)`: the payload size a return command carries after its
155/// 32-bit code.
156fn payload_size(cmd: u32) -> usize {
157    ((cmd >> 16) & 0x3fff) as usize
158}
159
160fn u32_at(bytes: &[u8], offset: usize) -> Option<u32> {
161    Some(u32::from_ne_bytes(
162        bytes.get(offset..offset + 4)?.try_into().ok()?,
163    ))
164}
165
166fn u64_at(bytes: &[u8], offset: usize) -> Option<u64> {
167    Some(u64::from_ne_bytes(
168        bytes.get(offset..offset + 8)?.try_into().ok()?,
169    ))
170}

172/// The descriptors the transactions in `commands`, a run of return commands
173/// as Linux wrote them, carry, in order.
174fn installed_descriptors<M: MemoryAccess>(
175    memory: &M,
176    commands: &[u8],
177) -> Result<Vec<RawFd>, &'static str> {
178    let mut fds = Vec::new();
179    let mut offset = 0;
180    while offset < commands.len() {
181        let cmd = u32_at(commands, offset).ok_or("a return command is cut short")?;
182        let payload = commands
183            .get(offset + 4..offset + 4 + payload_size(cmd))
184            .ok_or("a return command's payload is cut short")?;
185        if matches!(cmd, BR_TRANSACTION | BR_TRANSACTION_SEC_CTX | BR_REPLY) {
186            fds.extend(transaction_descriptors(memory, payload)?);
187        }
188        offset += 4 + payload.len();
189    }
190    Ok(fds)
191}

193/// The descriptors the transaction `data`, a `struct binder_transaction_data`
194/// (the head of a `_secctx` one too), carries.
195fn transaction_descriptors<M: MemoryAccess>(
196    memory: &M,
197    data: &[u8],
198) -> Result<Vec<RawFd>, &'static str> {
199    let short = "a transaction is cut short";
200    // The driver never reads TF_STATUS_CODE when it installs descriptors, so
201    // neither does this: a real status reply carries no objects anyway, and
202    // skipping on the flag would hide a descriptor Linux did install.
203    let offsets_size = u64_at(data, 40).ok_or(short)? as usize;
204    let buffer = u64_at(data, 48).ok_or(short)? as usize;
205    let offsets_at = u64_at(data, 56).ok_or(short)? as usize;
206    let count = offsets_size / 8;
207    if count == 0 {
208        return Ok(Vec::new());
209    }
210    if count > MAX_OBJECTS {
211        return Err("a transaction carries more objects than Detcore reads");
212    }
213    let mut offsets = vec![0u64; count];
214    let address = Addr::<u64>::from_raw(offsets_at).ok_or("a transaction's offsets are at 0")?;
215    memory
216        .read_values(address, &mut offsets)
217        .map_err(|_| "a transaction's offsets cannot be read")?;
218    let object = |index: usize, length: usize| -> Result<Vec<u8>, &'static str> {
219        let at = offsets
220            .get(index)
221            .ok_or("an object names a parent that is not in the transaction")?;
222        let address = Addr::<u8>::from_raw(buffer.wrapping_add(*at as usize))
223            .ok_or("a transaction's object is at 0")?;
224        let mut bytes = vec![0; length];
225        memory
226            .read_exact(address, &mut bytes)
227            .map_err(|_| "a transaction's object cannot be read")?;
228        Ok(bytes)
229    };
230    let mut fds = Vec::new();
231    for index in 0..count {
232        // Every object starts with its 32-bit type.
233        let header = object(index, 4)?;
234        match u32_at(&header, 0) {
235            // struct binder_fd_object: the type, a 32-bit pad, then the
236            // descriptor in the low half of a 64-bit union.
237            Some(BINDER_TYPE_FD) => {
238                let fd_object = object(index, 24)?;
239                fds.push(u32_at(&fd_object, 8).ok_or(short)? as RawFd);
240            }
241            // A BINDER_TYPE_FDA array's descriptors are the kernel's to
242            // close: it closes them itself when the receiver frees the
243            // transaction's buffer with BC_FREE_BUFFER, so a record of one
244            // would outlive it and describe whatever next takes the number.
245            // Left out until Detcore follows those frees
246            // (https://github.com/rrnewton/hermit/issues/ISSUE-FDA).
247            Some(BINDER_TYPE_FDA) => {
248                let array = object(index, 32)?;
249                let num_fds = u64_at(&array, 8).ok_or(short)?;
250                if num_fds != 0 {
251                    return Err(
252                        "a transaction carries a descriptor array, whose descriptors Linux \
253                         closes with the buffer",
254                    );
255                }
256            }
257            Some(_) => {}
258            None => return Err(short),
259        }
260    }
261    Ok(fds)
262}

270    /// A transaction's bytes, as a `struct binder_transaction_data`.
271    fn transaction(flags: u32, offsets: &[u64], buffer: usize) -> Vec<u8> {
272        let mut data = vec![0u8; 64];
273        data[20..24].copy_from_slice(&flags.to_ne_bytes());
274        data[40..48].copy_from_slice(&((offsets.len() * 8) as u64).to_ne_bytes());
275        data[48..56].copy_from_slice(&(buffer as u64).to_ne_bytes());
276        data[56..64].copy_from_slice(&(offsets.as_ptr() as u64).to_ne_bytes());
277        data
278    }

280    fn command(cmd: u32, payload: &[u8]) -> Vec<u8> {
281        let mut bytes = cmd.to_ne_bytes().to_vec();
282        bytes.extend_from_slice(payload);
283        bytes
284    }

286    fn fd_object(fd: u32) -> Vec<u8> {
287        let mut object = vec![0u8; 24];
288        object[0..4].copy_from_slice(&BINDER_TYPE_FD.to_ne_bytes());
289        object[8..12].copy_from_slice(&fd.to_ne_bytes());
290        object
291    }

293    /// The command codes are the uapi header's, from their `_IOR` parts.
294    #[test]
295    fn the_command_codes_encode_their_payload_sizes() {
296        let ior = |nr: u32, size: u32| (2 << 30) | (size << 16) | (u32::from(b'r') << 8) | nr;
297        assert_eq!(BR_TRANSACTION, ior(2, 64));
298        assert_eq!(BR_TRANSACTION_SEC_CTX, ior(2, 72));
299        assert_eq!(BR_REPLY, ior(3, 64));
300        let iowr = (3u64 << 30) | (48 << 16) | (u64::from(b'b') << 8) | 1;
301        assert_eq!(BINDER_WRITE_READ, iowr);
302        assert!(is_write_read(BINDER_WRITE_READ | 0xffff_ffff_0000_0000));
303        assert!(!is_write_read(0x4004_6205));
304    }

306    /// A reply carrying one descriptor object, after commands without
307    /// payload of interest, names it; a status-code reply names none.
308    #[test]
309    fn a_reply_names_the_descriptors_its_objects_hold() {
310        let mut buffer = [0u8; 64];
311        buffer[16..40].copy_from_slice(&fd_object(9));
312        let offsets = [16u64];
313        let noop = 0x0000_720c; // BR_NOOP, _IO('r', 12)
314        let complete = 0x0000_7206; // BR_TRANSACTION_COMPLETE, _IO('r', 6)
315        let mut commands = command(noop, &[]);
316        commands.extend(command(complete, &[]));
317        commands.extend(command(
318            BR_REPLY,
319            &transaction(0, &offsets, buffer.as_ptr() as usize),
320        ));
321        assert_eq!(
322            installed_descriptors(&LocalMemory::new(), &commands),
323            Ok(vec![9])
324        );
325
326        // A real status reply carries no objects. The flag itself is not
327        // read, because the driver does not read it when it installs
328        // descriptors: a reply that carries one still names it.
329        let status = command(BR_REPLY, &transaction(TF_STATUS_CODE, &[], 0));
330        assert_eq!(
331            installed_descriptors(&LocalMemory::new(), &status),
332            Ok(vec![])
333        );
334        let flagged = command(
335            BR_REPLY,
336            &transaction(TF_STATUS_CODE, &offsets, buffer.as_ptr() as usize),
337        );
338        assert_eq!(
339            installed_descriptors(&LocalMemory::new(), &flagged),
340            Ok(vec![9])
341        );
342    }

344    /// A descriptor array is refused: Linux closes its numbers with the
345    /// transaction's buffer, so a record of one would outlive it.
346    #[test]
347    fn a_descriptor_array_is_refused_rather_than_recorded() {
348        let numbers: [u32; 3] = [4, 5, 6];
349        let parent_data = {
350            let mut data = vec![0u8; 8];
351            numbers
352                .iter()
353                .for_each(|fd| data.extend_from_slice(&fd.to_ne_bytes()));
354            data
355        };
356        let mut buffer = [0u8; 128];
357        // The parent buffer object at 0.
358        buffer[0..4].copy_from_slice(&BINDER_TYPE_PTR.to_ne_bytes());
359        buffer[8..16].copy_from_slice(&(parent_data.as_ptr() as u64).to_ne_bytes());
360        buffer[16..24].copy_from_slice(&(parent_data.len() as u64).to_ne_bytes());
361        // The array object at 40: three descriptors at offset 8 of parent 0.
362        buffer[40..44].copy_from_slice(&BINDER_TYPE_FDA.to_ne_bytes());
363        buffer[48..56].copy_from_slice(&3u64.to_ne_bytes());
364        buffer[56..64].copy_from_slice(&0u64.to_ne_bytes());
365        buffer[64..72].copy_from_slice(&8u64.to_ne_bytes());
366        let offsets = [0u64, 40];
367        let commands = command(
368            BR_TRANSACTION_SEC_CTX,
369            &[
370                transaction(0, &offsets, buffer.as_ptr() as usize),
371                vec![0u8; 8],
372            ]
373            .concat(),
374        );
375        let refusal =
376            installed_descriptors(&LocalMemory::new(), &commands).expect_err("a descriptor array");
377        assert!(refusal.contains("Linux closes"), "{refusal}");
378        // An empty array names nothing and is not refused.
379        buffer[48..56].copy_from_slice(&0u64.to_ne_bytes());
380        let commands = command(
381            BR_TRANSACTION_SEC_CTX,
382            &[
383                transaction(0, &offsets, buffer.as_ptr() as usize),
384                vec![0u8; 8],
385            ]
386            .concat(),
387        );
388        assert_eq!(
389            installed_descriptors(&LocalMemory::new(), &commands),
390            Ok(vec![])
391        );
392    }

402    /// The whole call: a read that consumed nothing names nothing, one that
403    /// claims more than its size is refused, and one with a transaction
404    /// names its descriptors.
405    #[test]
406    fn a_write_read_names_what_its_read_delivered() {
407        let mut buffer = [0u8; 64];
408        buffer[0..24].copy_from_slice(&fd_object(12));
409        let offsets = [0u64];
410        let commands = command(
411            BR_TRANSACTION,
412            &transaction(0, &offsets, buffer.as_ptr() as usize),
413        );
414        let call = |read_size: u64, read_consumed: u64| {
415            [0, 0, 0, read_size, read_consumed, commands.as_ptr() as u64]
416        };
417        let memory = LocalMemory::new();
418        let words = call(256, commands.len() as u64);
419
420        assert_eq!(
421            delivered(&memory, words.as_ptr() as usize, 0),
422            Delivered {
423                fds: vec![12],
424                messages: vec![crate::syscalls::scm_rights::DeliveredMessage {
425                    count: 1,
426                    truncated: false,
427                }],
428                refused: None
429            }
430        );
431        let words = call(256, 0);
432        assert_eq!(
433            delivered(&memory, words.as_ptr() as usize, 0),
434            Delivered::default()
435        );
436        let words = call(4, commands.len() as u64);
437        assert!(
438            delivered(&memory, words.as_ptr() as usize, 0)
439                .refused
440                .is_some()
441        );
442
443        // An exactly full command buffer is parsed through its last FD.
444        // Cutting a larger one on a command boundary must refuse explicitly:
445        // otherwise its valid completion prefix would hide the installed FD.
446        let noop = 0x0000_720cu32; // BR_NOOP, _IO('r', 12)
447        let complete = 0x0000_7206u32; // BR_TRANSACTION_COMPLETE, _IO('r', 6)
448        let mut at_limit = noop.to_ne_bytes().to_vec();
449        at_limit.extend(
450            complete
451                .to_ne_bytes()
452                .repeat((MAX_COMMAND_BYTES - commands.len() - 4) / 4),
453        );
454        at_limit.extend_from_slice(&commands);
455        assert_eq!(at_limit.len(), MAX_COMMAND_BYTES);
456        let words = [
457            0,
458            0,
459            0,
460            at_limit.len() as u64,
461            at_limit.len() as u64,
462            at_limit.as_ptr() as u64,
463        ];
464        let expected = delivered(&memory, words.as_ptr() as usize, 0);
465        assert_eq!(expected.fds, vec![12]);
466        assert!(expected.refused.is_none());
467
468        let mut over_limit = noop.to_ne_bytes().to_vec();
469        over_limit.extend(complete.to_ne_bytes().repeat(MAX_COMMAND_BYTES / 4 - 1));
470        over_limit.extend_from_slice(&commands);
471        let words = [
472            0,
473            0,
474            0,
475            over_limit.len() as u64,
476            over_limit.len() as u64,
477            over_limit.as_ptr() as u64,
478        ];
479        assert_eq!(
480            delivered(&memory, words.as_ptr() as usize, 0),
481            refused("its newly returned commands exceed Detcore's read limit")
482        );
483        // Old commands do not consume this call's budget or get parsed again.
484        // Only the small FD transaction after the large old prefix is new.
485        assert_eq!(
486            consumed_before(
487                &memory,
488                [0, 0, 0, 0, MAX_COMMAND_BYTES as u64, 0].as_ptr() as usize
489            ),
490            MAX_COMMAND_BYTES as u64
491        );
492        assert_eq!(
493            delivered(&memory, words.as_ptr() as usize, MAX_COMMAND_BYTES as u64),
494            expected
495        );
496    }
