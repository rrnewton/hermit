9//! Descriptors a guest receives in a binder transaction.
10//!
11//! A binder transaction can carry file descriptors: a `BINDER_TYPE_FD`
12//! object, or a `BINDER_TYPE_FDA` array inside a buffer object. Linux opens
13//! each one in the receiving process when that process reads the transaction
14//! with `ioctl(BINDER_WRITE_READ)`, and writes the new numbers into the
15//! transaction's buffer in the receiver's binder mapping
16//! (`binder_apply_fd_fixups`, always with `O_CLOEXEC`). No call that names
17//! them as new descriptors ever reaches Detcore, as with `SCM_RIGHTS`
18//! (`crate::syscalls::scm_rights`): on a backend where the tool keeps its own
19//! descriptor table (ptrace, KVM), every later call through one of them that
20//! consults that table failed with `EBADF`.
21//!
22//! So after a `BINDER_WRITE_READ` that read something, Detcore walks the
23//! return commands Linux wrote (`read_buffer[..read_consumed]`), and for
24//! each delivered transaction (`BR_TRANSACTION`, `BR_TRANSACTION_SEC_CTX`,
25//! `BR_REPLY`) reads its offsets array and objects from the receiver's
26//! mapping and names the descriptors they now hold. They are then recorded as
27//! received `SCM_RIGHTS` descriptors are
28//! (`Detcore::record_received_descriptors`), from the kernel's own facts.
29//!
30//! Layouts are the 64-bit ABI of `include/uapi/linux/android/binder.h`, the
31//! only one an arm64 or x86-64 kernel builds (`binder_uintptr_t` and
32//! `binder_size_t` are 64 bits). Nothing here changes what the guest sees:
33//! the call has already returned, and its memory is only read. A buffer that
34//! cannot be read or parsed names no descriptor, and says why.

111/// Every request `binder_ioctl` serves at v6.1 (drivers/android/binder.c),
112/// each with its argument's size as `include/uapi/linux/android/binder.h`
113/// spells it: `BINDER_WRITE_READ`, `SET_MAX_THREADS`, `SET_CONTEXT_MGR`,
114/// `THREAD_EXIT`, `VERSION`, `GET_NODE_DEBUG_INFO`, `GET_NODE_INFO_FOR_REF`,
115/// `SET_CONTEXT_MGR_EXT`, `FREEZE`, `GET_FROZEN_INFO`,
116/// `ENABLE_ONEWAY_SPAM_DETECTION` and `GET_EXTENDED_ERROR`. The header's
117/// `SET_IDLE_TIMEOUT` and `SET_IDLE_PRIORITY` are left out: the driver
118/// answers them `EINVAL` like any unknown request, and the first is the very
119/// word of dma-buf's `DMA_BUF_IOCTL_IMPORT_SYNC_FILE`. The type byte `'b'`
120/// is shared with other drivers, dma-buf among them, so a request counts as
121/// binder's only when number, direction and size all match one of these;
122/// the descriptor is not consulted, as for `BINDER_WRITE_READ` elsewhere in
123/// this module.

125    BINDER_WRITE_READ,
126    binder_request(1, 5, size_of::<u32>()),
127    BINDER_SET_CONTEXT_MGR,
128    binder_request(1, 8, size_of::<i32>()),
129    binder_request(3, 9, size_of::<i32>()),
130    binder_request(3, 11, 24),
131    binder_request(3, 12, 24),
132    BINDER_SET_CONTEXT_MGR_EXT,
133    BINDER_FREEZE,
134    binder_request(3, 15, 12),
135    binder_request(1, 16, size_of::<u32>()),
136    binder_request(3, 17, 12),
137];

139/// Whether `request` is one of binder's own `ioctl` requests.

167/// `--strict`'s binder policy (`Config::strict_binder`) for one `ioctl`
168/// request: why the run must stop, or `None` for a request that is not
169/// binder's.
170///
171/// A `--strict` run never holds a binder descriptor: one is refused where it
172/// arrives (by an open, a receipt, or at the run's start), by the open file's
173/// identity ([`identity`]). This request-word refusal is the backstop for a
174/// descriptor whose identity is binder or cannot be told
175/// (`Detcore::binder_entry`); a request on a descriptor known not to be a
176/// binder device is left alone, as the same words name other drivers'
177/// requests.
178///
179/// Every binder request is refused, in a fresh run and in a replay alike.
180/// What a binder call returns, and when, is another process's to decide:
181/// the transactions, replies, reference requests and notices a read takes
182/// are bytes and timing from processes Hermit does not schedule, and Hermit
183/// records none of them, so no run, fresh or replayed, can promise to repeat
184/// a binder call until it does
185/// (<https://github.com/rrnewton/hermit/issues/5124>). A replay's record of
186/// where each backgrounded call rejoined fixes when the call is readmitted,
187/// not what it read. The decision reads only the request word, so it comes
188/// before Detcore waits for anything or reads the call's argument
189/// (`Detcore::binder_entry`).
190pub(crate) fn strict_refusal(request: u64) -> Option<String> {
191    if !is_binder_request(request) {
192        return None;
193    }
194    Some(format!(
195        "--strict refuses binder ({}): binder reply payloads are not recorded, so no run, fresh \
196         or replayed, can repeat what a binder call returns \
197         (https://github.com/rrnewton/hermit/issues/5124). Run a guest that uses binder without \
198         --strict, where each binder call that waits for another process is reported as \
199         external input",
200        request_name(request)
201    ))
202}

218/// tells a binderfs device by it (`is_binderfs_device`,

223/// The misc character devices' major number. Without binderfs, Linux
224/// registers each binder device that binder's `devices` parameter names as
225/// a misc device (`binder_init`, drivers/android/binder.c:7068 at v6.16, and
226/// `init_binder_device`, :7029), which any node with its number opens.
227const MISC_MAJOR: u32 = 10;

229/// The misc minors no kernel hands out dynamically: binder asks for
230/// `MISC_DYNAMIC_MINOR` (binder.c:7021), which Linux v6.16 allocates above
231/// 255 (`misc_minor_alloc`, drivers/char/misc.c:65-75) and earlier kernels
232/// allocated below 128. A minor in this range is one of the fixed
233/// assignments of include/linux/miscdevice.h (`FUSE_MINOR` 229, `TUN_MINOR`
234/// 200, `KVM_MINOR` 232, …), so its driver is known and is not binder.
235const STATIC_MISC_MINORS: std::ops::RangeInclusive<u32> = 128..=254;

251/// Whether the file a descriptor refers to is a binder device, told by the
252/// open file itself, never by a path:
253/// - `stat`: its mode and device number (`fstat`), or `None` when it cannot
254///   be read;
255/// - `file_system`: its file system's magic (`fstatfs`), or `None`;
256/// - `misc`: the contents of `/proc/misc`, which names each misc device by
257///   its minor (`misc_seq_show`, drivers/char/misc.c:101 at v6.16), or
258///   `None`;
259/// - `devices`: the binder driver's `devices` parameter
260///   ([`binder_devices_parameter`]), or `None` when none can be read.
261///
262/// A character device that is not a misc device is never binder's: binder
263/// registers no other (/dev/null, /dev/urandom, terminals and
264/// pseudo-terminals are not). Nor is a misc device with a fixed minor
265/// ([`STATIC_MISC_MINORS`]). A misc device with a dynamic minor is binder's
266/// when the driver's parameter names it; when `/proc/misc` or the parameter
267/// cannot be read, or does not list the minor, it is unknown. Each lookup is
268/// made only when the answer needs it.
269pub(crate) fn identity(
270    stat: Option<(u32, u64)>,
271    file_system: impl FnOnce() -> Option<i64>,
272    misc: impl FnOnce() -> Option<String>,
273    devices: impl FnOnce() -> Option<String>,
274) -> Identity {
275    let Some((mode, rdev)) = stat else {
276        return Identity::Unknown("its file cannot be inspected".into());
277    };
278    if mode & libc::S_IFMT != libc::S_IFCHR {
279        return Identity::NotBinder;
280    }
281    match file_system() {
282        Some(BINDERFS_SUPER_MAGIC) => return Identity::Binder("a binderfs device".into()),
283        Some(_) => {}
284        None => return Identity::Unknown("its file system cannot be inspected".into()),
285    }
286    let (major, minor) = (libc::major(rdev), libc::minor(rdev));
287    if major != MISC_MAJOR || STATIC_MISC_MINORS.contains(&minor) {
288        return Identity::NotBinder;
289    }
290    let Some(misc) = misc() else {
291        return Identity::Unknown("/proc/misc cannot be read".into());
292    };
293    let Some(name) = misc.lines().find_map(|line| {
294        let (number, name) = line.trim_start().split_once(' ')?;
295        (number.parse::<u32>().ok()? == minor).then(|| name.trim().to_owned())
296    }) else {
297        return Identity::Unknown(format!(
298            "/proc/misc does not list misc device {major}:{minor}"
299        ));
300    };
301    match devices() {
302        Some(devices) if devices.trim().split(',').any(|device| device == name) => {
303            Identity::Binder(format!(
304                "the binder device {name} (misc device {major}:{minor})"
305            ))
306        }
307        Some(_) => Identity::NotBinder,
308        None => Identity::Unknown(format!(
309            "misc device {name} ({major}:{minor}) has a dynamic minor, and no binder driver's \
310             devices parameter can be read"
311        )),
312    }
313}

315/// [`identity`] of the file `fd_link`, a `/proc/<tid>/fd/<n>` entry, refers
316/// to, from host procfs and sysfs.
317pub(crate) fn identity_of(fd_link: &std::path::Path) -> Identity {
318    use std::os::unix::ffi::OsStrExt;
319    use std::os::unix::fs::MetadataExt;
320    let stat = std::fs::metadata(fd_link)
321        .ok()
322        .map(|metadata| (metadata.mode(), metadata.rdev()));
323    identity(
324        stat,
325        || {
326            let path = std::ffi::CString::new(fd_link.as_os_str().as_bytes()).ok()?;
327            // SAFETY: a zeroed `statfs` is a valid value of the plain C struct.
328            let mut file_system: libc::statfs = unsafe { std::mem::zeroed() };
329            // SAFETY: a NUL-terminated path and a buffer this frame owns.
330            let status = unsafe { libc::statfs(path.as_ptr(), &mut file_system) };
331            (status == 0).then_some(file_system.f_type as i64)
332        },
333        || std::fs::read_to_string("/proc/misc").ok(),
334        || binder_devices_parameter(|path| std::fs::read_to_string(path).ok()),
335    )
336}

339/// ("opened", "received through SCM_RIGHTS", …) and whose identity is
340/// `identity`: a binder device, or one Detcore cannot tell is not. `None`

342pub(crate) fn strict_descriptor_refusal(

347    let what = match identity {
348        Identity::Binder(what) => format!("is {what}"),

351        }

354    Some(format!(
355        "--strict refuses binder: descriptor {fd}, {how}, {what}. A --strict run holds no binder \
356         descriptor: binder reply payloads are not recorded, so no run, fresh or replayed, can \
357         repeat what a binder call returns (https://github.com/rrnewton/hermit/issues/5124). Run \
358         a guest that uses binder without --strict"
359    ))
360}

419/// Which of [`BINDER_MODULES`] the kernel's own module directory lists, as
420/// `file_system` (the `statfs` magic of `/sys/module`) and `listing` (every
421/// entry it lists, [`directory_entries`]) say; or why `/sys/module` is not
422/// that directory: a missing, unmounted or unreadable `/sys`, or a directory
423/// of another file system standing in for it.
424pub(crate) fn sys_module_binder_modules(
425    file_system: std::io::Result<i64>,
426    listing: std::io::Result<Vec<String>>,
427) -> Result<Vec<&'static str>, String> {
428    match file_system {
429        Ok(SYSFS_MAGIC) => {}

431        Err(error) => return Err(format!("/sys/module cannot be inspected ({error})")),

434    Ok(BINDER_MODULES
435        .into_iter()
436        .filter(|module| names.iter().any(|name| name == module))
437        .collect())
438}

453/// The binder driver's `devices` parameter, from
454/// `/sys/module/<module>/parameters/devices` for each of [`BINDER_MODULES`]
455/// as `read` reads it. A parameter that exists but cannot be read is
456/// [`DevicesParameter::Unreadable`] whatever the others say; one that is
457/// absent (`NotFound`) is passed over.
458pub(crate) fn binder_devices_parameter_state(
459    read: impl Fn(&str) -> std::io::Result<String>,
460) -> DevicesParameter {
461    let mut present = Vec::new();
462    for module in BINDER_MODULES {
463        let path = format!("/sys/module/{module}/parameters/devices");
464        match read(&path) {
465            Ok(devices) => present.push(devices.trim().to_owned()),
466            Err(error) if error.kind() == std::io::ErrorKind::NotFound => {}
467            Err(error) => return DevicesParameter::Unreadable(format!("{path}: {error}")),
468        }
469    }
470    if present.is_empty() {
471        DevicesParameter::Missing
472    } else {
473        DevicesParameter::Present(present.join(","))
474    }
475}

477/// Why `--strict` refuses a run on a backend whose guest descriptors are not
478/// host descriptors Detcore can name (KVM) on this host, or `None`.
479///
480/// There no descriptor's identity can be read where it arrives or at the
481/// start (<https://github.com/rrnewton/hermit/issues/5129>). So, as a
482/// temporary conservative policy, strict KVM is refused on any host that
483/// offers binder in any form, whether or not the guest would ever hold a
484/// binder descriptor. The run is refused when:
485/// - `filesystems`, the contents of `/proc/filesystems`, lists binderfs.
486///   That a guest could reach a binderfs instance (one it mounts, or another
487///   mount namespace's through a `/proc/<pid>/root` alias) is not checked:
488///   the registration alone refuses;
489/// - `misc`, the contents of `/proc/misc`, lists a misc device whose name
490///   contains "binder", or one that the binder driver's `devices` parameter
491///   names (`devices`, [`binder_devices_parameter_state`]). The first does
492///   not depend on the parameter, which a binder module loaded under another
493///   name would not have;
494/// - the binder driver's `devices` parameter exists but cannot be read;
495/// - `sys_module`, which of [`BINDER_MODULES`] `/sys/module` lists
496///   ([`sys_module_binder_modules`]), is not `Ok`: there a missing parameter
497///   would not prove that there is no binder driver, which a misc device
498///   named neither like binder's nor by a parameter Hermit can read would
499///   let through;
500/// - `/sys/module` lists one of [`BINDER_MODULES`] whose `devices` parameter
501///   is missing;
502/// - `held`, the first of Hermit's own descriptors, which the guest inherits,
503///   that is not known to be other than a binder device
504///   ([`held_binder_descriptor`]), is not `None`;
505/// - or any of `filesystems`, `misc` or Hermit's descriptor table cannot be
506///   read.
507///
508/// A host with none of these is admitted: no binderfs registered, no misc
509/// device named like binder's or named by a binder driver, a visible
510/// `/sys/module`, and no binder descriptor handed to the guest. One binder
511/// device this lets through remains, on KVM only: one a binder module loaded
512/// under a name other than [`BINDER_MODULES`] registers under a name without
513/// "binder" in it (ptrace refuses it as of unknown identity). Narrowing the
514/// policy to the binder objects a guest can actually reach needs their
515/// identity through the guest, which also accounts for `/proc/<pid>/root`
516/// aliases and closes that remainder: #5129's follow-up.
517///
518/// The check trusts host files as the kernel presents them:
519/// `/proc/filesystems`, `/proc/misc`, `/sys/module` and Hermit's own
520/// descriptor table alike. A host that deliberately mounts a substitute
521/// over any of them (another sysfs subtree bound over `/sys/module`, say)
522/// defeats them all equally. That is outside `--strict`'s threat model,
523/// which guards a run against outside input, not against its host: the
524/// guest cannot mount. Where these files come from belongs to the same
525/// object-identity follow-up.
526pub(crate) fn kvm_strict_binder_refusal(
527    filesystems: Option<&str>,
528    misc: Option<&str>,
529    devices: &DevicesParameter,
530    sys_module: Result<Vec<&'static str>, String>,
531    held: std::io::Result<Option<(RawFd, Identity)>>,
532) -> Option<String> {
533    let reason = 'reason: {
534        let Some(filesystems) = filesystems else {
535            break 'reason "/proc/filesystems cannot be read, so it cannot tell whether the host \
536                           kernel offers binderfs"
537                .to_owned();
538        };
539        if filesystems
540            .lines()
541            .any(|line| line.split_whitespace().last() == Some("binder"))
542        {
543            break 'reason "the host kernel offers binderfs (/proc/filesystems lists binder)"
544                .to_owned();
545        }
546        let Some(misc) = misc else {
547            break 'reason "/proc/misc cannot be read, so it cannot tell whether the host kernel \
548                           has a binder device"
549                .to_owned();
550        };
551        let named: Vec<&str> = match devices {
552            DevicesParameter::Present(devices) => devices.split(',').map(str::trim).collect(),
553            DevicesParameter::Missing | DevicesParameter::Unreadable(_) => Vec::new(),
554        };
555        if let Some(line) = misc.lines().find(|line| {
556            line.trim_start().split_once(' ').is_some_and(|(_, name)| {
557                let name = name.trim();
558                name.contains("binder") || named.contains(&name)
559            })
560        }) {
561            break 'reason format!(
562                "the host kernel has a binder device (/proc/misc: \"{}\")",
563                line.trim()
564            );
565        }
566        if let DevicesParameter::Unreadable(why) = devices {
567            break 'reason format!(
568                "the binder driver's devices parameter cannot be read ({why}), so it cannot tell \
569                 which misc devices are binder's"
570            );
571        }
572        match (sys_module, devices) {
573            (Err(why), _) => {
574                break 'reason format!(
575                    "{why}, so the absence of a binder driver's devices parameter does not \
576                     prove that the host kernel has no binder driver"
577                );
578            }
579            (Ok(listed), DevicesParameter::Missing) if !listed.is_empty() => {
580                break 'reason format!(
581                    "/sys/module lists the binder module ({}) but its devices parameter is \
582                     missing, so it cannot tell which misc devices are binder's",
583                    listed.join(", ")
584                );
585            }
586            (Ok(_), _) => {}
587        }
588        match held {
589            Ok(None) => return None,
590            Ok(Some((fd, Identity::Binder(what)))) => {
591                format!("Hermit's descriptor {fd}, which the guest inherits, is {what}")
592            }
593            Ok(Some((fd, Identity::Unknown(why)))) => format!(
594                "Hermit's descriptor {fd}, which the guest inherits, may be a binder device: its \
595                 identity could not be read ({why})"
596            ),
597            Ok(Some((_, Identity::NotBinder))) => return None,
598            Err(error) => {
599                format!(
600                    "Hermit's own descriptors, which the guest inherits, cannot be listed ({error})"
601                )
602            }
603        }
604    };
605    Some(format!(
606        "--strict refuses binder: {reason}. Under the KVM backend Hermit cannot tell whether a \
607         guest descriptor is a binder device (https://github.com/rrnewton/hermit/issues/5129), and \
608         binder reply payloads are not recorded \
609         (https://github.com/rrnewton/hermit/issues/5124), so, for now, a strict run is admitted \
610         only on a host that offers binder in no form. Use the ptrace backend, or run without \
611         --strict"
612    ))
613}

683/// Only the low 32 bits are the request: the kernel's `cmd` is an

686    request & 0xffff_ffff == BINDER_WRITE_READ

689/// Why a binder `ioctl` waits in the kernel until another process acts, so
690/// that Detcore runs it as a blocking external call rather than in the
691/// caller's turn. `None` for every request that returns on its own.
692#[derive(Clone, Copy, Debug, PartialEq, Eq)]
693pub(crate) enum Blocking {
694    /// A `BINDER_WRITE_READ` whose read half waits for work for this thread:
695    /// a transaction, a reply, or a death notification
696    /// (`binder_thread_read` -> `binder_wait_for_work`).
697    Read,
698    /// A `BINDER_FREEZE` that waits for another process's outstanding
699    /// transactions to drain, up to its own timeout
700    /// (`binder_ioctl_freeze`).
701    Freeze,
702}

704/// What a binder call's argument held, read from guest memory once, after
705/// the exit-report barrier (`Detcore::binder_snapshot`): the
706/// `binder_write_read` of a `BINDER_WRITE_READ` and the `binder_freeze_info`
707/// of a `BINDER_FREEZE`, `None` where the request has none or it cannot be
708/// read. Every decision Detcore makes before the call (its route, its poll,
709/// its refusals) and the `read_consumed` its receipts parse from after it
710/// come from this one read, so none reads the argument twice or before the
711/// barrier.
712#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
713pub(crate) struct Snapshot {
714    pub(crate) write_read: Option<WriteRead>,
715    pub(crate) freeze: Option<[u32; 3]>,
716}
717
718impl Snapshot {
719    /// Whether `request` has an argument Detcore reads:
720    /// `BINDER_WRITE_READ` and `BINDER_FREEZE`.
721    pub(crate) fn reads(request: u64) -> bool {
722        is_write_read(request) || request & 0xffff_ffff == BINDER_FREEZE
723    }
724
725    /// Read the argument of binder `request` at `argument`.
726    pub(crate) fn read<M: MemoryAccess>(memory: &M, request: u64, argument: usize) -> Self {
727        Snapshot {
728            write_read: is_write_read(request)
729                .then(|| write_read(memory, argument))
730                .flatten(),
731            freeze: (request & 0xffff_ffff == BINDER_FREEZE)
732                .then(|| freeze_info(memory, argument))
733                .flatten(),
734        }
735    }
736
737    /// The `read_consumed` the call was given: the bytes of `read_buffer` an
738    /// earlier call wrote, which this one leaves alone. Zero where there is
739    /// no `binder_write_read` or it could not be read, so the whole buffer is
740    /// parsed, as before.
741    pub(crate) fn consumed_before(&self) -> u64 {
742        self.write_read.map_or(0, |call| call.read_consumed)
743    }
744}

746/// `WriteRead` of the call whose argument is at `argument`, read from guest
747/// memory; `None` when it cannot be read.
748fn write_read<M: MemoryAccess>(memory: &M, argument: usize) -> Option<WriteRead> {
749    let address = Addr::<WriteRead>::from_raw(argument)?;
750    memory.read_value(address).ok()
751}
752
753impl WriteRead {
754    /// Whether this call has a read half, which is the half that waits.
755    /// Linux runs `binder_thread_read` whenever `read_size` is nonzero
756    /// (`binder_ioctl_write_read`, drivers/android/binder.c:5073 at v6.4,
757    /// <https://git.kernel.org/pub/scm/linux/kernel/git/stable/linux.git/tree/drivers/android/binder.c?h=v6.4#n5073>),
758    /// and that read waits for work (`binder_wait_for_work`, :4412) before it
759    /// looks at how much of the buffer is left (:4445). A buffer the caller
760    /// has already filled (`read_consumed == read_size`) therefore waits as
761    /// long as an empty one, and then returns with nothing added.
762    pub(crate) fn reads(&self) -> bool {
763        self.read_size > 0
764    }
765
766    /// Whether this call has a write half left to run: `binder_thread_write`
767    /// runs the commands of `write_buffer[write_consumed..write_size]`
768    /// (drivers/android/binder.c:3804-3807 at v6.4), before the read half and
769    /// in the same call (:5060-5071).
770    pub(crate) fn writes(&self) -> bool {
771        self.write_size > self.write_consumed
772    }
773}

775/// Why `request` waits in the kernel, from the call's argument already read
776/// from guest memory: `None` when it returns on its own. `nonblocking` is the
777/// descriptor's `O_NONBLOCK`, which makes a read return `EAGAIN` instead of
778/// waiting (`binder_thread_read`, drivers/android/binder.c:4408-4410 at
779/// v6.4). A read is classified as waiting whenever it has a read half
780/// ([`WriteRead::reads`]); the caller may then find work already queued and
781/// run it in the caller's turn anyway ([`route`]).
782pub(crate) fn blocking(
783    request: u64,
784    write_read: Option<&WriteRead>,
785    freeze: Option<&[u32; 3]>,
786    nonblocking: bool,
787) -> Option<Blocking> {
788    match request & 0xffff_ffff {
789        BINDER_WRITE_READ if !nonblocking => write_read
790            .filter(|call| call.reads())
791            .map(|_| Blocking::Read),
792        BINDER_FREEZE => freeze
793            .filter(|info| info[1] != 0 && info[2] != 0)
794            .map(|_| Blocking::Freeze),
795        _ => None,
796    }
797}

799/// What a binder call that waits does besides waiting, which the background
800/// path would run at a moment the host chooses rather than one the schedule
801/// orders against the guest's other threads.
802#[derive(Clone, Copy, Debug, PartialEq, Eq)]
803pub(crate) enum Effect {
804    /// The write half of a `BINDER_WRITE_READ` that also reads: the
805    /// transactions, replies and commands it sends, which Linux runs first,
806    /// in the same call (`binder_ioctl_write_read`).
807    Write,
808    /// A `BINDER_FREEZE`: Linux marks the target frozen as the call starts
809    /// and then waits for its transactions to drain (`binder_ioctl_freeze`,
810    /// drivers/android/binder.c:5249-5256 at v6.4).
811    Freeze,
812}

814/// How Detcore issues a binder `ioctl` (`Detcore::run_binder_call_with`).
815#[derive(Clone, Copy, Debug, PartialEq, Eq)]
816pub(crate) enum Route {
817    /// It returns on its own, so it runs in the caller's turn.
818    InTurn,
819    /// A read and nothing else, which waits only when no work is queued for
820    /// its thread. A zero-timeout poll in the caller's turn asks first
821    /// (`binder_poll` asks what this read's own wait asks, whether the thread
822    /// has work it may take: drivers/android/binder.c:5017-5040 against
823    /// :4387-4412 at v6.4): with work the read keeps the turn, otherwise it
824    /// runs on the background path. No half of the call runs before the
825    /// read, so the answer holds for it, with one host-timed exception: work
826    /// queued for the whole process (not for this thread) that a sibling
827    /// thread, whose own read entered the kernel late on the background path,
828    /// takes first. Work that arrives while a sibling already waits goes to
829    /// that sibling's own queue (`binder_proc_transaction`, :2785), so this needs the
830    /// sibling's entry to trail its turn.
831    PollFirst,
832    /// It waits and has an effect besides waiting. The caller's readiness
833    /// says nothing about whether it waits: a write half can make the thread
834    /// unavailable for the process work the poll saw (a synchronous
835    /// `BC_TRANSACTION` pushes its transaction stack and defers its own
836    /// completion, drivers/android/binder.c:3616-3619 and :559-564 at v6.4),
837    /// and a freeze waits on another process's transactions, not on this
838    /// thread's work. It never polls, and its effect would run off the turn
839    /// on the background path, so the run must allow that
840    /// (`Detcore::admit_host_timed_binder_call`).
841    HostTimed(Blocking, Effect),
842}

844/// How Detcore issues `request` ([`Route`]), from the call's argument already
845/// read from guest memory, as [`blocking`] reads it.
846pub(crate) fn route(
847    request: u64,
848    write_read: Option<&WriteRead>,
849    freeze: Option<&[u32; 3]>,
850    nonblocking: bool,
851) -> Route {
852    match blocking(request, write_read, freeze, nonblocking) {
853        None => Route::InTurn,
854        Some(Blocking::Freeze) => Route::HostTimed(Blocking::Freeze, Effect::Freeze),
855        Some(Blocking::Read) if write_read.is_some_and(WriteRead::writes) => {
856            Route::HostTimed(Blocking::Read, Effect::Write)
857        }
858        Some(Blocking::Read) => Route::PollFirst,
859    }
860}

979/// Count a binder call that runs as external input, and warn once per guest
980/// process (`process`, its thread-group id, which only keys the warning: the
981/// line does not print it, so it reads the same whatever IDs the run has)
982/// that its binder calls are: their
983/// results come from processes Hermit does not schedule, so two runs can
984/// differ wherever those processes do. The run's summary reports the count
985/// (`external_calls`).
986///
987/// Returns whether it warned. Which calls go to the background is decided by
988/// host timing (`run_binder_call_with`'s zero-timeout poll), so whether a process
989/// warns at all, and the count, can differ between runs; the line's text
990/// cannot.
991pub(crate) fn note_blocking(waits: Blocking, process: i32) -> bool {
992    EXTERNAL_CALLS.fetch_add(1, std::sync::atomic::Ordering::Relaxed);
993    static WARNED: std::sync::Mutex<std::collections::BTreeSet<i32>> =
994        std::sync::Mutex::new(std::collections::BTreeSet::new());
995    let first = WARNED.lock().unwrap().insert(process);
996    if first {
997        let call = match waits {
998            Blocking::Read => "a binder read (BINDER_WRITE_READ)",
999            Blocking::Freeze => "a binder freeze (BINDER_FREEZE)",
1000        };
1001        tracing::warn!(
1002            "[detcore] HERMIT_EXTERNAL_INPUT source=binder: {call} waits for another process, \
1003             which Hermit does not schedule. It runs as a blocking external call, so when it \
1004             finishes, and what it reads, are this host's, not the schedule's (warned once per \
1005             guest process; the run summary counts every such call)"
1006        );
1007    }
1008    first
1009}

1012/// has granted its `BlockingExternalIO` request and the native call is
1013/// about to run. This is the first point that proves admission — the
1014/// earlier `HERMIT_EXTERNAL_INPUT` warning only proves the off-turn route
1015/// was chosen, before the request was even submitted. The receipt names
1016/// the operation (`ExternalOpId`: thread and per-thread syscall count),
1017/// so an outside observer can correlate it with the recorded release.
1018pub(crate) fn note_blocking_granted(op_id: crate::resources::ExternalOpId) {
1019    #[cfg(test)]

1055/// The descriptors a successful `BINDER_WRITE_READ`, whose
1056/// `struct binder_write_read` is at `argument`, made Linux install in the
1057/// caller, read from guest memory after the call. `already_consumed` is the
1058/// `read_consumed` the caller passed in, read before the call: Linux writes
1059/// this call's commands after it.
1060pub(crate) fn delivered<M: MemoryAccess>(
1061    memory: &M,
1062    argument: usize,
1063    already_consumed: u64,
1064) -> Delivered {
1065    let Some(address) = Addr::<[u64; 6]>::from_raw(argument) else {
1066        return refused("its binder_write_read is at address 0");
1067    };
1068    let Ok(words) = memory.read_value(address) else {
1069        return refused("its binder_write_read cannot be read");
1070    };
1071    // write_size, write_consumed, write_buffer, read_size, read_consumed,
1072    // read_buffer.
1073    let [_, _, _, read_size, read_consumed, read_buffer] = words;
1074    if read_consumed == 0 || read_consumed == already_consumed {
1075        return Delivered::default();
1076    }
1077    if read_consumed > read_size || already_consumed > read_consumed {
1078        return refused("its read_consumed is not within its read_size");
1079    }
1080    // Linux writes this call's commands after the read_consumed the caller
1081    // passed in, leaving any earlier ones alone: those are an older call's,
1082    // and their buffers may already be freed.
1083    let newly_consumed = read_consumed - already_consumed;
1084    if newly_consumed > MAX_COMMAND_BYTES as u64 {
1085        return refused("its newly returned commands exceed Detcore's read limit");
1086    }
1087    let length = newly_consumed as usize;
1088    let Some(address) = Addr::<u8>::from_raw((read_buffer + already_consumed) as usize) else {
1089        return refused("its read_buffer is at address 0");
1090    };
1091    let mut commands = vec![0; length];
1092    if memory.read_exact(address, &mut commands).is_err() {
1093        return refused("its return commands cannot be read");
1094    }
1095    match installed_descriptors(memory, &commands) {
1096        Ok(fds) => Delivered {
1097            // One transaction: its descriptors arrive as one message.
1098            messages: vec![crate::syscalls::scm_rights::DeliveredMessage {
1099                count: fds.len(),
1100                truncated: false,
1101                bytes: None,
1102            }],
1103            fds,
1104            refused: None,
1105            may_have_installed: false,
1106        },
1107        Err(why) => refused(why),
1108    }
1109}
1110
1111fn refused(why: &'static str) -> Delivered {
1112    Delivered {
1113        fds: Vec::new(),
1114        messages: Vec::new(),
1115        refused: Some(why),
1116        // A transaction Detcore cannot read may have carried descriptors.
1117        may_have_installed: true,
1118    }
1119}

1139/// The descriptors the transactions in `commands`, a run of return commands
1140/// as Linux wrote them, carry, in order.
1141fn installed_descriptors<M: MemoryAccess>(
1142    memory: &M,
1143    commands: &[u8],
1144) -> Result<Vec<RawFd>, &'static str> {
1145    let mut fds = Vec::new();
1146    let mut offset = 0;
1147    while offset < commands.len() {
1148        let cmd = u32_at(commands, offset).ok_or("a return command is cut short")?;
1149        let payload = commands
1150            .get(offset + 4..offset + 4 + payload_size(cmd))
1151            .ok_or("a return command's payload is cut short")?;
1152        if matches!(cmd, BR_TRANSACTION | BR_TRANSACTION_SEC_CTX | BR_REPLY) {
1153            fds.extend(transaction_descriptors(memory, payload)?);
1154        }
1155        offset += 4 + payload.len();
1156    }
1157    Ok(fds)
1158}

1160/// The descriptors the transaction `data`, a `struct binder_transaction_data`
1161/// (the head of a `_secctx` one too), carries.
1162fn transaction_descriptors<M: MemoryAccess>(
1163    memory: &M,
1164    data: &[u8],
1165) -> Result<Vec<RawFd>, &'static str> {
1166    let short = "a transaction is cut short";
1167    // The driver never reads TF_STATUS_CODE when it installs descriptors, so
1168    // neither does this: a real status reply carries no objects anyway, and
1169    // skipping on the flag would hide a descriptor Linux did install.
1170    let offsets_size = u64_at(data, 40).ok_or(short)? as usize;
1171    let buffer = u64_at(data, 48).ok_or(short)? as usize;
1172    let offsets_at = u64_at(data, 56).ok_or(short)? as usize;
1173    let count = offsets_size / 8;
1174    if count == 0 {
1175        return Ok(Vec::new());
1176    }
1177    if count > MAX_OBJECTS {
1178        return Err("a transaction carries more objects than Detcore reads");
1179    }
1180    let mut offsets = vec![0u64; count];
1181    let address = Addr::<u64>::from_raw(offsets_at).ok_or("a transaction's offsets are at 0")?;
1182    memory
1183        .read_values(address, &mut offsets)
1184        .map_err(|_| "a transaction's offsets cannot be read")?;
1185    let object = |index: usize, length: usize| -> Result<Vec<u8>, &'static str> {
1186        let at = offsets
1187            .get(index)
1188            .ok_or("an object names a parent that is not in the transaction")?;
1189        let address = Addr::<u8>::from_raw(buffer.wrapping_add(*at as usize))
1190            .ok_or("a transaction's object is at 0")?;
1191        let mut bytes = vec![0; length];
1192        memory
1193            .read_exact(address, &mut bytes)
1194            .map_err(|_| "a transaction's object cannot be read")?;
1195        Ok(bytes)
1196    };
1197    let mut fds = Vec::new();
1198    for index in 0..count {
1199        // Every object starts with its 32-bit type.
1200        let header = object(index, 4)?;
1201        match u32_at(&header, 0) {
1202            // struct binder_fd_object: the type, a 32-bit pad, then the
1203            // descriptor in the low half of a 64-bit union.
1204            Some(BINDER_TYPE_FD) => {
1205                let fd_object = object(index, 24)?;
1206                fds.push(u32_at(&fd_object, 8).ok_or(short)? as RawFd);
1207            }
1208            // A BINDER_TYPE_FDA array's descriptors are the kernel's to
1209            // close: it closes them itself when the receiver frees the
1210            // transaction's buffer with BC_FREE_BUFFER, so a record of one
1211            // would outlive it and describe whatever next takes the number.
1212            // Left out until Detcore follows those frees
1213            // (https://github.com/rrnewton/hermit/issues/5046).
1214            Some(BINDER_TYPE_FDA) => {
1215                let array = object(index, 32)?;
1216                let num_fds = u64_at(&array, 8).ok_or(short)?;
1217                if num_fds != 0 {
1218                    return Err(
1219                        "a transaction carries a descriptor array, whose descriptors Linux \
1220                         closes with the buffer",
1221                    );
1222                }
1223            }
1224            Some(_) => {}
1225            None => return Err(short),
1226        }
1227    }
1228    Ok(fds)
1229}
