21// Linux exposes USER_HZ, not the kernel's configurable scheduler HZ, through times(2).
22const CLOCK_TICKS_PER_SECOND: u64 = 100;
23const NANOS_PER_CLOCK_TICK: u64 = 1_000_000_000 / CLOCK_TICKS_PER_SECOND;
24
25fn clock_ticks(duration: crate::types::LogicalTime) -> u64 {
26    duration.as_nanos() / NANOS_PER_CLOCK_TICK
27}
28
29fn clock_t_from_ticks(ticks: u64) -> libc::clock_t {
30    ticks as libc::clock_t
31}
32
33const NANOS_PER_SECOND: u64 = 1_000_000_000;
34const NANOS_PER_MICROSECOND: u64 = 1_000;
35
36/// Render a logical CPU duration as the `timeval` `getrusage(2)` reports.
37///
38/// Linux truncates rusage CPU times to microsecond granularity, so the sub-microsecond
39/// remainder of the logical duration is discarded rather than rounded. Truncation (not
40/// rounding) is what keeps the value monotonic: a duration that grows by less than a
41/// microsecond must never make the reported total go backwards, and rounding-to-nearest
42/// on a shrinking remainder can do exactly that.
43fn timeval_from_logical(duration: crate::types::LogicalTime) -> libc::timeval {
44    let nanos = duration.as_nanos();
45    libc::timeval {
46        tv_sec: (nanos / NANOS_PER_SECOND) as libc::time_t,
47        tv_usec: ((nanos % NANOS_PER_SECOND) / NANOS_PER_MICROSECOND) as libc::suseconds_t,
48    }
49}
50
51fn logical_clock_ticks(
52    now: crate::types::LogicalTime,
53    boot: crate::types::LogicalTime,
54    uptime_offset_seconds: u64,
55) -> libc::clock_t {
56    let ticks = uptime_offset_seconds
57        .wrapping_mul(CLOCK_TICKS_PER_SECOND)
58        .wrapping_add(clock_ticks(now - boot));
59    clock_t_from_ticks(ticks)
60}

62/// Project elapsed logical time into Linux's integer `sysinfo(2)` uptime.
63///
64/// Linux rounds a positive fractional boottime up to the next second. Subtract
65/// the exact internal epoch first so its calendar fraction cannot affect the
66/// result. This projection does not alter the underlying logical clock.
67fn sysinfo_uptime_seconds(
68    now: crate::types::LogicalTime,
69    epoch: crate::types::LogicalTime,
70    uptime_offset_seconds: u64,
71) -> Result<u64, Error> {
72    let now_ns = now.as_nanos();
73    let epoch_ns = epoch.as_nanos();
74    let elapsed_ns = now_ns.checked_sub(epoch_ns).ok_or_else(|| {
75        Error::Tool(anyhow::anyhow!(
76            "sysinfo observed logical time {now_ns} ns before epoch {epoch_ns} ns"
77        ))
78    })?;
79    // Divide before rounding: adding NANOS_PER_SECOND - 1 to elapsed_ns can
80    // overflow. The rounded quotient of any u64 nanosecond duration fits u64.
81    let seconds =
82        elapsed_ns / NANOS_PER_SECOND + u64::from(!elapsed_ns.is_multiple_of(NANOS_PER_SECOND));
83    // Config accepts the full u64 offset; preserve its existing release-build
84    // wrapping extension, including the subsequent SysInfo -> c_long ABI cast.
85    // Ordinary Linux uptime semantics apply within the nonnegative c_long range.
86    Ok(uptime_offset_seconds.wrapping_add(seconds))
87}

89fn prlimit_targets_current_process(
90    target_pid: i32,
91    deterministic_pid: Option<i32>,
92    physical_pid: i32,
93) -> bool {
94    target_pid == 0 || target_pid == deterministic_pid.unwrap_or(physical_pid)
95}

97fn validate_resource_limit_mutation(
98    resource: u32,
99    previous: ResourceLimit,
100    requested: ResourceLimit,
101) -> Result<(), Errno> {
102    if requested.current > requested.maximum {
103        return Err(Errno::EINVAL);
104    }
105    // Linux accepts an exact no-op for every valid resource, including limits
106    // that an unprivileged process could not otherwise change. Recognize that
107    // case before applying Detcore's narrower virtual-mutation policy.
108    if requested == previous {
109        return Ok(());
110    }
111    // CORE is virtual compatibility state too: changing it cannot enable host
112    // core dumps, while Linux sanitizers routinely lower its soft limit.
113    if resource != libc::RLIMIT_STACK
114        && resource != libc::RLIMIT_NOFILE
115        && resource != libc::RLIMIT_CORE
116    {
117        return Err(Errno::EPERM);
118    }
119    if requested.maximum > previous.maximum {
120        return Err(Errno::EPERM);
121    }
122    Ok(())
123}

129    pub async fn handle_getrlimit<G: Guest<Self>>(
130        &self,
131        guest: &mut G,
132        call: syscalls::Getrlimit,
133    ) -> Result<i64, Error> {
134        let resource = u32::try_from(call.resource()).map_err(|_| Errno::EINVAL)?;
135        let address = call.rlim().ok_or(Errno::EFAULT)?;
136        let limit = guest
137            .thread_state()
138            .resource_limits
139            .lock()
140            .expect("resource limits mutex poisoned")
141            .get(resource)
142            .ok_or(Errno::EINVAL)?;
143        let result = libc::rlimit {
144            rlim_cur: limit.current,
145            rlim_max: limit.maximum,
146        };
147        guest.memory().write_value(address, &result)?;
148        Ok(0)
149    }

154    pub async fn handle_setrlimit<G: Guest<Self>>(
155        &self,
156        guest: &mut G,
157        call: syscalls::Setrlimit,
158    ) -> Result<i64, Error> {
159        let resource = u32::try_from(call.resource()).map_err(|_| Errno::EINVAL)?;
160        let address = call.rlim().ok_or(Errno::EFAULT)?;
161        let requested: libc::rlimit = guest.memory().read_value(address)?;
162        let requested = ResourceLimit {
163            current: requested.rlim_cur,
164            maximum: requested.rlim_max,
165        };
166        let resource_limits = guest.thread_state().resource_limits.clone();
167        let mut limits = resource_limits
168            .lock()
169            .expect("resource limits mutex poisoned");
170        let previous = limits.get(resource).ok_or(Errno::EINVAL)?;
171        validate_resource_limit_mutation(resource, previous, requested)?;
172        if requested != previous {
173            limits.set(resource, requested);
174        }
175        Ok(0)
176    }

178    /// Virtualize `prlimit64(2)` for the current guest process.
179    ///
180    /// Queries return process-local deterministic values. Exact no-op updates
181    /// succeed for every valid resource, as on Linux. Changes are kept virtual
182    /// and restricted to limits that do not grant access to host resources or
183    /// affect host scheduling. Accepted changes update only guest-observable
184    /// compatibility state; they are not a sandbox boundary and do not ask the
185    /// host kernel to enforce the virtual limit.
186    // AUTONOMOUS-BOT-IMPLEMENTED
187    // TODO-HUMAN-REVIEW(#534)
188    pub async fn handle_prlimit64<G: Guest<Self>>(
189        &self,
190        guest: &mut G,
191        call: syscalls::Prlimit64,
192    ) -> Result<i64, Error> {
193        let resource = call.resource();
194        let resource_limits = guest.thread_state().resource_limits.clone();
195        if resource_limits
196            .lock()
197            .expect("resource limits mutex poisoned")
198            .get(resource)
199            .is_none()
200        {
201            return Err(Errno::EINVAL.into());
202        }
203
204        let requested = if let Some(address) = call.new_rlim() {
205            let limit: libc::rlimit64 = guest.memory().read_value(address)?;
206            Some(ResourceLimit {
207                current: limit.rlim_cur,
208                maximum: limit.rlim_max,
209            })
210        } else {
211            None
212        };
213
214        let pid = call.pid();
215        let deterministic_pid = guest.thread_state().detpid.map(|detpid| detpid.as_raw());
216        if !prlimit_targets_current_process(pid, deterministic_pid, guest.pid().as_raw()) {
217            return Err(Errno::EPERM.into());
218        }
219
220        let previous = {
221            let mut limits = resource_limits
222                .lock()
223                .expect("resource limits mutex poisoned");
224            let previous = limits
225                .get(resource)
226                .expect("resource validity changed while handling prlimit64");
227
228            if let Some(requested) = requested {
229                validate_resource_limit_mutation(resource, previous, requested)?;
230                if requested != previous {
231                    limits.set(resource, requested);
232                }
233            }
234
235            previous
236        };
237
238        if let Some(address) = call.old_rlim() {
239            let previous = libc::rlimit64 {
240                rlim_cur: previous.current,
241                rlim_max: previous.maximum,
242            };
243            guest.memory().write_value(address, &previous)?;
244        }
245
246        crate::detlog!(
247            "prlimit64: pid={pid}, resource={resource}, mutation={}, old={}:{}",
248            requested.is_some(),
249            previous.current,
250            previous.maximum
251        );
252        Ok(0)
253    }
254    /// Return a deterministic resource-usage snapshot.
255    ///
256    /// `ru_utime`/`ru_stime` come from the SAME logical CPU accounting that backs `times(2)`
257    /// (see [`Self::handle_times`]), not from host scheduler counters. Reporting them as zero,
258    /// as this did previously, was both a fidelity bug and an internal contradiction: a guest
259    /// that called `times(2)` saw advancing CPU time while `getrusage(2)` insisted the same
260    /// process had consumed none. Deriving both from `ProcessCpuSnapshot` makes the two
261    /// syscalls agree by construction rather than by coincidence.
262    ///
263    /// The `who` values report different aggregates, matching Linux:
264    /// - `RUSAGE_SELF` — this process, summed across its threads.
265    /// - `RUSAGE_THREAD` — the calling thread alone. This reads the thread's own logical CPU
266    ///   counters rather than the process totals; substituting the process aggregate would
267    ///   over-report for every multithreaded guest.
268    /// - `RUSAGE_CHILDREN` — reaped children only, which is exactly what the `children_*`
269    ///   fields accumulate on `wait`.
270    ///
271    /// `ru_maxrss` is populated for the process/thread cases with the guest's peak resident set
272    /// size so that programs which require a positive maximum RSS (e.g. rr's `rusage` test)
273    /// behave like they do on Linux. This remains a best-effort host-procfs observation on
274    /// backends where [`Guest::pid`] names a host process; it is separate from the configured
275    /// system-wide memory reported by `sysinfo(2)` and virtual `/proc/meminfo`.
276    ///
277    /// Page-fault and context-switch counts remain zero: Detcore does not model them, and
278    /// synthesizing a plausible-looking number would be worse than reporting none.
279    pub async fn handle_getrusage<G: Guest<Self>>(
280        &self,
281        guest: &mut G,
282        call: syscalls::Getrusage,
283    ) -> Result<i64, Error> {
284        let who = call.who();
285        match who {
286            libc::RUSAGE_SELF | libc::RUSAGE_CHILDREN | libc::RUSAGE_THREAD => {}
287            _ => return Err(Errno::EINVAL.into()),
288        }
289
290        let usage_addr = call.usage().ok_or(Errno::EFAULT)?;
291
292        // SAFETY: `libc::rusage` is a plain-old-data C struct that is valid when zero-initialized.
293        let mut usage: libc::rusage = unsafe { std::mem::zeroed() };
294
295        let (user, system) = match who {
296            libc::RUSAGE_THREAD => guest.thread_state_mut().thread_cpu_time(),
297            libc::RUSAGE_CHILDREN => {
298                let cpu = guest.thread_state_mut().process_cpu_time();
299                (cpu.children_user, cpu.children_system)
300            }
301            // RUSAGE_SELF
302            _ => {
303                let cpu = guest.thread_state_mut().process_cpu_time();
304                (cpu.user, cpu.system)
305            }
306        };
307        usage.ru_utime = timeval_from_logical(user);
308        usage.ru_stime = timeval_from_logical(system);
309
310        // RUSAGE_SELF/RUSAGE_THREAD report this process's peak RSS. RUSAGE_CHILDREN aggregates
311        // terminated children only; with no such accounting we leave it zero, matching Linux when
312        // no child has exited.
313        if matches!(who, libc::RUSAGE_SELF | libc::RUSAGE_THREAD) {
314            usage.ru_maxrss = self.guest_peak_rss_kb(guest) as libc::c_long;
315        }
316
317        guest.memory().write_value(usage_addr, &usage)?;
318        Ok(0)
319    }

321    // AUTONOMOUS-BOT-IMPLEMENTED
322    // TODO-HUMAN-REVIEW(#797): Review logical elapsed and process CPU accounting semantics.
323    /// Return deterministic elapsed ticks and process CPU accounting for `times(2)`.
324    ///
325    /// Linux's host boot epoch and scheduler CPU counters are nondeterministic. Detcore instead
326    /// derives the return value from its global logical clock. Per-process logical CPU accounting
327    /// aggregates user instruction and syscall-system time across threads; forked processes start
328    /// fresh counters and contribute their totals to the parent's child counters when reaped.
329    pub async fn handle_times<G: Guest<Self>>(
330        &self,
331        guest: &mut G,
332        call: syscalls::Times,
333    ) -> Result<i64, Error> {
334        let now = thread_observe_time(guest).await;
335        let boot = crate::types::DetTime::new(&self.cfg).as_nanos();
336        let ticks = logical_clock_ticks(now, boot, self.cfg.sysinfo_uptime_offset);
337        let cpu = guest.thread_state_mut().process_cpu_time();
338
339        if let Some(address) = call.buf() {
340            let usage = libc::tms {
341                tms_utime: clock_t_from_ticks(clock_ticks(cpu.user)),
342                tms_stime: clock_t_from_ticks(clock_ticks(cpu.system)),
343                tms_cutime: clock_t_from_ticks(clock_ticks(cpu.children_user)),
344                tms_cstime: clock_t_from_ticks(clock_ticks(cpu.children_system)),
345            };
346            guest.memory().write_value(address, &usage)?;
347        }
348
349        Ok(ticks as i64)
350    }

352    /// The guest's peak resident set size ("high water mark") in kibibytes, matching the units of
353    /// Linux `getrusage`'s `ru_maxrss`. This reads host procfs through [`Guest::pid`], which only
354    /// identifies the guest process on backends where it names a host process; always returns a
355    /// positive value so guests can rely on a nonzero maximum RSS even if the read fails.
356    fn guest_peak_rss_kb<G: Guest<Self>>(&self, guest: &G) -> u64 {
357        Process::new(guest.pid().as_raw())
358            .and_then(|process| process.status())
359            .ok()
360            .and_then(|status| status.vmhwm.or(status.vmrss))
361            .unwrap_or(0)
362            .max(1)
363    }

379    pub(super) async fn calculate_uptime<G: Guest<Self>>(
380        &self,
381        guest: &mut G,
382    ) -> Result<u64, Error> {
383        let global_time = thread_observe_time(guest).await;
384        Ok(self.cfg.sysinfo_uptime_offset + global_time.as_secs()
385            - crate::types::DetTime::new(&self.cfg).as_nanos().as_secs())
386    }
387
388    async fn collect_sysinfo<G: Guest<Self>>(
389        &self,
390        guest: &mut G,
391    ) -> Result<syscalls::SysInfo, Error> {
392        let memory = configured_memory(self.cfg.memory);
393        let now = thread_observe_time(guest).await;
394        let epoch = crate::types::DetTime::new(&self.cfg).as_nanos();
395        Ok(syscalls::SysInfo {
396            uptime: sysinfo_uptime_seconds(now, epoch, self.cfg.sysinfo_uptime_offset)?,
397            loads_1: 1,
398            loads_5: 1,
399            loads_15: 1,
400            total_ram: memory.total_ram,
401            free_ram: memory.free_ram,
402            buffer_ram: memory.buffer_ram,
403            shared_ram: memory.shared_ram,
404            total_swap: memory.total_swap,
405            free_swap: memory.free_swap,
406            procs: 1,
407            total_high: memory.total_high,
408            free_high: memory.free_high,
409            mem_unit: memory.mem_unit,
410        })
411    }

725        // yield 2us and could make a later, larger duration report a SMALLER value once its
726        // remainder shrank -- i.e. CPU time going backwards. Pin truncation explicitly.
727        let tv = timeval_from_logical(LogicalTime::from_nanos(1_999));
728        assert_eq!(tv.tv_sec, 0);
729        assert_eq!(tv.tv_usec, 1);
730    }

732    #[test]
733    fn rusage_timeval_is_monotonic_in_the_logical_duration() {
734        // The property that matters to a guest: CPU time never goes backwards. Walk a range
735        // of nanosecond durations across microsecond and second boundaries and assert the
736        // rendered timeval is non-decreasing at every step.
737        let mut previous = (0_i64, 0_i64);
738        for nanos in (0..3_000_000u64).step_by(997) {
739            let tv = timeval_from_logical(LogicalTime::from_nanos(nanos));
740            let current = (tv.tv_sec, tv.tv_usec);
741            assert!(
742                current >= previous,
743                "rusage timeval went backwards at {nanos}ns: {previous:?} -> {current:?}"
744            );
745            previous = current;
746        }
747    }

756    #[test]
757    fn rusage_and_times_agree_within_one_clock_tick() {
758        // Both syscalls project the same logical duration, but times(2) is
759        // quantized to USER_HZ while getrusage(2) retains microseconds.
760        for nanos in [0u64, 1_000_000, 300_484_000, 7_000_000_000, 12_345_678_901] {
761            let duration = LogicalTime::from_nanos(nanos);
762            let tv = timeval_from_logical(duration);
763            let rusage_micros = tv.tv_sec as u64 * 1_000_000 + tv.tv_usec as u64;
764            let times_micros = clock_ticks(duration) * (NANOS_PER_CLOCK_TICK / 1_000);
765
766            assert!(rusage_micros >= times_micros);
767            assert!(rusage_micros - times_micros < NANOS_PER_CLOCK_TICK / 1_000);
768
769            // A tick-ALIGNED duration must agree EXACTLY, not merely to within
770            // one tick. The bounds above are satisfied at every sample by an
771            // implementation carrying a constant sub-tick offset, so without
772            // this the suite cannot distinguish that from a correct one.
773            if nanos % NANOS_PER_CLOCK_TICK == 0 {
774                assert_eq!(rusage_micros, times_micros);
775            }
776        }
777    }
