21const MB: u64 = 1024 * 1024;
22// Linux exposes USER_HZ, not the kernel's configurable scheduler HZ, through times(2).
23const CLOCK_TICKS_PER_SECOND: u64 = 100;
24const NANOS_PER_CLOCK_TICK: u64 = 1_000_000_000 / CLOCK_TICKS_PER_SECOND;
25
26fn clock_ticks(duration: crate::types::LogicalTime) -> u64 {
27    duration.as_nanos() / NANOS_PER_CLOCK_TICK
28}
29
30fn clock_t_from_ticks(ticks: u64) -> libc::clock_t {
31    ticks as libc::clock_t
32}
33
34const NANOS_PER_SECOND: u64 = 1_000_000_000;
35const NANOS_PER_MICROSECOND: u64 = 1_000;
36
37/// Render a logical CPU duration as the `timeval` `getrusage(2)` reports.
38///
39/// Linux truncates rusage CPU times to microsecond granularity, so the sub-microsecond
40/// remainder of the logical duration is discarded rather than rounded. Truncation (not
41/// rounding) is what keeps the value monotonic: a duration that grows by less than a
42/// microsecond must never make the reported total go backwards, and rounding-to-nearest
43/// on a shrinking remainder can do exactly that.
44fn timeval_from_logical(duration: crate::types::LogicalTime) -> libc::timeval {
45    let nanos = duration.as_nanos();
46    libc::timeval {
47        tv_sec: (nanos / NANOS_PER_SECOND) as libc::time_t,
48        tv_usec: ((nanos % NANOS_PER_SECOND) / NANOS_PER_MICROSECOND) as libc::suseconds_t,
49    }
50}
51
52fn logical_clock_ticks(
53    now: crate::types::LogicalTime,
54    boot: crate::types::LogicalTime,
55    uptime_offset_seconds: u64,
56) -> libc::clock_t {
57    let ticks = uptime_offset_seconds
58        .wrapping_mul(CLOCK_TICKS_PER_SECOND)
59        .wrapping_add(clock_ticks(now - boot));
60    clock_t_from_ticks(ticks)
61}

67    pub async fn handle_getrlimit<G: Guest<Self>>(
68        &self,
69        guest: &mut G,
70        call: syscalls::Getrlimit,
71    ) -> Result<i64, Error> {
72        let resource = u32::try_from(call.resource()).map_err(|_| Errno::EINVAL)?;
73        let address = call.rlim().ok_or(Errno::EFAULT)?;
74        let limit = guest
75            .thread_state()
76            .resource_limits
77            .lock()
78            .expect("resource limits mutex poisoned")
79            .get(resource)
80            .ok_or(Errno::EINVAL)?;
81        let result = libc::rlimit {
82            rlim_cur: limit.current,
83            rlim_max: limit.maximum,
84        };
85        guest.memory().write_value(address, &result)?;
86        Ok(0)
87    }

91    /// Update one virtual process resource limit through the legacy ABI.
92    pub async fn handle_setrlimit<G: Guest<Self>>(
93        &self,
94        guest: &mut G,
95        call: syscalls::Setrlimit,
96    ) -> Result<i64, Error> {
97        let resource = u32::try_from(call.resource()).map_err(|_| Errno::EINVAL)?;
98        let address = call.rlim().ok_or(Errno::EFAULT)?;
99        let requested: libc::rlimit = guest.memory().read_value(address)?;
100        let requested = ResourceLimit {
101            current: requested.rlim_cur,
102            maximum: requested.rlim_max,
103        };
104        let resource_limits = guest.thread_state().resource_limits.clone();
105        let mut limits = resource_limits
106            .lock()
107            .expect("resource limits mutex poisoned");
108        let previous = limits.get(resource).ok_or(Errno::EINVAL)?;
109        if requested.current > requested.maximum {
110            return Err(Errno::EINVAL.into());
111        }
112        if resource != libc::RLIMIT_STACK && resource != libc::RLIMIT_NOFILE {
113            return Err(Errno::EPERM.into());
114        }
115        if requested.maximum > previous.maximum {
116            return Err(Errno::EPERM.into());
117        }
118        limits.set(resource, requested);
119        Ok(0)
120    }

122    /// Virtualize `prlimit64(2)` for the current guest process.
123    ///
124    /// Queries return process-local deterministic values. Mutations are kept
125    /// virtual and restricted to limits that do not grant access to host
126    /// resources or affect host scheduling. Accepted mutations update only
127    /// guest-observable compatibility state; they are not a sandbox boundary
128    /// and do not ask the host kernel to enforce the virtual limit.

130    // TODO-HUMAN-REVIEW(#534)
131    pub async fn handle_prlimit64<G: Guest<Self>>(
132        &self,
133        guest: &mut G,
134        call: syscalls::Prlimit64,
135    ) -> Result<i64, Error> {
136        let resource = call.resource();
137        let resource_limits = guest.thread_state().resource_limits.clone();
138        if resource_limits
139            .lock()
140            .expect("resource limits mutex poisoned")
141            .get(resource)
142            .is_none()
143        {
144            return Err(Errno::EINVAL.into());
145        }
146
147        let requested = if let Some(address) = call.new_rlim() {
148            let limit: libc::rlimit64 = guest.memory().read_value(address)?;
149            Some(ResourceLimit {
150                current: limit.rlim_cur,
151                maximum: limit.rlim_max,
152            })
153        } else {
154            None
155        };
156
157        let pid = call.pid();
158        if pid != 0 && pid != guest.pid().as_raw() {
159            return Err(Errno::EPERM.into());
160        }
161
162        let previous = {
163            let mut limits = resource_limits
164                .lock()
165                .expect("resource limits mutex poisoned");
166            let previous = limits
167                .get(resource)
168                .expect("resource validity changed while handling prlimit64");
169
170            if let Some(requested) = requested {
171                if requested.current > requested.maximum {
172                    return Err(Errno::EINVAL.into());
173                }
174                if resource != libc::RLIMIT_STACK && resource != libc::RLIMIT_NOFILE {
175                    return Err(Errno::EPERM.into());
176                }
177                if requested.maximum > previous.maximum {
178                    return Err(Errno::EPERM.into());
179                }
180                limits.set(resource, requested);
181            }
182
183            previous
184        };
185
186        if let Some(address) = call.old_rlim() {
187            let previous = libc::rlimit64 {
188                rlim_cur: previous.current,
189                rlim_max: previous.maximum,
190            };
191            guest.memory().write_value(address, &previous)?;
192        }
193
194        crate::detlog!(
195            "prlimit64: pid={pid}, resource={resource}, mutation={}, old={}:{}",
196            requested.is_some(),
197            previous.current,
198            previous.maximum
199        );
200        Ok(0)
201    }
202    /// Return a deterministic resource-usage snapshot.
203    ///
204    /// `ru_utime`/`ru_stime` come from the SAME logical CPU accounting that backs `times(2)`
205    /// (see [`Self::handle_times`]), not from host scheduler counters. Reporting them as zero,
206    /// as this did previously, was both a fidelity bug and an internal contradiction: a guest
207    /// that called `times(2)` saw advancing CPU time while `getrusage(2)` insisted the same
208    /// process had consumed none. Deriving both from `ProcessCpuSnapshot` makes the two
209    /// syscalls agree by construction rather than by coincidence.
210    ///
211    /// The `who` values report different aggregates, matching Linux:
212    /// - `RUSAGE_SELF` — this process, summed across its threads.
213    /// - `RUSAGE_THREAD` — the calling thread alone. This reads the thread's own logical CPU
214    ///   counters rather than the process totals; substituting the process aggregate would
215    ///   over-report for every multithreaded guest.
216    /// - `RUSAGE_CHILDREN` — reaped children only, which is exactly what the `children_*`
217    ///   fields accumulate on `wait`.
218    ///
219    /// `ru_maxrss` is populated for the process/thread cases with the guest's peak resident set
220    /// size so that programs which require a positive maximum RSS (e.g. rr's `rusage` test)
221    /// behave like they do on Linux. The value comes from the same procfs memory accounting that
222    /// `sysinfo`'s free-memory reporting already relies on, which is deterministic across runs
223    /// under Detcore's fixed schedule.
224    ///
225    /// Page-fault and context-switch counts remain zero: Detcore does not model them, and
226    /// synthesizing a plausible-looking number would be worse than reporting none.
227    pub async fn handle_getrusage<G: Guest<Self>>(
228        &self,
229        guest: &mut G,
230        call: syscalls::Getrusage,
231    ) -> Result<i64, Error> {
232        let who = call.who();
233        match who {
234            libc::RUSAGE_SELF | libc::RUSAGE_CHILDREN | libc::RUSAGE_THREAD => {}
235            _ => return Err(Errno::EINVAL.into()),
236        }
237
238        let usage_addr = call.usage().ok_or(Errno::EFAULT)?;
239
240        // SAFETY: `libc::rusage` is a plain-old-data C struct that is valid when zero-initialized.
241        let mut usage: libc::rusage = unsafe { std::mem::zeroed() };
242
243        let (user, system) = match who {
244            libc::RUSAGE_THREAD => guest.thread_state_mut().thread_cpu_time(),
245            libc::RUSAGE_CHILDREN => {
246                let cpu = guest.thread_state_mut().process_cpu_time();
247                (cpu.children_user, cpu.children_system)
248            }
249            // RUSAGE_SELF
250            _ => {
251                let cpu = guest.thread_state_mut().process_cpu_time();
252                (cpu.user, cpu.system)
253            }
254        };
255        usage.ru_utime = timeval_from_logical(user);
256        usage.ru_stime = timeval_from_logical(system);
257
258        // RUSAGE_SELF/RUSAGE_THREAD report this process's peak RSS. RUSAGE_CHILDREN aggregates
259        // terminated children only; with no such accounting we leave it zero, matching Linux when
260        // no child has exited.
261        if matches!(who, libc::RUSAGE_SELF | libc::RUSAGE_THREAD) {
262            usage.ru_maxrss = self.guest_peak_rss_kb(guest) as libc::c_long;
263        }
264
265        guest.memory().write_value(usage_addr, &usage)?;
266        Ok(0)
267    }

269    // AUTONOMOUS-BOT-IMPLEMENTED
270    // TODO-HUMAN-REVIEW(#797): Review logical elapsed and process CPU accounting semantics.
271    /// Return deterministic elapsed ticks and process CPU accounting for `times(2)`.
272    ///
273    /// Linux's host boot epoch and scheduler CPU counters are nondeterministic. Detcore instead
274    /// derives the return value from its global logical clock. Per-process logical CPU accounting
275    /// aggregates user instruction and syscall-system time across threads; forked processes start
276    /// fresh counters and contribute their totals to the parent's child counters when reaped.
277    pub async fn handle_times<G: Guest<Self>>(
278        &self,
279        guest: &mut G,
280        call: syscalls::Times,
281    ) -> Result<i64, Error> {
282        let now = thread_observe_time(guest).await;
283        let boot = crate::types::DetTime::new(&self.cfg).as_nanos();
284        let ticks = logical_clock_ticks(now, boot, self.cfg.sysinfo_uptime_offset);
285        let cpu = guest.thread_state_mut().process_cpu_time();
286
287        if let Some(address) = call.buf() {
288            let usage = libc::tms {
289                tms_utime: clock_t_from_ticks(clock_ticks(cpu.user)),
290                tms_stime: clock_t_from_ticks(clock_ticks(cpu.system)),
291                tms_cutime: clock_t_from_ticks(clock_ticks(cpu.children_user)),
292                tms_cstime: clock_t_from_ticks(clock_ticks(cpu.children_system)),
293            };
294            guest.memory().write_value(address, &usage)?;
295        }
296
297        Ok(ticks as i64)
298    }

300    /// The guest's peak resident set size ("high water mark") in kibibytes, matching the units of
301    /// Linux `getrusage`'s `ru_maxrss`. Reads procfs like [`Self::free_ram`]; always returns a
302    /// positive value so guests can rely on a nonzero maximum RSS even if the read fails.
303    fn guest_peak_rss_kb<G: Guest<Self>>(&self, guest: &G) -> u64 {
304        Process::new(guest.pid().as_raw())
305            .and_then(|process| process.status())
306            .ok()
307            .and_then(|status| status.vmhwm.or(status.vmrss))
308            .unwrap_or(0)
309            .max(1)
310    }

313    pub async fn handle_sysinfo<G: Guest<Self>>(
314        &self,
315        guest: &mut G,
316        call: syscalls::Sysinfo,
317    ) -> Result<i64, Error> {
318        let sys_info = self.collect_sysinfo(guest).await?;
319        let mut memory = guest.memory();
320
321        if let Some(info_addr) = call.info() {
322            memory.write_value(info_addr, &sys_info.into())?;
323        }
324        Ok(0)
325    }

327    pub(super) async fn calculate_uptime<G: Guest<Self>>(
328        &self,
329        guest: &mut G,
330    ) -> Result<u64, Error> {
331        let global_time = thread_observe_time(guest).await;
332        Ok(self.cfg.sysinfo_uptime_offset + global_time.as_secs()
333            - crate::types::DetTime::new(&self.cfg).as_nanos().as_secs())
334    }

336    async fn collect_sysinfo<G: Guest<Self>>(
337        &self,
338        guest: &mut G,
339    ) -> Result<syscalls::SysInfo, Error> {
340        Ok(syscalls::SysInfo {
341            uptime: self.calculate_uptime(guest).await?,
342            loads_1: 1,
343            loads_5: 1,
344            loads_15: 1,
345            total_ram: self.cfg.memory,
346            free_ram: self.free_ram(guest, self.cfg.memory)?,
347            buffer_ram: MB,
348            shared_ram: MB,
349            total_swap: 0,
350            free_swap: 0,
351            procs: 1,
352            total_high: 0,
353            free_high: 0,
354            mem_unit: 1,
355        })

359    // TODO-HUMAN-REVIEW(PR-1054): Deterministic free-memory accounting for sysinfo(2).
360    /// Report guest-visible free memory as `total_ram` minus the guest's virtual
361    /// address-space size.
362    ///
363    /// The used-memory figure must be a function of guest computation alone, or
364    /// `sysinfo`'s `free_ram` (and therefore glibc `sysconf(_SC_AVPHYS_PAGES)`)
365    /// becomes nondeterministic across `--verify` and record/replay. The virtual
366    /// size (`statm.size`, i.e. `/proc/<pid>/statm` field 1) is the sum of the
367    /// guest's own mappings, which is fixed by the guest's brk/mmap sequence under
368    /// Detcore's deterministic schedule. The resident set size (`statm.resident`),
369    /// used previously, is physical page residency managed by the host kernel
370    /// (demand paging, reclaim, host memory pressure); it drifts by a page or two
371    /// between otherwise-identical runs and made `getconf -a` flake ~10% at L2.
372    fn free_ram<G: Guest<Self>>(&self, guest: &mut G, total_ram: u64) -> anyhow::Result<u64> {
373        let process = Process::new(guest.pid().as_raw())?;
374        let page_size = procfs::page_size();
375        let statm = process.statm()?;
376        let used_memory = statm.size * page_size;
377        if used_memory > total_ram {
378            return Ok(0);
379        }
380        Ok(total_ram - used_memory)
381    }

409    #[test]
410    fn rusage_timeval_truncates_sub_microsecond_rather_than_rounding() {
411        // 1_999 ns is a hair under 2us. Truncating yields 1us; rounding to nearest would
412        // yield 2us and could make a later, larger duration report a SMALLER value once its
413        // remainder shrank -- i.e. CPU time going backwards. Pin truncation explicitly.
414        let tv = timeval_from_logical(LogicalTime::from_nanos(1_999));
415        assert_eq!(tv.tv_sec, 0);
416        assert_eq!(tv.tv_usec, 1);
417    }

419    #[test]
420    fn rusage_timeval_is_monotonic_in_the_logical_duration() {
421        // The property that matters to a guest: CPU time never goes backwards. Walk a range
422        // of nanosecond durations across microsecond and second boundaries and assert the
423        // rendered timeval is non-decreasing at every step.
424        let mut previous = (0_i64, 0_i64);
425        for nanos in (0..3_000_000u64).step_by(997) {
426            let tv = timeval_from_logical(LogicalTime::from_nanos(nanos));
427            let current = (tv.tv_sec, tv.tv_usec);
428            assert!(
429                current >= previous,
430                "rusage timeval went backwards at {nanos}ns: {previous:?} -> {current:?}"
431            );
432            previous = current;
433        }
434    }

443    #[test]
444    fn rusage_and_times_agree_within_one_clock_tick() {
445        // Both syscalls project the same logical duration, but times(2) is
446        // quantized to USER_HZ while getrusage(2) retains microseconds.
447        for nanos in [0u64, 1_000_000, 300_484_000, 7_000_000_000, 12_345_678_901] {
448            let duration = LogicalTime::from_nanos(nanos);
449            let tv = timeval_from_logical(duration);
450            let rusage_micros = tv.tv_sec as u64 * 1_000_000 + tv.tv_usec as u64;
451            let times_micros = clock_ticks(duration) * (NANOS_PER_CLOCK_TICK / 1_000);
452
453            assert!(rusage_micros >= times_micros);
454            assert!(rusage_micros - times_micros < NANOS_PER_CLOCK_TICK / 1_000);
455
456            // A tick-ALIGNED duration must agree EXACTLY, not merely to within
457            // one tick. The bounds above are satisfied at every sample by an
458            // implementation carrying a constant sub-tick offset, so without
459            // this the suite cannot distinguish that from a correct one.
460            if nanos % NANOS_PER_CLOCK_TICK == 0 {
461                assert_eq!(rusage_micros, times_micros);
462            }
463        }
464    }
