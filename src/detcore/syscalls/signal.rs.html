80/// Signals hermit takes from the guest's namespace, and what a guest loses.
81///
82/// ⚠️ ENUMERATED BY MEASUREMENT, NOT BY READING (2026-08-25). Every signal 1..64
83/// was run under two delivery paths -- self-directed `raise` and a sibling
84/// thread's `pthread_kill` -- with a native run as the control for each. Exactly
85/// TWO differ from native, and they fail in different ways at different points:
86///
87///   SIGSTKFLT (16)  `rt_sigaction` is NO-OPED below, so the handler is never
88///                   installed and the default disposition terminates the guest:
89///                   observed exit 144 (= 128 + 16).
90///   SIGTRAP    (5)  `rt_sigaction` passes through and the handler IS installed,
91///                   but ptrace consumes every SIGTRAP (syscall stops, seccomp
92///                   stops, breakpoints), so the handler never runs. ⚠️ THE GUEST
93///                   THEN EXITS 0 WITH NO DIAGNOSTIC AT ALL -- a clean pass that
94///                   behaved differently from native, which no cell can catch.
95///
96/// Everything else in 1..31 matches native exactly; 9/19 and 32/33 refuse
97/// `sigaction` natively too and are not hermit's. Realtime 34..64 are
98/// delivered and handled like any other signal under ptrace
99/// (<https://github.com/rrnewton/reverie/issues/207>); they are not
100/// appropriation.
101const APPROPRIATED_SIGNALS: [(i32, &str); 2] = [
102    (
103        libc::SIGTRAP,
104        "ptrace consumes every SIGTRAP (syscall/seccomp stops, breakpoints)",
105    ),
106    (
107        libc::SIGSTKFLT,
108        "reverie uses it as PERF_EVENT_SIGNAL, the PMU preemption timer",
109    ),
110];

112/// Say, once per installation, that a guest handler will never run.
113///
114/// ⚠️ WHY A DIAGNOSTIC AND NOT A REFUSAL. Returning `EINVAL` from `sigaction`
115/// was considered and deliberately rejected for SIGSTKFLT -- see the comment at
116/// the no-op below: the Go runtime registers that handler, and refusing would
117/// break every Go guest at startup. That reasoning generalises: installing a
118/// handler defensively is common, actually raising these signals is rare, so
119/// refusal breaks MORE programs than the current behaviour. The contract
120/// question of what a guest is owed here is genuinely open.
121///
122/// What is NOT open is that hermit currently says NOTHING. This line commits to
123/// no policy, breaks no conforming program, and turns a silent wrong answer into
124/// a visible one -- the same reasoning as the `HERMIT_INTERNAL_FAILURE` marker.
125/// The decision, separated from the reporting so a test can exercise THIS and
126/// not a copy of it. A unit test that re-implements a predicate keeps passing
127/// when the real one is gutted -- measured in this project's own pipe work,
128/// where deleting the production wiring left every unit test green.
129fn appropriated_reason(signum: i32, handler: u64) -> Option<&'static str> {
130    // SIG_DFL (0) and SIG_IGN (1) lose nothing: the guest is not asking to be
131    // called back, so there is no expectation to disappoint.
132    if handler <= 1 {
133        return None;
134    }
135    APPROPRIATED_SIGNALS
136        .iter()
137        .find(|(s, _)| *s == signum)
138        .map(|(_, why)| *why)
139}

141fn warn_appropriated_signal(signum: i32, handler: u64) {
142    if let Some(why) = appropriated_reason(signum, handler) {
143        tracing::warn!(
144            "HERMIT_APPROPRIATED_SIGNAL signum={signum} effect=handler-installed-but-never-invoked reason={why}"
145        );
146    }
147}

161/// The mask Linux installs when a call names `mask` as the signal mask to sleep
162/// under (`rt_sigsuspend`, `ppoll`, `pselect6`): `set_current_blocked` drops
163/// `SIGKILL` and `SIGSTOP`, which can never be blocked.
164pub(crate) fn kernel_installed_signal_mask(mask: KernelSigset) -> KernelSigset {
165    let unblockable = (1_u64 << (libc::SIGKILL - 1)) | (1_u64 << (libc::SIGSTOP - 1));
166    mask & !unblockable
167}

169fn without_perf_event_signal(mask: KernelSigset) -> KernelSigset {
170    let bit = (reverie::PERF_EVENT_SIGNAL as u32) - 1;
171    mask & !(1_u64 << bit)
172}

174/// Read one raw kernel signal mask while preserving the kernel's user-access check.
175///
176/// `safeptrace::Stopped::read` deliberately uses `PTRACE_PEEKDATA` for reads of
177/// eight bytes or less. That operation can read a `PROT_NONE` page, unlike the
178/// kernel's `copy_from_user`, so using `MemoryAccess::read_value` alone would
179/// turn an `EFAULT` from a raw signal syscall into success. An invalid `how`
180/// value makes `rt_sigprocmask` copy exactly the kernel-sized input and then
181/// return `EINVAL` without changing the mask. Use that as a permission probe,
182/// then read the already-validated word while the guest is stopped.
183/// The x86-64 System V red zone below the stack pointer, which a leaf function
184/// may use and an injected call therefore must not.
185pub(super) const STACK_RED_ZONE: usize = 128;
186
187/// The two eight-byte scratch cells `handle_rt_sigsuspend` places below the
188/// red zone: the rt_sigpending result and the copy of the call's mask.
189const SCRATCH_CELLS_SIZE: usize = 2 * std::mem::size_of::<u64>();

191pub(super) async fn read_kernel_sigset<G, T>(
192    guest: &mut G,
193    address: Addr<'_, libc::sigset_t>,
194) -> Result<KernelSigset, Error>
195where
196    G: Guest<Detcore<T>>,
197    T: RecordOrReplay,
198{
199    let validation = syscalls::RtSigprocmask::new()
200        .with_how(-1)
201        .with_set(Some(address))
202        .with_oldset(None)
203        .with_sigsetsize(KERNEL_SIGSET_SIZE);
204    match guest.inject(validation).await {
205        Err(Errno::EINVAL) => {}
206        Err(errno) => return Err(errno.into()),
207        Ok(_) => {
208            // Both Linux and the KVM syscall implementation reject an unknown
209            // operation. Success means the backend did not validate the probe.
210            return Err(Errno::EIO.into());
211        }
212    }
213    Ok(guest.memory().read_value(address.cast())?)
214}

216/// Whether a thread-directed SIGKILL (`tgkill` or `tkill`) names `tid`, a
217/// thread-group leader that has exited while a sibling runs on. Linux keeps
218/// such a leader as a zombie, and a SIGKILL that names it does not end its
219/// group: `complete_signal` finds that the exiting leader does not want the
220/// signal, and a thread-directed signal goes to no other thread. So the send
221/// is not reserved as a group kill; it takes the ordinary path, and the
222/// kernel answers it as Linux does, with no effect. Only a leader the
223/// scheduler retired at its own `exit` counts: it commits no other turn until
224/// the kernel shows such a leader a zombie, so by this turn it is one, in
225/// every run (`Scheduler::names_a_retired_exited_leader`). A leader that left
226/// the scheduler at a host-timed deregistration, as on DBT, does not
227/// (https://github.com/rrnewton/hermit/issues/4445).
228pub(crate) async fn sigkill_names_an_exited_leader<G, T>(guest: &mut G, tid: DetTid) -> bool
229where
230    G: Guest<Detcore<T>>,
231    T: RecordOrReplay,
232{
233    names_a_retired_exited_leader(guest, tid).await
234}

252/// Query through the runtime, which reports the guest's action and mask even
253/// when it owns a physical SIGALRM handler. These queries run in an owned turn;
254/// a scheduler-side read of the callback's physical mask is not a substitute.
255async fn read_sigalrm_sleep_state<G, T>(guest: &mut G) -> Result<InGuestSleepSigalrm, Error>
256where
257    G: Guest<Detcore<T>>,
258    T: RecordOrReplay,
259{
260    let mut stack = guest.stack().await;
261    let action = stack.reserve::<KernelSigaction>();
262    let mask = stack.reserve::<KernelSigset>();
263    let _stack_guard = stack.commit()?;
264    guest
265        .inject(
266            syscalls::RtSigaction::new()
267                .with_signum(libc::SIGALRM)
268                .with_action(None)
269                .with_old_action(Some(action.cast()))
270                .with_sigsetsize(KERNEL_SIGSET_SIZE),
271        )
272        .await?;
273    guest
274        .inject(
275            syscalls::RtSigprocmask::new()
276                .with_how(libc::SIG_BLOCK)
277                .with_set(None)
278                .with_oldset(Some(mask.cast()))
279                .with_sigsetsize(KERNEL_SIGSET_SIZE),
280        )
281        .await?;
282    let action: KernelSigaction = guest.memory().read_value(action)?;
283    let mask: KernelSigset = guest.memory().read_value(mask)?;
284    let disposition = if action.handler == libc::SIG_DFL as u64 {
285        SigalrmSleepDisposition::Default
286    } else if action.handler == libc::SIG_IGN as u64 {
287        SigalrmSleepDisposition::Ignored
288    } else {
289        SigalrmSleepDisposition::Handled
290    };
291    Ok(InGuestSleepSigalrm {
292        disposition,
293        blocked: mask & kernel_sigset_bit(libc::SIGALRM) != 0,
294    })
295}

297/// Take one pending copy of `signo` from the calling thread's queues, its own
298/// first and then its process's shared one, as Linux dequeues it, with a
299/// zero-timeout `rt_sigtimedwait` injected while the thread holds its turn.
300/// Returns the signal taken, or the call's error (EAGAIN: none was pending).
301pub(crate) async fn take_pending_signal<G, T>(guest: &mut G, signo: i32) -> Result<i64, Error>
302where
303    G: Guest<Detcore<T>>,
304    T: RecordOrReplay,
305{
306    let mut stack = guest.stack().await;
307    let set = stack.push(kernel_sigset_bit(signo));
308    let timeout = stack.push(Timespec {
309        tv_sec: 0,
310        tv_nsec: 0,
311    });
312    let stack_guard = stack.commit()?;
313    // SAFETY: the kernel only reads the set; Reverie types the argument as
314    // mutable.
315    let set = unsafe { set.cast::<libc::sigset_t>().into_mut() };
316    let taken = guest
317        .inject(
318            syscalls::RtSigtimedwait::new()
319                .with_set(Some(set))
320                .with_info(None)
321                .with_timeout(Some(timeout))
322                .with_sigsetsize(KERNEL_SIGSET_SIZE),
323        )
324        .await;
325    drop(stack_guard);
326    Ok(taken?)
327}

329/// Claim a pending `SIGCHLD` for the calling thread, in its own turn, so that
330/// it is delivered to this thread as it next resumes and no other thread can
331/// take it. Returns whether there was one.
332///
333/// A harmless `getpid` is injected. Reverie runs an injected call by stepping
334/// the thread through user mode, so while a signal is pending and unblocked the
335/// kernel dequeues it at that step, Reverie holds it for delivery as the thread
336/// resumes, and the injection returns a restart errno instead of running. The
337/// caller calls this only while SIGCHLD is the signal the kernel dequeues next,
338/// so that held signal is the copy: claimed. If an outside thread took the copy
339/// first, nothing is pending, the call runs, and nothing is claimed. The call
340/// consumes no signal itself, so a stop that ends it can lose nothing (the Codex
341/// review of a version that took the copy with `rt_sigtimedwait` and queued it
342/// again).
343pub(crate) async fn claim_pending_sigchld<G, T>(guest: &mut G) -> Result<bool, Error>
344where
345    G: Guest<Detcore<T>>,
346    T: RecordOrReplay,
347{
348    match guest.inject(syscalls::Getpid::new()).await {
349        Ok(_) => Ok(false),
350        Err(errno) if is_restart_errno(errno) => Ok(true),
351        Err(errno) => Err(errno.into()),
352    }
353}

355/// Test-only fault hook, like `HERMIT_TEST_TIMERFD_FAULT`, acting once per run
356/// at the first claim of a `SIGCHLD` (`claim_pending_sigchld`), for the orders
357/// of https://github.com/rrnewton/hermit/issues/4346 that host timing alone
358/// gives too rarely to test:
359/// - `usr1-before-claim`: just before the claim, Detcore sends the claiming
360///   thread SIGUSR1 from its own process (an injected send would itself
361///   dequeue the pending SIGCHLD), so the thread's own queue holds a lower
362///   signal and the claim takes SIGUSR1 instead;
363/// - `usr1-after-claim`: just after the claim, Detcore sends the claiming
364///   thread SIGUSR1, which a guest that blocks it leaves pending, so the
365///   backend passes the claimed SIGCHLD on unreported;
366/// - `drop-claimed`: the claimed copy is dropped, as the ledger drops a
367///   duplicate (`Detcore::child_exit_sigchld_delivers`), at its delivery stop
368///   or where the backend passes it on unreported
369///   (`Tool::filter_unreported_signal`), so no handler runs and the sleep
370///   restarts: a relative `nanosleep` through the kernel's `restart_syscall`
371///   (`helpers::RestartCall::Nanosleep`).
372///
373/// Unset, nothing changes.
374const CLAIM_FAULT_ENV: &str = "HERMIT_TEST_SIGCHLD_CLAIM_FAULT";
375
376/// Whether the claim fault hook has fired in this process. It fires once.
377static CLAIM_FAULT_FIRED: std::sync::atomic::AtomicBool = std::sync::atomic::AtomicBool::new(false);
378
379/// A fault `HERMIT_TEST_SIGCHLD_CLAIM_FAULT` asks for.
380#[derive(Clone, Copy, Debug, PartialEq, Eq)]
381pub(crate) enum ClaimFault {
382    Usr1BeforeClaim,
383    Usr1AfterClaim,
384    DropClaimed,
385}
386
387/// Whether the hook asks for `fault` now: at most once per process.
388pub(crate) fn take_claim_fault(fault: ClaimFault) -> bool {
389    let asked = match std::env::var(CLAIM_FAULT_ENV).as_deref() {
390        Ok("usr1-before-claim") => ClaimFault::Usr1BeforeClaim,
391        Ok("usr1-after-claim") => ClaimFault::Usr1AfterClaim,
392        Ok("drop-claimed") => ClaimFault::DropClaimed,
393        _ => return false,
394    };
395    asked == fault && !CLAIM_FAULT_FIRED.swap(true, std::sync::atomic::Ordering::SeqCst)
396}

398/// For `usr1-before-claim` or `usr1-after-claim` (`fault`,
399/// `take_claim_fault`): send the claiming thread SIGUSR1 from Detcore's
400/// process, which shares the guest's pid namespace.
401fn send_claim_fault_signal<G, T>(guest: &G, fault: ClaimFault)
402where
403    G: Guest<Detcore<T>>,
404    T: RecordOrReplay,
405{
406    if take_claim_fault(fault) {
407        // SAFETY: tgkill only sends a signal.
408        let sent = unsafe {
409            libc::syscall(
410                libc::SYS_tgkill,
411                guest.pid().as_raw(),
412                guest.tid().as_raw(),
413                libc::SIGUSR1,
414            )
415        };
416        if sent == 0 {
417            tracing::warn!("claim fault hook: sent SIGUSR1 {:?}", fault);
418        } else {
419            tracing::error!(
420                "claim fault hook: could not send SIGUSR1 {:?}: {}",
421                fault,
422                std::io::Error::last_os_error()
423            );
424        }
425    }
426}

490fn timeval_to_logical_time(value: libc::timeval) -> Result<LogicalTime, Errno> {
491    let seconds = u64::try_from(value.tv_sec).map_err(|_| Errno::EINVAL)?;
492    let micros = u64::try_from(value.tv_usec).map_err(|_| Errno::EINVAL)?;
493    if micros >= 1_000_000 {
494        return Err(Errno::EINVAL);
495    }
496    let nanos = seconds
497        .checked_mul(1_000_000_000)
498        .and_then(|nanos| nanos.checked_add(micros * 1_000))
499        .ok_or(Errno::EINVAL)?;
500    Ok(LogicalTime::from_nanos(nanos))
501}

518/// `alarm()`'s return as Linux computes it (`alarm_setitimer`,
519/// `kernel/time/itimer.c`): the whole seconds left, plus one when the
520/// remainder is at least half a second, or when no whole second is left but
521/// the timer is still active. So an active timer, even one with less than a
522/// microsecond left, returns at least 1
523/// (https://github.com/rrnewton/hermit/issues/4139).
524fn linux_alarm_seconds(value: LogicalTime) -> i64 {
525    let ns = value.as_nanos();
526    let (secs, rem) = (ns / 1_000_000_000, ns % 1_000_000_000);
527    let round_up = (secs == 0 && rem != 0) || rem >= 500_000_000;
528    (secs + u64::from(round_up)) as i64
529}

543/// The siginfo a guest's send `call` of `signal` queues, as the receiver
544/// sees it (4262 design N8-5): the first 48 bytes of the caller's
545/// `siginfo_t` for `rt_sigqueueinfo` and `rt_tgsigqueueinfo`, or Linux's
546/// `SI_USER` (`kill`) or `SI_TKILL` (`tgkill`, `tkill`) record with the
547/// caller's process id and real uid; for `pidfd_send_signal`, the caller's
548/// siginfo or `SI_USER`. `None` when it cannot be read, or for a call
549/// Detcore does not decode, which the scheduler records as `SI_USER`.
550/// The process id is the container's, which is the receiver's unless either
551/// is in a nested PID namespace.
552fn sent_siginfo<T, G: Guest<Detcore<T>>>(
553    guest: &mut G,
554    call: &syscalls::Syscall,
555    signal: i32,
556) -> Option<KernelSiginfo>
557where
558    T: RecordOrReplay,
559{
560    if !crate::scheduler::discarded_at_send::records(guest.config()) {
561        return None;
562    }
563    let code = match call {
564        syscalls::Syscall::RtSigqueueinfo(call) => {
565            return read_siginfo(guest, call.siginfo()?)
566                .map(|bytes| KernelSiginfo::from_user(&bytes));
567        }
568        syscalls::Syscall::RtTgsigqueueinfo(call) => {
569            return read_siginfo(guest, call.siginfo()?)
570                .map(|bytes| KernelSiginfo::from_user(&bytes));
571        }
572        // `pidfd_send_signal` queues the caller's siginfo when it passes one,
573        // and Linux's `SI_USER` record otherwise.
574        syscalls::Syscall::Other(sysno, args) if *sysno == Sysno::pidfd_send_signal => {
575            match AddrMut::<libc::siginfo_t>::from_raw(args.arg2) {
576                Some(info) => {
577                    return read_siginfo(guest, info).map(|bytes| KernelSiginfo::from_user(&bytes));
578                }
579                None => libc::SI_USER,
580            }
581        }
582        syscalls::Syscall::Kill(_) => libc::SI_USER,
583        syscalls::Syscall::Tgkill(_) | syscalls::Syscall::Tkill(_) => libc::SI_TKILL,
584        _ => return None,
585    };
586    let pid = guest.thread_state().detpid?.as_raw();
587    let uid = crate::syscalls::real_uid(guest.tid().as_raw())?;
588    Some(crate::scheduler::signal_lists::ordinary_siginfo(
589        signal, code, pid, uid,
590    ))
591}

593fn read_siginfo<T, G: Guest<Detcore<T>>>(
594    guest: &mut G,
595    info: AddrMut<libc::siginfo_t>,
596) -> Option<[u8; 128]>
597where
598    T: RecordOrReplay,
599{
600    if let Ok(bytes) = guest.memory().read_value(info.cast::<[u8; 128]>()) {
601        return Some(bytes);
602    }
603    use std::os::unix::fs::FileExt;
604    let mem = std::fs::File::open(format!("/proc/{}/mem", guest.tid().as_raw())).ok()?;
605    let mut bytes = [0u8; 128];
606    mem.read_exact_at(&mut bytes, info.as_raw() as u64).ok()?;
607    Some(bytes)
608}

610/// The first bytes of the guest range at `address`, up to `len`, that are
611/// mapped: read a page at a time, through `/proc/<tid>/mem` where the page
612/// lacks read permission (`PROT_WRITE` alone, or `PROT_NONE`), and stopping at
613/// the first page that is not mapped. The kernel's copy to user mode stops at
614/// the first byte it cannot write, so every byte it can change in the range
615/// is in this prefix (the Codex re-check of
616/// https://github.com/rrnewton/hermit/pull/4351, P2-1).
617fn read_mapped_prefix<T, G: Guest<Detcore<T>>>(guest: &mut G, address: usize, len: usize) -> Vec<u8>
618where
619    T: RecordOrReplay,
620{
621    const PAGE: usize = 4096;
622    let mut bytes = Vec::with_capacity(len);
623    let mut mem = None;
624    while bytes.len() < len {
625        let at = address + bytes.len();
626        let take = (PAGE - at % PAGE).min(len - bytes.len());
627        let mut piece = vec![0u8; take];
628        let Some(remote) = AddrMut::<u8>::from_raw(at) else {
629            break;
630        };
631        if guest.memory().read_exact(remote, &mut piece).is_err() {
632            use std::os::unix::fs::FileExt;
633            if mem.is_none() {
634                mem = std::fs::File::open(format!("/proc/{}/mem", guest.tid().as_raw())).ok();
635            }
636            let Some(file) = mem.as_ref() else {
637                break;
638            };
639            if file.read_exact_at(&mut piece, at as u64).is_err() {
640                break;
641            }
642        }
643        bytes.extend_from_slice(&piece);
644    }
645    bytes
646}

648/// The most a signalfd read's records can need restored: one 128-byte record
649/// for each standard signal in each of the two queues it reads (the thread's
650/// and the process's). Real-time signals, which can queue more copies, are
651/// out of scope (https://github.com/rrnewton/hermit/issues/4319).
652const SIGNALFD_CAPTURED: usize = 2 * 64 * SIGNALFD_RECORD;
653
654/// What the guest's buffers `extents` (address, length) hold, in order, up to
655/// `limit` bytes and at most `SIGNALFD_CAPTURED`, read before a signalfd read
656/// writes them, so a record that read drops can leave them as they were
657/// (`apply_signalfd_answers`). The capture stops at the first byte that is not
658/// mapped, which the read cannot write either.
659pub(crate) fn capture_signalfd_buffer<T, G: Guest<Detcore<T>>>(
660    guest: &mut G,
661    extents: &[(usize, usize)],
662    limit: usize,
663) -> Vec<u8>
664where
665    T: RecordOrReplay,
666{
667    let total: usize = extents.iter().map(|(_, len)| *len).sum();
668    let len = total.min(limit).min(SIGNALFD_CAPTURED);
669    let mut bytes = Vec::with_capacity(len);
670    for (address, length) in signalfd_extent_pieces(extents, len) {
671        let piece = read_mapped_prefix(guest, address, length);
672        let short = piece.len() < length;
673        bytes.extend_from_slice(&piece);
674        if short {
675            break;
676        }
677    }
678    bytes
679}

681/// Put back, in the guest's siginfo buffer at `info`, the bytes that differ
682/// from `before`, the mapped prefix of its 128 bytes before a take whose copy
683/// is dropped (`read_mapped_prefix`). Only the bytes the kernel wrote differ,
684/// and it wrote only what user mode could, so no write reaches a page user
685/// mode could not write; a buffer that ends at an unmapped page keeps its
686/// prefix as it was (the Codex re-check of
687/// https://github.com/rrnewton/hermit/pull/4351, P2-1).
688fn restore_changed_bytes<T, G: Guest<Detcore<T>>>(
689    guest: &mut G,
690    info: AddrMut<libc::siginfo_t>,
691    before: &[u8],
692) where
693    T: RecordOrReplay,
694{
695    let now = read_mapped_prefix(guest, info.as_raw(), before.len());
696    let before = &before[..now.len()];
697    let mut at = 0;
698    while at < before.len() {
699        if now[at] == before[at] {
700            at += 1;
701            continue;
702        }
703        let start = at;
704        while at < before.len() && now[at] != before[at] {
705            at += 1;
706        }
707        if let Some(address) = AddrMut::<u8>::from_raw(info.as_raw() + start) {
708            let _ = guest.memory().write_exact(address, &before[start..at]);
709        }
710    }
711}

713/// The thread a process-directed signal goes to. `zombie` is the process's
714/// exit state (`Scheduler::exited_process_is_zombie`). `Ok(None)` is a zombie:
715/// no live threads, but its parent can still wait for it, so Linux still finds
716/// the process and the send succeeds (https://github.com/rrnewton/hermit/issues/4091).
717fn deterministic_kill_target(
718    targets: &[DetTid],
719    zombie: Option<bool>,
720    sig: libc::c_int,
721) -> Result<Option<DetTid>, Errno> {
722    match targets {
723        [] if zombie == Some(true) => Ok(None),
724        [] => Err(Errno::ESRCH),
725        [target] => Ok(Some(*target)),
726        [target, ..] if sig == 0 => Ok(Some(*target)),
727        _ => Err(Errno::ENOSYS),
728    }
729}

791        if !self.cfg.backend.virtualizes_guest_sigalrm {
792            self.stop_after_in_guest_sleep_signal_loss(

799            Ok(InGuestSleepSigalrm {

805                    guest,
806                    format!("its default/unblocked SIGALRM sleep snapshot changed: {other:?}"),

818                "could not stage its default-fatal SIGALRM after a sleep wake".into(),

944    /// A pause is really just an unbounded sleep.
945    pub async fn handle_pause<G: Guest<Self>>(
946        &self,
947        guest: &mut G,
948        call: syscalls::Pause,
949    ) -> Result<i64, Error> {
950        if guest.config().sequentialize_threads {
951            // `pause` has no deadline: it returns only when a signal is delivered.
952            // `LogicalTime::INDEFINITE` records that, and the scheduler refuses to
953            // fast-forward virtual time onto it (see `step2d_handle_empty_queue`),
954            // so the `Normal` arm below stays unreachable.
955            loop {
956                // A signal that already ended the call before it parked: one sent
957                // while this thread was stopped short of filing the request, as
958                // when its timeslice ended in the syscall's prehook.
959                if sleep_signal(guest, SleepCheck::BeforePark, true).await? == SleepSignal::Pending
960                {
961                    return Err(Errno::ERESTARTNOHAND.into());
962                }
963                let mut req = Self::sleep_request_abs(guest, LogicalTime::INDEFINITE).await;
964                req.fyi(crate::resources::PAUSE_FYI);
965                req.in_guest_sleep_sigalrm = self.in_guest_sleep_sigalrm_snapshot(guest).await;
966                match crate::tool_global::parked_wait_request(
967                    guest,
968                    req,
969                    crate::scheduler::parked::ParkedWaitPolicy::PauseNoHandlerRestart,
970                )
971                .await
972                {
973                    ResumeStatus::InGuestFatalAlarm => {
974                        self.stage_in_guest_fatal_alarm(guest).await;
975                        return Err(Errno::EINTR.into());
976                    }
977                    // A pause never asks for `NothingElseCanRun`.
978                    ResumeStatus::Normal | ResumeStatus::NothingElseCanRun => {
979                        panic!(
980                            "Internal violation: pause should never return from the scheduler except by interruption!"
981                        )
982                    }
983                    ResumeStatus::Signaled(signals) => {
984                        match sleep_signal(guest, SleepCheck::AfterWake(signals), true).await? {
985                            SleepSignal::Pending | SleepSignal::Claimed => {
986                                return Err(Errno::ERESTARTNOHAND.into());
987                            }
988                            SleepSignal::Held => return Err(Errno::EINTR.into()),
989                            SleepSignal::None => {
990                                info!(
991                                    "[dtid {}] pause woken for a signal no longer pending for it; waiting again",
992                                    guest.thread_state().dettid
993                                );
994                            }
995                        }
996                    }
997                }
998            }
999        } else {
1000            info!(
1001                "[dtid {}] Running without scheduler, so letting pause call through...",
1002                guest.thread_state().dettid
1003            );
1004            Ok(guest.inject(call).await?)
1005        }
1006    }

1008    /// Run rt_sigsuspend, without holding the deterministic scheduler turn where
1009    /// the call sleeps.
1010    ///
1011    /// The kernel must perform the temporary mask swap atomically and restore the
1012    /// original mask after signal delivery, so execute the real blocking syscall
1013    /// while marking this thread as blocked outside the runnable set.
1014    ///
1015    /// On a backend whose `rt_sigsuspend` never sleeps
1016    /// (`rt_sigsuspend_never_sleeps`: KVM), the call instead runs in the
1017    /// caller's own turn, and a refused wait yields the turn
1018    /// (<https://github.com/rrnewton/hermit/issues/4360>).
1019    pub async fn handle_rt_sigsuspend<G: Guest<Self>>(
1020        &self,
1021        guest: &mut G,
1022        call: syscalls::RtSigsuspend,
1023    ) -> Result<i64, Error> {
1024        // Invalid arguments return immediately from the kernel and therefore are
1025        // not signal-only waits.
1026        validate_kernel_sigset_size(call.sigsetsize())?;
1027        let Some(mask_addr) = call.mask() else {
1028            return Err(Errno::EFAULT.into());
1029        };
1030
1031        let temporary_mask = read_kernel_sigset(guest, mask_addr).await?;
1032        super::signal_shadow::note_wait_mask(guest, temporary_mask);
1033        // The temporary mask lets the kernel take a pending real-time timer
1034        // expiration that every thread blocked when it expired
1035        // (https://github.com/rrnewton/hermit/issues/4089).
1036        observe_under_mask(guest, temporary_mask).await?;
1037        // The real call sleeps under a private copy of the mask read here, not
1038        // under the guest's buffer. The scheduler decides from this copy which
1039        // signals can end the call (`Resources::blocked_signal_mask`), and
1040        // another process sharing the buffer (MAP_SHARED) may rewrite it before
1041        // the call runs; Linux copies the mask when the call starts, and from
1042        // the guest's view the call has already started.
1043        //
1044        // The copy and the rt_sigpending result live in two cells just below
1045        // the stack's red zone, the scratch area an injected call may use, but
1046        // never on the caller's buffer: a guest may keep its mask there while
1047        // it blocks every signal, and a copy placed on that very buffer would
1048        // be the original again, while the pending cell would overwrite the
1049        // guest's mask. If the buffer overlaps the two cells, they move to just
1050        // below it. They are written directly rather than through a scratch
1051        // stack, whose commit writes one contiguous region down from the red
1052        // zone and so would cover a buffer inside it.
1053        let caller_mask = mask_addr.as_raw()..mask_addr.as_raw() + KERNEL_SIGSET_SIZE;
1054        let below_red_zone = (guest.regs().await.rsp as usize).wrapping_sub(STACK_RED_ZONE);
1055        let cells = (below_red_zone - SCRATCH_CELLS_SIZE)..below_red_zone;
1056        let cells_start = if cells.start < caller_mask.end && caller_mask.start < cells.end {
1057            (caller_mask.start - SCRATCH_CELLS_SIZE) & !(std::mem::align_of::<u64>() - 1)
1058        } else {
1059            cells.start
1060        };
1061        let pending_addr = Addr::<u64>::from_raw(cells_start).ok_or(Errno::EFAULT)?;
1062        let mask_copy = AddrMut::<u64>::from_raw(cells_start + std::mem::size_of::<u64>())
1063            .ok_or(Errno::EFAULT)?;
1064        guest.memory().write_value(mask_copy, &temporary_mask)?;
1065        let pending_out = AddrMut::<libc::sigset_t>::from_raw(pending_addr.as_raw())
1066            .expect("stack address must be non-null");
1067        let pending_call = syscalls::RtSigpending::new()
1068            .with_set(Some(pending_out))
1069            .with_sigsetsize(KERNEL_SIGSET_SIZE);
1070        guest.inject_with_retry(pending_call).await?;
1071        let pending: u64 = guest.memory().read_value(pending_addr)?;
1072        let call = call.with_mask(Some(Addr::from_raw(mask_copy.as_raw()).expect("non-null")));
1073
1074        // On a backend whose `rt_sigsuspend` never sleeps
1075        // (`rt_sigsuspend_never_sleeps`), the call returns at once, so it runs
1076        // in this thread's own turn, as any call that cannot block does.
1077        // Backgrounded as a sleeper, it left nothing holding the scheduler
1078        // between the background grant and its continuation, so the scheduler
1079        // reported a deadlock or skipped virtual time to a timed waiter, as
1080        // host timing decided (https://github.com/rrnewton/hermit/issues/4360).
1081        //
1082        // The backend refuses a wait for a signal not yet pending (ENOSYS), and
1083        // a guest retries it, so that refusal hands the turn on first with a
1084        // strong yield (`ResourceID::SchedYield`): the caller is not selected
1085        // again until another thread has had a turn, and under chaos not until
1086        // a thread that did not yield has (a yield epoch), so the threads it
1087        // waits on run before it retries, in the order the scheduler gives. In
1088        // its own turn alone, the caller spun and never let its child's exit
1089        // be granted; a weak yield, which `sched_yield` makes under chaos,
1090        // would let a caller in a higher band win every turn (the Codex
1091        // review).
1092        if rt_sigsuspend_never_sleeps(&self.cfg.backend) {
1093            let result = self.record_or_replay(guest, call).await;
1094            if result == Err(Errno::ENOSYS) && self.cfg.sequentialize_threads {
1095                if self.cfg.max_timeslice.is_some() {
1096                    guest
1097                        .thread_state_mut()
1098                        .reset_timeslice_for_explicit_yield();
1099                }
1100                let request = Self::sched_yield_request(guest);
1101                resource_request(guest, request).await;
1102            }
1103            return Ok(result?);
1104        }
1105        // The scheduler records the mask the call sleeps under, read here in
1106        // this thread's turn, to choose a SIGCHLD target while the thread is
1107        // outside the runnable set (`Resources::blocked_signal_mask`).
1108        let installed_mask = kernel_installed_signal_mask(temporary_mask);
1109        if pending & !temporary_mask != 0 {
1110            // The kernel will consume an already-pending signal as soon as it
1111            // atomically installs the temporary mask. Keep this immediate case
1112            // out of the terminal-wait classification; the real syscall still
1113            // performs delivery and restores the old mask.
1114            self.record_or_replay_blocking_with_mask(guest, call.into(), Some(installed_mask))
1115                .await
1116        } else {
1117            self.record_or_replay_rt_sigsuspend(guest, call, installed_mask)
1118                .await
1119        }
1120    }

1122    /// rt_sigaction
1123    pub async fn handle_rt_sigaction<G: Guest<Self>>(
1124        &self,
1125        guest: &mut G,
1126        call: syscalls::RtSigaction,
1127    ) -> Result<i64, Error> {
1128        // Linux rejects an invalid size before inspecting either user pointer.
1129        // Preserve that EINVAL-before-EFAULT ordering even for the signal that
1130        // Hermit reserves for deterministic preemption.
1131        validate_kernel_sigset_size(call.sigsetsize())?;
1132
1133        // Copy the complete kernel object before inspecting the signal number or
1134        // writing old_action. This preserves Linux's action-before-old_action
1135        // pointer ordering, including when both arguments alias.
1136        let kernel_action = match call.action() {
1137            Some(action) => Some(read_kernel_sigaction(guest, action).await?),
1138            None => None,
1139        };
1140
1141        // Both appropriated signals are reported here, at the one point where the
1142        // guest states its expectation. SIGTRAP falls through to the ordinary
1143        // path below (its handler really is installed; ptrace just eats the
1144        // signal), so this must run before the SIGSTKFLT early return.
1145        if let Some(action) = kernel_action {
1146            warn_appropriated_signal(call.signum(), action.handler);
1147        }
1148
1149        // PERF_EVENT_SIGNAL is reserved.
1150        if call.signum() == reverie::PERF_EVENT_SIGNAL as i32 {
1151            // The go runtime attempts to register this (unused) signal handler.  We will never
1152            // deliver signals of this kind to the guest, so we just turn this action into a noop
1153            // rather than returning `Err(Errno::EINVAL.into())`. Preserve that established
1154            // policy while still honoring the raw syscall's pointer accesses: the virtual old
1155            // disposition is the default action, never Reverie's private preemption handler.
1156            if call.old_action().is_some() {
1157                // SIGKILL's disposition cannot be changed by userspace, so its
1158                // query copies the same four zero words as our virtual default.
1159                // Let the kernel copy them: a short write_value can finish with
1160                // PTRACE_POKEDATA and overwrite a protected final word. This also
1161                // preserves the kernel's partial-copy effects on EFAULT without
1162                // exposing Reverie's private preemption action.
1163                return Ok(guest
1164                    .inject(call.with_signum(libc::SIGKILL).with_action(None))
1165                    .await?);
1166            }
1167            return Ok(0);
1168        }
1169        // Signal phase 1 (step I3): on a backend that keeps a handled SIGALRM
1170        // virtual, the scheduler's ledger must agree with the runtime.
1171        let phase1 = self.cfg.backend.virtualizes_guest_sigalrm
1172            && call.signum() == libc::SIGALRM
1173            && kernel_action.is_some();
1174        let was_handled = guest.thread_state().sigalrm_handled;
1175        let to_handler = kernel_action.is_some_and(|action| {
1176            action.handler != libc::SIG_DFL as u64 && action.handler != libc::SIG_IGN as u64
1177        });
1178        if phase1 {
1179            if to_handler {
1180                self.refuse_sigalrm_handler(guest).await?;
1181            } else if was_handled
1182                && kernel_action.is_some_and(|action| action.handler == libc::SIG_DFL as u64)
1183            {
1184                refuse_sigalrm(guest, SigalrmControl::HandlerToDefault).await?;
1185            }
1186        }
1187        // Whether this call moves the signal from `SIG_IGN` to an action that
1188        // does not ignore it, read before the change: Linux then re-queues a
1189        // periodic POSIX timer's ignored expiry (`posixtimer_sig_unignore`,
1190        // <https://github.com/rrnewton/hermit/issues/4293>).
1191        let unignores = crate::scheduler::discarded_at_send::records(guest.config())
1192            && (1..32).contains(&call.signum())
1193            && kernel_action.is_some_and(|action| {
1194                action.handler != libc::SIG_IGN as u64
1195                    && !(action.handler == libc::SIG_DFL as u64
1196                        && [libc::SIGCONT, libc::SIGCHLD, libc::SIGWINCH, libc::SIGURG]
1197                            .contains(&call.signum()))
1198            })
1199            && crate::syscalls::threads::read_kernel_signal_state(guest.pid(), guest.tid())
1200                .is_ok_and(|state| state.ignored & kernel_sigset_bit(call.signum()) != 0);
1201        let result = if let Some(kernel_action) = kernel_action {
1202            // The kernel treats `action` as input. Sanitize a private copy instead
1203            // of writing through the guest pointer: the input may be read-only,
1204            // and changing it would make the syscall wrapper observable.
1205            let mut kernel_action = kernel_action;
1206            kernel_action.mask = without_perf_event_signal(kernel_action.mask);
1207            let mut stack = guest.stack().await;
1208            let sanitized_action = stack.push(kernel_action);
1209            let _stack_guard = stack.commit()?;
1210            guest
1211                .inject(call.with_action(Some(sanitized_action.cast())))
1212                .await
1213        } else {
1214            guest.inject(call).await
1215        };
1216        super::signal_shadow::record_rt_sigaction(
1217            guest,
1218            call.signum(),
1219            kernel_action.map(|action| KernelSigaction {
1220                mask: without_perf_event_signal(action.mask),
1221                ..action
1222            }),
1223            call.old_action(),
1224            &result,
1225        );
1226        // Publish the state the runtime now holds whenever the disposition may
1227        // have changed: from a handler, or to one the runtime did not refuse.
1228        // Any result counts, since an EFAULT copying the old action out comes
1229        // after the change. A default or ignored action set while none is
1230        // handled changes nothing the scheduler records, so it costs nothing.
1231        if phase1 && (was_handled || (to_handler && result != Err(Errno::EPERM))) {
1232            self.publish_virtual_sigalrm(guest).await?;
1233        }
1234        // Linux reaps a process's children automatically at exit when its
1235        // SIGCHLD action is SIG_IGN or carries SA_NOCLDWAIT, and a signal to
1236        // such a child then finds no zombie. Mirror the new action for the
1237        // scheduler's exit model, with the same success rule as above
1238        // (https://github.com/rrnewton/hermit/issues/4091).
1239        if call.signum() == libc::SIGCHLD
1240            && guest.config().sequentialize_threads
1241            && let Some(action) = kernel_action
1242            && matches!(result, Ok(_) | Err(Errno::EFAULT))
1243        {
1244            let dettid = guest.thread_state().dettid;
1245            if let Some(process) = registered_process_of(guest, dettid).await {
1246                set_sigchld_disposition(
1247                    guest,
1248                    process,
1249                    action.handler == libc::SIG_IGN as u64,
1250                    action.flags & libc::SA_NOCLDWAIT as u64 != 0,
1251                )
1252                .await;
1253            }
1254        }
1255        // Linux flushes a pending SIGCHLD when its action becomes SIG_IGN or
1256        // SIG_DFL, which ignores it (`do_sigaction`, `sig_handler_ignored`): the
1257        // child-exit ledger's send window goes with it, in this committed turn
1258        // (https://github.com/rrnewton/hermit/issues/4126).
1259        if call.signum() == libc::SIGCHLD
1260            && guest.config().sequentialize_threads
1261            && matches!(result, Ok(_) | Err(Errno::EFAULT))
1262            && kernel_action.is_some_and(|action| {
1263                action.handler == libc::SIG_IGN as u64 || action.handler == libc::SIG_DFL as u64
1264            })
1265        {
1266            crate::tool_global::child_exit_sigchld(
1267                guest,
1268                crate::scheduler::child_exit_sigchld::ChildExitSigchldControl::Flushed,
1269            )
1270            .await;
1271        }
1272        // Linux flushes the pending instances of a signal whose disposition
1273        // becomes SIG_IGN (`do_sigaction`), among them a pending real-time timer
1274        // expiration (https://github.com/rrnewton/hermit/issues/4089).
1275        if result.is_ok()
1276            && kernel_action.is_some_and(|action| action.handler == libc::SIG_IGN as u64)
1277            && (1..=64).contains(&call.signum())
1278        {
1279            let bit = kernel_sigset_bit(call.signum());
1280            let pending = pending_real_time_among(guest, bit).await;
1281            act_on_pending_real_time(guest, pending, 0, bit).await;
1282        }
1283        // The same flush takes a `SIGEV_THREAD_ID` timer's pending copy of a
1284        // standard signal, for SIG_IGN or for SIG_DFL on a signal whose
1285        // default is to ignore it (`sig_handler_ignored`), so it no longer
1286        // counts as the timer's own (https://github.com/rrnewton/hermit/issues/4164).
1287        let ignored = |handler: u64| {
1288            handler == libc::SIG_IGN as u64
1289                || (handler == libc::SIG_DFL as u64
1290                    && [libc::SIGCONT, libc::SIGCHLD, libc::SIGWINCH, libc::SIGURG]
1291                        .contains(&call.signum()))
1292        };
1293        // The kernel reads the new action from Detcore's sanitized copy, so an
1294        // EFAULT can only come from writing `old_action`, after the new action
1295        // is installed.
1296        if (result.is_ok() || matches!(result, Err(Errno::EFAULT)))
1297            && kernel_action.is_some_and(|action| ignored(action.handler))
1298            && (1..32).contains(&call.signum())
1299            && crate::scheduler::timer_copies_recorded()
1300        {
1301            crate::tool_global::forget_thread_timer_copies(guest, call.signum()).await;
1302        }
1303        // A handler's mask decides where a POSIX timer's next entry goes when a
1304        // delivery promotes it (<https://github.com/rrnewton/hermit/issues/4262>).
1305        if crate::scheduler::discarded_at_send::records(guest.config())
1306            && (result.is_ok() || matches!(result, Err(Errno::EFAULT)))
1307            && (1..32).contains(&call.signum())
1308            && let Some(action) = kernel_action
1309        {
1310            crate::tool_global::discarded_signal(
1311                guest,
1312                DiscardedSignalControl::Handler {
1313                    signal: call.signum(),
1314                    caught: action.handler != libc::SIG_IGN as u64
1315                        && action.handler != libc::SIG_DFL as u64,
1316                    mask: action.mask,
1317                    nodefer: action.flags & libc::SA_NODEFER as u64 != 0,
1318                },
1319            )
1320            .await;
1321        }
1322        // The flush takes every pending copy of the signal from every queue of
1323        // the process, so none of them is one Linux discarded any more; and an
1324        // unignore makes a still-current periodic timer expiry a kept one
1325        // (`scheduler::discarded_at_send`).
1326        if crate::scheduler::discarded_at_send::records(guest.config())
1327            && (result.is_ok() || matches!(result, Err(Errno::EFAULT)))
1328            && (1..32).contains(&call.signum())
1329        {
1330            let signal = call.signum();
1331            if kernel_action.is_some_and(|action| ignored(action.handler)) {
1332                crate::tool_global::discarded_signal(
1333                    guest,
1334                    DiscardedSignalControl::Flushed { signal },
1335                )
1336                .await;
1337            } else if unignores {
1338                crate::tool_global::discarded_signal(
1339                    guest,
1340                    DiscardedSignalControl::Unignored { signal },
1341                )
1342                .await;
1343            }
1344        }
1345        Ok(result?)
1346    }

1348    /// Signal phase 1's Detcore-side refusals of a new guest SIGALRM handler,
1349    /// each EPERM before anything changes (design section 6 and closure 5):
1350    /// threads not sequentialized (`alarm` would arm a kernel timer), a
1351    /// signalfd in the process's descriptor table, and a kernel producer of
1352    /// SIGALRM armed anywhere in the run.
1353    async fn refuse_sigalrm_handler<G: Guest<Self>>(&self, guest: &mut G) -> Result<(), Error> {
1354        if !guest.config().sequentialize_threads {
1355            return Err(Errno::EPERM.into());
1356        }
1357        // The model's own signalfds, and any the kernel's table holds that
1358        // the model labels otherwise (an inherited descriptor).
1359        let holds_signalfd = guest
1360            .thread_state()
1361            .file_metadata
1362            .lock()
1363            .expect("file metadata mutex poisoned")
1364            .file_handles
1365            .values()
1366            .any(|fd| fd.ty() == FdType::Signalfd)
1367            || crate::sigalrm_phase1::process_holds_signalfd(guest.pid().as_raw());

1369            return Err(Errno::EPERM.into());
1370        }
1371        Ok(refuse_sigalrm(guest, SigalrmControl::InstallHandler).await?)
1372    }

1400    /// rt_sigprocmask
1401    pub async fn handle_rt_sigprocmask<G: Guest<Self>>(
1402        &self,
1403        guest: &mut G,
1404        call: syscalls::RtSigprocmask,
1405    ) -> Result<i64, Error> {
1406        // The kernel checks sigsetsize before copying from either user pointer.
1407        validate_kernel_sigset_size(call.sigsetsize())?;
1408
1409        let (result, validated_set) = self.rt_sigprocmask_inner(guest, call).await;
1410        if guest.thread_state().signal_shadow.is_some() {
1411            use super::signal_shadow::SetInput;
1412            let errno_result = match &result {
1413                Ok(value) => Some(Ok(*value)),
1414                Err(Error::Errno(errno)) => Some(Err(*errno)),
1415                Err(_) => {
1416                    super::signal_shadow::give_up(
1417                        guest,
1418                        "rt_sigprocmask failed without an errno, so its effect is unknown",
1419                    );
1420                    None
1421                }
1422            };
1423            let set = match (call.set(), validated_set) {
1424                (None, _) => SetInput::None,
1425                (Some(_), ValidatedSet::Validated(set)) => {
1426                    SetInput::Validated(without_perf_event_signal(set))
1427                }
1428                (Some(_), ValidatedSet::NotRun) => SetInput::NotRun,
1429                // Read after the call, from memory the kernel itself just
1430                // read, and only when it succeeded; an old mask written over
1431                // any of the set's bytes leaves it unknown.
1432                (Some(set), ValidatedSet::NotValidated) => SetInput::Unvalidated(
1433                    (result.is_ok()
1434                        && !call.oldset().is_some_and(|old| {
1435                            old.as_raw().abs_diff(set.as_raw()) < KERNEL_SIGSET_SIZE
1436                        }))
1437                    .then(|| {
1438                        guest
1439                            .memory()
1440                            .read_value::<_, KernelSigset>(set.cast::<KernelSigset>())
1441                            .ok()
1442                    })
1443                    .flatten()
1444                    .map(without_perf_event_signal),
1445                ),
1446            };
1447            if let Some(errno_result) = errno_result {
1448                super::signal_shadow::record_rt_sigprocmask(
1449                    guest,
1450                    call.how(),
1451                    set,
1452                    call.oldset(),
1453                    &errno_result,
1454                );
1455            }
1456        }
1457        if call.how() != libc::SIG_BLOCK && call.set().is_some() {
1458            // The new mask may unblock a pending real-time timer expiration
1459            // that every thread blocked when it expired, which the kernel then
1460            // delivers (https://github.com/rrnewton/hermit/issues/4089). Whatever
1461            // the call returned: an EFAULT copying the old set out comes after
1462            // the change. A mask that cannot be read could unblock any of them.
1463            let pending = pending_real_time_among(guest, REAL_TIME_SIGNALS).await;
1464            if pending != 0 {
1465                let (observable, discarded) =
1466                    match read_thread_wait_signal_state(guest.pid(), guest.tid()) {
1467                        Ok(state) => unblocked_by(&state, state.blocked),
1468                        Err(_) => (REAL_TIME_SIGNALS, 0),
1469                    };
1470                act_on_pending_real_time(guest, pending, observable, discarded).await;
1471            }
1472        }
1473        if self.cfg.backend.virtualizes_guest_sigalrm && guest.thread_state().sigalrm_handled {
1474            // Signal phase 1: the scheduler's blocked bit follows the virtual
1475            // mask, whatever the call returned (an EFAULT copying the old set
1476            // out comes after the change).
1477            self.publish_virtual_sigalrm_blocked(guest).await?;
1478        }
1479        result
1480    }

1482    /// The call itself. Also says whether, and with what set, the kernel ran
1483    /// it: the `SIG_BLOCK` and `SIG_SETMASK` paths read the set through the
1484    /// kernel first and inject nothing when that fails.
1485    async fn rt_sigprocmask_inner<G: Guest<Self>>(
1486        &self,
1487        guest: &mut G,
1488        call: syscalls::RtSigprocmask,
1489    ) -> (Result<i64, Error>, ValidatedSet) {
1490        if call.how() != libc::SIG_BLOCK && call.how() != libc::SIG_SETMASK {
1491            let result = guest.inject_with_retry(call).await;
1492            (result.map_err(Into::into), ValidatedSet::NotValidated)
1493        } else if let Some(set) = call.set() {
1494            let set_mask = match read_kernel_sigset(guest, set).await {
1495                Ok(set_mask) => set_mask,
1496                Err(error) => return (Err(error), ValidatedSet::NotRun),
1497            };
1498            let mut stack = guest.stack().await;
1499            let new_set = stack.push(without_perf_event_signal(set_mask));
1500            let _stack_guard = match stack.commit() {
1501                Ok(guard) => guard,
1502                Err(error) => return (Err(error.into()), ValidatedSet::NotRun),
1503            };
1504            let modified_call = syscalls::RtSigprocmask::new()
1505                .with_how(call.how())
1506                .with_set(Some(new_set.cast()))
1507                .with_oldset(call.oldset())
1508                .with_sigsetsize(call.sigsetsize());
1509            // Keep returning to the handler so post_handler_hook can run, but
1510            // do not expose a tracer preemption as ERESTARTSYS to the guest.
1511            let result = guest.inject_with_retry(modified_call).await;
1512            (
1513                result.map_err(Into::into),
1514                ValidatedSet::Validated(set_mask),
1515            )
1516        } else {
1517            let result = guest.inject_with_retry(call).await;
1518            (result.map_err(Into::into), ValidatedSet::NotValidated)
1519        }
1520    }

1522    /// rt_sigtimedwait system call
1523    ///
1524    /// This is handled by the scheduler and not passed to the record/replay layer,
1525    /// because currently signals are not recorded.
1526    pub async fn handle_rt_sigtimedwait<G: Guest<Self>>(
1527        &self,
1528        guest: &mut G,
1529        call: syscalls::RtSigtimedwait,
1530    ) -> Result<i64, Error> {
1531        // Linux rejects an invalid mask width before inspecting its pointers.
1532        validate_kernel_sigset_size(call.sigsetsize())?;
1533        // Linux keeps this entry snapshot for the whole wait. The retry helper
1534        // currently injects from the caller's pointer again, so another thread
1535        // can change the set between retries. Fixing that requires sharing one
1536        // scratch-stack guard with the helper's zero-timeout object; do not hide
1537        // that remaining difference by treating this validation read as a snapshot.
1538
1539        let dettid = guest.thread_state().dettid;
1540
1541        let maybe_timeout = if let Some(timeout) = call.timeout() {
1542            let ts: Timespec = guest.memory().read_value(timeout)?;
1543            let ns_delta =
1544                Duration::from_secs(ts.tv_sec as u64) + Duration::from_nanos(ts.tv_nsec as u64);
1545            let base_time = thread_observe_time(guest).await;
1546            let target_time = base_time + ns_delta;
1547            Some(target_time)
1548        } else {
1549            None
1550        };
1551        let mut rsrc = Resources::new(dettid);
1552        rsrc.insert(ResourceID::InternalIOPolling, Permission::W);
1553        rsrc.fyi("rt_sigtimedwait");
1554        // While it polls, a signal in the set is taken at the next poll
1555        // whatever the thread's mask, which a process-directed signal another
1556        // thread sends must account for
1557        // (`Scheduler::own_process_signal_inputs`). An unreadable set fails in
1558        // the kernel at the first poll.
1559        let waited = match call.set() {
1560            Some(set) => read_kernel_sigset(guest, set.into()).await.ok(),
1561            None => None,
1562        };
1563        if waited.is_some() {
1564            set_sigtimedwait_set(guest, waited).await;
1565        }
1566        // Linux writes the siginfo only for a signal it returns. A dropped
1567        // copy's siginfo is replaced by what the buffer held before.
1568        let records = crate::scheduler::discarded_at_send::records(guest.config());
1569        let before = match call.info() {
1570            Some(info) if self.cfg.backend.reports_child_exit_publication || records => {
1571                Some(read_mapped_prefix(guest, info.as_raw(), 128))
1572            }
1573            _ => None,
1574        };
1575        let result = loop {
1576            let result =
1577                retry_nonblocking_syscall_with_timeout(guest, call, rsrc.clone(), maybe_timeout)
1578                    .await;
1579            // A consumed child-exit copy that the notification ledger drops
1580            // is not returned: the wait goes on, to the same deadline.
1581            let stands = self
1582                .synchronous_sigchld_stands(guest, call, waited, &result)
1583                .await;
1584            // Nor is a copy Linux discarded when it was sent, which Linux never
1585            // dequeues (https://github.com/rrnewton/hermit/issues/4293). The
1586            // take went into the guest's own buffer, so the kernel made the
1587            // copy-out with its own partial-write and EFAULT rules for a kept
1588            // copy; for a dropped one, the bytes it changed are put back.
1589            let outcome = if records {
1590                self.take_outcome_of(guest, waited, &result).await
1591            } else {
1592                crate::scheduler::posix_timer_coalesce::TakeOutcome {
1593                    skip: false,
1594                    rewrite: None,
1595                }
1596            };
1597            if stands && !outcome.skip {
1598                // A POSIX timer's entry, or a kept signal standing on another
1599                // sender's record, is returned with its own siginfo
1600                // (<https://github.com/rrnewton/hermit/issues/4262>); the
1601                // kernel's copy-out succeeded, so the buffer is writable.
1602                if let (Ok(_), Some(info), Some(rewrite)) = (&result, call.info(), outcome.rewrite)
1603                    && guest
1604                        .memory()
1605                        .write_value(info.cast::<[u8; 128]>(), &rewrite.to_user())
1606                        .is_err()
1607                {
1608                    tracing::warn!(
1609                        "[dtid {}] rt_sigtimedwait's siginfo keeps the physical record",
1610                        guest.thread_state().dettid
1611                    );
1612                }
1613                break result;
1614            }
1615            if let (Some(info), Some(bytes)) = (call.info(), before.as_ref()) {
1616                restore_changed_bytes(guest, info, bytes);
1617            }
1618        };
1619        if waited.is_some() {
1620            set_sigtimedwait_set(guest, None).await;
1621        }
1622        // O2: any completion of a wait for SIGALRM may have dequeued the
1623        // process's ITIMER_REAL copy, including one that then failed copying
1624        // out the siginfo (https://github.com/rrnewton/hermit/issues/4139).
1625        if waited.is_some_and(|set| {
1626            set & crate::syscalls::threads::kernel_sigset_bit(libc::SIGALRM) != 0
1627        }) {
1628            crate::tool_global::reconcile_alarm(guest).await;
1629        }
1630        result
1631    }

1639        guest: &mut G,

1642    ) -> crate::scheduler::posix_timer_coalesce::TakeOutcome {

1644        let skip = |skip| TakeOutcome {
1645            skip,
1646            rewrite: None,
1647        };

1651                    crate::syscalls::threads::read_kernel_signal_state(guest.pid(), guest.tid())

1657                let Some(set) = waited else {
1658                    return skip(false);
1659                };

1661                    crate::syscalls::threads::read_kernel_signal_state(guest.pid(), guest.tid())
1662                else {
1663                    return skip(false);
1664                };
1665                let at = guest.thread_state().thread_logical_time.as_nanos();

1667                    crate::tool_global::discarded_signal(
1668                        guest,

1674                        },

1676                    .await,

1679            }
1680            _ => skip(false),

1699    /// The `signalfd_siginfo` records a successful signalfd read returned:
1700    /// the first `read` bytes of the guest buffers `extents` (address,
1701    /// length), in order, as `read`, `readv`, `preadv` and `preadv2` scatter
1702    /// them. They are reconciled with the child-exit ledger as a SIGCHLD
1703    /// delivered to a handler (`Detcore::child_exit_sigchld_delivers`) or
1704    /// consumed by `rt_sigtimedwait` (`synchronous_sigchld_stands`) is: the
1705    /// scheduler's copy gets the siginfo Linux gives the parent, the kernel's
1706    /// coalesced copy for a SIGKILL cohort names the cohort's lowest pid, and
1707    /// a copy for an exit a SIGCHLD already stood for is dropped. Other
1708    /// records are kept as read. The kept records are scattered back over the
1709    /// same extents, in order, with the bytes the buffer held before the read
1710    /// (`before`) over the bytes of dropped ones. Returns the bytes kept, a
1711    /// whole number of records, or `None` when a record is dropped whose bytes
1712    /// `before` does not cover, which the caller refuses
1713    /// (`refuse_unrestorable_signalfd_record`).
1714    pub(crate) async fn reconcile_signalfd_extents<G: Guest<Self>>(
1715        &self,
1716        guest: &mut G,
1717        extents: &[(usize, usize)],
1718        read: usize,
1719        before: &[u8],
1720    ) -> Option<usize> {
1721        let len = read - read % SIGNALFD_RECORD;
1722        let pieces = signalfd_extent_pieces(extents, len);
1723        let mut bytes = Vec::with_capacity(len);
1724        for (address, length) in &pieces {
1725            let mut piece = vec![0u8; *length];
1726            let Some(address_mut) = AddrMut::<u8>::from_raw(*address) else {
1727                return Some(len);
1728            };
1729            if guest.memory().read_exact(address_mut, &mut piece).is_err() {
1730                use std::os::unix::fs::FileExt;
1731                let read = std::fs::File::open(format!("/proc/{}/mem", guest.tid().as_raw()))
1732                    .and_then(|mem| mem.read_exact_at(&mut piece, *address as u64));
1733                if read.is_err() {
1734                    // The kernel has just written these bytes, so the buffer
1735                    // is mapped and writable; `/proc/<tid>/mem` reads any
1736                    // such page.
1737                    return Some(len);
1738                }
1739            }
1740            bytes.extend_from_slice(&piece);
1741        }
1742        let controls = signalfd_record_controls(&bytes, std::process::id());
1743        let mut answers = Vec::with_capacity(controls.len());
1744        for control in controls {
1745            answers.push(match control {
1746                Some(control) => Some(crate::tool_global::child_exit_sigchld(guest, control).await),
1747                None => None,
1748            });
1749        }
1750        // A record of a copy Linux discarded when it was sent is dropped, as a
1751        // duplicate is: the guest never sees it
1752        // (`crate::syscalls::helpers::take_was_discarded`,
1753        // <https://github.com/rrnewton/hermit/issues/4293>).
1754        // A POSIX timer's entry, or a kept signal standing on another
1755        // sender's record, gets its own record in place of the kernel's
1756        // (<https://github.com/rrnewton/hermit/issues/4262>), and a stale
1757        // timer entry is dropped as a discarded copy is.
1758        if crate::scheduler::discarded_at_send::records(guest.config()) {
1759            for (record, answer) in bytes.chunks_mut(SIGNALFD_RECORD).zip(answers.iter_mut()) {
1760                let signal = signalfd_field(record, 0) as i32;
1761                if !(1..32).contains(&signal) {
1762                    continue;
1763                }
1764                let outcome = crate::syscalls::helpers::take_outcome(guest, signal, None).await;
1765                if outcome.skip {
1766                    *answer = Some(
1767                        crate::scheduler::child_exit_sigchld::ChildExitSigchldAnswer::Claimed(true),
1768                    );
1769                } else if let Some(rewrite) = outcome.rewrite {
1770                    record.copy_from_slice(&crate::scheduler::signal_lists::signalfd_record(
1771                        &rewrite,
1772                    ));
1773                }
1774            }
1775        }
1776        let rewritten = match apply_signalfd_answers(&bytes, &answers, before) {
1777            Ok(Some(rewritten)) => rewritten,
1778            Ok(None) => return Some(len),
1779            Err(()) => return None,
1780        };
1781        let mut offset = 0;
1782        for (address, length) in &pieces {
1783            let written = AddrMut::<u8>::from_raw(*address).map(|address| {
1784                guest
1785                    .memory()
1786                    .write_exact(address, &rewritten.bytes[offset..offset + length])
1787            });
1788            if !matches!(written, Some(Ok(()))) {
1789                tracing::warn!(
1790                    "[dtid {}] the signalfd records at {:#x} keep the kernel's siginfo",
1791                    guest.thread_state().dettid,
1792                    address,
1793                );
1794                return Some(len);
1795            }
1796            offset += length;
1797        }
1798        Some(rewritten.kept)
1799    }

1801    /// A child-exit SIGCHLD that a synchronous wait consumed goes through the
1802    /// same notification ledger as one delivered to a handler
1803    /// (`Detcore::child_exit_sigchld_delivers`), on a backend where both the
1804    /// kernel and the scheduler send it (`scheduler::child_exit_sigchld`).
1805    /// The scheduler's copy is returned with the siginfo Linux gives the
1806    /// parent, and a copy for an exit that a SIGCHLD already stood for is
1807    /// dropped. Returns whether the consumed signal stands.
1808    async fn synchronous_sigchld_stands<G: Guest<Self>>(
1809        &self,
1810        guest: &mut G,
1811        call: syscalls::RtSigtimedwait,
1812        waited: Option<u64>,
1813        result: &Result<i64, Error>,
1814    ) -> bool {
1815        if !self.cfg.backend.reports_child_exit_publication {
1816            return true;
1817        }
1818        // Linux copies the set in before it dequeues, and the siginfo out
1819        // after, so an EFAULT with a readable set (`waited`) took a signal it
1820        // never reports; an unreadable set dequeued nothing. Which copy it
1821        // took: a recorded private one if SIGCHLD is gone from the thread's
1822        // private queue, which only committed guest sends fill; otherwise,
1823        // with SIGCHLD the only signal in the set, the shared queue's. A wider
1824        // set falls back to the shared queue's pending bit (the Claude and
1825        // Codex re-checks of https://github.com/rrnewton/hermit/pull/4131).
1826        if matches!(result, Err(Error::Errno(Errno::EFAULT)))
1827            && guest.config().sequentialize_threads
1828            && let Some(set) = waited
1829        {
1830            let sigchld = kernel_sigset_bit(libc::SIGCHLD);
1831            if set & sigchld != 0
1832                && let Ok(state) = crate::syscalls::threads::read_thread_wait_signal_state(
1833                    guest.pid(),
1834                    guest.tid(),
1835                )
1836            {
1837                let shared_pending = if set & !sigchld == 0 {
1838                    false
1839                } else {
1840                    state.shared_pending & sigchld != 0
1841                };
1842                crate::tool_global::child_exit_sigchld(
1843                    guest,
1844                    crate::scheduler::child_exit_sigchld::ChildExitSigchldControl::WaitFaulted {
1845                        private_pending: state.thread_pending & sigchld != 0,
1846                        shared_pending,
1847                    },
1848                )
1849                .await;
1850            }
1851            return true;
1852        }
1853        if !matches!(result, Ok(signal) if *signal == libc::SIGCHLD as i64) {
1854            return true;
1855        }
1856        // Without a siginfo buffer the guest sees only the signal number,
1857        // which every copy shares; the ledger still learns that the wait took
1858        // one (`DequeuedCopy::Unknown`), which may be the shared queue's.
1859        let Some(info) = call.info() else {
1860            crate::tool_global::child_exit_sigchld(
1861                guest,
1862                crate::scheduler::child_exit_sigchld::ChildExitSigchldControl::Dequeue {
1863                    copy: crate::scheduler::child_exit_sigchld::DequeuedCopy::Unknown,
1864                    admits: false,
1865                },
1866            )
1867            .await;
1868            return true;
1869        };
1870        let Some(bytes) = read_siginfo(guest, info) else {
1871            return true;
1872        };
1873        Self::synchronous_sigchld_copy_stands(guest, info, &bytes).await
1874    }

2008    /// Resolve signal-zero existence checks in the fixed PID namespace, then route an
2009    /// unambiguous positive-PID process signal through the backend. Backends that can execute
2010    /// with guest PIDs preserve process-directed delivery; DBT translates it to the sole live
2011    /// thread because its native process uses a host PID. An unmaskable SIGKILL to a specific
2012    /// process group is also safe to preserve on backends whose guests use real namespace PIDs;
2013    /// other process-group and broadcast delivery remains refused until Detcore models eligible
2014    /// signal masks.
2015    pub async fn handle_kill<G: Guest<Self>>(
2016        &self,
2017        guest: &mut G,
2018        call: syscalls::Kill,
2019    ) -> Result<i64, Error> {
2020        self.record_orphaned_group_signals(guest, call.sig()).await;
2021        // A thread below the container's PID namespace names pids of its own
2022        // namespace, where none of Hermit's tasks is visible
2023        // (https://github.com/rrnewton/hermit/issues/4155).
2024        let nested = guest.thread_state().nested_pid_namespace;
2025        if !guest.config().sequentialize_threads {
2026            // Without sequentialization every other kill goes to the kernel as
2027            // is, but never one to a process group or to every process: the
2028            // guest's process group is inherited from Hermit's, so it can hold
2029            // Hermit and the user's shell or test runner, which a group kill
2030            // would reach (https://github.com/rrnewton/hermit/issues/4057).
2031            // Signal 0 sends nothing, and INT_MIN is ESRCH before any lookup.
2032            if call.sig() != 0 && call.pid() <= 0 && call.pid() != libc::pid_t::MIN {
2033                return self
2034                    .refuse_group_signal(guest, call.pid(), call.sig())
2035                    .await;
2036            }
2037            if !nested
2038                && self
2039                    .drop_signal_to_hermit(guest, Sysno::kill, None, call.pid(), call.sig())
2040                    .await
2041            {
2042                return Ok(0);
2043            }
2044            return Ok(self.record_or_replay(guest, call).await?);
2045        }
2046
2047        // A nested caller's process-group ids are its namespace's, which
2048        // Detcore does not translate yet; a SIGKILL to one would otherwise be
2049        // reserved for the container's group of that number.
2050        if nested && call.pid() <= 0 && call.pid() != libc::pid_t::MIN && call.sig() != 0 {
2051            return self
2052                .refuse_group_signal(guest, call.pid(), call.sig())
2053                .await;
2054        }
2055        let (named, own_init_caught) = if call.pid() > 0 {
2056            match self.name_guest_pid(guest, Sysno::kill, call.pid()).await {
2057                NamedGuestPid::Task {
2058                    task,
2059                    own_init_caught,
2060                } => (task.as_raw(), own_init_caught),
2061                NamedGuestPid::Answer(answer) => return answer,
2062            }
2063        } else {
2064            (call.pid(), None)
2065        };
2066        if Self::own_namespace_init_drops(own_init_caught, call.sig()) {
2067            let task = DetTid::from_raw(named);
2068            return self
2069                .send_to_own_init(guest, call, task, QueueId::Process(task), call.sig())
2070                .await;
2071        }
2072
2073        // Linux's `kill_something_info` answers ESRCH for pid INT_MIN before
2074        // anything else ("-INT_MIN is undefined"), so the kernel's answer is
2075        // deterministic and sends nothing; negating it here would overflow
2076        // (https://github.com/rrnewton/hermit/issues/4046).
2077        if call.sig() == 0 && call.pid() > 0 {
2078            // An existence check answers from the same model as a signal, so
2079            // that one pid never answers 0 to one and ESRCH to the other: a
2080            // process with no live threads is a zombie (0) or gone (ESRCH)
2081            // whenever Detcore saw it exit, whatever the host has reaped by
2082            // now (https://github.com/rrnewton/hermit/issues/4091).
2083            let (targets, zombie) =
2084                resolve_kill_targets_and_exit(guest, DetPid::from_raw(named)).await;
2085            if targets.is_empty() {
2086                match zombie {
2087                    Some(true) => return Ok(0),
2088                    Some(false) => return Err(Errno::ESRCH.into()),
2089                    None => {}
2090                }
2091            }
2092        }
2093        if call.sig() == 0 || call.pid() == libc::pid_t::MIN {
2094            return Ok(self.record_or_replay(guest, call).await?);
2095        }
2096
2097        let tgid = named;
2098        // A process-group or broadcast SIGALRM is refused by name below. The
2099        // scheduler resolves a thread id to its process itself.
2100        if call.sig() == libc::SIGALRM && tgid > 0 {
2101            refuse_sigalrm(guest, SigalrmControl::SendTo(DetTid::from_raw(tgid))).await?;
2102        }
2103        if self
2104            .drop_signal_to_hermit(guest, Sysno::kill, None, tgid, call.sig())
2105            .await
2106        {
2107            return Ok(0);
2108        }
2109        if can_forward_process_group_signal(
2110            tgid,
2111            call.sig(),
2112            guest
2113                .config()
2114                .backend
2115                .requires_thread_directed_process_signals,
2116        ) {

2118            // know the group was killed: a held process in it will die, so a

2511    /// What an `rt_sigqueueinfo` whose `siginfo_t` Hermit may not be able to
2512    /// read comes to, decided before admission (`do_rt_sigqueueinfo` in
2513    /// kernel/signal.c). No call that Linux could deliver reaches the kernel
2514    /// without admission (Codex re-check 2 of
2515    /// https://github.com/rrnewton/hermit/pull/4116).
2516    ///
2517    /// The kernel always copies the first 48 bytes (`kernel_siginfo_t`) and
2518    /// the other 80 only for a `si_code` whose layout it does not know
2519    /// (`copy_siginfo_from_user`). Its `copy_from_user` reads what user mode
2520    /// can: on x86 a page with any of read, write or execute permission; it
2521    /// faults on a `PROT_NONE` page; and a fault in a hole just below a
2522    /// grow-down mapping can grow that mapping and succeed. The tracer's own
2523    /// reads say nothing about that (`process_vm_readv` refuses a write-only
2524    /// page, and `/proc/<tid>/mem` force-reads a `PROT_NONE` one), so the
2525    /// mappings of `/proc/<tid>/smaps` decide first:
2526    /// - the first byte of the 48 that user mode cannot read lies in a
2527    ///   `PROT_NONE` mapping, or in a hole whose next mapping above is not
2528    ///   grow-down (`VmFlags` `gd`; only that mapping can grow into it):
2529    ///   EFAULT is certain, and Hermit answers it without the kernel;
2530    /// - any other byte of the 128 that user mode cannot read (a hole below a
2531    ///   grow-down mapping, a tail running into a hole or a `PROT_NONE` page),
2532    ///   or a listing that cannot be read: `Unreadable`, refused by name;
2533    /// - all 128 user-readable: the tracer reads them (`process_vm_readv`, then
2534    ///   `/proc/<tid>/mem` for a write-only page), and a page neither reaches
2535    ///   (a `VM_IO`/`VM_PFNMAP` page such as the vDSO's `[vvar]`) is
2536    ///   `Unreadable`. A `si_code` only the target thread itself may use
2537    ///   (`si_code >= 0` or `SI_TKILL`, from a caller that is not `pid`) is
2538    ///   then `KernelFails`: after a complete copy Linux fails it with EPERM,
2539    ///   or E2BIG first, and sends nothing, so it goes to the kernel unchanged.
2540    fn rt_sigqueueinfo_precheck<G: Guest<Self>>(
2541        &self,
2542        guest: &mut G,
2543        call: &syscalls::RtSigqueueinfo,
2544    ) -> RtSigqueueinfoPrecheck {
2545        const SIZE: usize = std::mem::size_of::<libc::siginfo_t>();
2546        // `sizeof(kernel_siginfo_t)` on x86_64: the part every copy reads.
2547        const KERNEL_SIGINFO: usize = 48;
2548        let address = call.siginfo().map_or(0, |address| address.as_raw());
2549        let Ok(smaps) = std::fs::read_to_string(format!("/proc/{}/smaps", guest.tid().as_raw()))
2550        else {
2551            return RtSigqueueinfoPrecheck::Unreadable;
2552        };
2553        let mappings = parse_smaps(&smaps);
2554        match first_user_unreadable(&mappings, address, KERNEL_SIGINFO) {
2555            None => {}
2556            Some(Unreadable::NoPermission)
2557            | Some(Unreadable::Hole {
2558                below_growsdown: false,
2559            }) => {
2560                return RtSigqueueinfoPrecheck::Efault;
2561            }
2562            Some(Unreadable::Hole {
2563                below_growsdown: true,
2564            }) => {
2565                return RtSigqueueinfoPrecheck::Unreadable;
2566            }
2567        }
2568        if first_user_unreadable(&mappings, address, SIZE).is_some() {
2569            return RtSigqueueinfoPrecheck::Unreadable;
2570        }
2571        let read = call.siginfo().and_then(|info| {
2572            guest.memory().read_value(info).ok().or_else(|| {
2573                use std::os::unix::fs::FileExt;
2574                let mut bytes = [0u8; SIZE];
2575                std::fs::File::open(format!("/proc/{}/mem", guest.tid().as_raw()))
2576                    .and_then(|mem| mem.read_exact_at(&mut bytes, address as u64))
2577                    .ok()?;
2578                // SAFETY: `siginfo_t` is plain data, and `bytes` holds exactly
2579                // one, read from the guest.
2580                Some(unsafe { std::ptr::read_unaligned(bytes.as_ptr().cast::<libc::siginfo_t>()) })
2581            })
2582        });
2583        let Some(info) = read else {
2584            return RtSigqueueinfoPrecheck::Unreadable;
2585        };
2586        let code = info.si_code;
2587        if (code >= 0 || code == libc::SI_TKILL)
2588            && guest.thread_state().dettid.as_raw() != call.tgid()
2589        {
2590            RtSigqueueinfoPrecheck::KernelFails
2591        } else {
2592            RtSigqueueinfoPrecheck::Proceeds
2593        }
2594    }

2800    pub async fn handle_tgkill<G: Guest<Self>>(
2801        &self,
2802        guest: &mut G,
2803        call: syscalls::Tgkill,
2804    ) -> Result<i64, Error> {
2805        self.record_orphaned_group_signals(guest, call.sig()).await;
2806        // A nested caller names its own namespace's ids; Detcore's model uses
2807        // the tasks they name, and the forwarded call keeps them
2808        // (https://github.com/rrnewton/hermit/issues/4155). Non-positive ids
2809        // are EINVAL in Linux before any lookup.
2810        let nested = guest.thread_state().nested_pid_namespace;
2811        let (tgid, tid) = if nested && call.tgid() > 0 && call.tid() > 0 {
2812            let (tid, own_init_caught) =
2813                match self.name_guest_pid(guest, Sysno::tgkill, call.tid()).await {
2814                    NamedGuestPid::Task {
2815                        task,
2816                        own_init_caught,
2817                    } => (task, own_init_caught),
2818                    NamedGuestPid::Answer(answer) => return answer,
2819                };
2820            let tgid = match self.name_guest_pid(guest, Sysno::tgkill, call.tgid()).await {
2821                NamedGuestPid::Task { task, .. } => task,
2822                NamedGuestPid::Answer(answer) => return answer,
2823            };

2825                return self
2826                    .send_to_own_init(guest, call, tid, QueueId::Thread(tid), call.sig())
2827                    .await;
2828            }
2829            (tgid.as_raw(), tid.as_raw())
2830        } else {
2831            (call.tgid(), call.tid())
2832        };

2834            refuse_sigalrm(guest, SigalrmControl::SendTo(DetTid::from_raw(tid))).await?;
2835        }

2837            && self
2838                .drop_signal_to_hermit(guest, Sysno::tgkill, Some(tgid), tid, call.sig())
2839                .await
2840        {
2841            return Ok(0);
2842        }
2843        let _reserved = self
2844            .reserve_controlled_self_sigkill_exit(
2845                guest,
2846                call.sig(),
2847                Some(DetPid::from_raw(tgid)),
2848                Some(DetTid::from_raw(tid)),
2849            )
2850            .await;

2852        // and is retired at this turn, like `kill`'s. Only a valid pair is
2853        // reserved: Linux rejects a thread outside the named group (ESRCH) or a
2854        // non-positive id (EINVAL) before anything happens, so such a send

2860            && guest.config().sequentialize_threads
2861            && tgid > 0
2862            && tid > 0
2863            && registered_process_of(guest, DetTid::from_raw(tid)).await
2864                == Some(DetPid::from_raw(tgid))
2865        {
2866            exited_leader = sigkill_names_an_exited_leader(guest, DetTid::from_raw(tid)).await;
2867            if !exited_leader
2868                && let Some(reservation) = reserve_sigkill(guest, DetTid::from_raw(tid)).await
2869            {
2870                return self.send_reserved_sigkill(guest, reservation, call).await;
2871            }
2872        }
2873        let value = self
2874            .send_guest_signal(
2875                guest,
2876                call,
2877                DetTid::from_raw(call.tid()),
2878                call.sig(),
2879                None,
2880                true,
2881                DetTid::from_raw(tid),
2882            )
2883            .await?;
2884        // The kernel accepted a SIGKILL that changes nothing: no thread takes
2885        // it and no process dies, so the scheduler is not told of a kill.
2886        if exited_leader {
2887            return Ok(value);
2888        }

2890        // guest THREAD signals a sibling. Like `kill`, a successful cross-task
2891        // send must tell the scheduler, or a target parked in a child wait is

2894            guest,
2895            DetTid::from_raw(tid),
2896            call.sig(),
2897            None,
2898            crate::tool_global::SignalScope::Thread,
2899        )
2900        .await;

2902    }

3105                    private: None,
3106                    shared: Some(process),

3710/// Only on a backend whose kernel reports the guest's signal state; on any
3711/// other, a sleep never checks before it parks, and a wake always ends it
3712/// (`Held`). A signal ends the sleep when it is pending for this thread,
3713/// unblocked, and caught or acted on by default
3714/// (`KernelSignalState::pending_interrupting`). Before the sleep parks, a stop
3715/// signal does so only when `stops_end_it`, which `nanosleep` leaves unset
3716/// because Detcore does not keep its deadline for the kernel's restart after a
3717/// stop, so the stop takes effect when the sleep ends, as before. After a wake,
3718/// which only the scheduler's own sends (a timer, a child-exit `SIGCHLD`) can
3719/// commit for a stop on a `nanosleep`, a pending stop ends it, as every wake
3720/// did before, so that the stop takes effect promptly.
3721///
3722/// `SIGCHLD` and the signals a host-timed source armed by a guest can post
3723/// (`host_timed_signals`) are held: the kernel can post them at a moment set by
3724/// host timing, and `/proc` cannot tell that copy from one a guest sent. They
3725/// never end a sleep before it parks, nor does the scheduler record a guest's
3726/// cross-task send of one for a sleep. A wake the scheduler committed for its
3727/// own child-exit `SIGCHLD` ends the sleep with `Held` only while the copy it
3728/// sent is still pending in the process's shared queue
3729/// (`ChildExitSigchldControl::PlacedPending`, which `/proc` cannot answer),
3730/// `/proc` shows it pending, and it is unblocked and caught or default-fatal
3731/// (`held_wake_ends_sleep`); a host-timed held signal, from the mask and
3732/// dispositions alone. Like the precise futex
3733/// wait's check, this reads the pending set before the sleep parks as well as
3734/// after a wake: a signal sent while the thread was stopped short of filing
3735/// its request, as when its timeslice ended in the syscall's prehook, is
3736/// pending and was never recorded for its sleep. After a wake whose
3737/// non-held signals are no longer pending, a `SIGCONT` having cancelled a stop
3738/// or `sigaction(SIG_IGN)` having flushed it, a Linux sleep would still be
3739/// asleep, so this returns `None`.
3740///
3741/// A state that cannot be read is handled as a wait's first read is
3742/// (`read_wait_signal_state`, here `read_thread_wait_signal_state`, since
3743/// Reverie can report the creator's pid for a process a raw `clone` made): a
3744/// thread that no longer exists ends the call
3745/// with `ERESTARTNOINTR`, which nothing observes, and otherwise the run ends
3746/// with a diagnostic.
3747pub(crate) async fn sleep_signal<G, T>(
3748    guest: &mut G,
3749    check: SleepCheck,
3750    stops_end_it: bool,
3751) -> Result<SleepSignal, Error>
3752where
3753    G: Guest<Detcore<T>>,
3754    T: RecordOrReplay,
3755{
3756    if !guest
3757        .config()
3758        .backend
3759        .supports_blocked_wait_signal_interruption
3760    {
3761        return Ok(match check {
3762            SleepCheck::BeforePark => SleepSignal::None,
3763            SleepCheck::AfterWake(_) => SleepSignal::Held,
3764        });
3765    }
3766    let host_timed = host_timed_signals(guest).await;
3767    let held = kernel_sigset_bit(libc::SIGCHLD) | host_timed;
3768    let before_park = check == SleepCheck::BeforePark;
3769    // `held_woken`: the held signals the scheduler woke this sleep for.
3770    // `unknown_held_wake`: a wake that names no signal.
3771    let (deciding, held_woken, unknown_held_wake) = match check {
3772        SleepCheck::BeforePark => (!held, 0, false),
3773        SleepCheck::AfterWake(None) => (0, 0, true),
3774        SleepCheck::AfterWake(Some(woken)) => {
3775            let woken = woken
3776                .iter()
3777                .fold(0, |set, signal| set | kernel_sigset_bit(signal.raw()));
3778            (woken & !held, woken & held, false)
3779        }
3780    };
3781    if deciding != 0 || held_woken != 0 {
3782        let state = read_thread_wait_signal_state(guest.pid(), guest.tid())?;
3783        let mut ending = state.pending_interrupting(state.blocked, false) & deciding;
3784        if !stops_end_it && before_park {
3785            ending &= !(state.default_job_control_stops() | kernel_sigset_bit(libc::SIGSTOP));
3786        }
3787        // A copy Linux discarded when it was sent ends nothing, except a
3788        // signal the scheduler committed at the wake, which is kept
3789        // (<https://github.com/rrnewton/hermit/issues/4293>).
3790        let committed = if before_park { 0 } else { deciding };
3791        let ending =
3792            crate::syscalls::threads::without_discarded(guest, &state, ending, committed).await;
3793        if ending != 0 {
3794            return Ok(SleepSignal::Pending);
3795        }
3796        // A committed `SIGCHLD` counts only while the schedule says this thread
3797        // will take it (`PlacedPending`): a guest's thread-directed copy in its
3798        // private queue, or the scheduler's or a guest's shared copy still in
3799        // the window. `/proc` cannot tell either from a kernel copy that host
3800        // timing posted, so its bits only prompt the question: a woken sleep
3801        // asks for the `SIGCHLD` it was woken for, and any sleep asks when its
3802        // private queue holds one, which a guest sent before it parked
3803        // (https://github.com/rrnewton/hermit/issues/4132).
3804        let sigchld = kernel_sigset_bit(libc::SIGCHLD) & !host_timed;
3805        let committed = if asks_committed_sigchld(held_woken, &state, sigchld)
3806            && matches!(
3807                crate::tool_global::child_exit_sigchld(
3808                    guest,
3809                    crate::scheduler::child_exit_sigchld::ChildExitSigchldControl::PlacedPending,
3810                )
3811                .await,
3812                crate::scheduler::child_exit_sigchld::ChildExitSigchldAnswer::Placed(true)
3813            ) {
3814            // A committed copy shows in `/proc`: an invariant, not a second
3815            // condition.
3816            if state.pending & sigchld == 0 {
3817                tracing::error!(
3818                    "[dtid {}] a committed SIGCHLD is pending by the schedule but not in /proc",
3819                    guest.thread_state().dettid
3820                );
3821                debug_assert!(false, "a committed SIGCHLD is not pending in /proc");
3822            }
3823            sigchld
3824        } else {
3825            0
3826        };
3827        if let Some(verdict) = committed_sigchld_verdict(&state, held_woken, committed) {
3828            return Ok(verdict);
3829        }
3830        // Where the process does not use the send window, a thread outside the
3831        // schedule could take the scheduler's copy at a moment host timing
3832        // chooses, so the window records none, and `PlacedPending` says no
3833        // whether or not that thread took it. Parked again, the sleep ran to its
3834        // deadline with the copy pending and the handler ran as it returned, an
3835        // outcome Linux never gives (https://github.com/rrnewton/hermit/issues/4229).
3836        // So this thread claims the copy itself, in its own turn
3837        // (`claim_pending_sigchld`), while SIGCHLD is the signal its kernel
3838        // dequeues next: held for delivery to it, where no other thread can
3839        // take it, so the handler runs as the sleep returns (`Claimed`). If the
3840        // outside thread took it first, the claim finds none and the sleep
3841        // parks again, as on Linux, where the woken sleeper finds no signal. Reading the pending bit and then ending the sleep
3842        // left a gap in which the outside thread could take it, and then the
3843        // sleep returned EINTR with no handler
3844        // (https://github.com/rrnewton/hermit/issues/4341).
3845        //
3846        // In record and replay too, but only while no other thread of the
3847        // process is outside the schedule in a call that could take the copy
3848        // (`Scheduler::outside_call_may_take_sigchld`, through
3849        // `WindowUnusable`). Then nothing but this claim can take it, so its
3850        // pending bit, and the claim, are the same in a recording and its
3851        // replay (https://github.com/rrnewton/hermit/issues/4343). While such a
3852        // call is out, whether it took the copy is host-timed, and replay
3853        // serves a signalfd read from the log without taking anything, so the
3854        // two could decide differently; the sleep parks again, as before.
3855        if committed == 0
3856            && held_woken & sigchld != 0
3857            && woken_sigchld_pending(&state, sigchld)
3858            && matches!(
3859                crate::tool_global::child_exit_sigchld(
3860                    guest,
3861                    crate::scheduler::child_exit_sigchld::ChildExitSigchldControl::WindowUnusable,
3862                )
3863                .await,
3864                crate::scheduler::child_exit_sigchld::ChildExitSigchldAnswer::Placed(true)
3865            )
3866            && state.next_dequeued(state.blocked).map(|(signal, _)| signal) == Some(libc::SIGCHLD)
3867            && {
3868                send_claim_fault_signal(guest, ClaimFault::Usr1BeforeClaim);
3869                let claimed = claim_pending_sigchld(guest).await?;
3870                if claimed {
3871                    send_claim_fault_signal(guest, ClaimFault::Usr1AfterClaim);
3872                }
3873                claimed
3874            }
3875        {
3876            // Its delivery stop places it, so it is not deferred behind a
3877            // runnable sibling (`ThreadState::claimed_sigchld`).
3878            guest.thread_state_mut().claimed_sigchld = true;
3879            return Ok(SleepSignal::Claimed);
3880        }
3881        let held_woken = held_woken & !sigchld;
3882        if held_wake_ends_sleep(&state, held_woken) {
3883            return Ok(SleepSignal::Held);
3884        }
3885        return Ok(SleepSignal::None);
3886    }
3887    Ok(if unknown_held_wake {
3888        SleepSignal::Held
3889    } else {
3890        SleepSignal::None
3891    })
3892}

3909/// Whether a `SIGCHLD` this thread claimed (`ThreadState::claimed_sigchld`,
3910/// `claimed`) is still claimed at a syscall entry, or after a `SIGCHLD` the
3911/// ledger dropped: SIGCHLD is still pending in its queue or its process's,
3912/// whatever its mask. The claim may have held another signal instead
3913/// (https://github.com/rrnewton/hermit/issues/4346), whose handler can block
3914/// SIGCHLD and make syscalls before its `rt_sigreturn` lets SIGCHLD be
3915/// delivered (the Codex review of the claimed-delivery fix), or a duplicate
3916/// the ledger drops while another copy waits (the Claude review).
3917/// Once SIGCHLD is not pending, its delivery never came to this thread. A
3918/// state that cannot be read (`read_state`, asked only for a claim) forgets
3919/// the claim, so the delivery is deferred as before the claim existed.
3920pub(crate) fn claim_still_pending<E>(
3921    claimed: bool,
3922    read_state: impl FnOnce() -> Result<crate::syscalls::threads::KernelSignalState, E>,
3923) -> bool {
3924    claimed && read_state().is_ok_and(|state| state.pending & kernel_sigset_bit(libc::SIGCHLD) != 0)
3925}

3927/// Whether `sigchld`, the scheduler's copy a sleep was woken for, is still
3928/// pending for it and ends it: in the thread's or its process's queue, and
3929/// unblocked and caught or acted on by default (`sleep_signal`, for a process
3930/// whose send window is unusable).
3931fn woken_sigchld_pending(
3932    state: &crate::syscalls::threads::KernelSignalState,
3933    sigchld: KernelSigset,
3934) -> bool {
3935    state.pending & state.interrupting(state.blocked) & sigchld != 0
3936}

3938/// Whether a sleep's own check (`sleep_signal`) asks the scheduler whether a
3939/// committed `SIGCHLD` is pending for it (`PlacedPending`): it was woken for
3940/// one (`held_woken`), or its private queue holds one. The latter is how a
3941/// guest's thread-directed copy sent before the sleeper filed its request is
3942/// not lost: the sender's turn found no sleep to wake, so only the sleeper's
3943/// check before it parks sees it (https://github.com/rrnewton/hermit/issues/4132).
3944fn asks_committed_sigchld(
3945    held_woken: KernelSigset,
3946    state: &crate::syscalls::threads::KernelSignalState,
3947    sigchld: KernelSigset,
3948) -> bool {
3949    (held_woken | state.thread_pending) & sigchld != 0
3950}

3952/// What a committed `SIGCHLD` (`committed`, the scheduler's answer) makes of
3953/// the sleep: it ends it if unblocked and caught or acted on by default, as
3954/// the held signal the wake named (`Held`) or as one pending for it
3955/// (`Pending`); `None` leaves the decision to the rest of the check.
3956fn committed_sigchld_verdict(
3957    state: &crate::syscalls::threads::KernelSignalState,
3958    held_woken: KernelSigset,
3959    committed: KernelSigset,
3960) -> Option<SleepSignal> {
3961    let verdict = if held_woken & committed != 0 {
3962        SleepSignal::Held
3963    } else {
3964        SleepSignal::Pending
3965    };
3966    (state.interrupting(state.blocked) & committed != 0).then_some(verdict)
3967}

3969/// The size of a `struct signalfd_siginfo`, which has ssi_signo at 0,
3970/// ssi_code at 8, ssi_pid at 12, ssi_uid at 16, ssi_status at 40, ssi_int at
3971/// 44, ssi_utime at 56 and ssi_stime at 64.
3972pub(crate) const SIGNALFD_RECORD: usize = 128;

3974/// The signals signalfd `fd` of process `pid` reads, from its `sigmask:`
3975/// line in `/proc/<pid>/fdinfo/<fd>` (a kernel sigset in hex), or `None`
3976/// when it cannot be read.
3977pub(crate) fn signalfd_mask(pid: i32, fd: i32) -> Option<u64> {
3978    let info = std::fs::read_to_string(format!("/proc/{pid}/fdinfo/{fd}")).ok()?;
3979    info.lines()
3980        .find_map(|line| line.strip_prefix("sigmask:"))
3981        .and_then(|mask| u64::from_str_radix(mask.trim(), 16).ok())
3982}

3984fn signalfd_field(record: &[u8], at: usize) -> u32 {
3985    u32::from_ne_bytes(record[at..at + 4].try_into().expect("four bytes"))
3986}

3988/// For each `signalfd_siginfo` record in `bytes`, the child-exit ledger
3989/// question it raises (`Detcore::reconcile_signalfd_records`): a SIGCHLD with
3990/// a child-exit code is the kernel's copy for that child, a SIGCHLD the
3991/// scheduler sent (`is_own_sigchld_mark`) is its copy, and any other record
3992/// raises none.
3993fn signalfd_record_controls(
3994    bytes: &[u8],
3995    tracer: u32,
3996) -> Vec<Option<crate::scheduler::child_exit_sigchld::ChildExitSigchldControl>> {
3997    use crate::scheduler::child_exit_sigchld::ChildExitSigchldControl;
3998    use crate::scheduler::child_exit_sigchld::DequeuedCopy;
3999    bytes
4000        .chunks(SIGNALFD_RECORD)
4001        .map(|record| {
4002            if signalfd_field(record, 0) != libc::SIGCHLD as u32 {
4003                return None;
4004            }
4005            let code = signalfd_field(record, 8) as i32;
4006            let pid = signalfd_field(record, 12);
4007            let value = signalfd_field(record, 44) as i32;
4008            let copy = if crate::scheduler::child_exit_sigchld::is_child_exit_code(code) {
4009                DequeuedCopy::Kernel {
4010                    child: DetPid::from_raw(pid as i32),
4011                }
4012            } else if crate::scheduler::child_exit_sigchld::is_own_sigchld_mark(
4013                code, pid as i32, value, tracer,
4014            ) {
4015                DequeuedCopy::Scheduler
4016            } else if code == libc::SI_TKILL {
4017                DequeuedCopy::Thread
4018            } else {
4019                DequeuedCopy::Other
4020            };
4021            // Read, a kept record makes no `InboundSignal` request.
4022            Some(ChildExitSigchldControl::Dequeue {
4023                copy,
4024                admits: false,
4025            })
4026        })
4027        .collect()
4028}

4030/// The pieces of the guest buffers `extents` (address, length) that hold the
4031/// first `len` bytes a vectored read scattered, in order; empty extents are
4032/// skipped.
4033pub(crate) fn signalfd_extent_pieces(
4034    extents: &[(usize, usize)],
4035    len: usize,
4036) -> Vec<(usize, usize)> {
4037    let mut left = len;
4038    let mut pieces = Vec::new();
4039    for (address, length) in extents {
4040        if left == 0 {
4041            break;
4042        }
4043        let take = (*length).min(left);
4044        if take > 0 {
4045            pieces.push((*address, take));
4046            left -= take;
4047        }
4048    }
4049    pieces
4050}

4052/// Signalfd records rewritten by the child-exit ledger's answers.
4053#[derive(Debug, PartialEq, Eq)]
4054struct RewrittenSignalfdRecords {
4055    /// The buffer to write back: the kept records in order, then what the
4056    /// buffer held before the read over the bytes of dropped ones, so no
4057    /// dropped record stays readable.
4058    bytes: Vec<u8>,
4059    /// How many bytes the read returns.
4060    kept: usize,
4061}
4062
4063/// Apply the ledger's `answers`, one per record of `bytes`
4064/// (`signalfd_record_controls`): a claimed duplicate or a duplicate scheduler
4065/// copy is dropped, a copy that stands for an owned child is given that
4066/// child's siginfo with zero CPU times, and every other record is kept as read.
4067/// `Ok(None)` when no record changed. `Err(())` when a record is dropped but
4068/// `before`, the buffer's bytes captured before the read
4069/// (`capture_signalfd_buffer`), does not cover the bytes it leaves: they are
4070/// never invented (the Codex re-check of
4071/// https://github.com/rrnewton/hermit/pull/4351, P2-2).
4072fn apply_signalfd_answers(
4073    bytes: &[u8],
4074    answers: &[Option<crate::scheduler::child_exit_sigchld::ChildExitSigchldAnswer>],
4075    before: &[u8],
4076) -> Result<Option<RewrittenSignalfdRecords>, ()> {
4077    use crate::scheduler::child_exit_sigchld::ChildExitSigchldAnswer;
4078    use crate::scheduler::child_exit_sigchld::OwnCopy;
4079    let mut kept = Vec::with_capacity(bytes.len());
4080    let mut changed = false;
4081    for (record, answer) in bytes.chunks(SIGNALFD_RECORD).zip(answers) {
4082        let mut record = record.to_vec();
4083        match answer {
4084            Some(ChildExitSigchldAnswer::Claimed(true))
4085            | Some(ChildExitSigchldAnswer::OwnCopy(OwnCopy::Duplicate)) => {
4086                changed = true;
4087                continue;
4088            }
4089            Some(ChildExitSigchldAnswer::OwnCopy(OwnCopy::Deliver(own)))
4090            | Some(ChildExitSigchldAnswer::ClaimedFor(own)) => {
4091                // The record Linux writes for a child exit (SIL_CHLD) starts
4092                // zeroed and fills in only these: nothing of the copy it
4093                // replaces is kept, such as the own copy's SI_QUEUE mark in
4094                // ssi_int and ssi_ptr. CPU times are 0 (natively, host time).
4095                let signo = signalfd_field(&record, 0);
4096                record.fill(0);
4097                record[0..4].copy_from_slice(&signo.to_ne_bytes());
4098                record[8..12].copy_from_slice(&own.code.to_ne_bytes());
4099                record[12..16].copy_from_slice(&(own.pid as u32).to_ne_bytes());
4100                record[16..20].copy_from_slice(&own.uid.to_ne_bytes());
4101                record[40..44].copy_from_slice(&own.status.to_ne_bytes());
4102                changed = true;
4103            }
4104            _ => {}
4105        }
4106        kept.extend_from_slice(&record);
4107    }
4108    if !changed {
4109        return Ok(None);
4110    }
4111    let kept_len = kept.len();
4112    let len = bytes.len() - bytes.len() % SIGNALFD_RECORD;
4113    // The bytes past the kept records are those the caller's buffer held
4114    // before the read: a dropped record must leave the buffer as Linux, which
4115    // never queued it, leaves it (https://github.com/rrnewton/hermit/issues/4293).
4116    if before.len() < len {
4117        return Err(());
4118    }
4119    kept.extend_from_slice(&before[kept_len..len]);
4120    Ok(Some(RewrittenSignalfdRecords {
4121        bytes: kept,
4122        kept: kept_len,
4123    }))
4124}

4542/// The refusal for `send`, or `None` when the kernel's choice of receiver and
4543/// the sender's dequeue are forced: the sender takes the signal on its way back
4544/// to user mode, before any other thread of the process can run
4545/// (https://github.com/rrnewton/hermit/issues/4034).
4546pub(crate) fn own_process_signal_refusal(
4547    send: &OwnProcessSignal,
4548) -> Option<OwnProcessSignalRefusal> {
4549    use OwnProcessSignalRefusal as R;
4550    let bit = kernel_sigset_bit(send.sig);
4551    if !OWN_PROCESS_SIGNALS.contains(&send.sig) {
4552        return Some(R::SignalNotAdmitted);
4553    }
4554    if (send.ignored | send.caught) & bit == 0 {
4555        return Some(R::DefaultDisposition);
4556    }
4557    if send.record_replay {
4558        return Some(R::RecordReplay);
4559    }
4560    if send.schedule_replay {
4561        return Some(R::ScheduleReplay);
4562    }
4563    if send.stacktrace_signal {
4564        return Some(R::StacktraceSignal);
4565    }
4566    if send.blocked & bit != 0 {
4567        return Some(R::SenderBlocks);
4568    }
4569    for (tid, state) in &send.inputs.siblings {
4570        if *state == SiblingSignalState::OutOfSchedule {
4571            return Some(R::OutOfSchedule(*tid));
4572        }
4573    }
4574    for (tid, state) in &send.inputs.siblings {
4575        if *state == SiblingSignalState::InSchedule(None) {
4576            return Some(R::UnknownMask(*tid));
4577        }
4578    }
4579    if !send.names_sender {
4580        for (tid, state) in &send.inputs.siblings {
4581            if matches!(state, SiblingSignalState::InSchedule(Some(mask)) if mask & bit == 0) {
4582                return Some(R::NotSoleReceiver(*tid));
4583            }
4584        }
4585    }
4586    let unblocked = !send.blocked;
4587    let private = without_perf_event_signal(send.thread_pending) & unblocked;
4588    if private != 0 {
4589        return Some(R::PrivatePending(private));
4590    }
4591    let host_timed = send.inputs.host_timed & unblocked;
4592    if host_timed != 0 {
4593        return Some(R::HostTimedSource(host_timed));
4594    }
4595    let foreign = send.inputs.foreign_exit_signals & unblocked;
4596    if foreign != 0 {
4597        return Some(R::ForeignExitSignal(foreign));
4598    }
4599    // A signal numbered at or below `sig` is dequeued first or coalesces.
4600    let at_or_below = bit | (bit - 1);
4601    let undelivered = send.inputs.undelivered_scheduler_signals & unblocked & at_or_below;
4602    if undelivered != 0 {
4603        return Some(R::UndeliveredSchedulerSignal(undelivered));
4604    }
4605    if send.posthook_anchor {
4606        return Some(R::PosthookAnchor);
4607    }
4608    if send.signalfd {
4609        return Some(R::Signalfd);
4610    }
4611    if send.inputs.exec_in_flight {
4612        return Some(R::ExecInFlight);
4613    }
4614    if send.inputs.guest_marker_signal {
4615        return Some(R::GuestMarkerSignal);
4616    }
4617    None
4618}

4663/// One mapping of a `/proc/<pid>/smaps` listing.

4667    end: usize,

4669    /// permission allows a read; none (`---`) faults.

4671    /// `VmFlags` `gd`: a fault just below it can grow it downwards.

4675/// The mappings of a `/proc/<pid>/smaps` listing, in address order.
4676fn parse_smaps(smaps: &str) -> Vec<Mapping> {
4677    let mut mappings: Vec<Mapping> = Vec::new();
4678    for line in smaps.lines() {
4679        if let Some(flags) = line.strip_prefix("VmFlags:") {
4680            if let Some(last) = mappings.last_mut() {
4681                last.growsdown = flags.split_whitespace().any(|flag| flag == "gd");
4682            }
4683            continue;
4684        }
4685        let mut fields = line.split_whitespace();
4686        let Some((start, end)) = fields.next().and_then(|range| range.split_once('-')) else {
4687            continue;
4688        };
4689        let (Ok(start), Ok(end), Some(perms)) = (
4690            usize::from_str_radix(start, 16),
4691            usize::from_str_radix(end, 16),
4692            fields.next(),
4693        ) else {
4694            continue;
4695        };
4696        if perms.len() < 4 {
4697            continue;
4698        }
4699        mappings.push(Mapping {
4700            start,
4701            end,
4702            user_readable: perms.chars().take(3).any(|perm| perm != '-'),
4703            growsdown: false,
4704        });
4705    }
4706    mappings.sort_by_key(|mapping| mapping.start);
4707    mappings
4708}

4710/// Why a byte is not readable by user mode (`first_user_unreadable`).
4711#[derive(Clone, Copy, Debug, PartialEq, Eq)]
4712enum Unreadable {
4713    /// In a mapping without read, write or execute permission.
4714    NoPermission,
4715    /// In no mapping; `below_growsdown` when the next mapping above it can
4716    /// grow down into it.
4717    Hole { below_growsdown: bool },
4718}

4720/// Where the first byte of `[address, address + len)` that user mode cannot
4721/// read lies, or `None` when user mode can read them all. A range that
4722/// overflows the address space counts as a hole.
4723fn first_user_unreadable(mappings: &[Mapping], address: usize, len: usize) -> Option<Unreadable> {
4724    let Some(end) = address.checked_add(len) else {
4725        return Some(Unreadable::Hole {
4726            below_growsdown: false,
4727        });
4728    };
4729    let mut next = address;
4730    while next < end {
4731        match mappings.iter().find(|mapping| mapping.end > next) {
4732            Some(mapping) if mapping.start <= next => {
4733                if !mapping.user_readable {
4734                    return Some(Unreadable::NoPermission);
4735                }
4736                next = mapping.end;
4737            }
4738            above => {
4739                return Some(Unreadable::Hole {
4740                    below_growsdown: above.is_some_and(|mapping| mapping.growsdown),
4741                });
4742            }
4743        }
4744    }
4745    None
4746}

4751/// orphan with a stopped job, and Linux then sends that whole group SIGHUP and
4752/// SIGCONT at the exiting process's physical exit (`kill_orphaned_pgrp` in
4753/// kernel/exit.c). Apart from a guest's send, only a terminal stops a guest
4754/// process, and a terminal's signals, these two among them, are recorded with
4755/// `terminal_signals`.
4756fn orphaned_group_signals(sig: libc::c_int) -> u64 {
4757    match sig {
4758        libc::SIGSTOP | libc::SIGTSTP | libc::SIGTTIN | libc::SIGTTOU => {
4759            kernel_sigset_bit(libc::SIGHUP) | kernel_sigset_bit(libc::SIGCONT)
4760        }
4761        _ => 0,
4762    }
4763}

4765/// Whether a held signal the scheduler woke a sleep for (`held_woken`, a
4766/// child-exit `SIGCHLD`) ends it: as Linux delivery would, only when this
4767/// thread's mask leaves it unblocked and it is caught or acted on by default
4768/// (`KernelSignalState::interrupting`). A blocked one stays pending, and a
4769/// default-ignored one (`SIGCHLD` without a handler) is discarded, and either
4770/// way the sleep goes on (https://github.com/rrnewton/hermit/issues/4126).
4771///
4772/// The caller has already removed a committed `SIGCHLD` that no longer counts:
4773/// the scheduler sends it to the process, so the kernel gives it to whichever
4774/// thread resumes first without blocking it, and a schedule may run a sibling
4775/// first (`--chaos-target-races`). Taken by that sibling, it is gone from the
4776/// child-exit ledger's send window (`ChildExitSigchldControl::PlacedPending`),
4777/// and the sleep goes on to its original deadline, as a futex wait woken for a
4778/// signal waits again (`futex_wait_held_after_wake`). Answering EINTR for it
4779/// anyway was a result Linux cannot give: the call interrupted, the handler run
4780/// in another thread (the Claude re-check of
4781/// https://github.com/rrnewton/hermit/pull/4131). A held signal a host-timed
4782/// source can also post is decided from the mask and dispositions alone.
4783fn held_wake_ends_sleep(
4784    state: &crate::syscalls::threads::KernelSignalState,
4785    held_woken: KernelSigset,
4786) -> bool {
4787    state.interrupting(state.blocked) & held_woken != 0
4788}

5624    /// The parent's ownership of the forked child: it is killed and reaped
5625    /// when the owner is dropped, so a panic or a cancelled await in the test
5626    /// leaves no process behind. `finish` does the same on the normal path
5627    /// and checks that the child was reaped.
5628    struct ChildOwner {
5629        pid: libc::pid_t,
5630        reaped: bool,
5631    }
5632
5633    impl ChildOwner {
5634        /// Wait for the child to stop. A wait that reports it exited or
5635        /// killed reaped it, so the owner is disarmed before anything can
5636        /// unwind: its pid may already belong to another process.
5637        /// An ECHILD means the child is no longer ours to wait for (it was
5638        /// reaped already), so it disarms the owner too.
5639        fn wait_stopped(&mut self) -> Result<(), Result<(libc::pid_t, libc::c_int), i32>> {
5640            let waited = wait_for(self.pid, libc::WUNTRACED);
5641            match waited {
5642                Ok((pid, status))
5643                    if pid == self.pid
5644                        && (libc::WIFEXITED(status) || libc::WIFSIGNALED(status)) =>
5645                {
5646                    self.reaped = true;
5647                }
5648                Err(libc::ECHILD) => self.reaped = true,
5649                _ => {}
5650            }
5651            match waited {
5652                Ok((pid, status)) if pid == self.pid && libc::WIFSTOPPED(status) => Ok(()),
5653                other => Err(other),
5654            }
5655        }
5656
5657        /// Whether the owner had disarmed itself, read once the child is
5658        /// known to be reaped. The owner is disarmed from here on whatever
5659        /// the answer, so a failed check of it unwinds without signalling a
5660        /// pid that may have been reused.
5661        fn disarmed_after_reap(&mut self) -> bool {
5662            std::mem::replace(&mut self.reaped, true)
5663        }
5664
5665        fn kill_and_reap(&mut self) -> Result<(libc::pid_t, libc::c_int), i32> {
5666            // SAFETY: killing our own child, which is not yet reaped.
5667            unsafe { libc::kill(self.pid, libc::SIGKILL) };
5668            let reaped = wait_for(self.pid, 0);
5669            self.reaped = true;
5670            reaped
5671        }
5672
5673        fn finish(mut self) {
5674            let reaped = self.kill_and_reap();
5675            assert!(
5676                matches!(reaped, Ok((pid, status)) if pid == self.pid
5677                    && libc::WIFSIGNALED(status)
5678                    && libc::WTERMSIG(status) == libc::SIGKILL),

5680            );
5681        }
