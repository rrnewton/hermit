56/// Signals hermit takes from the guest's namespace, and what a guest loses.
57///
58/// ⚠️ ENUMERATED BY MEASUREMENT, NOT BY READING (2026-08-25). Every signal 1..64
59/// was run under two delivery paths -- self-directed `raise` and a sibling
60/// thread's `pthread_kill` -- with a native run as the control for each. Exactly
61/// TWO differ from native, and they fail in different ways at different points:
62///
63///   SIGSTKFLT (16)  `rt_sigaction` is NO-OPED below, so the handler is never
64///                   installed and the default disposition terminates the guest:
65///                   observed exit 144 (= 128 + 16).
66///   SIGTRAP    (5)  `rt_sigaction` passes through and the handler IS installed,
67///                   but ptrace consumes every SIGTRAP (syscall stops, seccomp
68///                   stops, breakpoints), so the handler never runs. ⚠️ THE GUEST
69///                   THEN EXITS 0 WITH NO DIAGNOSTIC AT ALL -- a clean pass that
70///                   behaved differently from native, which no cell can catch.
71///
72/// Everything else in 1..31 matches native exactly; 9/19 and 32/33 refuse
73/// `sigaction` natively too and are not hermit's. Realtime 34..64 fail by a
74/// different mechanism (they cannot be represented at the reverie ptrace
75/// boundary) and are tracked separately -- they are not appropriation.
76const APPROPRIATED_SIGNALS: [(i32, &str); 2] = [
77    (
78        libc::SIGTRAP,
79        "ptrace consumes every SIGTRAP (syscall/seccomp stops, breakpoints)",
80    ),
81    (
82        libc::SIGSTKFLT,
83        "reverie uses it as PERF_EVENT_SIGNAL, the PMU preemption timer",
84    ),
85];
86
87/// Say, once per installation, that a guest handler will never run.
88///
89/// ⚠️ WHY A DIAGNOSTIC AND NOT A REFUSAL. Returning `EINVAL` from `sigaction`
90/// was considered and deliberately rejected for SIGSTKFLT -- see the comment at
91/// the no-op below: the Go runtime registers that handler, and refusing would
92/// break every Go guest at startup. That reasoning generalises: installing a
93/// handler defensively is common, actually raising these signals is rare, so
94/// refusal breaks MORE programs than the current behaviour. The contract
95/// question of what a guest is owed here is genuinely open.
96///
97/// What is NOT open is that hermit currently says NOTHING. This line commits to
98/// no policy, breaks no conforming program, and turns a silent wrong answer into
99/// a visible one -- the same reasoning as the `HERMIT_INTERNAL_FAILURE` marker.
100/// The decision, separated from the reporting so a test can exercise THIS and
101/// not a copy of it. A unit test that re-implements a predicate keeps passing
102/// when the real one is gutted -- measured in this project's own pipe work,
103/// where deleting the production wiring left every unit test green.
104fn appropriated_reason(signum: i32, handler: u64) -> Option<&'static str> {
105    // SIG_DFL (0) and SIG_IGN (1) lose nothing: the guest is not asking to be
106    // called back, so there is no expectation to disappoint.
107    if handler <= 1 {
108        return None;
109    }
110    APPROPRIATED_SIGNALS
111        .iter()
112        .find(|(s, _)| *s == signum)
113        .map(|(_, why)| *why)
114}

116fn warn_appropriated_signal(signum: i32, handler: u64) {
117    if let Some(why) = appropriated_reason(signum, handler) {
118        tracing::warn!(
119            "HERMIT_APPROPRIATED_SIGNAL signum={signum} effect=handler-installed-but-never-invoked reason={why}"
120        );
121    }
122}

125// x86_64. `libc::sigset_t` is the 128-byte userspace wrapper type and must not
126// be used to access these buffers. See:

128fn validate_kernel_sigset_size(sigsetsize: usize) -> Result<(), Errno> {
129    if sigsetsize == KERNEL_SIGSET_SIZE {
130        Ok(())
131    } else {
132        Err(Errno::EINVAL)
133    }

136/// The mask Linux installs when a call names `mask` as the signal mask to sleep
137/// under (`rt_sigsuspend`, `ppoll`, `pselect6`): `set_current_blocked` drops
138/// `SIGKILL` and `SIGSTOP`, which can never be blocked.
139pub(crate) fn kernel_installed_signal_mask(mask: KernelSigset) -> KernelSigset {
140    let unblockable = (1_u64 << (libc::SIGKILL - 1)) | (1_u64 << (libc::SIGSTOP - 1));
141    mask & !unblockable
142}

144fn without_perf_event_signal(mask: KernelSigset) -> KernelSigset {
145    let bit = (reverie::PERF_EVENT_SIGNAL as u32) - 1;
146    mask & !(1_u64 << bit)
147}

149/// Read one raw kernel signal mask while preserving the kernel's user-access check.
150///
151/// `safeptrace::Stopped::read` deliberately uses `PTRACE_PEEKDATA` for reads of
152/// eight bytes or less. That operation can read a `PROT_NONE` page, unlike the
153/// kernel's `copy_from_user`, so using `MemoryAccess::read_value` alone would
154/// turn an `EFAULT` from a raw signal syscall into success. An invalid `how`
155/// value makes `rt_sigprocmask` copy exactly the kernel-sized input and then
156/// return `EINVAL` without changing the mask. Use that as a permission probe,
157/// then read the already-validated word while the guest is stopped.
158/// The x86-64 System V red zone below the stack pointer, which a leaf function
159/// may use and an injected call therefore must not.
160const STACK_RED_ZONE: usize = 128;
161
162/// The two eight-byte scratch cells `handle_rt_sigsuspend` places below the
163/// red zone: the rt_sigpending result and the copy of the call's mask.
164const SCRATCH_CELLS_SIZE: usize = 2 * std::mem::size_of::<u64>();
165
166pub(super) async fn read_kernel_sigset<G, T>(
167    guest: &mut G,
168    address: Addr<'_, libc::sigset_t>,
169) -> Result<KernelSigset, Error>
170where
171    G: Guest<Detcore<T>>,
172    T: RecordOrReplay,
173{
174    let validation = syscalls::RtSigprocmask::new()
175        .with_how(-1)
176        .with_set(Some(address))
177        .with_oldset(None)
178        .with_sigsetsize(KERNEL_SIGSET_SIZE);
179    match guest.inject(validation).await {
180        Err(Errno::EINVAL) => {}
181        Err(errno) => return Err(errno.into()),
182        Ok(_) => {
183            // Both Linux and the KVM syscall implementation reject an unknown
184            // operation. Success means the backend did not validate the probe.
185            return Err(Errno::EIO.into());
186        }
187    }
188    Ok(guest.memory().read_value(address.cast())?)
189}

191/// Signal phase 1: the guest's SIGALRM disposition (a guest handler or not)
192/// and SIGALRM blocked bit, as its runtime keeps them virtual, read back with
193/// a query `rt_sigaction` and `rt_sigprocmask` that the runtime answers from
194/// its virtual state.
195async fn virtual_sigalrm_state<G, T>(guest: &mut G) -> Result<(bool, bool), Error>
196where
197    G: Guest<Detcore<T>>,
198    T: RecordOrReplay,
199{
200    let state = read_sigalrm_sleep_state(guest).await?;
201    Ok((
202        state.disposition == SigalrmSleepDisposition::Handled,
203        state.blocked,
204    ))
205}

207/// Query through the runtime, which reports the guest's action and mask even
208/// when it owns a physical SIGALRM handler. These queries run in an owned turn;
209/// a scheduler-side read of the callback's physical mask is not a substitute.
210async fn read_sigalrm_sleep_state<G, T>(guest: &mut G) -> Result<InGuestSleepSigalrm, Error>
211where
212    G: Guest<Detcore<T>>,
213    T: RecordOrReplay,
214{
215    let mut stack = guest.stack().await;
216    let action = stack.reserve::<KernelSigaction>();
217    let mask = stack.reserve::<KernelSigset>();
218    let _stack_guard = stack.commit()?;
219    guest
220        .inject(
221            syscalls::RtSigaction::new()
222                .with_signum(libc::SIGALRM)
223                .with_action(None)
224                .with_old_action(Some(action.cast()))
225                .with_sigsetsize(KERNEL_SIGSET_SIZE),
226        )
227        .await?;
228    guest
229        .inject(
230            syscalls::RtSigprocmask::new()
231                .with_how(libc::SIG_BLOCK)
232                .with_set(None)
233                .with_oldset(Some(mask.cast()))
234                .with_sigsetsize(KERNEL_SIGSET_SIZE),
235        )
236        .await?;
237    let action: KernelSigaction = guest.memory().read_value(action)?;
238    let mask: KernelSigset = guest.memory().read_value(mask)?;
239    let disposition = if action.handler == libc::SIG_DFL as u64 {
240        SigalrmSleepDisposition::Default
241    } else if action.handler == libc::SIG_IGN as u64 {
242        SigalrmSleepDisposition::Ignored
243    } else {
244        SigalrmSleepDisposition::Handled
245    };
246    Ok(InGuestSleepSigalrm {
247        disposition,
248        blocked: mask & kernel_sigset_bit(libc::SIGALRM) != 0,
249    })
250}

275/// Validate the entire action through the kernel before copying it privately.
276/// A partial `read_value` can fall back to ptrace for its final eight bytes,
277/// which would read through a protected page. SIGKILL accepts no new action,
278/// but both Linux and KVM copy the complete input before rejecting it.
279async fn read_kernel_sigaction<G, T>(
280    guest: &mut G,
281    address: Addr<'_, libc::sigaction>,
282) -> Result<KernelSigaction, Error>
283where
284    G: Guest<Detcore<T>>,
285    T: RecordOrReplay,

287    let validation = syscalls::RtSigaction::new()
288        .with_signum(libc::SIGKILL)
289        .with_action(Some(address))
290        .with_old_action(None)
291        .with_sigsetsize(KERNEL_SIGSET_SIZE);
292    match guest.inject(validation).await {
293        Err(Errno::EINVAL) => {}
294        Err(errno) => return Err(errno.into()),
295        Ok(_) => return Err(Errno::EIO.into()),
296    }
297    Ok(guest.memory().read_value(address.cast())?)
298}

302fn timeval_to_logical_time(value: libc::timeval) -> Result<LogicalTime, Errno> {
303    let seconds = u64::try_from(value.tv_sec).map_err(|_| Errno::EINVAL)?;
304    let micros = u64::try_from(value.tv_usec).map_err(|_| Errno::EINVAL)?;
305    if micros >= 1_000_000 {
306        return Err(Errno::EINVAL);
307    }
308    let nanos = seconds
309        .checked_mul(1_000_000_000)
310        .and_then(|nanos| nanos.checked_add(micros * 1_000))
311        .ok_or(Errno::EINVAL)?;
312    Ok(LogicalTime::from_nanos(nanos))
313}

351/// Whether one of Linux's three ordinary signal syscalls names the calling
352/// task exactly and asks for the one signal that cannot return successfully.
353///
354/// Keep process-group and broadcast spellings out of this predicate. Even when
355/// the caller belongs to the named group, KVM can refuse a group containing a
356/// second process; reserving an exit before that refusal would strand the
357/// scheduler's terminal barrier.
358fn self_sigkill_targets_current_task(
359    signal: libc::c_int,
360    target_process: Option<DetPid>,
361    target_thread: Option<DetTid>,
362    current_process: DetPid,
363    current_thread: DetTid,
364) -> bool {
365    signal == libc::SIGKILL
366        && (target_process.is_some() || target_thread.is_some())
367        && target_process.is_none_or(|target| target == current_process)
368        && target_thread.is_none_or(|target| target == current_thread)
369}

372    /// Bind a supported in-guest emulated sleep to its guest disposition/mask.
373    /// Ordinary admitted creation does not share signal dispositions between
374    /// processes, and each supported in-guest process has one guest thread.
375    /// The snapshot therefore stays current until this request resumes.
376    pub(crate) async fn in_guest_sleep_sigalrm_snapshot<G: Guest<Self>>(
377        &self,
378        guest: &mut G,
379    ) -> Option<InGuestSleepSigalrm> {
380        if !self.cfg.backend.virtualizes_guest_sigalrm {
381            return None;
382        }
383        match read_sigalrm_sleep_state(guest).await {
384            Ok(state) => Some(state),
385            Err(error) => {
386                self.stop_after_in_guest_sleep_signal_loss(
387                    guest,
388                    format!("could not query its SIGALRM action/mask before sleeping: {error}"),
389                )
390                .await
391            }
392        }
393    }

395    /// A typed wake means the scheduler generated a one-shot SIGALRM for this
396    /// exact sleep without sending it physically. Stage it through the frozen
397    /// runtime self-signal path, which runs signal and exit callbacks before
398    /// Linux kills the process. The caller must immediately return its handler:
399    /// it must neither resume guest code nor copy nanosleep's remainder out.
400    pub(crate) async fn stage_in_guest_fatal_alarm<G: Guest<Self>>(&self, guest: &mut G) {
401        if !self.cfg.backend.virtualizes_guest_sigalrm {
402            self.stop_after_in_guest_sleep_signal_loss(
403                guest,
404                "received an in-guest fatal alarm on another backend".into(),
405            )
406            .await;
407        }
408        match read_sigalrm_sleep_state(guest).await {
409            Ok(InGuestSleepSigalrm {
410                disposition: SigalrmSleepDisposition::Default,
411                blocked: false,
412            }) => {}
413            other => {
414                self.stop_after_in_guest_sleep_signal_loss(
415                    guest,
416                    format!("its default/unblocked SIGALRM sleep snapshot changed: {other:?}"),
417                )
418                .await;
419            }
420        }
421        let call = syscalls::Tgkill::new()
422            .with_tgid(guest.pid().as_raw())
423            .with_tid(guest.tid().as_raw())
424            .with_sig(libc::SIGALRM);
425        if guest.inject(call).await != Ok(0) {
426            self.stop_after_in_guest_sleep_signal_loss(
427                guest,
428                "could not stage its default-fatal SIGALRM after a sleep wake".into(),
429            )
430            .await;
431        }
432    }

480    /// Schedule a one-shot or periodic real-time interval timer on Detcore logical time.
481    pub async fn handle_setitimer<G: Guest<Self>>(
482        &self,
483        guest: &mut G,
484        call: syscalls::Setitimer,
485    ) -> Result<i64, Error> {
486        if !guest.config().sequentialize_threads {
487            info!(
488                "[dtid {}] Running without scheduler, so letting setitimer call through...",
489                guest.thread_state().dettid
490            );
491            return Ok(guest.inject(call).await?);
492        }
493        if call.which() != libc::ITIMER_REAL {
494            return Err(Error::Errno(Errno::ENOSYS));
495        }
496
497        let value = call.value().ok_or(Errno::EFAULT)?;
498        let timer: libc::itimerval = guest.memory().read_value(value)?;
499        let interval = timeval_to_logical_time(timer.it_interval)?;
500        let duration = timeval_to_logical_time(timer.it_value)?;
501        // A zero value disarms the timer, whatever its interval.
502        if duration.as_nanos() != 0 && interval.as_nanos() != 0 {
503            refuse_sigalrm(guest, SigalrmControl::ArmRecurringTimer).await?;
504        }
505        let (remaining, old_interval) =
506            register_alarm(guest, duration, interval, Signal::SIGALRM).await;
507        if let Some(old_value) = call.ovalue() {
508            let old_timer = libc::itimerval {
509                it_interval: logical_time_to_timeval(old_interval),
510                it_value: logical_time_to_timeval(remaining),
511            };
512            guest.memory().write_value(old_value, &old_timer)?;
513        }
514        Ok(0)
515    }

549    /// A pause is really just an unbounded sleep.
550    pub async fn handle_pause<G: Guest<Self>>(
551        &self,
552        guest: &mut G,
553        call: syscalls::Pause,
554    ) -> Result<i64, Error> {
555        if guest.config().sequentialize_threads {
556            // `pause` has no deadline: it returns only when a signal is delivered.
557            // `LogicalTime::INDEFINITE` records that, and the scheduler refuses to
558            // fast-forward virtual time onto it (see `step2d_handle_empty_queue`),
559            // so the `Normal` arm below stays unreachable.
560            loop {
561                // A signal that already ended the call before it parked: one sent
562                // while this thread was stopped short of filing the request, as
563                // when its timeslice ended in the syscall's prehook.
564                if sleep_signal(guest, SleepCheck::BeforePark, true).await? == SleepSignal::Pending
565                {
566                    return Err(Errno::ERESTARTNOHAND.into());
567                }
568                let mut req = Self::sleep_request_abs(guest, LogicalTime::INDEFINITE).await;
569                req.fyi(crate::resources::PAUSE_FYI);
570                req.in_guest_sleep_sigalrm = self.in_guest_sleep_sigalrm_snapshot(guest).await;
571                match crate::tool_global::parked_wait_request(
572                    guest,
573                    req,
574                    crate::scheduler::parked::ParkedWaitPolicy::PauseNoHandlerRestart,
575                )
576                .await
577                {
578                    ResumeStatus::InGuestFatalAlarm => {
579                        self.stage_in_guest_fatal_alarm(guest).await;
580                        return Err(Errno::EINTR.into());
581                    }
582                    ResumeStatus::Normal => {
583                        panic!(
584                            "Internal violation: pause should never return from the scheduler except by interruption!"
585                        )
586                    }
587                    ResumeStatus::Signaled(signals) => {
588                        match sleep_signal(guest, SleepCheck::AfterWake(signals), true).await? {
589                            SleepSignal::Pending => return Err(Errno::ERESTARTNOHAND.into()),
590                            SleepSignal::Held => return Err(Errno::EINTR.into()),
591                            SleepSignal::None => {
592                                info!(
593                                    "[dtid {}] pause woken for a signal no longer pending for it; waiting again",
594                                    guest.thread_state().dettid
595                                );
596                            }
597                        }
598                    }
599                }
600            }
601        } else {
602            info!(
603                "[dtid {}] Running without scheduler, so letting pause call through...",
604                guest.thread_state().dettid
605            );
606            Ok(guest.inject(call).await?)
607        }

610    /// Run rt_sigsuspend without holding the deterministic scheduler turn.
611    ///
612    /// The kernel must perform the temporary mask swap atomically and restore the
613    /// original mask after signal delivery, so execute the real blocking syscall
614    /// while marking this thread as blocked outside the runnable set.
615    pub async fn handle_rt_sigsuspend<G: Guest<Self>>(
616        &self,
617        guest: &mut G,
618        call: syscalls::RtSigsuspend,
619    ) -> Result<i64, Error> {
620        // Invalid arguments return immediately from the kernel and therefore are
621        // not signal-only waits.
622        validate_kernel_sigset_size(call.sigsetsize())?;
623        let Some(mask_addr) = call.mask() else {
624            return Err(Errno::EFAULT.into());
625        };
626
627        let temporary_mask = read_kernel_sigset(guest, mask_addr).await?;
628        // The real call sleeps under a private copy of the mask read here, not
629        // under the guest's buffer. The scheduler decides from this copy which
630        // signals can end the call (`Resources::blocked_signal_mask`), and
631        // another process sharing the buffer (MAP_SHARED) may rewrite it before
632        // the call runs; Linux copies the mask when the call starts, and from
633        // the guest's view the call has already started.
634        //
635        // The copy and the rt_sigpending result live in two cells just below
636        // the stack's red zone, the scratch area an injected call may use, but
637        // never on the caller's buffer: a guest may keep its mask there while
638        // it blocks every signal, and a copy placed on that very buffer would
639        // be the original again, while the pending cell would overwrite the
640        // guest's mask. If the buffer overlaps the two cells, they move to just
641        // below it. They are written directly rather than through a scratch
642        // stack, whose commit writes one contiguous region down from the red
643        // zone and so would cover a buffer inside it.
644        let caller_mask = mask_addr.as_raw()..mask_addr.as_raw() + KERNEL_SIGSET_SIZE;
645        let below_red_zone = (guest.regs().await.rsp as usize).wrapping_sub(STACK_RED_ZONE);
646        let cells = (below_red_zone - SCRATCH_CELLS_SIZE)..below_red_zone;
647        let cells_start = if cells.start < caller_mask.end && caller_mask.start < cells.end {
648            (caller_mask.start - SCRATCH_CELLS_SIZE) & !(std::mem::align_of::<u64>() - 1)
649        } else {
650            cells.start
651        };
652        let pending_addr = Addr::<u64>::from_raw(cells_start).ok_or(Errno::EFAULT)?;
653        let mask_copy = AddrMut::<u64>::from_raw(cells_start + std::mem::size_of::<u64>())
654            .ok_or(Errno::EFAULT)?;
655        guest.memory().write_value(mask_copy, &temporary_mask)?;
656        let pending_out = AddrMut::<libc::sigset_t>::from_raw(pending_addr.as_raw())
657            .expect("stack address must be non-null");
658        let pending_call = syscalls::RtSigpending::new()
659            .with_set(Some(pending_out))
660            .with_sigsetsize(KERNEL_SIGSET_SIZE);
661        guest.inject_with_retry(pending_call).await?;
662        let pending: u64 = guest.memory().read_value(pending_addr)?;
663        let call = call.with_mask(Some(Addr::from_raw(mask_copy.as_raw()).expect("non-null")));
664
665        // The scheduler records the mask the call sleeps under, read here in
666        // this thread's turn, to choose a SIGCHLD target while the thread is
667        // outside the runnable set (`Resources::blocked_signal_mask`).
668        let installed_mask = kernel_installed_signal_mask(temporary_mask);
669        if pending & !temporary_mask != 0 {
670            // The kernel will consume an already-pending signal as soon as it
671            // atomically installs the temporary mask. Keep this immediate case
672            // out of the terminal-wait classification; the real syscall still
673            // performs delivery and restores the old mask.
674            self.record_or_replay_blocking_with_mask(guest, call.into(), Some(installed_mask))
675                .await
676        } else {
677            self.record_or_replay_rt_sigsuspend(guest, call, installed_mask)
678                .await
679        }
680    }

682    /// rt_sigaction
683    pub async fn handle_rt_sigaction<G: Guest<Self>>(
684        &self,
685        guest: &mut G,
686        call: syscalls::RtSigaction,
687    ) -> Result<i64, Error> {
688        // Linux rejects an invalid size before inspecting either user pointer.
689        // Preserve that EINVAL-before-EFAULT ordering even for the signal that
690        // Hermit reserves for deterministic preemption.
691        validate_kernel_sigset_size(call.sigsetsize())?;
692
693        // Copy the complete kernel object before inspecting the signal number or
694        // writing old_action. This preserves Linux's action-before-old_action
695        // pointer ordering, including when both arguments alias.
696        let kernel_action = match call.action() {
697            Some(action) => Some(read_kernel_sigaction(guest, action).await?),
698            None => None,
699        };
700
701        // Both appropriated signals are reported here, at the one point where the
702        // guest states its expectation. SIGTRAP falls through to the ordinary
703        // path below (its handler really is installed; ptrace just eats the
704        // signal), so this must run before the SIGSTKFLT early return.
705        if let Some(action) = kernel_action {
706            warn_appropriated_signal(call.signum(), action.handler);
707        }
708
709        // PERF_EVENT_SIGNAL is reserved.
710        if call.signum() == reverie::PERF_EVENT_SIGNAL as i32 {
711            // The go runtime attempts to register this (unused) signal handler.  We will never
712            // deliver signals of this kind to the guest, so we just turn this action into a noop
713            // rather than returning `Err(Errno::EINVAL.into())`. Preserve that established
714            // policy while still honoring the raw syscall's pointer accesses: the virtual old
715            // disposition is the default action, never Reverie's private preemption handler.
716            if call.old_action().is_some() {
717                // SIGKILL's disposition cannot be changed by userspace, so its
718                // query copies the same four zero words as our virtual default.
719                // Let the kernel copy them: a short write_value can finish with
720                // PTRACE_POKEDATA and overwrite a protected final word. This also
721                // preserves the kernel's partial-copy effects on EFAULT without
722                // exposing Reverie's private preemption action.
723                return Ok(guest
724                    .inject(call.with_signum(libc::SIGKILL).with_action(None))
725                    .await?);
726            }
727            return Ok(0);
728        }
729        // Signal phase 1 (step I3): on a backend that keeps a handled SIGALRM
730        // virtual, the scheduler's ledger must agree with the runtime.
731        let phase1 = self.cfg.backend.virtualizes_guest_sigalrm
732            && call.signum() == libc::SIGALRM
733            && kernel_action.is_some();
734        let was_handled = guest.thread_state().sigalrm_handled;
735        let to_handler = kernel_action.is_some_and(|action| {
736            action.handler != libc::SIG_DFL as u64 && action.handler != libc::SIG_IGN as u64
737        });
738        if phase1 {
739            if to_handler {
740                self.refuse_sigalrm_handler(guest).await?;
741            } else if was_handled
742                && kernel_action.is_some_and(|action| action.handler == libc::SIG_DFL as u64)
743            {
744                refuse_sigalrm(guest, SigalrmControl::HandlerToDefault).await?;
745            }
746        }
747        let result = if let Some(kernel_action) = kernel_action {
748            // The kernel treats `action` as input. Sanitize a private copy instead
749            // of writing through the guest pointer: the input may be read-only,
750            // and changing it would make the syscall wrapper observable.
751            let mut kernel_action = kernel_action;
752            kernel_action.mask = without_perf_event_signal(kernel_action.mask);
753            let mut stack = guest.stack().await;
754            let sanitized_action = stack.push(kernel_action);
755            let _stack_guard = stack.commit()?;
756            guest
757                .inject(call.with_action(Some(sanitized_action.cast())))
758                .await
759        } else {
760            guest.inject(call).await
761        };
762        // Publish the state the runtime now holds whenever the disposition may
763        // have changed: from a handler, or to one the runtime did not refuse.
764        // Any result counts, since an EFAULT copying the old action out comes
765        // after the change. A default or ignored action set while none is
766        // handled changes nothing the scheduler records, so it costs nothing.
767        if phase1 && (was_handled || (to_handler && result != Err(Errno::EPERM))) {
768            self.publish_virtual_sigalrm(guest).await?;
769        }
770        Ok(result?)
771    }

774    /// each EPERM before anything changes (design section 6 and closure 5):
775    /// threads not sequentialized (`alarm` would arm a kernel timer), a
776    /// signalfd in the process's descriptor table, and a kernel producer of

862            // do not expose a tracer preemption as ERESTARTSYS to the guest.
863            Ok(guest.inject_with_retry(modified_call).await?)
864        } else {
865            Ok(guest.inject_with_retry(call).await?)
866        }

956    /// unambiguous positive-PID process signal through the backend. Backends that can execute
957    /// with guest PIDs preserve process-directed delivery; DBT translates it to the sole live
958    /// thread because its native process uses a host PID. An unmaskable SIGKILL to a specific

966    ) -> Result<i64, Error> {
967        if !guest.config().sequentialize_threads {
968            return Ok(self.record_or_replay(guest, call).await?);
969        }

971        // Linux's `kill_something_info` answers ESRCH for pid INT_MIN before
972        // anything else ("-INT_MIN is undefined"), so the kernel's answer is
973        // deterministic and sends nothing; negating it here would overflow
974        // (https://github.com/rrnewton/hermit/issues/4046).
975        if call.sig() == 0 || call.pid() == libc::pid_t::MIN {
976            return Ok(self.record_or_replay(guest, call).await?);
977        }
978
979        let tgid = call.pid();
980        // A process-group or broadcast SIGALRM is refused by name below.
981        if call.sig() == libc::SIGALRM && tgid > 0 {
982            refuse_sigalrm(guest, SigalrmControl::SendTo(DetTid::from_raw(tgid))).await?;
983        }

985            tgid,
986            call.sig(),

988                .config()
989                .backend
990                .requires_thread_directed_process_signals,

992            // The scheduler is not told about each killed process, but it must
993            // know the group was killed: a held process in it will die, so a
994            // wait for it is not part of a deadlock
995            // (https://github.com/rrnewton/hermit/issues/3904). Told before
996            // the kill, inside this committed turn: the sender may be in the
997            // group and die before the call returns. A kill that then fails
998            // only suppresses a deadlock report, which is the safe direction.
999            crate::tool_global::notify_unresolved_sigkill(guest, Some(DetPid::from_raw(-tgid)))
1000                .await;
1001            return Ok(self.record_or_replay(guest, call).await?);
1002        }
1003        if tgid <= 0 {
1004            return self.refuse_group_signal(guest, tgid, call.sig()).await;
1005        }
1006        // Exact self-SIGKILL is group-fatal, so it has no recipient-selection
1007        // ambiguity even when the process has several live threads. Reserve it
1008        // before the generic process-signal path rejects that thread set.
1009        if self
1010            .reserve_controlled_self_sigkill_exit(
1011                guest,
1012                call.sig(),
1013                Some(DetPid::from_raw(tgid)),
1014                None,
1015            )
1016            .await
1017        {
1018            return Ok(self.record_or_replay(guest, call).await?);
1019        }
1020        let tid = self
1021            .process_signal_target(guest, Sysno::kill, DetPid::from_raw(tgid), call.sig())
1022            .await?;
1023        let value = if !guest
1024            .config()
1025            .backend
1026            .requires_thread_directed_process_signals
1027        {
1028            self.record_or_replay(guest, call).await?
1029        } else {
1030            let targeted = syscalls::Tgkill::new()
1031                .with_tgid(tgid)
1032                .with_tid(tid.as_raw())
1033                .with_sig(call.sig());
1034            self.record_or_replay(guest, targeted).await?
1035        };
1036        self.notify_cross_task_signal(guest, tid, call.sig(), Some(DetPid::from_raw(tgid)))
1037            .await;
1038        Ok(value)
1039    }

1312/// Which check [`sleep_signal`] makes for an emulated `pause` or `nanosleep`.
1313#[derive(Clone, Debug, Eq, PartialEq)]
1314pub(crate) enum SleepCheck {
1315    /// Before the sleep parks: whether a signal already pending for the thread
1316    /// ends it at once.
1317    BeforePark,
1318    /// After the scheduler woke the sleep for these signals.
1319    AfterWake(Option<Vec<SigWrapper>>),
1320}
1321
1322/// What ends an emulated `pause` or `nanosleep`, per [`sleep_signal`].
1323#[derive(Clone, Copy, Debug, Eq, PartialEq)]
1324pub(crate) enum SleepSignal {
1325    /// A signal pending for this thread, unblocked and caught or acted on by
1326    /// default: the kernel delivers it as the call returns. `pause` returns
1327    /// `ERESTARTNOHAND` for it and `nanosleep` `EINTR`.
1328    Pending,
1329    /// The scheduler ended the sleep for a signal whose pending bit cannot
1330    /// decide (below), or this backend does not report the guest's signal
1331    /// state: the call returns `EINTR`, as every wake did before.
1332    Held,
1333    /// Nothing ends the sleep, and it waits (again) with its original deadline.
1334    None,
1335}
1336
1337/// Decides, in this thread's own turn, whether a signal ends an emulated
1338/// `pause` or `nanosleep` (<https://github.com/rrnewton/hermit/issues/3982>).
1339///
1340/// Only on a backend whose kernel reports the guest's signal state; on any
1341/// other, a sleep never checks before it parks, and a wake always ends it
1342/// (`Held`). A signal ends the sleep when it is pending for this thread,
1343/// unblocked, and caught or acted on by default
1344/// (`KernelSignalState::pending_interrupting`). Before the sleep parks, a stop
1345/// signal does so only when `stops_end_it`, which `nanosleep` leaves unset
1346/// because Detcore does not keep its deadline for the kernel's restart after a
1347/// stop, so the stop takes effect when the sleep ends, as before. After a wake,
1348/// which only the scheduler's own sends (a timer, a child-exit `SIGCHLD`) can
1349/// commit for a stop on a `nanosleep`, a pending stop ends it, as every wake
1350/// did before, so that the stop takes effect promptly.
1351///
1352/// `SIGCHLD` and the signals a host-timed source armed by a guest can post
1353/// (`host_timed_signals`) are held: the kernel can post them at a moment set by
1354/// host timing, and `/proc` cannot tell that copy from one a guest sent, so
1355/// their pending bits are never read. They never end a sleep before it parks,
1356/// nor does the scheduler record a guest's cross-task send of one for a sleep;
1357/// a wake the scheduler committed for one of its own, a child-exit `SIGCHLD`,
1358/// ends it with `Held`, as before. Like the precise futex
1359/// wait's check, this reads the pending set before the sleep parks as well as
1360/// after a wake: a signal sent while the thread was stopped short of filing
1361/// its request, as when its timeslice ended in the syscall's prehook, is
1362/// pending and was never recorded for its sleep. After a wake whose
1363/// non-held signals are no longer pending, a `SIGCONT` having cancelled a stop
1364/// or `sigaction(SIG_IGN)` having flushed it, a Linux sleep would still be
1365/// asleep, so this returns `None`.
1366///
1367/// A state that cannot be read is handled as a wait's first read is
1368/// (`read_wait_signal_state`, here `read_thread_wait_signal_state`, since
1369/// Reverie can report the creator's pid for a process a raw `clone` made): a
1370/// thread that no longer exists ends the call
1371/// with `ERESTARTNOINTR`, which nothing observes, and otherwise the run ends
1372/// with a diagnostic.
1373pub(crate) async fn sleep_signal<G, T>(
1374    guest: &mut G,
1375    check: SleepCheck,
1376    stops_end_it: bool,
1377) -> Result<SleepSignal, Error>
1378where
1379    G: Guest<Detcore<T>>,
1380    T: RecordOrReplay,
1381{
1382    if !guest
1383        .config()
1384        .backend_supports_blocked_wait_signal_interruption
1385    {
1386        return Ok(match check {
1387            SleepCheck::BeforePark => SleepSignal::None,
1388            SleepCheck::AfterWake(_) => SleepSignal::Held,
1389        });
1390    }
1391    let held = kernel_sigset_bit(libc::SIGCHLD) | host_timed_signals(guest).await;
1392    let before_park = check == SleepCheck::BeforePark;
1393    let (deciding, held_wake) = match check {
1394        SleepCheck::BeforePark => (!held, false),
1395        SleepCheck::AfterWake(None) => (0, true),
1396        SleepCheck::AfterWake(Some(woken)) => {
1397            let woken = woken
1398                .iter()
1399                .fold(0, |set, signal| set | kernel_sigset_bit(signal.raw()));
1400            (woken & !held, woken & held != 0)
1401        }
1402    };
1403    if deciding != 0 {
1404        let state = read_thread_wait_signal_state(guest.pid(), guest.tid())?;
1405        let mut ending = state.pending_interrupting(state.blocked, false) & deciding;
1406        if !stops_end_it && before_park {
1407            ending &= !(state.default_job_control_stops() | kernel_sigset_bit(libc::SIGSTOP));
1408        }
1409        if ending != 0 {
1410            return Ok(SleepSignal::Pending);
1411        }
1412    }
1413    Ok(if held_wake {
1414        SleepSignal::Held
1415    } else {
1416        SleepSignal::None
1417    })
1418}

1804    /// The real `rt_sigsuspend` sleeps under the mask Detcore read when it
1805    /// handled the call, the one the scheduler is given, even if the guest's
1806    /// buffer changes before the call runs. Linux copies the mask when the call
1807    /// starts, and from the guest's view it has started. Before, the call ran
1808    /// with the guest's pointer: another process sharing the buffer could make
1809    /// it block a signal the scheduler had counted on to end it, and the
1810    /// scheduler would hold every other thread waiting for that wake.
1811    #[tokio::test(flavor = "current_thread")]
1812    async fn rt_sigsuspend_sleeps_under_the_mask_detcore_read() {
1813        let config = Config::default();
1814        assert!(!config.sequentialize_threads);
1815        let tool = <Detcore as Tool>::new(Pid::from_raw(1), &config);
1816        let unblocks_sigchld: KernelSigset = kernel_sigset_bit(libc::SIGUSR2);
1817        let blocks_sigchld = unblocks_sigchld | kernel_sigset_bit(libc::SIGCHLD);
1818        let mut caller_mask: Box<KernelSigset> = Box::new(unblocks_sigchld);
1819        let caller_address = &mut *caller_mask as *mut KernelSigset as usize;
1820        let mut guest = SuspendGuest {
1821            thread: ThreadState::new(DetPid::from_raw(1), &config, ()),
1822            config: config.clone(),
1823            arena: Box::new([u64::MAX; ARENA_WORDS]),
1824            caller_mask: caller_address,
1825            rewritten: Some(blocks_sigchld),
1826            suspended_under: Vec::new(),
1827        };
1828        let call = syscalls::RtSigsuspend::new()
1829            .with_mask(Addr::from_raw(caller_address))
1830            .with_sigsetsize(KERNEL_SIGSET_SIZE);
1831
1832        let result = tool.handle_rt_sigsuspend(&mut guest, call).await;
1833        assert!(
1834            matches!(result, Err(Error::Errno(Errno::EINTR))),
1835            "{result:?}"

1841    /// A caller's mask that lies on the cells Detcore places below the red
1842    /// zone, as a raw syscall may pass one there in a shared stack: the copy
1843    /// must not be the caller's buffer, or a rewrite of the shared buffer
1844    /// reaches the real call again, and the rt_sigpending cell must not
1845    /// overwrite it. The cells move below the buffer, which keeps its own
1846    /// contents. Before, the copy slot was the caller's buffer, and the
1847    /// pending slot's buffer was overwritten with the pending set.
1848    #[tokio::test(flavor = "current_thread")]
1849    async fn rt_sigsuspend_scratch_never_lands_on_the_callers_mask() {
1850        let config = Config::default();
1851        let tool = <Detcore as Tool>::new(Pid::from_raw(1), &config);
1852        let original: KernelSigset = kernel_sigset_bit(libc::SIGUSR2);
1853        let rewritten = original | kernel_sigset_bit(libc::SIGCHLD);
1854        // The copy's slot, rewritten by a peer; then the pending cell's slot,
1855        // which nobody rewrites.
1856        for (slot, peer_rewrite) in [(COPY_WORD, Some(rewritten)), (COPY_WORD - 1, None)] {
1857            let mut arena = Box::new([u64::MAX; ARENA_WORDS]);
1858            arena[slot] = original;
1859            let caller_address = &mut arena[slot] as *mut u64 as usize;
1860            let mut guest = SuspendGuest {
1861                thread: ThreadState::new(DetPid::from_raw(1), &config, ()),
1862                config: config.clone(),
1863                arena,
1864                caller_mask: caller_address,
1865                rewritten: peer_rewrite,
1866                suspended_under: Vec::new(),
1867            };
1868            let call = syscalls::RtSigsuspend::new()
1869                .with_mask(Addr::from_raw(caller_address))
1870                .with_sigsetsize(KERNEL_SIGSET_SIZE);
1871
1872            let result = tool.handle_rt_sigsuspend(&mut guest, call).await;
1873            assert!(
1874                matches!(result, Err(Error::Errno(Errno::EINTR))),
1875                "slot {slot}: {result:?}"
1876            );
1877            assert_eq!(guest.suspended_under, [original], "slot {slot}");
1878            assert_eq!(
1879                guest.arena[slot],
1880                peer_rewrite.unwrap_or(original),
1881                "slot {slot}: the caller's buffer holds only what the guest wrote"
1882            );
1883        }
1884    }
