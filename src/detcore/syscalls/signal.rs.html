42/// Signals hermit takes from the guest's namespace, and what a guest loses.
43///
44/// ⚠️ ENUMERATED BY MEASUREMENT, NOT BY READING (2026-08-25). Every signal 1..64
45/// was run under two delivery paths -- self-directed `raise` and a sibling
46/// thread's `pthread_kill` -- with a native run as the control for each. Exactly
47/// TWO differ from native, and they fail in different ways at different points:
48///
49///   SIGSTKFLT (16)  `rt_sigaction` is NO-OPED below, so the handler is never
50///                   installed and the default disposition terminates the guest:
51///                   observed exit 144 (= 128 + 16).
52///   SIGTRAP    (5)  `rt_sigaction` passes through and the handler IS installed,
53///                   but ptrace consumes every SIGTRAP (syscall stops, seccomp
54///                   stops, breakpoints), so the handler never runs. ⚠️ THE GUEST
55///                   THEN EXITS 0 WITH NO DIAGNOSTIC AT ALL -- a clean pass that
56///                   behaved differently from native, which no cell can catch.
57///
58/// Everything else in 1..31 matches native exactly; 9/19 and 32/33 refuse
59/// `sigaction` natively too and are not hermit's. Realtime 34..64 fail by a
60/// different mechanism (they cannot be represented at the reverie ptrace
61/// boundary) and are tracked separately -- they are not appropriation.
62const APPROPRIATED_SIGNALS: [(i32, &str); 2] = [
63    (
64        libc::SIGTRAP,
65        "ptrace consumes every SIGTRAP (syscall/seccomp stops, breakpoints)",
66    ),
67    (
68        libc::SIGSTKFLT,
69        "reverie uses it as PERF_EVENT_SIGNAL, the PMU preemption timer",
70    ),
71];
72
73/// Say, once per installation, that a guest handler will never run.
74///
75/// ⚠️ WHY A DIAGNOSTIC AND NOT A REFUSAL. Returning `EINVAL` from `sigaction`
76/// was considered and deliberately rejected for SIGSTKFLT -- see the comment at
77/// the no-op below: the Go runtime registers that handler, and refusing would
78/// break every Go guest at startup. That reasoning generalises: installing a
79/// handler defensively is common, actually raising these signals is rare, so
80/// refusal breaks MORE programs than the current behaviour. The contract
81/// question of what a guest is owed here is genuinely open.
82///
83/// What is NOT open is that hermit currently says NOTHING. This line commits to
84/// no policy, breaks no conforming program, and turns a silent wrong answer into
85/// a visible one -- the same reasoning as the `HERMIT_INTERNAL_FAILURE` marker.
86/// The decision, separated from the reporting so a test can exercise THIS and
87/// not a copy of it. A unit test that re-implements a predicate keeps passing
88/// when the real one is gutted -- measured in this project's own pipe work,
89/// where deleting the production wiring left every unit test green.
90fn appropriated_reason(signum: i32, handler: u64) -> Option<&'static str> {
91    // SIG_DFL (0) and SIG_IGN (1) lose nothing: the guest is not asking to be
92    // called back, so there is no expectation to disappoint.
93    if handler <= 1 {
94        return None;
95    }
96    APPROPRIATED_SIGNALS
97        .iter()
98        .find(|(s, _)| *s == signum)
99        .map(|(_, why)| *why)
100}
101
102fn warn_appropriated_signal(signum: i32, handler: u64) {
103    if let Some(why) = appropriated_reason(signum, handler) {
104        tracing::warn!(
105            "HERMIT_APPROPRIATED_SIGNAL signum={signum} effect=handler-installed-but-never-invoked reason={why}"
106        );
107    }
108}

118fn timeval_to_logical_time(value: libc::timeval) -> Result<LogicalTime, Errno> {
119    let seconds = u64::try_from(value.tv_sec).map_err(|_| Errno::EINVAL)?;
120    let micros = u64::try_from(value.tv_usec).map_err(|_| Errno::EINVAL)?;
121    if micros >= 1_000_000 {
122        return Err(Errno::EINVAL);
123    }
124    let nanos = seconds
125        .checked_mul(1_000_000_000)
126        .and_then(|nanos| nanos.checked_add(micros * 1_000))
127        .ok_or(Errno::EINVAL)?;
128    Ok(LogicalTime::from_nanos(nanos))
129}

148fn deterministic_kill_target(targets: &[DetTid], sig: libc::c_int) -> Result<DetTid, Errno> {
149    match targets {
150        [] => Err(Errno::ESRCH),
151        [target] => Ok(*target),
152        [target, ..] if sig == 0 => Ok(*target),
153        _ => Err(Errno::ENOSYS),
154    }

171    pub async fn handle_alarm<G: Guest<Self>>(
172        &self,
173        guest: &mut G,
174        call: syscalls::Alarm,
175    ) -> Result<i64, Error> {
176        if guest.config().sequentialize_threads {
177            let remaining = register_alarm(
178                guest,
179                LogicalTime::from_secs(call.seconds() as u64),
180                LogicalTime::ZERO,
181                Signal::SIGALRM,
182            )
183            .await;
184            Ok(logical_time_to_alarm_seconds(remaining.0))
185        } else {
186            info!(
187                "[dtid {}] Running without scheduler, so letting alarm call through...",
188                guest.thread_state().dettid
189            );
190            Ok(guest.inject(call).await?)
191        }
192    }

197    /// Schedule a one-shot or periodic real-time interval timer on Detcore logical time.
198    pub async fn handle_setitimer<G: Guest<Self>>(
199        &self,
200        guest: &mut G,
201        call: syscalls::Setitimer,
202    ) -> Result<i64, Error> {
203        if !guest.config().sequentialize_threads {
204            info!(
205                "[dtid {}] Running without scheduler, so letting setitimer call through...",
206                guest.thread_state().dettid
207            );
208            return Ok(guest.inject(call).await?);
209        }
210        if call.which() != libc::ITIMER_REAL {
211            return Err(Error::Errno(Errno::ENOSYS));
212        }
213
214        let value = call.value().ok_or(Errno::EFAULT)?;
215        let timer: libc::itimerval = guest.memory().read_value(value)?;
216        let interval = timeval_to_logical_time(timer.it_interval)?;
217        let duration = timeval_to_logical_time(timer.it_value)?;
218        let (remaining, old_interval) =
219            register_alarm(guest, duration, interval, Signal::SIGALRM).await;
220        if let Some(old_value) = call.ovalue() {
221            let old_timer = libc::itimerval {
222                it_interval: logical_time_to_timeval(old_interval),
223                it_value: logical_time_to_timeval(remaining),
224            };
225            guest.memory().write_value(old_value, &old_timer)?;
226        }
227        Ok(0)
228    }

233    pub async fn handle_getitimer<G: Guest<Self>>(
234        &self,
235        guest: &mut G,
236        call: syscalls::Getitimer,
237    ) -> Result<i64, Error> {
238        if !guest.config().sequentialize_threads {
239            info!(
240                "[dtid {}] Running without scheduler, so letting getitimer call through...",
241                guest.thread_state().dettid
242            );
243            return Ok(guest.inject(call).await?);
244        }
245
246        let remaining = match call.which() {
247            libc::ITIMER_REAL => alarm_remaining(guest).await,
248            libc::ITIMER_VIRTUAL | libc::ITIMER_PROF => LogicalTime::ZERO,
249            _ => return Err(Errno::EINVAL.into()),
250        };
251        let value = call.value().ok_or(Errno::EFAULT)?;
252        let timer = libc::itimerval {
253            it_interval: logical_time_to_timeval(LogicalTime::ZERO),
254            it_value: logical_time_to_timeval(remaining),
255        };
256        guest.memory().write_value(value, &timer)?;
257        Ok(0)
258    }

260    /// A pause is really just an unbounded sleep.
261    pub async fn handle_pause<G: Guest<Self>>(
262        &self,
263        guest: &mut G,
264        call: syscalls::Pause,
265    ) -> Result<i64, Error> {
266        if guest.config().sequentialize_threads {
267            // `pause` has no deadline: it returns only when a signal is delivered.
268            // `LogicalTime::INDEFINITE` records that, and the scheduler refuses to
269            // fast-forward virtual time onto it (see `step2d_handle_empty_queue`),
270            // so the `Normal` arm below stays unreachable.
271            let req = Self::sleep_request_abs(guest, LogicalTime::INDEFINITE).await;
272            match resource_request(guest, req).await {
273                ResumeStatus::Normal => {
274                    panic!(
275                        "Internal violation: pause should never return from the scheduler except by interruption!"
276                    )
277                }
278                ResumeStatus::Signaled(_) => Err(reverie::Error::Errno(Errno::EINTR)),
279            }
280        } else {
281            info!(
282                "[dtid {}] Running without scheduler, so letting pause call through...",
283                guest.thread_state().dettid
284            );
285            Ok(guest.inject(call).await?)
286        }
287    }

289    /// Run rt_sigsuspend without holding the deterministic scheduler turn.
290    ///
291    /// The kernel must perform the temporary mask swap atomically and restore the
292    /// original mask after signal delivery, so execute the real blocking syscall
293    /// while marking this thread as blocked outside the runnable set.
294    pub async fn handle_rt_sigsuspend<G: Guest<Self>>(
295        &self,
296        guest: &mut G,
297        call: syscalls::RtSigsuspend,
298    ) -> Result<i64, Error> {
299        // Invalid arguments return immediately from the kernel and therefore are
300        // not signal-only waits.
301        if call.sigsetsize() != KERNEL_SIGSET_SIZE {
302            return Err(Errno::EINVAL.into());
303        }
304        let Some(mask_addr) = call.mask() else {
305            return Err(Errno::EFAULT.into());
306        };
307
308        let temporary_mask: u64 = guest
309            .memory()
310            .read_value(mask_addr.cast())
311            .map_err(|_| Errno::EFAULT)?;
312        let mut stack = guest.stack().await;
313        let pending_addr = stack.push(0_u64);
314        let pending_guard = stack.commit()?;
315        let pending_out = AddrMut::<libc::sigset_t>::from_raw(pending_addr.as_raw())
316            .expect("stack address must be non-null");
317        let pending_call = syscalls::RtSigpending::new()
318            .with_set(Some(pending_out))
319            .with_sigsetsize(KERNEL_SIGSET_SIZE);
320        guest.inject_with_retry(pending_call).await?;
321        let pending: u64 = guest.memory().read_value(pending_addr)?;
322        drop(pending_guard);
323
324        if pending & !temporary_mask != 0 {
325            // The kernel will consume an already-pending signal as soon as it
326            // atomically installs the temporary mask. Keep this immediate case
327            // out of the terminal-wait classification; the real syscall still
328            // performs delivery and restores the old mask.
329            self.record_or_replay_blocking(guest, call.into()).await
330        } else {
331            self.record_or_replay_rt_sigsuspend(guest, call).await
332        }
333    }

335    /// rt_sigaction
336    pub async fn handle_rt_sigaction<G: Guest<Self>>(
337        &self,
338        guest: &mut G,
339        call: syscalls::RtSigaction,
340    ) -> Result<i64, Error> {
341        // Both appropriated signals are reported here, at the one point where the
342        // guest states its expectation. SIGTRAP falls through to the ordinary
343        // path below (its handler really is installed; ptrace just eats the
344        // signal), so this must run before the SIGSTKFLT early return.
345        if let Some(action) = call.action() {
346            let handler = guest
347                .memory()
348                .read_value(AddrMut::<u64>::from_raw(action.as_raw()).unwrap())
349                .unwrap_or(0);
350            warn_appropriated_signal(call.signum(), handler);
351        }
352
353        // PERF_EVENT_SIGNAL is reserved.
354        if call.signum() == reverie::PERF_EVENT_SIGNAL as i32 {
355            // The go runtime attempts to register this (unused) signal handler.  We will never
356            // deliver signals of this kind to the guest, so we just turn this action into a noop
357            // rather than returning `Err(Errno::EINVAL.into())`.
358            return Ok(0);
359        }
360        Ok(if let Some(action) = call.action() {
361            let mut memory = guest.memory();
362            let sa_mask: AddrMut<libc::sigset_t> =
363                AddrMut::from_raw(SA_MASK_OFFET + action.as_raw()).unwrap();
364            let mut mask = memory.read_value(sa_mask)?;
365            unsafe { libc::sigdelset(&mut mask as *mut _, reverie::PERF_EVENT_SIGNAL as i32) };
366            memory.write_value(sa_mask, &mask)?;
367            guest.inject(call).await?
368        } else {
369            guest.inject(call).await?
370        })
371    }

375    /// rt_sigprocmask
376    pub async fn handle_rt_sigprocmask<G: Guest<Self>>(
377        &self,
378        guest: &mut G,
379        call: syscalls::RtSigprocmask,
380    ) -> Result<i64, Error> {
381        if call.how() != libc::SIG_BLOCK && call.how() != libc::SIG_SETMASK {
382            Ok(guest.inject_with_retry(call).await?)
383        } else if let Some(set) = call.set() {
384            let memory = guest.memory();
385            let mut stack = guest.stack().await;
386            let mut set_mask = memory.read_value(set)?;
387            unsafe { libc::sigdelset(&mut set_mask as *mut _, reverie::PERF_EVENT_SIGNAL as i32) };
388            let new_set = stack.push(set_mask);
389            stack.commit()?;
390            let modified_call = syscalls::RtSigprocmask::new()
391                .with_how(call.how())
392                .with_set(Some(new_set))
393                .with_oldset(call.oldset())
394                .with_sigsetsize(call.sigsetsize());
395            // Keep returning to the handler so post_handler_hook can run, but
396            // do not expose a tracer preemption as ERESTARTSYS to the guest.
397            Ok(guest.inject_with_retry(modified_call).await?)
398        } else {
399            Ok(guest.inject_with_retry(call).await?)
400        }
401    }

403    /// rt_sigtimedwait system call
404    ///
405    /// This is handled by the scheduler and not passed to the record/replay layer,
406    /// because currently signals are not recorded.
407    pub async fn handle_rt_sigtimedwait<G: Guest<Self>>(
408        &self,
409        guest: &mut G,
410        call: syscalls::RtSigtimedwait,
411    ) -> Result<i64, Error> {
412        let dettid = guest.thread_state().dettid;
413
414        let maybe_timeout = if let Some(timeout) = call.timeout() {
415            let ts: Timespec = guest.memory().read_value(timeout)?;
416            let ns_delta =
417                Duration::from_secs(ts.tv_sec as u64) + Duration::from_nanos(ts.tv_nsec as u64);
418            let base_time = thread_observe_time(guest).await;
419            let target_time = base_time + ns_delta;
420            Some(target_time)
421        } else {
422            None
423        };
424        let mut rsrc = Resources::new(dettid);
425        rsrc.insert(ResourceID::InternalIOPolling, Permission::W);
426        rsrc.fyi("rt_sigtimedwait");
427        retry_nonblocking_syscall_with_timeout(guest, call, rsrc, maybe_timeout).await
428    }

433    // TODO-HUMAN-REVIEW(PR-1119): Review unmaskable process-group SIGKILL forwarding.
434    /// Resolve signal-zero existence checks in the fixed PID namespace, then route an
435    /// unambiguous positive-PID process signal through the backend. Backends that can execute
436    /// with guest PIDs preserve process-directed delivery; DBT translates it to the sole live
437    /// thread because its native process uses a host PID. An unmaskable SIGKILL to a specific
438    /// process group is also safe to preserve on backends whose guests use real namespace PIDs;
439    /// other process-group and broadcast delivery remains refused until Detcore models eligible
440    /// signal masks.
441    pub async fn handle_kill<G: Guest<Self>>(
442        &self,
443        guest: &mut G,
444        call: syscalls::Kill,
445    ) -> Result<i64, Error> {
446        if !guest.config().sequentialize_threads {
447            return Ok(self.record_or_replay(guest, call).await?);
448        }
449
450        if call.sig() == 0 {
451            return Ok(self.record_or_replay(guest, call).await?);
452        }
453
454        let tgid = call.pid();
455        if can_forward_process_group_signal(
456            tgid,
457            call.sig(),
458            guest
459                .config()
460                .backend_requires_thread_directed_process_signals,
461        ) {
462            return Ok(self.record_or_replay(guest, call).await?);
463        }
464        if tgid <= 0 {
465            return Err(Errno::ENOSYS.into());
466        }
467        let targets = resolve_kill_targets(guest, DetPid::from_raw(tgid)).await;
468        let tid = deterministic_kill_target(&targets, call.sig())?;
469        let value = if !guest
470            .config()
471            .backend_requires_thread_directed_process_signals
472        {
473            self.record_or_replay(guest, call).await?
474        } else {
475            let targeted = syscalls::Tgkill::new()
476                .with_tgid(tgid)
477                .with_tid(tid.as_raw())
478                .with_sig(call.sig());
479            self.record_or_replay(guest, targeted).await?
480        };
481        self.notify_cross_task_signal(guest, tid, call.sig(), Some(DetPid::from_raw(tgid)))
482            .await;
483        Ok(value)
484    }

489    /// stable in the fresh PID namespace and delivery is scheduler-serialized.

491        &self,

496        // `pthread_kill` lowers to `tgkill`, so this is the ordinary way one

499        // never woken and the wait hangs.
500        self.notify_cross_task_signal(guest, DetTid::from_raw(call.tid()), call.sig(), None)
501            .await;
502        Ok(value)
503    }

524    /// Tell the scheduler that a successful thread-directed send left a signal
525    /// physically pending for another task.
526    ///
527    /// Shared by `kill`, `tgkill` and `tkill` so the three cannot drift: a fix
528    /// applied to one spelling of "signal another task" must apply to all of
529    /// them, which is exactly the gap that let a `pthread_kill` from a sibling
530    /// thread hang a `waitid` that a `kill` from a sibling process could
531    /// interrupt. Self-directed signals are excluded: the sender is running, so
532    /// it is not parked waiting to be woken.
533    async fn notify_cross_task_signal<G: Guest<Self>>(
534        &self,
535        guest: &mut G,
536        target: DetTid,
537        raw_signal: i32,
538        target_process: Option<DetPid>,
539    ) {
540        // ⚠️ NO `Signal::try_from` GATE. It used to stand here, and because
541        // `nix`'s `Signal` models only 1..=31 it rejected EVERY realtime signal:
542        // measured in-tree, the gate admitted exactly 1..=31 and zero of the 31
543        // realtime signals. The notification was skipped silently, so a target
544        // parked on `ResourceID::WaitChild` was never woken and the wait hung.
545        // `SigWrapper` now carries the raw number, so every deliverable signal
546        // can be represented and none is dropped for being unnameable. Signal
547        // zero is only an existence/permission probe and queues no signal.
548        if should_notify_cross_task_signal(guest.thread_state().dettid, target, raw_signal) {
549            notify_signal_pending(guest, target, SigWrapper(raw_signal), target_process).await;
550        }
551    }

555    /// Queue a thread-directed signal with an accompanying `siginfo_t`. Like
556    /// `tgkill`, the target is a specific thread named by stable guest TGID/TID
557    /// and delivery is scheduler-serialized; the guest-supplied siginfo is
558    /// deterministic input, so forwarding the kernel call is deterministic.
559    pub async fn handle_rt_tgsigqueueinfo<G: Guest<Self>>(
560        &self,
561        guest: &mut G,
562        call: syscalls::RtTgsigqueueinfo,
563    ) -> Result<i64, Error> {
564        let value = self.record_or_replay(guest, call).await?;
565        // Thread-directed like `tgkill`, so it carries the same wakeup
566        // obligation. `sigqueue`/`pthread_sigqueue` reach a parked sibling
567        // through here.
568        self.notify_cross_task_signal(guest, DetTid::from_raw(call.tid()), call.sig(), None)
569            .await;
570        Ok(value)
571    }
572
573    // AUTONOMOUS-BOT-IMPLEMENTED
574    // TODO-HUMAN-REVIEW(#812)
575    // TODO-HUMAN-REVIEW(PR-1058): Review queued process-signal preservation.
576    /// Queue a process-directed signal with an accompanying `siginfo_t`. Mirrors `handle_kill`:
577    /// preserve process-directed delivery when the backend accepts guest PIDs, otherwise route an
578    /// unambiguous positive-PID target to its sole live thread via `rt_tgsigqueueinfo`. Ambiguous
579    /// multithreaded process-directed delivery is refused until Detcore models eligible masks.
580    pub async fn handle_rt_sigqueueinfo<G: Guest<Self>>(
581        &self,
582        guest: &mut G,
583        call: syscalls::RtSigqueueinfo,
584    ) -> Result<i64, Error> {
585        if !guest.config().sequentialize_threads {
586            return Ok(self.record_or_replay(guest, call).await?);
587        }
588
589        if call.sig() == 0 {
590            return Ok(self.record_or_replay(guest, call).await?);
591        }
592
593        let tgid = call.tgid();
594        if tgid <= 0 {
595            return Err(Errno::ENOSYS.into());
596        }
597        let targets = resolve_kill_targets(guest, DetPid::from_raw(tgid)).await;
598        let tid = deterministic_kill_target(&targets, call.sig())?;
599        let value = if !guest
600            .config()
601            .backend_requires_thread_directed_process_signals
602        {
603            self.record_or_replay(guest, call).await?
604        } else {
605            let targeted = syscalls::RtTgsigqueueinfo::new()
606                .with_tgid(tgid)
607                .with_tid(tid.as_raw())
608                .with_sig(call.sig())
609                .with_siginfo(call.siginfo());
610            self.record_or_replay(guest, targeted).await?
611        };
612        self.notify_cross_task_signal(guest, tid, call.sig(), Some(DetPid::from_raw(tgid)))
613            .await;
614        Ok(value)
615    }
616
617    // AUTONOMOUS-BOT-IMPLEMENTED
618    // TODO-HUMAN-REVIEW(#663)
619    /// Read the kernel pending-signal mask after Detcore has serialized all signal
620    /// generation and delivery events that can change it.
621    pub async fn handle_rt_sigpending<G: Guest<Self>>(
622        &self,
623        guest: &mut G,
624        call: syscalls::RtSigpending,
625    ) -> Result<i64, Error> {
626        Ok(self.record_or_replay(guest, call).await?)
627    }
628}

630fn should_notify_cross_task_signal(sender: DetTid, target: DetTid, raw_signal: i32) -> bool {
631    raw_signal != 0 && target != sender
632}
