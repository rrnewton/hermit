12//! External network record and replay for outbound TCP clients.
13//!
14//! With a network trace mode configured, an outbound TCP `connect` becomes a
15//! channel of the [`NetworkEngine`] held in global state, and every later
16//! operation on that socket is answered by the engine:
17//!
18//! - Record mode connects for real, pulls whatever the host has received with
19//!   a nonblocking `recv` at each guest check, and hands it to the engine,
20//!   which stamps it with the global time of that check.
21//! - Replay mode never touches the host network. The engine releases each
22//!   recorded input once global time and the guest's own transmissions reach
23//!   the recorded point, and compares every outbound byte with the recording.
24//!
25//! A pull and the check it serves travel in one request, so the engine
26//! observes both at the same global time. Blocking operations retry on
27//! ordinary polling turns, which advance global time, so the release point of
28//! every input is reached in replay under any schedule.
29//!
30//! Host waits that the guest cannot observe as time passing -- the connect
31//! handshake and a full send buffer -- block the thread without yielding, so
32//! record and replay take the same scheduler turns. They are wall-clock
33//! waits of up to a minute each, unlike the virtual-time stall limit replay
34//! applies.
35//!
36//! Anything the engine cannot answer ends the run with the policy-refusal
37//! status, naming the reason and the remedy; replay never falls back to the
38//! host. That includes every operation that could reach the network outside a
39//! channel, because a recording shares the host's network namespace:
40//!
41//! - on an IPv4 or IPv6 socket, a UDP send, `bind`, `listen`, a TCP Fast Open
42//!   send, `epoll` registration, signal-driven I/O, and a nonzero send or
43//!   receive timeout, which Linux would apply in wall-clock time;
44//! - a socket of any family but `AF_UNIX`, `AF_INET` and `AF_INET6`, such as
45//!   netlink or packet sockets, and a raw IPv4 or IPv6 socket;
46//! - an abstract `AF_UNIX` address, which names the network namespace rather
47//!   than the file system;
48//! - receiving a socket of another family than `AF_UNIX` through
49//!   `SCM_RIGHTS`, which this module could not classify;
50//! - the interface and route ioctls (`SIOCGIFINDEX`, `SIOCGIFCONF`, ...) on a
51//!   socket of any family;
52//! - opening `/proc/net`, `/proc/sys/net`, their per-process spellings, or a
53//!   network interface under `/sys`.
54//!
55//! Creating an IPv4 or IPv6 socket and closing it stays allowed, because
56//! resolvers probe for IPv6 support that way. The checks are the same in both
57//! modes.
58//!
59//! A recording still runs in the host's network namespace, and some routes to
60//! that namespace's state remain open: `stat` and `readlink` on the paths
61//! above, an implicit abstract `AF_UNIX` autobind under `SO_PASSCRED`, and any
62//! host state a program reaches through a file this module does not know.
63//! A program that takes one of them can record a run that replays
64//! differently or refuses.

139/// The poll events Linux's `select` reports as readable, writable and
140/// exceptional (`POLLIN_SET`, `POLLOUT_SET` and `POLLEX_SET` in
141/// `fs/select.c`). Newer kernels include `POLLNVAL` in all three, so a
142/// descriptor closed during the wait reports ready in every set it is in;
143/// older ones ignore such a descriptor (no bit, no error).
144const SELECT_REPORTED: [i16; 3] = [
145    libc::POLLRDNORM
146        | libc::POLLRDBAND
147        | libc::POLLIN
148        | libc::POLLHUP
149        | libc::POLLERR
150        | libc::POLLNVAL,
151    libc::POLLWRBAND | libc::POLLWRNORM | libc::POLLOUT | libc::POLLERR | libc::POLLNVAL,
152    libc::POLLPRI | libc::POLLNVAL,
153];

155/// The poll events a descriptor in each `select` set asks for. `POLLERR` and
156/// `POLLHUP` need no request.
157const SELECT_REQUESTED: [i16; 3] = [
158    libc::POLLRDNORM | libc::POLLRDBAND | libc::POLLIN,
159    libc::POLLWRBAND | libc::POLLWRNORM | libc::POLLOUT,
160    libc::POLLPRI,
161];

168/// For each of `select`'s three sets, whether `pollfd`, built from those sets,
169/// reports ready in it.
170fn select_bits(pollfd: &libc::pollfd) -> [bool; 3] {
171    std::array::from_fn(|set| {
172        pollfd.events & SELECT_REQUESTED[set] != 0 && pollfd.revents & SELECT_REPORTED[set] != 0
173    })
174}

177fn select_ready(pollfds: &[libc::pollfd]) -> bool {

183/// The pollfd entries for `select`'s descriptor sets: one per descriptor below
184/// `nfds` named in any set, asking for that set's events.
185fn select_pollfds(nfds: usize, sets: &[Option<Vec<u8>>; 3]) -> Vec<libc::pollfd> {
186    (0..nfds)
187        .filter_map(|fd| {
188            let events = sets
189                .iter()
190                .zip(SELECT_REQUESTED)
191                .filter(|(set, _)| {
192                    set.as_ref()
193                        .is_some_and(|set| set[fd / 8] & (1 << (fd % 8)) != 0)
194                })
195                .fold(0, |events, (_, requested)| events | requested);
196            (events != 0).then_some(libc::pollfd {
197                fd: fd as RawFd,
198                events,
199                revents: 0,
200            })
201        })
202        .collect()
203}

205/// `select`'s result sets for `pollfds`, with `len` bytes for each set the
206/// call passed, and the number of bits set across them.
207fn select_result(
208    pollfds: &[libc::pollfd],
209    sets: &[Option<Vec<u8>>; 3],
210    len: usize,
211) -> ([Option<Vec<u8>>; 3], i64) {
212    let mut result = sets.clone().map(|set| set.map(|_| vec![0u8; len]));
213    let mut count = 0;
214    for pollfd in pollfds {
215        let fd = pollfd.fd as usize;
216        for (set, ready) in result.iter_mut().zip(select_bits(pollfd)) {
217            if let (Some(set), true) = (set, ready) {
218                set[fd / 8] |= 1 << (fd % 8);
219                count += 1;
220            }
221        }
222    }
223    (result, count)
224}

327/// Whether `call`, naming an IPv4 or IPv6 socket of `kind` that is not a
328/// channel, could reach the network unrecorded and must be refused.
329fn reaches_network_outside_channel(call: &Syscall, kind: NetworkSocketKind) -> bool {
330    let fast_open = |flags: i32| flags & libc::MSG_FASTOPEN != 0;
331    match kind {
332        NetworkSocketKind::NotInet => false,
333        NetworkSocketKind::InetOther => matches!(
334            call,
335            Syscall::Bind(_)
336                | Syscall::Listen(_)
337                | Syscall::Sendto(_)
338                | Syscall::Sendmsg(_)
339                | Syscall::Sendmmsg(_)
340        ),
341        NetworkSocketKind::InetStream { .. } => match call {
342            Syscall::Bind(_) | Syscall::Listen(_) => true,
343            Syscall::Sendto(c) => fast_open(c.flags() as i32),
344            Syscall::Sendmsg(c) => fast_open(c.flags()),
345            Syscall::Sendmmsg(c) => fast_open(c.flags()),
346            Syscall::Setsockopt(c) => {
347                c.level() == libc::IPPROTO_TCP && c.optname() == libc::TCP_FASTOPEN_CONNECT
348            }
349            _ => false,
350        },
351    }
352}

390/// Whether `path`, an absolute path the guest opened, names the host's
391/// network namespace state: `/proc/net` and its per-process and per-thread
392/// spellings, `/proc/sys/net`, or a network interface under `/sys`. A
393/// recording reads the host's interfaces, connections and settings there, a
394/// replay its own.
395fn names_host_network_state(path: &Path) -> bool {
396    let mut components: Vec<&OsStr> = Vec::new();
397    for component in path.components() {
398        match component {
399            Component::Normal(part) => components.push(part),
400            Component::ParentDir => {
401                components.pop();
402            }
403            _ => {}
404        }
405    }
406    let is_task = |name: &OsStr| {
407        name == "self"
408            || name == "thread-self"
409            || name.to_str().is_some_and(|name| {
410                !name.is_empty() && name.bytes().all(|byte| byte.is_ascii_digit())
411            })
412    };
413    match components.as_slice() {
414        [proc, net, ..] if *proc == "proc" && *net == "net" => true,
415        // The per-namespace settings, such as `net/core/somaxconn`.
416        [proc, sys, net, ..] if *proc == "proc" && *sys == "sys" && *net == "net" => true,
417        [proc, task, net, ..] if *proc == "proc" && is_task(task) && *net == "net" => true,
418        [proc, process, tasks, thread, net, ..]
419            if *proc == "proc"
420                && is_task(process)
421                && *tasks == "task"
422                && is_task(thread)
423                && *net == "net" =>
424        {
425            true
426        }
427        // `/sys/class/net/<interface>` and the device directories it links
428        // to, `/sys/devices/.../net/<interface>`.
429        [sys, rest @ ..] if *sys == "sys" => rest.iter().any(|part| *part == "net"),
430        _ => false,
431    }
432}

434/// The traced address of an IPv4 or IPv6 `sockaddr`, or `None` for any other
435/// family or a buffer shorter than Linux accepts. An IPv6 address without
436/// `sin6_scope_id` has scope zero, as in Linux.
437fn address_from_sockaddr(bytes: &[u8]) -> Option<NetworkAddressV1> {
438    let family = u16::from_ne_bytes(bytes.get(..2)?.try_into().ok()?);
439    if family == libc::AF_INET as u16 && bytes.len() >= size_of::<libc::sockaddr_in>() {
440        Some(NetworkAddressV1::Inet4 {
441            port: u16::from_be_bytes(bytes[2..4].try_into().ok()?),
442            address: bytes[4..8].try_into().ok()?,
443        })
444    } else if family == libc::AF_INET6 as u16 && bytes.len() >= SOCKADDR_IN6_RFC2133_LEN {
445        let scope_id = match bytes.get(24..28) {
446            Some(scope_id) => u32::from_ne_bytes(scope_id.try_into().ok()?),
447            None => 0,
448        };
449        Some(NetworkAddressV1::Inet6 {
450            port: u16::from_be_bytes(bytes[2..4].try_into().ok()?),
451            flowinfo: u32::from_be_bytes(bytes[4..8].try_into().ok()?),
452            address: bytes[8..24].try_into().ok()?,
453            scope_id,
454        })
455    } else {
456        None
457    }
458}

460/// Copy in a `sockaddr` as Linux's `move_addr_to_kernel` does: a length
461/// beyond `sockaddr_storage` or below zero fails with `EINVAL`, and a zero
462/// length copies nothing.
463fn read_sockaddr<M: MemoryAccess>(
464    memory: &M,
465    address: Option<AddrMut<'_, libc::sockaddr>>,
466    addrlen: i32,
467) -> Result<Vec<u8>, Errno> {
468    let length = usize::try_from(addrlen)
469        .ok()
470        .filter(|length| *length <= SOCKADDR_STORAGE_LEN)
471        .ok_or(Errno::EINVAL)?;
472    let mut bytes = vec![0; length];
473    if length > 0 {
474        let address = address.ok_or(Errno::EFAULT)?;
475        memory.read_exact(Addr::from(address.cast::<u8>()), &mut bytes)?;
476    }
477    Ok(bytes)
478}

505/// The descriptors carried by the `SCM_RIGHTS` messages in a sent or
506/// received control buffer.
507pub(crate) fn received_descriptors(control: &[u8]) -> Vec<RawFd> {
508    const HEADER: usize = size_of::<libc::cmsghdr>();
509    let align = |length: usize| length.next_multiple_of(size_of::<usize>());
510    let mut descriptors = Vec::new();
511    let mut offset = 0;
512    while let Some(header) = control.get(offset..offset + HEADER) {
513        let length = usize::from_ne_bytes(header[..8].try_into().unwrap());
514        let level = i32::from_ne_bytes(header[8..12].try_into().unwrap());
515        let kind = i32::from_ne_bytes(header[12..16].try_into().unwrap());
516        let Some(data) = control.get(offset + HEADER..offset.saturating_add(length)) else {
517            break;
518        };
519        if level == libc::SOL_SOCKET && kind == libc::SCM_RIGHTS {
520            descriptors.extend(
521                data.as_chunks::<{ size_of::<RawFd>() }>()
522                    .0
523                    .iter()
524                    .map(|fd| RawFd::from_ne_bytes(*fd)),
525            );
526        }
527        if length < HEADER {
528            break;
529        }
530        offset += align(length);
531    }
532    descriptors
533}

535/// The `sockaddr` bytes Linux reports for `address`.
536fn sockaddr_bytes(address: &NetworkAddressV1) -> Vec<u8> {
537    match address {
538        NetworkAddressV1::Inet4 { address, port } => {
539            let mut bytes = vec![0; size_of::<libc::sockaddr_in>()];
540            bytes[..2].copy_from_slice(&(libc::AF_INET as u16).to_ne_bytes());
541            bytes[2..4].copy_from_slice(&port.to_be_bytes());
542            bytes[4..8].copy_from_slice(address);
543            bytes
544        }
545        NetworkAddressV1::Inet6 {
546            address,
547            port,
548            flowinfo,
549            scope_id,
550        } => {
551            let mut bytes = vec![0; size_of::<libc::sockaddr_in6>()];
552            bytes[..2].copy_from_slice(&(libc::AF_INET6 as u16).to_ne_bytes());
553            bytes[2..4].copy_from_slice(&port.to_be_bytes());
554            bytes[4..8].copy_from_slice(&flowinfo.to_be_bytes());
555            bytes[8..24].copy_from_slice(address);
556            bytes[24..28].copy_from_slice(&scope_id.to_ne_bytes());
557            bytes

628    /// Whether the network trace handles `call`. Always false with no trace
629    /// mode, so ordinary runs never reach this module.
630    pub(crate) fn network_trace_owns<G: Guest<Self>>(&self, guest: &mut G, call: &Syscall) -> bool {
631        if self.network_mode() == NetworkTraceMode::Off {
632            return false;
633        }
634        match call {
635            // A connect may create a channel and a poll may name one; each
636            // falls back to its ordinary handler otherwise.
637            Syscall::Connect(_) | Syscall::Poll(_) | Syscall::Ppoll(_) => true,
638            // A received descriptor may be a socket this module never saw.
639            Syscall::Recvmsg(_) | Syscall::Recvmmsg(_) => true,
640            // Only AF_UNIX sockets stay out of the host's network namespace.
641            Syscall::Socket(c) => c.family() != libc::AF_UNIX,
642            Syscall::Select(c) => Self::select_names_channel(
643                guest,
644                c.nfds(),
645                [c.readfds(), c.writefds(), c.exceptfds()],
646            ),
647            Syscall::Pselect6(c) => Self::select_names_channel(
648                guest,
649                c.nfds(),
650                [c.readfds(), c.writefds(), c.exceptfds()],
651            ),
652            // The receive low-water mark may be set before the socket connects.
653            Syscall::Setsockopt(c)
654                if c.level() == libc::SOL_SOCKET && c.optname() == libc::SO_RCVLOWAT =>
655            {
656                true
657            }
658            // A registration, timeout or signal owner outlives the connect, so
659            // refuse it on any IPv4 or IPv6 socket rather than only on a
660            // channel.
661            Syscall::EpollCtl(c) => {
662                Self::network_socket_state(guest, c.fd()).1 != NetworkSocketKind::NotInet
663            }
664            Syscall::Setsockopt(c)
665                if c.level() == libc::SOL_SOCKET && TIMEOUT_OPTIONS.contains(&c.optname()) =>
666            {
667                Self::network_socket_state(guest, c.fd()).1 != NetworkSocketKind::NotInet
668            }
669            // Linux answers these from the network namespace on any socket.
670            Syscall::Ioctl(c) if requests_interface_state(c) => true,
671            Syscall::Fcntl(_) | Syscall::Ioctl(_) if requests_signal_driven_io(call) => {
672                get_fd(*call).is_some_and(|fd| {
673                    Self::network_socket_state(guest, fd).1 != NetworkSocketKind::NotInet
674                })
675            }
676            Syscall::Sendfile(c) => Self::names_channel(guest, [c.out_fd(), c.in_fd()]),
677            Syscall::Splice(c) => Self::names_channel(guest, [c.fd_in(), c.fd_out()]),
678            Syscall::Tee(c) => Self::names_channel(guest, [c.fd_in(), c.fd_out()]),
679            Syscall::CopyFileRange(c) => Self::names_channel(guest, [c.fd_in(), c.fd_out()]),
680            _ => {
681                let Some(fd) = get_fd(*call) else {
682                    return false;
683                };
684                match Self::network_socket_state(guest, fd) {
685                    (true, _) => !channel_call_takes_ordinary_path(call),
686                    (false, NetworkSocketKind::NotInet) => {
687                        Self::names_abstract_unix_destination(guest, call)
688                    }
689                    (false, kind) => reaches_network_outside_channel(call, kind),
690                }
691            }
692        }
693    }

695    /// Whether `call`, a `bind` or send, names an abstract `AF_UNIX` address.
696    /// An address the guest cannot read fails in Linux before any effect.
697    fn names_abstract_unix_destination<G: Guest<Self>>(guest: &mut G, call: &Syscall) -> bool {
698        let names = |guest: &mut G, address: Option<Addr<'_, u8>>, length: usize, binding| {
699            let mut bytes = [0u8; 3];
700            let prefix = &mut bytes[..length.min(3)];
701            address.is_some_and(|address| guest.memory().read_exact(address, prefix).is_ok())
702                && names_abstract_unix_address(prefix, length, binding)
703        };
704        let message_names = |guest: &mut G, header: Option<Addr<'_, libc::msghdr>>| {
705            let Some(header) = header.and_then(|header| guest.memory().read_value(header).ok())
706            else {
707                return false;
708            };
709            let header: libc::msghdr = header;
710            names(
711                guest,
712                Addr::from_raw(header.msg_name as usize),
713                header.msg_namelen as usize,
714                false,
715            )
716        };
717        let length = |addrlen: i32| usize::try_from(addrlen).unwrap_or(0);
718        match call {
719            Syscall::Bind(c) => names(
720                guest,
721                c.umyaddr().map(|a| Addr::from(a.cast())),
722                length(c.addrlen()),
723                true,
724            ),
725            Syscall::Sendto(c) => names(
726                guest,
727                c.addr().map(|a| Addr::from(a.cast())),
728                length(c.addr_len()),
729                false,
730            ),
731            Syscall::Sendmsg(c) => message_names(guest, c.msg()),
732            Syscall::Sendmmsg(c) => {
733                let Some(vector) = c.msgvec() else {
734                    return false;
735                };
736                let vector = vector.cast::<libc::mmsghdr>();
737                (0..c.vlen().min(libc::UIO_MAXIOV as u32) as usize).any(|index| {
738                    // SAFETY: only read through guest memory access, which
739                    // fails on an unmapped address.
740                    let entry = unsafe { vector.add(index) };
741                    message_names(guest, Some(entry.cast()))
742                })
743            }
744            _ => false,
745        }
746    }

748    /// Whether a `select` descriptor set below `nfds` names a channel. Linux
749    /// reads `nfds` bits, not a fixed `fd_set`, so a channel at or above
750    /// `FD_SETSIZE` counts too. Only the word holding each channel's bit is
751    /// read, which bounds the reads by the open channels rather than `nfds`.
752    /// An unreadable word leaves the call to its ordinary path, where Linux
753    /// fails it with `EFAULT`.
754    fn select_names_channel<G: Guest<Self>>(
755        guest: &mut G,
756        nfds: i32,
757        sets: [Option<AddrMut<'_, libc::fd_set>>; 3],
758    ) -> bool {
759        const WORD_BITS: usize = u64::BITS as usize;
760        let channels = guest.thread_state().network_channel_fds();
761        channels
762            .into_iter()
763            .filter(|&fd| fd >= 0 && fd < nfds)
764            .any(|fd| {
765                let fd = fd as usize;
766                sets.into_iter().flatten().any(|set| {
767                    // SAFETY: only read through guest memory access, which
768                    // fails on an unmapped address.
769                    let word = unsafe { set.cast::<u64>().add(fd / WORD_BITS) };
770                    guest
771                        .memory()
772                        .read_value(word)
773                        .is_ok_and(|word| word & (1 << (fd % WORD_BITS)) != 0)
774                })
775            })
776    }

778    /// Handle a call that [`Self::network_trace_owns`] accepted.
779    pub(crate) async fn handle_network_trace_syscall<G: Guest<Self>>(
780        &self,
781        guest: &mut G,
782        call: Syscall,
783    ) -> Result<i64, Error> {
784        match call {
785            Syscall::Socket(c) => return self.network_socket(guest, c).await,
786            Syscall::Connect(c) => return self.network_connect(guest, c).await,
787            Syscall::Poll(c) => return self.network_poll(guest, c).await,
788            Syscall::Ppoll(c) => return self.network_ppoll(guest, c).await,
789            Syscall::Setsockopt(c) => return self.network_setsockopt(guest, c).await,
790            Syscall::Recvmsg(c) if !Self::is_network_channel(guest, c.sockfd()) => {
791                return self.network_recv_outside_channel(guest, call).await;
792            }
793            Syscall::Recvmmsg(c) if !Self::is_network_channel(guest, c.fd()) => {
794                return self.network_recv_outside_channel(guest, call).await;
795            }
796            Syscall::Ioctl(c) if requests_interface_state(&c) => {
797                self.network_refuse(
798                    guest,
799                    &format!(
800                        "network record/replay does not model the interface and route ioctl \
801                         {:#x}, which Linux answers from the host's network namespace on a \
802                         recording and from a private one on a replay",
803                        c.request().into_raw().0
804                    ),
805                    UNSUPPORTED_REMEDY,
806                )
807                .await
808            }
809            Syscall::Fcntl(_) | Syscall::Ioctl(_) if requests_signal_driven_io(&call) => {
810                self.network_refuse(
811                    guest,
812                    "network record/replay does not model signal-driven I/O (O_ASYNC, F_SETOWN, \
813                     F_SETSIG) on an IPv4 or IPv6 socket",
814                    UNSUPPORTED_REMEDY,
815                )
816                .await
817            }
818            Syscall::EpollCtl(_) => {
819                self.network_refuse(
820                    guest,
821                    "network record/replay does not model epoll registration of an IPv4 or \
822                     IPv6 socket",
823                    UNSUPPORTED_REMEDY,
824                )
825                .await
826            }
827            Syscall::Select(c) => return self.network_select(guest, c).await,
828            Syscall::Pselect6(c) => return self.network_pselect6(guest, c).await,
829            Syscall::Sendfile(_)
830            | Syscall::Splice(_)
831            | Syscall::Tee(_)
832            | Syscall::CopyFileRange(_) => self.network_refuse_on_channel(guest, &call).await,
833            _ => {}
834        }
835        // Every other call the trace owns names one socket.
836        let fd = get_fd(call).expect("an owned call names a descriptor");
837        let (channel, kind) = Self::network_socket_state(guest, fd);
838        if !channel && kind == NetworkSocketKind::NotInet {
839            self.network_refuse(
840                guest,
841                &format!(
842                    "network record/replay does not model {} to an abstract AF_UNIX address, \
843                     which names the host's network namespace rather than a file",
844                    call.name()
845                ),
846                NAMESPACE_REMEDY,
847            )
848            .await
849        }
850        if !channel {
851            self.network_refuse(
852                guest,
853                &format!(
854                    "network record/replay does not model {} on an IPv4 or IPv6 socket that is \
855                     not a connected TCP client",
856                    call.name()
857                ),
858                UNSUPPORTED_REMEDY,
859            )
860            .await
861        }
862        match call {
863            Syscall::Read(c) => {
864                self.network_recv(
865                    guest,
866                    c.fd(),
867                    c.buf(),
868                    c.len(),
869                    0,
870                    c.signal_interrupt_errno(),
871                )
872                .await
873            }
874            Syscall::Recvfrom(c) => {
875                let received = self
876                    .network_recv(
877                        guest,
878                        c.fd(),
879                        c.buf(),
880                        c.len(),
881                        c.flags(),
882                        c.signal_interrupt_errno(),
883                    )
884                    .await?;
885                // As Linux does, copy out a source address only after a
886                // successful receive that asked for one. A connected TCP
887                // socket reports none: its length is zero.
888                if c.addr().is_some() {
889                    let length = c.addr_len().ok_or(Errno::EFAULT)?;
890                    let requested: i32 = guest
891                        .memory()
892                        .read_value(length.cast())
893                        .map_err(|_| Errno::EFAULT)?;
894                    if requested < 0 {
895                        return Err(Errno::EINVAL.into());
896                    }
897                    guest
898                        .memory()
899                        .write_value(length, &0u32)
900                        .map_err(|_| Errno::EFAULT)?;
901                }
902                Ok(received)
903            }
904            Syscall::Write(c) => self.network_send(guest, c.fd(), c.buf(), c.len(), 0).await,
905            Syscall::Sendto(c) => {
906                // A connected TCP socket ignores the destination address.
907                let flags = c.flags() as i32;
908                self.network_send(
909                    guest,
910                    c.fd(),
911                    c.buf().map(|buf| Addr::from(buf.cast::<u8>())),
912                    c.size(),
913                    flags,
914                )
915                .await
916            }
917            Syscall::Shutdown(c) => self.network_shutdown(guest, c).await,
918            Syscall::Getsockname(c) => {
919                self.network_address(guest, c.fd(), c.usockaddr(), c.usockaddr_len(), true)
920                    .await
921            }
922            Syscall::Getpeername(c) => {
923                self.network_address(guest, c.fd(), c.usockaddr(), c.usockaddr_len(), false)
924                    .await
925            }
926            Syscall::Getsockopt(c)
927                if c.level() == libc::SOL_SOCKET && c.optname() == libc::SO_ERROR =>
928            {
929                self.network_so_error(guest, c).await
930            }
931            // Configuration options are the same in record and replay; the
932            // host socket answers them in both modes.
933            Syscall::Getsockopt(c) if is_configuration_option(c.level(), c.optname()) => {
934                self.handle_getsockopt(guest, c).await
935            }
936            Syscall::Getsockopt(c) => {
937                self.network_refuse(
938                    guest,
939                    &format!(
940                        "network trace does not model getsockopt level {} option {} on a \
941                         recorded socket",
942                        c.level(),
943                        c.optname()
944                    ),
945                    UNSUPPORTED_REMEDY,
946                )
947                .await
948            }
949            other => self.network_refuse_on_channel(guest, &other).await,
950        }
951    }

1010    /// `socket`: refuse every family but IPv4 and IPv6, whose sockets reach
1011    /// the host's network namespace, and raw sockets, and classify IPv4 and
1012    /// IPv6 sockets for later calls.
1013    async fn network_socket<G: Guest<Self>>(
1014        &self,
1015        guest: &mut G,
1016        call: syscalls::Socket,
1017    ) -> Result<i64, Error> {
1018        let family = call.family();
1019        if family != libc::AF_INET && family != libc::AF_INET6 {
1020            let sockets = match family {
1021                libc::AF_NETLINK => "netlink sockets".to_owned(),
1022                libc::AF_PACKET => "packet sockets".to_owned(),
1023                other => format!("sockets of address family {other}"),
1024            };
1025            self.network_refuse(
1026                guest,
1027                &format!(
1028                    "network record/replay does not model {sockets}, which reach the host's \
1029                     network namespace"
1030                ),
1031                NAMESPACE_REMEDY,
1032            )
1033            .await
1034        }
1035        let socket_type = call.r#type() & 0xf;
1036        if socket_type == libc::SOCK_RAW {
1037            self.network_refuse(
1038                guest,
1039                "network record/replay does not model raw IPv4 or IPv6 sockets",
1040                UNSUPPORTED_REMEDY,
1041            )
1042            .await
1043        }
1044        let fd = self.handle_socket(guest, call).await?;
1045        let stream =
1046            socket_type == libc::SOCK_STREAM && matches!(call.protocol(), 0 | libc::IPPROTO_TCP);
1047        let kind = if stream {
1048            NetworkSocketKind::InetStream {
1049                ipv6: family == libc::AF_INET6,
1050            }
1051        } else {
1052            NetworkSocketKind::InetOther
1053        };
1054        guest
1055            .thread_state()
1056            .with_detfd(fd as RawFd, |detfd| detfd.set_network_socket(kind))?;
1057        Ok(fd)
1058    }

1060    /// `recvmsg` or `recvmmsg` outside a channel: take the ordinary path, then
1061    /// refuse if it delivered through `SCM_RIGHTS` a socket of another family
1062    /// than `AF_UNIX`, which this module never classified and so could not
1063    /// keep from the network. A descriptor that is no socket stays untraced.
1064    async fn network_recv_outside_channel<G: Guest<Self>>(
1065        &self,
1066        guest: &mut G,
1067        call: Syscall,
1068    ) -> Result<i64, Error> {
1069        // Raw addresses: a typed address is not `Send` across the receive.
1070        let headers: Vec<usize> = match &call {
1071            Syscall::Recvmsg(c) => c.msg().map(AddrMut::as_raw).into_iter().collect(),
1072            Syscall::Recvmmsg(c) => c.mmsg().map_or_else(Vec::new, |vector| {
1073                (0..c.vlen().min(libc::UIO_MAXIOV as u32) as usize)
1074                    // SAFETY: only read through guest memory access, which
1075                    // fails on an unmapped address.
1076                    .map(|index| unsafe { vector.add(index) }.as_raw())
1077                    .collect()
1078            }),
1079            _ => unreachable!("only receives with ancillary data reach here"),
1080        };
1081        // The control buffers the guest offered, before Linux overwrites
1082        // their lengths with what it delivered.
1083        let offered: Vec<Option<(usize, usize)>> = headers
1084            .iter()
1085            .map(|header| {
1086                let header = Addr::<libc::msghdr>::from_raw(*header)?;
1087                let header: libc::msghdr = guest.memory().read_value(header).ok()?;
1088                Some((header.msg_control as usize, header.msg_controllen))
1089            })
1090            .collect();
1091        let (result, received) = match call {
1092            Syscall::Recvmsg(c) => (self.handle_recvmsg(guest, c).await?, 1),
1093            // Ordinary dispatch first gives a sock_diag reply socket its
1094            // sanitized answer. That socket is netlink, which `socket` refuses
1095            // under a trace mode, so none exists here.
1096            Syscall::Recvmmsg(c) => {
1097                let result = self.handle_recvmmsg(guest, c).await?;
1098                (result, result as usize)
1099            }
1100            _ => unreachable!("only receives with ancillary data reach here"),
1101        };
1102        for (header, offered) in headers.into_iter().zip(offered).take(received) {
1103            let Some((control, capacity)) = offered else {
1104                continue;
1105            };
1106            let delivered = Addr::<libc::msghdr>::from_raw(header)
1107                .ok_or(Errno::EFAULT)
1108                .and_then(|header| guest.memory().read_value(header))
1109                .map(|header: libc::msghdr| {
1110                    header.msg_controllen.min(capacity).min(MAX_CONTROL_BYTES)
1111                });
1112            let mut bytes = vec![0; *delivered.as_ref().unwrap_or(&0)];
1113            let readable = delivered.is_ok()
1114                && (bytes.is_empty()
1115                    || Addr::<u8>::from_raw(control).is_some_and(|control| {
1116                        guest.memory().read_exact(control, &mut bytes).is_ok()
1117                    }));
1118            if !readable {
1119                self.network_refuse(
1120                    guest,
1121                    "network record/replay could not read the ancillary data Linux delivered, \
1122                     so cannot check it for sockets",
1123                    "The program changed its receive buffers during the call; fix the program.",
1124                )
1125                .await
1126            }
1127            for fd in received_descriptors(&bytes) {
1128                match self
1129                    .network_host_int_option(guest, fd, libc::SO_DOMAIN)
1130                    .await
1131                {
1132                    Ok(libc::AF_UNIX) | Err(Errno::ENOTSOCK) => {}
1133                    Ok(domain) => {
1134                        self.network_refuse(
1135                            guest,
1136                            &format!(
1137                                "network record/replay does not model a socket of address \
1138                                 family {domain} received through SCM_RIGHTS"
1139                            ),
1140                            UNSUPPORTED_REMEDY,
1141                        )
1142                        .await
1143                    }
1144                    Err(errno) => {
1145                        self.network_refuse(
1146                            guest,
1147                            &format!(
1148                                "network record/replay could not classify descriptor {fd} \
1149                                 received through SCM_RIGHTS: {errno}"
1150                            ),
1151                            UNSUPPORTED_REMEDY,
1152                        )
1153                        .await
1154                    }

1187    /// Wait in the host, without yielding, for `events` on `fd`. Returns the
1188    /// reported events, or `None` on timeout.
1189    async fn network_host_wait<G: Guest<Self>>(
1190        &self,
1191        guest: &mut G,
1192        fd: RawFd,
1193        events: i16,
1194        timeout_millis: i32,
1195    ) -> Result<Option<i16>, Error> {
1196        let mut stack = guest.stack().await;
1197        let pollfd: AddrMut<libc::pollfd> = stack.reserve();
1198        let _guard = stack.commit()?;
1199        guest.memory().write_value(
1200            pollfd,
1201            &libc::pollfd {
1202                fd,
1203                events,
1204                revents: 0,
1205            },
1206        )?;
1207        loop {
1208            let call = syscalls::Poll::new()
1209                .with_fds(Some(pollfd.cast()))
1210                .with_nfds(1)
1211                .with_timeout(timeout_millis);
1212            match guest.inject(call).await {
1213                Ok(0) => return Ok(None),
1214                Ok(_) => {
1215                    let pollfd: libc::pollfd = guest.memory().read_value(pollfd)?;
1216                    return Ok(Some(pollfd.revents));
1217                }
1218                Err(Errno::EINTR) => continue,
1219                Err(errno) => return Err(errno.into()),
1220            }
1221        }
1222    }

1306    /// `connect`: an outbound TCP connect to an IPv4 or IPv6 peer becomes a
1307    /// channel. Linux's argument checks run here, in the same order, so record
1308    /// and replay fail them alike before the host is reached. A connect on any
1309    /// other socket takes the ordinary path, unless it names an abstract
1310    /// `AF_UNIX` address.
1311    async fn network_connect<G: Guest<Self>>(
1312        &self,
1313        guest: &mut G,
1314        call: syscalls::Connect,
1315    ) -> Result<i64, Error> {
1316        let fd = call.fd();
1317        let socket = guest
1318            .thread_state()
1319            .with_detfd(fd, |detfd| {
1320                (detfd.ty() == FdType::Socket && detfd.open_file_id().is_socket()).then(|| {
1321                    (
1322                        detfd.open_file_id(),
1323                        detfd.is_nonblocking(),
1324                        detfd.is_network_channel(),
1325                        detfd.network_socket(),
1326                    )
1327                })
1328            })
1329            .ok()
1330            .flatten();
1331        let kind = socket.map_or(NetworkSocketKind::NotInet, |socket| socket.3);
1332        let ipv6 = match kind {
1333            NetworkSocketKind::NotInet => {
1334                if let Ok(bytes) = read_sockaddr(&guest.memory(), call.uservaddr(), call.addrlen())
1335                    && names_abstract_unix_address(&bytes, bytes.len(), false)
1336                {
1337                    self.network_refuse(
1338                        guest,
1339                        "network record/replay does not model connect to an abstract AF_UNIX \
1340                         address, which names the host's network namespace rather than a file",
1341                        NAMESPACE_REMEDY,
1342                    )
1343                    .await
1344                }
1345                return self.handle_connect(guest, call).await;
1346            }
1347            NetworkSocketKind::InetOther => {
1348                self.network_refuse(
1349                    guest,
1350                    "network record/replay does not model connect on an IPv4 or IPv6 socket \
1351                     other than TCP",
1352                    UNSUPPORTED_REMEDY,
1353                )
1354                .await
1355            }
1356            NetworkSocketKind::InetStream { ipv6 } => ipv6,
1357        };
1358        let (id, nonblocking, already_channel, _) =
1359            socket.expect("an IPv4 or IPv6 socket is tracked");
1360        let bytes = read_sockaddr(&guest.memory(), call.uservaddr(), call.addrlen())?;
1361        let family = match bytes.get(..2) {
1362            Some(family) => u16::from_ne_bytes([family[0], family[1]]),
1363            None => return Err(Errno::EINVAL.into()),
1364        };
1365        if family == libc::AF_UNSPEC as u16 {
1366            self.network_refuse(
1367                guest,
1368                "network record/replay does not model disconnecting a socket with an AF_UNSPEC \
1369                 connect",
1370                UNSUPPORTED_REMEDY,
1371            )
1372            .await
1373        }
1374        if already_channel {
1375            // Linux reports a finished connect's error once, then EISCONN.
1376            let NetworkReply::Errno(errno) = self
1377                .network_request(guest, NetworkRequest::TakeError(id))
1378                .await
1379            else {
1380                unreachable!()
1381            };
1382            return Err(Errno::new(if errno == 0 { libc::EISCONN } else { errno }).into());
1383        }
1384        let (minimum, expected) = if ipv6 {
1385            (SOCKADDR_IN6_RFC2133_LEN, libc::AF_INET6)
1386        } else {
1387            (size_of::<libc::sockaddr_in>(), libc::AF_INET)
1388        };
1389        if bytes.len() < minimum {
1390            return Err(Errno::EINVAL.into());
1391        }
1392        if family != expected as u16 {
1393            return Err(Errno::EAFNOSUPPORT.into());
1394        }
1395        let peer = address_from_sockaddr(&bytes).ok_or(Errno::EINVAL)?;
1396
1397        let request = if self.network_mode() == NetworkTraceMode::Record {
1398            if !peer.is_traceable_peer() {
1399                self.network_refuse_error(guest, &NetworkEngineError::UntraceablePeer(peer))
1400                    .await
1401            }
1402            let errno = match guest.inject(call).await {
1403                Ok(_) => 0,
1404                // An interrupted connect continues in Linux; finish it, so the
1405                // recording never holds a signal the replay cannot reproduce.
1406                Err(Errno::EINPROGRESS | Errno::EINTR) => {
1407                    if self
1408                        .network_host_wait(guest, fd, libc::POLLOUT, CONNECT_WAIT_MILLIS)
1409                        .await?
1410                        .is_none()
1411                    {
1412                        self.network_refuse(
1413                            guest,
1414                            "network connect did not finish in time",
1415                            HOST_REMEDY,
1416                        )
1417                        .await
1418                    }
1419                    self.network_host_int_option(guest, fd, libc::SO_ERROR)
1420                        .await?
1421                }
1422                // The argument checks above already ran, so every other
1423                // error is the host network's answer, which replay repeats.
1424                Err(errno) => errno.into_raw(),
1425            };
1426            let local = if errno == 0 {
1427                match self.network_host_local_address(guest, fd).await? {
1428                    Some(local) => Some(local),
1429                    None => {
1430                        self.network_refuse(
1431                            guest,
1432                            "network connect has no local address",
1433                            HOST_REMEDY,
1434                        )
1435                        .await
1436                    }
1437                }
1438            } else {
1439                None
1440            };
1441            NetworkRequest::RecordConnect {
1442                id,
1443                peer,
1444                local,
1445                errno,
1446                nonblocking,
1447            }
1448        } else {
1449            NetworkRequest::ReplayConnect {
1450                id,
1451                peer,
1452                nonblocking,
1453            }
1454        };
1455        let NetworkReply::Errno(errno) = self.network_request(guest, request).await else {
1456            unreachable!()
1457        };
1458        guest
1459            .thread_state()
1460            .with_detfd(fd, |detfd| detfd.set_network_channel())?;
1461        if errno == 0 {
1462            Ok(0)
1463        } else {
1464            Err(Errno::new(errno).into())
1465        }
1466    }

1468    /// `setsockopt`: track `SO_RCVLOWAT`, which the engine needs for receive
1469    /// targets and readiness, and refuse a send or receive timeout, which
1470    /// Linux would apply in wall-clock time. The host socket applies every
1471    /// other option, and fails a malformed timeout as Linux does.
1472    async fn network_setsockopt<G: Guest<Self>>(
1473        &self,
1474        guest: &mut G,
1475        call: syscalls::Setsockopt,
1476    ) -> Result<i64, Error> {
1477        if call.level() == libc::IPPROTO_TCP && call.optname() == libc::TCP_FASTOPEN_CONNECT {
1478            self.network_refuse(
1479                guest,
1480                "network record/replay does not model TCP Fast Open",
1481                UNSUPPORTED_REMEDY,
1482            )
1483            .await
1484        }
1485        if call.level() == libc::SOL_SOCKET
1486            && TIMEOUT_OPTIONS.contains(&call.optname())
1487            && call.optlen() as usize >= size_of::<libc::timeval>()
1488            && let Some(value) = call.optval()
1489            && let Ok(timeout) = guest.memory().read_value::<_, libc::timeval>(value.cast())
1490            && sets_a_timeout(timeout)
1491        {
1492            self.network_refuse(
1493                guest,
1494                "network record/replay does not model socket send or receive timeouts \
1495                 (SO_RCVTIMEO, SO_SNDTIMEO)",
1496                UNSUPPORTED_REMEDY,
1497            )
1498            .await
1499        }
1500        let result = self.handle_setsockopt(guest, call).await?;
1501        if call.level() == libc::SOL_SOCKET
1502            && call.optname() == libc::SO_RCVLOWAT
1503            && let Some(value) = call.optval()
1504        {
1505            let value: i32 = guest.memory().read_value(value.cast())?;
1506            let _ = guest.thread_state().with_detfd(call.fd(), |detfd| {
1507                detfd.set_network_lowat(normalized_lowat(value))
1508            });
1509        }
1510        Ok(result)
1511    }

1513    /// `recv`, `recvfrom` and `read` on a channel.
1514    async fn network_recv<G: Guest<Self>>(
1515        &self,
1516        guest: &mut G,
1517        fd: RawFd,
1518        buffer: Option<AddrMut<'_, u8>>,
1519        len: usize,
1520        flags: i32,
1521        interrupted: Errno,
1522    ) -> Result<i64, Error> {
1523        if flags & !RECV_FLAGS != 0 {
1524            self.network_refuse(
1525                guest,
1526                &format!("network trace does not model recv flags {flags:#x}"),
1527                UNSUPPORTED_REMEDY,
1528            )
1529            .await
1530        }
1531        if len == 0 {
1532            return Ok(0);
1533        }
1534        let buffer = buffer.ok_or(Errno::EFAULT)?;
1535        let (id, logically_nonblocking, lowat) = guest.thread_state().with_detfd(fd, |detfd| {
1536            (
1537                detfd.open_file_id(),
1538                detfd.is_nonblocking(),
1539                detfd.network_lowat(),
1540            )
1541        })?;
1542        let nonblocking = logically_nonblocking || flags & libc::MSG_DONTWAIT != 0;
1543        let target = if nonblocking {
1544            1
1545        } else if flags & libc::MSG_WAITALL != 0 {
1546            len
1547        } else {
1548            lowat.min(len)
1549        };
1550        let record = self.network_mode() == NetworkTraceMode::Record;
1551        let mut rsrc = Resources::new(guest.thread_state().dettid);
1552        rsrc.insert(ResourceID::InternalIOPolling, Permission::W);
1553        rsrc.fyi("network recv");
1554        loop {
1555            if !nonblocking
1556                && matches!(
1557                    resource_request(guest, rsrc.clone()).await,
1558                    ResumeStatus::Signaled(_)
1559                )
1560            {
1561                return Err(interrupted.into());
1562            }
1563            let arrivals = if record {
1564                self.network_pull(guest, fd, buffer, len)
1565                    .await?
1566                    .into_iter()
1567                    .collect()
1568            } else {
1569                Vec::new()
1570            };
1571            let request = NetworkRequest::Recv {
1572                id,
1573                arrivals,
1574                max_len: len,
1575                target,
1576                peek: flags & libc::MSG_PEEK != 0,
1577            };
1578            let NetworkReply::Recv(outcome) = self.network_request(guest, request).await else {
1579                unreachable!()
1580            };
1581            match outcome {
1582                NetworkRecvOutcome::Data(bytes) => {
1583                    if guest.memory().write_exact(buffer, &bytes).is_err() {
1584                        self.network_refuse_bad_buffer(guest).await
1585                    }
1586                    return Ok(bytes.len() as i64);
1587                }
1588                NetworkRecvOutcome::Eof => return Ok(0),
1589                NetworkRecvOutcome::Error(errno) => return Err(Errno::new(errno).into()),
1590                NetworkRecvOutcome::WouldBlock if nonblocking => {
1591                    return Err(Errno::EAGAIN.into());
1592                }
1593                NetworkRecvOutcome::WouldBlock => {
1594                    rsrc.poll_attempt += 1;
1595                    record_retry_event(guest, syscalls::Recvfrom::new().with_fd(fd)).await;
1596                }
1597            }
1598        }
1599    }

1601    /// `send`, `sendto` and `write` on a channel.
1602    async fn network_send<G: Guest<Self>>(
1603        &self,
1604        guest: &mut G,
1605        fd: RawFd,
1606        buffer: Option<Addr<'_, u8>>,
1607        len: usize,
1608        flags: i32,
1609    ) -> Result<i64, Error> {
1610        if flags & !SEND_FLAGS != 0 {
1611            self.network_refuse(
1612                guest,
1613                &format!("network trace does not model send flags {flags:#x}"),
1614                UNSUPPORTED_REMEDY,
1615            )
1616            .await
1617        }
1618        let (id, logically_nonblocking) = guest
1619            .thread_state()
1620            .with_detfd(fd, |detfd| (detfd.open_file_id(), detfd.is_nonblocking()))?;
1621        let mut bytes = vec![0; len];
1622        if len != 0 {
1623            guest
1624                .memory()
1625                .read_exact(buffer.ok_or(Errno::EFAULT)?, &mut bytes)
1626                .map_err(|_| Errno::EFAULT)?;
1627        }
1628        let NetworkReply::Failure(failure) = self
1629            .network_request(guest, NetworkRequest::SendFailure(id))
1630            .await
1631        else {
1632            unreachable!()
1633        };
1634        if let Some(errno) = failure {
1635            if errno == libc::EPIPE && flags & libc::MSG_NOSIGNAL == 0 {
1636                self.network_refuse(
1637                    guest,
1638                    "network trace does not model SIGPIPE",
1639                    "Send with MSG_NOSIGNAL, or run without --record-networking.",
1640                )
1641                .await
1642            }
1643            return Err(Errno::new(errno).into());
1644        }
1645        if len == 0 {
1646            return Ok(0);
1647        }
1648        let nonblocking = logically_nonblocking || flags & libc::MSG_DONTWAIT != 0;
1649        let buffer = buffer.ok_or(Errno::EFAULT)?;
1650        let mut sent = 0;
1651        // A blocking send that cannot go on yet yields to the scheduler, so
1652        // that other guest threads, which may be what the peer waits for,
1653        // keep running. Record waits until the host accepts more; replay
1654        // waits until the next fragment is due, which is after as many waits
1655        // as the recording took and no earlier in global time than the
1656        // recording accepted it. Both wait the same way, so a replayed send
1657        // returns at the same point of the guest's execution as the recorded
1658        // one. `waits` counts the waits since the send started or since its
1659        // last accepted chunk; after the first wait, every attempt goes on
1660        // from what the guest holds then.
1661        let mut rsrc = Resources::new(guest.thread_state().dettid);
1662        rsrc.insert(ResourceID::InternalIOPolling, Permission::W);
1663        rsrc.fyi("network send");
1664        let mut waits = 0;
1665        let mut waited = false;
1666        if self.network_mode() == NetworkTraceMode::Record {
1667            // A nonblocking send the host refuses reports EAGAIN, and the
1668            // recording holds the refusal so that replay reports it at the
1669            // same stream offset. A blocking send carries its mark across
1670            // each wait, and the engine refuses it if any other output event
1671            // on the channel happened in between, which network replay does
1672            // not model.
1673            let mut at = None;
1674            while sent < len && !(nonblocking && sent > 0) {
1675                if waited {
1676                    self.network_send_resume(guest, fd, id, buffer, sent, &mut bytes)
1677                        .await;
1678                }
1679                let call = syscalls::Sendto::new()
1680                    .with_fd(fd)
1681                    // SAFETY: `sent < len` stays within the guest's buffer.
1682                    .with_buf(AddrMut::from_raw(unsafe { buffer.add(sent) }.as_raw()))
1683                    .with_size(len - sent)
1684                    .with_flags(
1685                        (flags & libc::MSG_MORE | libc::MSG_DONTWAIT | libc::MSG_NOSIGNAL) as u32,
1686                    );
1687                match guest.inject(call).await {
1688                    Ok(written) => {
1689                        let written = written as usize;
1690                        let request = NetworkRequest::RecordSend {
1691                            id,
1692                            bytes: bytes[sent..sent + written].to_vec(),
1693                            at,
1694                            waits,
1695                        };
1696                        let NetworkReply::Recorded(end) =
1697                            self.network_request(guest, request).await
1698                        else {
1699                            unreachable!()
1700                        };
1701                        at = Some(end);
1702                        sent += written;
1703                        waits = 0;
1704                    }
1705                    Err(Errno::EINTR) => {}
1706                    Err(Errno::EAGAIN) if nonblocking => {
1707                        self.network_request(guest, NetworkRequest::RecordRefusedSend(id))
1708                            .await;
1709                        return Err(Errno::EAGAIN.into());
1710                    }
1711                    Err(Errno::EAGAIN) => {
1712                        if at.is_none() {
1713                            // Waiting before the first accepted byte: take
1714                            // the mark now; an empty send records nothing.
1715                            let request = NetworkRequest::RecordSend {
1716                                id,
1717                                bytes: Vec::new(),
1718                                at: None,
1719                                waits: 0,
1720                            };
1721                            let NetworkReply::Recorded(mark) =
1722                                self.network_request(guest, request).await
1723                            else {
1724                                unreachable!()
1725                            };
1726                            at = Some(mark);
1727                        }
1728                        waits += 1;
1729                        waited = true;
1730                        self.network_send_wait(guest, fd, &mut rsrc).await;
1731                    }
1732                    Err(errno) => {
1733                        self.network_refuse(
1734                            guest,
1735                            &format!("network send failed in the host with {errno}"),
1736                            HOST_REMEDY,
1737                        )
1738                        .await
1739                    }
1740                }
1741            }
1742        } else {
1743            while sent < len && !(nonblocking && sent > 0) {
1744                if waited {
1745                    self.network_send_resume(guest, fd, id, buffer, sent, &mut bytes)
1746                        .await;
1747                }
1748                let request = NetworkRequest::ReplaySend {
1749                    id,
1750                    bytes: bytes[sent..].to_vec(),
1751                    nonblocking,
1752                    waits,
1753                };
1754                match self.network_request(guest, request).await {
1755                    NetworkReply::Sent(accepted) => {
1756                        sent += accepted;
1757                        waits = 0;
1758                    }
1759                    NetworkReply::WouldBlock => return Err(Errno::EAGAIN.into()),
1760                    NetworkReply::NotYet => {
1761                        waits += 1;
1762                        waited = true;
1763                        self.network_send_wait(guest, fd, &mut rsrc).await;
1764                    }
1765                    _ => unreachable!(),
1766                }
1767            }

1796    /// Prepare a blocking send to go on after it waited, while other guest
1797    /// threads ran. Linux's in-flight send keeps the open file description it
1798    /// started on and reads the buffer when it copies each part. So `fd` must
1799    /// still name the channel `id`, and the unsent bytes are read again; the
1800    /// caller sends, and records, exactly what the guest's memory holds now.
1801    async fn network_send_resume<G: Guest<Self>>(
1802        &self,
1803        guest: &mut G,
1804        fd: RawFd,
1805        id: OpenFileId,
1806        buffer: Addr<'_, u8>,
1807        sent: usize,
1808        bytes: &mut [u8],
1809    ) {
1810        let same_channel = guest
1811            .thread_state()
1812            .with_detfd(fd, |detfd| detfd.open_file_id() == id)
1813            .unwrap_or(false);
1814        if !same_channel {
1815            self.network_refuse(
1816                guest,
1817                "network trace does not model closing or replacing a descriptor while a send \
1818                 on it waits for buffer space",
1819                UNSUPPORTED_REMEDY,
1820            )
1821            .await
1822        }
1823        // SAFETY: `sent < bytes.len()` stays within the guest's buffer.
1824        let unsent = unsafe { buffer.add(sent) };
1825        if guest
1826            .memory()
1827            .read_exact(unsent, &mut bytes[sent..])
1828            .is_err()
1829        {
1830            self.network_refuse(
1831                guest,
1832                "network trace cannot send from a guest buffer that became unreadable while \
1833                 the send waited for buffer space",
1834                "The program unmapped a buffer while sending from it; fix the program.",
1835            )
1836            .await
1837        }
1838    }

1840    /// The pollfd array at `address` and, for each entry, its channel and
1841    /// receive low-water mark. `None` when the array names no channel, so the
1842    /// call takes its ordinary path. Also `None`, before anything is read,
1843    /// when `nfds` (counted by [`poll_nfds`]) exceeds the process's
1844    /// RLIMIT_NOFILE: Linux refuses that count with EINVAL before it reads the
1845    /// array, and the ordinary path checks the rest of the call in Linux's
1846    /// order and gets that refusal.
1847    fn network_pollfds<G: Guest<Self>>(
1848        guest: &mut G,
1849        address: Option<AddrMut<'_, libc::pollfd>>,
1850        nfds: u32,
1851    ) -> Option<(Vec<libc::pollfd>, Vec<Option<PolledChannel>>)> {
1852        let address = address?;
1853        let open_file_limit = guest
1854            .thread_state()
1855            .resource_limits
1856            .lock()
1857            .expect("resource limits mutex poisoned")
1858            .get(libc::RLIMIT_NOFILE)
1859            .map_or(libc::RLIM64_INFINITY, |limit| limit.current);
1860        if u64::from(nfds) > open_file_limit {
1861            return None;
1862        }
1863        let mut pollfds = vec![
1864            libc::pollfd {
1865                fd: -1,
1866                events: 0,
1867                revents: 0,
1868            };
1869            nfds as usize
1870        ];
1871        guest
1872            .memory()
1873            .read_values(address.into(), &mut pollfds)
1874            .ok()?;
1875        let channels: Vec<_> = pollfds
1876            .iter()
1877            .map(|pollfd| Self::polled_channel(guest, pollfd.fd))
1878            .collect();
1879        channels
1880            .iter()
1881            .any(Option::is_some)
1882            .then_some((pollfds, channels))
1883    }

1944        &self,
1945        guest: &mut G,

1949            Self::network_pollfds(guest, call.fds(), poll_nfds(call.nfds()))

1962            Some(address) => Some(ppoll_timeout_duration(guest.memory().read_value(address)?)?),

2143    /// Wait for readiness on pollfd entries naming channels, until `ready`
2144    /// accepts the reported entries or the timeout expires, and return them.
2145    /// Each scheduler turn asks the engine about the channels and probes every
2146    /// other descriptor with a zero-timeout host poll, as a strict
2147    /// guest-internal poll does; channel entries are masked to -1 for that
2148    /// probe, which Linux skips. The probe uses the guest's own pollfd array
2149    /// at `address` and restores it, or, with no array, the pull scratch,
2150    /// which then bounds the entries to `PSELECT6_INTERNAL_MAX_NFDS`.
2151    #[allow(clippy::too_many_arguments)]
2152    async fn network_wait<G: Guest<Self>>(
2153        &self,
2154        guest: &mut G,
2155        address: Option<AddrMut<'_, libc::pollfd>>,
2156        mut pollfds: Vec<libc::pollfd>,
2157        channels: &[Option<PolledChannel>],
2158        zero_timeout: bool,
2159        deadline: Option<LogicalTime>,
2160        interrupted: Errno,
2161        retry: Syscall,
2162        ready: fn(&[libc::pollfd]) -> bool,
2163    ) -> Result<Vec<libc::pollfd>, Error> {
2164        let record = self.network_mode() == NetworkTraceMode::Record;
2165        let probe: Vec<libc::pollfd> = pollfds
2166            .iter()
2167            .zip(channels)
2168            .map(|(pollfd, channel)| libc::pollfd {
2169                fd: if channel.is_some() { -1 } else { pollfd.fd },
2170                events: pollfd.events,
2171                revents: 0,
2172            })
2173            .collect();
2174        let probe_host = probe.iter().any(|pollfd| pollfd.fd >= 0);
2175        let mut stack = guest.stack().await;
2176        let scratch: AddrMut<[u8; POLL_PULL_BYTES]> = stack.reserve();
2177        let _guard = stack.commit()?;
2178        let probe_address = match address {
2179            Some(address) => address,
2180            None => {
2181                assert!(probe.len() <= PSELECT6_INTERNAL_MAX_NFDS as usize);
2182                scratch.cast()
2183            }
2184        };
2185        let mut rsrc = Resources::new(guest.thread_state().dettid);
2186        rsrc.insert(ResourceID::InternalIOPolling, Permission::W);
2187        rsrc.fyi("network poll");
2188        loop {
2189            if matches!(
2190                resource_request(guest, rsrc.clone()).await,
2191                ResumeStatus::Signaled(_)
2192            ) {
2193                return Err(interrupted.into());
2194            }
2195            let mut request = Vec::new();
2196            for (pollfd, channel) in pollfds.iter().zip(channels) {
2197                if let Some((id, lowat)) = channel {
2198                    let arrivals = if record {
2199                        self.network_pull(guest, pollfd.fd, scratch.cast(), POLL_PULL_BYTES)
2200                            .await?
2201                            .into_iter()
2202                            .collect()
2203                    } else {
2204                        Vec::new()
2205                    };
2206                    request.push((*id, *lowat, arrivals));
2207                }
2208            }
2209            let NetworkReply::Events(events) = self
2210                .network_request(guest, NetworkRequest::Readiness(request))
2211                .await
2212            else {
2213                unreachable!()
2214            };
2215            let mut host_revents = vec![0; pollfds.len()];
2216            if probe_host {
2217                guest
2218                    .memory()
2219                    .write_values(probe_address, &probe)
2220                    .map_err(|_| Errno::EFAULT)?;
2221                let call = syscalls::Poll::new()
2222                    .with_fds(Some(probe_address.cast()))
2223                    .with_nfds(probe.len() as libc::nfds_t)
2224                    .with_timeout(0);
2225                let probed = guest.inject(call).await;
2226                let mut after = probe.clone();
2227                let read = guest.memory().read_values(probe_address.into(), &mut after);
2228                // Restore the guest's descriptors before reporting anything.
2229                if let Some(address) = address {
2230                    guest
2231                        .memory()
2232                        .write_values(address, &pollfds)
2233                        .map_err(|_| Errno::EFAULT)?;
2234                }
2235                probed?;
2236                read?;
2237                for (revents, pollfd) in host_revents.iter_mut().zip(&after) {
2238                    *revents = pollfd.revents;
2239                }
2240            }
2241            let mut events = events.into_iter();
2242            for ((pollfd, channel), host) in pollfds.iter_mut().zip(channels).zip(host_revents) {
2243                pollfd.revents = match channel {
2244                    Some(_) => masked_revents(events.next().unwrap(), pollfd.events),
2245                    None => host,
2246                };
2247            }
2248            let expired = zero_timeout
2249                || match deadline {
2250                    Some(deadline) => thread_observe_time(guest).await >= deadline,
2251                    None => false,
2252                };
2253            if ready(&pollfds) || expired {
2254                return Ok(pollfds);
2255            }
2256            rsrc.poll_attempt += 1;
2257            record_retry_event(guest, retry).await;
2258        }
2259    }
