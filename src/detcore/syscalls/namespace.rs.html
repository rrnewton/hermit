38fn is_proc_id(component: &OsStr) -> bool {
39    component == "self"
40        || component == "thread-self"
41        || component.to_str().is_some_and(|value| {
42            !value.is_empty() && value.bytes().all(|byte| byte.is_ascii_digit())
43        })
44}

49    match name.to_str()? {

62// AUTONOMOUS-BOT-IMPLEMENTED
63// TODO-HUMAN-REVIEW(#877)
64fn canonical_namespace_target(path: &Path) -> Option<&'static [u8]> {
65    if !path.is_absolute() {
66        return None;
67    }
68
69    let mut parts = Vec::new();
70    for component in path.components() {
71        match component {
72            Component::RootDir => {}
73            Component::Normal(part) => parts.push(part),
74            Component::CurDir | Component::ParentDir | Component::Prefix(_) => return None,
75        }
76    }
77
78    let namespace = match parts.as_slice() {
79        [proc, subject, ns, namespace] if *proc == "proc" && is_proc_id(subject) && *ns == "ns" => {
80            namespace
81        }
82        [proc, subject, task, tid, ns, namespace]
83            if *proc == "proc"
84                && is_proc_id(subject)
85                && *task == "task"
86                && is_proc_id(tid)
87                && *ns == "ns" =>
88        {
89            namespace
90        }
91        _ => return None,
92    };
93    canonical_namespace_name(namespace)
94}

98fn normalized_absolute_parts(path: &Path) -> Option<Vec<&OsStr>> {
99    if !path.is_absolute() {
100        return None;
101    }
102
103    let mut parts = Vec::new();
104    for component in path.components() {
105        match component {
106            Component::RootDir | Component::CurDir => {}
107            Component::Normal(part) => parts.push(part),
108            Component::ParentDir => {
109                parts.pop()?;
110            }
111            Component::Prefix(_) => return None,
112        }
113    }
114    Some(parts)
115}

117fn decimal_u32(component: &OsStr) -> Option<u32> {
118    let value = component.to_str()?;
119    (!value.is_empty() && value.bytes().all(|byte| byte.is_ascii_digit()))
120        .then(|| value.parse().ok())?
121}

123fn decimal_fd(component: &OsStr) -> Option<i32> {
124    let value = component.to_str()?;
125    (!value.is_empty() && value.bytes().all(|byte| byte.is_ascii_digit()))
126        .then(|| value.parse().ok())?
127}

129/// Returns the optional numeric proc subject and the descriptor number.
130fn proc_fd_target(path: &Path) -> Option<(Option<u32>, i32)> {
131    let parts = normalized_absolute_parts(path)?;
132    match parts.as_slice() {
133        [dev, fd_dir, fd] if *dev == "dev" && *fd_dir == "fd" => Some((None, decimal_fd(fd)?)),
134        [proc, subject, fd_dir, fd] if *proc == "proc" && *fd_dir == "fd" => {
135            let subject = match subject.to_str()? {
136                "self" | "thread-self" => None,
137                _ => Some(decimal_u32(subject)?),
138            };
139            Some((subject, decimal_fd(fd)?))
140        }
141        _ => None,
142    }
143}

154struct AnonymousProcFdIdentity {
155    kind: &'static str,
156    /// The inode number the link names. Its device is the kernel's pipe or
157    /// socket filesystem ([`anonymous_object_device`]).
158    raw_inode: u64,
159}

161/// The host devices of the kernel's pipe and socket filesystems, on which
162/// every `pipe:[N]` or `socket:[N]` link's inode lives. Each is one
163/// kernel-wide superblock, so a pipe and a socket made here show them. A host
164/// inode is identified by device AND number, and a pipe's `fstat` reports this
165/// device, so the identity a link names must carry it too.
166#[derive(Clone, Copy, Debug, Eq, PartialEq)]
167pub(crate) struct AnonymousObjectDevices {
168    pipe: u64,
169    socket: u64,
170}
171
172/// The [`AnonymousObjectDevices`], read once per process, or why they could
173/// not be read. Detcore reads them when it starts in a process
174/// ([`crate::Detcore`]'s `Tool::new`), before any guest syscall, so a guest's
175/// readlink never depends on whether the tracer can make a pipe or socket at
176/// that moment, and keeps the outcome either way: a later call neither retries
177/// nor panics.
178pub(crate) fn anonymous_object_devices() -> &'static Result<AnonymousObjectDevices, String> {
179    static DEVICES: std::sync::OnceLock<Result<AnonymousObjectDevices, String>> =
180        std::sync::OnceLock::new();
181    DEVICES.get_or_init(|| retained_probe_outcome(probe_anonymous_object_devices()))
182}
183
184/// The probe's outcome as Detcore keeps it. Under in-guest LiteInst the probe
185/// runs inside the guest before `main`, so a failure is described without the
186/// C library's errno message (see [`crate::util::describe_io_error`]).
187fn retained_probe_outcome(
188    probe: std::io::Result<AnonymousObjectDevices>,
189) -> Result<AnonymousObjectDevices, String> {
190    probe.map_err(|error| crate::util::describe_io_error(&error))
191}

193fn probe_anonymous_object_devices() -> std::io::Result<AnonymousObjectDevices> {
194    use crate::util::raw_syscall;
195    use crate::util::raw_zero_result;
196    // Raw syscalls (see crate::util::raw_syscall): Detcore::new runs this
197    // inside the guest under in-guest LiteInst.
198    let close = |fds: [i32; 2]| {
199        for fd in fds {
200            unsafe { raw_syscall(libc::SYS_close, [fd as u64, 0, 0, 0, 0, 0]) };
201        }
202    };
203    let device_of = |fd: i32| -> std::io::Result<u64> {
204        let mut metadata: libc::stat = unsafe { std::mem::zeroed() };
205        raw_zero_result(unsafe {
206            raw_syscall(
207                libc::SYS_fstat,
208                [fd as u64, (&raw mut metadata) as u64, 0, 0, 0, 0],
209            )
210        })?;
211        Ok(metadata.st_dev)
212    };
213    let mut pipe = [0_i32; 2];
214    raw_zero_result(unsafe {
215        raw_syscall(
216            libc::SYS_pipe2,
217            [pipe.as_mut_ptr() as u64, libc::O_CLOEXEC as u64, 0, 0, 0, 0],
218        )
219    })?;
220    let pipe_device = device_of(pipe[0]);
221    close(pipe);
222    let mut sockets = [0_i32; 2];
223    raw_zero_result(unsafe {
224        raw_syscall(
225            libc::SYS_socketpair,
226            [
227                libc::AF_UNIX as u64,
228                (libc::SOCK_DGRAM | libc::SOCK_CLOEXEC) as u64,
229                0,
230                sockets.as_mut_ptr() as u64,
231                0,
232                0,
233            ],
234        )
235    })?;
236    let socket_device = device_of(sockets[0]);
237    close(sockets);
238    Ok(AnonymousObjectDevices {
239        pipe: pipe_device?,
240        socket: socket_device?,
241    })
242}

244/// The device of a `kind` ("pipe" or "socket") link's inode, from the
245/// devices read at start (`devices`). When they could not be read, this is a
246/// tool error, not an errno: Detcore cannot give the link a faithful identity,
247/// so the run fails as an infrastructure failure through the backend's own
248/// failure handling, rather than a link identity getting an invented device or
249/// a guest getting an error Linux would not return.
250fn anonymous_object_device(
251    devices: &Result<AnonymousObjectDevices, String>,
252    kind: &str,
253) -> Result<u64, Error> {
254    let devices = devices.as_ref().map_err(|error| {
255        Error::Tool(anyhow::anyhow!(
256            "Detcore could not read the pipe and socket filesystem devices when it started \
257             ({error}), so it cannot identify the {kind} a /proc fd link names"
258        ))
259    })?;
260    Ok(match kind {
261        "pipe" => devices.pipe,
262        "socket" => devices.socket,
263        _ => unreachable!("anonymous_proc_fd_identity names only pipes and sockets, not {kind}"),
264    })
265}

267// Long enough for `socket:[` + every decimal u64 inode + `]`.
268const ANONYMOUS_PROC_FD_TARGET_CAPACITY: usize = 32;
269
270fn needs_anonymous_proc_fd_scratch(buffer_present: bool, buffer_len: usize) -> bool {
271    buffer_present && buffer_len != 0 && buffer_len < ANONYMOUS_PROC_FD_TARGET_CAPACITY
272}

274/// Recognize only a complete kernel pipe/socket symlink target.
275fn anonymous_proc_fd_identity(target: &[u8]) -> Option<AnonymousProcFdIdentity> {
276    for (kind, prefix) in [
277        ("pipe", b"pipe:[".as_slice()),
278        ("socket", b"socket:[".as_slice()),
279    ] {
280        let Some(digits) = target
281            .strip_prefix(prefix)
282            .and_then(|rest| rest.strip_suffix(b"]"))
283        else {
284            continue;
285        };
286        if digits.is_empty() || !digits.iter().all(u8::is_ascii_digit) {
287            continue;
288        }
289        let raw_inode = std::str::from_utf8(digits).ok()?.parse().ok()?;
290        return Some(AnonymousProcFdIdentity { kind, raw_inode });
291    }
292    None
293}

295/// Match a raw identity against cached current-process inherited-stdio identities.
296/// Callers exclude ordinary replacement objects before filling this array.
297///
298/// When stdio descriptors alias, the lowest one wins, as in the maps
299/// sanitizer's `stdio_by_raw_inode` table. Detcore gives all three inherited
300/// descriptors the stat of its own stdin, so they alias until the guest
301/// replaces them: a link to that object is a link to fd 0, and must read as
302/// the `pipe:[1000]` a readlink of fd 0 returns.
303fn deterministic_stdio_inode_for_raw(
304    raw_inode: RawInode,
305    stdio_raw_inodes: &[Option<RawInode>; 3],
306) -> Option<DetInode> {
307    let fd = stdio_raw_inodes
308        .iter()
309        .position(|cached| *cached == Some(raw_inode))?;
310    deterministic_stdio_inode(fd as i32)
311}

313fn canonical_anonymous_proc_fd_target(
314    identity: &AnonymousProcFdIdentity,
315    inode: DetInode,
316    buffer_len: usize,
317) -> Vec<u8> {
318    let mut target = format!("{}:[{}]", identity.kind, inode.as_raw()).into_bytes();
319    target.truncate(buffer_len);
320    target
321}

324    async fn canonicalize_other_proc_fd_target<G>(
325        &self,
326        guest: &mut G,
327        raw_target: &[u8],
328        buffer: Option<AddrMut<'_, libc::c_char>>,
329        buffer_len: usize,
330    ) -> Result<Option<i64>, Error>
331    where
332        G: Guest<Self>,
333    {
334        let Some(identity) = anonymous_proc_fd_identity(raw_target) else {
335            return Ok(None);
336        };
337        let mut stdio_raw_inodes = [None; 3];
338        for fd in libc::STDIN_FILENO..=libc::STDERR_FILENO {
339            stdio_raw_inodes[fd as usize] = guest
340                .thread_state()
341                .with_detfd(fd, |detfd| {
342                    deterministic_stdio_inode_for_resource(fd, detfd.resource())?;
343                    detfd.stat().map(|stat| stat.raw_inode())
344                })
345                .ok()
346                .flatten();
347        }
348        let raw_inode = RawInode::new(
349            anonymous_object_device(anonymous_object_devices(), identity.kind)?,
350            identity.raw_inode,
351        );
352        // Every link sends its numbering request, and a stdio match only
353        // chooses the rendering afterwards, as `/proc/*/maps` does: whether
354        // the raw identity matches stdio is host identity, which must not
355        // decide how many inode numbers the call consumes
356        // (<https://github.com/rrnewton/hermit/issues/2897>).
357        let pooled = determinize_inode(guest, raw_inode).await.0;
358        let inode =
359            deterministic_stdio_inode_for_raw(raw_inode, &stdio_raw_inodes).unwrap_or(pooled);
360        let target = canonical_anonymous_proc_fd_target(&identity, inode, buffer_len);
361        let buffer = buffer.ok_or(Errno::EFAULT)?;
362        guest.memory().write_exact(buffer.cast(), &target)?;
363        Ok(Some(target.len() as i64))
364    }

366    /// Canonicalize a pipe/socket link belonging to another virtual process.
367    /// The target descriptor is not in the caller's table, so its raw readlink
368    /// bytes are the only safe identity evidence available here.
369    async fn canonicalize_other_proc_fd_readlink<G>(
370        &self,
371        guest: &mut G,
372        buffer: Option<AddrMut<'_, libc::c_char>>,
373        buffer_len: usize,
374        result: i64,
375    ) -> Result<i64, Error>
376    where
377        G: Guest<Self>,
378    {
379        let buffer = buffer.expect("a successful readlink requires a non-null buffer");
380        let observed_len = usize::try_from(result)
381            .expect("a positive readlink result must fit usize")
382            .min(buffer_len);
383        let mut observed = vec![0; observed_len];
384        guest.memory().read_exact(buffer.cast(), &mut observed)?;
385        Ok(self
386            .canonicalize_other_proc_fd_target(guest, &observed, Some(buffer), buffer_len)
387            .await?
388            .unwrap_or(result))
389    }

391    async fn canonicalize_namespace_readlink_result<G>(
392        &self,
393        guest: &mut G,
394        path: PathBuf,
395        buffer: Option<AddrMut<'_, libc::c_char>>,
396        buffer_len: usize,
397        result: i64,
398    ) -> Result<i64, Error>
399    where
400        G: Guest<Self>,
401    {
402        if result <= 0 {
403            return Ok(result);
404        }
405
406        let target = if let Some(target) = canonical_namespace_target(&path) {
407            target.to_vec()
408        } else if let Some((subject, fd)) = proc_fd_target(&path) {
409            if let Some(subject) = subject {
410                let current_pid = guest.inject(syscalls::Getpid::new()).await?;
411                if current_pid != i64::from(subject) {
412                    return self
413                        .canonicalize_other_proc_fd_readlink(guest, buffer, buffer_len, result)
414                        .await;
415                }
416            }
417
418            let stat = self.inject_fstat(guest, fd).await?;
419            let kind = match stat.st_mode & libc::S_IFMT {
420                libc::S_IFIFO => "pipe",
421                libc::S_IFSOCK => "socket",
422                _ => return Ok(result),
423            };
424            let inode_override = guest
425                .thread_state()
426                .with_detfd(fd, |detfd| {
427                    deterministic_stdio_inode_for_resource(fd, detfd.resource())
428                })
429                .ok()
430                .flatten();
431            let inode = match inode_override {
432                Some(inode) => inode,
433                None => {
434                    determinize_inode(guest, RawInode::new(stat.st_dev, stat.st_ino))
435                        .await
436                        .0
437                }
438            };
439            format!("{kind}:[{inode}]").into_bytes()
440        } else {
441            return Ok(result);
442        };
443
444        let written = target.len().min(buffer_len);
445        let buffer = buffer.expect("a successful readlink requires a non-null buffer");
446        guest
447            .memory()
448            .write_exact(buffer.cast(), &target[..written])?;
449        Ok(written as i64)
450    }
451    async fn write_other_proc_fd_target<G>(
452        &self,
453        guest: &mut G,
454        raw_target: &[u8],
455        buffer: Option<AddrMut<'_, libc::c_char>>,
456        buffer_len: usize,
457    ) -> Result<i64, Error>
458    where
459        G: Guest<Self>,
460    {
461        if let Some(result) = self
462            .canonicalize_other_proc_fd_target(guest, raw_target, buffer, buffer_len)
463            .await?
464        {
465            return Ok(result);
466        }
467        let written = raw_target.len().min(buffer_len);
468        let buffer = buffer.ok_or(Errno::EFAULT)?;
469        guest
470            .memory()
471            .write_exact(buffer.cast(), &raw_target[..written])?;
472        Ok(written as i64)
473    }

475    async fn finish_other_proc_fd_readlink<G>(
476        &self,
477        guest: &mut G,
478        call: syscalls::Readlink,
479    ) -> Result<i64, Error>
480    where
481        G: Guest<Self>,
482    {
483        let buffer = call.buf();
484        let buffer_len = call.bufsize();
485        if !needs_anonymous_proc_fd_scratch(buffer.is_some(), buffer_len) {
486            let result = self.record_or_replay(guest, call).await?;
487            if result <= 0 {
488                return Ok(result);
489            }
490            return self
491                .canonicalize_other_proc_fd_readlink(guest, buffer, buffer_len, result)
492                .await;
493        }
494
495        let mut stack = guest.stack().await;
496        let scratch = stack
497            .reserve::<[u8; ANONYMOUS_PROC_FD_TARGET_CAPACITY]>()
498            .cast::<libc::c_char>();
499        let guard = stack.commit()?;
500        let physical_call = call
501            .with_buf(Some(scratch))
502            .with_bufsize(ANONYMOUS_PROC_FD_TARGET_CAPACITY);
503        let result = self.record_or_replay(guest, physical_call).await?;
504        let length = usize::try_from(result)
505            .expect("a successful readlink result must fit usize")
506            .min(ANONYMOUS_PROC_FD_TARGET_CAPACITY);
507        let mut raw_target = vec![0; length];
508        guest.memory().read_exact(scratch.cast(), &mut raw_target)?;
509        drop(guard);
510        self.write_other_proc_fd_target(guest, &raw_target, buffer, buffer_len)
511            .await
512    }

572    // AUTONOMOUS-BOT-IMPLEMENTED
573    // TODO-HUMAN-REVIEW(#877)
574    /// Preserve Linux readlink errors and canonicalize procfs namespace identities.
575    pub async fn handle_readlink<G: Guest<Self>>(
576        &self,
577        guest: &mut G,
578        call: syscalls::Readlink,
579    ) -> Result<i64, Error> {
580        let path: PathBuf = call.path().ok_or(Errno::EFAULT)?.read(&guest.memory())?;
581        let (host_path, other_proc_fd) = if let Some((Some(subject), _)) = proc_fd_target(&path) {
582            let current_pid = guest.inject(syscalls::Getpid::new()).await?;
583            (
584                host_self_proc_fd_alias(&path, current_pid),
585                current_pid != i64::from(subject),
586            )
587        } else {
588            (None, false)
589        };
590        if let Some(host_path) = host_path {
591            let bytes = host_path.as_os_str().as_bytes();
592            let mut path_buffer = [0_u8; 64];
593            path_buffer[..bytes.len()].copy_from_slice(bytes);
594            let mut stack = guest.stack().await;
595            let path_address = stack.push(path_buffer).cast::<libc::c_char>();
596            let stack_guard = stack.commit()?;
597            let physical_call = call.with_path(PathPtr::from_ptr(
598                path_address.as_raw() as *const libc::c_char
599            ));
600            let result = self.record_or_replay(guest, physical_call).await?;
601            drop(stack_guard);
602            return self
603                .canonicalize_namespace_readlink_result(
604                    guest,
605                    path,
606                    call.buf(),
607                    call.bufsize(),
608                    result,
609                )
610                .await;
611        }
612        if other_proc_fd {
613            return self.finish_other_proc_fd_readlink(guest, call).await;
614        }
615        self.finish_namespace_readlink(guest, path, call.buf(), call.bufsize(), call)
616            .await
617    }

619    // AUTONOMOUS-BOT-IMPLEMENTED
620    // TODO-HUMAN-REVIEW(#877)
621    /// Preserve Linux readlinkat errors and canonicalize absolute procfs namespace identities.
622    pub async fn handle_readlinkat<G: Guest<Self>>(
623        &self,
624        guest: &mut G,
625        call: syscalls::Readlinkat,
626    ) -> Result<i64, Error> {
627        let path: PathBuf = call.path().ok_or(Errno::EFAULT)?.read(&guest.memory())?;
628        let observed_path = if path.is_absolute() || call.dirfd() == libc::AT_FDCWD {
629            path
630        } else {
631            guest
632                .thread_state()
633                .with_detfd(call.dirfd(), |detfd| detfd.path())?
634                .map_or(path.clone(), |directory| directory.join(path))
635        };
636        if let Some((Some(subject), _)) = proc_fd_target(&observed_path) {
637            let current_pid = guest.inject(syscalls::Getpid::new()).await?;
638            if current_pid != i64::from(subject) {
639                return self.finish_other_proc_fd_readlinkat(guest, call).await;
640            }
641        }
642        self.finish_namespace_readlink(guest, observed_path, call.buf(), call.buf_len(), call)
643            .await
644    }
645}
