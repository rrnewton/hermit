9//! Linux's priority-inheritance futex protocol (FUTEX_LOCK_PI, FUTEX_LOCK_PI2,
10//! FUTEX_TRYLOCK_PI and FUTEX_UNLOCK_PI), as decisions over the futex word and
11//! what the scheduler holds for it.
12//!
13//! The word holds its owner's TID (`FUTEX_TID_MASK`), `FUTEX_WAITERS` once a
14//! waiter has queued in the kernel, and `FUTEX_OWNER_DIED` after a robust
15//! owner died (Documentation/locking/pi-futex.rst). User space takes and
16//! releases an uncontended lock with its own compare-and-exchange; the kernel
17//! sees only the slow paths. All line references are to Linux v6.16:
18//!
19//! - `futex_lock_pi_atomic` (kernel/futex/pi.c:515-609,
20//!   <https://git.kernel.org/pub/scm/linux/kernel/git/stable/linux.git/tree/kernel/futex/pi.c?h=v6.16#n515>)
21//!   decides LOCK_PI and TRYLOCK_PI ([`lock_pi_step`]);
22//! - `futex_unlock_pi` (pi.c:1132-1293,
23//!   <https://git.kernel.org/pub/scm/linux/kernel/git/stable/linux.git/tree/kernel/futex/pi.c?h=v6.16#n1132>)
24//!   and `wake_futex_pi` (pi.c:614-669) decide UNLOCK_PI ([`unlock_pi_step`]);
25//! - `fixup_pi_owner` and `__fixup_pi_state_owner` (pi.c:872-907, 671-841)
26//!   give a waiter the word after the owner died ([`owner_death_word`]).
27//!
28//! Detcore keeps no rt_mutex. The scheduler parks LOCK_PI waiters in its futex
29//! waiter pool, marked as PI waiters, and records each contended word's owner
30//! (`Scheduler::pi_futex_state`). Priority boosting has no meaning under
31//! Detcore's scheduler, and waiter order needs none: `__waiter_prio`
32//! (kernel/locking/rtmutex.c:349-357) gives every task that is not real-time
33//! or deadline the same rt_mutex priority, `DEFAULT_PRIO`, so their waiters
34//! are served first in, first out, and `rt_mutex_steal` (rtmutex.c:430-448)
35//! lets no equal-priority task take the lock from the top waiter. Detcore
36//! accepts `sched_setscheduler` as a no-op, so every guest thread is such a
37//! task in the kernel's eyes, and first in, first out is Linux's own order.

48/// How a FUTEX_LOCK_PI waiter was handed the lock it waited for.
49#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Serialize, Deserialize)]
50pub enum PiHandOver {
51    /// FUTEX_UNLOCK_PI, which already wrote the waiter's TID (`wake_futex_pi`).
52    Unlock,
53    /// The owner's exit or exec (`exit_pi_state_list`), after which the waiter
54    /// writes its TID itself ([`owner_death_word`]).
55    OwnerDeath,
56}

58/// What the scheduler holds for a futex word a PI command acts on: Linux's
59/// `futex_top_waiter` and the `pi_state` attached to it.
60#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Serialize, Deserialize)]
61pub enum PiFutexState {
62    /// No waiter is queued on the word, so the kernel holds no state for it.
63    /// Carries whether the TID the word names is a live guest thread, as
64    /// `find_get_task_by_vpid` decides in `attach_to_pi_owner` (pi.c:427).
65    Unqueued {
66        /// The word's owner TID names a live thread.
67        named_owner_alive: bool,
68        /// Queueing behind that thread would close a cycle of PI waits.
69        queueing_deadlocks: bool,
70    },
71    /// The first waiter is a plain FUTEX_WAIT waiter, which has no `pi_state`
72    /// (`attach_to_pi_state` [3], pi.c:214-215; `futex_unlock_pi`,
73    /// pi.c:1169-1171).
74    PlainWaiters,
75    /// A `pi_state` is live: PI waiters are queued, or its owner was handed the
76    /// lock and has not yet taken it. Carries the owner and the PI waiter next
77    /// in line (`rt_mutex_top_waiter`), if one is queued.
78    Contended {
79        /// The `pi_state` owner.
80        owner: DetTid,
81        /// The longest-waiting PI waiter.
82        next_owner: Option<DetTid>,
83        /// Queueing behind the owner would close a cycle of PI waits.
84        queueing_deadlocks: bool,
85    },
86}

88/// What FUTEX_LOCK_PI or FUTEX_TRYLOCK_PI does next ([`lock_pi_step`]).
89#[derive(Clone, Copy, Debug, PartialEq, Eq)]
90pub(crate) enum LockPiStep {
91    /// Return `errno`, after writing `write` to the word when it is set: a
92    /// first waiter sets `FUTEX_WAITERS` before it looks the owner up
93    /// (pi.c:599-608), and the bit stays when that fails.
94    Fail { errno: Errno, write: Option<u32> },
95    /// Take the lock: write this word and return 0.
96    Acquire(u32),
97    /// Queue behind `owner`, after writing `write` to the word when it is set.
98    Block { owner: DetTid, write: Option<u32> },
99}
100
101/// What FUTEX_UNLOCK_PI does once the caller is known to own the word
102/// ([`unlock_pi_step`]).
103#[derive(Clone, Copy, Debug, PartialEq, Eq)]
104pub(crate) enum UnlockPiStep {
105    /// Return this errno, leaving the word as it is.
106    Fail(Errno),
107    /// No waiter is queued: write 0 and return 0 (pi.c:1251-1276).
108    Release,
109    /// Write `write`, hand the lock to `to` and wake it (`wake_futex_pi`).
110    HandOff { to: DetTid, write: u32 },
111}
112
113/// The owner TID a futex word names.
114pub(crate) fn word_owner(word: u32) -> u32 {
115    word & FUTEX_TID_MASK
116}
117
118/// The TID a guest thread puts in a PI futex word, as the kernel's
119/// `task_pid_vnr` reports it. `DetTid` is the raw namespaced TID
120/// (`Detcore::run_robust_list_owner_death` explains why, and names the test
121/// that fails the moment the two diverge).
122pub(crate) fn guest_tid(thread: DetTid) -> u32 {
123    thread.as_raw() as u32
124}

126/// `futex_lock_pi_atomic` (pi.c:515-609) for `caller` on a word holding `word`,
127/// with what the scheduler holds for it, and `futex_lock_pi`'s TRYLOCK_PI
128/// answer when `trylock` (pi.c:989-993):
129///
130/// 1. the caller already owns the word: `EDEADLK` (pi.c:539-540);
131/// 2. a waiter is queued (`attach_to_pi_state`, pi.c:203-319): `EINVAL` if it
132///    is a plain waiter; otherwise the word must name the state's owner, or be
133///    marked `FUTEX_OWNER_DIED` with no owner TID (the owner died and the waiter
134///    handed the lock has not yet written its TID, [4]/[6]), else `EINVAL`
135///    ([5], [7], [9/10]). Then the caller queues without touching the word;
136/// 3. no waiter and no owner TID: take the lock, keeping `FUTEX_OWNER_DIED`
137///    (pi.c:559-592);
138/// 4. no waiter and an owner TID: set `FUTEX_WAITERS` (pi.c:599-602), then
139///    `ESRCH` if the TID names no live thread (`handle_exit_race`, pi.c:374),
140///    else queue behind it (`attach_to_pi_owner`).
141///
142/// A LOCK_PI that would queue behind an owner who waits, through PI futexes,
143/// for a lock the caller owns fails with `EDEADLK` instead, before it sleeps
144/// (`__rt_mutex_start_proxy_lock`'s chain walk, kernel/locking/rtmutex.c:860-861),
145/// leaving the `FUTEX_WAITERS` a first waiter set.
146///
147/// A TRYLOCK_PI that would queue fails with `EWOULDBLOCK` instead: the
148/// rt_mutex is owned, or its top waiter is someone else and no equal-priority
149/// task steals it (`rt_mutex_steal`). The `FUTEX_WAITERS` a first waiter set
150/// stays in the word.
151///
152/// The kernel's compare-and-exchange retries (`-EAGAIN` from
153/// `lock_pi_update_atomic`) cannot happen here: Detcore runs one guest thread
154/// at a time and the caller holds the turn, so the word cannot change between
155/// the read and the write. Nor can an owner be found mid-exit (`-EBUSY`): a
156/// thread is either still scheduled or already retired.
157pub(crate) fn lock_pi_step(
158    word: u32,
159    caller: u32,
160    state: PiFutexState,
161    trylock: bool,
162) -> LockPiStep {
163    let fail = |errno| LockPiStep::Fail { errno, write: None };
164    if word_owner(word) == caller {
165        return fail(Errno::EDEADLK);
166    }
167    match state {
168        PiFutexState::PlainWaiters => fail(Errno::EINVAL),
169        PiFutexState::Contended {
170            owner,
171            queueing_deadlocks,
172            ..
173        } => {
174            let named = word_owner(word);
175            let owner_died_unfixed = word & FUTEX_OWNER_DIED != 0 && named == 0;
176            if !owner_died_unfixed && named != guest_tid(owner) {
177                fail(Errno::EINVAL)
178            } else if trylock {
179                fail(Errno::EWOULDBLOCK)
180            } else if queueing_deadlocks {
181                fail(Errno::EDEADLK)
182            } else {
183                LockPiStep::Block { owner, write: None }
184            }
185        }
186        PiFutexState::Unqueued { .. } if word_owner(word) == 0 => {
187            LockPiStep::Acquire((word & FUTEX_OWNER_DIED) | caller)
188        }
189        PiFutexState::Unqueued {
190            named_owner_alive,
191            queueing_deadlocks,
192        } => {
193            let write = Some(word | FUTEX_WAITERS);
194            if !named_owner_alive {
195                LockPiStep::Fail {
196                    errno: Errno::ESRCH,
197                    write,
198                }
199            } else if trylock {
200                LockPiStep::Fail {
201                    errno: Errno::EWOULDBLOCK,
202                    write,
203                }
204            } else if queueing_deadlocks {
205                LockPiStep::Fail {
206                    errno: Errno::EDEADLK,
207                    write,
208                }
209            } else {
210                let owner = DetTid::from_raw(word_owner(word) as i32);
211                LockPiStep::Block { owner, write }
212            }
213        }
214    }
215}

217/// `futex_unlock_pi` (pi.c:1132-1293) for `caller`, whose TID the word
218/// already names (the `EPERM` check, pi.c:1148-1149, comes first):
219///
220/// - a plain first waiter, or a state the caller does not own, is `EINVAL`
221///   (pi.c:1169-1178);
222/// - with PI waiters, `wake_futex_pi` (pi.c:614-669) writes
223///   `FUTEX_WAITERS | next owner` (pi.c:631), dropping `FUTEX_OWNER_DIED`,
224///   and hands the lock over;
225/// - with no waiter the word becomes 0, keeping neither `FUTEX_WAITERS` nor
226///   `FUTEX_OWNER_DIED` (pi.c:1251-1258).
227pub(crate) fn unlock_pi_step(caller: DetTid, state: PiFutexState) -> UnlockPiStep {
228    match state {
229        PiFutexState::PlainWaiters => UnlockPiStep::Fail(Errno::EINVAL),
230        PiFutexState::Contended { owner, .. } if owner != caller => {
231            UnlockPiStep::Fail(Errno::EINVAL)
232        }
233        PiFutexState::Contended {
234            next_owner: Some(to),
235            ..
236        } => UnlockPiStep::HandOff {
237            to,
238            write: FUTEX_WAITERS | guest_tid(to),
239        },
240        PiFutexState::Contended {
241            next_owner: None, ..
242        }
243        | PiFutexState::Unqueued { .. } => UnlockPiStep::Release,
244    }
245}

247/// The word a PI waiter writes when its owner's death handed it the lock, as
248/// `fixup_pi_owner` (pi.c:872-886) and `__fixup_pi_state_owner` (pi.c:748-758)
249/// write it.
250///
251/// `exit_pi_state_list` (kernel/futex/core.c:1323-1402) clears the state's
252/// owner and leaves the word as the owner's exit made it: `FUTEX_OWNER_DIED`
253/// with no TID after a robust list walk, or the dead owner's TID. The waiter
254/// then writes its own TID with `FUTEX_WAITERS` and, because the state had no
255/// owner, `FUTEX_OWNER_DIED`, robust mutex or not, so glibc reports
256/// `EOWNERDEAD` for a robust mutex. A waiter FUTEX_UNLOCK_PI handed the lock
257/// writes nothing: the unlock already wrote its TID (pi.c:631).
258pub(crate) fn owner_death_word(waiter: u32) -> u32 {
259    FUTEX_OWNER_DIED | FUTEX_WAITERS | waiter
260}

269    fn tid(raw: u32) -> DetTid {
270        DetTid::from_raw(raw as i32)

273    fn unqueued(alive: bool) -> PiFutexState {
274        PiFutexState::Unqueued {
275            named_owner_alive: alive,
276            queueing_deadlocks: false,
277        }
278    }
279
280    fn contended(owner: u32, next: u32) -> PiFutexState {
281        PiFutexState::Contended {
282            owner: tid(owner),
283            next_owner: Some(tid(next)),
284            queueing_deadlocks: false,
285        }
286    }

288    #[test]
289    fn an_unowned_word_is_taken_keeping_only_owner_died() {
290        for trylock in [false, true] {
291            assert_eq!(
292                lock_pi_step(0, ME, unqueued(false), trylock),
293                LockPiStep::Acquire(ME)
294            );
295            // The waiters bit with no owner and no waiter (the owner unlocked
296            // after its last waiter timed out) is not kept.
297            assert_eq!(
298                lock_pi_step(FUTEX_WAITERS, ME, unqueued(false), trylock),
299                LockPiStep::Acquire(ME)
300            );
301            // A robust owner died with no waiter: glibc then sees
302            // FUTEX_OWNER_DIED and reports EOWNERDEAD.
303            assert_eq!(
304                lock_pi_step(FUTEX_OWNER_DIED, ME, unqueued(false), trylock),
305                LockPiStep::Acquire(FUTEX_OWNER_DIED | ME)
306            );
307        }
308    }
