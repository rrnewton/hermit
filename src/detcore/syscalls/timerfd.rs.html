9//! timerfd expirations on Detcore's virtual clock (timerfd_create(2)).
10//!
11//! Detcore keeps each timerfd's setting (`crate::fd::VirtualTimerfd`) and
12//! leaves the host timerfd disarmed, so the host never makes it readable on
13//! its own. Before a probe that can observe a timerfd's readiness or count --
14//! a `read`/`readv` of one, and every `poll`, `ppoll`, `select`, `pselect6`,
15//! `epoll_wait`, `epoll_pwait` and `epoll_pwait2` probe -- [`publish_expired`]
16//! counts the expirations at or before the guest clock and, for a timerfd that
17//! gained one, makes the host descriptor readable and waits until the host
18//! reports it so. Only then does the model record the expirations. A read
19//! returns the model's count; `timerfd_gettime` answers from it.
20//!
21//! A call that would wait in the host, outside these probes, on descriptors
22//! that may include a virtual timerfd would wait forever: the host is never
23//! armed. Such a call refuses the run by name ([`refuse_host_wait`]), as does
24//! any failure to make an expired timerfd readable ([`refuse`]), instead of
25//! returning an errno Linux would not.
26//!
27//! Determinism: the host descriptor becomes readable only inside the probing
28//! thread's turn, and that turn does not continue until the host reports it
29//! readable, so the host state never lags the model at any probe. Whether a
30//! probe finds the timerfd ready, and what a read returns, are therefore
31//! functions of the model and of the guest clock read at that probe. The guest
32//! clock read is `global_time_lower_bound` under the guest clock's floor, the
33//! same reading `clock_gettime` returns; it includes the time charged for
34//! internal polling retries, whose number is itself host-perturbed where a
35//! probe waits on an external, host-timed source, as for every other timeout
36//! Detcore serves. Host time decides only how long the readiness wait lasts,
37//! up to `READABLE_VALVE`. Expirations are counted exactly from the deadline
38//! and interval, never rounded.
39//!
40//! Scope: only where Detcore's model of an open file description is the one
41//! copy every process sees. Detcore outside the guest (ptrace, KVM) shares the
42//! description's model across a fork and keeps it across exec, as Linux
43//! shares the timer. A tool inside the guest copies its state into a forked
44//! child, so there the runtime must promote the description to Detcore's
45//! global state before each native fork or exec and carry it across exec as a
46//! survivor (`promotes_open_files_before_native_lifecycle`). DBT's runtime
47//! does, for the calls it performs natively; on in-guest LiteInst every such
48//! call goes through Detcore's own handler, which promotes first. SaBRe does
49//! not yet, so there a strict run's timerfd is refused by name
50//! ([`refuse_unshared_model`]) rather than left on host time. Also not under record/replay,
51//! whose recordings keep the host's answers, or without sequentialized
52//! threads. `TFD_TIMER_CANCEL_ON_SET` is accepted and never fires, because
53//! Hermit's virtual clock is never set; an absolute deadline is read on the
54//! guest clock for every clock id, as POSIX timers are.

80/// How long, in host time, Detcore waits for an expired host timerfd to
81/// become readable before it refuses the run, like the scheduler's other
82/// valves. An expired high-resolution timer fires within microseconds.
83const READABLE_VALVE: Duration = Duration::from_secs(30);

86/// the first publication only. `never-readable`: Detcore does not arm the
87/// host timerfd and waits one second for it, so the run ends on the named
88/// refusal. `signal-before-arm`: Detcore sends the guest thread a SIGCONT

103/// Whether Detcore's tool state runs outside the guest's processes, so one
104/// copy of an open file description's model serves every process that shares
105/// it: a fork shares the model as Linux shares the timer, and exec keeps it.
106///
107/// Reverie names no capability for where the tool runs, so this derives it
108/// from the two that follow from running inside the guest:
109/// `tool_shares_guest_descriptor_table` (the tool runs in the guest process
110/// and shares its descriptor table: SaBRe, DBT) and
111/// `process_exits_complete_asynchronously` (the tool runs in the guest
112/// process, so a process's exit completes only after its tool is gone, outside
113/// any turn: in-guest LiteInst). A backend whose tool runs inside the guest
114/// copies the tool's state into a forked child, so a timer's count would
115/// split. ptrace and KVM set neither; `detcore_runs_outside_the_guest_per_backend`
116/// pins the answer for every backend Reverie defines.
117pub(crate) fn detcore_runs_outside_the_guest(backend: &reverie::BackendCapabilities) -> bool {
118    !backend.tool_shares_guest_descriptor_table && !backend.process_exits_complete_asynchronously
119}

121/// Whether Detcore's model of an open file description is the one copy every
122/// guest process sharing it reaches: Detcore runs outside the guest, or such
123/// descriptions are promoted before every native fork or exec inside it and
124/// survive exec (by the backend's runtime, or by Detcore's own handler where
125/// the backend performs those calls only from it).
126pub(crate) fn models_are_shared(backend: &reverie::BackendCapabilities) -> bool {
127    detcore_runs_outside_the_guest(backend) || backend.promotes_open_files_before_native_lifecycle
128}

130/// Whether Detcore keeps timerfds on virtual time (see the module comment
131/// for the scope): virtual time under sequentialized threads, outside
132/// record/replay, where Detcore's model of a description is the one copy
133/// every process sees. That holds where Detcore runs outside the guest, and
134/// where the runtime inside it promotes timerfd descriptions before each
135/// native fork or exec and carries them across exec (see
136/// [`models_are_shared`]).
137pub(crate) fn enabled<T, G>(guest: &G) -> bool
138where
139    T: RecordOrReplay,
140    G: Guest<Detcore<T>>,
141{
142    let config = guest.config();
143    config.virtualize_time
144        && config.sequentialize_threads
145        && !config.recordreplay_modes
146        && models_are_shared(&config.backend)
147}

149/// Refuses a timerfd that Detcore would otherwise leave on the host clock in
150/// a run whose time it virtualizes: virtual time under sequentialized threads,
151/// outside record/replay, on a backend whose runtime does not share one model
152/// of the description ([`models_are_shared`]). The timer's expirations would
153/// follow host time there, so the run ends by name instead.
154pub(crate) fn refuse_unshared_model<T, G>(guest: &G)
155where
156    T: RecordOrReplay,
157    G: Guest<Detcore<T>>,
158{
159    let config = guest.config();
160    if config.virtualize_time
161        && config.sequentialize_threads
162        && !config.recordreplay_modes
163        && !models_are_shared(&config.backend)
164    {
165        refuse(format_args!(
166            "timerfd_create on a backend whose runtime does not keep one model of an open \
167             file description across fork and exec (it does not promote them before a \
168             native fork or exec), so the timer would follow the host clock"
169        ));
170    }
171}

204pub(crate) fn refuse_host_wait<T, G>(guest: &G, call: &str)

216/// Most descriptors a `poll`/`ppoll` list or `select`/`pselect6` set Detcore
217/// reads to decide whether it names a timerfd; a larger call counts as naming
218/// one.
219const MAX_SCANNED_DESCRIPTORS: usize = 1 << 16;
220
221/// Whether `call`, about to wait in the host, may name a timerfd: one of its
222/// `poll`/`ppoll` entries or `select`/`pselect6` set bits is a timerfd, or it
223/// reads one. An epoll descriptor in the list or set (its members, which
224/// Detcore cannot see, may include a timerfd), or a list Detcore cannot read,
225/// counts as naming one.
226///
227/// The list is read before the call is released to wait in the host, so a
228/// sibling thread that rewrites it in between could add a timerfd unseen: the
229/// same window the masked-`pselect6` staging describes. It needs a guest
230/// program that races its own poll arguments, and only ends in a wait where a
231/// refusal was due, never in a wrong result.
232fn call_may_name_timerfd<T, G>(guest: &G, call: Syscall) -> bool
233where
234    T: RecordOrReplay,
235    G: Guest<Detcore<T>>,
236{
237    let names = |fds: &mut dyn Iterator<Item = RawFd>| {
238        for fd in fds {
239            let ty = guest.thread_state().with_detfd(fd, |detfd| detfd.ty());
240            if matches!(ty, Ok(FdType::Timerfd) | Ok(FdType::Epoll)) {
241                return true;
242            }
243        }
244        false
245    };
246    let pollfds = |address: Option<usize>, nfds: u64| -> bool {
247        let count = nfds as u32 as usize;
248        if count == 0 {
249            return false;
250        }
251        let Some(address) = address.and_then(syscalls::Addr::<u8>::from_raw) else {
252            return true;
253        };
254        if count > MAX_SCANNED_DESCRIPTORS {
255            return true;
256        }
257        let mut bytes = vec![0u8; count * std::mem::size_of::<libc::pollfd>()];
258        if guest.memory().read_exact(address, &mut bytes).is_err() {
259            return true;
260        }
261        names(
262            &mut bytes
263                .as_chunks::<{ std::mem::size_of::<libc::pollfd>() }>()
264                .0
265                .iter()
266                .map(|entry| RawFd::from_ne_bytes(entry[..4].try_into().unwrap())),
267        )
268    };
269    let fd_sets = |nfds: i32, sets: [Option<usize>; 3]| -> bool {
270        let Ok(count) = usize::try_from(nfds) else {
271            return false;
272        };
273        if count > MAX_SCANNED_DESCRIPTORS {
274            return true;
275        }
276        let len = count.div_ceil(64) * 8;
277        sets.into_iter().flatten().any(|set| {
278            let Some(address) = syscalls::Addr::<u8>::from_raw(set) else {
279                return false;
280            };
281            let mut bits = vec![0u8; len];
282            if guest.memory().read_exact(address, &mut bits).is_err() {
283                return true;
284            }
285            names(
286                &mut (0..count)
287                    .filter(|fd| bits[fd / 8] & (1 << (fd % 8)) != 0)
288                    .map(|fd| fd as RawFd),
289            )
290        })
291    };
292    match call {
293        Syscall::Poll(poll) => pollfds(poll.fds().map(|fds| fds.as_raw()), poll.nfds()),
294        Syscall::Ppoll(ppoll) => pollfds(ppoll.fds().map(|fds| fds.as_raw()), ppoll.nfds()),
295        Syscall::Select(select) => fd_sets(
296            select.nfds(),
297            [select.readfds(), select.writefds(), select.exceptfds()]
298                .map(|set| set.map(|set| set.as_raw())),
299        ),
300        Syscall::Pselect6(pselect6) => fd_sets(
301            pselect6.nfds(),
302            [
303                pselect6.readfds(),
304                pselect6.writefds(),
305                pselect6.exceptfds(),
306            ]
307            .map(|set| set.map(|set| set.as_raw())),
308        ),
309        Syscall::Read(read) => is_timerfd(guest, read.fd()),
310        Syscall::Readv(readv) => is_timerfd(guest, readv.fd()),
311        _ => true,
312    }
313}

315/// Before `call` waits in the host, outside Detcore's probes: refuse the run
316/// by name if it can observe a timerfd, the process holds one on virtual
317/// time, and the call may name it (`call_may_name_timerfd`).
318pub(crate) fn refuse_host_wait_for<T, G>(guest: &G, call: Syscall)
319where
320    T: RecordOrReplay,
321    G: Guest<Detcore<T>>,
322{
323    if has_virtual_timerfds(guest)
324        && observes_timerfds(guest, call)
325        && call_may_name_timerfd(guest, call)
326    {
327        refuse(format_args!(
328            "{} would wait in the host on descriptors that may include a timerfd that only Detcore's probes make readable",
329            call.number().name()
330        ));
331    }
332}

334/// Whether a probe of `call` can observe a timerfd's readiness or count, so it
335/// must publish first and its poll deadline includes the next expiration: a
336/// readiness wait, or a read of a timerfd.
337pub(crate) fn observes_timerfds<T, G>(guest: &G, call: Syscall) -> bool
338where
339    T: RecordOrReplay,
340    G: Guest<Detcore<T>>,
341{
342    match call {
343        Syscall::Poll(_)
344        | Syscall::Ppoll(_)
345        | Syscall::Select(_)
346        | Syscall::Pselect6(_)
347        | Syscall::EpollWait(_)
348        | Syscall::EpollPwait(_) => true,
349        Syscall::Other(Sysno::epoll_pwait2, _) => true,
350        Syscall::Read(read) => is_timerfd(guest, read.fd()),
351        Syscall::Readv(readv) => is_timerfd(guest, readv.fd()),
352        _ => false,
353    }
354}

356fn valid(ts: &libc::timespec) -> bool {
357    ts.tv_sec >= 0 && (0..1_000_000_000).contains(&ts.tv_nsec)
358}
359
360fn timespec_ns(ts: &libc::timespec) -> u64 {
361    (ts.tv_sec as u64)
362        .saturating_mul(1_000_000_000)
363        .saturating_add(ts.tv_nsec as u64)
364}
365
366fn itimerspec(remaining_ns: u64, interval_ns: u64) -> libc::itimerspec {
367    let ts = |ns: u64| libc::timespec {
368        tv_sec: (ns / 1_000_000_000) as libc::time_t,
369        tv_nsec: (ns % 1_000_000_000) as libc::c_long,
370    };
371    libc::itimerspec {
372        it_interval: ts(interval_ns),
373        it_value: ts(remaining_ns),
374    }
375}

402/// Guest memory for making an expired timerfd readable, reserved by a caller
403/// that holds the scratch stack across its probes ([`reserve_scratch`]), so
404/// [`publish_expired_in`] need not take the stack itself.
405#[derive(Clone, Copy)]
406pub(crate) struct Scratch {
407    setting: AddrMut<'static, libc::itimerspec>,
408    pollfd: AddrMut<'static, libc::pollfd>,
409}

411/// Reserve a [`Scratch`] on `stack`, before the caller commits it.
412/// Whether this process holds an armed timerfd on Detcore's virtual clock:
413/// exactly when [`publish_expired_in`] can need guest scratch. A caller that
414/// stages its own cells on the stack reserves the scratch only then, so a call
415/// on a process with no armed timerfd keeps every byte of the stack it had
416/// before timerfds were virtualized (a large masked `pselect6` on a tight
417/// stack, `stage_private_cells`). It cannot change during one system call:
418/// the guest runs no code between staging and publication.
419pub(crate) fn holds_armed_virtual_timerfd<T, G>(guest: &G) -> bool
420where
421    T: RecordOrReplay,
422    G: Guest<Detcore<T>>,
423{
424    enabled(guest)
425        && guest
426            .thread_state()
427            .virtual_timerfds()
428            .iter()
429            .any(|(_, detfd)| detfd.virtual_timerfd().and_then(|t| t.deadline).is_some())
430}

432pub(crate) fn reserve_scratch<S: Stack>(stack: &mut S) -> Scratch {
433    Scratch {
434        setting: stack.reserve(),
435        pollfd: stack.reserve(),
436    }
437}

439/// Whether a probe of `call` must publish first and may need guest scratch to
440/// do so: it can observe a timerfd, and one is armed. A caller whose scratch
441/// stack is committed releases it before `publish_expired` when this holds.
442pub(crate) fn may_publish<T, G>(guest: &G, call: Syscall) -> bool
443where
444    T: RecordOrReplay,
445    G: Guest<Detcore<T>>,
446{
447    enabled(guest)
448        && observes_timerfds(guest, call)
449        && guest
450            .thread_state()
451            .virtual_timerfds()
452            .iter()
453            .any(|(_, detfd)| detfd.virtual_timerfd().and_then(|t| t.deadline).is_some())
454}

456/// [`publish_expired`] using `scratch` when the caller holds the scratch
457/// stack, or the stack itself when `scratch` is `None`.
458pub(crate) async fn publish_expired_in<T, G>(
459    guest: &mut G,
460    scratch: Option<Scratch>,
461) -> Option<LogicalTime>
462where
463    T: RecordOrReplay,
464    G: Guest<Detcore<T>>,
465{
466    if !holds_armed_virtual_timerfd(guest) {
467        return None;
468    }
469    let timerfds = guest.thread_state().virtual_timerfds();
470    let now = timerfd_now(guest).await;
471    let mut next: Option<LogicalTime> = None;
472    for (fd, detfd) in timerfds {
473        let Some(mut timer) = detfd.virtual_timerfd() else {
474            continue;
475        };
476        let added = timer.expire_through(now);
477        if added > 0 {
478            make_readable(guest, fd, scratch).await;
479            detfd.set_virtual_timerfd(Some(timer));
480            detlog!(
481                "[dtid {}] timerfd {} expired {} time(s) at virtual time {}, {} unread",
482                guest.thread_state().dettid,
483                fd,
484                added,
485                now,
486                timer.unread,
487            );
488        }
489        if let Some(deadline) = timer.deadline {
490            next = Some(next.map_or(deadline, |earliest| earliest.min(deadline)));
491        }
492    }
493    next
494}

496/// A polling request's deadline: the earlier of the call's own timeout and
497/// the next virtual timerfd expiration, so a backed-off poller is restored
498/// when a timerfd it may watch expires (`Resources::poll_deadline`).
499pub(crate) fn poll_deadline_with<T, G>(
500    guest: &G,
501    timeout: Option<LogicalTime>,
502    next_timerfd: Option<LogicalTime>,
503) -> Option<LogicalTime>
504where
505    T: RecordOrReplay,
506    G: Guest<Detcore<T>>,
507{
508    let deadline = match (timeout, next_timerfd) {
509        (Some(a), Some(b)) => Some(a.min(b)),
510        (a, b) => a.or(b),
511    };
512    poll_deadline_for(guest, deadline)
513}

515/// `publish_expired` before a probe of `call`, if it can observe a timerfd;
516/// returns the poll deadline the probe's request carries.
517pub(crate) async fn publish_before_probe<T, G>(
518    guest: &mut G,
519    call: Syscall,
520    timeout: Option<LogicalTime>,
521) -> Option<LogicalTime>
522where
523    T: RecordOrReplay,
524    G: Guest<Detcore<T>>,
525{
526    if !observes_timerfds(guest, call) {
527        return poll_deadline_for(guest, timeout);
528    }
529    let next = publish_expired(guest).await;
530    poll_deadline_with(guest, timeout, next)
531}

533/// The fault the hook asks of this publication, at most once per process.
534fn take_fault() -> Option<Fault> {
535    use std::sync::atomic::AtomicBool;
536    use std::sync::atomic::Ordering;
537    static FIRED: AtomicBool = AtomicBool::new(false);
538    let fault = match std::env::var(FAULT_ENV).as_deref() {
539        Ok("never-readable") => Fault::NeverReadable,
540        Ok("signal-before-arm") => Fault::SignalBeforeArm,
541        _ => return None,
542    };
543    (!FIRED.swap(true, Ordering::SeqCst)).then_some(fault)
544}

546/// Inject `call`, retrying an interruption (a pending host signal stops an
547/// injection with `ERESTARTSYS` or `EINTR`) until `start + valve`; any other
548/// failure, or interruptions past the limit, refuses the run, naming `what`
549/// and the timerfd `fd`.
550async fn inject_bounded<T, G, S>(
551    guest: &mut G,
552    call: S,
553    start: Instant,
554    valve: Duration,
555    what: &'static str,
556    fd: RawFd,
557) -> i64
558where
559    T: RecordOrReplay,
560    G: Guest<Detcore<T>>,
561    S: SyscallInfo + Copy,
562{
563    loop {
564        match guest.inject(call).await {
565            Ok(value) => return value,
566            Err(Errno::ERESTARTSYS | Errno::EINTR) if start.elapsed() < valve => {}
567            Err(errno) => refuse(format_args!("{what} {fd} failed with {errno}")),
568        }
569    }
570}

572/// Make the host timerfd `fd` readable and wait until the host reports it
573/// readable: arm it to expire at absolute time 1 ns on its own clock, which
574/// has passed, so the host fires it at once. Both injections retry an
575/// interruption, so a host signal stop changes nothing. Any failure, or no
576/// readiness (or only interruptions) within `READABLE_VALVE` of host time from
577/// the start of this publication, refuses the run.
578async fn make_readable<T, G>(guest: &mut G, fd: RawFd, scratch: Option<Scratch>)
579where
580    T: RecordOrReplay,
581    G: Guest<Detcore<T>>,
582{
583    let (setting, pollfd, _guard) = match scratch {
584        Some(scratch) => (scratch.setting, scratch.pollfd, None),
585        None => {
586            let mut stack = guest.stack().await;
587            let scratch = reserve_scratch(&mut stack);
588            let Ok(guard) = stack.commit() else {
589                refuse(format_args!(
590                    "no scratch memory to make expired timerfd {fd} readable"
591                ));
592            };
593            (scratch.setting, scratch.pollfd, Some(guard))
594        }
595    };
596    let fault = take_fault();
597    let never_readable = fault == Some(Fault::NeverReadable);
598    let valve = if never_readable {
599        Duration::from_secs(1)
600    } else {
601        READABLE_VALVE
602    };
603    let start = Instant::now();
604    if !never_readable {
605        if let Err(error) = guest.memory().write_value(setting, &itimerspec(1, 0)) {
606            refuse(format_args!(
607                "could not stage the setting that makes expired timerfd {fd} readable: {error}"
608            ));
609        }
610        let arm = syscalls::TimerfdSettime::new()
611            .with_fd(fd)
612            .with_flags(libc::TFD_TIMER_ABSTIME)
613            .with_new_value(Some(setting.into()))
614            .with_old_value(None);
615        // The guest is a host process only where the backend does not
616        // deliver its signals itself (ptrace); elsewhere its ids name no
617        // host thread, so the hook does nothing.
618        if fault == Some(Fault::SignalBeforeArm)
619            && !guest.config().backend.provides_process_signal_control
620        {
621            // SAFETY: tgkill takes only integer arguments.
622            unsafe {
623                libc::syscall(
624                    libc::SYS_tgkill,
625                    guest.pid().as_raw(),
626                    guest.tid().as_raw(),
627                    libc::SIGCONT,
628                );
629            }
630            if let Err(Errno::ERESTARTSYS | Errno::EINTR) = guest.inject(arm).await {
631                tracing::warn!(
632                    "[detcore, dtid {}] timerfd fault hook: a host signal interrupted arming timerfd {}; retrying",
633                    guest.thread_state().dettid,
634                    fd,
635                );
636            }
637        }
638        inject_bounded(
639            guest,
640            arm,
641            start,
642            valve,
643            "arming, to make its expiration readable, the host timerfd",
644            fd,
645        )
646        .await;
647    }
648    loop {
649        if let Err(error) = guest.memory().write_value(
650            pollfd,
651            &libc::pollfd {
652                fd,
653                events: libc::POLLIN,
654                revents: 0,
655            },
656        ) {
657            refuse(format_args!(
658                "could not stage the readiness probe of timerfd {fd}: {error}"
659            ));
660        }
661        // A 1 ms host wait where the backend honors it (ptrace); KVM probes
662        // without waiting, so this loop then spins until the timer fires.
663        let probe = syscalls::Poll::new()
664            .with_fds(Some(pollfd.cast()))
665            .with_nfds(1)
666            .with_timeout(1);
667        let ready = inject_bounded(
668            guest,
669            probe,
670            start,
671            valve,
672            "probing for its expiration the host timerfd",
673            fd,
674        )
675        .await;
676        if ready > 0 {
677            match guest.memory().read_value(pollfd) {
678                Ok(pollfd) if pollfd.revents & libc::POLLIN != 0 => return,
679                Ok(_) => {}
680                Err(error) => refuse(format_args!(
681                    "could not read the readiness probe of timerfd {fd}: {error}"
682                )),
683            }
684        }
685        if start.elapsed() >= valve {
686            refuse(format_args!(
687                "the host timerfd {fd} did not become readable within {valve:?} of being armed to expire at once"
688            ));
689        }
690    }
691}

693/// `timerfd_settime` on a virtual timerfd, or `None` when Detcore leaves the
694/// call to the host (virtual timerfds off, or `fd` not a timerfd, whose
695/// error the host then reports).
696pub(crate) async fn settime<T, G>(
697    guest: &mut G,
698    call: syscalls::TimerfdSettime,
699) -> Result<Option<i64>, Error>
700where
701    T: RecordOrReplay,
702    G: Guest<Detcore<T>>,
703{
704    let fd = call.fd();
705    if !enabled(guest) || !is_timerfd(guest, fd) {
706        return Ok(None);
707    }
708    let new_ptr = call.new_value().ok_or(Errno::EFAULT)?;
709    let new: libc::itimerspec = guest.memory().read_value(new_ptr)?;
710    if !valid(&new.it_value) || !valid(&new.it_interval) {
711        return Err(Errno::EINVAL.into());
712    }
713    // The host timerfd stays disarmed: the call runs with the guest's flags,
714    // less TFD_TIMER_CANCEL_ON_SET (which would let a host clock step make it
715    // readable at a host-timed moment), and a zero setting, so Linux still
716    // validates the flags and the descriptor, and discards any expiration the
717    // host holds, as a setting change does.
718    {
719        let mut stack = guest.stack().await;
720        let disarm: AddrMut<libc::itimerspec> = stack.reserve();
721        let _guard = stack.commit()?;
722        guest.memory().write_value(disarm, &itimerspec(0, 0))?;
723        guest
724            .inject_with_retry(
725                call.with_flags(call.flags() & !libc::TFD_TIMER_CANCEL_ON_SET)
726                    .with_new_value(Some(disarm.into()))
727                    .with_old_value(None),
728            )
729            .await?;
730    }
731    let now = timerfd_now(guest).await;
732    let detfd = guest.thread_state().with_detfd(fd, |detfd| detfd.clone())?;
733    let mut old = detfd.virtual_timerfd().unwrap_or_default();
734    old.expire_through(now);
735    let (old_remaining_ns, old_interval_ns) = old.remaining(now);
736    let value_ns = timespec_ns(&new.it_value);
737    let interval_ns = timespec_ns(&new.it_interval);
738    let deadline = if value_ns == 0 {
739        None
740    } else if call.flags() & libc::TFD_TIMER_ABSTIME != 0 {
741        Some(LogicalTime::from_nanos(value_ns))
742    } else {
743        Some(LogicalTime::from_nanos(
744            now.as_nanos().saturating_add(value_ns),
745        ))
746    };
747    // Linux keeps the interval of a disarmed timer for timerfd_gettime.
748    detfd.set_virtual_timerfd(Some(VirtualTimerfd::set(deadline, interval_ns)));
749    detlog!(
750        "[dtid {}] timerfd_settime(fd={}, flags={:#x}, interval_ns={}, value_ns={}) armed against virtual clock",
751        guest.thread_state().dettid,
752        fd,
753        call.flags(),
754        interval_ns,
755        value_ns,
756    );
757    // As Linux, the old setting is stored after the new one takes effect.
758    if let Some(old_ptr) = call.old_value() {
759        guest
760            .memory()
761            .write_value(old_ptr, &itimerspec(old_remaining_ns, old_interval_ns))?;
762    }
763    Ok(Some(0))
764}

766/// `timerfd_gettime` on a virtual timerfd, or `None` for the host to answer.
767pub(crate) async fn gettime<T, G>(
768    guest: &mut G,
769    call: syscalls::TimerfdGettime,
770) -> Result<Option<i64>, Error>
771where
772    T: RecordOrReplay,
773    G: Guest<Detcore<T>>,
774{
775    let fd = call.fd();
776    if !enabled(guest) || !is_timerfd(guest, fd) {
777        return Ok(None);
778    }
779    let detfd = guest.thread_state().with_detfd(fd, |detfd| detfd.clone())?;
780    if detfd.virtual_timerfd().is_none() {
781        return Ok(None);
782    }
783    let value_ptr = call.value().ok_or(Errno::EFAULT)?;
784    publish_expired(guest).await;
785    let now = timerfd_now(guest).await;
786    let timer = detfd.virtual_timerfd().unwrap_or_default();
787    let (remaining_ns, interval_ns) = timer.remaining(now);
788    guest
789        .memory()
790        .write_value(value_ptr, &itimerspec(remaining_ns, interval_ns))?;
791    Ok(Some(0))
792}

794/// The count a successful read of the virtual timerfd `fd` returns, which the
795/// read consumes: the expirations counted on the virtual clock, instead of
796/// the host's count of Detcore's own arming. `None` when there is nothing to
797/// replace.
798fn take_unread<T, G>(guest: &mut G, fd: RawFd) -> Option<u64>
799where
800    T: RecordOrReplay,
801    G: Guest<Detcore<T>>,
802{
803    let detfd = guest
804        .thread_state()
805        .with_detfd(fd, |detfd| detfd.clone())
806        .ok()?;
807    let mut timer = detfd.virtual_timerfd()?;
808    if timer.unread == 0 {
809        return None;
810    }
811    let unread = timer.unread;
812    timer.unread = 0;
813    detfd.set_virtual_timerfd(Some(timer));
814    Some(unread)
815}

817/// After a `read` of the timerfd `fd` into `buf` returned `result`.
818pub(crate) fn finish_read<T, G>(
819    guest: &mut G,
820    fd: RawFd,
821    buf: Option<AddrMut<u8>>,
822    result: Result<i64, Error>,
823) -> Result<i64, Error>
824where
825    T: RecordOrReplay,
826    G: Guest<Detcore<T>>,
827{
828    if !enabled(guest) || !matches!(result, Ok(8)) {
829        return result;
830    }
831    let Some(buf) = buf else {
832        return result;
833    };
834    if let Some(unread) = take_unread(guest, fd) {
835        guest.memory().write_value(buf.cast::<u64>(), &unread)?;
836    }
837    result
838}

840/// After a `readv` of the timerfd `fd` with the `iovcnt` iovecs at `iov`
841/// returned `result`. Linux copies the 8-byte count through the iovecs in
842/// order (`read_iter`), so the virtual count is scattered the same way.
843pub(crate) fn finish_readv<T, G>(
844    guest: &mut G,
845    fd: RawFd,
846    iov: Option<syscalls::Addr<libc::iovec>>,
847    iovcnt: usize,
848    result: Result<i64, Error>,
849) -> Result<i64, Error>
850where
851    T: RecordOrReplay,
852    G: Guest<Detcore<T>>,
853{
854    if !enabled(guest) || !matches!(result, Ok(8)) {
855        return result;
856    }
857    let Some(iov) = iov else {
858        return result;
859    };
860    let Some(unread) = take_unread(guest, fd) else {
861        return result;
862    };
863    let bytes = unread.to_ne_bytes();
864    let mut written = 0;
865    for index in 0..iovcnt {
866        if written == bytes.len() {
867            break;
868        }
869        let Some(slot) = syscalls::Addr::<libc::iovec>::from_raw(
870            iov.as_raw() + index * std::mem::size_of::<libc::iovec>(),
871        ) else {
872            break;
873        };
874        let entry: libc::iovec = guest.memory().read_value(slot)?;
875        let take = entry.iov_len.min(bytes.len() - written);
876        if take == 0 {
877            continue;
878        }
879        let Some(base) = AddrMut::<u8>::from_raw(entry.iov_base as usize) else {
880            break;
881        };
882        guest
883            .memory()
884            .write_exact(base, &bytes[written..written + take])?;
885        written += take;
886    }
887    result
888}

897    /// Detcore runs outside the guest on the ptrace family and KVM, not on

910            (
911                "liteinst in-guest",
912                BackendCapabilities::LITEINST_IN_GUEST,
913                false,
914                true,
915            ),
916        ] {
917            assert_eq!(detcore_runs_outside_the_guest(&backend), outside, "{name}");
