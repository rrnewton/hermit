28/// Inject the syscall that `call` makes for a buffer of `N` bytes holding
29/// `initial`, and return its result with what the buffer holds after it.
30///
31/// The buffer is staged as `Detcore::inject_fstat` stages its own. It goes
32/// first in the guest stack scratch, which on the ptrace backend lies below
33/// the red zone under the guest's stack pointer; the bytes it covers are
34/// saved and put back after the call. That memory is not always writable: a
35/// guest may run with its stack pointer just above a guard page (a thread,
36/// fiber or alternate signal stack), or within a few hundred bytes of the
37/// lowest page of the main-thread stack, which a tracer's write does not
38/// grow (<https://github.com/rrnewton/hermit/issues/3328>). The fault then
39/// belongs to Detcore's bookkeeping, not to the guest, so where the scratch
40/// cannot hold the buffer, and where the call fails with EFAULT there, the
41/// call is repeated with the buffer in a transient private page that is
42/// unmapped before the guest resumes.
43pub(crate) async fn inject_with_scratch<G, T, const N: usize>(
44    guest: &mut G,
45    initial: [u8; N],
46    call: impl Fn(AddrMut<'_, u8>) -> Syscall,
47) -> Result<(i64, [u8; N]), Errno>
48where
49    G: Guest<Detcore<T>>,
50    T: RecordOrReplay,
51{
52    match on_stack(guest, &initial, &call).await {
53        Err(Errno::EFAULT) => {
54            info!(
55                "Guest stack scratch cannot hold an injected syscall's {N}-byte buffer; using a \
56                 transient page instead."
57            );
58            in_transient_page(guest, &initial, &call).await
59        }
60        result => result,
61    }
62}

64/// [`inject_with_scratch`] with the buffer in the guest stack scratch.
65/// `Err(EFAULT)` when the scratch cannot hold it.
66async fn on_stack<G, T, const N: usize>(
67    guest: &mut G,
68    initial: &[u8; N],
69    call: &impl Fn(AddrMut<'_, u8>) -> Syscall,
70) -> Result<(i64, [u8; N]), Errno>
71where
72    G: Guest<Detcore<T>>,
73    T: RecordOrReplay,
74{
75    let mut stack = guest.stack().await;
76    let buffer = stack.reserve::<[u8; N]>().cast::<u8>();
77    let mut saved = [0_u8; N];
78    guest.memory().read_exact(buffer, &mut saved)?;
79    let guard = match stack.commit() {
80        Ok(guard) => guard,
81        Err(errno) => {
82            let _ = guest.memory().write_exact(buffer, &saved);
83            return Err(errno);
84        }
85    };
86    let outcome = run(guest, buffer, initial, call).await;
87    let restored = guest.memory().write_exact(buffer, &saved);
88    drop(guard);
89    let outcome = outcome?;
90    restored?;
91    Ok(outcome)
92}

94/// [`inject_with_scratch`] with the buffer in a transient private page.
95async fn in_transient_page<G, T, const N: usize>(
96    guest: &mut G,
97    initial: &[u8; N],
98    call: &impl Fn(AddrMut<'_, u8>) -> Syscall,
99) -> Result<(i64, [u8; N]), Errno>
100where
101    G: Guest<Detcore<T>>,
102    T: RecordOrReplay,
103{
104    let mapped = guest
105        .inject_with_retry(Syscall::Mmap(
106            syscalls::Mmap::new()
107                .with_addr(None)
108                .with_len(N)
109                .with_prot(ProtFlags::PROT_READ | ProtFlags::PROT_WRITE)
110                .with_flags(MapFlags::MAP_PRIVATE | MapFlags::MAP_ANONYMOUS)
111                .with_fd(-1)
112                .with_offset(0),
113        ))
114        .await?;
115    let page = usize::try_from(mapped)
116        .ok()
117        .and_then(AddrMut::<u8>::from_raw)
118        .unwrap_or_else(|| panic!("transient scratch page mmap returned {mapped}"));
119    let outcome = run(guest, page, initial, call).await;
120    if let Err(errno) = guest
121        .inject_with_retry(Syscall::Munmap(
122            syscalls::Munmap::new()
123                .with_addr(Some(page.cast::<libc::c_void>().into()))
124                .with_len(N),
125        ))
126        .await
127    {
128        // Not expected: the page was mapped by this call and its address
129        // never reached the guest.
130        warn!("[detcore] could not unmap the transient scratch page: {errno}");
131    }
132    outcome
133}

135/// Write `initial` at `buffer`, inject `call(buffer)`, and read the buffer
136/// back.
137async fn run<G, T, const N: usize>(
138    guest: &mut G,
139    buffer: AddrMut<'_, u8>,
140    initial: &[u8; N],
141    call: &impl Fn(AddrMut<'_, u8>) -> Syscall,
142) -> Result<(i64, [u8; N]), Errno>
143where
144    G: Guest<Detcore<T>>,
145    T: RecordOrReplay,
146{
147    guest.memory().write_exact(buffer, initial)?;
148    let result = guest.inject_with_retry(call(buffer)).await?;
149    let mut bytes = [0_u8; N];
150    guest.memory().read_exact(buffer, &mut bytes)?;
151    Ok((result, bytes))
152}
