9//! The caller's own `exe`, `cwd` and `root` links (`/proc/self/exe`,
10//! `/proc/thread-self/cwd`, `/proc/<own pid>/root`, ...), answered by Detcore
11//! from its own knowledge wherever that answer is exactly Linux's (package K
12//! slice E1): `cwd` from an injected `getcwd` of the calling thread and
13//! `root` as `/` on every backend, and `exe`, only where the backend's own
14//! procfs is not the guest's kernel procfs (KVM), from the exec record
15//! Detcore keeps for the process ([`crate::exec_record`]) while it still
16//! names exactly the image the process executes. No backend's `/proc` is
17//! read for them, so on KVM the executor's readlink answers for these links
18//! are reached only where Detcore cannot render one exactly.
19//!
20//! Linux (v6.17) answers each from the task the spelling names
21//! (`proc_exe_link`, `proc_cwd_link`, `proc_root_link`, fs/proc/base.c), as
22//! `d_path` renders the path under the reader's root
23//! (`do_proc_readlink`, fs/proc/base.c; `d_path`, fs/d_path.c): the image's
24//! path with " (deleted)" once it is unlinked, the working directory's path,
25//! and `/` for the caller's own root. A task without an address space or
26//! filesystem information (an exited thread-group leader) has no link
27//! (`ENOENT`). `readlink` copies at most its buffer's size, without a NUL.
28//!
29//! Elsewhere the call takes the path it took before this slice, so that every
30//! backend answers as it did: the kernel on ptrace, LiteInst and
31//! in-guest-trap, the launcher runs' own `exe` handling on DBT and SaBRe
32//! (`Detcore::launched_exe`), and the executor's existing readlink on KVM.
33//! That is: `exe` on a backend whose procfs is the guest's kernel procfs (the
34//! kernel names the image's dentry exactly, which no record can prove), and
35//! on KVM an image without a record or no longer exactly at its recorded
36//! path; a removed working directory, which `getcwd` refuses with `ENOENT`;
37//! a leader or another thread whose filesystem information the caller does
38//! not share; and a spelling that looks a name up past the link (a `.` after
39//! it), where Linux checks search permission (on KVM, see below).
40//!
41//! The backend split keys on
42//! [`crate::procfs_inode::tool_can_name_guest_descriptors`], which stands in
43//! for two filesystem facts no provider reports yet (the backend's procfs is
44//! the guest's kernel procfs; the tool's filesystem view is the guest's):
45//! <https://github.com/rrnewton/hermit/issues/4700>.
46//!
47//! Where the backend's procfs is not the guest's kernel procfs (KVM), so that
48//! its own readlink answer is not the kernel's, Detcore reads the caller's
49//! own `fd/<n>`, `exe` and `cwd` links from the backend's `link_path` fact
50//! (`reverie::Guest::link_path`, package K slice F2-h) before anything else:
51//! the name the backend's own records give the descriptor's file, the
52//! process's image and the thread's working directory, rendered as `d_path`
53//! renders them (a path, " (deleted)" once it is unlinked, or an object's
54//! name such as `pipe:[<inode>]` and `anon_inode:[eventfd]`). A pipe's and a
55//! socket's inode number, and a placeholder's name, are then Detcore's, as
56//! for every descriptor link. Where the backend does not provide the fact
57//! for a link, Detcore answers without the backend's `/proc` (package K
58//! slice F2-d0), which is not the guest's there:
59//! - an object of the backend's own (the KVM executor's rendering of the
60//!   process's `status` and `maps`) is the procfs file Detcore recorded at
61//!   the descriptor's open (`ProcfsFile::link_name`), and is refused by name
62//!   where it recorded none;
63//! - another thread's working or root directory is refused by name: the fact
64//!   names the calling thread's alone, and nothing Detcore keeps says whether
65//!   the other thread shares the caller's;
66//! - a `.` after a link is looked up from what the link leads to: from the
67//!   descriptor (`readlinkat(fd, ".")`), the working directory (`.`) or the
68//!   root (`/.`), each an injected lookup whose search permission the kernel
69//!   beneath checks, and `ENOTDIR` past the image.
70//!
71//! Another process's links still take the sources above and then the
72//! backend's readlink, as before.

94/// The longest path `d_path` renders into `do_proc_readlink`'s buffer, and

98/// Which of a task's three path links.
99#[derive(Clone, Copy, Debug, PartialEq, Eq)]
100pub(crate) enum OwnLink {
101    /// `exe`: the image the task's address space executes.
102    Exe,
103    /// `cwd`: the task's working directory.
104    Cwd,
105    /// `root`: the task's root directory.
106    Root,
107}

109impl OwnLink {
110    fn name(self) -> &'static str {
111        match self {
112            OwnLink::Exe => "exe",
113            OwnLink::Cwd => "cwd",
114            OwnLink::Root => "root",
115        }
116    }
117}

119/// The task a spelling of an own link names, before Detcore checks the
120/// numbers it does not know to be the caller's against its process table.
121#[derive(Clone, Copy, Debug, PartialEq, Eq)]
122pub(crate) enum NamedTask {
123    /// The caller itself: `thread-self`, or its own tid below its process.
124    Caller,
125    /// The caller's thread-group leader: `self` or the caller's pid.
126    Leader,
127    /// `/proc/<self or own pid>/task/<t>` for another thread `t`: one of
128    /// the caller's process if Detcore lists it so, else no such entry.
129    GroupThread(i32),
130    /// `/proc/<n>` for a number `n` other than the caller's pid: a thread of
131    /// the caller's process if Detcore lists it so, else another process.
132    Numbered(i32),
133    /// `/proc/<n>/task/<t>` for a number `n` other than the caller's pid.
134    NumberedThread(i32, i32),
135}

137/// How a spelling of an own link goes on past the link's name.
138#[derive(Clone, Copy, Debug, PartialEq, Eq)]
139pub(crate) enum Past {
140    /// It ends at the link: the link is read.
141    Nothing,
142    /// Only slashes follow: the link is followed, and what it leads to is
143    /// read, with no further lookup.
144    Slash,
145    /// A `.` follows: a lookup in what the link leads to, which Linux
146    /// checks for search permission there.
147    Walk,
148}

150/// A spelling of one of the caller's own path links ([`own_task_link`]).
151#[derive(Clone, Copy, Debug, PartialEq, Eq)]
152pub(crate) struct OwnTaskLink {
153    pub(crate) link: OwnLink,
154    pub(crate) task: NamedTask,
155    /// What follows the link's name.
156    pub(crate) past: Past,
157}

159/// Which own path link the absolute spelling `path` names, for a caller whose
160/// pid and tid as the guest numbers them are `pid` and `tid`:
161/// `/proc/<p>/{exe,cwd,root}` and `/proc/<p>/task/<t>/{exe,cwd,root}`, where
162/// `p` is `self` or a number and `t` a number, and
163/// `/proc/thread-self/{exe,cwd,root}`. Empty and `.` components are skipped,
164/// as Linux's walk skips them (`//proc/./self/exe` too). `None` for any
165/// other path and for a spelling with `..`. Numbers follow procfs's names:
166/// decimal, no leading zero.
167pub(crate) fn own_task_link(path: &[u8], pid: i32, tid: i32) -> Option<OwnTaskLink> {
168    let rest = path.strip_prefix(b"/")?;
169    let mut parts = Vec::new();
170    let mut past = Past::Nothing;
171    for part in rest.split(|byte| *byte == b'/') {
172        match part {
173            b"" if !parts.is_empty() && past == Past::Nothing => past = Past::Slash,
174            b"" => {}
175            b"." if !parts.is_empty() => past = Past::Walk,
176            b"." => {}
177            b".." => return None,
178            part => {
179                parts.push(part);
180                past = Past::Nothing;
181            }
182        }
183    }
184    let (first, parts) = parts.split_first()?;
185    if *first != b"proc" {
186        return None;
187    }
188    let number = |part: &[u8]| -> Option<i32> {
189        let digits = std::str::from_utf8(part).ok()?;
190        (!digits.is_empty()
191            && !digits.starts_with('0')
192            && digits.bytes().all(|byte| byte.is_ascii_digit()))
193        .then(|| digits.parse().ok())?
194    };
195    let link = |part: &[u8]| match part {
196        b"exe" => Some(OwnLink::Exe),
197        b"cwd" => Some(OwnLink::Cwd),
198        b"root" => Some(OwnLink::Root),
199        _ => None,
200    };
201    // `Some(None)`: the caller's own process directory; `Some(Some(n))`: the
202    // directory of another number.
203    let process = |part: &[u8]| -> Option<Option<i32>> {
204        if part == b"self" {
205            return Some(None);
206        }
207        let number = number(part)?;
208        Some((number != pid).then_some(number))
209    };
210    let thread = |own: Option<i32>, thread: i32| match own {
211        None if thread == tid => NamedTask::Caller,
212        None if thread == pid => NamedTask::Leader,
213        None => NamedTask::GroupThread(thread),
214        Some(process) => NamedTask::NumberedThread(process, thread),
215    };
216    let (task, name) = match parts {
217        [b"thread-self", name] => (NamedTask::Caller, name),
218        [subject, name] => match process(subject)? {
219            None => (NamedTask::Leader, name),
220            Some(other) => (NamedTask::Numbered(other), name),
221        },
222        [subject, b"task", named, name] => (thread(process(subject)?, number(named)?), name),
223        _ => return None,
224    };
225    Some(OwnTaskLink {
226        link: link(name)?,
227        task,
228        past,
229    })
230}

232/// How the absolute spelling `path` of a link goes on past the link's name,
233/// from its last bytes: only slashes ([`Past::Slash`]), a `.` component among
234/// them ([`Past::Walk`]), or nothing ([`Past::Nothing`]).
235pub(crate) fn past_the_link(path: &[u8]) -> Past {
236    let mut rest = path;
237    let mut past = Past::Nothing;
238    loop {
239        if let Some(before) = rest.strip_suffix(b"/") {
240            rest = before;
241            if past == Past::Nothing {
242                past = Past::Slash;
243            }
244        } else if let Some(before) = rest.strip_suffix(b"/.") {
245            rest = before;
246            past = Past::Walk;
247        } else {
248            return past;
249        }
250    }
251}

253/// The task an own link describes once Detcore checked its numbers.
254#[derive(Clone, Copy, Debug, PartialEq, Eq)]
255enum DescribedTask {
256    /// The calling thread.
257    Caller,
258    /// The caller's live thread-group leader, another thread.
259    Leader,
260    /// Another live thread of the caller's process.
261    Other,
262}
263
264/// What Detcore makes of an own link's spelling ([`Detcore::own_link_task`]).
265enum Resolved {
266    /// A task of the caller's process with this link.
267    Described(DescribedTask),
268    /// Linux finds no such link (`ENOENT`): a thread outside the process
269    /// below its `task` directory, or a task without an address space or
270    /// filesystem information (an exited leader).
271    Missing,
272    /// Not the caller's process: another process's link, as before.
273    Elsewhere,
274}

276/// The target `readlink` of the working-directory link shows for the path
277/// `getcwd` answered, or `None` where it cannot be rendered from it: a path
278/// that does not start with `/` (`getcwd`'s "(unreachable)" prefix for a
279/// directory outside the caller's root, where `d_path` shows the path from
280/// the filesystem's root instead).
281pub(crate) fn cwd_link_target(getcwd: &[u8]) -> Option<&[u8]> {
282    getcwd.starts_with(b"/").then_some(getcwd)
283}
284
285/// What `d_path` makes of a rendered `target` in `do_proc_readlink`'s
286/// `PATH_MAX` buffer: the target, or `ENAMETOOLONG` where it and its NUL do
287/// not fit.
288pub(crate) fn fits_d_path(target: &[u8]) -> Result<&[u8], Errno> {
289    if target.len() >= PATH_MAX {
290        Err(Errno::ENAMETOOLONG)
291    } else {
292        Ok(target)
293    }
294}

296impl<T: RecordOrReplay> Detcore<T> {
297    /// Answer `readlink` of one of the caller's own `exe`, `cwd` and `root`
298    /// links at the guest's absolute spelling `path` ([`own_task_link`]) from
299    /// Detcore's own knowledge, into the guest's `buffer` of `buffer_len`
300    /// bytes. `Ok(None)` leaves the call to the path it took before: for
301    /// any other path, under record and replay, in a nested PID namespace or
302    /// after a guest `chroot` (where `/proc`, the caller's numbers or its root
303    /// are not Detcore's), and where Detcore cannot render the link exactly
304    /// (the module's documentation).
305    pub(crate) async fn readlink_own_task_link<G: Guest<Self>>(
306        &self,
307        guest: &mut G,
308        sysno: Sysno,
309        path: &Path,
310        (buffer, buffer_len): (Option<AddrMut<'_, libc::c_char>>, usize),
311    ) -> Result<Option<i64>, Error> {
312        let (pid, tid) = {
313            let state = guest.thread_state();
314            if self.cfg.recordreplay_modes
315                || state.nested_pid_namespace
316                || state.changed_root.load(std::sync::atomic::Ordering::SeqCst)
317                || path.as_os_str().len() >= PATH_MAX
318            {
319                return Ok(None);
320            }
321            let Some(detpid) = state.detpid else {
322                return Ok(None);
323            };
324            (detpid.as_raw(), state.dettid.as_raw())
325        };
326        let Some(spelled) = own_task_link(path.as_os_str().as_bytes(), pid, tid) else {
327            return Ok(None);
328        };
329        // Where the backend's procfs is the guest's kernel procfs, the
330        // kernel names the image exactly, as Detcore's record cannot prove
331        // it would (the image's dentry may have moved while its inode stayed
332        // at the path): its answer stands, as before. A lookup past the link
333        // (a `.` after it) is left to that kernel too, which checks search
334        // permission where Detcore would not; where Detcore reads the links
335        // from the backend's facts it makes that lookup itself
336        // ([`Self::lookup_past_task_link`]).
337        if (spelled.link == OwnLink::Exe
338            && crate::procfs_inode::tool_can_name_guest_descriptors(&self.cfg.backend))
339            || (spelled.past == Past::Walk && !self.reads_link_facts())
340        {
341            return Ok(None);
342        }
343        // Linux refuses a size that is not positive before it looks the path up.
344        let size = super::readlink_size(buffer_len)?;
345        let described = match self.own_link_task(guest, spelled.task, (pid, tid)).await {
346            Resolved::Described(described) => described,
347            Resolved::Missing => return Err(Errno::ENOENT.into()),
348            Resolved::Elsewhere => return Ok(None),
349        };
350        if spelled.past == Past::Slash {
351            // The link followed, with no lookup past it: the image (a file)
352            // or a directory, which is no link.
353            return Err(match spelled.link {
354                OwnLink::Exe => Errno::ENOTDIR,
355                OwnLink::Cwd | OwnLink::Root => Errno::EINVAL,
356            }
357            .into());
358        }
359        // Another thread's working or root directory, where Detcore reads the
360        // links from the backend's facts: the fact names the calling thread's
361        // alone, and nothing Detcore keeps says whether the other thread
362        // shares the caller's (`CLONE_FS`). Linux gives that thread's
363        // directory; refused by name rather than left to the backend, whose
364        // own `/proc` is not the guest's.
365        if matches!(spelled.link, OwnLink::Cwd | OwnLink::Root)
366            && !self.shares_filesystem(guest, described)
367            && self.reads_link_facts()
368        {
369            return self
370                .refuse_unrendered_link(guest, sysno, &other_thread_directory_reason(path))
371                .await
372                .map(Some);
373        }
374        if spelled.past == Past::Walk {
375            return self
376                .lookup_past_task_link(guest, spelled.link, (buffer, size))
377                .await
378                .map(Some);
379        }
380        let target = match spelled.link {
381            // The process's image is one for all its threads: the caller's
382            // fact names it.
383            OwnLink::Exe => match self.own_link_fact(guest, PathLink::Executable)? {
384                Some(target) => Some(target),
385                None => self.own_exe_target(guest),
386            },
387            OwnLink::Cwd | OwnLink::Root if !self.shares_filesystem(guest, described) => None,
388            // The caller's working directory is the one the spelling names
389            // (`shares_filesystem`).
390            OwnLink::Cwd => match self.own_link_fact(guest, PathLink::WorkingDirectory)? {
391                Some(target) => Some(target),
392                None => match self.injected_working_directory(guest, buffer, size).await {
393                    Ok(directory) => cwd_link_target(&directory).map(<[u8]>::to_vec),
394                    // A removed working directory (`getcwd` refuses it, and
395                    // only its name, which Detcore does not know, would
396                    // render it), and any other failure: answered as before.
397                    Err(_) => None,
398                },
399            },
400            OwnLink::Root => Some(b"/".to_vec()),
401        };
402        // Where Detcore cannot render the link exactly, the call takes its
403        // previous path (the module's documentation).
404        let Some(target) = target else {
405            tracing::debug!(
406                "[detcore, tid {}] {sysno} of the caller's own {} link answered as before",
407                guest.tid(),
408                spelled.link.name()
409            );
410            return Ok(None);
411        };
412        let target = fits_d_path(&target)?;
413        let copied = &target[..target.len().min(size)];
414        let address = buffer.ok_or(Errno::EFAULT)?;
415        copy_link_out(guest, address.cast::<u8>(), copied)
416    }
417
418    /// Answer `readlink` of the caller's own descriptor link at the absolute
419    /// spelling `spelled` from the backend's `link_path` fact, where Detcore
420    /// reads it ([`Self::reads_link_facts`]), into the guest's `buffer` of
421    /// `buffer_len` bytes. `spelled` is the spelling the walk resolved (each
422    /// `..` within the caller's own task tree resolved, the components
423    /// before `proc` that name nothing dropped); `guest_bytes` the guest's own
424    /// spelling, whose last bytes say whether the link is followed
425    /// ([`past_the_link`]).
426    ///
427    /// The spellings are those [`super::own_descriptor_link`] names: the
428    /// process's table (`/proc/self/fd/N`, `/proc/<pid>/fd/N`, and
429    /// `/dev/fd/N` where `/dev/fd` is the link to `/proc/self/fd` it is on
430    /// Linux) and the calling thread's (`/proc/thread-self/fd/N`,
431    /// `/proc/<pid>/task/<tid>/fd/N`). Linux (v6.17) looks the descriptor
432    /// up in that task's table (`proc_lookupfd_common`, fs/proc/fd.c), so a
433    /// descriptor it does not hold is `ENOENT`; and renders the file's path
434    /// with `d_path` into `do_proc_readlink`'s buffer, copied without a NUL
435    /// and cut to the buffer's size (fs/proc/base.c). A spelling that follows
436    /// the link with a final `/` (`lookup_last`, fs/namei.c) needs a
437    /// directory there: `ENOTDIR` for any other file, and `EINVAL` for a
438    /// directory, which is no link (`do_readlinkat`, fs/stat.c).
439    ///
440    /// A spelling with a `.` after the link looks `.` up in the
441    /// descriptor's file, as Linux's walk does past the link it follows
442    /// (`link_path_walk`, fs/namei.c): Detcore issues that lookup from the
443    /// descriptor itself (`readlinkat(fd, ".")`), so the kernel beneath
444    /// answers `ENOTDIR` for a file that is not a directory, its search
445    /// permission check for a directory, and then `EINVAL`, which a
446    /// directory is (no link). A link the backend does not name (an object
447    /// it keeps itself) is the procfs file Detcore recorded at the
448    /// descriptor's open (`DetFd::procfs_link_name`), and is refused by name
449    /// where Detcore recorded none.
450    ///
451    /// `Ok(None)` leaves the call to the path it took before: the process's
452    /// table read by a thread that does not hold it, `/dev/fd` that is not
453    /// that link, and a backend that cannot write with the guest's
454    /// permissions.
455    pub(crate) async fn readlink_own_descriptor_fact<G: Guest<Self>>(
456        &self,
457        guest: &mut G,
458        sysno: Sysno,
459        (spelled, guest_bytes): (&[u8], &[u8]),
460        buffer: Option<AddrMut<'_, libc::c_char>>,
461        buffer_len: usize,
462    ) -> Result<Option<i64>, Error> {
463        if !self.reads_link_facts() {
464            return Ok(None);
465        }
466        let Some(own) = self.procfs_caller_ids(guest) else {
467            return Ok(None);
468        };
469        let Some(descriptor) =
470            super::own_descriptor_link(Path::new(std::ffi::OsStr::from_bytes(spelled)), Some(own))
471        else {
472            return Ok(None);
473        };
474        let past = past_the_link(guest_bytes);
475        if descriptor.table == super::DescriptorTable::Process {
476            let state = guest.thread_state();
477            if state.dettid.as_raw() != own.0 && !state.holds_process_files() {
478                return Ok(None);
479            }
480        }
481        if spelled.starts_with(b"/dev/")
482            && self
483                .injected_link_target(guest, b"/dev/fd")
484                .await
485                .as_deref()
486                != Some(Path::new("/proc/self/fd"))
487        {
488            return Ok(None);
489        }
490        // Linux refuses a size that is not positive before it looks the path up.
491        let size = super::readlink_size(buffer_len)?;
492        // `ENOENT` for a descriptor the table does not hold, before anything
493        // past the link is looked up.
494        let text = self.own_link_fact(guest, PathLink::Descriptor(descriptor.fd))?;
495        match past {
496            Past::Nothing => {}
497            // A `.` after the link: `.` looked up in the descriptor's file,
498            // as Linux's walk looks it up past the link it follows.
499            // A directory Detcore serves on a placeholder (package K slice
500            // K10) is no file of the kernel beneath to look `.` up in: the
501            // caller decides the spelling from Detcore's model
502            // (`served_directory_walk`).
503            Past::Walk => {
504                if self
505                    .served_directory_walk(guest, guest_bytes)
506                    .await
507                    .is_some()
508                {
509                    return Ok(None);
510                }
511                return self
512                    .inject_dot_lookup(guest, descriptor.fd, b".", (buffer, size))
513                    .await
514                    .map(Some);
515            }
516            Past::Slash => {
517                let stat = self.inject_fstat(guest, descriptor.fd).await?;
518                return Err(if stat.st_mode & libc::S_IFMT == libc::S_IFDIR {
519                    Errno::EINVAL
520                } else {
521                    Errno::ENOTDIR
522                }
523                .into());
524            }
525        }
526        let target = match text {
527            Some(text) => {
528                self.descriptor_link_target(guest, descriptor.fd, text)
529                    .await?
530            }
531            // An object the backend keeps itself and does not name (the KVM
532            // executor's own rendering of a process's `status` or `maps`):
533            // the procfs file Detcore recorded at the descriptor's open, or a
534            // refusal by name where it recorded none (a descriptor received
535            // over `SCM_RIGHTS`, the captured output's own descriptor).
536            None => match guest
537                .thread_state()
538                .with_detfd(descriptor.fd, |detfd| detfd.procfs_link_name())
539                .ok()
540                .flatten()
541            {
542                Some(name) => name.into_os_string().into_vec(),
543                None => {
544                    let reason = unnamed_descriptor_reason(spelled, descriptor.fd);
545                    return self
546                        .refuse_unrendered_link(guest, sysno, &reason)
547                        .await
548                        .map(Some);
549                }
550            },
551        };
552        let copied = &target[..target.len().min(size)];
553        let address = buffer.ok_or(Errno::EFAULT)?;
554        copy_link_out(guest, address.cast::<u8>(), copied)
555    }

557    /// The target the guest sees for the caller's descriptor `fd`, whose
558    /// link the backend names `text`: a placeholder's link, confirmed by the
559    /// kernel identity of the descriptor itself (an injected `fstat` of `fd`,
560    /// `placeholder_identity`), names the procfs file it stands for, as
561    /// Linux's link names the file, by every spelling of the link alike; a
562    /// pipe's or a socket's
563    /// (`pipe:[<inode>]`, `socket:[<inode>]`, the names `pipefs_dname` and
564    /// `sockfs_dname` give an object with no path) names the inode number
565    /// Detcore gives the descriptor's file (an inherited standard stream's,
566    /// or the pool's for its host identity), as its `fstat` does; any other
567    /// is the backend's text, a path (a named FIFO's included) or an
568    /// object's name.
569    async fn descriptor_link_target<G: Guest<Self>>(
570        &self,
571        guest: &mut G,
572        fd: i32,
573        text: Vec<u8>,
574    ) -> Result<Vec<u8>, Error> {
575        // Confirmed through the descriptor, not the guest's spelling: the
576        // calling thread's spelling, `/proc/<pid>/task/<tid>/fd/N`, names
577        // nothing the KVM executor's `/proc` looks up
578        // (`fd_directory::leader_process_spelling` respells a file of the
579        // leader's directory, not a path below its `fd`), so a stat of it
580        // failed and the placeholder's own name reached the guest. Every
581        // rendered file had that defect; `status`, rendered on KVM since
582        // package K item 2b slice 2b-2, is the one
583        // run_kvm_the_callers_own_link_targets_answer_as_linux reads so.
584        if let Some(path) = self.placeholder_link_within_level(guest, &text)
585            && (self.cfg.backend.tool_shares_guest_descriptor_table
586                || self.inject_fstat(guest, fd).await.is_ok_and(|stat| {
587                    crate::syscalls::files::rendered_procfs::placeholder_identity(
588                        stat.st_dev,
589                        stat.st_ino,
590                    )
591                    .is_some()
592                }))
593        {
594            return Ok(path.as_os_str().as_bytes().to_vec());
595        }
596        let Some(identity) = super::anonymous_proc_fd_identity(&text) else {
597            return Ok(text);
598        };
599        let standard = guest
600            .thread_state()
601            .with_detfd(fd, |detfd| {
602                crate::syscalls::deterministic_stdio_inode_for_resource(fd, detfd.resource())
603            })
604            .ok()
605            .flatten();
606        let inode = match standard {
607            Some(inode) => inode,
608            None => {
609                let stat = self.inject_fstat(guest, fd).await?;
610                crate::tool_global::determinize_inode(
611                    guest,
612                    crate::types::RawInode::new(stat.st_dev, stat.st_ino),
613                )
614                .await
615                .0
616            }
617        };
618        Ok(format!("{}:[{inode}]", identity.kind).into_bytes())
619    }

621    /// `readlink` of the caller's own `link` followed by a `.`, where Detcore
622    /// reads the links from the backend's facts, into the guest's `buffer`
623    /// of `size` bytes. Linux follows the link and looks `.` up in what it
624    /// leads to (`link_path_walk`, fs/namei.c): a file that is not a
625    /// directory, the image, is `ENOTDIR`; a directory is searched
626    /// (`may_lookup`'s `MAY_EXEC`, `EACCES` where that is denied), and is
627    /// then no link (`EINVAL`, `do_readlinkat`, fs/stat.c). The working
628    /// directory's and the root's lookup is the same lookup made from the
629    /// caller's own working directory (`.`) and root (`/.`), which Detcore
630    /// issues, so the kernel beneath decides the permission.
631    async fn lookup_past_task_link<G: Guest<Self>>(
632        &self,
633        guest: &mut G,
634        link: OwnLink,
635        (buffer, size): (Option<AddrMut<'_, libc::c_char>>, usize),
636    ) -> Result<i64, Error> {
637        match link {
638            OwnLink::Exe => Err(Errno::ENOTDIR.into()),
639            OwnLink::Cwd => {
640                self.inject_dot_lookup(guest, libc::AT_FDCWD, b".", (buffer, size))
641                    .await
642            }
643            OwnLink::Root => {
644                self.inject_dot_lookup(guest, libc::AT_FDCWD, b"/.", (buffer, size))
645                    .await
646            }
647        }
648    }

650    /// Inject `readlinkat(dirfd, name, buffer, size)` for a `name` that ends
651    /// in `.` (a directory, never a link), from the guest's stack, and give
652    /// its answer: the kernel beneath's walk to `name` from `dirfd`.
653    async fn inject_dot_lookup<G: Guest<Self>>(
654        &self,
655        guest: &mut G,
656        dirfd: i32,
657        name: &[u8],
658        (buffer, size): (Option<AddrMut<'_, libc::c_char>>, usize),
659    ) -> Result<i64, Error> {
660        let mut path_buffer = [0_u8; 8];
661        path_buffer[..name.len()].copy_from_slice(name);
662        let mut stack = guest.stack().await;
663        let path_address = stack.push(path_buffer).cast::<libc::c_char>();
664        let stack_guard = stack.commit()?;
665        let result = guest
666            .inject_with_retry(
667                syscalls::Readlinkat::new()
668                    .with_dirfd(dirfd)
669                    .with_path(PathPtr::from_ptr(
670                        path_address.as_raw() as *const libc::c_char
671                    ))
672                    .with_buf(buffer)
673                    .with_buf_len(size),
674            )
675            .await;
676        drop(stack_guard);
677        Ok(result?)
678    }

680    /// Refuse a `readlink` of one of the caller's own links that Detcore
681    /// cannot render where the backend's own `/proc` is not the guest's, by
682    /// name, through the unsupported-operation policy (`EACCES` where the
683    /// run allows unsupported operations).
684    async fn refuse_unrendered_link<G: Guest<Self>>(
685        &self,
686        guest: &mut G,
687        sysno: Sysno,
688        reason: &str,
689    ) -> Result<i64, Error> {

692            &format!(
693                "[dtid {}] {sysno} refused: {reason}",
694                guest.thread_state().dettid
695            ),

698            crate::tool_global::report_unsupported_syscall(guest, sysno).await;
699        }

701            .await

704    /// Whether Detcore reads the caller's own `fd/<n>`, `exe` and `cwd`
705    /// links from the backend's `link_path` fact: where the backend's procfs
706    /// is not the guest's kernel procfs
707    /// ([`crate::procfs_inode::tool_can_name_guest_descriptors`], KVM), so
708    /// that the backend's own readlink answer is not the kernel's, and not
709    /// under record and replay, which leave procfs to the kernel. Where it is
710    /// the kernel's (ptrace), the kernel answers, as before.
711    fn reads_link_facts(&self) -> bool {
712        !self.cfg.recordreplay_modes
713            && !crate::procfs_inode::tool_can_name_guest_descriptors(&self.cfg.backend)
714    }

716    /// What the calling thread's descriptor `fd` links to, read for one of
717    /// Detcore's own decisions rather than for a guest `readlink`, where
718    /// Detcore reads the caller's links from the backend's `link_path` fact
719    /// ([`Self::reads_link_facts`], KVM): `Some(Some(text))`, Linux's
720    /// readlink text; `Some(None)` where the backend does not name the file
721    /// (an object it keeps itself) or the read fails. `None` elsewhere: the
722    /// caller reads the link in the guest (`/proc/thread-self/fd/<fd>`, an
723    /// injected `readlinkat`), whose answer there is the kernel's. Where the
724    /// fact is read, an injected read would reach the backend's own `/proc`,
725    /// which is not the guest's: the KVM executor no longer answers the
726    /// guest's links itself (package K slice F2-d).
727    pub(crate) fn own_descriptor_link_from_fact<G: Guest<Self>>(
728        guest: &G,
729        fd: i32,
730    ) -> Option<Option<Vec<u8>>> {
731        let config = guest.config();
732        if config.recordreplay_modes
733            || crate::procfs_inode::tool_can_name_guest_descriptors(&config.backend)
734        {
735            return None;
736        }
737        Some(
738            guest
739                .link_path(guest.tid(), PathLink::Descriptor(fd))
740                .ok()
741                .map(|text| text.as_bytes().to_vec()),
742        )
743    }

745    /// The text of the caller's own path link `link` from the backend's
746    /// `link_path` fact, where Detcore reads it ([`Self::reads_link_facts`]):
747    /// `Ok(Some(text))`, Linux's readlink text; `Err(ENOENT)` for a
748    /// descriptor the caller's table does not hold, as procfs has no such
749    /// entry; `Err` with the backend's errno for a name it could not render
750    /// (`ENAMETOOLONG`, as `d_path`'s); `Ok(None)` where Detcore does not
751    /// read the fact or the backend does not provide it, and for an `ESRCH`
752    /// that would say the calling thread is not live, which the fact's
753    /// contract excludes: the caller's other sources answer.
754    fn own_link_fact<G: Guest<Self>>(
755        &self,
756        guest: &G,
757        link: PathLink,
758    ) -> Result<Option<Vec<u8>>, Errno> {
759        if !self.reads_link_facts() {
760            return Ok(None);
761        }
762        match guest.link_path(guest.tid(), link) {
763            Ok(text) => Ok(Some(text.as_bytes().to_vec())),
764            Err(FactError::Errno(Errno::EBADF)) if matches!(link, PathLink::Descriptor(_)) => {
765                Err(Errno::ENOENT)
766            }
767            Err(FactError::Errno(Errno::ESRCH)) => {
768                tracing::debug!(
769                    "[detcore, tid {}] the backend's link_path fact says the caller is not live; \
770                     its {link:?} link is answered as before",
771                    guest.tid()
772                );
773                Ok(None)
774            }
775            Err(FactError::Errno(errno)) => Err(errno),
776            Err(_) => Ok(None),
777        }
778    }

780    /// Which task of the caller's process, if any, the spelling's `task`
781    /// names, from Detcore's process table: a number Detcore does not know
782    /// to be the caller's is one of its threads only where the table lists
783    /// it in the caller's process, and the leader, where it is not the
784    /// caller, still has its address space only until it exits.
785    async fn own_link_task<G: Guest<Self>>(
786        &self,
787        guest: &mut G,
788        task: NamedTask,
789        (pid, tid): (i32, i32),
790    ) -> Resolved {
791        let thread_of = |thread: i32| {
792            if thread == tid {
793                DescribedTask::Caller
794            } else if thread == pid {
795                DescribedTask::Leader
796            } else {
797                DescribedTask::Other
798            }
799        };
800        let described = match task {
801            NamedTask::Caller => DescribedTask::Caller,
802            NamedTask::Leader => thread_of(pid),
803            NamedTask::GroupThread(thread) => {
804                if !thread_in_process(guest, pid, thread).await {
805                    return Resolved::Missing;
806                }
807                thread_of(thread)
808            }
809            NamedTask::Numbered(number) => {
810                if !thread_in_process(guest, pid, number).await {
811                    return Resolved::Elsewhere;
812                }
813                thread_of(number)
814            }
815            NamedTask::NumberedThread(number, thread) => {
816                if !thread_in_process(guest, pid, number).await {
817                    return Resolved::Elsewhere;
818                }
819                if thread != tid && thread != pid && !thread_in_process(guest, pid, thread).await {
820                    return Resolved::Missing;
821                }
822                thread_of(thread)
823            }
824        };
825        if described == DescribedTask::Leader {
826            // An exited leader stays in the table as a zombie while its
827            // threads run, without an address space or filesystem
828            // information: Linux's link lookup finds neither.
829            let answer = crate::tool_global::procfs_task(
830                guest,
831                crate::procfs::BoundTask {
832                    process: pid,
833                    thread: None,
834                },
835            )
836            .await;
837            if !answer.exists || answer.without_mm {
838                return Resolved::Missing;
839            }
840        }
841        Resolved::Described(described)
842    }

844    /// Whether the caller's working and root directories are those of the
845    /// `described` task: its own, or its live leader's where the caller
846    /// shares the leader's filesystem information (`CLONE_FS`,
847    /// `ThreadState::fs_apart_from_leader`). Another thread's are its own
848    /// to know.
849    fn shares_filesystem<G: Guest<Self>>(&self, guest: &G, described: DescribedTask) -> bool {
850        match described {
851            DescribedTask::Caller => true,
852            DescribedTask::Leader => !guest.thread_state().fs_apart_from_leader,
853            DescribedTask::Other => false,
854        }
855    }

857    /// The target of the caller's process's `exe` link, from its exec record:
858    /// the image's path, where that path still names exactly the file the
859    /// process executes ([`crate::exec_record::ExecRecord::names_path_exactly`]:
860    /// the file untouched at its path, with the link count and change time it
861    /// had at the exec, below the same directories), in the caller's view
862    /// ([`Self::exec_record_root`]). Whatever Linux's link would name
863    /// otherwise (a new path, " (deleted)"), it is not this answer: `None`,
864    /// as without a record, or where Detcore cannot see the caller's
865    /// filesystem.
866    fn own_exe_target<G: Guest<Self>>(&self, guest: &G) -> Option<Vec<u8>> {
867        let exe = guest.thread_state().exe.clone()?;
868        let root = self.exec_record_root(guest)?;

870            .then(|| exe.exe.as_os_str().as_bytes().to_vec())

873    /// The caller's root directory as the tool sees the guest's files, for an
874    /// exec record's paths: the root its procfs directory names where the
875    /// tool can read that ([`Self::guest_root`]), and otherwise, where guest
876    /// threads are no host tasks (KVM, whose executor opens every absolute
877    /// guest path in the tool's own filesystem view), `/`. `None` after a
878    /// guest `chroot`.
879    pub(crate) fn exec_record_root<G: Guest<Self>>(&self, guest: &G) -> Option<PathBuf> {
880        if guest
881            .thread_state()
882            .changed_root
883            .load(std::sync::atomic::Ordering::SeqCst)
884        {
885            return None;
886        }
887        if crate::procfs_inode::tool_can_name_guest_descriptors(&self.cfg.backend) {
888            self.guest_root(guest)
889        } else {
890            Some(PathBuf::from("/"))
891        }
892    }

894    /// The calling thread's working directory as its own `getcwd` answers
895    /// it, without the NUL: the call's buffer is the guest's readlink
896    /// `buffer` (`size` bytes) where it holds the answer, the guest may write
897    /// it, and no other guest thread can run meanwhile (sequentialized
898    /// threads: the call holds the turn), and is given back its bytes
899    /// afterwards; otherwise a private
900    /// page mapped for the call alone
901    /// ([`crate::syscalls::files::procfs_levels::map_transient`]). Nothing is
902    /// written below the guest's stack pointer.
903    pub(crate) async fn injected_working_directory<G: Guest<Self>>(
904        &self,
905        guest: &mut G,
906        buffer: Option<AddrMut<'_, libc::c_char>>,
907        size: usize,
908    ) -> Result<Vec<u8>, Errno> {
909        let buffer = buffer.filter(|_| self.cfg.sequentialize_threads);
910        let saved = super::saved_link_prefix(guest, buffer, size.min(PATH_MAX));
911        // The buffer takes getcwd's write only where the guest itself could
912        // write it: its bytes written back over themselves with its own
913        // permissions first (a backend's getcwd need not check them).
914        if let (Some(address), Some(saved)) = (buffer, saved.as_deref())
915            && saved.len() >= 2
916            && matches!(
917                copy_link_out(guest, address.cast::<u8>(), saved),
918                Ok(Some(_))
919            )
920        {
921            match getcwd_at(guest, address.cast::<u8>(), saved.len()).await {
922                Ok(directory) => {
923                    // Linux's readlink writes only the link: put back the
924                    // bytes getcwd wrote, the path and its NUL, with the
925                    // buffer's own permissions (it took getcwd's write).
926                    let written = &saved[..(directory.len() + 1).min(saved.len())];
927                    return match copy_link_out(guest, address.cast::<u8>(), written) {
928                        Ok(Some(_)) => Ok(directory),
929                        Ok(None) => guest
930                            .memory()
931                            .write_exact(address.cast::<u8>(), written)
932                            .map(|()| directory)
933                            .map_err(|_| Errno::EFAULT),
934                        Err(_) => Err(Errno::EFAULT),
935                    };
936                }
937                // Nothing written (ERANGE), or a fault the private page's
938                // answer meets again at the copy (EFAULT): no restore.
939                Err(Errno::ERANGE | Errno::EFAULT) => {}
940                Err(errno) => return Err(errno),
941            }
942        }
943        let Some(page) =
944            crate::syscalls::files::procfs_levels::map_transient(guest, PATH_MAX).await
945        else {
946            return Err(Errno::ENOMEM);
947        };
948        let answered = getcwd_at(guest, page, PATH_MAX).await;
949        crate::syscalls::files::procfs_levels::unmap_transient(guest, page, PATH_MAX).await;
950        answered
951    }
952}

954/// Copy a rendered link, `target` (already cut to the call's size), to the
955/// guest's readlink buffer at `address` as Linux's `copy_to_user` does, with
956/// the destination's own permissions: the bytes in order, a write-protected
957/// or unmapped byte ending the copy, whose prefix stays written, and then
958/// `EFAULT`; otherwise the number of bytes copied. `Ok(None)` where the
959/// backend cannot write with the guest's permissions (`ENOSYS`, before
960/// anything is copied): the call then takes its previous path.
961fn copy_link_out<T: RecordOrReplay, G: Guest<Detcore<T>>>(
962    guest: &mut G,
963    address: AddrMut<'_, u8>,
964    target: &[u8],
965) -> Result<Option<i64>, Error> {
966    let mut copied = 0;
967    while copied < target.len() {
968        // SAFETY: `copied < target.len()`, so the address stays inside the
969        // guest's buffer; nothing is dereferenced here.
970        let at = unsafe { address.add(copied) };
971        match guest.memory().write_with_user_access(at, &target[copied..]) {
972            Ok(0) | Err(Errno::EFAULT) => return Err(Errno::EFAULT.into()),
973            Ok(written) => copied += written,
974            Err(Errno::ENOSYS) if copied == 0 => return Ok(None),
975            Err(errno) => return Err(errno.into()),
976        }
977    }
978    Ok(Some(copied as i64))
979}

1005/// The path `readlink` of a descriptor link gives for a per-process procfs
1006/// file opened at the absolute spelling `observed` by a caller whose pid and
1007/// tid, as the guest numbers them, are `own`: the spelling with `self` and
1008/// `thread-self` resolved as Linux's lookup resolves them, which is the path
1009/// `d_path` renders for the opened file (`/proc/<pid>/status`,
1010/// `/proc/<pid>/task/<tid>/status`). `None` for any other spelling: one
1011/// with `..`, a name below a further directory, a number procfs would not
1012/// have (a leading zero), or no `own`.
1013pub(crate) fn procfs_file_link_name(observed: &Path, own: Option<(i32, i32)>) -> Option<PathBuf> {
1014    let (pid, tid) = own?;
1015    let below = observed.strip_prefix("/proc").ok()?;
1016    let names = below
1017        .components()
1018        .map(|component| match component {
1019            std::path::Component::Normal(name) => name.to_str(),
1020            _ => None,
1021        })
1022        .collect::<Option<Vec<_>>>()?;
1023    let number = |name: &str| {
1024        (!name.starts_with('0') && name.bytes().all(|byte| byte.is_ascii_digit()))
1025            .then(|| name.parse::<i32>().ok())
1026            .flatten()
1027    };
1028    let file = |name: &str| (!name.is_empty()).then_some(());
1029    Some(PathBuf::from(match names.as_slice() {
1030        ["self", name] => {
1031            file(name)?;
1032            format!("/proc/{pid}/{name}")
1033        }
1034        ["thread-self", name] => {
1035            file(name)?;
1036            format!("/proc/{pid}/task/{tid}/{name}")
1037        }
1038        ["self", "task", thread, name] => {
1039            file(name)?;
1040            format!("/proc/{pid}/task/{}/{name}", number(thread)?)
1041        }
1042        [process, name] => {
1043            file(name)?;
1044            format!("/proc/{}/{name}", number(process)?)
1045        }
1046        [process, "task", thread, name] => {
1047            file(name)?;
1048            format!("/proc/{}/task/{}/{name}", number(process)?, number(thread)?)
1049        }
1050        _ => return None,
1051    }))
1052}

1072/// Inject `getcwd(address, size)` and read back the path it wrote, without
1073/// its NUL.
1074async fn getcwd_at<T: RecordOrReplay, G: Guest<Detcore<T>>>(
1075    guest: &mut G,
1076    address: AddrMut<'_, u8>,
1077    size: usize,
1078) -> Result<Vec<u8>, Errno> {
1079    let length = guest
1080        .inject_with_retry(
1081            syscalls::Getcwd::new()
1082                .with_buf(Some(address.cast::<libc::c_char>()))
1083                .with_size(size),
1084        )
1085        .await?;
1086    // The length counts the NUL.
1087    let length = usize::try_from(length)
1088        .ok()
1089        .filter(|length| (1..=size).contains(length))
1090        .ok_or(Errno::EFAULT)?;
1091    let mut directory = vec![0; length - 1];
1092    guest
1093        .memory()
