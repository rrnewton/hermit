9//! Where the values of a descriptor's fdinfo that Detcore renders come from
10//! (package K slice K8-h of <https://github.com/rrnewton/hermit/issues/4213>);
11//! the text is `crate::procfs::fdinfo`'s.
12//!
13//! Where the guest's procfs is not the kernel's
14//! (`procfs_inode::tool_can_name_guest_descriptors` false: KVM), Detcore
15//! serves the caller's own process's `/proc/<pid>/fdinfo/<fd>`, by every
16//! spelling, as a per-process rendered file (`Detcore::rendered_procfs_at`),
17//! and renders it at its first read, and again where Linux's `seq_file` would
18//! run `seq_show` again, as for a rendered `stat`
19//! (`ProcfsFile::renders_task_record`). Each line comes from the source Linux
20//! takes it from, as the guest can observe it:
21//!
22//! - `pos`: the position the guest's own `lseek(fd, 0, SEEK_CUR)` answers:
23//!   Detcore's for a procfs file it serves from a snapshot, a `/proc`
24//!   directory it serves on a placeholder, and a directory it lists from a
25//!   stream; elsewhere an injected `lseek(fd, 0, SEEK_CUR)`, Linux's `f_pos`,
26//!   and 0 where the file cannot seek (a pipe, a socket, a terminal:
27//!   `ESPIPE`; an `O_PATH` descriptor: `EBADF`), whose position Linux never
28//!   moves.
29//! - `flags`: the status flags the guest's own `F_GETFL` returns
30//!   (`guest_status_flags`), with `O_CLOEXEC` where the guest's own
31//!   `F_GETFD` shows `FD_CLOEXEC`, as `seq_show` adds it.
32//! - `mnt_id`: the mount an injected `statx(STATX_MNT_ID)` names, through the
33//!   run's mount-ID pool, as the guest's own `statx` numbers it, and 0 where
34//!   it names none (KVM's captured standard output and error); for a file
35//!   Detcore serves, the guest's procfs mount.
36//! - `ino`: the inode number the guest's `fstat` of the descriptor shows
37//!   (`Detcore::fdinfo_inode`).
38//! - `lock:` lines: the locks the backend's `descriptor_status` fact reports
39//!   the open file description holds, in Linux's order, each owner from
40//!   Detcore's lock ledger (`procfs::locks::LockLedger::owners`): -1 for an
41//!   OFD lock; the guest process a call was proved to take any other in (the
42//!   description's locks just before the call against those after it,
43//!   `syscalls::files::locks_taken`); 0 for one no call was (a lock inherited
44//!   or received already held, whose taker the guest's PID namespace cannot
45//!   see), as Linux shows such a taker. The device and
46//!   inode are the guest's: the device as `mountinfo` shows the file's
47//!   superblock, the inode as the `ino` line. A guest's record lock
48//!   (`F_SETLK`) that the backend holds as an OFD lock (KVM's provider:
49//!   <https://github.com/rrnewton/reverie/issues/1223>) would show as an OFD
50//!   lock, where Linux shows a `POSIX` lock to its taker's table only: a read
51//!   that would show an OFD lock on such a file is refused by name
52//!   (`procfs::locks::LockLedger::record_lock_held_as_ofd`).
53//! - an eventfd's count from the fact; its ID as Detcore numbers the
54//!   description (the number the sanitizer of a kernel fdinfo prints, so
55//!   every backend shows one); `EFD_SEMAPHORE` as `eventfd2` set it.
56//! - a signalfd's mask as `signalfd4` last set it, on the description every
57//!   alias shares (an update changes the one context, as Linux's does).
58//! - a timerfd's clock and last `timerfd_settime` flags as the guest set
59//!   them, and its count and time left from Detcore's virtual timer, as
60//!   Linux's own timer would show them at this read on the virtual clock
61//!   (`LinuxTimerfd::view`).
62//! - a Unix socket's queued descriptors (`scm_fds`) from the fact, wherever
63//!   the fact reports them, whatever Detcore's type for the descriptor (an
64//!   inherited standard stream is modelled as a regular file).
65//! - a pidfd's process as Detcore's process table numbers it, or -1 once it
66//!   has been reaped.
67//!
68//! What the object is (a socket, an anonymous inode, a character device whose
69//! driver prints lines of its own) comes from the descriptor's own stat: for a
70//! container standard stream, whose model holds the supervisor's standard
71//! input's stat whatever the stream is (`setup_stdio`), from an injected
72//! `fstat` of the descriptor itself, as the guest's own `fstat` sees it.
73//!
74//! Named differences and residuals, rendered as stated rather than hidden:
75//!
76//! - a memfd's seals: Linux v7.1.3 prints none (shmem has no `show_fdinfo`),
77//!   so nothing is printed and nothing is injected for them;
78//! - where the backend declines the fact for a descriptor (`NotProvided`: an
79//!   object it keeps itself, such as, with captured output, a guest's own
80//!   `/dev/null` or terminal that is the supervisor's standard output), the
81//!   file is rendered without the fact's lines. A `/dev/null`, a terminal, a
82//!   pipe or a regular file prints none of its own, so only a lock held
83//!   through it is missing; a socket's would be its queued descriptors, so a
84//!   declined socket's read is refused by name;
85//! - a read is refused by name, through the unsupported-operation policy
86//!   (`Detcore::scratch_failure`: a fail-closed run stops with the
87//!   policy-refusal exit, and with `--allow-unsupported-syscalls` the read
88//!   fails with `EOPNOTSUPP`), never answered with an empty file, for an
89//!   epoll set (its `tfd:` lines), an inotify instance, a userfaultfd, an
90//!   anonymous inode Detcore does not classify, a descriptor Detcore keeps no
91//!   description of, an eventfd, signalfd, timerfd or pidfd whose facts
92//!   Detcore did not record (inherited, received without joining the
93//!   description its sender's table modeled, or rediscovered after an exec),
94//!   a timerfd where Detcore's virtual timerfd is off (a run that is not
95//!   strict: its timer is the host's), a socket the fact does not describe,
96//!   and a character device whose driver prints lines of its own (a
97//!   pseudo-terminal master's `tty-index`, the number of a host terminal
98//!   that other host programs' terminals decide, a tun or DRM device's). The
99//!   ptrace backend's sanitizer of a kernel fdinfo answers such a file with
100//!   an empty one, a residual of that path that this one does not share;
101//! - an eventfd's ID is Detcore's number for the description, not Linux's;
102//! - a read by a thread of another process (a descriptor inherited across
103//!   `fork` or received over `SCM_RIGHTS`) is refused by name: the fact and
104//!   the injected calls reach the reader's own descriptor table only. KVM
105//!   gives no thread a table of its own (`clone` without `CLONE_FILES` is
106//!   refused), so every thread of the process reads the one table.
107//!
108//! Determinism: every value is a function of the guest's own history at the
109//! read. The injected calls run in the reader's turn, with every other guest
110//! thread parked; the fact reports the backend's state, which only guest
111//! calls change (the counter, the queued messages, the locks), all in turns;
112//! the ledger's takers come from guest calls in turns; time is the guest's
113//! virtual clock at the read. Host identities (mount, device, inode) reach
114//! the text only through Detcore's pools, as everywhere else.

145/// What Detcore's model holds of the descriptor an fdinfo describes.
146struct DescriptorModel {
147    ty: FdType,
148    nonblocking: bool,
149    status_flags: i32,
150    stat: Option<DetStat>,
151    open_file: crate::types::OpenFileId,
152    resource: Option<crate::resources::ResourceID>,
153    facts: Option<FdinfoFacts>,
154    timerfd: Option<crate::fd::VirtualTimerfd>,
155    pidfd_target: Option<crate::types::DetPid>,
156    /// A procfs file Detcore renders.
157    rendered: Option<&'static Path>,
158    /// The position of a procfs file Detcore serves from a snapshot, rendered
159    /// or not, which the guest's own `lseek` answers.
160    procfs_position: Option<usize>,
161    /// The position of a directory Detcore lists from a stream
162    /// (`DirectoryStream`), which the guest's own `lseek` answers.
163    directory_position: Option<u64>,
164    placeholder: bool,
165}
166
167/// The value of an fdinfo `pos:` line (`pos:\t%lli`), if `text` has one.
168fn fdinfo_position(text: &[u8]) -> Option<i64> {
169    String::from_utf8_lossy(text)
170        .lines()
171        .find_map(|line| line.strip_prefix("pos:\t"))
172        .and_then(|position| position.parse().ok())
173}

176    /// Whether Detcore renders the fdinfo of the caller's own descriptors
177    /// (package K slice K8-h): where the guest's procfs is not the kernel's
178    /// (`procfs_inode::tool_can_name_guest_descriptors`, KVM), so no kernel
179    /// fdinfo of the guest's descriptors exists to sanitize. Elsewhere the
180    /// kernel's fdinfo is the guest's own and is sanitized, as before
181    /// (`procfs::sanitize_fdinfo`).
182    pub(crate) fn renders_fdinfo(&self) -> bool {
183        !procfs_inode::tool_can_name_guest_descriptors(&self.cfg.backend)
184    }

186    /// Whether `task` is a thread of the caller's own process, whose
187    /// descriptor table the backend's `descriptor_status` fact and the
188    /// injected calls reach: `/proc/self`, `/proc/thread-self`, the caller's
189    /// pid and its `task/<tid>` directories, and `/proc/<tid>` of a thread of
190    /// the caller's process, by Detcore's process table.
191    pub(super) async fn names_own_process<G: Guest<Self>>(
192        &self,
193        guest: &mut G,
194        task: BoundTask,
195    ) -> bool {
196        let own = self.own_detpid(guest).as_raw();
197        task.process == own
198            || crate::tool_global::task_stat_facts(guest, task)
199                .await
200                .process
201                == Some(own)
202    }
203
204    /// [`Self::names_own_process`] for a spelling of a task, bound to the
205    /// caller, which Detcore's process table must have.
206    pub(super) async fn names_own_process_spelling<G: Guest<Self>>(
207        &self,
208        guest: &mut G,
209        task: ProcfsTask,
210    ) -> bool {
211        let Some((pid, tid)) = self.procfs_caller_ids(guest) else {
212            return false;
213        };
214        let Some(task) = task.bound_for(pid, tid) else {
215            return false;
216        };
217        crate::tool_global::procfs_task(guest, task).await.exists
218            && self.names_own_process(guest, task).await
219    }

221    /// Whether the caller's descriptor table holds descriptor `fd`, as
222    /// Linux's lookup of `fdinfo/<fd>` requires (`proc_lookupfd_common`):
223    /// Detcore's table says, modelled or not; where a receive may have
224    /// installed descriptors Detcore does not know, the backend's
225    /// `descriptor_status` fact says whether one it does not know is open.
226    pub(super) fn table_holds<G: Guest<Self>>(guest: &G, fd: RawFd) -> bool {
227        let (numbers, _, unaccounted) = guest.thread_state().descriptor_table();
228        numbers.binary_search(&fd).is_ok()
229            || (unaccounted
230                && !matches!(
231                    guest.descriptor_status(fd),
232                    Err(reverie::FactError::Errno(Errno::EBADF))
233                ))
234    }

236    /// The text of the fdinfo Detcore opened as rendered at descriptor `fd`,
237    /// as Linux's `seq_show` would print it at this read
238    /// (`procfs::fdinfo::render`); `None` for any other file. The task it
239    /// belongs to must still be the one it was bound to, and its descriptor
240    /// must still be open in the reader's table (ENOENT otherwise, as
241    /// `seq_show` fails). A read by a thread of another process is refused by
242    /// name through the unsupported-operation policy.
243    pub(super) async fn render_own_fdinfo<G: Guest<Self>>(
244        &self,
245        guest: &mut G,
246        fd: RawFd,
247    ) -> Result<Option<Vec<u8>>, Error> {
248        let Some(path) = guest
249            .thread_state()
250            .with_detfd(fd, |detfd| detfd.procfs_rendered_path())?
251        else {
252            return Ok(None);
253        };
254        let Some(file) = ProcfsFile::from_rendered_path(path).filter(ProcfsFile::is_fdinfo) else {
255            return Ok(None);
256        };
257        let (Some(task), Some(target)) = (file.bound_task(), file.rendered_fdinfo_target()) else {
258            return Err(Errno::ENOENT.into());
259        };
260        match self.rendered_task_state(guest, path).await {
261            Some((_, true)) => {}
262            _ => return Err(Errno::ENOENT.into()),
263        }
264        if !self.names_own_process(guest, task).await {
265            let reason = format!(
266                "a thread of another process read the rendered fdinfo of descriptor {target} of \
267                 task {task:?} (a descriptor inherited across fork or received over SCM_RIGHTS); \
268                 Detcore reads the reader's own descriptor table only"
269            );
270            let sysno = Sysno::new(guest.regs().await.orig_rax as usize).unwrap_or(Sysno::read);
271            return Err(Self::scratch_failure(guest, sysno, reason).await);
272        }
273        if !Self::table_holds(guest, target) {
274            return Err(Errno::ENOENT.into());
275        }
276        Ok(Some(self.fdinfo_text(guest, target).await?))
277    }

279    /// Refuse the read of descriptor `target`'s rendered fdinfo by name,
280    /// through the unsupported-operation policy (`Self::scratch_failure`):
281    /// Detcore cannot render it as Linux prints it, for the reason `why`. Never
282    /// an empty file, which would be a successful read of nothing (see the
283    /// module comment).
284    async fn refuse_fdinfo<G: Guest<Self>>(guest: &mut G, target: RawFd, why: String) -> Error {
285        let reason = format!(
286            "the rendered fdinfo of descriptor {target}: {why}; Detcore cannot render it as \
287             Linux prints it"
288        );
289        let sysno = Sysno::new(guest.regs().await.orig_rax as usize).unwrap_or(Sysno::read);
290        Self::scratch_failure(guest, sysno, reason).await
291    }

293    /// The fdinfo text of descriptor `target` of the reader's table, now.
294    async fn fdinfo_text<G: Guest<Self>>(
295        &self,
296        guest: &mut G,
297        target: RawFd,
298    ) -> Result<Vec<u8>, Error> {
299        let model = guest
300            .thread_state()
301            .with_detfd(target, |detfd| DescriptorModel {
302                ty: detfd.ty(),
303                nonblocking: detfd.is_nonblocking(),
304                status_flags: detfd.status_flags(),
305                stat: detfd.stat(),
306                open_file: detfd.open_file_id(),
307                resource: detfd.resource(),
308                facts: detfd.fdinfo_facts(),
309                timerfd: detfd.virtual_timerfd(),
310                pidfd_target: detfd.pidfd_target(),
311                rendered: detfd.procfs_rendered_path(),
312                procfs_position: detfd.procfs_position().map(|(offset, _)| offset),
313                directory_position: detfd
314                    .has_directory_stream()
315                    .then(|| detfd.directory_stream_position()),
316                placeholder: detfd.placeholder().is_some(),
317            })
318            .ok();
319        // A descriptor Detcore keeps no description of (a received anonymous
320        // inode): no flags, type or identity to render. Refused by name.
321        let Some(model) = model else {
322            return Err(Self::refuse_fdinfo(
323                guest,
324                target,
325                "Detcore keeps no description of the descriptor (a descriptor received over \
326                 SCM_RIGHTS that it did not record), so it has no flags, type or identity to \
327                 print"
328                    .to_owned(),
329            )
330            .await);
331        };
332        // What the backend reports that no call reads back. `EBADF`: the
333        // descriptor is not open after all, as `seq_show` finds it.
334        let status = match guest.descriptor_status(target) {
335            Ok(status) => Some(status),
336            Err(reverie::FactError::Errno(Errno::EBADF)) => return Err(Errno::ENOENT.into()),
337            Err(reverie::FactError::NotProvided) => None,
338            Err(error) => {
339                let reason = format!(
340                    "the backend's descriptor_status of descriptor {target}, which its fdinfo \
341                     prints, failed: {error:?}"
342                );
343                let sysno = Sysno::new(guest.regs().await.orig_rax as usize).unwrap_or(Sysno::read);
344                return Err(Self::scratch_failure(guest, sysno, reason).await);
345            }
346        };
347        let placeholder = if model.placeholder {
348            Box::pin(self.placeholder_fdinfo(guest, target)).await?
349        } else {
350            None
351        };
352        // The position, as the guest's own `lseek(fd, 0, SEEK_CUR)` answers
353        // it: Detcore's for a file it serves from a snapshot, a directory it
354        // serves on a placeholder, or one it lists from a stream; the
355        // kernel's for every other file.
356        let pos = if let Some(offset) = model.procfs_position {
357            offset as i64
358        } else if let Some((text, _)) = &placeholder {
359            fdinfo_position(text).unwrap_or(0)
360        } else if let Some(position) = model.directory_position {
361            position as i64
362        } else {
363            guest
364                .inject(
365                    syscalls::Lseek::new()
366                        .with_fd(target)
367                        .with_offset(0)
368                        .with_whence(syscalls::Whence::SEEK_CUR),
369                )
370                .await
371                .unwrap_or(0)
372        };
373        // The status flags, as the guest's own `F_GETFL` answers them, with
374        // `O_CLOEXEC` from the descriptor slot.
375        let status_flags = if placeholder.is_some() {
376            i64::from(model.status_flags & !libc::O_CLOEXEC)
377        } else {
378            let physical = guest
379                .inject(
380                    syscalls::Fcntl::new()
381                        .with_fd(target)
382                        .with_cmd(syscalls::FcntlCmd::F_GETFL),
383                )
384                .await?;
385            super::guest_status_flags(
386                physical,
387                model.nonblocking,
388                model.rendered.is_some() && model.status_flags & libc::O_NOFOLLOW != 0,
389            )
390        };
391        // `O_CLOEXEC` as the descriptor table holds it (`close_on_exec`), which
392        // the guest's own `F_GETFD` reads.
393        let descriptor_flags = guest
394            .inject(
395                syscalls::Fcntl::new()
396                    .with_fd(target)
397                    .with_cmd(syscalls::FcntlCmd::F_GETFD),
398            )
399            .await?;
400        let cloexec = descriptor_flags & i64::from(libc::FD_CLOEXEC) != 0;
401        let flags = status_flags as u32 | if cloexec { libc::O_CLOEXEC as u32 } else { 0 };
402        // The mount, as the guest's own `statx` numbers it, and 0 where that
403        // reports none: KVM's captured standard output and error under
404        // `--verify` (pipes its executor keeps itself) report no mount.
405        let raw_mount_id = if model.rendered.is_some() || placeholder.is_some() {
406            Some(super::rendered_procfs::guest_procfs_mount_id()?)
407        } else {
408            Self::inject_statx_mount_id(guest, target).await?
409        };
410        let mnt_id = self.fdinfo_mount_id(guest, raw_mount_id).await;
411        // The inode, as the descriptor's `fstat` numbers it, and what the
412        // object is, which its type lines are keyed on (`file_kind`): from the
413        // descriptor's own `fstat` where Detcore's model has no stat of it, or
414        // one that need not be its own: the container's standard streams
415        // share the one stat `setup_stdio` took of the supervisor's standard
416        // input, whatever each is.
417        let stat = match model.stat {
418            Some(stat) if !super::is_container_stream(model.resource.clone()) => stat,
419            _ => DetStat::from(self.inject_fstat(guest, target).await?),
420        };
421        let ino = self
422            .fdinfo_inode(
423                guest,
424                target,
425                &stat,
426                model.resource.clone(),
427                model.rendered,
428                placeholder.as_ref().map(|(_, inode)| *inode),
429            )
430            .await
431            .as_raw();
432        // What the file's own `show_fdinfo` adds, or why Detcore cannot
433        // render it.
434        let kind = match self
435            .fdinfo_kind(guest, &model, &stat, status.as_ref())
436            .await
437        {
438            Ok(kind) => kind,
439            Err(why) => return Err(Self::refuse_fdinfo(guest, target, why).await),
440        };
441        // The locks held through the description, with their owners, unless a
442        // line would show a guest's record lock the backend holds as an OFD
443        // lock.
444        let mut locks = Vec::new();
445        if let Some(status) = &status
446            && !status.locks.is_empty()
447        {
448            let held = super::held_locks_of_status(status);
449            let owners = match crate::tool_global::lock_owners(guest, model.open_file, held).await {
450                Ok(owners) => owners,
451                Err(why) => return Err(Self::refuse_fdinfo(guest, target, why).await),
452            };
453            for (lock, owner) in status.locks.iter().zip(owners) {
454                locks.push(FdinfoLock {
455                    class: lock.class,
456                    mandatory: lock.mandatory,
457                    write: lock.write,
458                    owner,
459                    device: Self::lock_line_device(guest, lock.raw_device).await,
460                    inode: ino,
461                    start: lock.start,
462                    end: lock.end,
463                });
464            }
465        }
466        Ok(crate::procfs::fdinfo::render(&Fdinfo {
467            pos,
468            flags,
469            mnt_id,
470            ino,
471            locks,
472            kind,
473        }))
474    }

476    /// The lines the file's own `show_fdinfo` adds, from the descriptor's
477    /// model and the backend's fact (`status`, where it reports one), or why
478    /// Detcore cannot render them, for which the read is refused by name (see
479    /// the module comment).
480    async fn fdinfo_kind<G: Guest<Self>>(
481        &self,
482        guest: &mut G,
483        model: &DescriptorModel,
484        stat: &DetStat,
485        status: Option<&reverie::DescriptorStatus>,
486    ) -> Result<FdinfoKind, String> {
487        /// Why a descriptor's facts are missing.
488        const UNRECORDED: &str = "(inherited, received over SCM_RIGHTS without joining the \
489                                  description its sender's table modeled, or rediscovered \
490                                  after an exec)";
491        Ok(match model.ty {
492            FdType::Eventfd => {
493                match (model.facts, status.and_then(|status| status.eventfd_count)) {
494                    (Some(FdinfoFacts::Eventfd { semaphore }), Some(count)) => {
495                        FdinfoKind::Eventfd {
496                            count,
497                            id: model.open_file.deterministic_socket_cookie(),
498                            semaphore,
499                        }
500                    }
501                    (None, _) => {
502                        return Err(format!(
503                            "an eventfd whose flags Detcore did not record {UNRECORDED}"
504                        ));
505                    }
506                    _ => {
507                        return Err(
508                            "an eventfd whose count the backend's fact did not report".to_owned()
509                        );
510                    }
511                }
512            }
513            FdType::Signalfd => match model.facts {
514                Some(FdinfoFacts::Signalfd { mask }) => FdinfoKind::Signalfd { mask },
515                _ => {
516                    return Err(format!(
517                        "a signalfd whose mask Detcore did not record {UNRECORDED}"
518                    ));
519                }
520            },
521            FdType::Timerfd => {
522                let Some(FdinfoFacts::Timerfd {
523                    clockid,
524                    settime_flags,
525                    linux,
526                }) = model.facts
527                else {
528                    return Err(format!(
529                        "a timerfd whose clock and settings Detcore did not record {UNRECORDED}"
530                    ));
531                };
532                if !crate::syscalls::timerfd::enabled(guest) {
533                    return Err("a timerfd whose timer is the host's: Detcore's virtual \
534                                timerfd is off in a run that is not strict"
535                        .to_owned());
536                }
537                let (ticks, value_ns, interval_ns) = match model.timerfd {
538                    Some(timer) => {
539                        let now = crate::syscalls::timerfd::timerfd_now(guest).await;
540                        let (ticks, value_ns) = linux.view(now);
541                        (ticks, value_ns, timer.interval_ns)
542                    }
543                    None => (0, 0, 0),
544                };
545                FdinfoKind::Timerfd(TimerfdLines {
546                    clockid,
547                    ticks,
548                    settime_flags,
549                    value_ns,
550                    interval_ns,
551                })
552            }
553            FdType::Pidfd => {
554                let Some(target) = model.pidfd_target else {
555                    return Err(format!(
556                        "a pidfd whose process Detcore did not record {UNRECORDED}"
557                    ));
558                };
559                let task = BoundTask {
560                    process: target.as_raw(),
561                    thread: None,
562                };
563                let exists = crate::tool_global::procfs_task(guest, task).await.exists;
564                FdinfoKind::Pidfd {
565                    pid: if exists {
566                        i64::from(target.as_raw())
567                    } else {
568                        -1
569                    },
570                }
571            }
572            FdType::Epoll => {
573                return Err(
574                    "an epoll instance, whose tfd: lines name the host's descriptors \
575                            and inodes of its watched files"
576                        .to_owned(),
577                );
578            }
579            FdType::Inotify => {
580                return Err(
581                    "an inotify instance, whose watch lines name the host's inodes and \
582                            devices"
583                        .to_owned(),
584                );
585            }
586            FdType::Userfaultfd => {
587                return Err("a userfaultfd, whose lines Detcore does not model".to_owned());
588            }
589            FdType::Socket | FdType::Regular | FdType::Pipe | FdType::Memfd | FdType::Rng => {
590                file_kind(model.ty, stat, status)?
591            }
592        })
593    }
594}

596/// The lines a file's own `show_fdinfo` adds for a descriptor Detcore models
597/// as a socket, a pipe, a memfd, a random device or a regular file, keyed on
598/// what the object is rather than on Detcore's type for the descriptor (an
599/// inherited standard stream is modelled as a regular file whatever it is),
600/// or why Detcore cannot render them, for which the read is refused by name.
601///
602/// - A Unix socket's queued descriptors, wherever the backend's fact reports
603///   them: it reports `scm_fds` for a Unix socket only, as Linux prints it
604///   for every one (`unix_show_fdinfo`, net/unix/af_unix.c).
605/// - A socket the fact does not describe (the backend declined it): whether
606///   it is a Unix socket, whose fdinfo prints its queued descriptors, is
607///   unknown, so the read is refused, as an eventfd without its count is.
608///   One the fact describes without `scm_fds` is another kind of socket,
609///   whose fdinfo has no lines of its own.
610/// - An anonymous inode (no file type in its mode) Detcore does not classify,
611///   which may have lines of its own: refused.
612/// - A character device whose driver prints lines of its own
613///   (`show_fdinfo`): a pseudo-terminal master's `tty-index`
614///   (`pty_show_fdinfo`, drivers/tty/pty.c; the ptmx device, 5:2), a tun
615///   device's `iff` (10:200) and a DRM device's (major 226): refused, as a
616///   file whose lines Detcore cannot render.
617fn file_kind(
618    ty: FdType,
619    stat: &DetStat,
620    status: Option<&reverie::DescriptorStatus>,
621) -> Result<FdinfoKind, String> {
622    if let Some(scm_fds) = status.and_then(|status| status.scm_fds) {
623        return Ok(FdinfoKind::UnixSocket { scm_fds });
624    }
625    let file_type = stat
626        .mask
627        .contains(StatxMask::STATX_TYPE)
628        .then_some(stat.mode & libc::S_IFMT);
629    if ty == FdType::Socket || file_type == Some(libc::S_IFSOCK) {
630        return match status {
631            Some(_) => Ok(FdinfoKind::Plain),
632            None => Err(
633                "a socket the backend's descriptor_status fact does not describe, \
634                         whose fdinfo would print its queued descriptors (scm_fds) if it is a \
635                         Unix socket"
636                    .to_owned(),
637            ),
638        };
639    }
640    if file_type == Some(0) {
641        return Err(
642            "an anonymous inode Detcore does not classify, which may print lines of \
643                    its own"
644                .to_owned(),
645        );
646    }
647    if file_type == Some(libc::S_IFCHR) && driver_prints_fdinfo(stat.rdev) {
648        return Err(format!(
649            "a character device {}:{} whose driver prints lines of its own (a pseudo-terminal \
650             master's tty-index, a tun device's iff, a DRM device's), which Detcore does not \
651             render",
652            libc::major(stat.rdev),
653            libc::minor(stat.rdev)
654        ));
655    }
656    Ok(FdinfoKind::Plain)
657}
658
659/// Whether the character device `rdev` is one whose Linux v7.1.3 driver
660/// prints fdinfo lines of its own: the ptmx device (5:2, a pseudo-terminal
661/// master), the tun device (10:200) and the DRM devices (major 226).
662fn driver_prints_fdinfo(rdev: u64) -> bool {
663    let (major, minor) = (libc::major(rdev), libc::minor(rdev));
664    (major, minor) == (5, 2) || (major, minor) == (10, 200) || major == 226
665}

667impl<T: RecordOrReplay> Detcore<T> {
668    /// The device a lock line shows for the superblock device `raw_device`
669    /// the backend reported, as the guest's `mountinfo` shows that file
670    /// system's device: through a proven mountinfo-to-metadata rewrite
671    /// (`Config::mountinfo_device_rewrites`), then the run's device pool where
672    /// metadata is virtualized, as `mountinfo` and `stat` number it.
673    async fn lock_line_device<G: Guest<Self>>(guest: &mut G, raw_device: u64) -> u64 {
674        let device = guest
675            .config()
676            .mountinfo_device_rewrites
677            .iter()
678            .find(|(mountinfo, _)| *mountinfo == raw_device)
679            .map_or(raw_device, |(_, metadata)| *metadata);
680        if guest.config().virtualize_metadata {
681            crate::tool_global::determinize_device(guest, device).await
682        } else {
683            device
684        }
685    }
686
687    /// The mount ID the backend reports for descriptor `fd`, from an injected
688    /// `statx(fd, "", AT_EMPTY_PATH, STATX_MNT_ID)`, as Linux's fdinfo
689    /// prints it (`real_mount(file->f_path.mnt)->mnt_id`, the mount `statx`
690    /// reports): `None` where the call did not report one. The empty path
691    /// and the buffer are on the guest's stack scratch.
692    async fn inject_statx_mount_id<G: Guest<Self>>(
693        guest: &mut G,
694        fd: RawFd,
695    ) -> Result<Option<u64>, Error> {
696        let mut stack = guest.stack().await;
697        let path = stack.push(c_string::<8>(b""));
698        let statx = stack.reserve::<reverie::syscalls::linux_abi::Statx>();
699        let stack_guard = stack.commit()?;
700        let result = guest
701            .inject(
702                syscalls::Statx::new()
703                    .with_dirfd(fd)
704                    .with_path(PathPtr::from_ptr(path.as_raw() as *const libc::c_char))
705                    .with_flags(AtFlags::AT_EMPTY_PATH)
706                    .with_mask(StatxMask::STATX_MNT_ID)
707                    .with_statx(Some(reverie::syscalls::StatxPtr(statx))),
708            )
709            .await;
710        let answer = match result {
711            Ok(_) => {
712                let value: reverie::syscalls::linux_abi::Statx =
713                    guest.memory().read_value(statx)?;
714                (value.stx_mask & reverie::syscalls::linux_abi::STATX_MNT_ID != 0)
715                    .then_some(value.stx_mnt_id)
716            }
717            Err(_) => None,
718        };
719        drop(stack_guard);
720        Ok(answer)
721    }
722}

724#[cfg(test)]
725mod file_kind_tests {
726    use super::*;
727
728    fn stat(mode: u32, rdev: u64) -> DetStat {
729        DetStat {
730            mask: StatxMask::STATX_TYPE,
731            mode,
732            rdev,
733            ..DetStat::default()
734        }
735    }
736
737    fn status(fdinfo: &str) -> reverie::DescriptorStatus {
738        reverie::DescriptorStatus::from_fdinfo(fdinfo.as_bytes()).unwrap()
739    }
740
741    /// A socket's, a device's or a plain file's fdinfo lines follow what the
742    /// object is, not Detcore's type for the descriptor: a Unix socket's
743    /// queued descriptors wherever the fact reports them (an inherited
744    /// standard stream is a regular file to Detcore); no lines for another
745    /// socket the fact describes; a named refusal for a socket the fact
746    /// declines, for an anonymous inode, and for a character device whose
747    /// driver prints lines of its own (a pseudo-terminal master, tun, DRM),
748    /// which Detcore cannot render; no lines for `/dev/null` or a regular
749    /// file.
750    #[test]
751    fn a_files_lines_follow_what_the_object_is() {
752        let unix = status("pos:\t0\nflags:\t02\nmnt_id:\t9\nino:\t5\nscm_fds: 2\n");
753        let inet = status("pos:\t0\nflags:\t02\nmnt_id:\t9\nino:\t5\n");
754        let socket = stat(libc::S_IFSOCK | 0o777, 0);
755        assert_eq!(
756            file_kind(FdType::Regular, &socket, Some(&unix)),
757            Ok(FdinfoKind::UnixSocket { scm_fds: 2 })
758        );
759        assert_eq!(
760            file_kind(FdType::Socket, &socket, Some(&unix)),
761            Ok(FdinfoKind::UnixSocket { scm_fds: 2 })
762        );
763        assert_eq!(
764            file_kind(FdType::Socket, &socket, Some(&inet)),
765            Ok(FdinfoKind::Plain)
766        );
767        for ty in [FdType::Regular, FdType::Socket] {
768            let refusal = file_kind(ty, &socket, None).unwrap_err();
769            assert!(refusal.contains("a socket"), "{refusal}");
770        }
771        let refusal = file_kind(FdType::Regular, &stat(0, 0), Some(&inet)).unwrap_err();
772        assert!(refusal.contains("an anonymous inode"), "{refusal}");
773        let device = |major, minor| stat(libc::S_IFCHR | 0o666, libc::makedev(major, minor));
774        for (major, minor) in [(5, 2), (10, 200), (226, 0), (226, 128)] {
775            let refusal = file_kind(FdType::Regular, &device(major, minor), Some(&inet))
776                .expect_err("a driver that prints lines of its own is refused");
777            assert!(
778                refusal.contains(&format!("a character device {major}:{minor} ")),
779                "{refusal}"
780            );
781        }
782        assert_eq!(
783            file_kind(FdType::Regular, &device(1, 3), None),
784            Ok(FdinfoKind::Plain)
785        );
786        assert_eq!(
787            file_kind(
788                FdType::Regular,
789                &stat(libc::S_IFREG | 0o644, 0),
790                Some(&inet)
791            ),
792            Ok(FdinfoKind::Plain)
793        );
794    }
