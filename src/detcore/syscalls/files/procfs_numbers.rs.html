9//! A `/proc` path that names a process or thread by the guest's number
10//! reaches the kernel's directory for it where the kernel numbers processes
11//! otherwise than Detcore (DBT; slice B3a-4 of
12//! <https://github.com/rrnewton/hermit/issues/4266>; see
13//! [`crate::procfs::ids`]).
14//!
15//! The guest's call goes to the kernel with each such path spelled with the
16//! kernel's numbers instead, staged on a transient page, as the calls for a
17//! name outside the `--proc` level are (see [`super::procfs_levels`]). Every
18//! check before the call, and every decision Detcore takes about the path,
19//! sees the guest's spelling; only the kernel sees the other. A number that
20//! names no process or thread `/proc` lists is spelled `0`, which names
21//! nothing, so the kernel's process of that number, if any, is never reached.
22

54/// A call whose paths are respelled (with the kernel's numbers, or below a
55/// descriptor directory), and where they are staged, which the caller
56/// releases once the call returns ([`Detcore::finish_translated_call`]).
57pub(crate) struct TranslatedCall {
58    pub(crate) call: Syscall,
59    staging: Staging,
60}
61
62/// Where a [`TranslatedCall`]'s respelled paths are staged.
63enum Staging {
64    /// In the guest stack scratch below the stack pointer, which needs no
65    /// release.
66    Scratch,
67    /// In a transient private page, unmapped once the call returns.
68    Page {
69        page: AddrMut<'static, u8>,
70        length: usize,
71    },
72}
73
74/// How far below the stack pointer the guest stack scratch for respelled
75/// paths starts: past the red zone and the whole of the scratch Reverie
76/// hands out (`Guest::stack`: the red zone's 128 bytes and 896 more, on
77/// ptrace), so a stack guard alive around the call is never overlapped and
78/// none is checked out (a second checkout while one is alive panics,
79/// `reverie-ptrace/src/stack.rs`).
80const SCRATCH_GAP: usize = 1024 + 64;
81
82/// The most bytes of respelled paths, NULs included, staged in the guest
83/// stack scratch; more go to a transient page. A path respelled below a
84/// descriptor directory starts with a short spelling of it
85/// (`/proc/self/fd/N`).
86const SCRATCH_LIMIT: usize = 512;

89    /// Whether this guest thread's `/proc` paths may need translating: where
90    /// its backend registered the thread's kernel ids (DBT, and an in-guest
91    /// native exec, whose numberings agree, so its paths come back
92    /// unchanged), outside record and replay (whose replayer answers for the
93    /// recorded paths), a nested PID namespace (whose numbers are not the
94    /// container's) and a guest `chroot` (after which `/proc` need not be
95    /// procfs).
96    pub(crate) fn translates_procfs_numbers<G: Guest<Self>>(guest: &G) -> bool {
97        let state = guest.thread_state();
98        state.physical_tid.is_some()
99            && !guest.config().recordreplay_modes
100            && !state.nested_pid_namespace
101            && !state.changed_root.load(Ordering::SeqCst)
102    }

104    /// The directory the guest's relative spelling, looked up from `dirfd`,
105    /// starts at, resolved: the kernel's name for the directory `dirfd` is
106    /// open on, or for the working directory with `AT_FDCWD`, which the
107    /// tool, on the guest's own thread where numbers are translated (DBT),
108    /// reads as the guest's. It is spelled with the kernel's numbers, and
109    /// names no link. Only the number of its components, and whether it is
110    /// the init's, matter ([`ids::Lookup`]). `None` where the tool has no
111    /// view of the guest's descriptors, and for a directory that was
112    /// removed.
113    fn relative_base<G: Guest<Self>>(guest: &G, dirfd: i32) -> Option<PathBuf> {
114        if !guest.config().backend.tool_shares_guest_descriptor_table {
115            return None;
116        }
117        let link = if dirfd == libc::AT_FDCWD {
118            PathBuf::from("/proc/thread-self/cwd")
119        } else {
120            PathBuf::from(format!("/proc/thread-self/fd/{dirfd}"))
121        };
122        let base = std::fs::read_link(link).ok()?;
123        (base.is_absolute() && !base.as_os_str().as_bytes().ends_with(b" (deleted)"))
124            .then_some(base)
125    }

127    /// The guest's `spelling`, absolute, or relative to `dirfd` (or the
128    /// working directory for `AT_FDCWD`), whose resolved name is `base`, with
129    /// each number it names as a process or thread replaced in place by the
130    /// kernel's number for it (see the module documentation). Every other
131    /// component is kept, and a relative spelling stays relative, so the
132    /// kernel looks the path up as it would the guest's: from the same
133    /// directory, through the same links, counted against the same limit.
134    ///
135    /// Where the meaning of a component depends on where a link before it
136    /// leads (a number, or a `..` a number follows, past `cwd`, `root`,
137    /// `exe`, a descriptor's link, or a directory outside `/proc`), the
138    /// tool, on the guest's own thread, opens the components before it as
139    /// the kernel looks them up (`O_PATH`, from the same directory), and
140    /// walks the rest from the kernel's name for what it opened. That only
141    /// decides which components are numbers: what the tool opens is closed
142    /// again. Where it cannot be opened, the kernel's lookup of the same
143    /// components fails too, so the rest stays as it is. `None` where no
144    /// number changes.
145    async fn kernel_spelling<G: Guest<Self>>(
146        guest: &mut G,
147        dirfd: i32,
148        base: Option<PathBuf>,
149        spelling: &Path,
150    ) -> Option<PathBuf> {
151        let names = ids::components(spelling);
152        let shares = guest.config().backend.tool_shares_guest_descriptor_table;
153        let mut replaced: Vec<(usize, OsString)> = Vec::new();
154        let mut base = base;
155        let mut offset = 0;
156        // The directory the components from `offset` on are looked up from,
157        // where a resolution opened one.
158        let mut directory: Option<OwnedFd> = None;
159        while offset < names.len() {
160            let absolute = offset == 0 && spelling.is_absolute();
161            let rest = if absolute {
162                spelling.to_path_buf()
163            } else {
164                ids::rebuilt_like(names[offset..].to_vec(), Path::new(""))
165            };
166            let lookup = ids::Lookup::new(base.as_deref(), &rest);
167            let named = lookup
168                .ids()
169                .into_iter()
170                .collect::<BTreeSet<_>>()
171                .into_iter()
172                .collect::<Vec<_>>();
173            let map = if named.is_empty() {
174                BTreeMap::new()
175            } else {
176                let kernel = crate::tool_global::procfs_ids(guest, true, named.clone()).await;
177                named.into_iter().zip(kernel).collect::<BTreeMap<_, _>>()
178            };
179            replaced.extend(
180                lookup
181                    .translations(|id| map.get(&id).copied().flatten())
182                    .into_iter()
183                    .map(|(position, name)| (offset + position, name)),
184            );
185            // Only the tool on the guest's own thread looks up as the guest.
186            let Some(at) = lookup.stop().filter(|_| shares) else {
187                break;
188            };
189            let mut prefix = names[offset..offset + at].to_vec();
190            for (position, name) in &replaced {
191                if (offset..offset + at).contains(position) {
192                    prefix[position - offset] = name.clone();
193                }
194            }
195            let from = directory.as_ref().map_or(dirfd, |fd| fd.as_raw_fd());
196            let Some(opened) = open_path(from, prefix, absolute) else {
197                break;
198            };
199            directory = Some(opened);
200            offset += at;
201            base = directory.as_ref().and_then(resolved_name);
202            // A removed directory has no name to walk from, and no entries:
203            // only `.` and `..` lead on from it.
204            while base.is_none() && offset < names.len() {
205                let name = &names[offset];
206                if name != "." && name != ".." {
207                    break;
208                }
209                let from = directory.as_ref().map_or(dirfd, |fd| fd.as_raw_fd());
210                let Some(opened) = open_path(from, vec![name.clone()], false) else {
211                    break;
212                };
213                directory = Some(opened);
214                offset += 1;
215                base = directory.as_ref().and_then(resolved_name);
216            }
217            if base.is_none() {
218                break;
219            }
220        }
221        if replaced.is_empty() {
222            return None;
223        }
224        let mut names = names;
225        for (position, name) in replaced {
226            names[position] = name;
227        }
228        Some(ids::rebuilt_like(names, spelling))
229    }

231    /// Whether [`Self::procfs_translated_call`] can translate anything in
232    /// `call`: numbers are translated on this thread and `call` looks up a
233    /// path, or a path of `call` may lead below a descriptor directory
234    /// (`Detcore::may_respell_below_descriptor_directory`). Where this is
235    /// false, that function answers `Ok(None)` at once, so a caller that asks
236    /// first may skip it.
237    pub(crate) fn may_translate_procfs_call<G: Guest<Self>>(guest: &G, call: &Syscall) -> bool {
238        (Self::translates_procfs_numbers(guest) && path_operands(call.number()).is_some())
239            || Self::may_respell_below_descriptor_directory(guest, call)
240    }

242    /// `call` with each path it looks up that names a process or thread by
243    /// the guest's number spelled with the kernel's numbers instead (see the
244    /// module documentation and [`Self::kernel_spelling`]): `Ok(None)` where
245    /// no path changes, and an error, never the untranslated call, where the
246    /// respelled paths cannot be staged. A path is respelled where it is
247    /// absolute, or relative to a directory the tool can resolve
248    /// ([`Self::relative_base`]).
249    ///
250    /// A path that leads below a descriptor directory Detcore serves is
251    /// respelled as Linux resolves it instead
252    /// (`Detcore::respelled_below_descriptor_directory`), and looked up from
253    /// the working directory (`AT_FDCWD`) at its absolute spelling. The
254    /// respelled paths are staged for this one call alone
255    /// ([`Self::stage_translated`]): in the guest stack scratch, or, where
256    /// that cannot hold them, a transient private page mapped for the call
257    /// and unmapped as it returns ([`Self::finish_translated_call`]); the
258    /// guest never learns either address. Every call so translated is one
259    /// that holds the turn while it runs (no path call is committed as a
260    /// background turn), so no other guest thread runs while they are
261    /// staged, outside `--no-sequentialize-threads`.
262    pub(crate) async fn procfs_translated_call<G: Guest<Self>>(
263        guest: &mut G,
264        call: &Syscall,
265    ) -> Result<Option<TranslatedCall>, Errno> {
266        Self::translated_call_with_scratch(guest, call, true).await
267    }

269    async fn translated_call_with_scratch<G: Guest<Self>>(
270        guest: &mut G,
271        call: &Syscall,
272        allow_scratch: bool,
273    ) -> Result<Option<TranslatedCall>, Errno> {
274        if !Self::may_translate_procfs_call(guest, call) {
275            return Ok(None);
276        }
277        let (sysno, args) = call.into_parts();
278        let mut values = [
279            args.arg0, args.arg1, args.arg2, args.arg3, args.arg4, args.arg5,
280        ];
281        if Self::may_respell_below_descriptor_directory(guest, call) {
282            // Boxed: the respelling's future is large, and every handler that
283            // injects a path call through this one would carry it inline,
284            // while a path below a placeholder descriptor is rare.
285            let respelled =
286                Box::pin(Self::respelled_below_descriptor_directory(guest, call)).await?;
287            if !respelled.is_empty() {
288                let staged = respelled
289                    .into_iter()
290                    .map(|operand| {
291                        if let Some(dirfd) = operand.dirfd {
292                            values[dirfd] = libc::AT_FDCWD as isize as usize;
293                        }
294                        let mut bytes = operand.bytes;
295                        bytes.push(0);
296                        (operand.path, bytes)
297                    })
298                    .collect();
299                return Self::stage_translated(guest, sysno, values, staged, allow_scratch)
300                    .await
301                    .map(Some);
302            }
303        }
304        if !Self::translates_procfs_numbers(guest) {
305            return Ok(None);
306        }
307        let Some(operands) = path_operands(sysno) else {
308            return Ok(None);
309        };
310        let mut respelled = Vec::new();
311        for &(dirfd, argument) in operands {
312            let Some(spelling) = PathPtr::from_ptr(values[argument] as *const libc::c_char)
313                .and_then(|address| address.read(&guest.memory()).ok())
314            else {
315                continue;
316            };
317            // Only a spelling that can name a process or a thread is looked
318            // at further.
319            if !spelling
320                .as_os_str()
321                .as_bytes()
322                .iter()
323                .any(u8::is_ascii_digit)
324            {
325                continue;
326            }
327            let base = if spelling.is_absolute() {
328                None
329            } else {
330                let dirfd = dirfd.map_or(libc::AT_FDCWD, |position| values[position] as i32);
331                match Self::relative_base(guest, dirfd) {
332                    Some(base) => Some(base),
333                    None => continue,
334                }
335            };
336            let dirfd = dirfd.map_or(libc::AT_FDCWD, |position| values[position] as i32);
337            if let Some(translated) = Self::kernel_spelling(guest, dirfd, base, &spelling).await {
338                let mut bytes = translated.into_os_string().into_vec();
339                bytes.push(0);
340                respelled.push((argument, bytes));
341            }
342        }
343        if respelled.is_empty() {
344            return Ok(None);
345        }
346        Self::stage_translated(guest, sysno, values, respelled, allow_scratch)
347            .await
348            .map(Some)
349    }

351    /// The call `sysno` with arguments `values`, each path argument in
352    /// `respelled` (by index, with its NUL) pointing at its bytes, staged as
353    /// `Detcore::inject_fstat` stages its buffer
354    /// (<https://github.com/rrnewton/hermit/issues/3328>): in the guest stack
355    /// scratch where the paths fit in [`SCRATCH_LIMIT`] bytes, the scratch
356    /// overlaps no other argument the call may read, and it is writable;
357    /// otherwise in a transient private page mapped for the call. The
358    /// scratch maps nothing, so the guest's address space and the placement
359    /// of its later mappings stay as they were, and nothing is left behind
360    /// if the thread is killed inside the call. Only where neither can hold
361    /// them does the call fail (ENOMEM) rather than reach a directory other
362    /// than the one the guest named.
363    async fn stage_translated<G: Guest<Self>>(
364        guest: &mut G,
365        sysno: reverie::syscalls::Sysno,
366        values: [usize; 6],
367        respelled: Vec<(usize, Vec<u8>)>,
368        allow_scratch: bool,
369    ) -> Result<TranslatedCall, Errno> {
370        if allow_scratch
371            && let Some(staged) = Self::stage_in_scratch(guest, sysno, values, &respelled).await
372        {
373            return Ok(staged);
374        }
375        Self::stage_in_page(guest, sysno, values, respelled).await
376    }

378    /// [`Self::stage_translated`] in the guest stack scratch, or `None`
379    /// where it cannot be: [`SCRATCH_GAP`] below the stack pointer and
380    /// down, memory the x86-64 ABI leaves dead (anything below the red zone
381    /// may be overwritten at any moment, by a signal frame), written with
382    /// the call's paths and read by the call alone, in the caller's turn. A
383    /// stack pointer too low, or memory that cannot be written there (a
384    /// guard page), falls back to the page.
385    async fn stage_in_scratch<G: Guest<Self>>(
386        guest: &mut G,
387        sysno: reverie::syscalls::Sysno,
388        mut values: [usize; 6],
389        respelled: &[(usize, Vec<u8>)],
390    ) -> Option<TranslatedCall> {
391        let length: usize = respelled.iter().map(|(_, bytes)| bytes.len()).sum();
392        if length > SCRATCH_LIMIT {
393            return None;
394        }
395        let rsp = usize::try_from(guest.regs().await.rsp).ok()?;
396        let top = rsp.checked_sub(SCRATCH_GAP)?;
397        let lowest = top.checked_sub(length)? & !15;
398        // Another argument the call reads (a second path, a buffer) must not
399        // lie in the scratch, nor in the page below it, where a buffer may
400        // start.
401        let respelled_arguments = respelled
402            .iter()
403            .map(|(argument, _)| *argument)
404            .collect::<Vec<_>>();
405        let overlaps = values.iter().enumerate().any(|(index, value)| {
406            !respelled_arguments.contains(&index)
407                && (lowest.saturating_sub(4096)..top).contains(value)
408        });
409        if overlaps {
410            return None;
411        }
412        let mut contents = Vec::with_capacity(length);
413        for (argument, bytes) in respelled {
414            values[*argument] = lowest + contents.len();
415            contents.extend_from_slice(bytes);
416        }
417        let target = AddrMut::<u8>::from_raw(lowest)?;
418        guest.memory().write_exact(target, &contents).ok()?;
419        Some(TranslatedCall {
420            call: Syscall::from_raw(sysno, raw_args(values)),
421            staging: Staging::Scratch,
422        })
423    }

425    /// [`Self::stage_translated`] in a transient private page.
426    async fn stage_in_page<G: Guest<Self>>(
427        guest: &mut G,
428        sysno: reverie::syscalls::Sysno,
429        mut values: [usize; 6],
430        respelled: Vec<(usize, Vec<u8>)>,
431    ) -> Result<TranslatedCall, Errno> {
432        let length: usize = respelled.iter().map(|(_, bytes)| bytes.len()).sum();
433        let page = map_transient(guest, length).await.ok_or(Errno::ENOMEM)?;
434        let mut contents = Vec::with_capacity(length);
435        for (argument, bytes) in &respelled {
436            values[*argument] = page.as_raw() + contents.len();
437            contents.extend_from_slice(bytes);
438        }
439        if guest.memory().write_exact(page, &contents).is_err() {
440            unmap_transient(guest, page, length).await;
441            return Err(Errno::ENOMEM);
442        }
443        Ok(TranslatedCall {
444            call: Syscall::from_raw(sysno, raw_args(values)),
445            staging: Staging::Page { page, length },
446        })
447    }

460    /// An auxiliary lookup whose translated paths must not borrow guest
461    /// stack scratch: the original syscall may still need bytes that are
462    /// absent from this lookup's arguments. Keep the same translation, but
463    /// stage changed paths in the existing transient page instead.
464    pub(crate) async fn inject_translated_in_page<G: Guest<Self>>(
465        guest: &mut G,
466        call: Syscall,
467    ) -> Result<i64, Errno> {
468        let translated = Self::translated_call_with_scratch(guest, &call, false).await?;
469        let injected = translated
470            .as_ref()
471            .map_or(call, |translated| translated.call);
472        let result = guest.inject(injected).await;
473        if let Some(translated) = translated {
474            Self::finish_translated_call(guest, translated).await;
475        }
476        result
477    }

479    /// [`Self::inject_translated`], made again while a signal interrupts it
480    /// where `retry` is set (`Guest::inject_with_retry`).
481    pub(crate) async fn inject_translated_retrying<G: Guest<Self>>(
482        guest: &mut G,
483        call: Syscall,
484        retry: bool,
485    ) -> Result<i64, Errno> {
486        let translated = Self::procfs_translated_call(guest, &call).await?;
487        let injected = translated
488            .as_ref()
489            .map_or(call, |translated| translated.call);
490        let result = if retry {
491            guest.inject_with_retry(injected).await
492        } else {
493            guest.inject(injected).await
494        };
495        if let Some(translated) = translated {
496            Self::finish_translated_call(guest, translated).await;
497        }
498        result
499    }

501    /// Release where a [`TranslatedCall`]'s paths were staged once its call
502    /// returned: drop the scratch guard, or unmap the page.
503    pub(crate) async fn finish_translated_call<G: Guest<Self>>(
504        guest: &mut G,
505        translated: TranslatedCall,
506    ) {
507        match translated.staging {
508            Staging::Scratch => {}
509            Staging::Page { page, length } => unmap_transient(guest, page, length).await,
510        }
511    }

531    /// The guest's spelling of the `/proc` directory open as `fd`, with the
532    /// kernel's numbers in the kernel's name for it replaced by the guest's
533    /// (see [`Self::translates_procfs_numbers`]): bound when the descriptor
534    /// was opened, as the kernel binds it, so a `/proc/self/task` a child
535    /// inherited still names its opener's threads. `None` for a descriptor
536    /// not below `/proc`, or one that names a process or thread `/proc` does
537    /// not list for the guest. The tool shares the guest's descriptor table
538    /// there (DBT), so its own view of the descriptor is the guest's.
539    pub(crate) async fn guest_procfs_spelling<G: Guest<Self>>(
540        guest: &mut G,
541        fd: std::os::fd::RawFd,
542    ) -> Option<PathBuf> {
543        if !Self::translates_procfs_numbers(guest)
544            || !guest.config().backend.tool_shares_guest_descriptor_table
545        {
546            return None;
547        }
548        let kernel = std::fs::read_link(format!("/proc/thread-self/fd/{fd}")).ok()?;
549        let held = ids::named_ids(&kernel);
550        if held.is_empty() {
551            return kernel.starts_with("/proc").then_some(kernel);
552        }
553        let numbers = Self::guest_procfs_numbers(guest, &held).await?;
554        if held.iter().any(|kernel| !numbers.contains_key(kernel)) {
555            return None;
556        }
557        Some(ids::translate(&kernel, |id| numbers.get(&id).copied()).unwrap_or(kernel))
558    }

560    /// The guest's number for each kernel process or thread number in `ids`,
561    /// where numbers are translated (see [`Self::translates_procfs_numbers`]),
562    /// as `/proc` spells it; `None` for a number of no process or thread
563    /// `/proc` lists, and for every number elsewhere.
564    pub(crate) async fn guest_procfs_numbers<G: Guest<Self>>(
565        guest: &mut G,
566        kernel: &[i32],
567    ) -> Option<BTreeMap<i32, i32>> {
568        if !Self::translates_procfs_numbers(guest) || kernel.is_empty() {
569            return None;
570        }
571        let guest_ids = crate::tool_global::procfs_ids(guest, false, kernel.to_vec()).await;
572        Some(
573            kernel
574                .iter()
575                .zip(guest_ids)
576                .filter_map(|(kernel, guest_id)| Some((*kernel, guest_id?)))
577                .collect(),
578        )
579    }
580}

582/// Open the components `names`, absolute or relative to `from`, as the
583/// kernel looks up a directory on the way to a path's last component
584/// (following every link), for its name only (`O_PATH`). `"."` for none.
585/// `None` where the kernel's lookup fails.
586fn open_path(from: RawFd, names: Vec<OsString>, absolute: bool) -> Option<OwnedFd> {
587    let mut path = if absolute {
588        PathBuf::from("/")
589    } else {
590        PathBuf::new()
591    };
592    path.extend(names);
593    if path.as_os_str().is_empty() {
594        path.push(".");
595    }
596    let path = CString::new(path.into_os_string().into_vec()).ok()?;
597    // SAFETY: a valid, NUL-terminated path; the result is a new descriptor
598    // or -1.
599    let fd = unsafe {
600        libc::openat(
601            from,
602            path.as_ptr(),
603            libc::O_PATH | libc::O_DIRECTORY | libc::O_CLOEXEC,
604        )
605    };
606    // SAFETY: a descriptor openat just returned, owned by nothing else.
607    (fd >= 0).then(|| unsafe { OwnedFd::from_raw_fd(fd) })
608}

610/// The kernel's name for the directory open as `fd`: absolute, with no link
611/// among its components and the kernel's numbers. `None` for a directory
612/// that was removed.
613fn resolved_name(fd: &OwnedFd) -> Option<PathBuf> {
614    let name = std::fs::read_link(format!("/proc/thread-self/fd/{}", fd.as_raw_fd())).ok()?;
615    (name.is_absolute() && !name.as_os_str().as_bytes().ends_with(b" (deleted)")).then_some(name)
616}
