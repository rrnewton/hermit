9//! A `/proc` path that names a process or thread by the guest's number
10//! reaches the kernel's directory for it where the kernel numbers processes
11//! otherwise than Detcore (DBT; slice B3a-4 of
12//! <https://github.com/rrnewton/hermit/issues/4266>; see
13//! [`crate::procfs::ids`]).
14//!
15//! The guest's call goes to the kernel with each such path spelled with the
16//! kernel's numbers instead, staged on a transient page, as the calls for a
17//! name outside the `--proc` level are (see [`super::procfs_levels`]). Every
18//! check before the call, and every decision Detcore takes about the path,
19//! sees the guest's spelling; only the kernel sees the other. A number that
20//! names no process or thread `/proc` lists is spelled `0`, which names
21//! nothing, so the kernel's process of that number, if any, is never reached.
22

55/// A call whose paths are respelled with the kernel's numbers, and the
56/// transient page holding them, which the caller unmaps once the call
57/// returns.
58pub(crate) struct TranslatedCall {
59    pub(crate) call: Syscall,
60    pub(crate) page: AddrMut<'static, u8>,
61    pub(crate) length: usize,
62}

64impl<T: RecordOrReplay> Detcore<T> {
65    /// Whether this guest thread's `/proc` paths may need translating: where
66    /// its backend registered the thread's kernel ids (DBT, and an in-guest
67    /// native exec, whose numberings agree, so its paths come back
68    /// unchanged), outside record and replay (whose replayer answers for the
69    /// recorded paths), a nested PID namespace (whose numbers are not the
70    /// container's) and a guest `chroot` (after which `/proc` need not be
71    /// procfs).
72    pub(crate) fn translates_procfs_numbers<G: Guest<Self>>(guest: &G) -> bool {
73        let state = guest.thread_state();
74        state.physical_tid.is_some()
75            && !guest.config().recordreplay_modes
76            && !state.nested_pid_namespace
77            && !state.changed_root.load(Ordering::SeqCst)
78    }

80    /// The directory the guest's relative spelling, looked up from `dirfd`,
81    /// starts at, resolved: the kernel's name for the directory `dirfd` is
82    /// open on, or for the working directory with `AT_FDCWD`, which the
83    /// tool, on the guest's own thread where numbers are translated (DBT),
84    /// reads as the guest's. It is spelled with the kernel's numbers, and
85    /// names no link. Only the number of its components, and whether it is
86    /// the init's, matter ([`ids::Lookup`]). `None` where the tool has no
87    /// view of the guest's descriptors, and for a directory that was
88    /// removed.
89    fn relative_base<G: Guest<Self>>(guest: &G, dirfd: i32) -> Option<PathBuf> {
90        if !guest.config().backend.tool_shares_guest_descriptor_table {
91            return None;
92        }
93        let link = if dirfd == libc::AT_FDCWD {
94            PathBuf::from("/proc/thread-self/cwd")
95        } else {
96            PathBuf::from(format!("/proc/thread-self/fd/{dirfd}"))
97        };
98        let base = std::fs::read_link(link).ok()?;
99        (base.is_absolute() && !base.as_os_str().as_bytes().ends_with(b" (deleted)"))
100            .then_some(base)
101    }

103    /// The guest's `spelling`, absolute, or relative to `dirfd` (or the
104    /// working directory for `AT_FDCWD`), whose resolved name is `base`, with
105    /// each number it names as a process or thread replaced in place by the
106    /// kernel's number for it (see the module documentation). Every other
107    /// component is kept, and a relative spelling stays relative, so the
108    /// kernel looks the path up as it would the guest's: from the same
109    /// directory, through the same links, counted against the same limit.
110    ///
111    /// Where the meaning of a component depends on where a link before it
112    /// leads (a number, or a `..` a number follows, past `cwd`, `root`,
113    /// `exe`, a descriptor's link, or a directory outside `/proc`), the
114    /// tool, on the guest's own thread, opens the components before it as
115    /// the kernel looks them up (`O_PATH`, from the same directory), and
116    /// walks the rest from the kernel's name for what it opened. That only
117    /// decides which components are numbers: what the tool opens is closed
118    /// again. Where it cannot be opened, the kernel's lookup of the same
119    /// components fails too, so the rest stays as it is. `None` where no
120    /// number changes.
121    async fn kernel_spelling<G: Guest<Self>>(
122        guest: &mut G,
123        dirfd: i32,
124        base: Option<PathBuf>,
125        spelling: &Path,
126    ) -> Option<PathBuf> {
127        let names = ids::components(spelling);
128        let shares = guest.config().backend.tool_shares_guest_descriptor_table;
129        let mut replaced: Vec<(usize, OsString)> = Vec::new();
130        let mut base = base;
131        let mut offset = 0;
132        // The directory the components from `offset` on are looked up from,
133        // where a resolution opened one.
134        let mut directory: Option<OwnedFd> = None;
135        while offset < names.len() {
136            let absolute = offset == 0 && spelling.is_absolute();
137            let rest = if absolute {
138                spelling.to_path_buf()
139            } else {
140                ids::rebuilt_like(names[offset..].to_vec(), Path::new(""))
141            };
142            let lookup = ids::Lookup::new(base.as_deref(), &rest);
143            let named = lookup
144                .ids()
145                .into_iter()
146                .collect::<BTreeSet<_>>()
147                .into_iter()
148                .collect::<Vec<_>>();
149            let map = if named.is_empty() {
150                BTreeMap::new()
151            } else {
152                let kernel = crate::tool_global::procfs_ids(guest, true, named.clone()).await;
153                named.into_iter().zip(kernel).collect::<BTreeMap<_, _>>()
154            };
155            replaced.extend(
156                lookup
157                    .translations(|id| map.get(&id).copied().flatten())
158                    .into_iter()
159                    .map(|(position, name)| (offset + position, name)),
160            );
161            // Only the tool on the guest's own thread looks up as the guest.
162            let Some(at) = lookup.stop().filter(|_| shares) else {
163                break;
164            };
165            let mut prefix = names[offset..offset + at].to_vec();
166            for (position, name) in &replaced {
167                if (offset..offset + at).contains(position) {
168                    prefix[position - offset] = name.clone();
169                }
170            }
171            let from = directory.as_ref().map_or(dirfd, |fd| fd.as_raw_fd());
172            let Some(opened) = open_path(from, prefix, absolute) else {
173                break;
174            };
175            directory = Some(opened);
176            offset += at;
177            base = directory.as_ref().and_then(resolved_name);
178            // A removed directory has no name to walk from, and no entries:
179            // only `.` and `..` lead on from it.
180            while base.is_none() && offset < names.len() {
181                let name = &names[offset];
182                if name != "." && name != ".." {
183                    break;
184                }
185                let from = directory.as_ref().map_or(dirfd, |fd| fd.as_raw_fd());
186                let Some(opened) = open_path(from, vec![name.clone()], false) else {
187                    break;
188                };
189                directory = Some(opened);
190                offset += 1;
191                base = directory.as_ref().and_then(resolved_name);
192            }
193            if base.is_none() {
194                break;
195            }
196        }
197        if replaced.is_empty() {
198            return None;
199        }
200        let mut names = names;
201        for (position, name) in replaced {
202            names[position] = name;
203        }
204        Some(ids::rebuilt_like(names, spelling))
205    }

207    /// `call` with each path it looks up that names a process or thread by
208    /// the guest's number spelled with the kernel's numbers instead (see the
209    /// module documentation and [`Self::kernel_spelling`]): `Ok(None)` where
210    /// no path changes, and an error, never the untranslated call, where the
211    /// respelled paths cannot be staged. A path is respelled where it is
212    /// absolute, or relative to a directory the tool can resolve
213    /// ([`Self::relative_base`]).
214    pub(crate) async fn procfs_translated_call<G: Guest<Self>>(
215        guest: &mut G,
216        call: &Syscall,
217    ) -> Result<Option<TranslatedCall>, Errno> {
218        if !Self::translates_procfs_numbers(guest) {
219            return Ok(None);
220        }
221        let (sysno, args) = call.into_parts();
222        let Some(Lookups::Paths(operands)) = looked_up(sysno) else {
223            return Ok(None);
224        };
225        let mut values = [
226            args.arg0, args.arg1, args.arg2, args.arg3, args.arg4, args.arg5,
227        ];
228        let mut respelled = Vec::new();
229        for (dirfd, argument) in operands {
230            let Some(spelling) = PathPtr::from_ptr(values[argument] as *const libc::c_char)
231                .and_then(|address| address.read(&guest.memory()).ok())
232            else {
233                continue;
234            };
235            // Only a spelling that can name a process or a thread is looked
236            // at further.
237            if !spelling
238                .as_os_str()
239                .as_bytes()
240                .iter()
241                .any(u8::is_ascii_digit)
242            {
243                continue;
244            }
245            let base = if spelling.is_absolute() {
246                None
247            } else {
248                let dirfd = dirfd.map_or(libc::AT_FDCWD, |position| values[position] as i32);
249                match Self::relative_base(guest, dirfd) {
250                    Some(base) => Some(base),
251                    None => continue,
252                }
253            };
254            let dirfd = dirfd.map_or(libc::AT_FDCWD, |position| values[position] as i32);
255            if let Some(translated) = Self::kernel_spelling(guest, dirfd, base, &spelling).await {
256                let mut bytes = translated.into_os_string().into_vec();
257                bytes.push(0);
258                respelled.push((argument, bytes));
259            }
260        }
261        if respelled.is_empty() {
262            return Ok(None);
263        }
264        let length: usize = respelled.iter().map(|(_, bytes)| bytes.len()).sum();
265        // With no room for the respelled paths, the call fails rather than
266        // reaching the kernel's process of the guest's number.
267        let page = map_transient(guest, length).await.ok_or(Errno::ENOMEM)?;
268        let mut contents = Vec::with_capacity(length);
269        for (argument, bytes) in &respelled {
270            values[*argument] = page.as_raw() + contents.len();
271            contents.extend_from_slice(bytes);
272        }
273        if guest.memory().write_exact(page, &contents).is_err() {
274            unmap_transient(guest, page, length).await;
275            return Err(Errno::ENOMEM);
276        }
277        Ok(Some(TranslatedCall {
278            call: Syscall::from_raw(sysno, raw_args(values)),
279            page,
280            length,
281        }))
282    }

284    /// Inject `call` directly, with its `/proc` paths translated as
285    /// [`Self::procfs_translated_call`] translates them, for the handlers and
286    /// the lookups Detcore makes that inject a call rather than hand it to
287    /// the record-or-replay tool.
288    pub(crate) async fn inject_translated<G: Guest<Self>>(
289        guest: &mut G,
290        call: Syscall,
291    ) -> Result<i64, Errno> {
292        let translated = Self::procfs_translated_call(guest, &call).await?;
293        let result = guest
294            .inject(
295                translated
296                    .as_ref()
297                    .map_or(call, |translated| translated.call),
298            )
299            .await;
300        if let Some(translated) = translated {
301            Self::finish_translated_call(guest, translated).await;
302        }
303        result
304    }

306    /// Unmap the page of a [`TranslatedCall`] once its call returned.
307    pub(crate) async fn finish_translated_call<G: Guest<Self>>(
308        guest: &mut G,
309        translated: TranslatedCall,
310    ) {
311        unmap_transient(guest, translated.page, translated.length).await;
312    }

314    /// The kernel's number for the guest's process or thread number `id`
315    /// where numbers are translated (see [`Self::translates_procfs_numbers`]),
316    /// `id` itself elsewhere, and `None` for a number of no process or thread
317    /// `/proc` lists.
318    pub(crate) async fn kernel_procfs_number<G: Guest<Self>>(
319        guest: &mut G,
320        id: i32,
321    ) -> Option<i32> {
322        if !Self::translates_procfs_numbers(guest) {
323            return Some(id);
324        }
325        crate::tool_global::procfs_ids(guest, true, vec![id])
326            .await
327            .into_iter()
328            .next()
329            .flatten()
330    }

332    /// The guest's spelling of the `/proc` directory open as `fd`, with the
333    /// kernel's numbers in the kernel's name for it replaced by the guest's
334    /// (see [`Self::translates_procfs_numbers`]): bound when the descriptor
335    /// was opened, as the kernel binds it, so a `/proc/self/task` a child
336    /// inherited still names its opener's threads. `None` for a descriptor
337    /// not below `/proc`, or one that names a process or thread `/proc` does
338    /// not list for the guest. The tool shares the guest's descriptor table
339    /// there (DBT), so its own view of the descriptor is the guest's.
340    pub(crate) async fn guest_procfs_spelling<G: Guest<Self>>(
341        guest: &mut G,
342        fd: std::os::fd::RawFd,
343    ) -> Option<PathBuf> {
344        if !Self::translates_procfs_numbers(guest)
345            || !guest.config().backend.tool_shares_guest_descriptor_table
346        {
347            return None;
348        }
349        let kernel = std::fs::read_link(format!("/proc/thread-self/fd/{fd}")).ok()?;
350        let held = ids::named_ids(&kernel);
351        if held.is_empty() {
352            return kernel.starts_with("/proc").then_some(kernel);
353        }
354        let numbers = Self::guest_procfs_numbers(guest, &held).await?;
355        if held.iter().any(|kernel| !numbers.contains_key(kernel)) {
356            return None;
357        }
358        Some(ids::translate(&kernel, |id| numbers.get(&id).copied()).unwrap_or(kernel))
359    }

361    /// The guest's number for each kernel process or thread number in `ids`,
362    /// where numbers are translated (see [`Self::translates_procfs_numbers`]),
363    /// as `/proc` spells it; `None` for a number of no process or thread
364    /// `/proc` lists, and for every number elsewhere.
365    pub(crate) async fn guest_procfs_numbers<G: Guest<Self>>(
366        guest: &mut G,
367        kernel: &[i32],
368    ) -> Option<BTreeMap<i32, i32>> {
369        if !Self::translates_procfs_numbers(guest) || kernel.is_empty() {
370            return None;
371        }
372        let guest_ids = crate::tool_global::procfs_ids(guest, false, kernel.to_vec()).await;
373        Some(
374            kernel
375                .iter()
376                .zip(guest_ids)
377                .filter_map(|(kernel, guest_id)| Some((*kernel, guest_id?)))
378                .collect(),
379        )
380    }

383/// Open the components `names`, absolute or relative to `from`, as the
384/// kernel looks up a directory on the way to a path's last component
385/// (following every link), for its name only (`O_PATH`). `"."` for none.
386/// `None` where the kernel's lookup fails.
387fn open_path(from: RawFd, names: Vec<OsString>, absolute: bool) -> Option<OwnedFd> {
388    let mut path = if absolute {
389        PathBuf::from("/")
390    } else {
391        PathBuf::new()
392    };
393    path.extend(names);
394    if path.as_os_str().is_empty() {
395        path.push(".");
396    }
397    let path = CString::new(path.into_os_string().into_vec()).ok()?;
398    // SAFETY: a valid, NUL-terminated path; the result is a new descriptor
399    // or -1.
400    let fd = unsafe {
401        libc::openat(
402            from,
403            path.as_ptr(),
404            libc::O_PATH | libc::O_DIRECTORY | libc::O_CLOEXEC,
405        )
406    };
407    // SAFETY: a descriptor openat just returned, owned by nothing else.
408    (fd >= 0).then(|| unsafe { OwnedFd::from_raw_fd(fd) })
409}

411/// The kernel's name for the directory open as `fd`: absolute, with no link
412/// among its components and the kernel's numbers. `None` for a directory
413/// that was removed.
414fn resolved_name(fd: &OwnedFd) -> Option<PathBuf> {
415    let name = std::fs::read_link(format!("/proc/thread-self/fd/{}", fd.as_raw_fd())).ok()?;
416    (name.is_absolute() && !name.as_os_str().as_bytes().ends_with(b" (deleted)")).then_some(name)
