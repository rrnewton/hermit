9//! Procfs files that Detcore renders without the backend opening them
10//! (<https://github.com/rrnewton/hermit/issues/4213>).
11//!
12//! For a file whose every byte comes from Detcore's own state
13//! (`ProcfsFile::rendered_path`), Detcore decides on every backend that it
14//! exists, what its metadata is and what it holds. (`/proc/cpuinfo` is one
15//! only where Detcore answers the guest's CPUID instruction from its own
16//! table, `procfs::RenderedFiles`; elsewhere it describes the host's CPU
17//! and the kernel opens it.) The backend opens no
18//! procfs file. The guest's descriptor is a placeholder: a memfd named
19//! `PLACEHOLDER_NAME_PREFIX` and the file's path, reopened read-only with
20//! the guest's own flags, a kernel object every backend has. Reopened that
21//! way, the kernel itself refuses writes, write locks and size changes, and
22//! reports the guest's status flags and a read-only descriptor link, as it
23//! does for the procfs file. Detcore answers the rest as Linux answers for a
24//! procfs file only global root may write (0444, or 0644 for the fixed
25//! tunables of `procfs::fixed`), to a process that is not global root. A
26//! per-process file (`/proc/<pid>/...`, `procfs::ProcfsTask`) belongs instead
27//! to its process's owner, the guest's own identity, so the guest owns it and
28//! its capabilities decide write access (`FileAccess`); a placeholder it opens
29//! for writing is reopened writable, and Detcore answers every write through
30//! it. Either way Detcore answers:
31//!
32//! - opening it, including the flag errors Linux reports;
33//! - reads, from the procfs snapshot, as for every procfs file;
34//! - `stat` of the path and `fstat` of the descriptor, with Detcore's procfs
35//!   device (`procfs_inode::VIRTUAL_PROC_DEVICE`) and an inode that its path
36//!   alone names (`procfs_inode::rendered_inode`), the same on every backend
37//!   and for every spelling that reaches the file;
38//! - the descriptor's link in any process's `/proc/*/fd`, and reopening
39//!   through it;
40//! - its `fdinfo`, `statfs`, and the calls such a procfs file refuses:
41//!   `mmap`, `fsync`, `fchmod`, extended attributes, file `ioctl`s and seals.
42//!
43//! Not under record and replay, whose Replayer stands its own placeholder in
44//! for every open, and not after the process changed its root with `chroot`,
45//! where the kernel decides whether the file exists. `epoll` refuses the
46//! placeholder (`EPERM`), which has no poll method, where it accepts the
47//! procfs file: a known limitation, see
48//! <https://github.com/rrnewton/hermit/issues/4263>.

50// determinization strategy: Detcore serves a file the kernel never opened.

121/// How many low bits of a carried uptime the nanoseconds of a placeholder's
122/// modification time hold ([`carried_uptime_mtime`]): fewer than 10^9 values,
123/// as `utimensat` requires.
124const CARRIED_UPTIME_NSEC_BITS: u32 = 29;
125
126/// The modification time that makes a placeholder carry `uptime_seconds`, the
127/// uptime of the first read of a file whose snapshot that read takes
128/// (`ProcfsFile::snapshot_at_first_read`): a time before 1970, the uptime's
129/// high bits counting down from -1 seconds and its low bits in the
130/// nanoseconds. Every `u64` has one. A memfd's own modification time is the
131/// host's time at its creation, which Linux never lets be before 1970
132/// (`settimeofday` refuses a negative time), so no placeholder carries an
133/// uptime it was not given.
134fn carried_uptime_mtime(uptime_seconds: u64) -> Timespec {
135    Timespec {
136        tv_sec: -1 - (uptime_seconds >> CARRIED_UPTIME_NSEC_BITS) as i64,
137        tv_nsec: (uptime_seconds & ((1 << CARRIED_UPTIME_NSEC_BITS) - 1)) as i64,
138    }
139}

162/// The uptime a placeholder whose modification time is `seconds` and
163/// `nanoseconds` carries ([`carried_uptime_mtime`]), if it carries one.
164fn carried_uptime(seconds: i64, nanoseconds: i64) -> Option<u64> {
165    if seconds >= 0 || !(0..1 << CARRIED_UPTIME_NSEC_BITS).contains(&nanoseconds) {
166        return None;
167    }
168    let high = (-1 - seconds) as u64;
169    (high < 1 << (64 - CARRIED_UPTIME_NSEC_BITS))
170        .then_some((high << CARRIED_UPTIME_NSEC_BITS) | nanoseconds as u64)
171}

275/// The rendered file that the kernel's file with metadata `stat` is, if it
276/// is one: the guest reached it through a spelling Detcore did not resolve,
277/// as a symbolic link or a directory listing, under a backend whose procfs
278/// files Detcore does not name by path
279/// (`procfs_inode::guest_tids_name_host_tasks`).
280///
281/// Each candidate costs a stat, so only a regular file with one link and
282/// permissions 0444 or 0644, as every rendered file is in the kernel's
283/// procfs, is checked, and an identity found not to be a rendered file is
284/// remembered (`NOT_RENDERED`): the checks of one procfs file cost their
285/// stats once per tool process. The search covers every file Detcore can
286/// render, so that what it remembers holds for any session; the answer is
287/// then the file only if this session renders it (`files`).
288pub(crate) fn rendered_path_of_raw(stat: &DetStat, files: RenderedFiles) -> Option<&'static Path> {
289    let has = |field| stat.mask.contains(field);
290    if (has(StatxMask::STATX_TYPE) && stat.mode & libc::S_IFMT != libc::S_IFREG)
291        || (has(StatxMask::STATX_NLINK) && stat.nlink != 1)
292        || (has(StatxMask::STATX_MODE) && !matches!(stat.mode & 0o7777, 0o444 | 0o644))
293    {
294        return None;
295    }
296    let identity = (stat.dev, stat.inode);
297    if NOT_RENDERED
298        .lock()
299        .expect("raw identity cache poisoned")
300        .contains(&identity)
301    {
302        return None;
303    }
304    let found = rendered_path_among(stat.dev, stat.inode, rendered_paths(RenderedFiles::ALL));
305    if found.is_none() && proc_device() == Some(stat.dev) {
306        NOT_RENDERED
307            .lock()
308            .expect("raw identity cache poisoned")
309            .insert(identity);
310    }
311    found
312        .and_then(ProcfsFile::from_path)
313        .and_then(|file| file.rendered_path(files))
314}

331/// The rendered file among `paths` that the kernel's procfs inode
332/// `(dev, ino)` is, if it is one.
333fn rendered_path_among(
334    dev: u64,
335    ino: u64,
336    mut paths: impl Iterator<Item = &'static std::ffi::CStr>,
337) -> Option<&'static Path> {
338    if proc_device() != Some(dev) {
339        return None;
340    }
341    paths.find_map(|path| {
342        let stat = GuestProcfs::stat(path).ok()?;
343        (stat.st_dev == dev && stat.st_ino == ino)
344            .then(|| Path::new(OsStr::from_bytes(path.to_bytes())))
345            .and_then(|path| ProcfsFile::from_path(path)?.rendered_path(RenderedFiles::ALL))
346    })
347}

349/// Facts about the guest's procfs mount that a rendered file shares with the
350/// kernel's procfs files. The tool runs in the guest's mount and user
351/// namespaces on every backend outside record and replay, so its `/proc` is
352/// the guest's. Read with raw system calls: under in-guest backends the tool
353/// shares the guest's C library.
354///
355/// Each fact is read once, when `/proc` is first reachable, and kept: the
356/// container's procfs mount does not change during the run, while the tool's
357/// own view of it can disappear, as when an in-guest tool runs in a process
358/// that later changed its root. A fact that cannot be read is the errno the
359/// guest's call gets, never a failure of the run.
360struct GuestProcfs;
361
362impl GuestProcfs {
363    fn stat(path: &std::ffi::CStr) -> Result<libc::stat, Errno> {
364        let mut stat = std::mem::MaybeUninit::<libc::stat>::zeroed();
365        // SAFETY: the path is NUL-terminated and `stat` is writable.
366        let result = unsafe {
367            raw_syscall(
368                libc::SYS_newfstatat,
369                [
370                    libc::AT_FDCWD as u64,
371                    path.as_ptr() as u64,
372                    stat.as_mut_ptr() as u64,
373                    0,
374                    0,
375                    0,
376                ],
377            )
378        };
379        raw_errno(result)?;
380        // SAFETY: the call succeeded and initialized the structure.
381        Ok(unsafe { stat.assume_init() })
382    }
383
384    /// The owner Linux reports for the guest's procfs sysctl files, and for
385    /// the other system-wide procfs files (`/proc/version`, `/proc/cmdline`):
386    /// global root as the guest's user namespace sees it.
387    fn sysctl_owner() -> Result<(u32, u32), Errno> {
388        static OWNER: std::sync::OnceLock<(u32, u32)> = std::sync::OnceLock::new();
389        if let Some(owner) = OWNER.get() {
390            return Ok(*owner);
391        }
392        let stat = Self::stat(c"/proc/sys/kernel")?;
393        Ok(*OWNER.get_or_init(|| (stat.st_uid, stat.st_gid)))
394    }
395
396    /// The guest's procfs statistics, and its mount flags (`f_flags`, which
397    /// the C library's structure keeps private).
398    fn statfs() -> Result<(libc::statfs, u64), Errno> {
399        // The kernel's `struct statfs` on x86_64: fifteen 64-bit words, of
400        // which the eleventh is `f_flags`.
401        const WORDS: usize = 15;
402        const _: () = assert!(std::mem::size_of::<libc::statfs>() == WORDS * 8);
403        static WORDS_READ: std::sync::OnceLock<[u64; WORDS]> = std::sync::OnceLock::new();
404        let words = match WORDS_READ.get() {
405            Some(words) => *words,
406            None => {
407                let mut words = [0_u64; WORDS];
408                // SAFETY: the path is NUL-terminated and `words` is writable
409                // and as large as the kernel's structure.
410                let result = unsafe {
411                    raw_syscall(
412                        libc::SYS_statfs,
413                        [
414                            c"/proc".as_ptr() as u64,
415                            words.as_mut_ptr() as u64,
416                            0,
417                            0,
418                            0,
419                            0,
420                        ],
421                    )
422                };
423                raw_errno(result)?;
424                *WORDS_READ.get_or_init(|| words)
425            }
426        };
427        let flags = words[10];
428        // SAFETY: the two have the same size and layout, of plain integers.
429        let statfs = unsafe { std::mem::transmute::<[u64; WORDS], libc::statfs>(words) };
430        Ok((statfs, flags))
431    }

433    /// The raw mount ID of the guest's procfs mount, as Linux reports it in
434    /// the fdinfo of a procfs file.
435    fn mount_id() -> Result<u64, Error> {
436        static MOUNT_ID: std::sync::OnceLock<u64> = std::sync::OnceLock::new();
437        if let Some(mount_id) = MOUNT_ID.get() {
438            return Ok(*mount_id);
439        }
440        let mut statx = std::mem::MaybeUninit::<libc::statx>::zeroed();
441        // SAFETY: the path is NUL-terminated and `statx` is writable.
442        let result = unsafe {
443            raw_syscall(
444                libc::SYS_statx,
445                [
446                    libc::AT_FDCWD as u64,
447                    c"/proc".as_ptr() as u64,
448                    0,
449                    u64::from(libc::STATX_MNT_ID),
450                    statx.as_mut_ptr() as u64,
451                    0,
452                ],
453            )
454        };
455        raw_errno(result)?;
456        // SAFETY: the call succeeded and initialized the structure.
457        let statx = unsafe { statx.assume_init() };
458        if statx.stx_mask & libc::STATX_MNT_ID == 0 {
459            // A kernel too old to report it: the guest's read fails with
460            // the errno a missing feature gives, not the run.
461            return Err(Errno::EOPNOTSUPP.into());
462        }
463        Ok(*MOUNT_ID.get_or_init(|| statx.stx_mnt_id))
464    }

466    /// Whether the guest's user namespace maps user ID `id` (`gid` false) or
467    /// group ID `id` (`gid` true). -1 never names an ID.
468    fn maps_id(gid: bool, id: u32) -> bool {
469        static UID_MAP: std::sync::OnceLock<Vec<u8>> = std::sync::OnceLock::new();
470        static GID_MAP: std::sync::OnceLock<Vec<u8>> = std::sync::OnceLock::new();
471        if id == u32::MAX {
472            return false;
473        }
474        let (cell, path) = if gid {
475            (&GID_MAP, c"/proc/self/gid_map")
476        } else {
477            (&UID_MAP, c"/proc/self/uid_map")
478        };
479        let map = match cell.get() {
480            Some(map) => map,
481            None => match Self::read_small(path) {
482                Some(map) => cell.get_or_init(|| map),
483                None => return false,
484            },
485        };
486        String::from_utf8_lossy(map).lines().any(|line| {
487            let fields: Vec<u64> = line
488                .split_whitespace()
489                .filter_map(|field| field.parse().ok())
490                .collect();
491            matches!(fields.as_slice(), [inside, _, count]
492                if (*inside..inside.saturating_add(*count)).contains(&u64::from(id)))
493        })
494    }

496    /// The first 4 KiB of the small procfs file at `path`, read with raw
497    /// system calls.
498    fn read_small(path: &std::ffi::CStr) -> Option<Vec<u8>> {
499        // SAFETY: the path is NUL-terminated.
500        let fd = unsafe {
501            raw_syscall(
502                libc::SYS_openat,
503                [
504                    libc::AT_FDCWD as u64,
505                    path.as_ptr() as u64,
506                    (libc::O_RDONLY | libc::O_CLOEXEC) as u64,
507                    0,
508                    0,
509                    0,
510                ],
511            )
512        };
513        let fd = raw_errno(fd).ok()?;
514        let mut buffer = vec![0_u8; 4096];
515        // SAFETY: `buffer` is writable for its length.
516        let length = unsafe {
517            raw_syscall(
518                libc::SYS_read,
519                [fd, buffer.as_mut_ptr() as u64, buffer.len() as u64, 0, 0, 0],
520            )
521        };
522        // SAFETY: closing the descriptor opened above.
523        unsafe { raw_syscall(libc::SYS_close, [fd, 0, 0, 0, 0, 0]) };
524        buffer.truncate(raw_errno(length).ok()? as usize);
525        Some(buffer)
526    }

528    /// Read every fact while `/proc` is reachable.
529    fn remember() {
530        let _ = Self::sysctl_owner();
531        let _ = Self::statfs();
532        let _ = Self::mount_id();
533        let _ = Self::maps_id(false, 0);
534        let _ = Self::maps_id(true, 0);
535    }
536}

596/// What decides Linux's answer to a call that asks for write access to a
597/// rendered file, or for its owner: whether the caller owns it, and the
598/// capabilities that override its permissions.
599///
600/// A system-wide file belongs to global root, whom the guest's user
601/// namespace does not map, so no capability of the guest overrides its
602/// permissions and the guest is not its owner ([`FileAccess::GLOBAL_ROOT`]).
603/// A per-process file belongs to its process's owner, Detcore's virtual root,
604/// which is every guest's own identity, so the guest owns it, and
605/// `CAP_DAC_OVERRIDE` overrides its 0444 mode; `CAP_SYS_ADMIN` in procfs's
606/// user namespace decides a `security.` attribute. Both come from the
607/// guest's own capability sets, as `capget` reports them
608/// (`Detcore::rendered_file_access`).
609#[derive(Clone, Copy, Debug, Eq, PartialEq)]
610pub(super) struct FileAccess {
611    /// The caller owns the file.
612    owner: bool,
613    /// `CAP_DAC_OVERRIDE`: write access despite the mode.
614    dac_override: bool,
615    /// `CAP_SYS_ADMIN`: a `security.` attribute may be changed.
616    sys_admin: bool,
617}
618
619impl FileAccess {
620    /// A system-wide rendered file, owned by global root.
621    pub(super) const GLOBAL_ROOT: Self = Self {
622        owner: false,
623        dac_override: false,
624        sys_admin: false,
625    };
626
627    /// A per-process file, which the caller owns, with the capabilities
628    /// `capabilities` (a `capget` set) of the caller.
629    fn owned_with(capabilities: u64) -> Self {
630        Self {
631            owner: true,
632            dac_override: capabilities & (1 << CAP_DAC_OVERRIDE) != 0,
633            sys_admin: capabilities & (1 << CAP_SYS_ADMIN) != 0,
634        }
635    }
636
637    /// Whether the caller may write the file: every rendered file's mode
638    /// grants no one write access (0444; a fixed tunable's 0644 grants only
639    /// global root), so only `CAP_DAC_OVERRIDE` over an owner the caller's
640    /// namespace maps does.
641    fn may_write(&self) -> bool {
642        self.dac_override
643    }
644}

681/// Linux's answer to opening an existing rendered procfs file (writable by
682/// no one but global root, `access`) with `flags`, from a process that is not
683/// global root, before it opens anything: `build_open_flags`, then the
684/// lookup's `O_EXCL` and type checks, then `may_open` and `do_dentry_open`.
685/// On success, the flags the placeholder is reopened with: read-only, or with
686/// the guest's access mode where it may write (a per-process file under
687/// `CAP_DAC_OVERRIDE`, whose writes Detcore then answers,
688/// `answer_rendered_procfs_operation`), never with `O_TRUNC`, which truncates
689/// nothing in procfs.
690fn rendered_open_flags(flags: OFlag, access: FileAccess) -> Result<OFlag, Errno> {
691    if let Some(errno) = open_flags_refused(flags) {
692        return Err(errno);
693    }
694    // `O_PATH` keeps only its own flags (`O_TMPFILE` includes `O_DIRECTORY`).
695    if flags.contains(OFlag::O_PATH) {
696        if flags.contains(OFlag::O_DIRECTORY) {
697            return Err(Errno::ENOTDIR);
698        }
699        return Ok(OFlag::O_PATH | (flags & OFlag::O_CLOEXEC));
700    }
701    // A temporary file is made in a directory, which a file is not.
702    if flags.contains(OFlag::O_TMPFILE) {
703        return Err(Errno::ENOTDIR);
704    }
705    if flags.contains(OFlag::O_CREAT | OFlag::O_EXCL) {
706        return Err(Errno::EEXIST);
707    }
708    if flags.contains(OFlag::O_DIRECTORY) {
709        return Err(Errno::ENOTDIR);
710    }
711    // No mode bit allows writing; `O_TRUNC` asks for write access.
712    let writes = flags & OFlag::O_ACCMODE != OFlag::O_RDONLY || flags.contains(OFlag::O_TRUNC);
713    if writes && !access.may_write() {
714        return Err(Errno::EACCES);
715    }
716    // `O_NOATIME` needs the file's owner.
717    if flags.contains(OFlag::O_NOATIME) && !access.owner {
718        return Err(Errno::EPERM);
719    }
720    // Procfs files cannot do direct I/O.
721    if flags.contains(OFlag::O_DIRECT) {
722        return Err(Errno::EINVAL);
723    }
724    let kept = OFlag::O_APPEND
725        | OFlag::O_ASYNC
726        | OFlag::O_CLOEXEC
727        | OFlag::O_DSYNC
728        | OFlag::O_LARGEFILE
729        | OFlag::O_NOATIME
730        | OFlag::O_NONBLOCK
731        | OFlag::O_SYNC;
732    let mode = flags & OFlag::O_ACCMODE;
733    Ok(mode | (flags & kept))
734}

736/// What `build_open_flags` refuses with EINVAL before any path is looked up,
737/// once `build_open_how` has kept only `O_PATH`'s own flags: a malformed
738/// `__O_TMPFILE` (it needs `O_DIRECTORY` and no `O_CREAT`), a temporary file
739/// without write access, and `O_CREAT` with `O_DIRECTORY`.
740pub(super) fn open_flags_refused(flags: OFlag) -> Option<Errno> {
741    if flags.contains(OFlag::O_PATH) {
742        return None;
743    }
744    let tmpfile = OFlag::O_TMPFILE & !OFlag::O_DIRECTORY;
745    if flags.intersects(tmpfile)
746        && (flags & (OFlag::O_TMPFILE | OFlag::O_CREAT) != OFlag::O_TMPFILE
747            || flags & OFlag::O_ACCMODE == OFlag::O_RDONLY)
748    {
749        return Some(Errno::EINVAL);
750    }
751    flags
752        .contains(OFlag::O_CREAT | OFlag::O_DIRECTORY)
753        .then_some(Errno::EINVAL)
754}

756/// The checks `do_mmap` makes of a mapping of a procfs file before it looks
757/// for a mapping the range overlaps (`MAP_FIXED_NOREPLACE`'s `EEXIST`), in
758/// its order: the arguments and the address. Returns the page-aligned length.
759fn rendered_mmap_address_checks(
760    addr: usize,
761    len: usize,
762    offset: libc::off_t,
763    flags: MapFlags,
764) -> Result<usize, Errno> {
765    // `ksys_mmap_pgoff` refuses MAP_HUGETLB for a file that is not on
766    // hugetlbfs before `do_mmap` looks at anything.
767    if flags.contains(MapFlags::MAP_HUGETLB) || len == 0 || offset & 4095 != 0 {
768        return Err(Errno::EINVAL);
769    }
770    // A length that cannot fit the address space fails before the file is
771    // looked at (`PAGE_ALIGN` overflowing, then `__get_unmapped_area`).
772    // The 47-bit user address space of x86_64 without 5-level paging, which
773    // Hermit assumes: on a host with LA57, Linux would take a higher fixed
774    // address and answer ENODEV where this answers ENOMEM.
775    const TASK_SIZE: usize = 0x7fff_ffff_f000;
776    let Some(len) = len.checked_add(4095).map(|length| length & !4095) else {
777        return Err(Errno::ENOMEM);
778    };
779    if len > TASK_SIZE {
780        return Err(Errno::ENOMEM);
781    }
782    // A fixed address is taken as given; then it must leave room for the
783    // mapping below TASK_SIZE, and be page aligned (measured: EINVAL before
784    // ENODEV).
785    if flags.intersects(MapFlags::MAP_FIXED | MapFlags::MAP_FIXED_NOREPLACE) {
786        if addr > TASK_SIZE - len {
787            return Err(Errno::ENOMEM);
788        }
789        if addr & 4095 != 0 {
790            return Err(Errno::EINVAL);
791        }
792        // `security_mmap_addr`: below `vm.mmap_min_addr` (the machine's, which
793        // does not depend on the configuration) needs CAP_SYS_RAWIO, which a
794        // guest lacks.
795        let mmap_min_addr = crate::machine::MachineDescription::default()
796            .limits
797            .mmap_min_addr;
798        if (addr as u64) < mmap_min_addr {
799            return Err(Errno::EPERM);
800        }
801    }
802    Ok(len)
803}

805/// The checks `do_mmap` and `mmap_region` make of a mapping of a procfs file
806/// Detcore renders after the overlap check, in their order, and the error the
807/// mapping ends with: a sysctl file has no `mmap` method (`ENODEV`); a
808/// system-wide `seq_file` has `proc_reg_mmap` (`procfs::rendered_path_is_seq_file`),
809/// so a growing-down mapping is refused (`EINVAL`) and otherwise the method
810/// finds nothing behind it (`EIO`); a per-process file's operations have no
811/// `mmap` (`ENODEV`, `procfs::rendered_path_is_per_process`). `len` is page
812/// aligned. `noexec` says whether procfs is mounted `noexec`. `readable` and
813/// `writable` are the descriptor's access mode: only a per-process file's can
814/// be writable, or not readable.
815fn rendered_mmap_file_checks(
816    len: usize,
817    offset: libc::off_t,
818    prot: ProtFlags,
819    flags: MapFlags,
820    noexec: bool,
821    seq_file: bool,
822    (readable, writable): (bool, bool),
823) -> Errno {
824    // `file_mmap_ok`: the mapping must end within the largest offset a
825    // regular file can have (`MAX_LFS_FILESIZE`), the offset taken unsigned.
826    const MAX_LFS_FILESIZE: u64 = i64::MAX as u64;
827    if offset as u64 / 4096 > (MAX_LFS_FILESIZE - len as u64) >> 12 {
828        return Errno::EOVERFLOW;
829    }
830    let map_type = flags.bits() & libc::MAP_TYPE;
831    if map_type != libc::MAP_SHARED
832        && map_type != libc::MAP_PRIVATE
833        && map_type != libc::MAP_SHARED_VALIDATE
834    {
835        return Errno::EINVAL;
836    }
837    // MAP_SHARED_VALIDATE refuses every flag outside `LEGACY_MAP_MASK`, as
838    // procfs supports none (`mmap_supported_flags`): MAP_SYNC and
839    // MAP_FIXED_NOREPLACE among them.
840    if map_type == libc::MAP_SHARED_VALIDATE && flags.bits() & !LEGACY_MAP_MASK != 0 {
841        return Errno::EOPNOTSUPP;
842    }
843    if map_type != libc::MAP_PRIVATE && prot.contains(ProtFlags::PROT_WRITE) && !writable {
844        return Errno::EACCES;
845    }
846    // Every mapping needs a readable descriptor (`MAP_SHARED` falls through
847    // to `MAP_PRIVATE`'s check).
848    if !readable {
849        return Errno::EACCES;
850    }
851    if noexec && prot.contains(ProtFlags::PROT_EXEC) {
852        return Errno::EPERM;
853    }
854    if !seq_file {
855        return Errno::ENODEV;
856    }
857    if flags.contains(MapFlags::MAP_GROWSDOWN) {
858        return Errno::EINVAL;
859    }
860    Errno::EIO
861}

864/// under MAP_SHARED_VALIDATE, with the type bits.

866    | libc::MAP_PRIVATE
867    | libc::MAP_FIXED
868    | libc::MAP_ANONYMOUS
869    | libc::MAP_DENYWRITE
870    | libc::MAP_EXECUTABLE
871    | 0x400_0000 // MAP_UNINITIALIZED
872    | libc::MAP_GROWSDOWN
873    | libc::MAP_LOCKED
874    | libc::MAP_NORESERVE
875    | libc::MAP_POPULATE
876    | libc::MAP_NONBLOCK
877    | libc::MAP_STACK
878    | libc::MAP_HUGETLB
879    | libc::MAP_32BIT
880    | 0x80 // MAP_ABOVE4G
881    | (21 << 26) // MAP_HUGE_2MB
882    | (30 << 26); // MAP_HUGE_1GB

886/// [`rendered_mmap_address_checks`], then [`rendered_mmap_file_checks`]. The

927/// Linux's answer to setting the extended attribute `name` of a procfs file,
928/// which has no attribute handlers, to `value` (`size` bytes, `None` if the
929/// guest's buffer cannot be read), or to removing it (`size` is `None`), in
930/// the kernel's order: the name, the size, the value, then the namespace.
931/// Measured against /proc/sys/kernel/random/uuid as a user its user namespace
932/// does not own ([`FileAccess::GLOBAL_ROOT`]), and against /proc/self/cgroup
933/// as its owner with and without the capabilities `access` names.
934fn xattr_change_answer(
935    name: &[u8],
936    size: Option<usize>,
937    value: Option<&[u8]>,
938    access: FileAccess,
939) -> Result<i64, Errno> {
940    if name.is_empty() || name.len() > 255 {
941        return Err(Errno::ERANGE);
942    }
943    if size.is_some_and(|size| size > XATTR_SIZE_MAX) {
944        return Err(Errno::E2BIG);
945    }
946    if size.is_some_and(|size| size > 0) && value.is_none() {
947        return Err(Errno::EFAULT);
948    }
949    let setting = size.is_some();
950    if name.starts_with(b"trusted.") {
951        Err(Errno::EPERM)
952    } else if name == b"system.posix_acl_access" || name == b"system.posix_acl_default" {
953        // A list is parsed first; procfs then has no way to keep it.
954        if setting && posix_acl_parser_refuses(value.unwrap_or_default()) {
955            Err(Errno::EINVAL)
956        } else {
957            Err(Errno::EOPNOTSUPP)
958        }
959    } else if name == b"security.capability" && setting {
960        // The capability module parses the value first.
961        Err(Errno::EINVAL)
962    } else if name.starts_with(b"security.") {
963        // Changing a security attribute needs CAP_SYS_ADMIN; procfs then has
964        // no way to keep it.
965        Err(if access.sys_admin {
966            Errno::EOPNOTSUPP
967        } else {
968            Errno::EPERM
969        })
970    } else if name.starts_with(b"system.") {
971        Err(Errno::EOPNOTSUPP)
972    } else {
973        // `user.` and unknown namespaces need write permission first.
974        Err(if access.may_write() {
975            Errno::EOPNOTSUPP
976        } else {
977            Errno::EACCES
978        })
979    }
980}

982/// Whether the kernel's parser (`posix_acl_from_xattr`) refuses `value` as a
983/// POSIX access control list: a value too short for its header, a size that
984/// is not a whole number of entries, an unknown tag, or a named user or group
985/// the caller's user namespace does not map. A bad version, an empty value or
986/// an empty list are not refused here; procfs then has no way to keep any
987/// list (EOPNOTSUPP). The checks of a list's order, permissions and counts
988/// (`posix_acl_valid`) come after that, so a procfs file never reaches them.
989/// Measured on Linux 7.1, as a user and as root in a user namespace.
990fn posix_acl_parser_refuses(value: &[u8]) -> bool {
991    if value.is_empty() {
992        return false;
993    }
994    let Some((header, entries)) = value.split_first_chunk::<4>() else {
995        return true;
996    };
997    if u32::from_le_bytes(*header) != 2 {
998        return false;
999    }
1000    if entries.len() % 8 != 0 {
1001        return true;
1002    }
1003    entries.as_chunks::<8>().0.iter().any(|entry| {
1004        let tag = u16::from_le_bytes([entry[0], entry[1]]);
1005        let id = u32::from_le_bytes([entry[4], entry[5], entry[6], entry[7]]);
1006        match tag {
1007            // ACL_USER_OBJ, ACL_GROUP_OBJ, ACL_MASK, ACL_OTHER.
1008            0x01 | 0x04 | 0x10 | 0x20 => false,
1009            0x02 => !GuestProcfs::maps_id(false, id),
1010            0x08 => !GuestProcfs::maps_id(true, id),
1011            _ => true,
1012        }
1013    })
1014}

1096    ) -> Result<(), Errno> {

1098            .thread_state()
1099            .with_detfd(fd, |detfd| detfd.procfs_rendered_path())
1100            .ok()

1156    /// `/dev/fd/N` link to a descriptor Detcore rendered. A spelling that
1157    /// asks for a directory fails as Linux fails it.
1158    pub(super) async fn rendered_procfs_at<G: Guest<Self>>(
1159        &self,
1160        guest: &mut G,
1161        path: &Path,
1162    ) -> Result<Option<RenderedTarget>, Error> {
1163        // A `..` component is the kernel's to resolve: it walks through the
1164        // components before it, which need not exist or be directories, while
1165        // Detcore's names drop them. (A valid `..` spelling of a rendered file
1166        // so reaches the kernel's own: a known limitation, see
1167        // <https://github.com/rrnewton/hermit/issues/4263>.)
1168        if !self.renders_procfs(guest)
1169            || path
1170                .components()
1171                .any(|component| component == std::path::Component::ParentDir)
1172        {
1173            return Ok(None);
1174        }
1175        if let Some(mut procfs) = ProcfsFile::from_path(path) {
1176            // A per-process file belongs to the task its path names, which
1177            // Detcore's process table decides exists, as Linux's lookup of
1178            // `/proc/<pid>` and `task/<tid>` does: a task that is not there
1179            // has no file (ENOENT), before any other check. In a nested PID
1180            // namespace, whose numbers are its own, the kernel decides.
1181            // AUTONOMOUS-BOT-IMPLEMENTED
1182            // TODO-HUMAN-REVIEW(#4213): Review per-process rendered files: the
1183            // task a spelling names, its existence and identity from Detcore's
1184            // process table, and its owner's access (FileAccess).
1185            if procfs.per_process_name().is_some() {
1186                let Some((pid, tid)) = self.procfs_caller_ids(guest) else {
1187                    return Ok(None);
1188                };
1189                procfs.bind_task(pid, tid);
1190                let Some(task) = procfs.bound_task() else {
1191                    return Err(Errno::ENOENT.into());
1192                };
1193                let answer = crate::tool_global::procfs_task(guest, task).await;
1194                if !answer.exists {
1195                    return Err(Errno::ENOENT.into());
1196                }
1197                self.bind_task_identity(guest, &mut procfs, answer).await?;
1198            }
1199            if procfs.rendered_path(self.rendered_files(guest)).is_none() {
1200                return Ok(None);
1201            }
1202            if names_a_directory(path) {
1203                return Err(Errno::ENOTDIR.into());
1204            }
1205            return Ok(Some(RenderedTarget {
1206                procfs,
1207                via_link: false,
1208            }));
1209        }
1210        // procfs names descriptors without leading zeros, so `fd/03` does not
1211        // exist even where descriptor 3 is a placeholder. Detcore answers so
1212        // on every backend (KVM's own procfs resolves the spelling).
1213        if let Some((subject, fd)) = proc_fd_target_with_leading_zeros(path) {
1214            let own = match subject {
1215                Some(subject) => self.guest_visible_own_pid(guest).await? == i64::from(subject),
1216                None => true,
1217            };

1219                .thread_state()
1220                .with_detfd(fd, |detfd| detfd.procfs_rendered_path().is_some())
1221                .unwrap_or(false);
1222            if own && rendered {
1223                return Err(Errno::ENOENT.into());
1224            }
1225            return Ok(None);
1226        }
1227        let Some((subject, fd)) = proc_fd_target(path) else {
1228            return Ok(None);
1229        };
1230        if let Some(subject) = subject
1231            && self.guest_visible_own_pid(guest).await? != i64::from(subject)
1232        {
1233            return Ok(None);
1234        }
1235        // The descriptor is a placeholder, so the file it reopens is one
1236        // Detcore opened as rendered: the target keeps that state, whatever
1237        // the session renders by path.
1238        let target = guest
1239            .thread_state()
1240            .with_detfd(fd, |detfd| detfd.procfs_rendered_path())
1241            .ok()
1242            .flatten()
1243            .and_then(ProcfsFile::from_rendered_path)
1244            .map(|procfs| RenderedTarget {
1245                procfs: procfs.into_rendered(),
1246                via_link: true,
1247            });
1248        // The link reaches a file, which a trailing slash or `/.` refuses.
1249        if target.is_some() && names_a_directory(path) {
1250            return Err(Errno::ENOTDIR.into());
1251        }
1252        Ok(target)
1253    }

1255    /// The rendered procfs file that descriptor `fd`, which the kernel just
1256    /// opened, is a placeholder for: one the guest reached through a link
1257    /// Detcore does not resolve itself, such as another process's
1258    /// `/proc/<pid>/fd/N` or `/proc/*/task/*/fd/N`. A placeholder this tool
1259    /// process made is known by its kernel identity. One it did not (another
1260    /// process's, under a backend whose tool runs in each guest process) is
1261    /// known by its memfd's name and seals, which a placeholder carries where
1262    /// the backend supports seals.
1263    pub(super) async fn rendered_procfs_behind<G: Guest<Self>>(
1264        &self,
1265        guest: &mut G,
1266        fd: RawFd,
1267    ) -> Result<Option<&'static Path>, Error> {
1268        if !self.renders_procfs(guest) {
1269            return Ok(None);
1270        }
1271        if let Ok(stat) = self.inject_fstat(guest, fd).await
1272            && let Some(path) = placeholder_identity(stat.st_dev, stat.st_ino)
1273        {
1274            return Ok(Some(path));
1275        }
1276        // Where the tool runs in one process for every guest, it made every
1277        // placeholder, and its identity is the whole answer. Elsewhere a name
1278        // and seals a guest could imitate decide: a known limitation, see
1279        // <https://github.com/rrnewton/hermit/issues/4263>.
1280        if !self.cfg.backend.tool_shares_guest_descriptor_table {
1281            return Ok(None);
1282        }

1284        if named.is_none() {
1285            return Ok(None);
1286        }
1287        let seals = guest
1288            .inject(
1289                syscalls::Fcntl::new()
1290                    .with_fd(fd)
1291                    .with_cmd(syscalls::FcntlCmd::F_GET_SEALS),

1297        Ok(named.filter(|_| match seals {
1298            Ok(seals) => seals & wanted == wanted,
1299            Err(errno) => errno == Errno::EBADF,
1300        }))
1301    }

1398    /// image of this one (inherited across `fork` or `exec`), is learned here:
1399    /// an empty regular file with no link, as a placeholder is, is checked by

1402    /// descriptor shows no seals, so for a path, or an `O_PATH` descriptor,
1403    /// the name alone decides. (A guest could imitate the name and seals: a known

1417            && (!has(StatxMask::STATX_NLINK) || stat.nlink == 0)
1418            && (!has(StatxMask::STATX_SIZE) || stat.size == 0);

1420            || self.cfg.recordreplay_modes
1421            || !may_be_placeholder

1426            StatTarget::Descriptor(fd) => (fd, false),

1484    /// Who owns the rendered file `path` and what the caller may override
1485    /// ([`FileAccess`]). A per-process file whose task has exited or is gone
1486    /// belongs to global root. For a live task's file, the caller's capability set
1487    /// `set` decides the overrides, asked of the kernel (or the provider that

1490    /// other call's answer depends on ownership alone, so a read pays nothing.
1491    /// A set that cannot be read overrides nothing.
1492    pub(super) async fn rendered_file_access<G: Guest<Self>>(

1494        guest: &mut G,
1495        path: &Path,
1496        needed: bool,
1497        set: CapabilitySet,
1498    ) -> FileAccess {

1501        }

1503        // and Linux gives its files to global root (`task_dump_owner`): then
1504        // the caller owns nothing and no capability of its namespace applies.
1505        match self.rendered_task_state(guest, path).await {

1507            _ => return FileAccess::GLOBAL_ROOT,
1508        }

1510            Self::inject_capget(guest, set).await.unwrap_or(0)

1513        };

1517    /// The caller's capability set `set` (its low 64 bits), as `capget`
1518    /// reports it for the calling thread.
1519    async fn inject_capget<G: Guest<Self>>(guest: &mut G, set: CapabilitySet) -> Option<u64> {
1520        let mut stack = guest.stack().await;
1521        let header = stack.push([LINUX_CAPABILITY_VERSION_3, 0_u32]);
1522        let data = stack.reserve::<[u32; 6]>();
1523        let stack_guard = stack.commit().ok()?;
1524        let result = guest
1525            .inject(
1526                syscalls::Capget::new()
1527                    .with_header(AddrMut::from_raw(header.as_raw()))
1528                    .with_data(AddrMut::from_raw(data.as_raw())),
1529            )
1530            .await;
1531        let words = match result {
1532            Ok(_) => guest.memory().read_value(data).ok(),
1533            Err(_) => None,
1534        };
1535        drop(stack_guard);
1536        // Each 32-bit word holds `effective`, `permitted`, `inheritable`.
1537        let words = words?;
1538        let word = match set {
1539            CapabilitySet::Effective => 0,
1540            CapabilitySet::Permitted => 1,
1541        };
1542        Some(u64::from(words[word]) | (u64::from(words[word + 3]) << 32))
1543    }

1545    /// Open the rendered procfs file `target` names with the guest's `flags`.
1546    pub(super) async fn open_rendered_procfs<G: Guest<Self>>(
1547        &self,
1548        guest: &mut G,
1549        flags: OFlag,
1550        target: RenderedTarget,
1551    ) -> Result<i64, Error> {
1552        // A descriptor link is a symbolic link: `O_NOFOLLOW` refuses it, once
1553        // the flags themselves passed. (With `O_PATH`, the caller let the
1554        // kernel open the link itself.)
1555        if let Some(errno) = open_flags_refused(flags) {
1556            return Err(errno.into());
1557        }
1558        // `O_CREAT|O_EXCL` follows no final link: the link exists (EEXIST).
1559        if target.via_link
1560            && flags.contains(OFlag::O_CREAT | OFlag::O_EXCL)
1561            && !flags.contains(OFlag::O_PATH)
1562        {
1563            return Err(Errno::EEXIST.into());
1564        }
1565        if target.via_link && flags.contains(OFlag::O_NOFOLLOW) {
1566            return Err(Errno::ELOOP.into());
1567        }
1568        let procfs = target.procfs.into_rendered();
1569        let path = procfs
1570            .opened_path()
1571            .expect("a file opened as rendered has a path");
1572        let writes = flags & OFlag::O_ACCMODE != OFlag::O_RDONLY || flags.contains(OFlag::O_TRUNC);
1573        let access = self
1574            .rendered_file_access(guest, path, writes, CapabilitySet::Effective)
1575            .await;
1576        let reopen = rendered_open_flags(flags, access)?;
1577        GuestProcfs::remember();
1578        // Read before the placeholder exists, so that a failure leaves the
1579        // guest nothing open.
1580        let metadata = self.rendered_metadata(guest, path).await?;
1581        // The snapshot this open takes, which its placeholder's name carries,
1582        // so that every backend has it after `exec`; a file whose every
1583        // snapshot is the same is rendered again there
1584        // (`ProcfsFile::snapshot_is_rerendered`). A file Linux renders when it
1585        // is read takes none until its first read
1586        // (`ProcfsFile::snapshot_at_first_read`).
1587        let snapshot =
1588            (!procfs.snapshot_at_first_read()).then(|| self.render_snapshot(guest, &procfs));
1589        let carried: &[u8] = match &snapshot {
1590            Some(snapshot) if !procfs.snapshot_is_rerendered() => snapshot,
1591            _ => &[],
1592        };
1593        let (fd, identity) = self
1594            .inject_placeholder(guest, path, carried, reopen)
1595            .await?;
1596        let carried = carried.to_vec();
1597        // The status flags Linux keeps, which include `O_NOFOLLOW`: the
1598        // placeholder cannot be reopened with it through its link.
1599        guest.thread_state().add_fd(
1600            fd,
1601            reopen | (flags & OFlag::O_NOFOLLOW),
1602            FdType::Regular,
1603            Some(metadata),
1604        )?;
1605        let procfs = match snapshot {
1606            Some(snapshot) => procfs.with_rendered_snapshot(snapshot),
1607            None => procfs,
1608        };
1609        let handle = guest.thread_state().with_detfd(fd, |detfd| {
1610            detfd.set_path(path);
1611            detfd.set_procfs(procfs.clone());
1612            detfd.open_file_handle()
1613        })?;
1614        register_placeholder_open(identity, handle, carried);
1615        Ok(fd as i64)
1616    }

1618    /// The snapshot of the rendered file `procfs` an open takes, from
1619    /// Detcore's state, as its first read used to: a UUID from the thread's
1620    /// generator, Detcore's count of open ptys, the virtual CPU's table
1621    /// (`Detcore::virtual_cpuid`, which answers the guest's CPUID
1622    /// instruction), or a fact of the machine description.
1623    fn render_snapshot<G: Guest<Self>>(&self, guest: &mut G, procfs: &ProcfsFile) -> Vec<u8> {
1624        use rand::RngExt as _;
1625        let mut file = procfs.clone();
1626        let random_uuid = file
1627            .needs_random_uuid()
1628            .then(|| guest.thread_state_mut().thread_prng().random::<[u8; 16]>());
1629        let virtual_pty_count = guest
1630            .thread_state()

1649    /// The uptime, in seconds, that the first read in this process of the
1650    /// rendered file `path` behind `fd`, one whose snapshot its first read
1651    /// takes (`ProcfsFile::snapshot_at_first_read`), renders: `uptime_seconds`,
1652    /// the uptime now, unless an earlier read of the same open file already
1653    /// took the snapshot and carried its uptime on the placeholder.
1654    ///
1655    /// Linux keeps that snapshot in the open file, for every process sharing
1656    /// it and across `exec`. Detcore keeps it in its state of the open file,
1657    /// which ptrace and KVM keep across `fork` and `exec` too; where the tool
1658    /// runs in each guest process (DBT, SaBRe, in-guest LiteInst), that state
1659    /// is the process's own, and a process that shares the open file, or the
1660    /// image after `exec`, starts without it. Every other byte of the
1661    /// snapshot is fixed for the run, so the uptime is all it needs, and the
1662    /// placeholder, one kernel object per open file, carries it in its
1663    /// modification time ([`carried_uptime_mtime`]). The guest never sees
1664    /// that time (every stat of a rendered file is Detcore's
1665    /// `rendered_stat`) and cannot set it (Detcore answers the guest's
1666    /// `utimensat` of a rendered file, by descriptor too).
1667    ///
1668    /// The carry is read and written only where nothing can change what the
1669    /// descriptor number names between Detcore's look and its write
1670    /// ([`may_carry_uptime`]): with threads sequentialized, Detcore holds the
1671    /// scheduler turn across both injected calls, no guest code runs inside
1672    /// the handler, and the descriptor must be the placeholder of `path` this
1673    /// tool process registered, by its kernel identity. Under
1674    /// `--no-sequentialize-threads` another thread could replace the number
1675    /// in between, and on SaBRe and in-guest LiteInst an accepted native
1676    /// signal handler could, so the write could land on another file: there
1677    /// is no carry there, and a process without Detcore's state of the open
1678    /// file (after `exec` or `fork` on SaBRe) takes its own snapshot at its
1679    /// next read, as before.
1680    ///
1681    /// If the write fails (the process no longer owns the placeholder after
1682    /// changing its user), the snapshot is still taken, and is not carried.
1683    pub(super) async fn first_read_uptime<G: Guest<Self>>(
1684        &self,
1685        guest: &mut G,
1686        fd: RawFd,
1687        path: &'static Path,
1688        uptime_seconds: u64,
1689    ) -> u64 {
1690        let stable = descriptor_stable_across_injections(
1691            guest.config().sequentialize_threads,
1692            guest.config().backend.runs_guest_code_between_injections,
1693        );
1694        if !stable {
1695            return uptime_seconds;
1696        }
1697        let Ok(stat) = self.inject_fstat(guest, fd).await else {
1698            return uptime_seconds;
1699        };
1700        if !may_carry_uptime(stable, placeholder_identity(stat.st_dev, stat.st_ino), path) {
1701            return uptime_seconds;
1702        }
1703        if let Some(carried) = carried_uptime(stat.st_mtime, stat.st_mtime_nsec) {
1704            return carried;
1705        }
1706        let times = [
1707            Timespec {
1708                tv_sec: 0,
1709                tv_nsec: libc::UTIME_OMIT,
1710            },
1711            carried_uptime_mtime(uptime_seconds),
1712        ];
1713        if let Err(errno) = Self::inject_futimens(guest, fd, times).await {
1714            warn!(
1715                "The first read of {} on fd {fd} could not carry its uptime on the \
1716                 placeholder ({errno}); a process without Detcore's state of the open \
1717                 file takes its own snapshot.",
1718                path.display()
1719            );
1720        }
1721        uptime_seconds
1722    }

1724    /// `utimensat(fd, NULL, times, 0)`, its times in guest stack scratch, or in
1725    /// a private anonymous page mapped for this call only where the stack
1726    /// cannot hold them (as `inject_fstat` falls back,
1727    /// https://github.com/rrnewton/hermit/issues/3328).
1728    async fn inject_futimens<G: Guest<Self>>(
1729        guest: &mut G,
1730        fd: RawFd,
1731        times: [Timespec; 2],
1732    ) -> Result<(), Errno> {
1733        let call = |times| {
1734            syscalls::Utimensat::new()
1735                .with_dirfd(fd)
1736                .with_path(None)
1737                .with_times(Some(times))
1738                .with_flags(0)
1739        };
1740        let mut stack = guest.stack().await;
1741        let staged = stack.push(times);
1742        // Keep the guard until the call has read the times: on backends whose
1743        // scratch is a Tool-owned arena (DBT, SaBRe), dropping it frees them.
1744        if let Ok(stack_guard) = stack.commit() {
1745            let result = guest.inject(call(staged)).await;
1746            drop(stack_guard);
1747            if result != Err(Errno::EFAULT) {
1748                return result.map(drop);
1749            }
1750        }
1751        let len = std::mem::size_of::<[Timespec; 2]>();
1752        let mapped = guest
1753            .inject_with_retry(Syscall::Mmap(
1754                syscalls::Mmap::new()
1755                    .with_addr(None)
1756                    .with_len(len)
1757                    .with_prot(ProtFlags::PROT_READ | ProtFlags::PROT_WRITE)
1758                    .with_flags(MapFlags::MAP_PRIVATE | MapFlags::MAP_ANONYMOUS)
1759                    .with_fd(-1)
1760                    .with_offset(0),
1761            ))
1762            .await?;
1763        let page = usize::try_from(mapped).map_err(|_| Errno::EFAULT)?;
1764        let staged = Addr::<[Timespec; 2]>::from_raw(page).ok_or(Errno::EFAULT)?;
1765        let written = AddrMut::<[Timespec; 2]>::from_raw(page)
1766            .ok_or(Errno::EFAULT)
1767            .and_then(|address| guest.memory().write_value(address, &times));
1768        let result = match written {
1769            Ok(()) => guest.inject(call(staged)).await.map(drop),
1770            Err(errno) => Err(errno),
1771        };
1772        let _ = guest
1773            .inject_with_retry(Syscall::Munmap(
1774                syscalls::Munmap::new()
1775                    .with_addr(Some(staged.cast()))
1776                    .with_len(len),
1777            ))
1778            .await;
1779        result
1780    }

1782    /// Inject a placeholder for `path` open with `flags`, at the lowest free
1783    /// descriptor number, as `open` would place the file.
1784    async fn inject_placeholder<G: Guest<Self>>(
1785        &self,
1786        guest: &mut G,
1787        path: &'static Path,
1788        snapshot: &[u8],
1789        flags: OFlag,
1790    ) -> Result<(RawFd, (u64, u64)), Error> {
1791        let name = placeholder_name(path, snapshot);
1792        let mut stack = guest.stack().await;
1793        let name = stack.push(c_string::<PLACEHOLDER_NAME_CAPACITY>(name.as_bytes()));
1794        let link = stack.reserve::<[u8; 32]>();
1795        let stack_guard = stack.commit()?;
1796        let name = name.as_raw();
1797        let create = move |flags| {
1798            syscalls::MemfdCreate::new()
1799                .with_name(PathPtr::from_ptr(name as *const libc::c_char))
1800                .with_flags(flags)
1801        };
1802        // Not executable, which a hardened `vm.memfd_noexec` requires and
1803        // kernels before 6.3 do not know; sealed, so that a spelling Detcore
1804        // does not recognise cannot change the empty file.
1805        let sealable = libc::MFD_CLOEXEC | libc::MFD_ALLOW_SEALING;
1806        let memfd = match guest.inject(create(sealable | libc::MFD_NOEXEC_SEAL)).await {
1807            Err(Errno::EINVAL) => guest.inject(create(sealable)).await?,
1808            created => created?,
1809        } as RawFd;
1810        // A backend without seals (KVM refuses `F_ADD_SEALS`) leaves the
1811        // placeholder unsealed; Detcore still answers every call it knows.
1812        let _ = guest
1813            .inject(
1814                syscalls::Fcntl::new()
1815                    .with_fd(memfd)
1816                    .with_cmd(syscalls::FcntlCmd::F_ADD_SEALS(PLACEHOLDER_SEALS)),
1817            )
1818            .await;
1819        // By the calling thread's own table (`/proc/thread-self/fd`): the
1820        // leader's (`/proc/self/fd`) lacks the memfd in a thread made without
1821        // CLONE_FILES, and the reopen then failed with ENOENT.
1822        guest.memory().write_exact(
1823            link.cast::<u8>(),
1824            &c_string::<32>(format!("/proc/thread-self/fd/{memfd}").as_bytes()),
1825        )?;
1826        // Reopening through the descriptor link gives a description with the
1827        // guest's flags, which then takes the memfd's number.
1828        let reopened = guest
1829            .inject(
1830                syscalls::Openat::new()
1831                    .with_dirfd(libc::AT_FDCWD)
1832                    .with_path(PathPtr::from_ptr(link.as_raw() as *const libc::c_char))
1833                    .with_flags(flags | OFlag::O_CLOEXEC),
1834            )
1835            .await;
1836        let moved = match reopened {
1837            Ok(reopened) => {
1838                let moved = guest
1839                    .inject(
1840                        syscalls::Dup3::new()
1841                            .with_oldfd(reopened as i32)
1842                            .with_newfd(memfd)
1843                            .with_flags(flags & OFlag::O_CLOEXEC),
1844                    )
1845                    .await;
1846                let _ = guest
1847                    .inject(syscalls::Close::new().with_fd(reopened as i32))
1848                    .await;
1849                moved
1850            }
1851            Err(errno) => Err(errno),
1852        };
1853        drop(stack_guard);
1854        // The guest gets the errno that kept the file from opening, as an open
1855        // that runs out of descriptors or memory reports it; nothing is left
1856        // open.
1857        if let Err(errno) = moved {
1858            let _ = guest.inject(syscalls::Close::new().with_fd(memfd)).await;
1859            return Err(errno.into());
1860        }
1861        let identity = match self.inject_fstat(guest, memfd).await {
1862            Ok(stat) => stat,
1863            Err(errno) => {
1864                let _ = guest.inject(syscalls::Close::new().with_fd(memfd)).await;
1865                return Err(errno.into());
1866            }
1867        };
1868        register_placeholder(identity.st_dev, identity.st_ino, path);
1869        Ok((memfd, (identity.st_dev, identity.st_ino)))
1870    }

1959    /// Detcore keeps at each one's cursor after every read and seek

1964        guest: &mut G,
1965        fd: RawFd,

1969        }

1971            .thread_state()
1972            .with_detfd(fd, |detfd| detfd.procfs_keeps_kernel_position())
1973            .unwrap_or(false);

1992    }

1994    /// Keep a placeholder's kernel position at Detcore's cursor, which a
1995    /// backend that rediscovers descriptors after `exec`, or another process
1996    /// sharing the open file, reads back (`adopt_rendered_procfs_position`).
1997    pub(super) async fn sync_rendered_procfs_position<G: Guest<Self>>(
1998        &self,
1999        guest: &mut G,
2000        fd: RawFd,
2001    ) -> Result<(), Error> {
2002        let (offset, placeholder) = guest.thread_state().with_detfd(fd, |detfd| {
2003            let offset = detfd
2004                .procfs_keeps_kernel_position()
2005                .then(|| detfd.procfs_position().map(|(offset, _)| offset))
2006                .flatten();
2007            (offset, detfd.procfs_rendered_path().is_some())
2008        })?;
2009        if let Some(offset) = offset {
2010            let moved = guest
2011                .inject(
2012                    syscalls::Lseek::new()
2013                        .with_fd(fd)
2014                        .with_offset(offset as libc::off_t)
2015                        .with_whence(syscalls::Whence::SEEK_SET),

2017                .await;
2018            // A placeholder takes any position. The kernel's own file may
2019            // refuse one past its contents, which differ from Detcore's;
2020            // Detcore's cursor still answers the guest.
2021            if placeholder {
2022                moved?;
2023            }
2024        }
2025        Ok(())

2209    /// Linux's answer to a mapping of a procfs file Detcore renders, which the
2210    /// placeholder standing in for it (a memfd) would accept. In `do_mmap`'s
2211    /// order: the arguments and the address; `MAP_FIXED_NOREPLACE` over a
2212    /// mapping is `EEXIST`, found by asking the kernel for the same range
2213    /// with nothing behind it; then the file's checks
2214    /// (`rendered_mmap_file_checks`). A `seq_file`'s mapping reaches
2215    /// `mmap_region`, which with `MAP_FIXED` unmaps whatever the range held
2216    /// before `proc_reg_mmap` fails, so the range is unmapped here too.
2217    ///
2218    /// The probe and the unmap are injected while the thread holds its turn,
2219    /// so with threads sequentialized (the default) no sibling runs between
2220    /// them. Under `--no-sequentialize-threads` a sibling could map into the
2221    /// probed range between the probe and its unmap and lose its mapping,
2222    /// where Linux makes the whole call under the mm's lock; there neither is
2223    /// made, so an overlap is not found and the range is not unmapped.
2224    pub(crate) async fn answer_rendered_procfs_mapping<G: Guest<Self>>(
2225        &self,
2226        guest: &mut G,
2227        call: &Syscall,
2228    ) -> Option<Result<i64, Error>> {
2229        let Syscall::Mmap(c) = call else {
2230            return None;
2231        };
2232        if c.flags().contains(MapFlags::MAP_ANONYMOUS) {
2233            return None;
2234        }
2235        let (path, status_flags) = guest
2236            .thread_state()
2237            .with_detfd(c.fd(), |detfd| {
2238                (detfd.status_flags() & libc::O_PATH == 0)
2239                    .then(|| detfd.procfs_rendered_path())
2240                    .flatten()
2241                    .map(|path| (path, detfd.status_flags()))
2242            })
2243            .ok()
2244            .flatten()?;
2245        // A per-process file has no `mmap` method, as a sysctl file has none.
2246        let seq_file = crate::procfs::rendered_path_is_seq_file(path)
2247            && !crate::procfs::rendered_path_is_per_process(path);
2248        let access_mode = status_flags & libc::O_ACCMODE;
2249        // Exactly: access mode 3 can neither read nor write.
2250        let mode = (
2251            access_mode == libc::O_RDONLY || access_mode == libc::O_RDWR,
2252            access_mode == libc::O_WRONLY || access_mode == libc::O_RDWR,
2253        );
2254        let noexec = match GuestProcfs::statfs() {
2255            Ok((_, flags)) => flags & libc::ST_NOEXEC != 0,
2256            Err(errno) => return Some(Err(errno.into())),
2257        };
2258        let addr = c.addr().map_or(0, |addr| addr.as_raw());
2259        let flags = c.flags();
2260        let len = match rendered_mmap_address_checks(addr, c.len(), c.offset(), flags) {
2261            Ok(len) => len,
2262            Err(errno) => return Some(Err(errno.into())),
2263        };
2264        let sequentialized = guest.config().sequentialize_threads;
2265        if sequentialized && flags.contains(MapFlags::MAP_FIXED_NOREPLACE) {
2266            let probe = guest
2267                .inject_with_retry(Syscall::Mmap(
2268                    syscalls::Mmap::new()
2269                        .with_addr(c.addr())
2270                        .with_len(len)
2271                        .with_prot(ProtFlags::PROT_NONE)
2272                        .with_flags(
2273                            MapFlags::MAP_PRIVATE
2274                                | MapFlags::MAP_ANONYMOUS
2275                                | MapFlags::MAP_FIXED_NOREPLACE,
2276                        )
2277                        .with_fd(-1)
2278                        .with_offset(0),
2279                ))
2280                .await;
2281            match probe {
2282                // The backend refused the range itself: EEXIST for an overlap,
2283                // or EPERM where DBT answers an overlap so for every mapping
2284                // (https://github.com/rrnewton/hermit/issues/4429; an anonymous
2285                // PROT_NONE probe meets no other EPERM, the address having
2286                // passed `security_mmap_addr` above). That is its answer for
2287                // this range too.
2288                Err(errno @ (Errno::EEXIST | Errno::EPERM)) => return Some(Err(errno.into())),
2289                // A probe refused for another reason says nothing of an
2290                // overlap: its limits (`map_count` before the overlap check,
2291                // `RLIMIT_AS` after the file's) are not this mapping's order.
2292                // Nothing is unmapped for MAP_FIXED_NOREPLACE, so going on to
2293                // the file's checks cannot remove a mapping.
2294                Err(_) => {}
2295                // Unmap the probe where it was placed. A backend that takes
2296                // MAP_FIXED_NOREPLACE as a hint (KVM's executor) places it
2297                // elsewhere when the range is occupied: that is the overlap.
2298                // If that unmap fails, the PROT_NONE probe stays and the guest
2299                // sees the unmap's error, which Linux never gives here; the
2300                // unmap of a range the kernel just mapped does not fail in
2301                // practice.
2302                Ok(placed) => {
2303                    let placed_addr = Addr::<libc::c_void>::from_raw(placed as usize);
2304                    if let Err(errno) = guest
2305                        .inject_with_retry(Syscall::Munmap(
2306                            syscalls::Munmap::new().with_addr(placed_addr).with_len(len),
2307                        ))
2308                        .await
2309                    {
2310                        return Some(Err(errno.into()));
2311                    }
2312                    if placed as usize != addr {
2313                        return Some(Err(Errno::EEXIST.into()));
2314                    }
2315                }
2316            }
2317        }
2318        let errno =
2319            rendered_mmap_file_checks(len, c.offset(), c.prot(), flags, noexec, seq_file, mode);
2320        // MAP_FIXED_NOREPLACE implies MAP_FIXED in Linux, but its range is
2321        // vacant here, so there is nothing to unmap.
2322        let unmaps = sequentialized
2323            && errno == Errno::EIO
2324            && flags.contains(MapFlags::MAP_FIXED)
2325            && !flags.contains(MapFlags::MAP_FIXED_NOREPLACE);
2326        if unmaps
2327            && let Err(unmapped) = guest
2328                .inject_with_retry(Syscall::Munmap(
2329                    syscalls::Munmap::new().with_addr(c.addr()).with_len(len),
2330                ))
2331                .await
2332        {
2333            return Some(Err(unmapped.into()));
2334        }
2335        if unmaps {
2336            // As `handle_munmap` does, so Detcore's record of shared
2337            // mappings (futex keys, the heap) forgets the range.
2338            guest.thread_state().unmap_memory(addr, len);
2339        }
2340        Some(Err(errno.into()))
2341    }

2343    /// Linux's answer to a call that writes through the descriptor it names
2344    /// (`write`, `writev`, `pwritev2` at the file position, `ftruncate`,
2345    /// `fallocate`, and `sendfile` or `splice` into it) when that descriptor
2346    /// is a placeholder opened for writing, or with access mode 3, which only
2347    /// a per-process file under `CAP_DAC_OVERRIDE` is (`rendered_open_flags`).
2348    /// Detcore answers every such call, so none reaches the memfd. Linux opens such a
2349    /// file without `FMODE_CAN_WRITE` (it has no write method), so a write is
2350    /// `EINVAL` after the access mode is checked; `proc_setattr` accepts any
2351    /// size and keeps none; and it has no `fallocate` method. A positioned
2352    /// write is the `seq_file`'s `ESPIPE` (`answer_rendered_procfs_operation`).
2353    /// Measured on Linux 7.1 against `/proc/self/cgroup` as its owner with
2354    /// `CAP_DAC_OVERRIDE`. `sendfile` and `splice` into it answer in Linux's
2355    /// order, which looks at the source descriptor and the offsets first
2356    /// ([`Self::answer_rendered_procfs_transfer`]).
2357    async fn answer_rendered_procfs_write<G: Guest<Self>>(
2358        &self,
2359        guest: &mut G,
2360        sysno: Sysno,
2361        args: &reverie::syscalls::SyscallArgs,
2362    ) -> Option<Result<i64, Errno>> {
2363        let fd = match sysno {
2364            Sysno::write | Sysno::writev | Sysno::ftruncate | Sysno::fallocate => args.arg0,
2365            Sysno::pwritev2 if args.arg3 as i64 == -1 => args.arg0,
2366            Sysno::sendfile => args.arg0,
2367            Sysno::splice => args.arg2,
2368            _ => return None,
2369        } as RawFd;
2370        let status = guest
2371            .thread_state()
2372            .with_detfd(fd, |detfd| {
2373                (detfd.procfs_rendered_path().is_some() && detfd.status_flags() & libc::O_PATH == 0)
2374                    .then(|| detfd.status_flags())
2375            })
2376            .ok()
2377            .flatten();
2378        let writable = match status.map(|status| status & libc::O_ACCMODE) {
2379            None | Some(libc::O_RDONLY) => return None,
2380            Some(mode) => mode == libc::O_WRONLY || mode == libc::O_RDWR,
2381        };
2382        if matches!(sysno, Sysno::sendfile | Sysno::splice) {
2383            // `O_APPEND`, which `F_SETFL` changes, is the kernel's answer for
2384            // the description, not Detcore's copy, which another process
2385            // sharing it (a native fork's child under DBT) may have changed.
2386            // The access mode is fixed at the open.
2387            let kernel = guest
2388                .inject(
2389                    syscalls::Fcntl::new()
2390                        .with_fd(fd)
2391                        .with_cmd(syscalls::FcntlCmd::F_GETFL),
2392                )
2393                .await
2394                .ok()
2395                .map(|status| status as i32);
2396            let appends = kernel
2397                .or(status)
2398                .is_some_and(|status| status & libc::O_APPEND != 0);
2399            return Some(
2400                self.answer_rendered_procfs_transfer(guest, sysno, args, writable, appends)
2401                    .await,
2402            );
2403        }
2404        // Access mode 3 (O_ACCMODE) asks for both permissions at the open and
2405        // gives the description neither: it can neither read nor write
2406        // (`FMODE_READ` and `FMODE_WRITE` unset), so a write is EBADF, and
2407        // `ftruncate` the EINVAL of a description not open for writing.
2408        if !writable {
2409            return Some(match sysno {
2410                Sysno::ftruncate => Err(Errno::EINVAL),
2411                _ => Err(Errno::EBADF),
2412            });
2413        }
2414        Some(match sysno {
2415            Sysno::ftruncate if (args.arg1 as i64) < 0 => Err(Errno::EINVAL),
2416            Sysno::ftruncate => Ok(0),
2417            Sysno::fallocate if (args.arg2 as i64) < 0 || (args.arg3 as i64) <= 0 => {
2418                Err(Errno::EINVAL)
2419            }
2420            Sysno::fallocate => Err(Errno::EOPNOTSUPP),
2421            _ => Err(Errno::EINVAL),
2422        })
2423    }

2425    /// Linux's answer to `sendfile` or `splice` into a placeholder opened
2426    /// for writing (`writable`) or with access mode 3, in the order Linux
2427    /// checks (measured on Linux 7.1 as root against `/proc/self/cgroup`
2428    /// opened `O_RDWR`). The source descriptor's state is the kernel's
2429    /// answer (`F_GETFL`, `fstat`), not Detcore's record of it, which a
2430    /// received descriptor need not have.
2431    ///
2432    /// `sendfile(out, in, offset, count)`: an unreadable `offset` is EFAULT;
2433    /// a source that is not open, is `O_PATH`, or cannot read is EBADF; an
2434    /// offset for a pipe or socket source is ESPIPE; a negative offset is
2435    /// EINVAL; then a destination that cannot write is EBADF and an
2436    /// `O_APPEND` one EINVAL; a count of 0 transfers nothing (0); anything
2437    /// else is the EINVAL of a file with no `splice_write`.
2438    ///
2439    /// `splice(in, off_in, out, off_out, len, flags)`: a length of 0 is 0
2440    /// before anything is looked at; unknown flags are EINVAL; a source that
2441    /// is not open or is `O_PATH` is EBADF; an `off_in` for a pipe source is
2442    /// ESPIPE; an unreadable `off_in` or `off_out` is EFAULT; a source that
2443    /// cannot read, or a destination that cannot write, is EBADF; anything
2444    /// else is EINVAL (no `splice_write`, or no pipe on either side).
2445    async fn answer_rendered_procfs_transfer<G: Guest<Self>>(
2446        &self,
2447        guest: &mut G,
2448        sysno: Sysno,
2449        args: &reverie::syscalls::SyscallArgs,
2450        writable: bool,
2451        appends: bool,
2452    ) -> Result<i64, Errno> {
2453        const SPLICE_F_ALL: usize = 0xf;
2454        fn offset_at<M: MemoryAccess>(memory: &M, address: usize) -> Result<i64, Errno> {
2455            let address = reverie::syscalls::Addr::<i64>::from_raw(address).ok_or(Errno::EFAULT)?;
2456            memory.read_value(address).map_err(|_| Errno::EFAULT)
2457        }
2458        let (source, length) = match sysno {
2459            Sysno::sendfile => (args.arg1 as RawFd, args.arg3),
2460            _ => (args.arg0 as RawFd, args.arg4),
2461        };
2462        if sysno == Sysno::splice {
2463            if length == 0 {
2464                return Ok(0);
2465            }
2466            if args.arg5 & !SPLICE_F_ALL != 0 {
2467                return Err(Errno::EINVAL);
2468            }
2469        }
2470        let offset = if sysno == Sysno::sendfile && args.arg2 != 0 {
2471            Some(offset_at(&guest.memory(), args.arg2)?)
2472        } else {
2473            None
2474        };
2475        let source_status = guest
2476            .inject(
2477                syscalls::Fcntl::new()
2478                    .with_fd(source)
2479                    .with_cmd(syscalls::FcntlCmd::F_GETFL),
2480            )
2481            .await
2482            .map_err(|_| Errno::EBADF)? as i32;
2483        if source_status & libc::O_PATH != 0 {
2484            return Err(Errno::EBADF);
2485        }
2486        let source_reads = matches!(
2487            source_status & libc::O_ACCMODE,
2488            libc::O_RDONLY | libc::O_RDWR
2489        );
2490        // The source's file type, which decides an offset's ESPIPE: a pipe
2491        // or FIFO has no position for either call, and a socket none for
2492        // `sendfile` (no `FMODE_PREAD`).
2493        let source_type =
2494            |stat: Result<libc::stat, Errno>| stat.map_or(0, |stat| stat.st_mode & libc::S_IFMT);
2495        if sysno == Sysno::sendfile {
2496            if !source_reads {
2497                return Err(Errno::EBADF);
2498            }
2499            if let Some(offset) = offset {
2500                let kind = source_type(self.inject_fstat(guest, source).await);
2501                if matches!(kind, libc::S_IFIFO | libc::S_IFSOCK) {
2502                    return Err(Errno::ESPIPE);
2503                }
2504                if offset < 0 {
2505                    return Err(Errno::EINVAL);
2506                }
2507            }
2508            if !writable {
2509                return Err(Errno::EBADF);
2510            }
2511            if appends {
2512                return Err(Errno::EINVAL);
2513            }
2514            return if length == 0 {
2515                Ok(0)
2516            } else {
2517                Err(Errno::EINVAL)
2518            };
2519        }
2520        // `splice`.
2521        let (off_in, off_out) = (args.arg1, args.arg3);
2522        if off_in != 0 && source_type(self.inject_fstat(guest, source).await) == libc::S_IFIFO {
2523            return Err(Errno::ESPIPE);
2524        }
2525        if off_in != 0 {
2526            offset_at(&guest.memory(), off_in)?;
2527        }
2528        if off_out != 0 {
2529            offset_at(&guest.memory(), off_out)?;
2530        }
2531        if !source_reads || !writable {
2532            return Err(Errno::EBADF);
2533        }
2534        Err(Errno::EINVAL)
2535    }

2537    /// Adopt, before `call` runs, each descriptor it names that Detcore has
2538    /// no record of and that is a placeholder this tool process made: a
2539    /// rendered procfs descriptor that reached this table by a route Detcore
2540    /// does not model, `SCM_RIGHTS` (where the tool does not discover
2541    /// descriptors as they are used: ptrace and KVM; an in-guest backend
2542    /// discovers it first, `discover_fd_from_current_process`). Then every
2543    /// call through it (reads, the cursor, a mapping, the write family, the
2544    /// task's reap) is answered as for a descriptor Detcore opened, and none
2545    /// reaches the memfd.
2546    ///
2547    /// The descriptor shares the sender's `struct file`, so it is an alias of
2548    /// the sender's open file description while Detcore still models that
2549    /// description (`PLACEHOLDER_OPENS`): one cursor, one snapshot, one set
2550    /// of status flags, exactly as Linux shares them. Once no modeled
2551    /// descriptor refers to it (the sender closed its own), the description
2552    /// is rebuilt from what the kernel keeps of it, as an in-guest backend
2553    /// rebuilds a placeholder it rediscovers (`adopt_placeholder`): the file
2554    /// and task its identity names, the status flags (`F_GETFL`), the cursor
2555    /// Detcore keeps at the kernel position (`sync_rendered_procfs_position`),
2556    /// and the snapshot its name carries; a file whose snapshot is taken at
2557    /// its first read takes it then (`initialize_procfs_snapshot`).
2558    ///
2559    /// Only once this tool process has made a placeholder, and only for a
2560    /// descriptor Detcore has no record of: one injected `fstat`, for each
2561    /// such descriptor a listed call names, every time (no cache: a
2562    /// descriptor number's identity can only be learned by asking).
2563    // TODO-HUMAN-REVIEW(#4213): Review the adoption of rendered descriptors
2564    // received over SCM_RIGHTS: the alias of a modeled description, the
2565    // rebuild of one no longer modeled, and the refusal without
2566    // sequentialized threads.
2567    pub(crate) async fn adopt_received_placeholders<G: Guest<Self>>(
2568        &self,
2569        guest: &mut G,
2570        call: &Syscall,
2571    ) -> Result<(), Error> {
2572        if !placeholders_exist() || !self.renders_procfs(guest) {
2573            return Ok(());
2574        }
2575        let (sysno, args) = call.into_parts();
2576        for fd in descriptor_arguments(&guest.memory(), sysno, &args) {
2577            if fd < 0 || guest.thread_state().with_detfd(fd, |_| ()).is_ok() {
2578                continue;
2579            }
2580            info!(
2581                "Descriptor {fd}, which Detcore has no record of, is checked for a rendered \
2582                 procfs placeholder."
2583            );
2584            let Ok(stat) = self.inject_fstat(guest, fd).await else {
2585                continue;
2586            };
2587            // A placeholder is an empty regular file with no link, as the
2588            // in-guest discovery requires too (`discover_fd_from_current_process`).
2589            // Its identity is its memfd's (dev, ino) on the kernel's internal
2590            // shmem mount, which numbers inodes from the superblock's 64-bit
2591            // `next_ino` in per-CPU batches, without the 32-bit wrap of a
2592            // user-visible tmpfs (`shmem_reserve_inode` for an
2593            // `SB_KERNMOUNT`, Linux 5.9 and later): ASSUMED here, a number
2594            // that named a placeholder this run is not given to another
2595            // memfd, so the registry, which is never pruned, names no other
2596            // file.
2597            let may_be_placeholder = stat.st_mode & libc::S_IFMT == libc::S_IFREG
2598                && stat.st_nlink == 0
2599                && stat.st_size == 0;
2600            if !may_be_placeholder {
2601                continue;
2602            }
2603            let identity = (stat.st_dev, stat.st_ino);
2604            let Some(path) = placeholder_identity(identity.0, identity.1) else {
2605                continue;
2606            };
2607            // Without sequentialized threads (a declared relaxation) another
2608            // thread may close the descriptor, or `dup2` another file over
2609            // it, between the `fstat` above and the adoption, which would
2610            // then install a stale description. Refused by name instead,
2611            // through the unsupported-operation policy: a fail-closed run
2612            // stops with the reason, and --allow-unsupported-syscalls gives
2613            // the call EOPNOTSUPP. Never a silent memfd effect.
2614            if !guest.config().sequentialize_threads {
2615                let reason = format!(
2616                    "descriptor {fd}, a rendered procfs file Detcore has no record of \
2617                     (received over SCM_RIGHTS), cannot be adopted without sequentialized \
2618                     threads (--no-sequentialize-threads)"
2619                );
2620                // Named where the run's own output shows it: a backend whose
2621                // tool runs in Hermit's process prints no stored reason.
2622                error!("Detcore refuses {sysno}: {reason}");
2623                return crate::syscalls::helpers::refuse_unserviceable_operation_with(
2624                    guest,
2625                    self.cfg.panic_on_unsupported_syscalls,
2626                    sysno,
2627                    Errno::EOPNOTSUPP,
2628                    Some(&reason),
2629                )
2630                .await
2631                .map(|_| ());
2632            }
2633            self.adopt_received_placeholder(guest, fd, path, identity)
2634                .await?;
2635        }
2636        Ok(())
2637    }

2639    /// Make descriptor `fd`, which Detcore has no record of, the placeholder
2640    /// `identity` for the rendered file `path` it is
2641    /// ([`Self::adopt_received_placeholders`]).
2642    async fn adopt_received_placeholder<G: Guest<Self>>(
2643        &self,
2644        guest: &mut G,
2645        fd: RawFd,
2646        path: &'static Path,
2647        identity: (u64, u64),
2648    ) -> Result<(), Error> {
2649        let descriptor_flags = guest
2650            .inject(
2651                syscalls::Fcntl::new()
2652                    .with_fd(fd)
2653                    .with_cmd(syscalls::FcntlCmd::F_GETFD),
2654            )
2655            .await?;
2656        let cloexec = descriptor_flags & i64::from(libc::FD_CLOEXEC) != 0;
2657        let registered = placeholder_open(identity);
2658        if let Some(alias) = registered
2659            .as_ref()
2660            .and_then(|(description, _)| description.descriptor(fd, cloexec))
2661        {
2662            guest.thread_state().install_alias(alias);
2663            return Ok(());
2664        }
2665        let status = guest
2666            .inject(
2667                syscalls::Fcntl::new()
2668                    .with_fd(fd)
2669                    .with_cmd(syscalls::FcntlCmd::F_GETFL),
2670            )
2671            .await? as i32;
2672        // An `O_PATH` descriptor has no position (EBADF).
2673        let offset = guest
2674            .inject(
2675                syscalls::Lseek::new()
2676                    .with_fd(fd)
2677                    .with_offset(0)
2678                    .with_whence(syscalls::Whence::SEEK_CUR),
2679            )
2680            .await
2681            .map_or(0, |offset| offset as usize);
2682        // OBLIGATION for every rendered file added after `cgroup`: this
2683        // rebuilt description has the snapshot the placeholder's name
2684        // carries, and no other. A file whose snapshot its first read takes
2685        // takes it again at its next read here; at position 0 that is
2686        // Linux's `show` running again, exact, but at a position above 0
2687        // Linux reads on from the buffer the first read filled in another
2688        // process. That is exact only while every snapshot of the file is
2689        // the same, as `cgroup`'s template is. A per-process file whose bytes
2690        // change (`status`, `stat`) must carry its first read's snapshot on
2691        // its placeholder (as `/proc/uptime` carries its first read's uptime,
2692        // `first_read_uptime`) before it is rendered, or its rebuilt
2693        // descriptor reads other bytes than the sender's buffer held.
2694        let snapshot = registered.map(|(_, snapshot)| snapshot).unwrap_or_default();
2695        let metadata = self.rendered_metadata(guest, path).await?;
2696        let mut flags = OFlag::from_bits_truncate(status);
2697        flags.set(OFlag::O_CLOEXEC, cloexec);
2698        guest
2699            .thread_state()
2700            .add_fd(fd, flags, FdType::Regular, Some(metadata))?;
2701        let procfs = ProcfsFile::from_rendered_path(path)
2702            .expect("a placeholder's path is a rendered procfs file")
2703            .into_rendered();
2704        let procfs = if snapshot.is_empty() {
2705            procfs
2706        } else {
2707            procfs.with_rendered_snapshot(snapshot.clone())
2708        };
2709        let handle = guest.thread_state().with_detfd(fd, |detfd| {
2710            detfd.set_path(path);
2711            detfd.set_procfs(procfs.clone());
2712            detfd.set_procfs_offset(offset);
2713            detfd.open_file_handle()
2714        })?;
2715        register_placeholder_open(identity, handle, snapshot);
2716        Ok(())
2717    }

2719    /// Answer `call` if it must not reach the placeholder behind a rendered
2720    /// procfs descriptor: what a procfs file refuses, in the order Linux
2721    /// checks. A read-only placeholder's description already makes the kernel
2722    /// refuse writes and size changes, and an `O_PATH` one every call but
2723    /// `fstat`, which Detcore answers. A per-process file opened for writing
2724    /// (`rendered_open_flags`) has a writable placeholder, so Detcore answers
2725    /// every call that would write it ([`Self::answer_rendered_procfs_write`]):
2726    /// none reaches the memfd, which KVM, without seals, would let change.
2727    pub(crate) async fn answer_rendered_procfs_operation<G: Guest<Self>>(
2728        &self,
2729        guest: &mut G,
2730        call: &Syscall,
2731    ) -> Option<Result<i64, Error>> {
2732        let (sysno, args) = call.into_parts();
2733        if let Some(answer) = self.answer_rendered_procfs_write(guest, sysno, &args).await {
2734            return Some(answer.map_err(Error::from));
2735        }
2736        let fd = match sysno {
2737            Sysno::pwrite64
2738            | Sysno::pwritev
2739            | Sysno::pwritev2
2740            | Sysno::fsync
2741            | Sysno::fdatasync
2742            | Sysno::fchmod
2743            | Sysno::fgetxattr
2744            | Sysno::flistxattr
2745            | Sysno::fsetxattr
2746            | Sysno::fremovexattr
2747            | Sysno::ioctl
2748            | Sysno::fcntl => args.arg0 as i32,
2749            _ => return None,
2750        };
2751        let (path, status_flags) = guest
2752            .thread_state()
2753            .with_detfd(fd, |detfd| {
2754                detfd
2755                    .procfs_rendered_path()
2756                    .map(|path| (path, detfd.status_flags()))
2757            })
2758            .ok()
2759            .flatten()?;
2760        if status_flags & libc::O_PATH != 0 {
2761            return None;
2762        }
2763        let seq_file = crate::procfs::rendered_path_is_seq_file(path);
2764        let writable = matches!(
2765            status_flags & libc::O_ACCMODE,
2766            libc::O_WRONLY | libc::O_RDWR
2767        );
2768        let per_process = crate::procfs::rendered_path_is_per_process(path);
2769        // Who owns the file matters only to the calls below that ask for the
2770        // owner or for write access, and only an attribute change asks for
2771        // the caller's capabilities; no other call pays for either.
2772        let command = args.arg1 as libc::c_int;
2773        let needs_access = matches!(sysno, Sysno::fsetxattr | Sysno::fremovexattr)
2774            || (sysno == Sysno::fcntl
2775                && (matches!(command, libc::F_SETLEASE | 1036)
2776                    || (command == libc::F_SETFL
2777                        && args.arg2 as libc::c_int & libc::O_NOATIME != 0)));
2778        let access = if needs_access {
2779            self.rendered_file_access(
2780                guest,
2781                path,
2782                matches!(sysno, Sysno::fsetxattr | Sysno::fremovexattr),
2783                CapabilitySet::Effective,
2784            )
2785            .await
2786        } else {
2787            FileAccess::GLOBAL_ROOT
2788        };

2790            // A sysctl file's placeholder refuses a positioned write as the
2791            // sysctl does: it is read-only (EBADF), or the position is
2792            // negative (EINVAL).
2793            Syscall::Pwrite64(_) | Syscall::Pwritev(_) | Syscall::Pwritev2(_) if !seq_file => {
2794                return None;
2795            }
2796            Syscall::Pwrite64(_) | Syscall::Pwritev(_) | Syscall::Pwritev2(_) => {
2797                return rendered_seq_file_pwrite_answer(sysno, args.arg3 as i64)
2798                    .map(|answer| answer.map_err(Error::from));
2799            }
2800            Syscall::Fsync(_) | Syscall::Fdatasync(_) => Err(Errno::EINVAL),
2801            Syscall::Fchmod(_) => Err(Errno::EPERM),
2802            Syscall::Fgetxattr(c) => match c.name().map(|name| name.read(&guest.memory())) {
2803                Some(Ok(name)) => xattr_get_answer(name.as_bytes()),
2804                Some(Err(errno)) => Err(errno),
2805                None => Err(Errno::EFAULT),
2806            },
2807            Syscall::Flistxattr(_) => Ok(0),
2808            Syscall::Fsetxattr(c)
2809                if c.flags() & !(libc::XATTR_CREATE | libc::XATTR_REPLACE) != 0 =>
2810            {
2811                Err(Errno::EINVAL)
2812            }
2813            Syscall::Fsetxattr(c) => match c.name().map(|name| name.read(&guest.memory())) {
2814                Some(Ok(name)) => xattr_change_answer(
2815                    name.as_bytes(),
2816                    Some(c.size()),
2817                    read_value(&guest.memory(), args.arg2 as u64, c.size()).as_deref(),
2818                    access,
2819                ),
2820                Some(Err(errno)) => Err(errno),
2821                None => Err(Errno::EFAULT),
2822            },
2823            Syscall::Fremovexattr(c) => match c.name().map(|name| name.read(&guest.memory())) {
2824                Some(Ok(name)) => xattr_change_answer(name.as_bytes(), None, None, access),
2825                Some(Err(errno)) => Err(errno),
2826                None => Err(Errno::EFAULT),
2827            },
2828            Syscall::Ioctl(_) => match args.arg1 as u32 as usize {
2829                // The generic file ioctls Linux answers before the file's own:
2830                // FIBMAP and FIFREEZE/FITHAW need capabilities, FS_IOC_FIEMAP
2831                // a file system that maps extents, and FICLONE/FICLONERANGE a
2832                // source on the same file system.
2833                1 | 0xc004_5877 | 0xc004_5878 => Err(Errno::EPERM),
2834                0xc020_660b => Err(Errno::EOPNOTSUPP),
2835                0x4004_9409 => Self::clone_ioctl_answer(guest, args.arg2 as i32),
2836                0x4020_940d => {
2837                    // The whole `struct file_clone_range` is read first.
2838                    match reverie::syscalls::Addr::<[u64; 4]>::from_raw(args.arg2)
2839                        .map(|address| guest.memory().read_value(address))
2840                    {
2841                        Some(Ok(range)) => Self::clone_ioctl_answer(guest, range[0] as i32),
2842                        _ => Err(Errno::EFAULT),
2843                    }
2844                }
2845                // FIONCLEX, FIOCLEX, FIOASYNC and FIONBIO, which the kernel
2846                // answers alike for both files.
2847                0x5450..=0x5452 | 0x5421 => return None,
2848                // FIONREAD: the file's size, 0, less the position, as an int.
2849                // Where the tool runs in each guest process, the kernel answers
2850                // from the empty placeholder's position, which Detcore keeps
2851                // current and which processes sharing the open file share.
2852                // Elsewhere Detcore answers from its own cursor, shared there
2853                // too (KVM's executor has no answer for a memfd).

2855                0x541b => match AddrMut::<libc::c_int>::from_raw(args.arg2) {
2856                    Some(address) => {
2857                        let position = guest
2858                            .thread_state()
2859                            .with_detfd(args.arg0 as RawFd, |detfd| {
2860                                detfd.procfs_position().map_or(0, |(offset, _)| offset)
2861                            })
2862                            .unwrap_or(0);
2863                        let unread = 0_i64.wrapping_sub(position as i64) as libc::c_int;
2864                        guest.memory().write_value(address, &unread).map(|()| 0)
2865                    }
2866                    None => Err(Errno::EFAULT),
2867                },
2868                // FIOQSIZE: the bytes the file uses, none for procfs.
2869                0x5460 => match AddrMut::<libc::loff_t>::from_raw(args.arg2) {
2870                    Some(address) => guest.memory().write_value(address, &0).map(|()| 0),
2871                    None => Err(Errno::EFAULT),
2872                },
2873                // FIGETBSZ: procfs's block size.
2874                2 => match AddrMut::<libc::c_int>::from_raw(args.arg2) {
2875                    Some(address) => guest
2876                        .memory()
2877                        .write_value(address, &(procfs_inode::PROC_BLOCK_SIZE as libc::c_int))
2878                        .map(|()| 0),
2879                    None => Err(Errno::EFAULT),
2880                },
2881                _ => Err(Errno::ENOTTY),
2882            },
2883            Syscall::Fcntl(_) => match args.arg1 as libc::c_int {
2884                // Adding seals needs a writable descriptor, before Linux
2885                // looks for seals the file cannot have.
2886                libc::F_ADD_SEALS if !writable => Err(Errno::EPERM),
2887                libc::F_ADD_SEALS | libc::F_GET_SEALS => Err(Errno::EINVAL),
2888                // A lease needs the file's owner, so a system-wide file never
2889                // has one; its owner may ask, and procfs takes none.
2890                libc::F_SETLEASE if access.owner => Err(Errno::EINVAL),
2891                libc::F_SETLEASE => Err(Errno::EACCES),
2892                // F_SET_RW_HINT needs the file's owner, then reads and checks
2893                // the hint (`rw_hint_valid`), which procfs keeps on the inode.
2894                // Detcore keeps it, for every descriptor and reopening of a
2895                // per-process file (`RenderedFileState`).
2896                1036 if access.owner => {
2897                    match Addr::<u64>::from_raw(args.arg2)
2898                        .map(|hint| guest.memory().read_value(hint))
2899                    {
2900                        Some(Ok(hint)) if hint <= 5 => {
2901                            crate::tool_global::rendered_file_state(
2902                                guest,
2903                                procfs_inode::rendered_inode(path),
2904                                crate::procfs::listing::RenderedFileChange::WriteHint(hint),
2905                            )
2906                            .await;
2907                            Ok(0)
2908                        }
2909                        Some(Ok(_)) => Err(Errno::EINVAL),
2910                        _ => Err(Errno::EFAULT),
2911                    }
2912                }
2913                1036 => Err(Errno::EPERM),
2914                // F_GET_RW_HINT: the hint the inode keeps, which any
2915                // descriptor reads. A system-wide file's is never set, and its
2916                // placeholder answers 0 as the file does.
2917                1035 if per_process => {
2918                    let state = crate::tool_global::rendered_file_state(
2919                        guest,
2920                        procfs_inode::rendered_inode(path),
2921                        crate::procfs::listing::RenderedFileChange::Read,
2922                    )
2923                    .await;
2924                    match AddrMut::<u64>::from_raw(args.arg2) {
2925                        Some(address) => guest
2926                            .memory()
2927                            .write_value(address, &state.write_hint)
2928                            .map(|()| 0),
2929                        None => Err(Errno::EFAULT),
2930                    }
2931                }
2932                libc::F_GETLEASE => Ok(i64::from(libc::F_UNLCK)),
2933                // `O_NOATIME` needs the file's owner (the owner's reaches the
2934                // placeholder, which the guest owns too), and procfs files
2935                // cannot do direct I/O.
2936                libc::F_SETFL
2937                    if args.arg2 as libc::c_int & libc::O_NOATIME != 0 && !access.owner =>
2938                {
2939                    Err(Errno::EPERM)
2940                }
2941                libc::F_SETFL if args.arg2 as libc::c_int & libc::O_DIRECT != 0 => {
2942                    Err(Errno::EINVAL)
2943                }
2944                _ => return None,
2945            },
2946            _ => return None,
2947        };
2948        Some(answer.map_err(Into::into))
2949    }

2961    GetXattr,
2962    ListXattr,
2963    /// `setxattr` with the value's size, the value at guest address `value`,
2964    /// and its flags; or `removexattr`.
2965    ChangeXattr {
2966        size: Option<usize>,
2967        value: u64,
2968        flags: u64,
2969    },
2970    Chmod {
2971        flags: u64,

2987/// Linux's answer to `request` for a rendered procfs file (0444, or 0644 for
2988/// a fixed tunable: only global root may write either), owned and overridden
2989/// as `access` says, given the attribute `name` when it changes one.
2990fn path_request_answer(
2991    request: PathRequest,
2992    name: Option<&[u8]>,
2993    value: Option<&[u8]>,
2994    access: FileAccess,
2995) -> Result<i64, Errno> {
2996    match request {
2997        PathRequest::Access { mode, flags } => {
2998            if access_arguments_invalid(mode, flags) {
2999                Err(Errno::EINVAL)
3000            } else if mode & libc::X_OK as u64 != 0
3001                || (mode & libc::W_OK as u64 != 0 && !access.may_write())
3002            {
3003                // No execute bit, which even CAP_DAC_OVERRIDE needs one of.
3004                Err(Errno::EACCES)
3005            } else {
3006                Ok(0)
3007            }
3008        }
3009        PathRequest::GetXattr => xattr_get_answer(name.ok_or(Errno::EFAULT)?),
3010        PathRequest::ListXattr => Ok(0),
3011        PathRequest::ChangeXattr { size, flags, .. } => {
3012            if flags & !(libc::XATTR_CREATE | libc::XATTR_REPLACE) as u64 != 0 {
3013                return Err(Errno::EINVAL);
3014            }
3015            xattr_change_answer(name.ok_or(Errno::EFAULT)?, size, value, access)
3016        }
3017        PathRequest::Chmod { flags } => {
3018            if flags & !(libc::AT_SYMLINK_NOFOLLOW | libc::AT_EMPTY_PATH) as u64 != 0 {
3019                Err(Errno::EINVAL)
3020            } else {
3021                Err(Errno::EPERM)
3022            }
3023        }
3024        PathRequest::Truncate { length } if length < 0 => Err(Errno::EINVAL),
3025        // procfs's `proc_setattr` accepts a size and keeps none.
3026        PathRequest::Truncate { .. } if access.may_write() => Ok(0),
3027        PathRequest::Truncate { .. } => Err(Errno::EACCES),
3028        // Leaving both times alone changes nothing; the current time needs
3029        // write permission or the owner, and any other time the owner.
3030        // (`proc_setattr` keeps the owner's times on the inode until Linux
3031        // drops it; Detcore's metadata of the file keeps the epoch.)
3032        PathRequest::Times { explicit: None } if access.owner || access.may_write() => Ok(0),
3033        PathRequest::Times { explicit: None } => Err(Errno::EACCES),
3034        PathRequest::Times {
3035            explicit: Some(false),
3036        } => Ok(0),
3037        PathRequest::Times {
3038            explicit: Some(true),
3039        } if access.owner => Ok(0),
3040        PathRequest::Times {
3041            explicit: Some(true),
3042        } => Err(Errno::EPERM),
3043        PathRequest::Chdir => Err(Errno::ENOTDIR),
3044        PathRequest::Invalid => Err(Errno::EINVAL),
3045        PathRequest::Fault => Err(Errno::EFAULT),
3046    }
3047}

3049/// How `utimensat`'s `times` asks to change a file's times: `None` for the
3050/// current time (NULL, or both `UTIME_NOW`), `Some(false)` to change nothing
3051/// (both `UTIME_OMIT`), `Some(true)` for any other time, including one
3052/// `UTIME_NOW` with one `UTIME_OMIT`, which needs the owner.
3053fn utimens_request(times: Option<[libc::timespec; 2]>) -> Option<bool> {
3054    let times = times?;
3055    if times.iter().all(|time| time.tv_nsec == libc::UTIME_OMIT) {
3056        Some(false)
3057    } else if times.iter().all(|time| time.tv_nsec == libc::UTIME_NOW) {
3058        None
3059    } else {
3060        Some(true)
3061    }
3062}

3216                        PathRequest::Invalid
3217                    },

3223                a0,
3224                0,
3225                legacy_times_request(&guest.memory(), a1, false),
3226                0,
3227            ),
3228            Sysno::utimes => (
3229                at,
3230                a0,
3231                0,
3232                legacy_times_request(&guest.memory(), a1, true),
3233                0,
3234            ),
3235            // `futimesat` is ENOSYS under Detcore for every file.
3236            Sysno::chdir => (at, a0, 0, PathRequest::Chdir, 0),
3237            _ => return Ok(None),
3238        };
3239        // Linux refuses invalid access arguments before it looks up the path
3240        // or the descriptor, for every file; leave that order to the backend.
3241        if let PathRequest::Access { mode, flags } = request
3242            && access_arguments_invalid(mode, flags)
3243        {
3244            return Ok(None);
3245        }
3246        // Likewise invalid setxattr flags, before the name, the value or the
3247        // path, for every file (`setxattr_copy`). KVM's executor reads the
3248        // name first, so answer here rather than leave it to the backend.
3249        if let PathRequest::ChangeXattr { flags, .. } = request
3250            && flags & !(libc::XATTR_CREATE | libc::XATTR_REPLACE) as u64 != 0
3251            && self.renders_procfs(guest)
3252        {
3253            return Err(Errno::EINVAL.into());
3254        }
3255        let dirfd = dirfd as i32;
3256        let empty_path_allowed = flags & libc::AT_EMPTY_PATH as u64 != 0;
3257        // An unreadable path is the kernel's EFAULT to report.
3258        let path_text = match PathPtr::from_ptr(path as *const libc::c_char) {
3259            Some(path) => match path.read(&guest.memory()) {
3260                Ok(path) => Some(path),
3261                Err(_) => return Ok(None),
3262            },
3263            None => None,
3264        };
3265        // A descriptor named by an empty path (`AT_EMPTY_PATH`), or by none
3266        // (`futimens`).
3267        let by_descriptor = match &path_text {
3268            None => sysno == Sysno::utimensat,
3269            Some(path) => path.as_os_str().is_empty() && empty_path_allowed,
3270        };
3271        // Linux then copies and checks the attribute name, the size and the
3272        // value, still before it looks up the path (`setxattr_copy`,
3273        // `import_xattr_name`), so a path error cannot hide them. Detcore
3274        // checks them first where it may answer: a spelling that could name
3275        // a rendered file. Any other path goes to the kernel untouched.
3276        let session_files = self.rendered_files(guest);
3277        let xattr_arguments = if self.renders_procfs(guest)
3278            && (by_descriptor
3279                || path_text
3280                    .as_deref()
3281                    .is_some_and(|text| could_name_rendered(text, session_files)))
3282        {
3283            checked_xattr_arguments(&guest.memory(), request, name)?
3284        } else {
3285            None
3286        };
3287        let rendered = if by_descriptor {
3288            let known = guest
3289                .thread_state()
3290                .with_detfd(dirfd, |detfd| {
3291                    detfd
3292                        .procfs_rendered_path()
3293                        .map(|path| (path, detfd.status_flags() & libc::O_PATH != 0))
3294                })
3295                .ok()
3296                .flatten();
3297            // A placeholder Detcore did not record as one (an alias the kernel
3298            // opened) is known by its kernel identity, as a stat of it is: a
3299            // time set on it would change no file of the guest's and could be
3300            // read as an uptime a first read carried (`first_read_uptime`).
3301            let rendered = match known {
3302                Some(known) => Some(known),
3303                None if self.renders_procfs(guest) => {
3304                    match self.placeholder_descriptor(guest, dirfd).await {
3305                        Some(path) => {
3306                            let status = guest
3307                                .inject(
3308                                    syscalls::Fcntl::new()
3309                                        .with_fd(dirfd)
3310                                        .with_cmd(syscalls::FcntlCmd::F_GETFL),
3311                                )
3312                                .await
3313                                .unwrap_or(0);
3314                            Some((path, status & i64::from(libc::O_PATH) != 0))
3315                        }
3316                        None => None,
3317                    }
3318                }
3319                None => None,
3320            };
3321            // `futimens` needs a descriptor opened for I/O: an `O_PATH` one
3322            // is EBADF, after the flags are checked and before the times are
3323            // (measured on Linux 7.1).
3324            if let Some((_, true)) = rendered
3325                && path_text.is_none()
3326                && !decided_before_descriptor
3327            {
3328                return Err(Errno::EBADF.into());
3329            }
3330            rendered.map(|(path, _)| path)
3331        } else {
3332            let Some(text) = path_text else {
3333                return Ok(None);
3334            };
3335            if text.as_os_str().is_empty() || !self.renders_procfs(guest) {
3336                return Ok(None);
3337            }
3338            // The spellings Detcore resolves itself, which end in a rendered
3339            // file's name or a descriptor number.
3340            let files = self.rendered_files(guest);
3341            let lexical = if could_name_rendered(&text, files) {
3342                let observed = self.procfs_spelling_at(guest, dirfd, &text).await?;
3343                match self.rendered_procfs_at(guest, &observed).await? {
3344                    // A call that does not follow a descriptor link acts on
3345                    // the link itself, which the kernel has.
3346                    Some(target) if target.via_link && flags & nofollow() != 0 => return Ok(None),
3347                    Some(target) => target.procfs.rendered_path(files),
3348                    None => None,
3349                }
3350            } else {
3351                None
3352            };
3353            match lexical {
3354                Some(rendered) => Some(rendered),
3355                // Any other spelling (a symbolic link to a descriptor link,
3356                // `/dev/stdin`): what the kernel reaches, by its identity.
3357                // Only while this process holds a rendered descriptor, so a
3358                // process that holds none pays nothing.
3359                None => {
3360                    self.placeholder_reached(
3361                        guest,
3362                        dirfd,
3363                        PathPtr::from_ptr(path as *const libc::c_char),
3364                        flags & nofollow() == 0,
3365                    )
3366                    .await
3367                }
3368            }
3369        };
3370        if rendered.is_none() {
3371            return Ok(None);
3372        }
3373        // A spelling that reached a rendered file another way (a link) has
3374        // its attribute arguments checked here, after the path.
3375        let xattr_arguments = match xattr_arguments {
3376            Some(arguments) => Some(arguments),
3377            None => checked_xattr_arguments(&guest.memory(), request, name)?,
3378        };
3379        let (name, value) = match xattr_arguments {
3380            Some((name, value)) => (Some(name), value),
3381            None => (None, None),
3382        };
3383        let rendered = rendered.expect("checked above");
3384        // A request for write access (an access probe for it, a truncation,
3385        // an attribute change) asks for the caller's capabilities; `access`
3386        // and `faccessat` without `AT_EACCESS` check the real IDs' set.
3387        let needs_capabilities = match request {
3388            PathRequest::Access { mode, .. } => mode & libc::W_OK as u64 != 0,
3389            PathRequest::Truncate { .. } | PathRequest::ChangeXattr { .. } => true,
3390            _ => false,
3391        };
3392        let set = match request {
3393            PathRequest::Access { flags, .. } if flags & libc::AT_EACCESS as u64 == 0 => {
3394                CapabilitySet::Permitted
3395            }
3396            _ => CapabilitySet::Effective,
3397        };
3398        let access = self
3399            .rendered_file_access(guest, rendered, needs_capabilities, set)
3400            .await;
3401        let answer = path_request_answer(request, name.as_deref(), value.as_deref(), access);
3402        // A time set on a per-process file stays on its inode (`proc_setattr`
3403        // keeps what `setattr_copy` copies), which every alias of the file
3404        // shows: Detcore keeps the modification time in its global tool. The
3405        // access and change times are the epoch for every rendered file, as
3406        // Detcore's metadata shows them for every file it virtualizes.
3407        if answer == Ok(0)
3408            && matches!(request, PathRequest::Times { .. })
3409            && access.owner
3410            && crate::procfs::per_process_name(rendered).is_some()
3411            && let Some((seconds, nanoseconds)) =
3412                self.requested_mtime(guest, sysno, [a0, a1, a2]).await
3413        {
3414            crate::tool_global::rendered_file_state(
3415                guest,
3416                procfs_inode::rendered_inode(rendered),
3417                crate::procfs::listing::RenderedFileChange::Mtime(seconds, nanoseconds),
3418            )
3419            .await;
3420        }
3421        answer.map(Some).map_err(Into::into)
3422    }

3424    /// The modification time a successful `utimensat`, `utime` or `utimes`
3425    /// with arguments `args` sets: the given time, or the virtual time now for
3426    /// a NULL `times` or `UTIME_NOW`; `None` where it leaves it alone
3427    /// (`UTIME_OMIT`) or its times cannot be read.
3428    async fn requested_mtime<G: Guest<Self>>(
3429        &self,
3430        guest: &mut G,
3431        sysno: Sysno,
3432        [_, a1, a2]: [u64; 3],
3433    ) -> Option<(i64, i64)> {
3434        let times = if sysno == Sysno::utimensat { a2 } else { a1 };
3435        let explicit = if times == 0 {
3436            None
3437        } else if sysno == Sysno::utimensat {
3438            let [_, mtime] = read_pair::<libc::timespec, _>(&guest.memory(), times)
3439                .ok()
3440                .flatten()?;
3441            match mtime.tv_nsec {
3442                libc::UTIME_OMIT => return None,
3443                libc::UTIME_NOW => None,
3444                _ => Some((mtime.tv_sec, mtime.tv_nsec)),
3445            }
3446        } else if sysno == Sysno::utime {
3447            let [_, modtime] = read_pair::<libc::time_t, _>(&guest.memory(), times)
3448                .ok()
3449                .flatten()?;
3450            Some((modtime, 0))
3451        } else {
3452            let [_, mtime] = read_pair::<libc::timeval, _>(&guest.memory(), times)
3453                .ok()
3454                .flatten()?;
3455            Some((mtime.tv_sec, mtime.tv_usec * 1000))

3457        match explicit {
3458            Some(time) => Some(time),
3459            None => {
3460                let now: Timespec = crate::tool_global::thread_observe_time(guest).await.into();
3461                Some((now.tv_sec, now.tv_nsec))
3462            }
3463        }
3464    }

3542/// An extended attribute's name, and the value a change sets.
3543type XattrArguments = (Vec<u8>, Option<Vec<u8>>);

3545/// An extended-attribute call's name and value, checked as Linux checks them
3546/// before the path (`setxattr_copy`, `import_xattr_name`): the name (EFAULT,
3547/// or ERANGE when empty or longer than `XATTR_NAME_MAX`), then the size
3548/// (E2BIG), then the value (EFAULT). None for any other request.
3549fn checked_xattr_arguments<M: MemoryAccess>(
3550    memory: &M,
3551    request: PathRequest,
3552    name: u64,
3553) -> Result<Option<XattrArguments>, Errno> {
3554    if !matches!(
3555        request,
3556        PathRequest::GetXattr | PathRequest::ChangeXattr { .. }
3557    ) {
3558        return Ok(None);
3559    }
3560    let name = read_xattr_name(memory, name)?;
3561    if name.is_empty() || name.len() > XATTR_NAME_MAX {
3562        return Err(Errno::ERANGE);
3563    }
3564    let value = match request {
3565        PathRequest::ChangeXattr {
3566            size: Some(size),
3567            value,
3568            ..
3569        } => {
3570            if size > XATTR_SIZE_MAX {
3571                return Err(Errno::E2BIG);
3572            }
3573            let read = read_value(memory, value, size);
3574            if size > 0 && read.is_none() {
3575                return Err(Errno::EFAULT);
3576            }
3577            read
3578        }
3579        _ => None,
3580    };
3581    Ok(Some((name, value)))
3582}

3584/// An extended attribute's name at guest address `address`, imported as
3585/// Linux does (`import_xattr_name`): at most `XATTR_NAME_MAX + 1` bytes are
3586/// read, so a name with no NUL within them is too long (ERANGE) even when
3587/// the bytes after them cannot be read, and only a fault within them is
3588/// EFAULT. The answer checks the length.
3589///
3590/// Each read stays within one page and uses the guest's own access rights
3591/// (`read_exact_with_user_access`): an exact read that faults only when a
3592/// needed byte does, as the kernel's copy faults.
3593fn read_xattr_name<M: MemoryAccess>(memory: &M, address: u64) -> Result<Vec<u8>, Errno> {
3594    const PAGE: u64 = 4096;
3595    let mut name = Vec::new();
3596    let limit = XATTR_NAME_MAX + 1;
3597    while name.len() < limit {
3598        let start = address
3599            .checked_add(name.len() as u64)
3600            .ok_or(Errno::EFAULT)?;
3601        let page_left = (PAGE - start % PAGE) as usize;
3602        let mut chunk = vec![0u8; page_left.min(limit - name.len())];
3603        let chunk_address =
3604            reverie::syscalls::Addr::<u8>::from_raw(start as usize).ok_or(Errno::EFAULT)?;
3605        memory
3606            .read_exact_with_user_access(chunk_address, &mut chunk)
3607            .map_err(|_| Errno::EFAULT)?;
3608        match chunk.iter().position(|byte| *byte == 0) {
3609            Some(end) => {
3610                name.extend_from_slice(&chunk[..end]);
3611                return Ok(name);
3612            }
3613            None => name.extend_from_slice(&chunk),
3614        }
3615    }
3616    Ok(name)
3617}

3639/// An attribute value of `size` bytes at guest address `address` (at most
3640/// `XATTR_SIZE_MAX`, which the answer checks first), if the guest can read it.
3641fn read_value<M: MemoryAccess>(memory: &M, address: u64, size: usize) -> Option<Vec<u8>> {
3642    let size = size.min(XATTR_SIZE_MAX);
3643    if size == 0 {
3644        return Some(Vec::new());
3645    }
3646    let address = reverie::syscalls::Addr::<u8>::from_raw(address as usize)?;
3647    let mut value = vec![0; size];
3648    // Exact, with the guest's own access rights, as the kernel copies it.
3649    memory
3650        .read_exact_with_user_access(address, &mut value)
3651        .ok()?;
3652    Some(value)
3653}

3655/// What `utime` (a `struct utimbuf`) or `utimes`/`futimesat` (two `struct
3656/// timeval`, when `timevals`) asks of a file's times at guest address
3657/// `times`: NULL is the current time; the times are read, and for `timeval`s
3658/// their microseconds checked, before the file is.
3659fn legacy_times_request<M: MemoryAccess>(memory: &M, times: u64, timevals: bool) -> PathRequest {
3660    if times == 0 {
3661        return PathRequest::Times { explicit: None };
3662    }
3663    if !timevals {
3664        // A `struct utimbuf`: two `time_t`, with nothing to check.
3665        return match read_pair::<libc::time_t, M>(memory, times) {
3666            Err(_) => PathRequest::Fault,
3667            Ok(_) => PathRequest::Times {
3668                explicit: Some(true),
3669            },
3670        };
3671    }
3672    match read_pair::<libc::timeval, M>(memory, times) {
3673        Err(_) => PathRequest::Fault,
3674        Ok(Some(values))
3675            if values
3676                .iter()
3677                .any(|value| !(0..1_000_000).contains(&value.tv_usec)) =>
3678        {
3679            PathRequest::Invalid
3680        }
3681        Ok(_) => PathRequest::Times {
3682            explicit: Some(true),
3683        },
3684    }
3685}
