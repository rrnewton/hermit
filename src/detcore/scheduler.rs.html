184fn take_matching_futex_waiters(waiters: &mut Vec<FutexWaiter>, wake_mask: u32) -> Vec<FutexWaiter> {
185    let (matching, remaining) = std::mem::take(waiters)
186        .into_iter()
187        .partition(|waiter| waiter.bitset & wake_mask != 0);
188    *waiters = remaining;
189    matching
190}

192/// Actions that are blocked on another internal action of the guest, such as a pipe communication,
193/// or are blocked on external conditions such as a network request.  These cannot consume a logical
194/// turn until a matching unblocking action is ready.
195///
196/// This structure will NOT include blocking operations that are implemented via polling.
197/// See NOTE [Blocking Syscalls via Internal Polling] in this folder.
198#[derive(Debug, Clone, Default)]
199pub struct BlockedPool {
200    /// BLOCKED futex transactions, waiting for wakers. Multiple threads may be blocked on
201    /// the same futex.
202    ///
203    /// INVARIANT: because Futexes aren't currently modeled with `ResourceID`, a thread
204    /// waiting on a futex will have a request filled in `next_turns` but for zero resources.
205    pub futex_waiters: HashMap<FutexID, Vec<FutexWaiter>>,
206
207    /// Futex waiters whose deadlines expired and must receive `ETIMEDOUT` when scheduled.
208    /// Timed-out waiters are removed from `futex_waiters` before entering the run queue.
209    pub timed_out_futex_waiters: HashSet<DetTid>,
210
211    /// Threads whose next event is waiting on a point in time to proceed.
212    ///
213    /// This is sorted by soonest time of occurrence.
214    /// NOTE: futex waiters will ALSO appear in here if they have timeouts.
215    pub timed_waiters: TimedEvents,
216
217    /// Blockers on external IO that are in the middle of executing (or have finished) and
218    /// are waiting for permission from the scheduler to resume.
219    ///
220    /// The protocol here is that the `(request,response)` pair (in `next_turns`) for
221    /// threads in `external_io_blockers` will have the request filled in with an
222    /// `BlockedExternalContinue` request when the thread is past its blocking action and
223    /// waiting for permission to resume. A failed operation governed by `BlockingVfork` instead
224    /// reports `VforkFailed`, which follows the same re-admission path after cancelling its
225    /// barrier. The request will stay empty while the thread is doing the blocking action. This is
226    /// different than the normal relationship
227    pub external_io_blockers: BTreeMap<DetTid, ExternalOpId>,
228
229    /// Parents parked awaiting deterministic delivery of a host-async `SIGCHLD`.
230    ///
231    /// When a guest child process exits, the kernel raises `SIGCHLD` on the
232    /// parent at a moment decided purely by host timing. If the resulting
233    /// `InboundSignal` turn is committed as soon as it arrives, its position
234    /// races whatever guest work was already runnable -- classically a `make -jN`
235    /// jobserver `pselect6` continuation -- and `--strict --verify` diverges.
236    ///
237    /// Instead the parent is parked here, out of the run queue, and re-admitted
238    /// by `step2e_process_signal_deferred` only once no ordinary (non-poller)
239    /// guest work remains: the same deterministic-work-first policy that governs
240    /// `external_io_blockers`. The physical signal has already been delivered by
241    /// the kernel, so the handler's `wait4`/`waitpid` still reaps a real host
242    /// zombie and no synthetic signal is ever generated.
243    pub sigchld_deferred: BTreeSet<DetTid>,
244
245    /// Deferred `SIGCHLD` parents that `step2e_process_signal_deferred` has
246    /// re-admitted to the run queue. Their `InboundSignal` turn must now be
247    /// granted rather than deferred again on the turn the scheduler selects them.
248    pub sigchld_ready: BTreeSet<DetTid>,
249}

251impl BlockedPool {
252    /// Returns true if there are NO blocked threads waiting outside the run-queue.
253    fn is_empty(&self) -> bool {
254        self.no_futex_waiters()
255            && self.timed_waiters.is_empty()
256            && self.external_io_blockers.is_empty()
257            && self.sigchld_deferred.is_empty()
258    }
259
260    /// True if there are no runnable threads, and the only blocked ones are externally-blocked.
261    fn only_external_blocked(&self) -> bool {
262        self.no_futex_waiters()
263            && self.timed_waiters.is_empty()
264            && !self.external_io_blockers.is_empty()
265    }
266
267    /// Returns true if there are zero threads blocked on futexes.
268    fn no_futex_waiters(&self) -> bool {
269        self.futex_waiters.iter().all(|(_, v)| v.is_empty())
270    }
271}

283/// Runtime state for enforcing a [`HappensBeforeProgram`] inside the scheduler.
284///
285/// The scheduler holds each edge's AFTER anchor -- removing that thread from the
286/// run queue -- until the edge's BEFORE anchor has *fired*, so an authored
287/// partial order deterministically reproduces a known race instead of relying on
288/// a seed lottery. An anchor "fires" when its thread is granted passage past the
289/// corresponding checkpoint (see [`Scheduler::hb_checkpoint`]).
290///
291/// Only [`Position::SyscallCount`] anchors are enforced in this milestone. Other
292/// position kinds are retained for diagnostics but never fire; [`HbRuntime::new`]
293/// warns about them so a run never silently ignores an ordering constraint.
294#[derive(Debug)]
295struct HbRuntime {
296    /// The validated, normalized program (anchors indexed by name, plus edges).
297    program: HappensBeforeProgram,
298    /// Names of anchors that have fired. Monotonic: an anchor fires at most once,
299    /// when its thread is first granted passage past it.
300    fired: BTreeSet<String>,
301    /// Threads currently parked at an AFTER anchor, out of the run queue, awaiting
302    /// their gating BEFORE anchor(s). A `BTreeSet` keeps re-admission order
303    /// deterministic.
304    parked: BTreeSet<DetTid>,
305    /// Threads observed at creation time, in deterministic spawn order, so an
306    /// anchor addressed by `spawn_ordinal` resolves to a concrete `DetTid`.
307    /// Index 0 is the root thread; index N (1-based) is the Nth spawned child,
308    /// matching [`ThreadRef::spawn_ordinal`] semantics.
309    spawn_order: Vec<DetTid>,
310    /// Set when a newly fired anchor may have opened a parked thread's gate, so
311    /// [`Scheduler::hb_flush_wakes`] re-admits parked threads at the next
312    /// `step3` boundary. Re-admission is *deferred* to that boundary because it
313    /// pushes to the run queue, which is illegal while a `tentative_pop`
314    /// selection is in progress (as it is inside `block_for_one_resource`,
315    /// where anchors fire).
316    wake_pending: bool,
317}
318
319impl HbRuntime {
320    /// Build runtime state from a normalized program, warning about any anchor
321    /// whose position kind this milestone does not enforce.
322    fn new(program: HappensBeforeProgram) -> Self {
323        for anchor in program.unenforced_positions() {
324            tracing::warn!(
325                "[happens-before] anchor {} uses position '{}', which the scheduler does not yet \
326                 enforce (only 'after N syscalls' is enforced); this ordering constraint will NOT \
327                 be applied",
328                anchor.name,
329                anchor.position,
330            );
331        }
332        Self {
333            program,
334            fired: BTreeSet::new(),
335            parked: BTreeSet::new(),
336            spawn_order: Vec::new(),
337            wake_pending: false,
338        }
339    }
340
341    /// Record a thread at creation time for `spawn_ordinal` resolution. Idempotent
342    /// and cheap; the root thread lands at index 0, the Nth child at index N.
343    fn note_spawn(&mut self, dettid: DetTid) {
344        if !self.spawn_order.contains(&dettid) {
345            self.spawn_order.push(dettid);
346        }
347    }
348
349    /// True when `tref` resolves to `dettid`, by explicit `DetTid` or by
350    /// `spawn_ordinal` against the observed spawn order.
351    fn thread_matches(&self, tref: &ThreadRef, dettid: DetTid) -> bool {
352        if let Some(d) = tref.dettid {
353            return d == dettid;
354        }
355        if let Some(ord) = tref.spawn_ordinal {
356            return self.spawn_order.get(ord as usize).copied() == Some(dettid);
357        }
358        false
359    }
360
361    /// Names of anchors on `dettid` whose enforced position is exactly
362    /// `SyscallCount(count)`.
363    fn anchors_at_syscall(&self, dettid: DetTid, count: u64) -> Vec<String> {
364        self.program
365            .anchors
366            .values()
367            .filter(|a| {
368                matches!(a.position, Position::SyscallCount(n) if n == count)
369                    && self.thread_matches(&a.thread, dettid)
370            })
371            .map(|a| a.name.clone())
372            .collect()
373    }
374
375    /// True when anchor `name` is the AFTER endpoint of a Hard edge whose BEFORE
376    /// endpoint has not yet fired -- i.e. a thread reaching `name` must be held.
377    fn anchor_blocked(&self, name: &str) -> bool {
378        self.program.edges.iter().any(|e| {
379            e.after == name && e.strength == Strength::Hard && !self.fired.contains(&e.before)
380        })
381    }
382}

384/// Which end of a thread's priority band a run-queue admission targets.
385#[derive(Debug, Clone, Copy, PartialEq, Eq)]
386pub(crate) enum AdmitSide {
387    /// Run before an equal-priority peer (`runqueue_push_front`).
388    Front,
389    /// Ordinary tail admission (`runqueue_push_back`).
390    Back,
391}
392
393/// How a run-queue admission's side is determined.
394///
395/// The side is *resolved* — not merely applied — at the deterministic drain
396/// ([`Scheduler::drain_pending_run_queue_admissions`]). Buffering the *intent*
397/// rather than an already-chosen `AdmitSide` is what keeps the admission a pure
398/// function of deterministic scheduler state: any PRNG draw that picks the side
399/// (`RunsPostFork::Random`) is consumed at the drain, in canonical `DetTid`
400/// order, instead of in host RPC / lock-acquisition order at the handler. See
401/// [`Scheduler::admit_to_run_queue`].
402#[derive(Debug, Clone, Copy, PartialEq, Eq)]
403pub(crate) enum AdmitIntent {
404    /// The side is fixed regardless of scheduler state; consumes no PRNG.
405    Fixed(AdmitSide),
406    /// The side follows the post-fork policy; `RunsPostFork::Random` draws from
407    /// the scheduler PRNG at resolution time.
408    PostFork(RunsPostFork),
409}

411/// Why a raw TID must be removed from the physical run queue at the next
412/// deterministic drain.
413#[derive(Debug, Clone, Copy, PartialEq, Eq)]
414enum RemovalDisposition {
415    /// The current thread incarnation is gone. Any admission recorded for the
416    /// same raw TID is stale and must be cancelled.
417    Retire,
418    /// Linux nonleader exec destroyed the old process leader and reassigned
419    /// its raw TID to the caller's replacement image. Remove the old physical
420    /// queue slot, but preserve the one causally paired replacement admission.
421    ReplaceThenAdmit,
422}

641    /// Internal helper. Add a [child] process to the tree, with the parent being `None`
642    /// if it's the root of the tree.
643    fn add_edge(&mut self, parent: Option<DetTid>, child: DetTid) {
644        match parent {
645            None => {
646                self.root = Some(child);
647                // Ensure an entry, even if the children vector is empty:
648                let _vec = self.tree.entry(child).or_default();
649            }
650            Some(p) => {
651                let vec = self.tree.entry(p).or_default();
652                vec.push(child);
653                let _vec = self.tree.entry(child).or_default();
654            }
655        }
656    }

658    /// Read the children of a thread, which is assumed to have an entry in the tree.
659    pub fn get_children(&mut self, parent: &DetTid) -> &Vec<DetTid> {
660        self.tree
661            .get(parent)
662            .expect("Internal failure: tid was not found in ThreadTree")
663    }

665    /// Convert to pretty-printed document.
666    ///
667    /// For example, a binary tree of depth two may print as `(1 (2 3 4) (5 6 7))`,
668    /// showing each thread ID grouped with its children.
669    ///
670    /// The thread_group_leaders argument is used for additional context into account when
671    /// pretty-printing a `ThreadTree`.  This will indicate which children are within new
672    /// thread groups using square brackets:
673    ///
674    ///   `[1 [2 [3] 4] (5 6 7)]`
675    // TODO: it would also be nice to store a fixed prefix of the binary name and listing
676    // that along with the thread ID.
677    pub fn pretty_print(&self) -> String {
678        fn walk<'a>(
679            tt: &'a HashMap<DetTid, Vec<DetTid>>,
680            tgl: &HashSet<DetTid>,
681            current: &DetTid,
682        ) -> RcDoc<'a, ()> {
683            if let Some(children) = tt.get(current) {
684                if tgl.contains(current) {
685                    RcDoc::text("[")
686                        .append(RcDoc::as_string(current))
687                        .append(if children.is_empty() {
688                            RcDoc::text("")
689                        } else {
690                            RcDoc::text(" ").append(
691                                RcDoc::intersperse(
692                                    children.iter().map(|x| walk(tt, tgl, x)),
693                                    Doc::line(),
694                                )
695                                .nest(1)
696                                .group(),
697                            )
698                        })
699                        .append(RcDoc::text("]"))
700                } else if children.is_empty() {
701                    RcDoc::as_string(current)
702                } else {
703                    RcDoc::text("(")
704                        .append(RcDoc::as_string(current))
705                        .append(RcDoc::text(" "))
706                        .append(
707                            RcDoc::intersperse(
708                                children.iter().map(|x| walk(tt, tgl, x)),
709                                Doc::line(),
710                            )
711                            .nest(1)
712                            .group(),
713                        )
714                        .append(RcDoc::text(")"))
715                }
716            } else {
717                // This should be unreachable if the invariants are maintained:
718                RcDoc::text("<ThreadTree corrupt, missing tid: ")
719                    .append(RcDoc::as_string(current))
720                    .append(RcDoc::text(">"))
721            }
722        }
723
724        let root = match self.root {
725            None => return "[]".into(),
726            Some(root) => root,
727        };
728
729        let doc = walk(&self.tree, &self.thread_group_leaders, &root);
730        let width = 100;
731        let mut vec = Vec::new();
732        doc.render(width, &mut vec).unwrap();
733        String::from_utf8(vec).unwrap()
734    }

738    pub fn size(&self) -> usize {
739        self.tree.len()
740    }

742    /// Simultaneously update the thread tree and leader tracking to reflect the creation
743    /// of a new child thread.
744    pub fn add_child(
745        &mut self,
746        parent_dettid: DetTid,
747        child_dettid: DetTid,
748        is_group_leader: bool,
749    ) {
750        // TODO(T78538674): virtualize pid/tid:
751        if parent_dettid == child_dettid {
752            self.add_edge(None, child_dettid);
753        } else {
754            self.add_edge(Some(parent_dettid), child_dettid);
755        }
756        if is_group_leader {
757            self.thread_group_leaders.insert(child_dettid);
758            self.thread_to_leader.insert(child_dettid, child_dettid);
759            // Record the creating process (the parent thread's group leader) so a
760            // deterministic child-exit SIGCHLD can later be targeted at it. The
761            // root process (parent == child) has no parent process.
762            if parent_dettid != child_dettid
763                && let Some(parent_leader) = self.thread_to_leader.get(&parent_dettid).copied()
764            {
765                self.process_parent.insert(child_dettid, parent_leader);
766            }
767        } else {
768            let parent_leader: DetPid =
769                    *self
770                        .thread_to_leader
771                        .get(&parent_dettid)
772                        .unwrap_or_else(|| {
773                            panic!("recv_create_child_thread: parent {} of child dtid {} does not exist in thread_to_leader map!",
774                                   parent_dettid, child_dettid);
775                        });
776            self.thread_to_leader.insert(child_dettid, parent_leader);
777        }
778    }

782    /// exited; callers deliver through `select_signal_target`, which drops a
783    /// signal to a `Gone` target.
784    pub fn parent_process(&self, pid: &DetPid) -> Option<DetPid> {
785        self.process_parent.get(pid).copied()
786    }

788    /// Return the set of thread IDs in the "same process" as me (same TGID), including
789    /// myself.
790    ///
791    /// Locks: takes scheduler lock.
792    pub fn my_thread_group(&mut self, me: &DetTid) -> Vec<DetTid> {
793        let root_tid: DetTid = if self.thread_group_leaders.contains(me) {
794            *me
795        } else {
796            *self
797                .thread_to_leader
798                .get(me)
799                .expect("thread must be in to_leader table")
800        };
801        let mut stack: Vec<DetTid> = vec![root_tid];
802        let mut acc: Vec<DetTid> = vec![];
803
804        while let Some(first) = stack.pop() {
805            if self.thread_group_leaders.contains(&first) && first != root_tid {
806                continue; // Stop traversal when we walk into child processes.
807            } else {
808                acc.push(first);
809            }
810            let children = self.get_children(&first);
811            stack.extend_from_slice(children);
812        }
813        assert!(acc.contains(me));
814        acc
815    }

827/// A simple backoff strategy while we have any realtime/polling elements in the system.
828/// When all external polling is removed, we can remove this.
829struct Backoff {
830    count: u64,
831}
832
833impl Backoff {
834    fn new() -> Self {
835        Backoff { count: 0 }
836    }
837
838    async fn further(&mut self, blocking: bool) {
839        self.count += 1;
840        const YIELDS_FIRST: u64 = 10;
841        if blocking {
842            if self.count <= YIELDS_FIRST {
843                std::thread::yield_now();
844            } else {
845                let round = self.count - YIELDS_FIRST;
846                let micros = if round > 13 { 10_000 } else { 2 ^ round };
847                std::thread::sleep(Duration::from_micros(micros));
848            }
849        } else if self.count <= YIELDS_FIRST {
850            tokio::task::yield_now().await;
851        } else {
852            let round = self.count - YIELDS_FIRST;
853            let micros = if round > 13 { 10_000 } else { 2 ^ round };
854            tokio::time::sleep(Duration::from_micros(micros)).await;
855        }
856    }
857
858    fn reset(&mut self) {
859        self.count = 0;
860    }
861}

864    fn default() -> Self {
865        Self::new()
866    }

883async fn sched_loop_inner(
884    sched: Arc<Mutex<Scheduler>>,
885    timer: Arc<Mutex<GlobalTime>>,
886    blocking_backoff: bool,
887    observer: Option<SchedulerObserver>,
888) {
889    info!("[scheduler] daemon task starting up, waiting for guest thread start..");
890    if let Some(observer) = &observer {
891        observer("daemon task starting; waiting for guest thread");
892    }
893    let (iv, stop_after_iter) = {
894        // Block until queue is populated.
895        let sched = sched.lock().unwrap();
896        (sched.started_up.clone(), sched.stop_after_iter)
897    };
898    iv.get().await;
899    info!("[scheduler] guest in queue, scheduler proceeding..",);
900    if let Some(observer) = &observer {
901        observer("guest registered; deterministic scheduler proceeding");
902    }
903    let mut iter: u64 = 0;
904    // We keep track of whether the last turn was a SKIP:
905    let mut last_res = Err(SkipTurn);
906    let mut backoff = Backoff::new();
907    let mut observed_turn = false;
908
909    loop {
910        // TODO (T137183027, T137184765): as part of the current strategy for blocking IO ops (see
911        // SPINNING below), we need to make sure that other threads can progress so we don't
912        // busy-wait too tightly.
913        if last_res.is_err() {
914            backoff.further(blocking_backoff).await;
915        } else {
916            backoff.reset();
917        }
918
919        trace!("[scheduler] loop iteration {}", iter);
920        if stop_after_iter.is_some() && iter > stop_after_iter.unwrap() {
921            let sched = sched.lock().unwrap();
922            tracing::warn!(
923                "[scheduler] Early exit during sched loop iteration {} due to --stop-after-iter.  Summary:\n\n{}",
924                iter,
925                sched.full_summary()
926            );
927            immediate_fatal_exit(); // We don't want a backtrace of this thread.
928        }
929        iter += 1;
930
931        // If there are NO threads left in the system, then we're truly done:
932        {
933            let sched = sched.lock().unwrap();
934            if sched.run_queue.is_empty()
935                && sched.blocked.is_empty()
936                && sched.pending_physical_process_exits.is_empty()
937                && sched.pending_run_queue_admissions.is_empty()
938                && sched.pending_run_queue_removals.is_empty()
939            {
940                info!("[scheduler] run queue empty, exiting sched_loop.");
941                if let Some(observer) = &observer {
942                    observer("run queue empty; scheduler completed");
943                }
944                return;
945            } else if let Some(stop) = sched.stop_after_turn
946                && sched.turn > stop
947            {
948                tracing::warn!(
949                    "[scheduler] Early exit during turn {} due to --stop-after-turn.  Summary:\n\n{}",
950                    sched.turn,
951                    sched.full_summary()
952                );
953                immediate_fatal_exit(); // We don't want a backtrace of this thread.
954            }
955        }
956
957        // Otherwise we trust the turn function to either choose a runnable thread or wait
958        // until something blocked is ready to run again.
959        last_res = do_a_turn_blocking(sched.clone(), timer.clone(), &last_res).await;
960        if last_res.is_ok() && !observed_turn {
961            if let Some(observer) = &observer {
962                observer("completed a deterministic scheduling turn");
963            }
964            observed_turn = true;
965        }
966    }
967}

973/// Advance turn by 1 turn, blocking when necessary to make it happen.
974/// Return the outcome of the turn as well as which resources were used, if any.
975///
976/// WARNING: this is duplicated with the non-blocking `step` function below.
977/// TODO: this duplication is temporary and they should be either combined or one removed soon.
978pub async fn do_a_turn_blocking(
979    sched: Arc<Mutex<Scheduler>>,
980    global_time: Arc<Mutex<GlobalTime>>,
981    last_turn: &Result<Resources, SkipTurn>,
982) -> Result<Resources, SkipTurn> {
983    // Loop until all threads are parked, then proceed:
984    //
985    // TODO: First, this check can move after step2, before we commit the action. Second,
986    // it can later grow more sophisticated to only check the completion of dependent
987    // actions, not all outsanding guest actions.
988    loop {
989        // We must read the queue carefully, because it can grow in the background
990        // everytime we await.  However, while it can *grow*, it cannot change order, as
991        // only the scheduler thread (us) actually rotates entries from the front to the back.
992        let req_ivar = {
993            let mut mg = sched.lock().unwrap();
994            let arc = global_time.clone();
995
996            let next_outstanding = mg.step1_check_quiescence(&arc, last_turn);
997            match next_outstanding {
998                None => {
999                    trace!("Scheduler observed full quiescense, proceeding...");
1000                    break;
1001                }
1002                Some(iv) => iv.clone(),
1003            }
1004        };
1005        trace!("Scheduler wait for full quiescense, on {}...", req_ivar);
1006        let _ = req_ivar.await;
1007    }
1008
1009    // Here we copy some information while holding the sched lock, and then release it so
1010    // we can `.await` below:
1011    let (next_dtid, req, resp) = {
1012        let mut sched = sched.lock().unwrap();
1013        sched.step2_process_blocked(&global_time)?;
1014        sched.step3_peek().ok_or(SkipTurn)?
1015    };
1016
1017    // Step 1B: wait for the selected thread to make its request.
1018    trace!(
1019        "[sched-daemon] waiting for next thread (dtid {}) to park...",
1020        next_dtid
1021    );
1022    let rsrcs: Resources = match req.get().await {
1023        Err(ThreadExited) => {
1024            debug!(
1025                "[sched-daemon] woke up on request {}, but fizzling because next thread, {}, exited.",
1026                &req, &next_dtid
1027            );
1028            // The selected thread died while we awaited its request, so this turn
1029            // is skipped without reaching step4-6. `step3_peek` opened a
1030            // tentative_pop for `next_dtid`; close it here (undo, since the turn
1031            // did not commit) before returning. Otherwise the tentative selection
1032            // outlives the turn, and the next pass's step2 removal drain calls
1033            // `remove_tid` while `tentative_selection` is still `Some`, tripping
1034            // the run queue's transaction guard -- the "reconnect panic moved one
1035            // pass" defect. The dead thread's buffered removal drains
1036            // deterministically on the next pass, once this undo has closed the
1037            // window.
1038            sched.lock().unwrap().run_queue.undo_tentative_pop();
1039            return Err(SkipTurn);
1040        }
1041        Ok(r) => r,
1042    };
1043    trace!("[sched-daemon] daemon woke up on {}...", &req);
1044
1045    // Since the scheduler is asynchronous, we need to check our assumptions.  Polling is
1046    // sufficient here because the thread cannot be racing with us to exit since we know
1047    // it is *already* parked.
1048    let mut mg = sched.lock().unwrap();
1049    mg.abort_turn_if_thread_vanished(next_dtid)?;
1050
1051    // The logical COMMIT point for the turn is during step4:
1052    mg.step4_resource_block(next_dtid, &rsrcs, &resp)?;
1053    mg.step5_guest_unblock(next_dtid, &rsrcs, &resp)?;
1054    let sched_yield = rsrcs.resources.contains_key(&ResourceID::SchedYield);
1055    mg.step6_reenquue(next_dtid, sched_yield);
1056    if let Some(call) = rsrcs.as_exit_syscall() {
1057        mg.step7_simulate_exit_posthook(next_dtid, call, &global_time);
1058    }
1059    Ok(rsrcs)
1060}

1215    /// Handle a happens-before checkpoint issued by `dettid` after its `count`th
1216    /// intercepted syscall (see `Detcore::handle_syscall_event`).
1217    ///
1218    /// Grants passage (firing every anchor at `SyscallCount(count)` on this
1219    /// thread and re-admitting any parked threads whose gate may now be open)
1220    /// unless a reached anchor is the AFTER endpoint of a Hard edge whose BEFORE
1221    /// anchor has not fired, in which case the thread is parked out of the run
1222    /// queue until a later firing wakes it. Mirrors the `SleepUntil` park/skip
1223    /// protocol: the request/response ivars are left intact so the re-admitted
1224    /// thread re-evaluates this same checkpoint on its next turn.
1225    fn hb_checkpoint(&mut self, dettid: DetTid, count: u64) -> Result<(), SkipTurn> {
1226        let (reached, blocked) = {
1227            let hb = self
1228                .happens_before
1229                .as_ref()
1230                .expect("hb checkpoint issued without a happens-before program");
1231            let reached = hb.anchors_at_syscall(dettid, count);
1232            let blocked = reached.iter().any(|name| hb.anchor_blocked(name));
1233            (reached, blocked)
1234        };
1235
1236        if reached.is_empty() {
1237            // No anchor addresses this (thread, count); nothing to gate or fire.
1238            return Ok(());
1239        }
1240
1241        if blocked {
1242            info!(
1243                "[scheduler] >>>>>>>\n\n NONCOMMIT turn {}, SKIP dettid {} held at happens-before \
1244                 anchor(s) {:?} (syscall count {}) awaiting a BEFORE anchor",
1245                self.turn, dettid, reached, count
1246            );
1247            self.happens_before.as_mut().unwrap().parked.insert(dettid);
1248            return self.skip_turn_blocked(dettid);
1249        }
1250
1251        // Grant passage: fire the reached anchors. Only wake parked threads when a
1252        // new anchor actually fired, so an idempotent re-grant causes no churn.
1253        let mut newly_fired = false;
1254        {
1255            let hb = self.happens_before.as_mut().unwrap();
1256            for name in &reached {
1257                if hb.fired.insert(name.clone()) {
1258                    newly_fired = true;
1259                }
1260            }
1261        }
1262        if newly_fired {
1263            debug!(
1264                "[happens-before] dettid {} fired anchor(s) {:?} at syscall count {}",
1265                dettid, reached, count
1266            );
1267            // Defer the actual re-admission: we are inside `block_for_one_resource`
1268            // with a `tentative_pop` selection live, and pushing to the run queue
1269            // now would trip the queue's transaction assertion. `step3` flushes.
1270            self.happens_before.as_mut().unwrap().wake_pending = true;
1271        }
1272        Ok(())
1273    }
1274
1275    /// If a happens-before anchor fired since the last check, re-admit every
1276    /// parked thread to the run queue so it re-evaluates its gate on its next
1277    /// turn. Threads still blocked re-park; the request/response ivars are
1278    /// untouched, so no request needs re-filling. Deterministic: parked threads
1279    /// are iterated in `DetTid` order.
1280    ///
1281    /// Called from `step3_peek` *before* the turn's `tentative_pop`, the only
1282    /// safe point to push to the run queue: anchors fire deep inside
1283    /// `block_for_one_resource` while a selection transaction is live, so the
1284    /// actual re-admission must be deferred to here.
1285    fn hb_flush_wakes(&mut self) {
1286        match self.happens_before.as_mut() {
1287            Some(hb) if hb.wake_pending => hb.wake_pending = false,
1288            _ => return,
1289        }
1290        let parked: Vec<DetTid> = self
1291            .happens_before
1292            .as_ref()
1293            .map(|hb| hb.parked.iter().copied().collect())
1294            .unwrap_or_default();
1295        for dettid in parked {
1296            self.happens_before.as_mut().unwrap().parked.remove(&dettid);
1297            if !self.run_queue.contains_tid(dettid) {
1298                let pos = self.runqueue_push_back(dettid);
1299                trace!(
1300                    "[happens-before] re-admitting parked dettid {} at queue position {}",
1301                    dettid, pos
1302                );
1303            }
1304        }
1305    }

1333                    if existing

1335                        .keys()
1336                        .any(|r| matches!(r, ResourceID::InboundSignal(_)))

1541    // AUTONOMOUS-BOT-IMPLEMENTED
1542    // TODO-HUMAN-REVIEW(PR-1173): Review SaBRe exec incarnation reconciliation.
1543    /// Apply Linux's successful-exec rule when a backend reloads its tool.
1544    ///
1545    /// Every sibling disappears. If a non-leader called exec, Linux also changes
1546    /// that surviving task's TID to the process leader's TID. In that case the old
1547    /// caller registration is retired and a fresh leader registration is installed
1548    /// before it is removed, so process-exit barriers cannot observe a transiently
1549    /// empty thread group.
1550    pub fn reconnect_after_exec(&mut self, reconnect: ExecReconnect) -> Vec<DetTid> {
1551        let ExecReconnect {
1552            caller,
1553            new_leader,
1554            detpid,
1555            pre_exec_mm,
1556            post_exec_mm,
1557            child_tid_addr,
1558            reconnect_priority,
1559        } = reconnect;
1560        let group = self.thread_tree.my_thread_group(&detpid);
1561        assert!(group.contains(&caller));
1562        assert!(group.contains(&new_leader));
1563        self.exec_incarnations.insert(new_leader, post_exec_mm);
1564
1565        if caller == new_leader {
1566            let siblings: Vec<_> = group.into_iter().filter(|tid| *tid != caller).collect();
1567            for sibling in &siblings {
1568                self.logically_kill_thread(sibling, &detpid, pre_exec_mm);
1569                self.timeslices.remove(sibling);
1570            }
1571            self.remove_exec_vfork_barriers(&siblings);
1572            return siblings;
1573        }
1574
1575        let survivor_priority = self
1576            .priorities
1577            .get(&caller)
1578            .copied()
1579            .or(reconnect_priority)
1580            .expect("exec caller must have a scheduler priority");
1581        let mut retired = Vec::new();
1582        for old_tid in group.into_iter().filter(|tid| *tid != caller) {
1583            self.logically_kill_thread(&old_tid, &detpid, pre_exec_mm);
1584            self.timeslices.remove(&old_tid);
1585            retired.push(old_tid);
1586        }
1587
1588        // The leader identity was occupied by a thread the kernel destroyed as
1589        // part of this exec. This is the one intentional exception to permanent
1590        // raw-TID tombstones: the pending exec record proves why Linux reused it.
1591        self.logically_killed_threads.remove(&new_leader);
1592        self.deregistration_accounted.remove(&new_leader);
1593        self.pending_physical_process_exits.remove(&detpid);
1594        assert!(
1595            self.next_turns
1596                .insert(
1597                    new_leader,
1598                    ThreadNextTurn {
1599                        dettid: new_leader,
1600                        child_tid_addr,
1601                        req: Ivar::new(),
1602                        resp: Ivar::new(),
1603                    },
1604                )
1605                .is_none(),
1606            "retired exec leader still had a scheduler registration"
1607        );
1608        self.priorities.insert(new_leader, survivor_priority);
1609        if let Some(writer) = &mut self.preemption_writer {
1610            writer.set_current(new_leader, survivor_priority);
1611        }
1612        // Post-exec reconnection can arrive asynchronously on backends whose
1613        // exec-child self-bootstraps outside a scheduler turn (DBT), so route
1614        // the new leader's admission (a run-queue *push*) through the
1615        // tentative-safe buffer rather than pushing directly.
1616        //
1617        // The exec caller's fresh next-turn request is the causal anchor. Step5
1618        // installed that empty Ivar before the caller began executing exec, so
1619        // the daemon cannot pass step1 while the successful exec is in flight.
1620        // This handler records the complete old-leader removal/replacement
1621        // admission pair before `logically_kill_thread(caller)` resolves that
1622        // request. The scheduler mutex then prevents step2 from observing only
1623        // half of the handoff. Host delay can move the handler relative to the
1624        // daemon's wait, but cannot change first-drain membership.
1625        //
1626        // The *removals* in this handler (`logically_kill_thread` ->
1627        // `run_queue.remove_tid`, for the exec caller and its siblings, above
1628        // and below) are likewise tentative-safe: `logically_kill_thread` now
1629        // routes the run-queue removal through `deschedule_or_defer`, which
1630        // buffers it to the same deterministic `step2` drain. The old leader's
1631        // removal is explicitly classified as `ReplaceThenAdmit`, so the drain
1632        // removes its physical queue slot without cancelling the new
1633        // incarnation. `are_all_quiesced` filters every pending removal key;
1634        // ordinary targets are logically dead, while the replacement key is not
1635        // runnable until that old slot has been removed and its admission
1636        // applied. No handler mutates the queue inside a tentative window.
1637        self.replace_retired_run_queue_incarnation(new_leader, AdmitIntent::Fixed(AdmitSide::Back));
1638        self.started_up.try_put(());
1639
1640        self.logically_kill_thread(&caller, &detpid, pre_exec_mm);
1641        self.timeslices.remove(&caller);
1642        retired.push(caller);
1643        self.remove_exec_vfork_barriers(&retired);
1644        retired
1645    }

1740    /// Put a Futex waiter to sleep, to be awoken by `wake_futex_waiter`.
1741    pub fn sleep_futex_waiter(
1742        &mut self,
1743        dettid: &DetTid,
1744        futexid: FutexID,
1745        maybe_timeout: Option<LogicalTime>,
1746        bitset: u32,
1747    ) {

1749            .next_turns
1750            .get(dettid)

1753        entry.push(FutexWaiter {

1755            response: nxt.resp.clone(),

1759        // not quite the same as when an active thread in the runqueue blocks on a resource, because

1946    /// Keep scheduling inside an active vfork until the parent can continue.
1947    /// Before child registration no guest may run; afterward step 3 admits only
1948    /// the child. A failed clone reaches the parent continuation without a child.
1949    ///
1950    /// On the ptrace backend the kernel keeps the vfork parent blocked inside the injected
1951    /// `clone(2)` until the child execs or exits, so a registered child (barrier `Some`) is always
1952    /// present by the time the parent posts its continuation; an unfulfilled barrier (`None`) at
1953    /// that point therefore means the clone failed and the barrier must be dropped. On a backend
1954    /// that defers the child spawn (see `backend_defers_vfork_child_registration`, e.g. KVM) the
1955    /// child registers only *after* the parent posts its continuation, so an unfulfilled barrier at
1956    /// parent continuation means the child is still on its way and the barrier must be kept.
1957    fn step2a_wait_for_vfork_barrier(&mut self) -> Result<(), SkipTurn> {
1958        // AUTONOMOUS-BOT-IMPLEMENTED
1959        // TODO-HUMAN-REVIEW(PR-1152): Review deferred vfork child registration.
1960        let defers_registration = self.backend_defers_vfork_child_registration;
1961        let completed_parents: Vec<_> = self
1962            .vfork_barriers
1963            .iter()
1964            .filter_map(|(parent, registered_child)| {
1965                let child_registered = registered_child.is_some();
1966                let remove = match self
1967                    .next_turns
1968                    .get(parent)
1969                    .and_then(|turn| turn.req.try_read())
1970                {
1971                    // The parent exited: there will be no child; drop the barrier.
1972                    Some(Err(ThreadExited)) => true,
1973                    Some(Ok(resources)) => {
1974                        let vfork_failed = resources
1975                            .resources
1976                            .keys()
1977                            .any(|resource| matches!(resource, ResourceID::VforkFailed(_)));
1978                        let at_continue = resources.resources.keys().any(|resource| {
1979                            matches!(resource, ResourceID::BlockedExternalContinue(_))
1980                        });
1981                        // A failed injected clone is an explicit deterministic outcome: no child
1982                        // can ever register, so cancel the barrier on every backend. Successful
1983                        // deferred spawns retain the ordinary continuation and keep waiting.
1984                        // At parent continuation, drop a fulfilled barrier as normal cleanup. An
1985                        // unfulfilled barrier is a failed clone only when the backend kept the
1986                        // parent blocked until the child registered; when the backend defers child
1987                        // registration the child is still coming, so keep waiting.
1988                        vfork_failed || (at_continue && (child_registered || !defers_registration))
1989                    }
1990                    _ => false,
1991                };
1992                remove.then_some(*parent)
1993            })
1994            .collect();
1995        for parent in completed_parents {
1996            self.vfork_barriers.remove(&parent);
1997        }
1998
1999        if self.vfork_barriers.values().all(Option::is_some) {
2000            Ok(())
2001        } else {
2002            trace!(
2003                "waiting for vfork child registration from parents {:?}",
2004                self.vfork_barriers
2005            );
2006            Err(SkipTurn)
2007        }
2008    }

2010    /// Check whether it is time for the *earliest* time-based event to execute INSTEAD of
2011    /// dispatching from the normal run queue.  Manipulates scheduler data structures
2012    /// accordingly.
2013    fn step2b_process_timed(&mut self) {
2014        if let Some((time_ns, evt)) = self
2015            .blocked
2016            .timed_waiters
2017            .pop_if_before(self.committed_time)
2018        {
2019            match evt {
2020                TimedEvent::ThreadEvt(dtid) => self.wake_timed_event(time_ns, dtid),
2021                TimedEvent::SignalEvt(
2022                    timed_waiters::SignalTimerId::ChildExit { parent, .. },
2023                    dtid,
2024                    sig,
2025                ) => {
2026                    // Deterministic child-exit SIGCHLD. If the host-async signal
2027                    // already arrived and was parked by the InboundSignal deferral
2028                    // gate, release it now at this logical time — that real signal
2029                    // is sufficient, so do not also synthesize one (avoids a
2030                    // duplicate delivery). Otherwise synthesize the delivery so the
2031                    // parent is notified deterministically regardless of host
2032                    // signal latency. Either way mark the parent `sigchld_ready` so
2033                    // its InboundSignal turn is granted here rather than deferred a
2034                    // second time. Releasing the deferred signal at a logical
2035                    // deadline (rather than only at run-queue quiescence, as
2036                    // step2e does) is what breaks the redis_deep starvation
2037                    // deadlock: a busy sibling can no longer starve the reaper.
2038                    self.blocked.sigchld_ready.insert(parent);
2039                    if self.blocked.sigchld_deferred.remove(&parent) {
2040                        self.run_queue.push_eager_io_repoll(parent);
2041                    } else {
2042                        self.fire_alarm(parent, dtid, sig);
2043                    }
2044                }
2045                TimedEvent::SignalEvt(id, dtid, sig) => self.fire_alarm(id.process(), dtid, sig),
2046            }
2047        }
2048    }

2244    /// Check on threads that were backgrounded performing external IO.
2245    fn step2c_process_io_blockers(&mut self) -> Result<(), SkipTurn> {
2246        if !self.blocked.external_io_blockers.is_empty() {
2247            // A nondeterministic snapshot of which blocking IO actions are ready right now:
2248            let ready: Vec<DetTid> = self
2249                .blocked
2250                .external_io_blockers
2251                .iter()
2252                .filter(|(dtid, op_id)| {
2253                    let nt = self
2254                        .next_turns
2255                        .get(dtid)
2256                        .expect("internal invariant broken");
2257                    if let Some(Ok(req)) = nt.req.try_read() {
2258                        assert_eq!(external_continue_id(&req), **op_id);
2259                        true
2260                    } else {
2261                        false
2262                    }
2263                })
2264                .map(|(dtid, _)| *dtid)
2265                .collect();
2266            debug!(
2267                "Nondeterministic status of blocking IO: out of {}, completed on {}, dtids: {:?}",
2268                self.blocked.external_io_blockers.len(),
2269                ready.len(),
2270                ready
2271            );
2272
2273            // FIXME TODO (T137183027): for record/replay to work properly, we need to ALLOW the
2274            // "Nondeterminstic algorithm" below, but record & replay those scheduler events. In
2275            // the meantime, use a deterministic eager policy once there is no other runnable work.
2276            if self.recordreplay_modes {
2277                // Only *real* deterministic work should defer external-IO harvesting.
2278                // Internal pollers sit at LAST_PRIORITY and are frequently spinning on
2279                // the very result an external-IO blocker will produce (e.g. the reader
2280                // side of `echo hi | cat`, where one stage uses InternalIOPolling while
2281                // its peer is backgrounded on BlockingExternalIO). Treat a run queue
2282                // that holds only pollers as "no deterministic work" so we still harvest
2283                // completed IO below; a queued poller must not starve a ready blocker.
2284                let only_pollers = if let Some(fp) = self.run_queue.first_priority() {
2285                    fp >= LAST_PRIORITY
2286                } else {
2287                    true
2288                };
2289
2290                // Deterministic work is runnable: let it proceed. Waiting here while such
2291                // work exists can deadlock thread creation -- the parent and new child
2292                // cannot complete clone while an existing worker blocks indefinitely in
2293                // epoll_wait.
2294                if !self.run_queue.is_empty() && !only_pollers {
2295                    return Ok(());
2296                }
2297
2298                // Reschedule every blocker whose IO has completed so its continuation can
2299                // consume the recorded result. Harvesting all ready blockers (not just the
2300                // first) and doing so even when pollers are queued is what breaks the
2301                // record-mode pipe livelock: previously a queued poller made this branch
2302                // return early forever, so the completed read/write was never rescheduled
2303                // and the poller spun on data that never arrived.
2304                if !ready.is_empty() {
2305                    for ready_dtid in &ready {
2306                        info!(
2307                            "[step2] Reschedule formerly (external IO) blocked dtid {:?}",
2308                            ready_dtid
2309                        );
2310                        self.blocked.external_io_blockers.remove(ready_dtid);
2311                        self.run_queue.push_eager_io_repoll(*ready_dtid);
2312                    }
2313                    return Ok(());
2314                }
2315
2316                // No completed IO yet, and nothing but pollers (or nothing) to run. The
2317                // blocking syscalls are executing in the host kernel; go around the loop
2318                // and re-check readiness. (Still a busy-wait; see T137183027 for the
2319                // record-the-nondeterministic-event fix.)
2320                trace!(
2321                    "[step2] eagerly waiting on external IO for dtids {:?}. spinning.",
2322                    &self.blocked.external_io_blockers
2323                );
2324                std::thread::yield_now();
2325                return Err(SkipTurn);
2326            } // End region which should be deleted.
2327
2328            // Use the same deterministic-work-first policy as record/replay. Host
2329            // completion timing must not decide whether a ready continuation overtakes
2330            // guest work that was already runnable. Pollers are excluded because they
2331            // commonly wait for the completed operation and would otherwise starve it.
2332            let only_pollers = if let Some(fp) = self.run_queue.first_priority() {
2333                fp >= LAST_PRIORITY
2334            } else {
2335                true
2336            };
2337            if !self.run_queue.is_empty() && !only_pollers {
2338                return Ok(());
2339            }
2340
2341            if !ready.is_empty() {
2342                for ready_dtid in &ready {
2343                    info!(
2344                        "[step2] Reschedule formerly (external IO) blocked dtid {:?}",
2345                        ready_dtid
2346                    );
2347                    self.blocked.external_io_blockers.remove(ready_dtid);
2348                    self.run_queue.push_eager_io_repoll(*ready_dtid);
2349                }
2350            }
2351            if self.run_queue.is_empty()
2352                && self.blocked.timed_waiters.is_empty()
2353                && !self.blocked.external_io_blockers.is_empty()
2354            {
2355                // TODO (T137184765): for now we just WAIT eagerly whenever there is blocking
2356                // external IO and else to do. We implement a busy-wait by going around the
2357                // scheduler loop again.
2358                trace!(
2359                    "[step2] TEMPORARY2: eagerly blocking on external IO for dtids {:?}.  SPINNING!",
2360                    &self.blocked.external_io_blockers
2361                );
2362                std::thread::yield_now();
2363                Err(SkipTurn)
2364            } else {
2365                // Productive work to do, irrespcetive of what's blocked, so let's get to it.
2366                Ok(())
2367            }
2368        } else {
2369            Ok(())
2370        }
2371    }

2373    fn step2d_handle_empty_queue(
2374        &mut self,
2375        global_time: &Arc<Mutex<GlobalTime>>,
2376    ) -> Result<(), SkipTurn> {
2377        let timed_empty = self.blocked.timed_waiters.is_empty();
2378        let blockers_empty = self.blocked.external_io_blockers.is_empty();
2379        let futex_empty = self.blocked.no_futex_waiters();
2380
2381        if self.run_queue.is_empty() {
2382            if !self.pending_physical_process_exits.is_empty() {
2383                // The SaBRe plugin has run the child process's logical exit hook, but the ptrace
2384                // supervisor has not received its final wait status. Fast-forwarding the next
2385                // timer here can fire a parent's timeout before the child becomes waitable.
2386                trace!(
2387                    "waiting for physical process exits before empty-queue timer fast-forward: {:?}",
2388                    self.pending_physical_process_exits
2389                );
2390                std::thread::yield_now();
2391                return Err(SkipTurn);
2392            }
2393            // When the run queue is empty, we sometimes need to give things a kick.
2394            if futex_empty && timed_empty && blockers_empty {
2395                info!("scheduler (step2_process_blocked): zero threads left anywhere, fizzling.");
2396                return Err(SkipTurn);
2397            } else if !futex_empty && timed_empty && blockers_empty {
2398                panic!(
2399                    "Deadlock detected: thread(s) waiting on futex, but no runnable threads left.\n \
2400                 queue: {:?}\n  next_turns: {:?}\n  blocked: {:?} \n",
2401                    self.run_queue, self.next_turns, self.blocked
2402                )
2403            } else if !timed_empty {
2404                debug!(
2405                    "[scheduler] Deadlock avoidance! Empty run-queue, so waking next timed event."
2406                );
2407                let (event_ns, evt) = self
2408                    .blocked
2409                    .timed_waiters
2410                    .pop()
2411                    .expect("internal error: no timed events found");
2412                info!("[scheduler] Skipping global time ahead to {}.", event_ns);
2413                {
2414                    let mut gt = global_time.lock().unwrap();
2415                    let gt_now_ns = gt.as_nanos();
2416                    let delta = event_ns.duration_since(gt_now_ns);
2417                    detlog_debug!(
2418                        "[sched] add extra global time for deadlock avoidance {:?} on current time {}",
2419                        delta,
2420                        gt_now_ns,
2421                    );
2422                    gt.add_extra_time(delta);
2423                }
2424
2425                match evt {
2426                    TimedEvent::ThreadEvt(dtid) => self.wake_timed_event(event_ns, dtid),
2427                    TimedEvent::SignalEvt(id, dtid, sig) => {
2428                        self.fire_alarm(id.process(), dtid, sig)
2429                    }
2430                }
2431                return Err(SkipTurn);
2432            }
2433        }
2434        Ok(())

2437    /// Step: Find the next thread to run for this scheduling run.
2438    /// Sometimes the next thread is from the run queue, but it can also be a timed event.
2439    /// Return `None` if the queue is empty.
2440    ///
2441    /// This is a "peek" in the sense that it leaves the thread in the run queue.
2442    fn step3_peek(&mut self) -> Option<(DetTid, Ivar<SchedRequest>, Ivar<SchedResponse>)> {
2443        // Re-admit any happens-before threads whose gate opened since last turn.
2444        // Must precede `tentative_pop_next`: the run queue forbids pushes while a
2445        // selection transaction is live.
2446        self.hb_flush_wakes();
2447        debug!(
2448            "[sched-step3] Stepping scheduler, queue len {}, current turn {}, committed_time {}",
2449            self.run_queue.len(),
2450            self.turn,
2451            self.committed_time
2452        );
2453
2454        // Enable for FULL detail:
2455        {
2456            trace!(
2457                "[sched-step3] queue {:?}, io-blocked {:?}, next_turns: ",
2458                &self.run_queue, self.blocked.external_io_blockers
2459            );
2460            for (dtid, nxt) in self.next_turns.iter() {
2461                trace!(" ==> dtid {}, req {}, resp {}", dtid, nxt.req, nxt.resp);
2462            }
2463            if !self.blocked.timed_waiters.is_empty() {
2464                trace!("Timed events: {:?}", self.blocked.timed_waiters);
2465            }
2466        }
2467
2468        if self.run_queue.is_empty() {
2469            None
2470        } else {
2471            let next_dtid = if self.vfork_barriers.is_empty() {
2472                self.run_queue.tentative_pop_next().expect("impossible")
2473            } else {
2474                let child = self
2475                    .vfork_barriers
2476                    .values()
2477                    .flatten()
2478                    .find(|child| self.run_queue.contains_tid(**child))
2479                    .copied()?;
2480                self.run_queue
2481                    .tentative_pop_tid(child)
2482                    .expect("vfork child disappeared from run queue")
2483            };
2484            let nextturn = self.next_turns.get(&next_dtid).unwrap_or_else(|| {
2485                panic!(
2486                "[sched-step3] internal error: dettid {} queued but missing entry in next_turns",
2487                    next_dtid
2488            )
2489            });
2490            Some((next_dtid, nextturn.req.clone(), nextturn.resp.clone()))
2491        }
2492    }

2507    /// `Ok`), but did its `next_turns` entry survive until the daemon got the

2509    ///
2510    /// Returns `Err(SkipTurn)` for the abandoned case, after closing the
2511    /// tentative window `step3_peek` opened. Two things hang on that `Err`:
2512    ///
2513    /// * The tentative pop is undone rather than committed, so the selection
2514    ///   does not outlive the turn and the next pass's step2 removal drain does
2515    ///   not call `remove_tid` against a live `tentative_selection` (the same
2516    ///   hygiene the `Err(ThreadExited)` arm needs).

2518    ///   steps 4-7, so nothing is blocked, unblocked or re-enqueued -- yet

2545    /// Step: Determine if action will block based on current information.  E.g. will it block
2546    /// on a pipe read with no writer? If so, register it in the blocked_pool and issue a "skip".
2547    /// We can go ahead and take resource locks and physically issue the blocking effect if we
2548    /// like.  It's immaterial whether we do that now or later.
2549    ///
2550    /// Postcondition:
2551    ///  - If returning SkipTurn, this function ENDS the Scheduler turn, advancing to the
2552    ///    next (skipping subsequent steps within theturn).  Otherwise, it waits for a
2553    ///    later step end the turn.
2554    #[allow(clippy::unnecessary_wraps)]
2555    fn step4_resource_block(
2556        &mut self,
2557        dettid: DetTid,
2558        rs: &Resources,
2559        resp: &Ivar<SchedResponse>,
2560    ) -> Result<(), SkipTurn> {
2561        if rs.poll_attempt > 0 {
2562            // The thread is polling and hasn't been "remade" as runnable yet.
2563            info!(
2564                "[scheduler] >>>>>>>\n\n NONCOMMIT turn {}, SKIP dettid {} polling resource {:?}",
2565                self.turn, dettid, rs
2566            );
2567            // Requeue the thread as a poller
2568            let popped = self.run_queue.commit_tentative_pop();
2569            assert_eq!(dettid, popped);
2570            self.run_queue
2571                .push_poller(dettid, self.get_priority(dettid), rs.poll_attempt);
2572            trace!(
2573                "[dtid {}] after deprioritizing polling request, run queue: {:?}",
2574                dettid, &self.run_queue
2575            );
2576            self.upgrade_polled_to_runnable(dettid, rs); // Indicate the thread gets to run next time
2577            self.skip_turn()
2578        } else {
2579            match rs.resources.len() {
2580                0 => Ok(()),
2581                1 => {
2582                    let (rid, perm) = rs.resources.iter().next().unwrap();
2583                    self.block_for_one_resource(dettid, rid, perm, resp)
2584                }
2585                _ => {
2586                    panic!(
2587                        "Requests for more than one resource at a time are not supported yet: {:?}",
2588                        rs
2589                    )
2590                }
2591            }
2592        }
2593    }

2617    /// Helper function. Same postcondition as step4_resource_block
2618    fn block_for_one_resource(
2619        &mut self,
2620        dettid: DetTid,
2621        rid: &ResourceID,
2622        _perm: &Permission,
2623        resp: &Ivar<SchedResponse>,
2624    ) -> Result<(), SkipTurn> {
2625        match rid {
2626            ResourceID::SleepUntil(target_ns) => {
2627                if *target_ns <= self.committed_time {
2628                    trace!(
2629                        "[dtid {}] time-based action ready to execute, target time {} is before committed global time {}",
2630                        dettid, target_ns, self.committed_time
2631                    );
2632                    Ok(())
2633                } else {
2634                    trace!(
2635                        "[dtid {}] time-based action not ready yet, registering waiter at future time {}. Current time is {}",
2636                        dettid, target_ns, self.committed_time
2637                    );
2638                    info!(
2639                        "[scheduler] >>>>>>>\n\n NONCOMMIT turn {}, SKIP dettid {} which wanted resource {:?} (blocking)",
2640                        self.turn, dettid, rid
2641                    );
2642                    self.blocked.timed_waiters.insert(*target_ns, dettid);
2643                    self.skip_turn_blocked(dettid)
2644                }
2645            }
2646
2647            // Thread BEGINS [potentially] blocking external IO
2648            ResourceID::BlockingExternalIO(op_id) | ResourceID::BlockingVfork(op_id) => {
2649                if matches!(rid, ResourceID::BlockingVfork(_)) {
2650                    assert!(self.vfork_barriers.insert(dettid, None).is_none());
2651                }
2652                info!(
2653                    "[scheduler] >>>>>>>\n\n COMMIT turn {}, BACKGROUND dettid {} (maybe-blocking)",
2654                    self.turn, dettid
2655                );
2656                // Here we allow the action to execute asynchrounously, in the
2657                // background. The protocol is that it must:
2658                //   (1) not interfere with other internal/external actions (independence),
2659                //   (2) Request a BlockedExternalContinue as the first thing after the external IO is complete.
2660                self.run_queue.undo_tentative_pop(); // Begun in step3
2661                assert!(self.run_queue.remove_tid(dettid)); // Deschedule while in background.
2662
2663                // TODO: Register the action that is occuring in the background:
2664                // let act = self.new_action(Ivar::new());
2665                // self.bg_action_pool.insert(act.action_id, act);
2666
2667                // Unblock guest so that potentially-blocking IO action can get
2668                // started. This intentionally races with subsequent turns the
2669                // scheduler commits, and thus it leans on an assumption of
2670                // non-interference, or on interference *only* affecting the external
2671                // actions that will be recorded anyway.
2672                self.run_queue.consume_yield_exclusion();
2673                self.unblock_guest(dettid, resp);
2674
2675                // Only once the ivars are cleared, and the guest is officially past the
2676                // BlockingExternalIO phase ready to issue BlockedExternalContinue, do we
2677                // then put it into the external_io_blockers struct.
2678                let old = self.blocked.external_io_blockers.insert(dettid, *op_id);
2679                assert!(old.is_none(), "thread started a second external operation");
2680                Err(SkipTurn)
2681            }
2682
2683            // Thread CONTINUES after completing [potentially] blocking IO.
2684            ResourceID::BlockedExternalContinue(_) | ResourceID::VforkFailed(_) => {
2685                // We leave the thread out of the run-queue.  At the point we put it back
2686                // in, this resource request is immediately granted.
2687                Ok(())
2688            }
2689
2690            // Thread requests change in priority
2691            ResourceID::PriorityChangePoint(prio, change_time, rcbs, epochs) => {
2692                self.perform_priority_changepoint(dettid, *prio, *change_time, *rcbs, epochs)
2693            }
2694
2695            // For now, all other resource types are immediately granted.
2696            // (TODO/FIXME: handle the entire set of resource requests.)
2697            ResourceID::FileContents(_) => Ok(()),
2698            ResourceID::FileMetadata(_) => Ok(()),
2699            ResourceID::DirectoryContents(_) => Ok(()),
2700            ResourceID::MemAddrSpace(_) => Ok(()),
2701            ResourceID::Path(_) => Ok(()),
2702            ResourceID::PathsTransitive(_) => Ok(()),
2703            ResourceID::Device(_) => Ok(()),
2704            // The scheduler-ordered `Exit` grant is the deterministic moment a
2705            // child process leaves the run set. Register a one-shot child-exit
2706            // `SIGCHLD` for the reaping parent, to be delivered at a deterministic
2707            // logical time by `step2b_process_timed`, instead of relying on the
2708            // host-async kernel `SIGCHLD` whose arrival time is host-timed (the
2709            // `make -jN` / redis `--strict --verify` nondeterminism source).
2710            ResourceID::Exit { group, process, .. } => {
2711                if *group && let Some(parent) = self.thread_tree.parent_process(process) {
2712                    // Fire strictly after the current committed time so the event
2713                    // is dispatched on a subsequent scheduler pass (DetTid == DetPid
2714                    // for a group leader, so `parent` is also the parent thread id).
2715                    let deadline = self.committed_time + LogicalTime::from_nanos(1);
2716                    self.blocked
2717                        .timed_waiters
2718                        .insert_child_exit(deadline, *process, parent, parent);
2719                }
2720                Ok(())
2721            }
2722            ResourceID::ParentContinue { .. } => Ok(()),
2723            ResourceID::InternalIOPolling => Ok(()),
2724            ResourceID::FutexWait => Ok(()),
2725            ResourceID::TraceReplay => Ok(()),
2726            ResourceID::SchedYield => Ok(()),
2727
2728            // A guest thread checking in at a happens-before anchor point. Delegate
2729            // to the enforcement logic, which either grants passage (firing anchors)
2730            // or parks the thread until its gating BEFORE anchor fires.
2731            ResourceID::HappensBeforeCheckpoint(count) => self.hb_checkpoint(dettid, *count),
2732
2733            // A host-async SIGCHLD (a guest child process exited) is delivered to
2734            // the parent at a moment decided by host timing. Committing that turn
2735            // immediately makes the signal race whatever guest work was already
2736            // runnable (e.g. a `make -jN` jobserver `pselect6` continuation),
2737            // which diverges under `--strict --verify`. Defer it deterministic-
2738            // work-first: park the parent out of the run queue and let
2739            // `step2e_process_signal_deferred` re-admit it once no ordinary guest
2740            // work remains, mirroring the `external_io_blockers` policy. Signals
2741            // that the scheduler itself synthesizes deterministically (timers via
2742            // `fire_alarm`) are never SIGCHLD and are unaffected.
2743            ResourceID::InboundSignal(SigWrapper(sig)) => {
2744                // `sigchld_ready` marks a parent step2e has already re-admitted;
2745                // grant it now rather than deferring it a second time.
2746                let already_readmitted = self.blocked.sigchld_ready.remove(&dettid);
2747                if *sig == Signal::SIGCHLD
2748                    && !already_readmitted
2749                    && self.run_queue.has_runnable_besides(dettid)
2750                {
2751                    self.run_queue.undo_tentative_pop(); // Begun in step3.
2752                    assert!(self.run_queue.remove_tid(dettid));
2753                    self.blocked.sigchld_deferred.insert(dettid);
2754                    Err(SkipTurn)
2755                } else {
2756                    Ok(())
2757                }
2758            }
2759        }
2760    }

2795    fn perform_priority_changepoint(
2796        &mut self,
2797        dettid: DetTid,
2798
2799        new_priority: Priority,
2800        guest_time: LogicalTime,
2801        guest_rcbs: u64,

2912    /// Tick global logical time due to represent the work of the scheduler itself.
2913    /// Also, update committed time.
2914    /// Prerequisite: all threads are parked, with their time contributions frozen.
2915    fn bump_global_time(
2916        &mut self,
2917        global_time: &Mutex<GlobalTime>,
2918        last_turn: &Result<Resources, SkipTurn>,
2919    ) {
2920        // An internal IO-polling retry (see `is_polling_turn`) must still advance logical
2921        // time -- finite poll/epoll/select/futex timeouts are enforced by comparing observed
2922        // logical time against the deadline in `retry_nonblocking_syscall_helper`, so freezing
2923        // time here would turn a timed wait into an infinite spin. But because the *count* of
2924        // these retries is host-timing nondeterministic, we keep their time-advance out of the
2925        // determinism log (DETLOG). This makes the `--verify` deterministic comparison
2926        // insensitive to retry count; the matching `{InternalIOPolling: ...}` COMMIT turn is
2927        // likewise excluded in `logdiff::is_internal_io_poll_commit`. Time values still shift
2928        // between runs, but those are numerically normalized before comparison.
2929        let last_turn_was_polling = last_turn
2930            .as_ref()
2931            .map(|resources| {
2932                Self::is_polling_turn(resources)
2933                    || self.is_sabre_internal_pipe_io_turn(resources)
2934                    || self.is_sabre_loopback_poll_yield_turn(resources)
2935            })
2936            .unwrap_or(false);
2937
2938        // At this moment, when threads are parked, we know that the global_time is
2939        // frozen and we can read it without any race.
2940        let snapshot: LogicalTime = {
2941            let mut gtime = global_time.lock().unwrap();
2942
2943            if self.run_queue.is_empty() && self.blocked.only_external_blocked() {
2944                // TODO(T112017687): rationalize the occurence of
2945                // BlockingExternalIO in strict runs. For example, we should
2946                // probably inject nanosleep and actually wait the intervening
2947                // time, so we don't appear too fast to external observers.
2948                trace!(
2949                    "[scheduler] skipping scheduler time advance because we're ONLY waiting for external events"
2950                );
2951            } else if last_turn.is_err() {
2952                // Note: if the last turn was a skip, it shouldn't really have time-bumped. But since we
2953                // can't see the future, we just cancel out the bump by not doing a bump this turn.
2954                trace!(
2955                    "[scheduler] skipping scheduler time advance because just-finished turn did not progress (i.e. SkipTurn)"
2956                );
2957            } else if last_turn
2958                .as_ref()
2959                .map(Self::is_internal_turn)
2960                .unwrap_or(false)
2961            {
2962                trace!(
2963                    "[scheduler] skipping scheduler time advance because just-finished turn was an internal book-keeping one"
2964                );
2965            } else {
2966                let newtime = gtime.add_scheduler_time();
2967                if last_turn_was_polling {
2968                    // Advance time (needed for timeout enforcement) but keep it off the DETLOG.
2969                    trace!(
2970                        "[sched] advance global time for internal IO-polling retry (suppressed from detlog), new time {:?}",
2971                        newtime,
2972                    );
2973                } else {
2974                    detlog_debug!(
2975                        "[sched] advance global time for scheduler turn, new time {:?}",
2976                        newtime,
2977                    );
2978                }
2979            }
2980            gtime.as_nanos()
2981        };
2982
2983        match snapshot.cmp(&self.committed_time) {
2984            std::cmp::Ordering::Less => {
2985                panic!(
2986                    "bump_global_time: invariant broken, global time went backwards from {} to {}",
2987                    self.committed_time, snapshot
2988                );
2989            }
2990            std::cmp::Ordering::Equal => {}
2991            std::cmp::Ordering::Greater => {
2992                // NB: `committed_time` still tracks the (host-timing-perturbed) global clock,
2993                // including the time advanced by suppressed IO-polling retries above, so this
2994                // line's presence is retry-count sensitive. It is therefore excluded from the
2995                // deterministic `--verify` comparison in `logdiff::is_scheduler_committed_time`
2996                // (it is redundant with the per-turn "advance global time" DETLOG anyway).
2997                detlog_debug!(
2998                    "[sched-step1] advancing committed_time from {} to {}",
2999                    self.committed_time,
3000                    snapshot
3001                );
3002                self.committed_time = snapshot;
3003            }
3004        }
3005    }

3196    /// Record an intent to admit `dtid` to the run queue, applied by the daemon
3197    /// at the next deterministic drain point ([`step2`](Self::step2_process_blocked)).
3198    ///
3199    /// Global-request handlers (`recv_create_child_thread`,
3200    /// `reconnect_after_exec`) hold the scheduler lock but run on whichever
3201    /// backend worker fielded the RPC, not on the scheduler daemon's turn. The
3202    /// point in the daemon's loop at which such a handler acquires the lock is
3203    /// host-timing-dependent on asynchronous backends (e.g. DBT): it may land
3204    /// inside the tentative-pop window (between `step3_peek` and `step4`'s
3205    /// commit, where the lock is released across `req.get().await`) *or* outside
3206    /// it (during the quiescence-wait / backoff awaits at the top of
3207    /// `do_a_turn_blocking`, where `tentative_selection` is `None`). A design
3208    /// that pushed directly whenever the window happened to be closed would make
3209    /// the *admission order* — and, under `RunsPostFork::Random`, the PRNG draw
3210    /// order — a function of that host timing: two equal-priority admissions
3211    /// could enter the queue in either relative order across otherwise-identical
3212    /// runs, and a fixed seed could explore different schedules.
3213    ///
3214    /// So admission is *always* deferred, never applied directly here. Handlers
3215    /// only record the unresolved [`AdmitIntent`]; the daemon resolves the side
3216    /// (drawing any `RunsPostFork::Random` value) and pushes the run queue at the
3217    /// single `step2` drain, in canonical `DetTid` order, before `step3` opens a
3218    /// tentative window. Draining is also the only place `remove_tid`'s tentative
3219    /// guard is guaranteed to hold.
3220    ///
3221    /// # What this does and does not make deterministic
3222    ///
3223    /// **Synchronous backends (ptrace): fully deterministic, and byte-identical
3224    /// to the pre-deferral behavior.** Handlers run post-commit, one per turn, so
3225    /// at most one admission is buffered per turn and it drains at the next
3226    /// `step2` — before that turn's `step3` selection — yielding the same
3227    /// selection sequence and the same one-draw-per-fork PRNG order as an
3228    /// immediate push.
3229    ///
3230    /// **Asynchronous backends: every production admission site has a causal or
3231    /// explicit barrier that fixes its drain, and order within that drain is
3232    /// canonical.** A bare off-turn handler would still be insufficient: a
3233    /// `BTreeMap` only canonicalizes items already in one snapshot. The current
3234    /// sites additionally bind snapshot membership to deterministic scheduler
3235    /// state:
3236    ///
3237    /// * **Ordinary clone — anchored, causally.** `CreateChildThread` issues the
3238    ///   parent's `ParentContinue` request only *after* buffering the child's
3239    ///   admission, so no thread can run between the two and the admission
3240    ///   cannot straddle a drain boundary.
3241    /// * **`vfork` — anchored, by barrier.** `vfork_barriers` /
3242    ///   [`Scheduler::step2a_wait_for_vfork_barrier`] hold the parent until the
3243    ///   child has registered, which fixes the drain.
3244    /// * **Multi-threaded exec reconnect — anchored, causally.** Step5 installs
3245    ///   the caller's empty next-turn request before it executes exec. The
3246    ///   reconnect handler atomically buffers the old-leader removal and new
3247    ///   incarnation admission, then retires the caller and resolves that
3248    ///   request. Step1 therefore cannot release step2 before the complete pair
3249    ///   exists. [`Scheduler::replace_retired_run_queue_incarnation`] binds the
3250    ///   same-raw-TID handoff explicitly.
3251    ///
3252    /// Thus both membership and within-drain resolution are functions of
3253    /// deterministic state for all current sites. The exec regression test
3254    /// forces the daemon to wait before reconnect, varies host yields, and
3255    /// compares the exact first-drain queue plus the next post-fork PRNG draw.
3256    pub(crate) fn admit_to_run_queue(&mut self, dtid: DetTid, intent: AdmitIntent) {
3257        let prev = self.pending_run_queue_admissions.insert(dtid, intent);
3258        debug_assert!(
3259            prev.is_none(),
3260            "thread {:?} recorded for run-queue admission twice before draining",
3261            dtid
3262        );
3263    }

3317    /// Record an intent to remove `dtid` from the run queue, applied by the
3318    /// daemon at the next deterministic drain point ([`step2`](Self::step2_process_blocked)).
3319    ///
3320    /// The mirror of [`Scheduler::admit_to_run_queue`] for the removal side, and
3321    /// deferred for the same reason: a global-request handler
3322    /// (`reconnect_after_exec` -> `logically_kill_thread`) runs on a backend
3323    /// worker and may hold the lock inside the daemon's tentative-pop window,
3324    /// where `RunQueue::remove_tid`'s `tentative_selection.is_none()` assert
3325    /// would trip and poison the scheduler mutex. Recording the removal and
3326    /// applying it at `step2` (window closed, guard holds) avoids that. The
3327    /// caller has already made the thread logically dead (cleared `next_turns`,
3328    /// resolved its request to `ThreadExited`), so leaving its stale run-queue
3329    /// entry in place until the drain is inert: the daemon skips it for any
3330    /// intervening turn (`step3`'s pick is validated against `next_turns`) and
3331    /// `are_all_quiesced` filters it out. On ptrace the drain runs at the next
3332    /// `step2`, before that turn's `step3` selection, so the removal is
3333    /// observationally immediate — the dead thread is never selected.
3334    fn deschedule_or_defer(&mut self, dtid: DetTid) {
3335        // A later logical death of a not-yet-drained exec replacement must
3336        // override `ReplaceThenAdmit`: `remove_blocking_entries` clears its
3337        // admission and this `Retire` disposition prevents resurrection.
3338        self.pending_run_queue_removals
3339            .insert(dtid, RemovalDisposition::Retire);
3340    }
3341
3342    /// Drain removals deferred by [`Scheduler::deschedule_or_defer`] at the same
3343    /// deterministic `step2` point as admissions, and *before* them, so a thread
3344    /// killed while an admission was still buffered is not re-enqueued. The
3345    /// window is closed here (`tentative_selection` is `None`), so
3346    /// `remove_tid`'s guard holds. The `BTreeMap` makes removal order canonical;
3347    /// each disposition determines whether a same-raw-TID admission is stale or
3348    /// is the explicitly paired exec replacement.
3349    fn drain_pending_run_queue_removals(&mut self) {
3350        if self.pending_run_queue_removals.is_empty() {
3351            return;
3352        }
3353        let pending = std::mem::take(&mut self.pending_run_queue_removals);
3354        for (dtid, disposition) in pending {
3355            match disposition {
3356                RemovalDisposition::Retire => {
3357                    // Ordinary logical death cancels a buffered admission for
3358                    // the same thread incarnation.
3359                    self.pending_run_queue_admissions.remove(&dtid);
3360                }
3361                RemovalDisposition::ReplaceThenAdmit => {
3362                    assert!(
3363                        self.next_turns.contains_key(&dtid),
3364                        "exec replacement {:?} lost its scheduler registration",
3365                        dtid
3366                    );
3367                    assert!(
3368                        self.pending_run_queue_admissions.contains_key(&dtid),
3369                        "exec replacement {:?} lost its paired admission",
3370                        dtid
3371                    );
3372                    assert!(
3373                        !self.thread_is_logically_killed(dtid),
3374                        "logically dead exec replacement {:?} reached the drain",
3375                        dtid
3376                    );
3377                }
3378            }
3379            // Always remove the old physical queue slot before a replacement
3380            // admission is applied. This prevents the new image from inheriting
3381            // the destroyed leader's round-robin position.
3382            let _ = self.run_queue.remove_tid(dtid);
3383        }
3384    }

3402    /// Drain admissions deferred by [`Scheduler::admit_to_run_queue`] into the
3403    /// run queue at a single deterministic point: the very start of `step2`,
3404    /// before `step3_peek` opens a tentative window (so `tentative_selection` is
3405    /// guaranteed `None` here). Draining a `BTreeMap` visits `DetTid`s in sorted
3406    /// order, so the resulting run-queue state is a pure function of the
3407    /// deterministic schedule rather than of RPC/lock-acquisition timing.
3408    fn drain_pending_run_queue_admissions(&mut self) {
3409        if self.pending_run_queue_admissions.is_empty() {
3410            return;
3411        }
3412        let pending = std::mem::take(&mut self.pending_run_queue_admissions);
3413        // `pending` is a `BTreeMap`, so iteration visits `DetTid`s in sorted
3414        // order. Resolving each intent here (rather than at the racing handler)
3415        // means any `RunsPostFork::Random` PRNG draw is consumed in this
3416        // canonical order, so both the admission order *and* the chosen side are
3417        // pure functions of deterministic state.
3418        for (dtid, intent) in pending {
3419            // A thread retired (exec/kill) between record and drain is skipped.
3420            // `remove_blocking_entries` also clears the buffer on teardown, so
3421            // this guard is defensive against any teardown path that does not.
3422            if !self.next_turns.contains_key(&dtid) {
3423                trace!(
3424                    "[step2] skipping deferred admission of retired thread {:?}",
3425                    dtid
3426                );
3427                continue;
3428            }
3429            let side = self.resolve_admit_intent(intent);
3430            self.admit_now(dtid, side);
3431        }
3432    }
