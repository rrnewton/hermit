22/// Detcore's execution policy for a named Linux syscall.
23pub(crate) enum SyscallClassification {
24    /// Detcore models the syscall or applies an explicit deterministic refusal policy.
25    Determinized,
26    /// The syscall is intentionally forwarded under documented container assumptions.
27    PassThrough,
28    /// The syscall lacks a deterministic implementation and uses the configured fallback policy.
29    // TODO-HUMAN-REVIEW(PR-643): Review the issue-backed unsupported classification policy.
30    Unsupported,
31}

1214// TODO-HUMAN-REVIEW(PR-978): Review the aggregate deterministic-refusal
1215// boundary used to enforce the same fixed refusal in backends that execute
1216// guest syscalls outside Detcore's `handle_syscall_event` dispatcher.
1217/// True for every syscall Detcore refuses with a fixed errno in the strict
1218/// dispatcher *without consulting the host* -- the deterministic-refusal
1219/// boundary. Classifying a syscall this way in the ptrace path only enforces the
1220/// isolation boundary where the shared Detcore dispatcher runs. A backend that
1221/// lets a guest execute a syscall natively outside that dispatcher (most
1222/// importantly the DBT copied pre-exec child fast path, which runs on the
1223/// DynamoRIO client stack with no Detcore tool) would otherwise reach the host
1224/// and leak or mutate the very state the classification is meant to hide. Such a
1225/// backend must fail closed for these syscalls in strict mode instead.
1226///
1227/// This is the union of the individual family predicates whose dispatch arm in
1228/// `Detcore::handle_syscall_event` returns `Err(Error::Errno(..))` unconditionally
1229/// or under `panic_on_unsupported_syscalls`. It deliberately excludes families
1230/// Detcore *emulates* with real semantics (for example the credential-identity
1231/// no-ops, `timer_create`, and the socket-identity handlers), because failing a
1232/// copied child closed for an emulated syscall would diverge from the ptrace
1233/// path rather than match it. Keep this in sync with those dispatch arms.
1234pub(crate) const fn is_strict_only_deterministic_refusal_syscall(sysno: Sysno) -> bool {
1235    // These families preserve native behavior outside strict mode for rr and
1236    // application compatibility, but fail closed under strict execution.
1237    matches!(sysno, Sysno::rseq)
1238        || is_zero_copy_pipe_syscall(sysno)
1239        || is_kernel_keyring_syscall(sysno)
1240}

1248        || is_unsupported_async_ipc_syscall(sysno)
1249        || is_remap_file_pages_enosys_syscall(sysno)
1250        || is_landlock_sandbox_syscall(sysno)
1251        || is_process_isolation_refused_syscall(sysno)
1252        || is_privileged_observation_refused_syscall(sysno)
1253        || is_perf_event_enosys_syscall(sysno)
