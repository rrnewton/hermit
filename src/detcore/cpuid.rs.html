17/// The CPUID table a guest sees: the fixed x86-64-v2 table, or the x86-64-v3
18/// table built from it (<https://github.com/rrnewton/hermit/issues/4243>).
19#[derive(Debug, Clone, Copy)]
20pub struct InterceptedCpuid {
21    profile: CpuProfile,
22    /// Under v3, the guest's leaf 0xD rows for subleaves 0-9, derived from the
23    /// host's: a guest's XGETBV and XSAVE run on the host, so leaf 0xD must
24    /// describe the host's XCR0 and layout.
25    xsave: [[u32; 4]; XSAVE_SUBLEAVES],
26}
27
28/// Leaf 0xD subleaves the v3 table answers from the host: 0, and components
29/// 2-9 (the components the qualification covers stop at PKRU, bit 9).
30const XSAVE_SUBLEAVES: usize = 10;
31
32/// Leaf 1 ECX bits v3 adds to v2: FMA (12), MOVBE (22), XSAVE (26), OSXSAVE
33/// (27) and F16C (29). AVX (28), SSE4.2, POPCNT and the rest are already there.
34pub const V3_LEAF1_ECX_ADDED: u32 = (1 << 12) | (1 << 22) | (1 << 26) | (1 << 27) | (1 << 29);
35/// Leaf 1 ECX bit 31, "running under a hypervisor": a platform declaration,
36/// not an instruction the guest needs, which v3 clears so that the host check
37/// does not refuse every bare-metal host.
38pub const LEAF1_ECX_HYPERVISOR: u32 = 1 << 31;
39/// Leaf 7 EBX bits v3 clears: HLE (4) and RTM (11). With AVX2 and RTM both
40/// advertised, glibc picks its `*_avx2_rtm` functions, which raise SIGILL
41/// where TSX is absent (<https://github.com/rrnewton/hermit/issues/4251>).
42pub const V3_LEAF7_EBX_CLEARED: u32 = (1 << 4) | (1 << 11);
43/// Leaf 0x80000001 ECX bit v3 adds: LZCNT/ABM (5).
44pub const V3_EXT1_ECX_ADDED: u32 = 1 << 5;
45
46/// See [`crate::v3_feature_words`].
47pub(crate) fn v3_feature_words() -> Vec<(u32, u32, usize, u32)> {
48    let host = HostXstate {
49        vendor: [0; 3],
50        signature: 0,
51        xcr0: 0,
52        xsave_leaves: Vec::new(),
53        xcomp_supp: 0,
54        xcomp_perm: 0,
55        xcomp_guest_perm: 0,
56        xfeature_max: 0,
57        probe_cases: 0,
58        probe_forced_exit: false,
59    };
60    let v3 = InterceptedCpuid::v3(&host);
61    let mut words = Vec::new();
62    for (leaf, registers) in [
63        (0x1, [2, 3].as_slice()),
64        (0x7, &[1, 2, 3]),
65        (0x8000_0001, &[2, 3]),
66    ] {
67        let r = v3.cpuid(leaf, 0).unwrap_or(cpuid_result(0, 0, 0, 0));
68        let values = [r.eax, r.ebx, r.ecx, r.edx];
69        for &register in registers {
70            words.push((leaf, 0, register, values[register]));
71        }
72    }
73    words
74}

76impl InterceptedCpuid {
77    /// The x86-64-v2 table.
78    pub fn new() -> Self {
79        InterceptedCpuid {
80            profile: CpuProfile::X86_64V2,
81            xsave: [[0; 4]; XSAVE_SUBLEAVES],
82        }
83    }
84
85    /// The x86-64-v3 table for a host that passed the qualification.
86    pub fn v3(host: &HostXstate) -> Self {
87        let mut xsave = [[0; 4]; XSAVE_SUBLEAVES];
88        // Subleaf 0: the enabled components (EAX, EDX) and the save-area size
89        // for them (EBX). ECX, the size for every supported component, is
90        // given as EBX: the guest can enable nothing beyond XCR0.
91        let size = host.xsave_leaf(0).map_or(0, |[_, ebx, _, _]| ebx);
92        xsave[0] = [host.xcr0 as u32, size, size, (host.xcr0 >> 32) as u32];
93        // Subleaf 1 (XSAVEOPT, XSAVEC, XGETBV1, XSAVES) stays zero.
94        for (component, row) in xsave.iter_mut().enumerate().skip(2) {
95            if host.xcr0 & (1 << component) != 0
96                && let Some([eax, ebx, ecx, _]) = host.xsave_leaf(component as u32)
97            {
98                *row = [eax, ebx, ecx, 0];
99            }
100        }
101        InterceptedCpuid {
102            profile: CpuProfile::X86_64V3,
103            xsave,
104        }
105    }
106
107    /// The table a run's configuration selects. Under v3 the CLI sets the host
108    /// facts after the host qualifies; a v3 configuration without them is a
109    /// launch defect, so it is refused rather than given a guessed leaf 0xD.
110    pub fn for_config(cfg: &crate::Config) -> Self {
111        match cfg.cpu_profile {
112            CpuProfile::X86_64V2 => Self::new(),
113            CpuProfile::X86_64V3 => Self::v3(cfg.cpu_profile_host.as_ref().expect(
114                "--cpu-profile x86-64-v3 reached Detcore without the qualified host's xstate facts",
115            )),
116        }
117    }
118
119    pub fn cpuid(&self, leaf: u32, subleaf: u32) -> Option<CpuIdResult> {
120        let request = leaf as usize;
121        let mut result = if request >= 0x80000000 && request < 0x80000000 + EXTENDED_CPUIDS.len() {
122            EXTENDED_CPUIDS[request - 0x80000000]
123        } else if request < CPUIDS.len() {
124            CPUIDS[request]
125        } else {
126            return None;
127        };
128        if self.profile == CpuProfile::X86_64V3 {
129            match leaf {
130                0x1 => result.ecx = (result.ecx | V3_LEAF1_ECX_ADDED) & !LEAF1_ECX_HYPERVISOR,
131                0x7 => result.ebx &= !V3_LEAF7_EBX_CLEARED,
132                0xd => {
133                    let [eax, ebx, ecx, edx] =
134                        self.xsave.get(subleaf as usize).copied().unwrap_or([0; 4]);
135                    return Some(cpuid_result(eax, ebx, ecx, edx));
136                }
137                0x8000_0001 => result.ecx |= V3_EXT1_ECX_ADDED,
138                _ => {}
139            }
140        }
141
142        if subleaf != 0 && uses_subleaf(leaf) {
143            Some(cpuid_result(0, 0, 0, 0))
144        } else {
145            Some(result)
146        }
147    }
148
149    /// SHA-256 over a canonical encoding of everything this table answers and
150    /// the host facts a v3 run binds, for the recording to carry: the profile
151    /// name (length-prefixed), then each basic and extended leaf with each of
152    /// its subleaves 0-63 (only subleaf 0 for leaves without subleaves) as
153    /// little-endian u32 leaf, subleaf, EAX, EBX, ECX, EDX, sorted by leaf then
154    /// subleaf; then, under v3, XCR0, the three XCOMP masks (u64 each) and
155    /// XFEATURE_MAX (u32).
156    pub fn digest(&self, host: Option<&HostXstate>) -> [u8; 32] {
157        let mut hash = Sha256::new();
158        let name = self.profile.name().as_bytes();
159        hash.update((name.len() as u32).to_le_bytes());
160        hash.update(name);
161        let leaves =
162            (0..CPUIDS.len() as u32).chain(0x8000_0000..0x8000_0000 + EXTENDED_CPUIDS.len() as u32);
163        for leaf in leaves {
164            let subleaves = if uses_subleaf(leaf) { 64 } else { 1 };
165            for subleaf in 0..subleaves {
166                let r = self
167                    .cpuid(leaf, subleaf)
168                    .unwrap_or(cpuid_result(0, 0, 0, 0));
169                for word in [leaf, subleaf, r.eax, r.ebx, r.ecx, r.edx] {
170                    hash.update(word.to_le_bytes());
171                }
172            }
173        }
174        if let Some(host) = host {
175            for word in [
176                host.xcr0,
177                host.xcomp_supp,
178                host.xcomp_perm,
179                host.xcomp_guest_perm,
180            ] {
181                hash.update(word.to_le_bytes());
182            }
183            hash.update(host.xfeature_max.to_le_bytes());
184        }
185        hash.finalize().into()
186    }
187}

189impl InterceptedCpuid {
190    /// `/proc/cpuinfo` for the virtual CPU this table describes, so that the
191    /// file and the CPUID instruction show a guest the same CPU
192    /// (<https://github.com/rrnewton/hermit/issues/4161>). Without this, the
193    /// file shows the host's model, flags and caches, which differ from host
194    /// to host and contradict CPUID.
195    ///
196    /// It is the block Linux's `show_cpuinfo` (arch/x86/kernel/cpu/proc.c)
197    /// prints for CPU 0 of a one-CPU machine whose CPUID answers as this table
198    /// does, with each value derived as Linux 7.1.3 derives it:
199    /// - the vendor, family, model, stepping and `cpuid level` from leaves 0
200    ///   and 1, and the model name from leaves 0x80000002-4, trimmed, or
201    ///   `family/model` in hex when they hold no name (`identify_cpu`);
202    /// - the cache size, and the TLB size when Linux computes one, by the
203    ///   vendor's path: for GenuineIntel `init_intel_cacheinfo` (leaf 4,
204    ///   else leaf 2's descriptors, falling back to the extended leaves when
205    ///   no L2 is found), otherwise `cpu_detect_cache_sizes` (leaves
206    ///   0x80000005 and 0x80000006);
207    /// - the `clflush` size and `cache_alignment` from leaf 1, with
208    ///   `get_cpu_address_sizes`'s defaults, and the address sizes from leaf
209    ///   0x80000008;
210    /// - `flags`: the named bits of every CPUID word in Linux's capability
211    ///   array ([`CPUINFO_FLAGS`]), less `pat` on the Intel family 6 models
212    ///   `early_init_intel` clears it for, and `power management` from leaf
213    ///   0x80000007's EDX.
214    ///
215    /// The rest have no CPUID source and are fixed: one CPU at 1000 MHz with
216    /// the matching TSC-calibrated `bogomips` (2000.00), and no microcode
217    /// revision. `bugs` is [`CPUINFO_BUGS`], the set Linux derives for this
218    /// table; it is not recomputed from the table, and
219    /// `tests::cpuinfo_describes_the_virtual_cpu` pins it together with the
220    /// table so that a table change must revisit it. Linux's synthetic flags
221    /// (such as `constant_tsc`), which describe the kernel rather than CPUID,
222    /// are not shown.
223    pub fn cpuinfo(&self) -> Vec<u8> {
224        let bugs = match self.profile {
225            CpuProfile::X86_64V2 => CPUINFO_BUGS,
226            CpuProfile::X86_64V3 => CPUINFO_BUGS_V3,
227        };
228        render_cpuinfo(
229            |leaf, subleaf| {
230                self.cpuid(leaf, subleaf)
231                    .unwrap_or(cpuid_result(0, 0, 0, 0))
232            },
233            bugs,
234        )
235    }
236}

238/// [`InterceptedCpuid::cpuinfo`] for the CPU whose CPUID answers as `leaf`
239/// does.
240fn render_cpuinfo(leaf: impl Fn(u32, u32) -> CpuIdResult, bugs: &str) -> Vec<u8> {
241    let basic = leaf(0, 0);
242    let vendor: Vec<u8> = [basic.ebx, basic.edx, basic.ecx]
243        .iter()
244        .flat_map(|register| register.to_le_bytes())
245        .collect();
246    let signature = leaf(1, 0);
247    let base_family = (signature.eax >> 8) & 0xf;
248    let family = if base_family == 0xf {
249        base_family + ((signature.eax >> 20) & 0xff)
250    } else {
251        base_family
252    };
253    let mut model = (signature.eax >> 4) & 0xf;
254    if family >= 6 {
255        model |= ((signature.eax >> 16) & 0xf) << 4;
256    }
257    let extended_max = leaf(0x8000_0000, 0).eax;
258    let model_name = if extended_max >= 0x8000_0004 {
259        let bytes: Vec<u8> = (0x8000_0002..=0x8000_0004)
260            .map(|brand| leaf(brand, 0))
261            .flat_map(|r| [r.eax, r.ebx, r.ecx, r.edx])
262            .flat_map(|register| register.to_le_bytes())
263            .take_while(|byte| *byte != 0)
264            .collect();
265        String::from_utf8_lossy(&bytes)
266            .trim_start_matches(' ')
267            .trim_end()
268            .to_owned()
269    } else {
270        String::new()
271    };
272    let model_name = if model_name.is_empty() {
273        format!("{family:02x}/{model:02x}")
274    } else {
275        model_name
276    };
277    let intel = vendor == b"GenuineIntel";
278    let (cache_kb, tlb_pages) = if intel {
279        intel_cache_sizes(&leaf, basic.eax, extended_max)
280    } else {
281        extended_cache_sizes(&leaf, extended_max, 0)
282    };
283    // `get_cpu_address_sizes`: 64 bytes without leaf 0x80000008, and 32
284    // when leaf 1 enumerates none.
285    let mut clflush = 64;
286    if signature.edx & (1 << 19) != 0 {
287        clflush = ((signature.ebx >> 8) & 0xff) * 8;
288    }
289    if extended_max < 0x8000_0008 {
290        clflush = 64;
291    } else if clflush == 0 {
292        clflush = 32;
293    }
294    let cache_alignment = if intel && family == 15 {
295        clflush * 2
296    } else {
297        clflush
298    };
299    let (physical_bits, virtual_bits) = if extended_max >= 0x8000_0008 {
300        let sizes = leaf(0x8000_0008, 0).eax;
301        (sizes & 0xff, (sizes >> 8) & 0xff)
302    } else {
303        (36, 48)
304    };
305    let flags: Vec<&str> = CPUINFO_FLAGS
306        .iter()
307        .flat_map(|word| {
308            let value = word.register.of(leaf(word.leaf, word.subleaf));
309            word.names()
310                .filter(move |(bit, _)| value & (1 << bit) != 0)
311                .map(|(_, name)| name)
312        })
313        // `early_init_intel` clears PAT from the Pentium Pro to the Core
314        // Yonah (family 6, models 0x01-0x0e) for an erratum.
315        .filter(|name| !(*name == "pat" && intel && family == 6 && (1..=0xe).contains(&model)))
316        .collect();
317    let power = if extended_max >= 0x8000_0007 {
318        leaf(0x8000_0007, 0).edx
319    } else {
320        0
321    };
322    let power: String = (0..32)
323        .filter(|bit| power & (1 << bit) != 0)
324        .map(|bit| match POWER_FLAGS.get(bit) {
325            Some(&"") => String::new(),
326            Some(name) => format!(" {name}"),
327            None => format!(" [{bit}]"),
328        })
329        .collect();
330
331    let mut out = String::new();
332    out += "processor\t: 0\n";
333    out += &format!("vendor_id\t: {}\n", String::from_utf8_lossy(&vendor));
334    out += &format!("cpu family\t: {family}\n");
335    out += &format!("model\t\t: {model}\n");
336    out += &format!("model name\t: {model_name}\n");
337    out += &format!("stepping\t: {}\n", signature.eax & 0xf);
338    if signature.edx & (1 << 4) != 0 {
339        out += "cpu MHz\t\t: 1000.000\n";
340    }
341    if cache_kb != 0 {
342        out += &format!("cache size\t: {cache_kb} KB\n");
343    }
344    out += "physical id\t: 0\nsiblings\t: 1\ncore id\t\t: 0\ncpu cores\t: 1\n";
345    out += "apicid\t\t: 0\ninitial apicid\t: 0\n";
346    out += "fpu\t\t: yes\nfpu_exception\t: yes\n";
347    out += &format!("cpuid level\t: {}\n", basic.eax);
348    out += "wp\t\t: yes\n";
349    out += &format!("flags\t\t: {}\n", flags.join(" "));
350    out += &format!("bugs\t\t: {bugs}\n");
351    out += "bogomips\t: 2000.00\n";
352    if tlb_pages != 0 {
353        out += &format!("TLB size\t: {tlb_pages} 4K pages\n");
354    }
355    out += &format!("clflush size\t: {clflush}\n");
356    out += &format!("cache_alignment\t: {cache_alignment}\n");
357    out +=
358        &format!("address sizes\t: {physical_bits} bits physical, {virtual_bits} bits virtual\n");
359    out += &format!("power management:{power}\n\n");
360    out.into_bytes()
361}

363/// Linux's cache and TLB sizes for GenuineIntel (`init_intel_cacheinfo`,
364/// arch/x86/kernel/cpu/cacheinfo.c), in KB and 4K pages: from leaf 4 when it
365/// enumerates a cache, else from leaf 2's descriptors; with no L2 found,
366/// `cpu_detect_cache_sizes` decides. The size is the L3, else the L2, else
367/// L1i + L1d. Only `cpu_detect_cache_sizes` sets a TLB size.
368fn intel_cache_sizes(
369    leaf: &impl Fn(u32, u32) -> CpuIdResult,
370    max_leaf: u32,
371    extended_max: u32,
372) -> (u32, u32) {
373    let (mut l1i, mut l1d, mut l2, mut l3) = (0, 0, 0, 0);
374    // `intel_cacheinfo_0x4`: subleaves until the first NULL cache type.
375    let leaf4: Vec<CpuIdResult> = if max_leaf >= 4 {
376        (0..64)
377            .map(|subleaf| leaf(4, subleaf))
378            .take_while(|cache| cache.eax & 0x1f != 0)
379            .collect()
380    } else {
381        Vec::new()
382    };
383    if !leaf4.is_empty() {
384        for cache in leaf4 {
385            let kb = (((cache.ecx) + 1)
386                * ((cache.ebx & 0xfff) + 1)
387                * (((cache.ebx >> 12) & 0x3ff) + 1)
388                * ((cache.ebx >> 22) + 1))
389                / 1024;
390            match ((cache.eax >> 5) & 0x7, cache.eax & 0x1f) {
391                (1, 1) => l1d = kb,
392                (1, 2) => l1i = kb,
393                (2, _) => l2 = kb,
394                (3, _) => l3 = kb,
395                _ => {}
396            }
397        }
398    } else if max_leaf >= 2 {
399        // `intel_cacheinfo_0x2`: descriptor bytes 1-15, all NULL unless the
400        // iteration count (byte 0) is 1, and a register's NULL when its top
401        // bit is set (`cpuid_leaf_0x2`).
402        let descriptors = leaf(2, 0);
403        let registers = [
404            descriptors.eax,
405            descriptors.ebx,
406            descriptors.ecx,
407            descriptors.edx,
408        ];
409        let bytes: Vec<u8> = registers
410            .iter()
411            .flat_map(|register| {
412                if register & (1 << 31) != 0 {
413                    [0; 4]
414                } else {
415                    register.to_le_bytes()
416                }
417            })
418            .collect();
419        if bytes[0] == 0x01 {
420            for descriptor in &bytes[1..] {
421                match leaf2_cache(*descriptor) {
422                    Some(('i', kb)) => l1i += kb,
423                    Some(('d', kb)) => l1d += kb,
424                    Some(('2', kb)) => l2 += kb,
425                    Some(('3', kb)) => l3 += kb,
426                    _ => {}
427                }
428            }
429        }
430    }
431    // `intel_cacheinfo_done`.
432    let cache_kb = if l3 != 0 {
433        l3
434    } else if l2 != 0 {
435        l2
436    } else {
437        l1i + l1d
438    };
439    if l2 == 0 {
440        return extended_cache_sizes(leaf, extended_max, cache_kb);
441    }
442    (cache_kb, 0)
443}

445/// `cpu_detect_cache_sizes` (arch/x86/kernel/cpu/common.c): the cache size
446/// in KB and the TLB size in 4K pages from leaves 0x80000005 and 0x80000006,
447/// each replacing `cache_kb`, the size known before, when present.
448fn extended_cache_sizes(
449    leaf: &impl Fn(u32, u32) -> CpuIdResult,
450    extended_max: u32,
451    mut cache_kb: u32,
452) -> (u32, u32) {
453    let mut tlb_pages = 0;
454    if extended_max >= 0x8000_0005 {
455        let l1 = leaf(0x8000_0005, 0);
456        cache_kb = (l1.ecx >> 24) + (l1.edx >> 24);
457    }
458    if extended_max >= 0x8000_0006 {
459        let l2 = leaf(0x8000_0006, 0);
460        cache_kb = l2.ecx >> 16;
461        tlb_pages = ((l2.ebx >> 16) & 0xfff) + (l2.ebx & 0xfff);
462    }
463    (cache_kb, tlb_pages)
464}

466/// A leaf 2 cache descriptor's level (`i`/`d` for L1 instruction or data,
467/// `2`, `3`) and size in KB.
468fn leaf2_cache(descriptor: u8) -> Option<(char, u32)> {
469    LEAF2_CACHES.split_whitespace().find_map(|entry| {
470        let mut fields = entry.split(':');
471        let code = u8::from_str_radix(fields.next()?, 16).ok()?;
472        let level = fields.next()?.chars().next()?;
473        let kb = fields.next()?.parse().ok()?;
474        (code == descriptor).then_some((level, kb))
475    })
476}

478/// Linux 7.1.3's leaf 2 cache descriptors, `code:level:KB`
479/// (arch/x86/kernel/cpu/cpuid_0x2_table.c, `CACHE_ENTRY` rows; the TLB rows
480/// do not bear on /proc/cpuinfo).
481const LEAF2_CACHES: &str = "06:i:8 08:i:16 09:i:32 0a:d:8 0c:d:16 0d:d:16 0e:d:24 21:2:256 22:3:512 \
482    23:3:1024 25:3:2048 29:3:4096 2c:d:32 30:i:32 39:2:128 3a:2:192 3b:2:128 \
483    3c:2:256 3d:2:384 3e:2:512 3f:2:256 41:2:128 42:2:256 43:2:512 44:2:1024 \
484    45:2:2048 46:3:4096 47:3:8192 48:2:3072 49:3:4096 4a:3:6144 4b:3:8192 \
485    4c:3:12288 4d:3:16384 4e:2:6144 60:d:16 66:d:8 67:d:16 68:d:32 78:2:1024 \
486    79:2:128 7a:2:256 7b:2:512 7c:2:1024 7d:2:2048 7f:2:512 80:2:512 82:2:256 \
487    83:2:512 84:2:1024 85:2:2048 86:2:512 87:2:1024 d0:3:512 d1:3:1024 d2:3:2048 \
488    d6:3:1024 d7:3:2048 d8:3:4096 dc:3:2048 dd:3:4096 de:3:8192 e2:3:2048 \
489    e3:3:4096 e4:3:8192 ea:3:12288 eb:3:18432 ec:3:24576";

491/// The `bugs` line Linux 7.1.3 derives for this table (`cpu_set_bug_bits`,
492/// arch/x86/kernel/cpu/common.c), in bug-bit order: an Intel family 6 model 6
493/// matches no entry of its vulnerability lists, the table enumerates no
494/// IA32_ARCH_CAPABILITIES (leaf 7 EDX bit 29), so every `*_NO` bit is clear,
495/// and it advertises RTM (`taa`) and the hypervisor bit (`bhi`, `its`).
496///
497/// It is a constant, not derived from the table: a CPU profile such as
498/// <https://github.com/rrnewton/hermit/issues/4243> proposes, or any other
499/// table change, must recompute it from Linux's `cpu_set_bug_bits` for the new
500/// vendor, family, model and features.
501const CPUINFO_BUGS: &str = "cpu_meltdown spectre_v1 spectre_v2 spec_store_bypass l1tf mds swapgs taa \
502    itlb_multihit bhi spectre_v2_user its";
503
504/// The `bugs` set Linux 7.1.3's `cpu_set_bug_bits` derives for the v3 table:
505/// the v2 set without `taa` (it needs RTM, which v3 clears), `bhi` (it needs

512/// A CPUID register.

514enum Register {
515    Eax,
516    Ebx,
517    Ecx,
518    Edx,

521impl Register {
522    fn of(self, result: CpuIdResult) -> u32 {
523        match self {
524            Register::Eax => result.eax,

526            Register::Ecx => result.ecx,
527            Register::Edx => result.edx,
528        }

533/// register, with the names `/proc/cpuinfo` shows for its bits.

535    leaf: u32,

542impl CpuinfoFlagWord {
543    fn names(&self) -> impl Iterator<Item = (u32, &'static str)> {
544        self.names.split_whitespace().map(|entry| {
545            let (bit, name) = entry.split_once(':').expect("a `bit:name` entry");
546            (bit.parse().expect("a bit number"), name)
547        })
548    }

795    /// `render_cpuinfo` for a CPU whose CPUID answers with `entries`
796    /// (`(leaf, subleaf, [eax, ebx, ecx, edx])`) and zero elsewhere, and the
797    /// value of each line it renders.
798    fn render(entries: &[(u32, u32, [u32; 4])]) -> std::collections::BTreeMap<String, String> {
799        let block = render_cpuinfo(
800            |leaf, subleaf| {
801                entries
802                    .iter()
803                    .find(|(l, s, _)| *l == leaf && *s == subleaf)
804                    .map_or(cpuid_result(0, 0, 0, 0), |(_, _, [a, b, c, d])| {
805                        cpuid_result(*a, *b, *c, *d)
806                    })
807            },
808            CPUINFO_BUGS,
809        );
810        String::from_utf8(block)
811            .unwrap()
812            .lines()
813            .filter_map(|line| line.split_once(':'))
814            .map(|(key, value)| (key.trim().to_owned(), value.trim().to_owned()))
815            .collect()
816    }

984        let v3 = InterceptedCpuid::v3(&epyc_9d85());
985        let regs = |t: &InterceptedCpuid, leaf, sub| registers(t.cpuid(leaf, sub).unwrap());
986        let [_, _, ecx2, _] = regs(&v2, 1, 0);
987        let [_, _, ecx3, _] = regs(&v3, 1, 0);
988        assert_eq!(ecx3, (ecx2 | V3_LEAF1_ECX_ADDED) & !LEAF1_ECX_HYPERVISOR);
989        assert_ne!(
990            ecx2 & LEAF1_ECX_HYPERVISOR,
991            0,
992            "v2 keeps its hypervisor bit"
993        );
994        // AVX with OSXSAVE now, and FMA, MOVBE, XSAVE, F16C.
995        for bit in [12, 22, 26, 27, 28, 29] {
996            assert_ne!(ecx3 & (1 << bit), 0, "leaf 1 ECX bit {bit}");
997        }
998        // RDRAND stays masked.
999        assert_eq!(ecx3 & (1 << 30), 0);
1000        let [_, ebx7, _, _] = regs(&v3, 7, 0);
1001        assert_eq!(ebx7 & V3_LEAF7_EBX_CLEARED, 0, "HLE and RTM cleared");
1002        // AVX2, BMI1, BMI2 stay; RDSEED stays masked.
1003        for bit in [3, 5, 8] {
1004            assert_ne!(ebx7 & (1 << bit), 0, "leaf 7 EBX bit {bit}");
1005        }
1006        assert_eq!(ebx7 & (1 << 18), 0);
1007        assert_ne!(regs(&v3, 0x8000_0001, 0)[2] & V3_EXT1_ECX_ADDED, 0, "LZCNT");
1008        // Leaf 0xD: the host's enabled components and size, ECX = EBX.
1009        assert_eq!(regs(&v3, 0xd, 0), [0x2e7, 0x988, 0x988, 0]);
1010        assert_eq!(
1011            regs(&v3, 0xd, 1),
1012            [0; 4],
1013            "no XSAVEOPT, XSAVEC, XGETBV1, XSAVES"
1014        );
1015        assert_eq!(regs(&v3, 0xd, 2), [0x100, 0x240, 0, 0]);
1016        assert_eq!(regs(&v3, 0xd, 9), [0x8, 0x980, 0, 0]);
1017        assert_eq!(regs(&v3, 0xd, 3), [0; 4], "MPX is not in XCR0");
1018        assert_eq!(regs(&v3, 0xd, 11), [0; 4], "CET is not in XCR0");
1019        assert_eq!(regs(&v3, 0xd, 63), [0; 4]);
1020        // Everything else is v2's.
1021        for leaf in [0, 2, 4, 0xb, 0x8000_0000, 0x8000_0002, 0x8000_0008] {
1022            assert_eq!(regs(&v3, leaf, 0), regs(&v2, leaf, 0), "leaf {leaf:#x}");
1023        }
1024    }
