9//! Linux's CPU-time clocks, read from Detcore's logical CPU accounting.
10//!
11//! A CPU clock id is one of the two fixed ids, `CLOCK_PROCESS_CPUTIME_ID` (2)
12//! and `CLOCK_THREAD_CPUTIME_ID` (3), or a negative id that encodes a task and
13//! what to measure (include/linux/posix-timers_types.h:20-32 at v6.16):
14//! `CPUCLOCK_PID(id) = ~(id >> 3)` names the task, bit 2
15//! (`CPUCLOCK_PERTHREAD_MASK`) makes it a thread's clock rather than a
16//! process's, and the low two bits (`CPUCLOCK_WHICH`) choose PROF (0), VIRT (1)
17//! or SCHED (2). The low three bits 3 are a clock descriptor (`CLOCKFD`), not a
18//! CPU clock. `pthread_getcpuclockid` and `clock_getcpuclockid` build these ids.
19//!
20//! Detcore models a thread's CPU time as the user and system time its own
21//! logical clock has charged since the thread was created: retired branches
22//! (and RDTSC/CPUID) as user time, intercepted syscalls at their modeled cost
23//! as system time ([`DetTime::user_cpu_time`], [`DetTime::system_cpu_time`],
24//! `ThreadState::thread_cpu_time`). That is the accounting `getrusage(2)` and
25//! `times(2)` already report, so every CPU-time view of a task agrees.
26//!
27//! [`DetTime::user_cpu_time`]: crate::types::DetTime::user_cpu_time
28//! [`DetTime::system_cpu_time`]: crate::types::DetTime::system_cpu_time
29

41/// What a CPU clock measures: `CPUCLOCK_WHICH` (`cpu_clock_sample`,
42/// kernel/time/posix-cpu-timers.c:194-212 at v6.16).
43#[derive(Debug, Clone, Copy, PartialEq, Eq)]
44pub(crate) enum CpuClockKind {
45    /// `CPUCLOCK_PROF`: user plus system time.
46    Prof,
47    /// `CPUCLOCK_VIRT`: user time.
48    Virt,
49    /// `CPUCLOCK_SCHED`: the scheduler's runtime, which is what
50    /// `CLOCK_THREAD_CPUTIME_ID` and `CLOCK_PROCESS_CPUTIME_ID` read. Linux
51    /// scales `getrusage`'s and `times`'s user and system time so that they
52    /// add up to it (`cputime_adjust`, kernel/sched/cputime.c:536-541), so in
53    /// Detcore's model it is user plus system time too.
54    Sched,
55}
56
57impl CpuClockKind {
58    /// The clock's reading for a task that has used `user` and `system` time.
59    pub(crate) fn sample(self, user: LogicalTime, system: LogicalTime) -> LogicalTime {
60        match self {
61            Self::Virt => user,
62            Self::Prof | Self::Sched => user + system,
63        }
64    }
65}

67/// Whose CPU time a clock measures, by the pid its id carries. Pid 0 names
68/// the caller: its own thread for a thread's clock, its own process for a
69/// process's (`pid_for_clock`, kernel/time/posix-cpu-timers.c:66-71).
70#[derive(Debug, Clone, Copy, PartialEq, Eq)]
71pub(crate) enum CpuClockOwner {
72    Thread(i32),
73    Process(i32),
74}

76/// A CPU clock id, decoded.
77#[derive(Debug, Clone, Copy, PartialEq, Eq)]
78pub(crate) struct CpuClock {
79    pub(crate) owner: CpuClockOwner,
80    pub(crate) kind: CpuClockKind,
81}

83/// What a clock id is, as far as the id alone decides.
84#[derive(Debug, Clone, Copy, PartialEq, Eq)]
85pub(crate) enum ClockIdClass {
86    /// Not a CPU clock: a fixed clock other than 2 and 3, a clock descriptor,
87    /// or an id that names no clock at all.
88    Other,
89    /// A CPU clock of a thread or a process.
90    Cpu(CpuClock),
91    /// A negative thread-clock id whose clock bits are `CPUCLOCK_MAX` (3):
92    /// no CPU clock. `pid_for_clock` refuses it from the bits alone
93    /// (kernel/time/posix-cpu-timers.c:63-64), so every call answers EINVAL.
94    NoCpuClock,
95}

97/// Decode `clockid` (include/linux/posix-timers_types.h:20-32 at v6.16).
98pub(crate) fn classify(clockid: i32) -> ClockIdClass {
99    const PERTHREAD: i32 = 4;
100    const WHICH: i32 = 3;
101    let clock = |owner| {
102        ClockIdClass::Cpu(CpuClock {
103            owner,
104            kind: CpuClockKind::Sched,
105        })
106    };
107    match clockid {
108        libc::CLOCK_PROCESS_CPUTIME_ID => clock(CpuClockOwner::Process(0)),
109        libc::CLOCK_THREAD_CPUTIME_ID => clock(CpuClockOwner::Thread(0)),
110        id if id >= 0 => ClockIdClass::Other,
111        id => {
112            let thread = id & PERTHREAD != 0;
113            let kind = match id & WHICH {
114                0 => CpuClockKind::Prof,
115                1 => CpuClockKind::Virt,
116                2 => CpuClockKind::Sched,
117                _ if thread => return ClockIdClass::NoCpuClock,
118                _ => return ClockIdClass::Other,
119            };
120            let pid = !(id >> 3);
121            let owner = if thread {
122                CpuClockOwner::Thread(pid)
123            } else {
124                CpuClockOwner::Process(pid)
125            };
126            ClockIdClass::Cpu(CpuClock { owner, kind })
127        }
128    }
129}

131/// The task a CPU clock names, found as Linux's `pid_for_clock` finds it.
132#[derive(Debug, Clone, Copy, PartialEq, Eq)]
133pub(crate) enum ClockTask {
134    /// The calling thread.
135    OwnThread,
136    /// Another live thread of the calling process.
137    Sibling(DetTid),
138    /// The calling process.
139    OwnProcess,
140    /// Another live process, whose CPU accounting Detcore keeps in that
141    /// process's own threads, out of the caller's reach.
142    OtherProcess,
143}
144
145/// Which call looks the clock's task up. Only `clock_gettime` also takes the
146/// calling thread's own id as naming its process
147/// (kernel/time/posix-cpu-timers.c:82-89); `clock_getres` and
148/// `clock_nanosleep` require a process's id (`validate_clock_permissions`,
149/// :97-106, and `posix_cpu_timer_create`, :386-397).
150#[derive(Debug, Clone, Copy, PartialEq, Eq)]
151pub(crate) enum ClockLookup {
152    Read,
153    Other,
154}
155
156impl<T: RecordOrReplay> Detcore<T> {
157    /// The task `owner` names for the calling thread (`pid_for_clock`,
158    /// kernel/time/posix-cpu-timers.c:57-95 at v6.16): EINVAL for an id that
159    /// names no live thread of the caller's process (a thread's clock) or no
160    /// live process (a process's clock). Which process a task belongs to is
161    /// the scheduler's thread tree, the answer a `SIGEV_THREAD_ID` timer's
162    /// thread and a signal's target get.
163    pub(crate) async fn cpu_clock_task<G: Guest<Self>>(
164        &self,
165        guest: &mut G,
166        sysno: Sysno,
167        owner: CpuClockOwner,
168        lookup: ClockLookup,
169    ) -> Result<ClockTask, Error> {
170        let (named, thread) = match owner {
171            CpuClockOwner::Thread(pid) => (pid, true),
172            CpuClockOwner::Process(pid) => (pid, false),
173        };
174        if named == 0 {
175            return Ok(if thread {
176                ClockTask::OwnThread
177            } else {
178                ClockTask::OwnProcess
179            });
180        }
181        let named = self.cpu_clock_pid_task(guest, sysno, named).await?;
182        let dettid = guest.thread_state().dettid;
183        let detpid = guest.thread_state().detpid.expect("detpid unset");
184        if thread && named == dettid {
185            return Ok(ClockTask::OwnThread);
186        }
187        if !thread && (named == detpid || (lookup == ClockLookup::Read && named == dettid)) {
188            return Ok(ClockTask::OwnProcess);
189        }
190        // A thread's clock needs a thread of the caller's process
191        // (`same_thread_group`, :77-80), and a process's clock a process's
192        // id: a thread's id names its process for a signal, but not for a
193        // clock (`pid_has_task(pid, PIDTYPE_TGID)`, :91-94).
194        match crate::tool_global::signal_target_process(guest, named).await {
195            Some(process) if thread && process == detpid => Ok(ClockTask::Sibling(named)),
196            Some(process) if !thread && process == named => Ok(ClockTask::OtherProcess),
197            _ => Err(Errno::EINVAL.into()),
198        }
199    }

201    /// The task the positive `pid` of a CPU clock id names. At depth 0 the pid
202    /// is the DetTid. In a nested PID namespace it is a pid of that namespace,
203    /// translated through the scheduler's record (`scheduler::pid_namespaces`):
204    /// a pid no task of the namespace has is EINVAL, as `pid_for_clock`'s
205    /// `find_vpid` answers (:73-75); one the scheduler could not record is
206    /// refused.
207    async fn cpu_clock_pid_task<G: Guest<Self>>(
208        &self,
209        guest: &mut G,
210        sysno: Sysno,
211        pid: i32,
212    ) -> Result<DetTid, Error> {
213        if !guest.thread_state().nested_pid_namespace {
214            return Ok(DetTid::from_raw(pid));
215        }
216        match crate::tool_global::guest_pid(guest, pid).await {
217            GuestPidAnswer::Task { task, .. } => Ok(task),
218            GuestPidAnswer::NoSuchPid => Err(Errno::EINVAL.into()),
219            GuestPidAnswer::Untranslatable(reason) => Err(self
220                .refuse_cpu_clock(
221                    guest,
222                    sysno,
223                    &format!(
224                        "a CPU clock id naming pid {pid} in a nested PID namespace: {reason} (https://github.com/rrnewton/hermit/issues/4155)"
225                    ),
226                )
227                .await),
228        }
229    }
230
231    /// The current reading of `clock` for the calling thread, its task found
232    /// as `clock_gettime` finds it (`posix_cpu_clock_get`,
233    /// kernel/time/posix-cpu-timers.c:358-379 at v6.16).
234    pub(crate) async fn read_cpu_clock<G: Guest<Self>>(
235        &self,
236        guest: &mut G,
237        sysno: Sysno,
238        clock: CpuClock,
239    ) -> Result<LogicalTime, Error> {
240        let task = self
241            .cpu_clock_task(guest, sysno, clock.owner, ClockLookup::Read)
242            .await?;
243        self.cpu_clock_reading(guest, sysno, task, clock.kind).await
244    }
245
246    /// What a `kind` clock of `task` reads now: the calling thread's own CPU
247    /// time or its process's, or a sibling thread's as of that thread's last
248    /// accounting, which with threads sequentialized is where it last stopped
249    /// running. A process's time includes its exited threads'
250    /// (`thread_group_cputime`, kernel/sched/cputime.c:310-350). Another
251    /// process's clock is refused by the unsupported-operation policy.
252    pub(crate) async fn cpu_clock_reading<G: Guest<Self>>(
253        &self,
254        guest: &mut G,
255        sysno: Sysno,
256        task: ClockTask,
257        kind: CpuClockKind,
258    ) -> Result<LogicalTime, Error> {
259        let (user, system) = match task {
260            ClockTask::OwnThread => guest.thread_state().thread_cpu_time(),
261            ClockTask::Sibling(thread) => guest.thread_state().sibling_cpu_time(thread),
262            ClockTask::OwnProcess => {
263                let process = guest.thread_state_mut().process_cpu_time();
264                (process.user, process.system)
265            }
266            ClockTask::OtherProcess => {
267                return Err(self
268                    .refuse_cpu_clock(
269                        guest,
270                        sysno,
271                        "reading another process's CPU clock: Detcore keeps a process's CPU accounting in that process's own threads",
272                    )
273                    .await);
274            }
275        };
276        Ok(kind.sample(user, system))
277    }
278
279    /// Refuse a CPU-clock operation Detcore cannot serve deterministically,
280    /// by the unsupported-operation policy (`refuse_unserviceable_operation`):
281    /// a fail-closed run ends naming `reason`; with
282    /// `--allow-unsupported-syscalls` the guest gets ENOSYS.
283    pub(crate) async fn refuse_cpu_clock<G: Guest<Self>>(
284        &self,
285        guest: &mut G,
286        sysno: Sysno,
287        reason: &str,
288    ) -> Error {
289        self.refuse_unserviceable_operation_because(guest, sysno, Errno::ENOSYS, reason)
290            .await
291            .expect_err("an unserviceable operation is never served")
292    }
293}

302    /// `MAKE_PROCESS_CPUCLOCK(pid, clock)`, include/linux/posix-timers.h.
303    fn process_clock(pid: i32, which: i32) -> i32 {
304        (!pid << 3) | which
305    }
306
307    /// `MAKE_THREAD_CPUCLOCK(tid, clock)`.
308    fn thread_clock(tid: i32, which: i32) -> i32 {
309        (!tid << 3) | 4 | which
310    }
311
312    fn cpu(owner: CpuClockOwner, kind: CpuClockKind) -> ClockIdClass {
313        ClockIdClass::Cpu(CpuClock { owner, kind })
314    }

316    #[test]
317    fn fixed_cpu_clocks_name_the_caller() {
318        assert_eq!(
319            classify(libc::CLOCK_PROCESS_CPUTIME_ID),
320            cpu(CpuClockOwner::Process(0), CpuClockKind::Sched)
321        );
322        assert_eq!(
323            classify(libc::CLOCK_THREAD_CPUTIME_ID),
324            cpu(CpuClockOwner::Thread(0), CpuClockKind::Sched)
