9//! Detcore is a Reverie tool that determinizes the execution of a process.
10//!
11//! # Backend-abstraction commandment
12//!
13//! Detcore is a *tool* written against Reverie's **abstract** instrumentation
14//! interface (the `reverie` crate). It depends only on those traits and types
15//! and is deliberately ignorant of how a guest is actually traced.
16//!
17//! Detcore MUST NEVER depend on or import a concrete Reverie backend or support
18//! crate -- any `reverie-*` crate other than the abstract `reverie-core`
19//! interface. Choosing and instantiating a backend, and running a detcore tool
20//! against it, is the sole responsibility of the `hermit-cli` package. There
21//! are no backend-specific hacks in detcore: any tracing-mechanism-specific
22//! behavior belongs behind the Reverie abstraction, not here.
23//!
24//! Why: Hermit follows Reverie's abstract model. A backend dependency in
25//! detcore would couple the determinism engine to one tracing mechanism and
26//! break the clean abstraction boundary that lets the same tool run over any
27//! backend.
28//!
29//! The one allowed exception is test-only: detcore's own integration tests
30//! (under `detcore/tests/`, wired via the `reverie-ptrace` **dev-dependency**)
31//! drive a real tracer to exercise the tool. That coupling never reaches the
32//! shipped library. This invariant is enforced in CI by
33//! `scripts/check-detcore-backend-abstraction.sh`.

226#[cold]
227fn report_rcb_overshoot(
228    panic_on_rcb_overshoot: bool,
229    clock_value: u64,
230    delta_rcbs: u64,
231    last_timer: u64,
232) {
233    let message = format!(
234        "prehook: PMU RCB overshoot! Clock_value: {}. Stepped forward {} RCBs, but should have trapped at {}",
235        clock_value, delta_rcbs, last_timer
236    );
237    if panic_on_rcb_overshoot {
238        panic!("{}", message);
239    }
240    error!("{}", message);
241}

243fn rcb_timer_overshot(delta_rcbs: u64, last_timer: u64) -> bool {
244    delta_rcbs > last_timer
245}

247fn choose_rcb_timer(
248    max_rcbs_remaining: u64,
249    current_rcbs: u64,
250    next_interrupt: Option<u64>,
251) -> (u64, bool) {
252    if let Some(next_interrupt) = next_interrupt {
253        let interrupt_rcbs = next_interrupt - current_rcbs;
254        if interrupt_rcbs < max_rcbs_remaining {
255            return (interrupt_rcbs, false);
256        }
257    }
258    (max_rcbs_remaining, true)
259}

320    /// Defense-in-depth determinism for the registers the syscall instruction
321    /// clobbers.
322    ///
323    /// On x86-64 the `syscall` instruction destroys `%rcx` (which the CPU loads
324    /// with the return instruction pointer) and `%r11` (the saved `RFLAGS`).
325    /// After a syscall these are architecturally "undefined", so hermit must not
326    /// assume a well-behaved guest ignores them: a misbehaving guest that reads
327    /// `%rcx`/`%r11` must still observe deterministic values. Reverie's
328    /// injected-syscall path can otherwise leave its *private trampoline page's*
329    /// RIP/RFLAGS in these registers, which is both nondeterministic and an
330    /// information leak of tracer internals.
331    ///
332    /// This forces both registers to the guest's own (deterministic) RIP and
333    /// RFLAGS, which is exactly what a faithful `SYSRET` would leave there. It is
334    /// a no-op when they already hold the canonical values (the common path), so
335    /// it only writes registers when something diverged. Register-preserved
336    /// arguments (`%rdi`..`%r9`, callee-saved) are deliberately left untouched:
337    /// the Linux ABI preserves them, so zeroing them would break faithful,
338    /// well-behaved programs.
339    #[cfg(target_arch = "x86_64")]
340    async fn canonicalize_syscall_clobbers<G: Guest<Self>>(&self, guest: &mut G) {
341        let mut regs = guest.regs().await;
342        // A faithful SYSRET leaves the return RIP in %rcx and RFLAGS in %r11.
343        if regs.rcx != regs.rip || regs.r11 != regs.eflags {
344            regs.rcx = regs.rip;
345            regs.r11 = regs.eflags;
346            if let Err(err) = guest.set_regs(regs).await {
347                // Best-effort: some backends cannot write registers. Do not fail
348                // the syscall over a defense-in-depth hardening step.
349                debug!(
350                    "canonicalize_syscall_clobbers: set_regs unsupported/failed: {}",
351                    err
352                );
353            }
354        }
355    }

361    /// Update logical thread time with any outstanding ticks of the Reverie clock.  Returns a list
362    /// of corresponding Branch/OtherInstructions events if schedule recording is enabled.
363    ///
364    /// # Arguments
365    ///
366    /// * `precise_branch`: if true, there were no non-branch instructions since the last recorded branch instruction.
367    async fn update_logical_time_rcbs<G: Guest<Self>>(
368        &self,
369        guest: &mut G,
370        precise_branch: bool,
371    ) -> Option<Vec<SchedEvent>> {
372        if self.cfg.max_timeslice.is_some() {
373            let precise_timers = !guest.config().imprecise_timers;
374            // TODO(T86591083): we might need to not always increment as a hack fix
375            // for deterministic virtual time without sequentialize threads.
376            let clock_value = guest.read_clock().expect("Couldn't read clock");
377            // N.B. clock_value does not yet include any updates for the inbound
378            // syscall/instruction because this function is the very first thing that
379            // happens in each type of handler.
380            let thread_state = guest.thread_state_mut();
381            let dettid = thread_state.dettid;
382            assert!(thread_state.committed_clock_value <= clock_value);
383            let delta_rcbs: u64 = clock_value - thread_state.committed_clock_value;
384            if self.cfg.use_rcb_time() {
385                // AUTONOMOUS-BOT-IMPLEMENTED
386                // TODO-HUMAN-REVIEW(PR-1151)
387                if thread_state.chaos_slowdown_active {
388                    let factor = thread_state.rcb_time_multiplier();
389                    thread_state
390                        .thread_logical_time
391                        .add_rcbs_with_multiplier(delta_rcbs, factor);
392                } else {
393                    thread_state.thread_logical_time.add_rcbs(delta_rcbs);
394                }
395            }
396            thread_state.account_process_cpu_time();
397            thread_state.committed_clock_value = clock_value;
398
399            if thread_state.end_of_timeslice.is_some() {
400                if let Some(last_timer) = thread_state.last_rcb_timer
401                    && rcb_timer_overshot(delta_rcbs, last_timer)
402                    && precise_timers
403                {
404                    report_rcb_overshoot(
405                        self.cfg.panic_on_rcb_overshoot,
406                        clock_value,
407                        delta_rcbs,
408                        last_timer,
409                    );
410                    // Preserve timer state. `pre_handler_hook` will yield through the normal
411                    // scheduler path if the slice expired; `post_handler_hook` will otherwise
412                    // re-arm an overshot `interrupt_at` timer.
413                }
414                // Otherwise we're very early, at the prehook of handle_thread_start.
415            } else {
416                panic!(
417                    "Invariant violation: end_of_timeslice is None during update_logical_time_rcbs..."
418                )
419            }
420
421            trace!(
422                "[dtid {}] updated rcb clock, new logical time: {:?}, i.e. {}, timeslice end: {}, local rcb clock_value {:?}",
423                dettid,
424                &thread_state.thread_logical_time,
425                thread_state.thread_logical_time.as_nanos(),
426                thread_state
427                    .end_of_timeslice
428                    .map_or_else(|| "".to_string(), |x| format!("{}", x)),
429                clock_value,
430            );
431            if self.cfg.use_rcb_time() && self.cfg.should_trace_schedevent() {
432                let mut vec = Vec::new();
433                let ev = with_guest_time(
434                    guest,
435                    SchedEvent::branches(
436                        dettid,
437                        delta_rcbs
438                            .try_into()
439                            .expect("should not have more than 2^32 branches at once"),
440                    ),
441                );
442                let ev = if precise_branch {
443                    with_guest_rip(guest, ev).await
444                } else {
445                    ev
446                };
447
448                if delta_rcbs > 0 {
449                    // We don't fill the end_rip here, because the current rip is NOT precisely the
450                    // end of this block of branch events.  Other instructions may have occured
451                    // since the last branch.
452                    vec.push(ev)
453                } else {
454                    trace!(
455                        "[detcore, dtid {}] Refusing to record zero-braches event: {:?}",
456                        &ev.dettid, ev
457                    );
458                }
459                if !precise_branch {
460                    // This will ALWAYS record, even if the branches above are zero.
461                    let ev2 = with_guest_time(
462                        guest,
463                        SchedEvent {
464                            dettid,
465                            op: Op::OtherInstructions,
466                            count: 1,
467                            start_rip: None,
468                            end_rip: None,
469                            end_time: None,
470                        },
471                    );
472                    // Fill in end_rip because current rip represents the end of this event.
473                    let ev2 = with_guest_rip(guest, ev2).await;
474                    vec.push(ev2);
475                }
476                Some(vec)
477            } else {
478                None
479            }
480        } else {
481            None
482        }
483    }

532    /// A common hook called at the end of *every* handler, just before returning control
533    /// to the guest. This enforces the logical target and resets the PMU maximum timer.
534    ///
535    /// However, note that the thread's timeslice (turn) may have expired DURING this handler.
536    /// Therefore the timeslice can end in the posthook as well as in the prehook.
537    async fn post_handler_hook<G: Guest<Self>>(&self, guest: &mut G) {
538        self.end_timeslice_if_needed(guest).await;
539
540        let dettid = guest.thread_state().dettid;
541        let mut current_time = guest.thread_state().thread_logical_time.as_nanos();
542
543        if let Some(mut max_timeslice_end) = guest.thread_state().max_timeslice_end {
544            assert!(guest.config().max_timeslice.is_some());
545            let mut replay_rcb_end = guest.thread_state().replay_rcb_end;
546            // TODO: get rid of fractional NANOS_PER_RCB so it's clear that this does not lose precision:
547            // AUTONOMOUS-BOT-IMPLEMENTED
548            // TODO-HUMAN-REVIEW(PR-1151)
549            let clock_multiplier = guest.config().clock_multiplier.unwrap_or(1.0)
550                * guest.thread_state().rcb_time_multiplier().as_f64();
551            let epsilon = Duration::from_nanos((NANOS_PER_RCB * clock_multiplier).ceil() as u64);
552
553            if replay_rcb_end.is_none() && current_time + epsilon > max_timeslice_end {
554                trace!(
555                    "posthook [dtid {}] less than one RCB remains before PMU maximum {}; ending slice",
556                    dettid, max_timeslice_end
557                );
558                self.end_timeslice(guest).await;
559                max_timeslice_end = guest
560                    .thread_state()
561                    .max_timeslice_end
562                    .expect("ending a PMU-backed timeslice must install a new maximum");
563                current_time = guest.thread_state().thread_logical_time.as_nanos();
564                replay_rcb_end = guest.thread_state().replay_rcb_end;
565            }
566            if replay_rcb_end.is_none() && current_time + epsilon > max_timeslice_end {
567                panic!(
568                    "Ended time slice, but current time {} is still beyond PMU maximum {}",
569                    current_time, max_timeslice_end
570                );
571            }
572
573            let current_rcbs = guest.thread_state().thread_logical_time.rcbs();
574            let current_pmu_rcbs = guest.thread_state().committed_clock_value;
575            let (ns_remaining, max_rcbs_remaining) = if let Some(replay_rcb_end) = replay_rcb_end {
576                assert!(
577                    replay_rcb_end > current_pmu_rcbs,
578                    "recorded PMU RCB deadline {} is not ahead of current {}",
579                    replay_rcb_end,
580                    current_pmu_rcbs
581                );
582                let logical_remaining = if max_timeslice_end > current_time {
583                    max_timeslice_end - current_time
584                } else {
585                    LogicalTime::ZERO
586                };
587                (logical_remaining, replay_rcb_end - current_pmu_rcbs)
588            } else {
589                let logical_remaining = max_timeslice_end - current_time;
590                (
591                    logical_remaining,
592                    logical_remaining.into_rcbs_with_multiplier(clock_multiplier),
593                )
594            };
595            let next_interrupt = self
596                .cfg
597                .use_rcb_time()
598                .then(|| {
599                    guest
600                        .thread_state()
601                        .interrupt_at
602                        .range((current_rcbs + 1)..)
603                        .next()
604                        .copied()
605                })
606                .flatten();
607            let (rcbs_remaining, timer_is_max) =
608                choose_rcb_timer(max_rcbs_remaining, current_rcbs, next_interrupt);
609            if let Some(next_interrupt) = next_interrupt {
610                debug!(
611                    "[dtid: {}] current rcbs: {}, next interrupt_at: {}",
612                    dettid, current_rcbs, next_interrupt
613                )
614            }
615
616            trace!(
617                "posthook [dtid {}] {} remaining before PMU maximum ({} rcbs).",
618                dettid, ns_remaining, rcbs_remaining,
619            );
620
621            if replay_rcb_end.is_none() && ns_remaining.is_zero() {
622                panic!(
623                    "Timer invariant broken: we should not exit a handler with 0 timeslice remaining."
624                );
625            }
626            assert!(rcbs_remaining > 0);
627            trace!(
628                "posthook [dtid {}] Resetting timer to {:?} RCBs in the future (current {})",
629                dettid,
630                rcbs_remaining,
631                guest.thread_state().thread_logical_time.rcbs()
632            );
633            {
634                let thread_state = guest.thread_state_mut();
635                thread_state.last_rcb_timer = Some(rcbs_remaining);
636                thread_state.last_rcb_timer_is_max = timer_is_max;
637            }
638
639            if guest.config().imprecise_timers {
640                guest
641                    .set_timer(TimerSchedule::Rcbs(rcbs_remaining))
642                    .expect("Failed to set timer");
643            } else {
644                guest
645                    .set_timer_precise(TimerSchedule::Rcbs(rcbs_remaining))
646                    .expect("Failed to set timer");
647            }
648        } else {
649            assert!(guest.config().max_timeslice.is_none());
650            guest.thread_state_mut().last_rcb_timer = None;
651            guest.thread_state_mut().last_rcb_timer_is_max = false;
652        }
653
654        if guest.thread_state().guest_past_first_execve() {
655            detlog_debug!(
656                "(post) registers [dtid {}][rcbs {}]. {}",
657                dettid,
658                guest.thread_state().thread_logical_time.rcbs(),
659                guest.regs().await.display(),
660            );
661        }
662    }

668    ///  - priority change / yield RPC

677    async fn end_timeslice_with_sched_yield<G: Guest<Self>>(
678        &self,
679        guest: &mut G,
680        mut explicit_sched_yield: bool,
681    ) {
682        let chaos = guest.config().chaos;
683        loop {
684            let thread_state = guest.thread_state();
685            let dettid = thread_state.dettid;
686            let end_time = thread_state.thread_logical_time.as_nanos();
687            info!(
688                "[detcore, dtid {}] ending timeslice T{}. {} syscalls and {} signals this timeslice.",
689                dettid,
690                thread_state.stats.timeslice_count,
691                thread_state.stats.timeslice_syscall_count,
692                thread_state.stats.timeslice_signal_count,
693            );
694            let maybe_prio = guest.thread_state_mut().next_timeslice(&self.cfg);
695
696            // Depending on chaos mode, a received timer event is either a preemption or a changepoint
697            let req = if let Some(prio) = maybe_prio {
698                Self::priority_changepoint_request(guest, end_time, prio)
699            } else if chaos {
700                Self::random_priority_changepoint_request(guest, end_time)
701            } else if explicit_sched_yield && self.cfg.replay_schedule_from.is_none() {
702                Self::sched_yield_request(guest)
703            } else {
704                Self::yield_request(guest)
705            };
706            resource_request(guest, req).await;
707
708            // AUTONOMOUS-BOT-IMPLEMENTED
709            // TODO-HUMAN-REVIEW(PR-1151)
710            // Multiple scheduler commits can occur without an intervening
711            // conditional branch. Exact-RCB replay represents those as
712            // adjacent zero-RCB slices, which must be consumed before the
713            // guest resumes.
714            if !guest.thread_state().timeslice_expired() {
715                break;
716            }
717            explicit_sched_yield = false;
718        }
719    }

721    /// Hash the guest REGISTER FILE and log it.
722    ///
723    /// # The sampling boundary: GUEST-LOGICAL CONTROL, never handler interior
724    ///
725    /// This is called from exactly one place -- immediately after a syscall has finished and its
726    /// result has been written back, before the guest resumes. At that instant the guest
727    /// LOGICALLY HAS CONTROL: the architectural state is what the guest itself would observe at
728    /// its own RIP, and it is the same instant at which stack and heap are already hashed.
729    ///
730    /// It is deliberately NOT sampled anywhere inside a tool handler. A backend that runs its
731    /// handler IN-GUEST (sabre, liteinst, e9patch) executes instructions the ptrace reference
732    /// never executes, using guest registers as scratch while it does. Registers there
733    /// legitimately differ across backends, so comparing them would report correct behaviour as a
734    /// divergence and burn the prefix-depth ratchet on artifacts. Handler-interior state is out of
735    /// the domain, not excluded from it by a filter -- the same "define the domain" rule the heap
736    /// definition follows.
737    ///
738    /// # What is in the hash, and what is deliberately not
739    ///
740    /// Included: the general-purpose registers the guest can observe, `rip`, `rsp`, `rflags`,
741    /// `orig_rax`, and the TLS bases `fs_base`/`gs_base`.
742    ///
743    /// EXCLUDED, with reasons rather than by convenience:
744    /// * `rcx` and `r11` -- architecturally clobbered by the `SYSCALL` instruction, which stores
745    ///   the return RIP and RFLAGS in them. They carry no information beyond `rip`/`eflags`, which
746    ///   ARE hashed, and a patching backend that reaches the kernel by some route other than a
747    ///   bare `SYSCALL` will leave different values there for a reason that is not a determinism
748    ///   defect.
749    /// * The segment selectors `cs`/`ss`/`ds`/`es`/`fs`/`gs` -- constant for a 64-bit userspace
750    ///   guest, so they add no signal; the TLS BASES are what a guest actually observes and those
751    ///   are hashed.
752    fn detlog_registers<G: Guest<Self>>(
753        &self,
754        guest: &mut G,
755        regs: &libc::user_regs_struct,
756        seq: u64,
757    ) {
758        if !self.cfg.detlog_regs {
759            return;
760        }
761        // COST TIER: cadence 1 == full (every control point); N > 1 == spot-check every Nth.
762        // The cadence index is a PER-THREAD counter, NOT a shared one: a global atomic would be
763        // incremented in whatever order threads happen to reach it, so the cadence -- and
764        // therefore which points got sampled -- would itself be nondeterministic. A determinism
765        // instrument must not have a nondeterministic sampling schedule.
766        //
767        // It is `stats.syscall_count`, which starts at ZERO, rather than the syscall ORDINAL used
768        // in the log (which starts at 2). With the ordinal, a guest whose control points never
769        // land on a multiple of the cadence emitted NOTHING and the run still reported PASS -- a
770        // spot-tier green backed by zero samples. Indexing from zero makes the first control point
771        // of every thread always sampled, so a spot-tier run can never be silently empty.
772        let cadence = self.cfg.detlog_regs_cadence.max(1);
773        let index = {
774            let stats = &mut guest.thread_state_mut().stats;
775            let i = stats.regs_sample_index;
776            stats.regs_sample_index = i.saturating_add(1);
777            i
778        };
779        let _ = seq;
780        if !index.is_multiple_of(cadence) {
781            return;
782        }
783        let tier = if cadence == 1 {
784            "full".to_string()
785        } else {
786            format!("spot-1/{cadence}")
787        };
788        let mut bytes = Vec::with_capacity(19 * 8);
789        for v in [
790            regs.rax,
791            regs.rbx,
792            regs.rdx,
793            regs.rsi,
794            regs.rdi,
795            regs.rbp,
796            regs.rsp,
797            regs.r8,
798            regs.r9,
799            regs.r10,
800            regs.r12,
801            regs.r13,
802            regs.r14,
803            regs.r15,
804            regs.rip,
805            regs.eflags,
806            regs.orig_rax,
807            regs.fs_base,
808            regs.gs_base,
809        ] {
810            bytes.extend_from_slice(&v.to_le_bytes());
811        }
812        detlog!(
813            "[registers][dtid {}] control_point=syscall-exit tier={} {}",
814            guest.thread_state().dettid,
815            tier,
816            Digest::new(&bytes)
817        );
818    }

820    fn detlog_memory_maps<G: Guest<Self>>(&self, guest: &mut G) -> Result<(), reverie::Error> {
821        if !(self.cfg.detlog_stack || self.cfg.detlog_heap) {
822            // Don't incur the *significant* performance penalty for reading
823            // /proc/maps unless one of these flags is enabled.
824            return Ok(());
825        }
826        // Out-of-process backends (e.g. KVM) report their guest memory regions
827        // directly, because `guest.pid()` is the host VMM process there and its
828        // `/proc/<pid>/maps` describes the VMM, not the guest address space.
829        // Reading those host addresses through `guest.memory()` (guest-address
830        // space) would fault and abort the syscall. When the backend supplies
831        // regions, hash those guest ranges; otherwise fall back to the ptrace
832        // path of parsing `/proc/<pid>/maps`.
833        if let Some(regions) = guest.detlog_memory_regions() {
834            for region in regions {
835                let want = match region.kind {
836                    reverie::DetlogRegionKind::Stack => self.cfg.detlog_stack,
837                    reverie::DetlogRegionKind::Heap => self.cfg.detlog_heap,
838                };
839                if !want {
840                    continue;
841                }
842                let dettid = guest.thread_state().dettid;
843                detlog!(
844                    "[memory][dtid {}] {:?} {:#x}-{:#x}->{}",
845                    dettid,
846                    region.kind,
847                    region.start,
848                    region.end,
849                    procmaps::compute_hash_range(guest, region.start, region.end)?
850                )
851            }
852            return Ok(());
853        }
854        for mmap in procmaps::from_pid(guest.pid(), |map| match map.pathname {
855            procmaps::MMapPath::Stack if self.cfg.detlog_stack => true,
856            procmaps::MMapPath::Heap if self.cfg.detlog_heap => true,
857            _ => false,
858        })? {
859            let dettid = guest.thread_state().dettid;
860            detlog!(
861                "[memory][dtid {}] {}->{}",
862                dettid,
863                procmaps::display(&mmap),
864                procmaps::compute_hash(guest, &mmap)?
865            )
866        }
867        Ok(())
868    }

1239    fn init_thread_state(
1240        &self,
1241        tid: Tid,

1243    ) -> Self::ThreadState {

1247            .record_or_replay

1382    async fn handle_thread_start<G: Guest<Self>>(&self, guest: &mut G) -> Result<(), Error> {
1383        let detpid = DetPid::from_raw(guest.pid().into());
1384        let new_dettid = DetTid::from_raw(guest.tid().into()); // TODO(T78538674): virtualize pid/tid:
1385        trace!(
1386            "[tid {}] detcore handle_thread_start, pid={}",
1387            guest.tid(),
1388            detpid
1389        );
1390
1391        // Delayed initialization of thread_state for this new thread:
1392        let thread_state = guest.thread_state_mut();
1393        thread_state.detpid = Some(detpid);
1394        if thread_state.recover_process_mm_id(detpid) {
1395            debug!(
1396                "[detcore, dtid {}] recovered process memory identity {} for unparented thread state",
1397                new_dettid, detpid
1398            );
1399        }
1400
1401        assert_eq!(new_dettid, guest.thread_state().dettid);
1402
1403        if let Some(vfork) = guest.thread_state_mut().pending_vfork.take() {
1404            create_vfork_child_thread(guest, new_dettid, vfork).await;
1405        } else if guest.is_root_thread() {
1406            // There is no fork event to catch for the root thread.
1407            debug!(
1408                "[detcore, dtid {}] root thread start, scheduling.. full config:\n {:?}",
1409                &new_dettid,
1410                guest.config()
1411            );
1412            if let Some(post_exec_mm) = create_child_thread(guest, new_dettid, 0, None).await {
1413                guest.thread_state_mut().mm_id = post_exec_mm;
1414            }
1415        }
1416
1417        // Except for the root task, let's block until it's our turn to go:
1418        let th = tool_global::thread_start_request(&self.cfg, guest, self.detpid).await;
1419
1420        // Finish the delayed initialization of the full threadstate:
1421        {
1422            let ts = guest.thread_state_mut();
1423            ts.preemption_points = th.map(|x| x.into_iter());
1424            ts.next_timeslice(&self.cfg); // Must be after preemption_points is set.
1425        }
1426
1427        // The prehook is a noop for a thread just starting.  Can't end the timeslice.  There's no
1428        // RCB progress to record.  However, we call it for consistency with all the other handlers.
1429        self.pre_handler_hook(guest, true).await;
1430        // ^ precise_branch=true: There should have been ZERO prior instructions before this,
1431        // because the thread hasn't done anything yet.
1432
1433        self.record_or_replay
1434            .handle_thread_start(&mut guest.into_guest())
1435            .await?;
1436
1437        self.post_handler_hook(guest).await;
1438        Ok(())
1439    }

1441    async fn handle_post_exec<G: Guest<Self>>(&self, guest: &mut G) -> Result<(), Errno> {
1442        guest.thread_state_mut().past_global_first_execve = true;
1443        tool_global::mark_past_first_execve(guest).await;
1444        self.pre_handler_hook(guest, false).await;
1445
1446        if let Some(ptr) = guest.auxv().at_random() {
1447            // It is safe to mutate this address since libc has not yet had a
1448            // chance to modify or copy the auxv table.
1449            let bytes: [u8; 16] = guest.thread_state_mut().thread_prng().random();
1450            detlog!(
1451                "[post_exec, dtid {}] init auxv AT_RANDOM value to {:?}",
1452                guest.thread_state().dettid,
1453                bytes
1454            );
1455            let ptr = unsafe { ptr.into_mut() };
1456            guest.memory().write_value(ptr, &bytes)?;
1457        }
1458
1459        self.post_handler_hook(guest).await;
1460        Ok(())
1461    }

1464    async fn handle_timer_event<G: Guest<Self>>(&self, guest: &mut G) {
1465        info!(
1466            "[detcore, dtid {}] inbound timer preemption event",
1467            guest.thread_state().dettid
1468        );
1469        if guest.config().preemption_stacktrace {
1470            let mut file_writer: Box<dyn Write> =
1471                match &guest.config().preemption_stacktrace_log_file {
1472                    Some(path) => Box::new(
1473                        File::create(path).expect("Failed to open preemption stacktrace log file"),
1474                    ),
1475                    None => Box::new(std::io::stderr()),
1476                };
1477            let ts = guest.thread_state();
1478            writeln!(
1479                file_writer,
1480                "\n>>> Guest tid {} preempted at thread time {} with stack trace:",
1481                ts.dettid,
1482                ts.thread_logical_time.as_nanos(),
1483            )

1485            if let Some(backtrace) = guest.backtrace() {
1486                if let Ok(pbt) = backtrace.pretty() {
1487                    writeln!(file_writer, "{}", pbt).unwrap();
1488                } else {
1489                    writeln!(file_writer, "{}", backtrace).unwrap();
1490                }
1491            } else {
1492                warn!("Could not read backtrace!");
1493            }
1494        }
1495        // This may LOOK like a noop, but actually all of the logic for ending the timeslice is in
1496        // the prehook.  All the timer has to do is interrupt the guest and generate an extra call
1497        // to this prehook.
1498        self.pre_handler_hook(guest, true).await;
1499        if guest.config().no_rcb_time && guest.thread_state().last_rcb_timer_is_max {
1500            let max_timeslice_end = guest
1501                .thread_state()
1502                .max_timeslice_end
1503                .expect("PMU maximum timer requires a deadline");
1504            guest
1505                .thread_state_mut()
1506                .thread_logical_time
1507                .advance_to(max_timeslice_end);
1508            if self.cfg.should_trace_schedevent() {
1509                let dettid = guest.thread_state().dettid;
1510                let ev = with_guest_time(
1511                    guest,
1512                    SchedEvent {
1513                        dettid,
1514                        op: Op::OtherInstructions,
1515                        count: 1,
1516                        start_rip: None,
1517                        end_rip: None,
1518                        end_time: None,
1519                    },
1520                );
1521                let ev = with_guest_rip(guest, ev).await;
1522                trace_schedevent(guest, ev, false).await;
1523            }
1524            if self.cfg.replay_schedule_from.is_some() {
1525                let fallback_deadline = max_timeslice_end
1526                    + Duration::from_nanos(u64::from(
1527                        self.cfg
1528                            .max_timeslice
1529                            .expect("PMU maximum must be configured"),
1530                    ));
1531                let thread_state = guest.thread_state_mut();
1532                let replay_deadline = thread_state
1533                    .end_of_timeslice
1534                    .filter(|deadline| *deadline > max_timeslice_end)
1535                    .unwrap_or(fallback_deadline);
1536                thread_state.end_of_timeslice = Some(replay_deadline);
1537                thread_state.max_timeslice_end = Some(replay_deadline);
1538                thread_state.last_rcb_timer = None;
1539                thread_state.last_rcb_timer_is_max = false;
1540                thread_state.stats.reset_timeslice();
1541            } else {
1542                self.end_timeslice(guest).await;
1543            }
1544        }
1545        self.post_handler_hook(guest).await;
1546    }

1548    async fn handle_syscall_event<G: Guest<Self>>(
1549        &self,
1550        guest: &mut G,
1551        call: Syscall,
1552    ) -> Result<i64, Error> {
1553        self.pre_handler_hook(guest, false).await;
1554
1555        let dettid = guest.thread_state().dettid;
1556
1557        if guest.thread_state().guest_past_first_execve() {
1558            detlog!(
1559                "[syscall][detcore, dtid {}] inbound syscall: {} = ?",
1560                dettid,
1561                call.display(&guest.memory())
1562            );
1563        }
1564
1565        // The hot syscall path only reads a few Copy flags from the config, so copy
1566        // just those out instead of cloning the entire Config on every intercepted
1567        // syscall (previously flagged inline as an unnecessary copy). guest.config()
1568        // borrows guest immutably; bind the flags in a tight scope so the borrow ends
1569        // before the later thread_state_mut()/&mut guest borrows below.
1570        let (sequentialize_threads, virtualize_time, panic_on_unsupported_syscalls) = {
1571            let config = guest.config();
1572            (
1573                config.sequentialize_threads,
1574                config.virtualize_time,
1575                config.panic_on_unsupported_syscalls,
1576            )
1577        };
1578
1579        if sequentialize_threads && self.cfg.should_trace_schedevent() {
1580            trace_schedevent(
1581                guest,
1582                with_guest_time(
1583                    guest,
1584                    SchedEvent::syscall(dettid, call.number(), SyscallPhase::Prehook),
1585                ),
1586                true,
1587            )
1588            .await;
1589        }
1590
1591        let syscall_cost_ns = syscall_time::cost_ns(call.number());
1592        let new_count = {
1593            // which results from not being able to borrow guest twice.
1594            let thread_state = guest.thread_state_mut();
1595            thread_state.stats.count_syscall();
1596
1597            // Every intercepted syscall advances logical time, including configurations that do
1598            // not serialize threads. This keeps virtual clocks productive during syscall loops.
1599            thread_state
1600                .thread_logical_time
1601                .add_syscall_with_cost(syscall_cost_ns);
1602            thread_state.account_process_cpu_time();
1603            thread_state.stats.syscall_count
1604        };
1605
1606        // Happens-before enforcement checkpoint. When the run carries a
1607        // happens-before program with syscall-count anchors, every intercepted
1608        // syscall checks in with the scheduler at its prehook, carrying this
1609        // thread's running syscall count. The scheduler fires any anchor at
1610        // `Position::SyscallCount(new_count)` on this thread and parks the thread
1611        // (out of the run queue) when that anchor is the AFTER endpoint of a Hard
1612        // edge whose BEFORE endpoint has not fired yet. This is the gate that
1613        // makes an authored partial order reproduce a known race deterministically
1614        // (see detcore-model `happens_before`). It requires sequentialized
1615        // threads (enforced by the CLI) so the scheduler owns ordering.
1616        if guest
1617            .config()
1618            .happens_before
1619            .as_ref()
1620            .is_some_and(|p| p.has_syscall_count_anchors())
1621        {
1622            let request = guest.thread_state().mk_request(
1623                ResourceID::HappensBeforeCheckpoint(new_count),
1624                Permission::R,
1625            );
1626            resource_request(guest, request).await;
1627        }
1628
1629        let res = match classify_syscall(call.number()) {
1630            // Rseq is not type-safe in the pinned Reverie revision. Dispatch by Sysno so a
1631            // future typed representation preserves this explicit policy.
1632            SyscallClassification::Determinized if call.number() == Sysno::rseq => {
1633                if panic_on_unsupported_syscalls {
1634                    Err(Error::Errno(Errno::ENOSYS))
1635                } else {
1636                    self.passthrough(guest, call).await
1637                }
1638            }
1639            // AUTONOMOUS-BOT-IMPLEMENTED
1640            // TODO-HUMAN-REVIEW(#663)
1641            // The pinned Reverie revision exposes process_madvise only as a raw call.
1642            SyscallClassification::Determinized if call.number() == Sysno::process_madvise => {
1643                match call {
1644                    Syscall::Other(_, args) => Self::handle_process_madvise(args.arg0, args.arg4),
1645                    _ => unreachable!("process_madvise unexpectedly gained a typed variant"),
1646                }
1647            }
1648            // AUTONOMOUS-BOT-IMPLEMENTED
1649            // TODO-HUMAN-REVIEW(PR-1175): The pinned Reverie revision exposes
1650            // pidfd_send_signal/pidfd_getfd only as raw calls, so dispatch on the
1651            // Sysno. See the handlers in syscalls/files.rs for the determinism
1652            // argument.
1653            SyscallClassification::Determinized if call.number() == Sysno::pidfd_send_signal => {
1654                match call {
1655                    Syscall::Other(_, args) => {
1656                        self.handle_pidfd_send_signal(
1657                            guest,
1658                            call,
1659                            args.arg0 as RawFd,
1660                            args.arg3 as u32,
1661                        )
1662                        .await
1663                    }
1664                    _ => unreachable!("pidfd_send_signal unexpectedly gained a typed variant"),
1665                }
1666            }
1667            // AUTONOMOUS-BOT-IMPLEMENTED
1668            // TODO-HUMAN-REVIEW(PR-1175): pidfd_getfd, likewise untyped.
1669            SyscallClassification::Determinized if call.number() == Sysno::pidfd_getfd => {
1670                match call {
1671                    Syscall::Other(_, args) => {
1672                        self.handle_pidfd_getfd(guest, call, args.arg0 as RawFd, args.arg2 as u32)
1673                            .await
1674                    }
1675                    _ => unreachable!("pidfd_getfd unexpectedly gained a typed variant"),
1676                }
1677            }
1678            // AUTONOMOUS-BOT-IMPLEMENTED
1679            // TODO-HUMAN-REVIEW(#715): Deterministic ENOSYS for syscalls the pinned
1680            // x86_64 kernel leaves unimplemented (sys_ni_syscall). A fixed -ENOSYS is
1681            // deterministic by construction and identical to the modern kernel's own
1682            // return, so no guest-visible behavior changes versus the legacy
1683            // pass-through. These are untyped (Syscall::Other) in the pinned Reverie,
1684            // so dispatch on the Sysno before the typed match below.
1685            SyscallClassification::Determinized
1686                if is_unimplemented_enosys_syscall(call.number()) =>
1687            {
1688                Err(Error::Errno(Errno::ENOSYS))
1689            }
1690            // AUTONOMOUS-BOT-IMPLEMENTED
1691            // TODO-HUMAN-REVIEW(PR-852): Review the futex2 fallback contract.
1692            // Detcore models legacy futex but not the newer vector/sized futex2
1693            // ABI. Match a kernel without futex2 so runtimes take their
1694            // established legacy-futex fallback without consulting the host.
1695            SyscallClassification::Determinized if is_futex2_enosys_syscall(call.number()) => {
1696                Err(Error::Errno(Errno::ENOSYS))
1697            }
1698            // AUTONOMOUS-BOT-IMPLEMENTED
1699            // TODO-HUMAN-REVIEW(PR-836): Host filesystem and mount
1700            // introspection are outside the deterministic model. Return the
1701            // portable feature-absence errno so callers use /proc fallbacks.
1702            // AUTONOMOUS-BOT-IMPLEMENTED
1703            // TODO-HUMAN-REVIEW(PR-859): Extend this boundary to obsolete ustat
1704            // host-filesystem capacity counters.
1705            SyscallClassification::Determinized
1706                if is_mount_introspection_enosys_syscall(call.number()) =>
1707            {
1708                Err(Error::Errno(Errno::ENOSYS))
1709            }
1710            // AUTONOMOUS-BOT-IMPLEMENTED
1711            // TODO-HUMAN-REVIEW(PR-848): Hide unmodeled shared keyrings and
1712            // request-key upcalls behind the portable CONFIG_KEYS-absent errno.
1713            // TODO-HUMAN-REVIEW(PR-916): Only fail closed under the strict
1714            // (panic-on-unsupported) policy. A non-strict run keeps the pre-848
1715            // host pass-through so the guest observes a real working keyring;
1716            // this restores the enabled rr `keyctl` compatibility test, whose
1717            // guest asserts add_key + keyctl(SETPERM) succeed. Under strict mode
1718            // the deterministic ENOSYS boundary is preserved.
1719            SyscallClassification::Determinized if is_kernel_keyring_syscall(call.number()) => {
1720                if panic_on_unsupported_syscalls {
1721                    Err(Error::Errno(Errno::ENOSYS))
1722                } else {
1723                    self.passthrough(guest, call).await
1724                }
1725            }
1726            // AUTONOMOUS-BOT-IMPLEMENTED
1727            // TODO-HUMAN-REVIEW(PR-855): Strict runs cannot expose unmodeled
1728            // pipe-buffer ownership or vmsplice page pinning. Return ENOSYS so
1729            // callers use read/write fallbacks, but preserve the legacy normal-
1730            // mode pass-through used by the existing rr splice compatibility
1731            // test.
1732            SyscallClassification::Determinized if is_zero_copy_pipe_syscall(call.number()) => {
1733                if panic_on_unsupported_syscalls {
1734                    Err(Error::Errno(Errno::ENOSYS))
1735                } else {
1736                    self.passthrough(guest, call).await
1737                }
1738            }
1739            // AUTONOMOUS-BOT-IMPLEMENTED
1740            // TODO-HUMAN-REVIEW(PR-860): Host LSM attributes are outside
1741            // Detcore's model. Present a stable feature-absence boundary
1742            // instead of forwarding probes.
1743            SyscallClassification::Determinized
1744                if is_host_security_identity_probe_syscall(call.number()) =>
1745            {
1746                Err(Error::Errno(Errno::ENOSYS))
1747            }
1748            // AUTONOMOUS-BOT-IMPLEMENTED
1749            // TODO-HUMAN-REVIEW(#722): Deterministic EPERM for privileged
1750            // system-administration syscalls (module load/unload, kexec, reboot,
1751            // swap, raw I/O ports, root-mount pivot, host/domain name, tty
1752            // hangup, disk quotas). The deterministic guest does not hold the
1753            // capabilities these require against the host kernel, so a fixed
1754            // -EPERM matches the unprivileged errno, never perturbs global host
1755            // state, and is identical across --verify and record/replay. These
1756            // are untyped (Syscall::Other) in the pinned Reverie, so dispatch on
1757            // the Sysno before the typed match below.
1758            SyscallClassification::Determinized
1759                if is_privileged_admin_refused_syscall(call.number()) =>
1760            {

2413    async fn on_exit_thread<G: GlobalRPC<Self::GlobalState>>(
2414        &self,
2415        tid: Tid,
2416        global_state: &G,
2417        mut thread_state: Self::ThreadState,
2418        exit_status: ExitStatus,
2419    ) -> Result<(), Error> {
2420        let dettid = thread_state.dettid;
2421        debug!(
2422            "[detcore, dtid {}] thread exit hook, deregistering from scheduler.",
2423            dettid
2424        );
2425        // Close the final in-progress timeslice so this thread contributes its
2426        // last (partial) slice to the run report, even if it never exhausted a
2427        // full slice.
2428        let now = thread_state.thread_logical_time.as_nanos();
2429        thread_state.stats.close_final_timeslice(now);
2430        // Reverie invokes this callback while the backend still owns the exit
2431        // event, before the guest parent can consume it with wait. Ptrace also
2432        // guarantees that the process leader exits after the other threads, so
2433        // the final published aggregate is complete when wait returns.
2434        let detpid = thread_state.detpid.expect("Missing DetPid");
2435        if dettid == detpid {
2436            thread_state.record_exited_child_process_cpu_time(detpid);
2437        } else {
2438            thread_state.account_process_cpu_time();
2439        }
2440        let mm_id = thread_state.mm_id;
2441        let pending_chaos_epochs = thread_state.take_pending_chaos_epochs();
2442        deregister_thread(
2443            thread_state.thread_logical_time.clone(),
2444            &self.cfg,
2445            global_state,
2446            ThreadDeregistration {
2447                dettid,
2448                detpid,
2449                mm: mm_id,
2450                timeslice_stats: thread_state.stats.timeslice_stats,
2451                chaos_epochs: pending_chaos_epochs,
2452            },
2453        )
2454        .await;
2455
2456        self.record_or_replay
2457            .on_exit_thread(
2458                tid,
2459                global_state,
2460                thread_state.record_or_replay,
2461                exit_status,
2462            )
2463            .await?;
2464
2465        Ok(())
2466    }
