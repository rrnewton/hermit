9//! Which system-wide `/proc` names exist at each `--proc` level.
10//!
11//! A system-wide or `/proc/sys` name exists only if this table lists it, or
12//! lists a name below it; so does a name in a process's directory
13//! (`/proc/<pid>`, `/proc/self`) or a thread's (`/proc/<pid>/task/<tid>`,
14//! `/proc/thread-self`). Every other name is ENOENT and missing from
15//! listings, on every backend. Still not decided here, so still answered by
16//! the kernel or the KVM executor: whether a process or thread directory
17//! exists, the entries of `fd`, `fdinfo`, `ns` and `net`, what the `cwd`,
18//! `root` and `exe` links reach, and the few names Hermit's own tests and
19//! tools still need the kernel to answer ([`PROCESS_KERNEL`]). The table is the checklist in the
20//! dev-hermit design note `ai_docs/proc-virtualization-design.md` (section
21//! 5): its V and F entries, split into the two levels of its section 1b.
22

140/// The names in a process's directory below which this table decides
141/// nothing: links that leave `/proc` (`cwd`, `root`, `exe`), the directories
142/// whose entries are a process's descriptors (`fd`, `fdinfo`), and `ns` and
143/// `net`, whose entries are not decided yet (design note, slices B3a and B3b).
144const PROCESS_UNDECIDED_BELOW: &[&str] = &["cwd", "exe", "fd", "fdinfo", "net", "ns", "root"];
145
146/// The names in a process's directory that the checklist omits but that
147/// stay the kernel's for now, because something Hermit runs needs the
148/// kernel's answer and Detcore does not serve one yet:
149/// - `mem`: Detcore's guard on writes through it, and its test
150///   (`verify_claim_names_its_limit`), rr's suite and debuggers;
151/// - `uid_map`: the DBT isolation test reads it in the guest
152///   (`run_dbt_binds_in_a_user_namespace_of_its_own`);
153/// - `gid_map`, `setgroups`: kept with `uid_map`, the files a user-namespace
154///   setup writes together, until Detcore models user namespaces. Their
155///   writers in this repository run before Hermit or stop at Detcore's
156///   refusal of `unshare` before writing, so nothing intercepted needs them
157///   yet (https://github.com/rrnewton/hermit/issues/4263).
158const PROCESS_KERNEL: &[&str] = &["gid_map", "mem", "setgroups", "uid_map"];

160/// What the table says about a path under `/proc`.
161#[derive(Debug, Clone, Copy, PartialEq, Eq)]
162pub(crate) enum ProcfsName {
163    /// A per-process name, or one reached through a link the table does not
164    /// follow: not decided here.
165    Undecided,
166    /// A name that exists from this level up.
167    From(ProcLevel),
168    /// A name in no level.
169    Omitted,
170}
171
172impl ProcfsName {
173    /// Whether the name is absent at `level`.
174    pub(crate) fn absent_at(self, level: ProcLevel) -> bool {
175        match self {
176            Self::Undecided => false,
177            Self::From(from) => from > level,
178            Self::Omitted => true,
179        }
180    }
181}
182
183/// The level from which the procfs name `path` exists, for an absolute path
184/// with no `..` component; `None` for a path outside `/proc`. `path` is the
185/// spelling the guest gave, so a link elsewhere into `/proc` is not followed.
186pub(crate) fn procfs_name(path: &Path) -> Option<ProcfsName> {
187    let mut components = path.components().filter(|c| *c != Component::CurDir);
188    if components.next() != Some(Component::RootDir) {
189        return None;
190    }
191    match components.next() {
192        Some(Component::Normal(name)) if name == "proc" => {}
193        _ => return None,
194    }
195    let mut names = Vec::new();
196    for component in components {
197        match component {
198            Component::Normal(name) => names.push(name.as_bytes()),
199            // A `..` is the kernel's to resolve (see the callers).
200            _ => return Some(ProcfsName::Undecided),
201        }
202    }
203    let Some(first) = names.first() else {
204        return Some(ProcfsName::From(ProcLevel::Minimal));
205    };
206    if *first == b"self" || *first == b"thread-self" || first.iter().all(u8::is_ascii_digit) {
207        return Some(process_name(&names[1..]));
208    }
209    // Below the `net` and `mounts` links is the process's own directory.
210    if names.len() > 1 && (*first == b"net" || *first == b"mounts") {
211        return Some(ProcfsName::Undecided);
212    }
213    // The name, a directory on the way to it, or a name below a file (which
214    // the kernel refuses with ENOTDIR, as Linux does).
215    let listed = |table: &[&str]| {
216        table.iter().any(|entry| {
217            names
218                .iter()
219                .zip(entry.split('/'))
220                .all(|(name, part)| *name == part.as_bytes())
221        })
222    };
223    Some(if listed(MINIMAL) {
224        ProcfsName::From(ProcLevel::Minimal)
225    } else if listed(STANDARD) {
226        ProcfsName::From(ProcLevel::Standard)
227    } else {
228        ProcfsName::Omitted
229    })
230}

232/// What the table says about the names below a process's or a thread's
233/// directory.
234fn process_name(names: &[&[u8]]) -> ProcfsName {
235    let Some((first, rest)) = names.split_first() else {
236        // Whether the process or thread exists is the kernel's.
237        return ProcfsName::Undecided;
238    };
239    let is = |table: &[&str]| table.iter().any(|entry| entry.as_bytes() == *first);
240    if *first == b"task" {
241        return match rest.split_first() {
242            None => ProcfsName::From(ProcLevel::Minimal),
243            // A thread's directory and the names below it.
244            Some((tid, below)) if tid.iter().all(u8::is_ascii_digit) => process_name(below),
245            // `task` holds only threads.
246            Some(_) => ProcfsName::Omitted,
247        };
248    }
249    if (!rest.is_empty() && is(PROCESS_UNDECIDED_BELOW)) || is(PROCESS_KERNEL) {
250        return ProcfsName::Undecided;
251    }
252    // A name below a file stays the file's (the kernel's ENOTDIR).
253    if is(PROCESS_MINIMAL) {
254        ProcfsName::From(ProcLevel::Minimal)
255    } else if is(PROCESS_STANDARD) {
256        ProcfsName::From(ProcLevel::Standard)
257    } else {
258        ProcfsName::Omitted
259    }
260}

262/// The names the table gives the system-wide directory `/proc/<names>` at
263/// `level` (the root for no names), each with its `d_type`, in the table's
264/// order; `None` when the table holds no directory there. A name with names
265/// below it is a directory; the root's `mounts` and `net` are links into the
266/// caller's own directory.
267pub(crate) fn system_entries(names: &[&[u8]], level: ProcLevel) -> Option<Vec<(Vec<u8>, u8)>> {
268    let tables: &[&[&str]] = match level {
269        ProcLevel::Minimal => &[MINIMAL],
270        ProcLevel::Standard => &[MINIMAL, STANDARD],
271    };
272    let mut entries: Vec<(Vec<u8>, u8)> = Vec::new();
273    let mut found = names.is_empty();
274    for entry in tables.iter().flat_map(|table| table.iter()) {
275        let mut parts = entry.split('/');
276        if !names
277            .iter()
278            .all(|name| parts.next().is_some_and(|part| part.as_bytes() == *name))
279        {
280            continue;
281        }
282        let Some(child) = parts.next() else {
283            // `names` is this entry itself, a file.
284            return None;
285        };
286        found = true;
287        let ty = if parts.next().is_some() {
288            libc::DT_DIR
289        } else if names.is_empty() && (child == "mounts" || child == "net") {
290            libc::DT_LNK
291        } else {
292            libc::DT_REG
293        };
294        match entries
295            .iter_mut()
296            .find(|(name, _)| name == child.as_bytes())
297        {
298            Some(existing) if ty == libc::DT_DIR => existing.1 = ty,
299            Some(_) => {}
300            None => entries.push((child.as_bytes().to_vec(), ty)),
301        }
302    }
303    found.then_some(entries)
304}

306/// The names in a process's directory, or a thread's (`thread`), at
307/// `level`, each with its `d_type`: the level's names and the kernel's
308/// ([`PROCESS_KERNEL`]), less `task` and `timerslack_ns` in a thread's
309/// directory and `children` in a process's, as on Linux
310/// (`tgid_base_stuff`, `tid_base_stuff`).
311pub(crate) fn process_entries(thread: bool, level: ProcLevel) -> Vec<(&'static str, u8)> {
312    let mut names: Vec<&'static str> = PROCESS_MINIMAL.to_vec();
313    if level >= ProcLevel::Standard {
314        names.extend_from_slice(PROCESS_STANDARD);
315    }
316    names.extend_from_slice(PROCESS_KERNEL);
317    names
318        .into_iter()
319        .filter(|name| match *name {
320            "task" | "timerslack_ns" => !thread,
321            "children" => thread,
322            _ => true,
323        })
324        .map(|name| {
325            let ty = match name {
326                "fd" | "fdinfo" | "net" | "ns" | "task" => libc::DT_DIR,
327                "cwd" | "exe" | "root" => libc::DT_LNK,
328                _ => libc::DT_REG,
329            };
330            (name, ty)
331        })
332        .collect()
333}

335/// Whether the entry `name` of the directory at the canonical path
336/// `directory` is absent at `level`.
337pub(crate) fn absent_entry(directory: &Path, name: &[u8], level: ProcLevel) -> bool {
338    if name == b"." || name == b".." {
339        return false;
340    }
341    procfs_name(&directory.join(OsStr::from_bytes(name)))
342        .is_some_and(|entry| entry.absent_at(level))
343}

441    #[test]
442    fn each_level_contains_the_one_before() {
443        let interrupts = name("/proc/interrupts").unwrap();
444        assert!(interrupts.absent_at(ProcLevel::Minimal));
445        assert!(!interrupts.absent_at(ProcLevel::Standard));
446        let version = name("/proc/version").unwrap();
447        assert!(!version.absent_at(ProcLevel::Minimal));
448        assert!(!version.absent_at(ProcLevel::Standard));
449        let kallsyms = name("/proc/kallsyms").unwrap();
450        assert!(kallsyms.absent_at(ProcLevel::Minimal));
451        assert!(kallsyms.absent_at(ProcLevel::Standard));
452        // Every rendered or fixed file is in a level.

458                !procfs_name(path).unwrap().absent_at(ProcLevel::Standard),
459                "{path:?}"
