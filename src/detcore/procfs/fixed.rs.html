9//! System-wide procfs files whose contents are fixed (owner decision, Oct
10//! 13; <https://github.com/rrnewton/hermit/issues/4266>): the files that
11//! describe the kernel and its limits, each one value from the
12//! [`MachineDescription`], and the kernel's tables (interrupts, memory zones,
13//! modules and others), rendered for the description's CPUs and memory with
14//! every counter zero. `/proc/diskstats` is not one yet: its disks must be
15//! those `/sys/block` lists, which still come from the host. Detcore renders them with no kernel open,
16//! the same on every backend and host.
17//!
18//! Before, the host-process backends passed the kernel files through
19//! unchanged and kept each table's host shape (its interrupt names, zones
20//! and one column per host CPU) with the counters zeroed, while the
21//! KVM executor refused most of them and made up the rest.
22//!
23//! The tables follow the formats of Linux 7.1, the kernel of the reference
24//! host, where `uname` reports release 5.2.0; the few format changes between
25//! the two are not worth a second copy of each table.
26

284    pub(crate) fn from_path(path: &str) -> Option<Self> {

290    /// The permission bits Linux gives the file: read-only for the kernel's
291    /// descriptions, writable by its owner (global root) for the tunables.
292    /// Detcore refuses every write, as Linux refuses a process that is not
293    /// global root. A guest that is global root would otherwise change the
294    /// host's setting, so it is refused too: a known difference.
295    pub(crate) fn permissions(self) -> u32 {
296        match self {
297            Self::OverflowUid
298            | Self::OverflowGid
299            | Self::PidMax
300            | Self::OvercommitMemory
301            | Self::MaxMapCount
302            | Self::MmapMinAddr
303            | Self::FileMax
304            | Self::AioMaxNr
305            | Self::PipeMaxSize => 0o644,
306            _ => 0o444,
307        }
308    }

310    /// Whether the file is an integer sysctl (`proc_dointvec`,
311    /// `proc_doulongvec_minmax` and the `fs` counters built on it), whose
312    /// reads at a nonzero position Linux answers with end of file. The others
313    /// are strings (`proc_dostring`, `proc_do_uuid`) or `seq_file` files,
314    /// which serve the rest of their contents.
315    pub(crate) fn is_integer(self) -> bool {
316        !self.is_seq_file()
317            && !matches!(
318                self,
319                Self::Osrelease | Self::Ostype | Self::KernelVersion | Self::BootId
320            )
321    }
322
323    /// Whether the file is a `seq_file` (every file outside `/proc/sys`),
324    /// which seeks only from its start or position (`seq_lseek`), rather
325    /// than a sysctl file.
326    pub(crate) fn is_seq_file(self) -> bool {
327        !self.c_path().to_bytes().starts_with(b"/proc/sys/")
328    }
329
330    /// Whether Linux reads the file through `read_iter`, as every file here
331    /// does except `/proc/modules` and `/proc/swaps`, whose `seq_file`
332    /// operations have only `read`. A vectored read of one of those two goes
333    /// through `do_loop_readv_writev`, which refuses every `RWF_` flag but
334    /// `RWF_HIPRI` (measured on Linux 7.1).
335    pub(crate) fn has_read_iter(self) -> bool {
336        !matches!(self, Self::Modules | Self::Swaps)
337    }

339    /// The file's contents on `machine`.
340    pub(crate) fn contents(self, machine: &MachineDescription) -> Vec<u8> {
341        let kernel = &machine.kernel;
342        let limits = &machine.limits;
343        let cpus = machine.cpus;
344        let table = match self {
345            Self::Modules => Some(String::new()),
346            Self::Filesystems => Some(FILESYSTEMS.to_owned()),
347            Self::Interrupts => Some(interrupts(cpus)),
348            Self::Softirqs => Some(softirqs(cpus)),
349            Self::Swaps => Some("Filename\t\t\t\tType\t\tSize\t\tUsed\t\tPriority\n".to_owned()),
350            Self::Buddyinfo => Some(buddyinfo()),
351            Self::Zoneinfo => Some(zoneinfo(cpus)),
352            Self::Schedstat => Some(schedstat(cpus)),
353            Self::KeyUsers => Some(KEY_USERS.to_owned()),
354            Self::Vmstat => Some(vmstat()),
355            Self::Cgroups => Some(cgroups()),
356            Self::TtyDrivers => Some(TTY_DRIVERS.to_owned()),
357            Self::Meminfo => Some(meminfo(machine.memory)),
358            Self::Loadavg => Some("0.00 0.00 0.00 1/1 1\n".to_owned()),
359            _ => None,
360        };
361        if let Some(table) = table {
362            return table.into_bytes();
363        }
364        let text = match self {
365            Self::Version => kernel.banner(),
366            Self::Cmdline => kernel.command_line(),
367            Self::Osrelease => kernel.release.clone(),
368            Self::Ostype => kernel.sysname.clone(),
369            Self::KernelVersion => kernel.version(),
370            Self::BootId => kernel.boot_id.clone(),
371            Self::EntropyAvail => limits.entropy_avail.to_string(),
372            Self::OverflowUid => limits.overflow_uid.to_string(),
373            Self::OverflowGid => limits.overflow_gid.to_string(),
374            Self::PidMax => limits.pid_max.to_string(),
375            Self::CapLastCap => limits.cap_last_cap.to_string(),
376            Self::OvercommitMemory => limits.overcommit_memory.to_string(),
377            Self::MaxMapCount => limits.max_map_count.to_string(),
378            Self::MmapMinAddr => limits.mmap_min_addr.to_string(),
379            // Nothing is counted: no open files, inodes or dentries in use,
380            // no asynchronous I/O requests.
381            Self::FileNr => format!("0\t0\t{}", limits.file_max),
382            Self::FileMax => limits.file_max.to_string(),
383            Self::InodeNr => "0\t0".to_owned(),
384            Self::InodeState => "0\t0\t0\t0\t0\t0\t0".to_owned(),
385            // The fourth field is the age limit, 45 seconds.
386            Self::DentryState => "0\t0\t45\t0\t0\t0".to_owned(),
387            Self::AioNr => "0".to_owned(),
388            Self::AioMaxNr => limits.aio_max_nr.to_string(),
389            Self::PipeMaxSize => limits.pipe_max_size.to_string(),
390            _ => unreachable!("{self:?} is a table"),
391        };
392        format!("{text}\n").into_bytes()
393    }
394}

417const INTERRUPT_LINES: [(u32, u32, &str); 2] = [(0, 2, "timer"), (8, 8, "rtc0")];

446/// `/proc/interrupts` (`show_interrupts`) in Linux 7.1's layout, with
447/// three-digit line numbers: a count per CPU, each `" %10u"`; for a line,
448/// the chip left-aligned in 14 columns, the pin right-aligned in 15, and the
449/// trigger and action.
450fn interrupts(cpus: u32) -> String {
451    let counts = || "          0".repeat(cpus as usize);
452    let mut out = " ".repeat(11);
453    for cpu in 0..cpus {
454        out += &format!("CPU{cpu:<8}");
455    }
456    out.push('\n');
457    for (line, pin, action) in INTERRUPT_LINES {
458        out += &format!(
459            "{line:>3}:{}  {:<14} {pin:>14}-{:<8}  {action}\n",
460            counts(),
461            "IO-APIC",
462            "edge"
463        );
464    }
465    for (name, description) in ARCH_INTERRUPTS {
466        out += &format!("{name:>3}:{}  {description}\n", counts());
467    }
468    out
469}

476/// `/proc/softirqs` (`show_softirqs`).
477fn softirqs(cpus: u32) -> String {
478    let mut out = " ".repeat(20);
479    for cpu in 0..cpus {
480        out += &format!("CPU{cpu:<8}");
481    }
482    out.push('\n');
483    for name in SOFTIRQS {
484        out += &format!("{name:>12}:{}\n", "          0".repeat(cpus as usize));
485    }
486    out
487}

489/// `/proc/buddyinfo` (`frag_show_print`): node 0's populated zones, eleven
490/// orders each, none free.
491fn buddyinfo() -> String {
492    ["DMA", "DMA32", "Normal"]
493        .map(|zone| format!("Node 0, zone {zone:>8} {}\n", "     0 ".repeat(11)))
494        .concat()
495}

497/// `/proc/zoneinfo` of the reference kernel's node 0, every number zero, with
498/// a page set per CPU.
499fn zoneinfo(cpus: u32) -> String {
500    let pagesets = (0..cpus)
501        .map(|cpu| ZONEINFO_PAGESET.replace("@CPU@", &cpu.to_string()))
502        .collect::<String>();
503    ZONEINFO.replace("@PAGESETS@\n", &pagesets)
504}

513/// `/proc/schedstat` (`show_schedstat`), version 17: a line per CPU. One
514/// CPU has no scheduling domains.
515fn schedstat(cpus: u32) -> String {
516    let mut out = "version 17\ntimestamp 0\n".to_owned();
517    for cpu in 0..cpus {
518        out += &format!("cpu{cpu} 0 0 0 0 0 0 0 0 0\n");
519    }
520    out
521}

531/// The controllers `/proc/cgroups` lists, as the reference kernel does.
532const CGROUP_CONTROLLERS: [&str; 10] = [
533    "cpu",
534    "cpuacct",
535    "blkio",
536    "devices",
537    "freezer",
538    "net_cls",
539    "perf_event",
540    "net_prio",
541    "hugetlb",
542    "pids",
543];

545/// `/proc/cgroups` (`proc_cgroupstats_show`): every controller on the
546/// unified hierarchy (hierarchy 0), enabled, in one cgroup.
547fn cgroups() -> String {
548    let mut out = "#subsys_name\thierarchy\tnum_cgroups\tenabled\n".to_owned();
549    for controller in CGROUP_CONTROLLERS {
550        out += &format!("{controller}\t0\t1\t1\n");
551    }
552    out
553}

555/// `/proc/meminfo` (`meminfo_proc_show`): `memory` bytes, all free, nothing
556/// cached or swapped, each value right-aligned in 8 columns after a
557/// 16-column label, as Linux prints them.
558fn meminfo(memory: u64) -> String {
559    let kb = memory / 1024;
560    [
561        ("MemTotal:", kb),
562        ("MemFree:", kb),
563        ("MemAvailable:", kb),
564        ("Buffers:", 0),
565        ("Cached:", 0),
566        ("SwapCached:", 0),
567        ("Active:", 0),
568        ("Inactive:", 0),
569        ("Shmem:", 0),
570        ("SReclaimable:", 0),
571        ("SwapTotal:", 0),
572        ("SwapFree:", 0),
573    ]
574    .map(|(label, value)| format!("{label:<16}{value:>8} kB\n"))
575    .concat()
576}

763    /// The tables agree with each other and with the description: a column,
764    /// line or page set per CPU, and the same populated zones.
765    #[test]
766    fn the_tables_describe_the_machine_consistently() {
767        // `/proc/stat` counts each interrupt number below NR_IRQS, which
768        // holds every line `/proc/interrupts` lists.
769        for (line, _, _) in INTERRUPT_LINES {
770            assert!((line as usize) < NR_IRQS, "{line}");
771        }
772        for cpus in [1, 2, 3] {
773            let machine = MachineDescription { cpus, ..machine() };
774            let table = |file: FixedProcfsFile| String::from_utf8(file.contents(&machine)).unwrap();
775            let columns = |text: &str| text.lines().next().unwrap().split_whitespace().count();
776            assert_eq!(columns(&table(FixedProcfsFile::Interrupts)), cpus as usize);
777            assert_eq!(columns(&table(FixedProcfsFile::Softirqs)), cpus as usize);
778            for line in table(FixedProcfsFile::Interrupts).lines().skip(1) {
779                let counts = line.split_once(": ").unwrap().1;
780                assert_eq!(
781                    counts
782                        .split_whitespace()
783                        .take_while(|word| *word == "0")
784                        .count(),
785                    cpus as usize,
786                    "{line}"
787                );
788            }
789            assert_eq!(
790                table(FixedProcfsFile::Schedstat)
791                    .lines()
792                    .filter(|line| line.starts_with("cpu"))
793                    .count(),
794                cpus as usize
795            );
796            let zoneinfo = table(FixedProcfsFile::Zoneinfo);
797            let populated = zoneinfo.matches("  pagesets\n").count();
798            assert_eq!(populated, 3);
799            assert_eq!(
800                zoneinfo.matches("    cpu: ").count(),
801                populated * cpus as usize
802            );
803        }
804        // Buddyinfo lists the zones whose page sets zoneinfo shows.
805        let zoneinfo = contents(FixedProcfsFile::Zoneinfo);
806        let populated = zoneinfo
807            .split("Node 0, zone ")
808            .filter(|zone| zone.contains("  pagesets\n"))
809            .map(|zone| zone.lines().next().unwrap().trim().to_owned())
810            .collect::<Vec<_>>();
811        let buddyinfo = contents(FixedProcfsFile::Buddyinfo)
812            .lines()
813            .map(|line| line.split_whitespace().nth(3).unwrap().to_owned())
814            .collect::<Vec<_>>();
815        assert_eq!(populated, buddyinfo);
816        // The configured memory is what meminfo reports.
817        let machine = MachineDescription {
818            memory: 2 << 30,
819            ..machine()
820        };
821        let meminfo = String::from_utf8(FixedProcfsFile::Meminfo.contents(&machine)).unwrap();
822        assert!(meminfo.starts_with("MemTotal:        2097152 kB\n"));
823    }
