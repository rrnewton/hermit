9//! A process's `smaps` and `smaps_rollup` rendered from the backend's
10//! memory map (`reverie::Guest::memory_map`), where Detcore renders them
11//! (`procfs::RenderedFiles::address_space`: where the guest's procfs is not
12//! the kernel's, KVM). Each is the text Linux prints for the listed mappings
13//! (`show_smap`, `__show_smap` and `show_smaps_rollup` in
14//! `fs/proc/task_mmu.c`, v6.17), which the sanitizer that serves the
15//! kernel's file of the same name on the other backends then normalizes
16//! (`procfs::sanitize_smaps`, `sanitize_smaps_rollup`). So for every mapping
17//! it lists, the guest reads the
18//! bytes the sanitizer gives the kernel's file for that mapping on ptrace,
19//! but for `ProtectionKey`, which follows the kernel serving the process
20//! (below).
21//!
22//! A mapping's line of `maps` is its header in `smaps` ([`super::render`]'s,
23//! from the memory map's facts). Linux prints more about it than the memory
24//! map lists, and each such field is derived the same way on every backend,
25//! from facts, or the file is refused by name:
26//!
27//! - `Size` is the range's; `KernelPageSize` and `MMUPageSize` are 4 kB for
28//!   every mapping that is not hugetlbfs memory (`vma_kernel_pagesize`).
29//! - The accounting fields count the pages the host holds; each sanitizer
30//!   prints every one of them as `0 kB` (`procfs::SMAPS_ACCOUNTING_FIELDS`),
31//!   and they are rendered as those of an untouched mapping, zero.
32//! - `THPeligible` (`thp_vma_allowable_orders(vma, vma->vm_flags, TVA_SMAPS,
33//!   THP_ORDERS_ALL)`, include/linux/huge_mm.h and mm/huge_memory.c, v7.1.3):
34//!   0 for a mapping whose flags hold `VM_NOHUGEPAGE`, which `MAP_STACK`
35//!   sets (`vma_thp_disabled`). A mapping whose flags hold `VM_HUGEPAGE` is
36//!   refused: Detcore answers every `MADV_HUGEPAGE` and `MADV_NOHUGEPAGE`
37//!   itself and passes neither on (`syscalls::memory`), so only a backend's
38//!   own flags could hold it, and such a mapping's orders follow host
39//!   settings Detcore does not read. For anonymous memory (`vma_is_anonymous`:
40//!   a private anonymous mapping, the heap, the stack), 1 where the host's
41//!   transparent huge pages allow an order ([`HostThp`]), the address space
42//!   has not disabled them (`PR_SET_THP_DISABLE`, which Detcore passes to the
43//!   backend and asks it about), and the mapping holds a naturally aligned
44//!   block of that order, tried from the largest. The host's settings are
45//!   those the kernel beneath ptrace consults for the same answer, and the
46//!   ones a KVM guest reads in `/sys/kernel/mm/transparent_hugepage`, so the
47//!   value follows the host's settings exactly as ptrace's sanitized file
48//!   does. Where the host lets an anonymous mapping have huge pages without
49//!   `MADV_HUGEPAGE` (`always`), the value also follows the mapping's size and
50//!   alignment, so KVM's 8188 kB stack window, which holds an aligned 2 MiB
51//!   block, prints 1 where a native-sized stack prints 0: the same layout
52//!   difference as its range and `Size`. Any other mapping (a file's, shared
53//!   memory's) prints 0 where no order a file may have fits it, a block
54//!   aligned to its size in the address space and in the file
55//!   (`thp_vma_suitable_order`), whatever backs it. Where the backend asked
56//!   the kernel it runs on (`MappingFact::thp_eligible`: the value that
57//!   kernel prints for a mapping of its own of the same open file, with the
58//!   same protection, sharing, length and offset within a PMD-sized block,
59//!   which LinuxEmu asks for a file on overlayfs or tmpfs), that answer is
60//!   the value, but 0 where the address space disabled huge pages
61//!   (`PR_SET_THP_DISABLE`, asked of the backend as for anonymous memory:
62//!   either of its modes refuses them to a mapping `MADV_HUGEPAGE` did not
63//!   mark, `vma_thp_disabled`). The kernel beneath ptrace gives the guest's
64//!   own mapping of the file the same answer. Where the backend did not ask,
65//!   the address space's refusal still comes first, as Linux tests it before
66//!   any rule of what backs the mapping (`__thp_vma_allowable_orders`,
67//!   mm/huge_memory.c:136-137 at v7.1.3), so the value is 0 there.
68//!   Otherwise what backs it decides, after the host's `enabled` setting
69//!   ([`HostThp::enabled`]): a file on a disk filesystem (btrfs, ext2/3/4,
70//!   XFS), which the host's mount table names for the device the memory map
71//!   gives ([`HostFilesystems`]), prints 0 under `madvise` and `never`.
72//!   Under `always` it depends on the filesystem's huge page fault handler
73//!   and on whether any task holds the file open for writing
74//!   (`file_thp_enabled`), which no fact says, so the file is refused.
75//!   Shared memory follows shmem's own settings, its mount's `huge=` option
76//!   and the file's size (`shmem_allowable_huge_orders`), and overlayfs the
77//!   filesystem of a layer it does not show, so where the backend did not
78//!   ask, a file on tmpfs, on overlayfs, on any other filesystem or on a
79//!   device the mount table does not list (the kernel's internal mount that
80//!   backs shared anonymous memory and memfds) is refused by name. Detcore
81//!   asks the backend whether the address space disabled huge pages wherever
82//!   that decides a mapping's value ([`thp_disabled_decides`]).
83//! - `ProtectionKey` is printed where the kernel serving the process enables
84//!   protection keys (`arch_pkeys_enabled`, `cpu_feature_enabled(OSPKE)`): a
85//!   fact of that kernel, not of the CPU the guest is shown (ptrace's guests
86//!   are shown a virtual CPU without OSPKE, and the host kernel beneath them
87//!   prints the line). Detcore asks that kernel, through the backend
88//!   (`Detcore::serving_kernel_enables_protection_keys`). Where it enables
89//!   none, as KVM's does (its vCPU runs with CR4.PKE clear, and LinuxEmu
90//!   answers the protection-key calls as Linux does on such a CPU), the line
91//!   is not printed, as Linux prints none. Where it enables them, the file is
92//!   refused: a mapping's key then follows more calls than Detcore records
93//!   (`mprotect(PROT_EXEC)` alone gives the execute-only key,
94//!   `__arch_override_mprotect_pkey`), so a 0 could be wrong. The renderer
95//!   still prints the line with the key 0 from facts that say keys are
96//!   enabled and unused ([`SmapsFacts::protection_keys_enabled`]), as the host
97//!   kernel does for the test process ([`host_protection_keys_enabled`]).
98//! - `VmFlags`: the flags the memory map gives the mapping
99//!   ([`MappingFact::vm_flags`], which the backend establishes), as
100//!   `show_smap_vma_flags` prints them: the two-letter name of each, in the
101//!   order of its bit, each with the space after it ([`vm_flags_line`]). A
102//!   mapping's flags depend on how it was made (`MAP_GROWSDOWN`,
103//!   `MAP_NORESERVE`, the file's open mode and mount, the history of
104//!   VM_ACCOUNT), so where the memory map gives none `smaps` is refused, but
105//!   for the process's initial stack (`[stack]`), whose flags Detcore derives:
106//!   the flags `setup_arg_pages` (fs/exec.c) leaves it with,
107//!   `VM_STACK_FLAGS` with the protection the memory map lists: VM_READ,
108//!   VM_WRITE and VM_EXEC as the protection says, VM_MAYREAD, VM_MAYWRITE,
109//!   VM_MAYEXEC, VM_GROWSDOWN and VM_ACCOUNT (`rd wr mr mw me gd ac`).
110//!   `mprotect(2)` changes only the protection bits there: it keeps
111//!   VM_ACCOUNT on a mapping that has an `anon_vma` (`mprotect_fixup`), which
112//!   the stack always has, as exec wrote the arguments into it. The fork and
113//!   dump advice of `madvise(2)` sets flags (`dc`, `dd`, `wf`) no memory map
114//!   gives, and Detcore does not record which mapping such a call reached, so
115//!   a map of an address space where it, access-pattern advice or `mlock(2)`
116//!   reached the backend for a listed mapping
117//!   (`MemoryMetadata::mapping_flags_changed`) is refused; a call on a mapping
118//!   the memory map does not list changes nothing the file shows.
119//!
120//! `smaps_rollup` prints one header for the whole address space, from its
121//! first mapping's start to its last one's end (`show_smaps_rollup`, the gate
122//! area excluded, which is not in the address space's tree), and the sums of
123//! the accounting fields, which the sanitizer prints as `0 kB`. Only a
124//! complete memory map says where the address space starts: an incomplete
125//! one (KVM's where a page's mapping is not known, as for an image loaded
126//! from bytes) is refused, since a reader must not infer from a missing
127//! entry that no mapping is there (`reverie::MemoryMap::complete`).
128//!
129//! An incomplete list renders only the mappings it lists, as [`super::render`]
130//! does for `maps`: the others are missing from the file, a named difference
131//! of the backend (on KVM, <https://github.com/rrnewton/reverie/issues/1211>).
132//! An incomplete list that lists nothing is refused.

208/// What the host kernel's transparent huge page settings allow a mapping, as
209/// `/sys/kernel/mm/transparent_hugepage` shows them: the inputs of
210/// `thp_vma_allowable_orders`'s mask under the sysfs settings.
211#[derive(Clone, Copy, Debug, Eq, PartialEq)]
212pub(crate) struct HostThp {
213    /// The global setting (`enabled`), read when the settings are.
214    pub(crate) enabled: ThpEnabled,
215    /// The orders whose own setting (`hugepages-<size>kB/enabled`) is
216    /// `always` (`huge_anon_orders_always`).
217    pub(crate) always: u64,
218    /// The orders whose own setting is `inherit`
219    /// (`huge_anon_orders_inherit`): the PMD order alone where the kernel has
220    /// no per-size settings.
221    pub(crate) inherit: u64,
222}
223
224impl HostThp {
225    /// The orders an anonymous mapping no `MADV_HUGEPAGE` marked may have
226    /// (`thp_vma_allowable_orders`): those set to `always`, and those set to
227    /// `inherit` where the global setting is `always`.
228    pub(crate) fn anonymous_orders(&self) -> u64 {
229        let mut orders = self.always;
230        if self.enabled == ThpEnabled::Always {
231            orders |= self.inherit;
232        }
233        orders & THP_ORDERS_ALL_ANON
234    }
235
236    /// The settings under `root`, the kernel's
237    /// `/sys/kernel/mm/transparent_hugepage`: `Ok(None)` where it does not
238    /// exist, a kernel without transparent huge pages or on hardware that has
239    /// none (`hugepage_init` makes no directory then), where every mapping's
240    /// answer is 0. `Err` names a setting that could not be read or parsed.
241    pub(crate) fn read(root: &Path) -> Result<Option<Self>, String> {
242        let global = match std::fs::read(root.join("enabled")) {
243            Ok(text) => text,
244            Err(error) if error.kind() == std::io::ErrorKind::NotFound && !root.exists() => {
245                return Ok(None);
246            }
247            Err(error) => return Err(format!("{}/enabled: {error}", root.display())),
248        };
249        let enabled = match selected(&global) {
250            Some("always") => ThpEnabled::Always,
251            Some("madvise") => ThpEnabled::Madvise,
252            Some("never") => ThpEnabled::Never,
253            Some(_) => {
254                return Err(format!(
255                    "{}/enabled has no setting Linux gives",
256                    root.display()
257                ));
258            }
259            None => {
260                return Err(format!(
261                    "{}/enabled has no selected setting",
262                    root.display()
263                ));
264            }
265        };
266        let mut thp = HostThp {
267            enabled,
268            always: 0,
269            inherit: 0,
270        };
271        // Every size directory, `hugepages-<size>kB`, listed with raw
272        // `getdents64` (`util::find_in_directory`), in the order of size.
273        let mut sizes = Vec::new();
274        crate::util::find_in_directory(root, |name| {
275            if let Some(kilobytes) = name
276                .to_str()
277                .and_then(|name| name.strip_prefix("hugepages-"))
278                .and_then(|name| name.strip_suffix("kB"))
279                .and_then(|size| size.parse::<u64>().ok())
280            {
281                sizes.push(kilobytes);
282            }
283            None::<()>
284        })
285        .map_err(|error| format!("{}: {error}", root.display()))?;
286        sizes.sort_unstable();
287        let per_size = !sizes.is_empty();
288        for kilobytes in sizes {
289            let directory = root.join(format!("hugepages-{kilobytes}kB"));
290            let pages = kilobytes * 1024 / PAGE_SIZE;
291            if !pages.is_power_of_two() || pages < 2 {
292                return Err(format!("{}: an unexpected size", directory.display()));
293            }
294            let order = pages.trailing_zeros();
295            let path = directory.join("enabled");
296            // A size anonymous memory cannot have (order 1, which only files
297            // and shared memory use) has no `enabled`.
298            let text = match std::fs::read(&path) {
299                Ok(text) => text,
300                Err(error) if error.kind() == std::io::ErrorKind::NotFound => continue,
301                Err(error) => return Err(format!("{}: {error}", path.display())),
302            };
303            match selected(&text) {
304                Some("always") => thp.always |= 1 << order,
305                Some("inherit") => thp.inherit |= 1 << order,
306                Some("madvise" | "never") => {}
307                _ => return Err(format!("{} has no setting Linux gives", path.display())),
308            }
309        }
310        if !per_size {
311            // Before per-size settings, the PMD size followed the global
312            // setting alone (`huge_anon_orders_inherit`'s default).
313            thp.inherit = 1 << 9;
314        }
315        Ok(Some(thp))
316    }
317}

319/// The host kernel's settings ([`HostThp::read`] of
320/// `/sys/kernel/mm/transparent_hugepage`), read once per tool process, so
321/// that every `smaps` the process renders takes the same answer even if the
322/// host's administrator changes them meanwhile. Only KVM renders `smaps`
323/// (`Detcore::renders_address_space_files`), and there one tool process
324/// serves every guest process of a run, so that is every `smaps` of the run.
325/// The settings are not recorded, and each run reads them again (each run
326/// of a `--verify` pair, each run of a bisection), so two runs agree only
327/// while the host keeps its settings: they are an input from outside the
328/// run, as the host's block devices are to the `/proc/diskstats` a run
329/// renders (`tool_global::host_diskstats`, read once per run). A KVM guest
330/// can read the same settings in sysfs.
331pub(crate) fn host_thp() -> Result<Option<HostThp>, String> {
332    static HOST: std::sync::OnceLock<Result<Option<HostThp>, String>> = std::sync::OnceLock::new();
333    HOST.get_or_init(|| HostThp::read(Path::new("/sys/kernel/mm/transparent_hugepage")))
334        .clone()
335}

337/// The word a sysfs choice file selects, the one in brackets, as
338/// `always [madvise] never` selects `madvise`.
339fn selected(text: &[u8]) -> Option<&str> {
340    std::str::from_utf8(text)
341        .ok()?
342        .split_whitespace()
343        .find_map(|word| {
344            word.strip_prefix('[')
345                .and_then(|word| word.strip_suffix(']'))
346        })
347}

349/// The filesystems of the supervisor's mounts, each by the device of its
350/// superblock, as its `/proc/self/mountinfo` lists them: the `major:minor`
351/// field, the superblock's `s_dev`, and the filesystem type after the
352/// separator (`show_mountinfo`, fs/proc_namespace.c). A file's device in the
353/// memory map (`MappingBacking::File`) is that field of that table, the
354/// mount the backend opened the file through, so it names the filesystem the
355/// file is on; a mount of the kernel's own (the internal one that backs
356/// shared anonymous memory and memfds) is in no table.
357#[derive(Clone, Debug, Default, Eq, PartialEq)]
358pub(crate) struct HostFilesystems {
359    types: std::collections::BTreeMap<u64, String>,
360}
361
362impl HostFilesystems {
363    /// The filesystems `mountinfo`, the text of a `mountinfo` file, lists.
364    /// `Err` names a line that is not one Linux prints.
365    pub(crate) fn parse(mountinfo: &str) -> Result<Self, String> {
366        let mut types = std::collections::BTreeMap::new();
367        for line in mountinfo.lines() {
368            let fields: Vec<&str> = line.split(' ').collect();
369            let device = fields.get(2).and_then(|device| {
370                let (major, minor) = device.split_once(':')?;
371                Some(libc::makedev(major.parse().ok()?, minor.parse().ok()?))
372            });
373            // The optional fields end at the separator, which the type follows.
374            let filesystem = fields
375                .iter()
376                .skip(6)
377                .position(|field| *field == "-")
378                .and_then(|at| fields.get(6 + at + 1));
379            match (device, filesystem) {
380                (Some(device), Some(filesystem)) => {
381                    types.insert(device, (*filesystem).to_string());
382                }
383                _ => return Err(format!("a mountinfo line Detcore cannot read: {line}")),
384            }
385        }
386        Ok(Self { types })
387    }
388
389    /// The type of the filesystem whose superblock has device `raw_device`
390    /// (`libc::makedev`'s encoding), where a mount of it is listed.
391    fn filesystem(&self, raw_device: u64) -> Option<&str> {
392        self.types.get(&raw_device).map(String::as_str)
393    }
394}

396/// The supervisor's filesystems ([`HostFilesystems::parse`] of its
397/// `/proc/self/mountinfo`), read once per tool process, as the host's
398/// transparent huge page settings are ([`host_thp`]), so every `smaps` the
399/// process renders, every one of a KVM run, takes the same answer. A run's
400/// mounts are made before its guest starts, and the guest makes none
401/// (Detcore refuses `mount(2)` and `umount2(2)`): a device a later mount of
402/// the host's added is in no table, so a mapping of it is refused rather
403/// than misread. A device number names one superblock until that superblock
404/// is gone, and a mapped file keeps its superblock, so the table names a
405/// mapped file's filesystem unless the host itself unmounted a filesystem
406/// and mounted another while the run went on. Like the settings, the table
407/// is not recorded: another run reads its own, and gives the same answers
408/// where its mounts put the same filesystems on the devices its mappings
409/// name.
410pub(crate) fn host_filesystems() -> Result<&'static HostFilesystems, &'static str> {
411    static HOST: std::sync::OnceLock<Result<HostFilesystems, String>> = std::sync::OnceLock::new();
412    HOST.get_or_init(|| {
413        let text = std::fs::read_to_string("/proc/self/mountinfo")
414            .map_err(|error| format!("/proc/self/mountinfo: {error}"))?;
415        HostFilesystems::parse(&text)
416    })
417    .as_ref()
418    .map_err(String::as_str)
419}

421/// Whether the host kernel prints `ProtectionKey` in `smaps`
422/// (`arch_pkeys_enabled`): on x86-64, whether it enabled the CPU's
423/// protection keys, which the OSPKE bit of CPUID leaf 7 reports (it mirrors
424/// CR4.PKE), read in the test's own process, which runs on the host kernel.
425/// Only for the test that compares the renderer with that kernel's file: a
426/// guest's answer comes from the kernel serving it (the module's docs).
427/// `None` on another architecture.
428#[cfg(test)]
429pub(crate) fn host_protection_keys_enabled() -> Option<bool> {
430    #[cfg(target_arch = "x86_64")]
431    {
432        // Leaf 7 is read only where leaf 0 says the CPU has it.
433        if std::arch::x86_64::__get_cpuid_max(0).0 < 7 {
434            return Some(false);
435        }
436        Some(std::arch::x86_64::__cpuid_count(7, 0).ecx & (1 << 4) != 0)
437    }
438    #[cfg(not(target_arch = "x86_64"))]
439    {
440        None
441    }
442}

470    /// anonymous memory, the kernel's answer for a mapping
471    /// (`MappingFact::thp_eligible`) is 1, or a file's or shared memory's
472    /// mapping without that answer, which a block fits, would be refused for
473    /// what backs it; `None` where it was not asked because none holds.
474    pub(crate) thp_disabled: Option<bool>,
475}

477/// Linux's `THPeligible` for an anonymous mapping `[start, end)` that no
478/// `MADV_HUGEPAGE` or `MADV_NOHUGEPAGE` marked, which may have huge pages of
479/// `orders` (`thp_vma_allowable_orders`), in an address space that has
480/// (`disabled`) or has not disabled them: the first of those orders, from the
481/// largest, for which the mapping holds a naturally aligned block of that
482/// size ([`holds_block`]).
483fn thp_eligible(orders: u64, disabled: bool, start: u64, end: u64) -> bool {
484    if orders == 0 || disabled {
485        return false;
486    }
487    (2..=9)
488        .rev()
489        .filter(|order| orders & (1 << order) != 0)
490        .any(|order| holds_block(start, end, PAGE_SIZE << order))
491}
492
493/// Whether `thp_vma_suitable_order` (include/linux/huge_mm.h:220-238 at
494/// v7.1.3) finds a block of `size` bytes, aligned to its size, in the mapping
495/// `[start, end)`, where `__thp_vma_allowable_orders` looks for it: at `end`
496/// less the size, rounded down to the size (mm/huge_memory.c:163). Linux
497/// computes both bounds in unsigned arithmetic, which wraps, so a mapping that
498/// ends below `size` passes them: on Linux 7.1.3 a 4 KiB anonymous mapping at
499/// 0x100000 that `MADV_HUGEPAGE` marked prints `THPeligible: 1` with only the
500/// 2 MiB order allowed, where the same mapping high in the address space
501/// prints 0.
502fn holds_block(start: u64, end: u64, size: u64) -> bool {
503    let aligned = end.wrapping_sub(size) & !(size - 1);
504    aligned >= start && aligned.wrapping_add(size) <= end
505}
506
507/// Whether a mapping that is not anonymous memory, `[start, end)` from page
508/// `pgoff` of what backs it, holds a block of one of `orders` aligned to the
509/// block's size both in the address space and in the file: the filter
510/// `__thp_vma_allowable_orders` (mm/huge_memory.c:158-171 at v7.1.3) applies
511/// with `thp_vma_suitable_order`, which for such a mapping also requires the
512/// file's offset to be aligned as its address is ([`holds_block`]), from the
513/// largest order. Where none fits, Linux's answer is 0 whatever backs the
514/// mapping.
515fn file_order_fits(orders: u64, start: u64, end: u64, pgoff: u64) -> bool {
516    (1..=9)
517        .rev()
518        .filter(|order| orders & (1 << order) != 0)
519        .any(|order| {
520            let pages = 1u64 << order;
521            (start / PAGE_SIZE).wrapping_sub(pgoff) & (pages - 1) == 0
522                && holds_block(start, end, PAGE_SIZE << order)
523        })
524}

526/// Linux's `THPeligible` for `mapping` with the flags `vm_flags` (`None`:
527/// the initial stack's derived flags, which hold neither huge page flag), or
528/// the refusal that names what decides it and no fact says (the module's
529/// docs; `thp_vma_allowable_orders` and `__thp_vma_allowable_orders`,
530/// include/linux/huge_mm.h:288-313 and mm/huge_memory.c:111-219 at v7.1.3).
531/// `filesystems` is the host's mount table, or why it could not be read.
532fn thp_eligibility(
533    mapping: &MappingFact,
534    vm_flags: Option<u64>,
535    facts: &SmapsFacts,
536    filesystems: Result<&HostFilesystems, &str>,
537) -> Result<bool, Refusal> {
538    let flags = vm_flags.unwrap_or(0);
539    // No transparent huge pages on the host, or the mapping's own refusal of
540    // them (`vma_thp_disabled`).
541    let Some(enabled) = facts.thp_enabled else {
542        return Ok(false);
543    };
544    if flags & reverie::VM_NOHUGEPAGE != 0 {
545        return Ok(false);
546    }
547    if flags & reverie::VM_HUGEPAGE != 0 {
548        return Err(format!(
549            "Linux's smaps prints whether {} may have transparent huge pages, and for a \
550             mapping MADV_HUGEPAGE marked that follows host settings Detcore does not read",
551            named(mapping)
552        ));
553    }
554    let backing = match &mapping.backing {
555        MappingBacking::Anonymous
556        | MappingBacking::Stack
557        | MappingBacking::Heap
558        | MappingBacking::NamedAnonymous(_) => {
559            let thp_disabled = match (facts.thp_orders, facts.thp_disabled) {
560                (0, _) => false,
561                (_, Some(disabled)) => disabled,
562                (_, None) => {
563                    return Err(format!(
564                        "Linux's smaps prints whether {} may have transparent huge pages, and \
565                         whether the address space disabled them was not asked",
566                        named(mapping)
567                    ));
568                }
569            };
570            return Ok(thp_eligible(
571                facts.thp_orders,
572                thp_disabled,
573                mapping.start,
574                mapping.end,
575            ));
576        }
577        // The kernel's own mappings have no file and no `huge_fault`.
578        MappingBacking::Vdso | MappingBacking::Vvar | MappingBacking::Special(_) => {
579            return Ok(false);
580        }
581        MappingBacking::File { raw_device, .. } => Some(*raw_device),
582        MappingBacking::NamedSharedAnonymous { .. } => None,
583        _ => {
584            return Err(format!(
585                "Linux's smaps prints whether {} may have transparent huge pages, and Detcore \
586                 does not know what backs it",
587                named(mapping)
588            ));
589        }
590    };
591    if !file_order_fits(
592        THP_ORDERS_ALL_FILE_DEFAULT,
593        mapping.start,
594        mapping.end,
595        mapping.offset / PAGE_SIZE,
596    ) {
597        return Ok(false);
598    }
599    if let Some(eligible) = mapping.thp_eligible {
600        // The value the kernel the backend runs on printed for a mapping of
601        // its own of the same file and shape (`MappingFact::thp_eligible`),
602        // which answers for the mapping alone: the address space's own
603        // refusal of huge pages (`PR_SET_THP_DISABLE`, `vma_thp_disabled`)
604        // makes it 0, in either of that call's modes, as no `MADV_HUGEPAGE`
605        // marked the mapping (refused above).
606        return match (eligible, facts.thp_disabled) {
607            (false, _) | (true, Some(false)) => Ok(eligible),
608            (true, Some(true)) => Ok(false),
609            (true, None) => Err(format!(
610                "Linux's smaps prints whether {} may have transparent huge pages, and whether \
611                 the address space disabled them was not asked",
612                named(mapping)
613            )),
614        };
615    }
616    // Without the kernel's answer, the address space's own refusal of huge
617    // pages still comes first: Linux tests it before any rule of what backs
618    // the mapping (`__thp_vma_allowable_orders`, mm/huge_memory.c:136-137 at
619    // v7.1.3), and in either mode it refuses them to a mapping no
620    // `MADV_HUGEPAGE` marked (`vma_thp_disabled`,
621    // include/linux/huge_mm.h:331-353).
622    if facts.thp_disabled == Some(true) {
623        return Ok(false);
624    }
625    let Some(raw_device) = backing else {
626        return Err(format!(
627            "Linux's smaps prints whether {} may have transparent huge pages, which for shared \
628             anonymous memory follow shmem's settings, which Detcore does not model",
629            named(mapping)
630        ));
631    };
632    let filesystems = filesystems.map_err(|error| {
633        format!(
634            "Linux's smaps prints whether {} may have transparent huge pages, which the \
635             filesystem of the file decides, and the host's mount table could not be read: \
636             {error}",
637            named(mapping)
638        )
639    })?;
640    match filesystems.filesystem(raw_device) {
641        Some(filesystem) if DISK_FILESYSTEMS.contains(&filesystem) => match enabled {
642            ThpEnabled::Madvise | ThpEnabled::Never => Ok(false),
643            ThpEnabled::Always => Err(format!(
644                "Linux's smaps prints whether {} may have transparent huge pages, which for a \
645                 file on {filesystem} under the host's setting always depends on the \
646                 filesystem's huge page fault handler and on whether any task holds the file \
647                 open for writing, which no fact says",
648                named(mapping)
649            )),
650        },
651        Some(filesystem) => Err(format!(
652            "Linux's smaps prints whether {} may have transparent huge pages, which for a file \
653             on {filesystem} follow rules Detcore does not model (shmem's mount option and the \
654             file's size on tmpfs, the filesystem of a layer on overlayfs)",
655            named(mapping)
656        )),
657        None => Err(format!(
658            "Linux's smaps prints whether {} may have transparent huge pages, which the \
659             filesystem of the file decides, and the host's mount table lists none for its \
660             device (the kernel's internal mount backs shared anonymous memory and memfds)",
661            named(mapping)
662        )),
663    }
664}

666/// Whether the answer to whether the address space disabled transparent huge
667/// pages (`PR_GET_THP_DISABLE`) decides `mapping`'s `THPeligible`, so that
668/// Detcore asks the backend for it ([`SmapsFacts::thp_disabled`]):
669/// [`thp_eligibility`] refuses the mapping without the answer, and gives 0
670/// where the address space disabled them. `facts` are the reader's, whatever
671/// their answer, and `filesystems` the host's mount table, as
672/// [`render_smaps`] takes them.
673pub(crate) fn thp_disabled_decides(
674    mapping: &MappingFact,
675    facts: &SmapsFacts,
676    filesystems: Result<&HostFilesystems, &str>,
677) -> bool {
678    let unasked = SmapsFacts {
679        thp_disabled: None,
680        ..*facts
681    };
682    let disabled = SmapsFacts {
683        thp_disabled: Some(true),
684        ..*facts
685    };
686    thp_eligibility(mapping, mapping.vm_flags, &unasked, filesystems).is_err()
687        && thp_eligibility(mapping, mapping.vm_flags, &disabled, filesystems) == Ok(false)
688}

715/// The `VmFlags` line Linux prints for a mapping with the flags `vm_flags`
716/// (`show_smap_vma_flags`, fs/proc/task_mmu.c:1221-1229 at v7.1.3): the name
717/// of each flag, in the order of its bit, with the space after it. `None`
718/// where a flag has no name here, which no memory map gives
719/// (`reverie::MappingFact::validate`).
720fn vm_flags_line(vm_flags: u64) -> Option<String> {
721    let named = VM_FLAG_NAMES.iter().fold(0, |all, (bit, _)| all | bit);
722    if vm_flags & !named != 0 {
723        return None;
724    }
725    let mut line = String::from("VmFlags: ");
726    for (bit, name) in VM_FLAG_NAMES {
727        if vm_flags & bit != 0 {
728            line.push_str(name);
729            line.push(' ');
730        }
731    }
732    line.push('\n');
733    Some(line)
734}

736/// The `VmFlags` line Linux prints for the initial stack with protection
737/// `prot` (the module's docs): its flags in the order of their bits
738/// (`show_smap_vma_flags`), each with the space after it.
739fn stack_vm_flags(prot: u32) -> String {
740    let mut flags = String::from("VmFlags: ");
741    for (bit, mnemonic) in [
742        (libc::PROT_READ, "rd "),
743        (libc::PROT_WRITE, "wr "),
744        (libc::PROT_EXEC, "ex "),
745    ] {
746        if prot & bit as u32 != 0 {
747            flags.push_str(mnemonic);
748        }
749    }
750    // VM_MAYREAD, VM_MAYWRITE, VM_MAYEXEC, VM_GROWSDOWN and VM_ACCOUNT.
751    flags.push_str("mr mw me gd ac \n");
752    flags
753}

787/// The text of a process's `smaps` for `map`, as Linux prints it
788/// (`show_smap`), with the fields the memory map does not list derived from
789/// `facts` and the host's mount table `filesystems` (or why it could not be
790/// read), or the refusal that names why Detcore cannot give one of them.
791pub(crate) fn render_smaps(
792    map: &MemoryMap,
793    facts: &SmapsFacts,
794    filesystems: Result<&HostFilesystems, &str>,
795) -> Result<Vec<u8>, Refusal> {
796    refuse_empty_incomplete(map, "smaps")?;
797    let mut out = Vec::new();
798    for mapping in &map.mappings {
799        // The initial stack's flags Detcore derives where the memory map
800        // gives none (the module's docs).
801        let vm_flags = match (mapping.vm_flags, &mapping.backing) {
802            (Some(vm_flags), _) => Some(vm_flags),
803            (None, MappingBacking::Stack) => None,
804            (None, _) => {
805                return Err(format!(
806                    "Linux's smaps prints the flags it keeps on each mapping, and for {} \
807                     they are not in the backend's memory map",
808                    named(mapping)
809                ));
810            }
811        };
812        if facts.mapping_flags_changed {
813            return Err(format!(
814                "Linux's smaps prints the flags it keeps on each mapping, and madvise \
815                 advice or mlock that reached the backend may have set flags on {} that \
816                 the backend's memory map does not list",
817                named(mapping)
818            ));
819        }
820        if facts.protection_keys_enabled && facts.protection_keys_used {
821            return Err(format!(
822                "Linux's smaps prints each mapping's protection key, and the address space \
823                 allocated or applied one, which the backend's memory map does not list \
824                 for {}",
825                named(mapping)
826            ));
827        }
828        let eligible = thp_eligibility(mapping, vm_flags, facts, filesystems)?;
829        let flags_line = match vm_flags {
830            Some(vm_flags) => vm_flags_line(vm_flags).ok_or_else(|| {
831                format!(
832                    "the backend's memory map gives {} a flag Detcore has no name for: \
833                     {vm_flags:#x}",
834                    named(mapping)
835                )
836            })?,
837            None => stack_vm_flags(mapping.prot),
838        };
839        super::line(mapping, &mut out).ok_or_else(|| {
840            format!(
841                "the backend's memory map lists a mapping Detcore cannot print: {:08x}-{:08x}",
842                mapping.start, mapping.end
843            )
844        })?;
845        let kilobytes = (mapping.end - mapping.start) >> 10;
846        out.extend_from_slice(
847            format!(
848                "Size:           {kilobytes:>8} kB\n\
849                 KernelPageSize: {:>8} kB\n\
850                 MMUPageSize:    {:>8} kB\n",
851                PAGE_SIZE >> 10,
852                PAGE_SIZE >> 10,
853            )
854            .as_bytes(),
855        );
856        accounting(&mut out, false);
857        out.extend_from_slice(format!("THPeligible:    {:>8}\n", u8::from(eligible)).as_bytes());
858        if facts.protection_keys_enabled {
859            out.extend_from_slice(format!("ProtectionKey:  {:>8}\n", 0).as_bytes());
860        }
861        out.extend_from_slice(flags_line.as_bytes());
862    }
863    Ok(out)
864}

877/// Why Detcore cannot render a process's `smaps_rollup` from `map`, if it
878/// cannot: an incomplete list does not say where the address space starts.
879/// The one rule [`render_smaps_rollup`] refuses by, and the one a listing
880/// of the process's directory leaves the name out by
881/// (`Detcore::task_placeholder_snapshot`), so that a listing never names a
882/// `smaps_rollup` its read refuses (dh-62azc).
883pub(crate) fn smaps_rollup_refusal(map: &MemoryMap) -> Option<Refusal> {
884    (!map.complete).then(|| {
885        "Linux's smaps_rollup gives the range from the address space's first mapping to \
886         its last one, and the backend's memory map lists only some of its mappings"
887            .to_string()
888    })
889}

891/// The text of a process's `smaps_rollup` for `map`, as Linux prints it
892/// (`show_smaps_rollup`), or the refusal that names why Detcore cannot give
893/// it ([`smaps_rollup_refusal`]).
894pub(crate) fn render_smaps_rollup(map: &MemoryMap) -> Result<Vec<u8>, Refusal> {
895    if let Some(refusal) = smaps_rollup_refusal(map) {
896        return Err(refusal);
897    }
898    let mut own = map.mappings.iter().filter(|mapping| !is_gate_area(mapping));
899    let first = own.next();
900    let (start, end) = match first {
901        Some(first) => (
902            first.start,
903            own.next_back().map_or(first.end, |last| last.end),
904        ),
905        // `vma_start` and `last_vma_end` stay 0 for an empty tree.
906        None => (0, 0),
907    };
908    let header = format!("{start:08x}-{end:08x} ---p 00000000 00:00 0 ");
909    let mut out = header.clone().into_bytes();
910    // `seq_pad`: the column `show_vma_header_prefix` set, then one space.
911    out.extend(std::iter::repeat_n(
912        b' ',
913        super::NAME_COLUMN.saturating_sub(header.len()) + 1,
914    ));
915    out.extend_from_slice(b"[rollup]\n");
916    accounting(&mut out, true);
917    Ok(out)
918}
