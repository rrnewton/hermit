9//! The process and thread numbers a `/proc` path names, and the same path
10//! with those numbers translated between the guest's numbering (Detcore's)
11//! and the kernel's. They differ on DBT, whose kernel numbers the container's
12//! processes otherwise than Detcore does (slice B3a-4 of
13//! <https://github.com/rrnewton/hermit/issues/4266>): there `/proc/3`, the
14//! guest's own directory, is the kernel's `/proc/4`, and the kernel's `/proc/3`
15//! is no guest process at all.
16
17use std::ffi::OsStr;
18use std::ffi::OsString;
19use std::os::unix::ffi::OsStrExt;
20use std::path::Path;
21use std::path::PathBuf;
22
23/// A process or thread number as `/proc` spells it: decimal digits with no
24/// leading zero (Linux names nothing `03`).
25fn number(name: &[u8]) -> Option<i32> {
26    if name.first() == Some(&b'0') || !name.iter().all(u8::is_ascii_digit) {
27        return None;
28    }
29    std::str::from_utf8(name).ok()?.parse().ok()
30}

32/// Where a lookup is, walking a path's components: what the components so
33/// far name, as far as the numbers below `/proc` go.
34#[derive(Clone, Copy, PartialEq, Eq)]
35enum At {
36    Root,
37    /// `/proc` itself.
38    Proc,
39    /// A process's directory; `init` for the namespace's init (`/proc/1`).
40    Process {
41        init: bool,
42    },
43    /// A process's `task`.
44    Tasks {
45        init: bool,
46    },
47    /// A thread's directory.
48    Thread,
49    /// A directory of links (`fd`, `ns`, `map_files`).
50    Links,
51    /// Past a link that leads anywhere (`cwd`, `root`, `exe`, `fd/N`): where
52    /// the lookup is depends on where the link leads.
53    Opaque,
54    /// A directory outside `/proc`; `resolved` where it is known to be no
55    /// link (a component of a resolved directory), so that its `..` is its
56    /// lexical parent and its entries are no process's.
57    Elsewhere {
58        resolved: bool,
59    },
60    /// Anything else inside `/proc`, which has no other links: no numbers
61    /// below.
62    Other,
63}
64
65/// One number a path names: its position among the path's components (see
66/// [`components`]), the number, and whether it is a thread of the
67/// namespace's init (`/proc/1/task/<T>`), Hermit's own, which keeps its
68/// number in both numberings.
69#[derive(Clone, Copy, Debug, PartialEq, Eq)]
70struct Named {
71    position: usize,
72    id: i32,
73    init_thread: bool,
74}
75
76/// `path`'s components after the root, every `.` and `..` included, so
77/// that a path rebuilt from them resolves as the original does. Split on the
78/// bytes: `Path::components` drops a `.` that is not first, which changes
79/// how a lookup ends (`stat/.` is ENOTDIR, `cwd/.` follows the link).
80pub(crate) fn components(path: &Path) -> Vec<OsString> {
81    path.as_os_str()
82        .as_bytes()
83        .split(|byte| *byte == b'/')
84        .filter(|name| !name.is_empty())
85        .map(|name| OsString::from(OsStr::from_bytes(name)))
86        .collect()
87}

89/// A walk of an absolute path's components (see [`walk`]).
90struct Walk {
91    /// The numbers named before `unresolved`.
92    named: Vec<Named>,
93    /// The position of the first component whose meaning depends on where a
94    /// link before it leads: a number, or a `..` a number follows, past a
95    /// link, or past a directory outside `/proc` that may be one. The walk
96    /// stops there.
97    unresolved: Option<usize>,
98    /// Whether the path, walked to its end, is the namespace's init's
99    /// directory or below it.
100    init: bool,
101}
102
103/// Walk the absolute path `path`'s components as a lookup does, for the
104/// numbers it names as a process (an entry of `/proc`) or a thread (an
105/// entry of a process's `task`). `..` goes back to the directory it left,
106/// also from `self` (to `/proc`), `thread-self` (to the process's `task`)
107/// and `/proc/net` (`self/net`). The first `resolved` components are known
108/// to be no links. The walk stops where the meaning of a component depends
109/// on where a link before it leads ([`Walk::unresolved`]).
110fn walk(path: &Path, resolved: usize) -> Walk {
111    let mut named = Vec::new();
112    let mut stack = vec![At::Root];
113    if !path.is_absolute() {
114        return Walk {
115            named,
116            unresolved: None,
117            init: false,
118        };
119    }
120    let names = components(path);
121    // A `..` whose meaning a link decides matters only where a number
122    // follows: nothing else is translated.
123    let last_number = names
124        .iter()
125        .rposition(|name| number(name.as_bytes()).is_some());
126    for (position, name) in names.iter().enumerate() {
127        let top = *stack.last().expect("the walk starts at the root");
128        let name = name.as_bytes();
129        let unresolved = matches!(top, At::Opaque | At::Elsewhere { resolved: false })
130            && (number(name).is_some()
131                || (name == b".." && last_number.is_some_and(|last| last > position)));
132        if unresolved {
133            return Walk {
134                named,
135                unresolved: Some(position),
136                init: false,
137            };
138        }
139        match name {
140            b"." => continue,
141            b".." => {
142                if stack.len() > 1 {
143                    stack.pop();
144                }
145                continue;
146            }
147            _ => {}
148        }
149        let elsewhere = At::Elsewhere {
150            resolved: position < resolved,
151        };
152        match top {
153            At::Root if name == b"proc" => stack.push(At::Proc),
154            At::Root | At::Elsewhere { .. } => stack.push(elsewhere),
155            At::Proc => match name {
156                b"self" => stack.push(At::Process { init: false }),
157                b"thread-self" => stack.extend([
158                    At::Process { init: false },
159                    At::Tasks { init: false },
160                    At::Thread,
161                ]),
162                b"net" => stack.extend([At::Process { init: false }, At::Other]),
163                _ => match number(name) {
164                    Some(id) => {
165                        named.push(Named {
166                            position,
167                            id,
168                            init_thread: false,
169                        });
170                        stack.push(At::Process { init: id == 1 });
171                    }
172                    None => stack.push(At::Other),
173                },
174            },
175            At::Process { init } if name == b"task" => stack.push(At::Tasks { init }),
176            At::Process { .. } | At::Thread => match name {
177                b"fd" | b"ns" | b"map_files" => stack.push(At::Links),
178                b"cwd" | b"root" | b"exe" => stack.push(At::Opaque),
179                _ => stack.push(At::Other),
180            },
181            At::Tasks { init } => match number(name) {
182                Some(id) => {
183                    named.push(Named {
184                        position,
185                        id,
186                        init_thread: init,
187                    });
188                    stack.push(At::Thread);
189                }
190                None => stack.push(At::Other),
191            },
192            At::Links | At::Opaque => stack.push(At::Opaque),
193            At::Other => stack.push(At::Other),
194        }
195    }
196    Walk {
197        named,
198        unresolved: None,
199        init: stack.contains(&At::Process { init: true }),
200    }
201}

203/// The process and thread numbers the absolute path `path` names that the
204/// two numberings may number otherwise (see [`walk`]): every one before the
205/// walk stops but the threads of the namespace's init.
206pub(crate) fn named_ids(path: &Path) -> Vec<i32> {
207    walk(path, 0)
208        .named
209        .into_iter()
210        .filter(|named| !named.init_thread)
211        .map(|named| named.id)
212        .collect()
213}

215/// `names` as a path, absolute or relative, with the trailing slash of
216/// `like` where `trailing`.
217fn rebuilt(names: Vec<OsString>, absolute: bool, like: &Path, trailing: bool) -> PathBuf {
218    let mut rebuilt = if absolute {
219        PathBuf::from("/")
220    } else {
221        PathBuf::new()
222    };
223    rebuilt.extend(names);
224    if trailing && like.as_os_str().as_bytes().ends_with(b"/") {
225        rebuilt.as_mut_os_string().push("/");
226    }
227    rebuilt
228}
229
230/// `names` as a path like `like`: absolute where it is, with its trailing
231/// slash.
232pub(crate) fn rebuilt_like(names: Vec<OsString>, like: &Path) -> PathBuf {
233    rebuilt(names, like.is_absolute(), like, true)
234}

236/// The lookup of `rest`, absolute, or relative to the resolved directory
237/// `base` (absolute, with no link among its components), as far as the
238/// numbers `/proc` names go.
239pub(crate) struct Lookup<'a> {
240    rest: &'a Path,
241    first: usize,
242    walk: Walk,
243}
244
245impl<'a> Lookup<'a> {
246    /// Walk `rest` from `base` (see [`walk`]). An absolute `rest` ignores
247    /// `base`.
248    pub(crate) fn new(base: Option<&Path>, rest: &'a Path) -> Self {
249        let base = base.filter(|_| !rest.is_absolute());
250        let (first, walk) = match base {
251            None => (0, walk(rest, 0)),
252            Some(base) => {
253                let first = components(base).len();
254                (first, walk(&base.join(rest), first))
255            }
256        };
257        Lookup { rest, first, walk }
258    }
259
260    /// The numbers within `rest` that the two numberings may number
261    /// otherwise: every one before the walk stops but the threads of the
262    /// namespace's init. By position: a number in `base` is not `rest`'s.
263    pub(crate) fn ids(&self) -> Vec<i32> {
264        self.translatable().map(|named| named.id).collect()
265    }
266
267    fn translatable(&self) -> impl Iterator<Item = &Named> {
268        self.walk
269            .named
270            .iter()
271            .filter(|named| named.position >= self.first && !named.init_thread)
272    }
273
274    /// Each component of `rest` (by its position among [`components`] of
275    /// `rest`) that is a number [`Self::ids`] finds and `map` changes, and
276    /// what it becomes: `map` of it, or `0` where `map` has none, so that a
277    /// lookup fails as it does for a number that names no process.
278    pub(crate) fn translations(&self, map: impl Fn(i32) -> Option<i32>) -> Vec<(usize, OsString)> {
279        self.translatable()
280            .filter_map(|named| {
281                let translated = map(named.id).unwrap_or(0);
282                (translated != named.id).then(|| {
283                    (
284                        named.position - self.first,
285                        OsString::from(translated.to_string()),
286                    )
287                })
288            })
289            .collect()
290    }
291
292    /// `rest` with its [`Self::translations`]. Every other component, `.`
293    /// and `..` included, is kept, so the kernel resolves the path as it
294    /// would the guest's.
295    pub(crate) fn respelled(&self, map: impl Fn(i32) -> Option<i32>) -> PathBuf {
296        let mut names = components(self.rest);
297        for (position, name) in self.translations(map) {
298            names[position] = name;
299        }
300        rebuilt_like(names, self.rest)
301    }
302
303    /// Where the walk stops (see [`Walk::unresolved`]), by position among
304    /// [`components`] of `rest`: the directory the components before it
305    /// name decides what the rest means.
306    pub(crate) fn stop(&self) -> Option<usize> {
307        Some(self.walk.unresolved? - self.first)
308    }
309}
310
311/// The absolute path `path` with each number it names (see [`named_ids`])
312/// replaced (see [`Lookup::translations`]). `None` when nothing changes,
313/// which includes every path that names no number.
314pub(crate) fn translate(path: &Path, map: impl Fn(i32) -> Option<i32>) -> Option<PathBuf> {
315    let lookup = Lookup::new(None, path);
316    let respelled = lookup.respelled(&map);
317    (!lookup.translations(map).is_empty()).then_some(respelled)
318}

320/// Whether the absolute path `path`, walked to its end, is the namespace's
321/// init's directory or below it (`/proc/1/...`): Hermit's own process, whose
322/// numbers are the same in both numberings.
323pub(crate) fn names_init(path: &Path) -> bool {
324    walk(path, 0).init
325}
326
327/// Whether `name`, an entry of the `/proc` directory spelled `directory`
328/// (resolved), is a process or thread number: an entry of `/proc` itself or
329/// of a `task` directory.
330pub(crate) fn names_an_id(directory: &Path, name: &OsStr) -> bool {
331    !Lookup::new(Some(directory), Path::new(name))
332        .ids()
333        .is_empty()
334}

349    #[test]
350    fn named_ids_are_the_process_and_its_task() {
351        assert_eq!(named_ids(Path::new("/proc/3/stat")), [3]);
352        assert_eq!(named_ids(Path::new("/proc/3/task/5/stat")), [3, 5]);
353        assert_eq!(named_ids(Path::new("/proc/self/task/5")), [5]);
354        // A thread's directory has no `task`.
355        assert!(named_ids(Path::new("/proc/thread-self/task/5")).is_empty());
356        assert!(named_ids(Path::new("/proc/self/stat")).is_empty());
357        assert!(named_ids(Path::new("/proc/sys/kernel/pid_max")).is_empty());
358        // A leading zero names nothing; nor does a path outside /proc or a
359        // relative one.
360        assert!(named_ids(Path::new("/proc/03/stat")).is_empty());
361        assert!(named_ids(Path::new("/tmp/3")).is_empty());
362        assert!(named_ids(Path::new("proc/3")).is_empty());
363        // Below the process, only `task/<T>` names a number.
364        assert!(named_ids(Path::new("/proc/3/fd/5")).len() == 1);
365    }

411    /// The relative `rest`, looked up from the resolved `base`, respelled
412    /// where a number changes.
413    fn relative(base: &str, rest: &str) -> Option<PathBuf> {
414        let lookup = Lookup::new(Some(Path::new(base)), Path::new(rest));
415        (!lookup.translations(kernel).is_empty()).then(|| lookup.respelled(kernel))
416    }
417
418    /// Where the walk of `rest` stops: the components before the stop,
419    /// spelled with the kernel's numbers, and the rest.
420    fn unresolved(base: Option<&str>, rest: &str) -> Option<(PathBuf, PathBuf)> {
421        let lookup = Lookup::new(base.map(Path::new), Path::new(rest));
422        let at = lookup.stop()?;
423        let names = components(&lookup.respelled(kernel));
424        let directory = rebuilt(
425            names[..at].to_vec(),
426            rest.starts_with('/'),
427            Path::new(rest),
428            false,
429        );
430        let directory = match base {
431            Some(base) => Path::new(base).join(directory),
432            None => directory,
433        };
434        Some((
435            directory,
436            rebuilt(names[at..].to_vec(), false, Path::new(rest), true),
437        ))
438    }
439
440    fn pair(directory: &str, rest: &str) -> Option<(PathBuf, PathBuf)> {
441        Some((PathBuf::from(directory), PathBuf::from(rest)))
442    }

444    #[test]
445    fn dot_dot_goes_back_where_a_lookup_goes() {
446        // `..` leaves the process for /proc, so the next number is a
447        // process; each component stays, so the kernel resolves the same.
448        assert_eq!(named_ids(Path::new("/proc/3/../5/stat")), [3, 5]);
449        assert_eq!(
450            translate(Path::new("/proc/3/../5/stat"), kernel),

453        // `self` is a process of /proc, `thread-self` a thread of its task,
454        // and `/proc/net` is `self/net`.

457            Some(PathBuf::from("/proc/self/../4/stat"))

459        assert_eq!(

461            Some(PathBuf::from("/proc/thread-self/../9"))

487    #[test]
488    fn past_a_link_the_walk_stops_where_the_link_decides() {
489        // A `..` or a number past a link that leads anywhere depends on
490        // where it leads: the walk stops there, with the directory to resolve
491        // spelled with the kernel's numbers.
492        assert_eq!(named_ids(Path::new("/proc/3/cwd/../5")), [3]);
493        assert_eq!(
494            unresolved(None, "/proc/3/cwd/../5"),
495            pair("/proc/4/cwd", "../5")
496        );
497        assert_eq!(named_ids(Path::new("/proc/3/fd/7/../5")), [3]);
498        assert_eq!(named_ids(Path::new("/proc/3/task/5/cwd/../../6")), [3, 5]);
499        // A descriptor of /proc, by its link.
500        assert_eq!(
501            unresolved(None, "/proc/self/fd/4/3/exe"),
502            pair("/proc/self/fd/4", "3/exe")
503        );
504        // No number follows, so no matter where the link leads.
505        assert_eq!(unresolved(None, "/proc/3/cwd/file"), None);
506        assert_eq!(unresolved(None, "/proc/3/cwd/../file"), None);
507        assert_eq!(unresolved(None, "/usr/lib/../lib64/ld.so"), None);
508        assert_eq!(
509            translate(Path::new("/proc/3/fd/7/x"), kernel),
510            Some("/proc/4/fd/7/x".into())
511        );
512        // Outside /proc, a directory may be a link: `/tmp/U/link` may lead
513        // anywhere, so its `..` is not `/tmp/U`.
514        assert_eq!(
515            unresolved(None, "/tmp/U/link/../../../proc/3/stat"),
516            pair("/tmp/U/link", "../../../proc/3/stat")
517        );
518        assert_eq!(
519            translate(Path::new("/tmp/U/link/../../../proc/3/stat"), kernel),
520            None
521        );
522        assert_eq!(unresolved(None, "/tmp/link/3"), pair("/tmp/link", "3"));
523        // A resolved directory is no link: its `..` is its parent.
524        assert_eq!(
525            relative("/tmp", "../proc/3/stat"),
526            Some("../proc/4/stat".into())
527        );
528        assert_eq!(unresolved(Some("/tmp"), "../proc/3/stat"), None);
529        assert_eq!(
530            unresolved(Some("/tmp/U"), "link/../3"),
531            pair("/tmp/U/link", "../3")
532        );
533        // A path that names no number past a directory outside /proc.
534        assert_eq!(
535            unresolved(None, "/usr/lib/x86_64-linux-gnu/libc.so.6"),
536            None
537        );
538    }
