9//! A process's `maps` rendered from the backend's memory map
10//! (`reverie::Guest::memory_map`), where the backend gives one: each listed
11//! mapping as one line, as Linux prints it (`show_map_vma` and
12//! `show_vma_header_prefix` in `fs/proc/task_mmu.c`), in the order listed.
13//!
14//! The backend's list is exact for each mapping it lists, but it need not be
15//! complete (`reverie::MemoryMap::complete`). On ptrace it is the kernel's own
16//! text, so the rendered file is the one the kernel prints. On KVM it holds the
17//! stack window's mappings only, `[stack]` and the parts a split left of it:
18//! the image, the interpreter, the vDSO, the heap and every mapping `mmap(2)`
19//! made are missing from the rendered file, a named difference of the KVM
20//! provider (<https://github.com/rrnewton/reverie/issues/1211>). Nothing here
21//! infers from a missing line that no mapping is there, and an incomplete list
22//! that lists nothing is no answer: KVM's, once LinuxEmu has lost track of the
23//! stack window (an `mprotect(2)` across a hole in it, which LinuxEmu refuses
24//! whole where Linux changes the part before the hole, or a change a `vfork`
25//! partner made). The file is then read as it was before this rendering.
26//!
27//! One line the kernel prints is left out, and only here: `[uprobes]`, the
28//! page the kernel adds to a process the first time a uprobe fires in it
29//! (`xol_add_vma` in `kernel/events/uprobes.c`). Whether it is there depends
30//! on what other processes on the host do (a profiler installing a uprobe on a
31//! file the guest runs), not on the guest, so it can be in one run of a guest
32//! and not in the next one on the same host. Lines that depend on the host or
33//! on the backend but stay the same from run to run on one host are kept:
34//! `[vsyscall]` (the kernel's `vsyscall=` mode), `[vvar_vclock]` (the kernel's
35//! version), the sizes of `[vdso]` and `[vvar]`, and the pages a backend maps
36//! into the guest for itself, which only `super::private_maps` removes, for
37//! the backends that report them.
38//!
39//! The page is left out of the file this module renders, and nowhere else: it
40//! still shows in `smaps` and `numa_maps`, in another process's `maps`, and in
41//! a `maps` that is read from the kernel's file (by another process that
42//! inherited the descriptor, or after an `exec`). Nor is the page's effect on
43//! the rest of the map hidden: the kernel asks for it one page below the top
44//! of the address space, which on a host with 4-level paging is inside the
45//! stack, so there it takes a page a later mapping could have had (read from
46//! the source, not measured: this host has 5-level paging, where that address
47//! is free). x86's `[uprobes-trampoline]` page, which the kernel maps near a
48//! probed instruction it optimizes (`create_uprobe_trampoline` in
49//! `arch/x86/kernel/uprobes.c`), is not left out.

62/// The column `seq_pad` fills a header to before a name, with
63/// `seq_setwidth(m, 25 + sizeof(void *) * 6 - 1)` on a 64-bit kernel.
64const NAME_COLUMN: usize = 25 + 8 * 6 - 1;
65
66/// The name the kernel gives the page uprobes execute their copied
67/// instructions from (`xol_add_vma`).
68const UPROBES: &[u8] = b"[uprobes]";
69
70/// The bytes of a `maps` file listing `map`'s mappings, but for any
71/// `[uprobes]` page. `None` where `map` is an incomplete list that lists
72/// nothing once that page is left out, or where a mapping's backing is one
73/// this renderer does not know, so that the caller reads the file another way
74/// rather than print a file Linux would not.
75pub(crate) fn render(map: &MemoryMap) -> Option<Vec<u8>> {
76    let mut out = Vec::new();
77    for mapping in &map.mappings {
78        if matches!(&mapping.backing, MappingBacking::Special(name) if name == UPROBES) {
79            continue;
80        }
81        line(mapping, &mut out)?;
82    }
83    if !map.complete && out.is_empty() {
84        return None;
85    }
86    Some(out)
87}

89/// Append `mapping`'s line, newline included.
90fn line(mapping: &MappingFact, out: &mut Vec<u8>) -> Option<()> {
91    let (device, inode, name): (u64, u64, Option<Vec<u8>>) = match &mapping.backing {
92        MappingBacking::Anonymous => (0, 0, None),
93        MappingBacking::Stack => (0, 0, Some(b"[stack]".to_vec())),
94        MappingBacking::Heap => (0, 0, Some(b"[heap]".to_vec())),
95        MappingBacking::Vdso => (0, 0, Some(b"[vdso]".to_vec())),
96        MappingBacking::Vvar => (0, 0, Some(b"[vvar]".to_vec())),
97        MappingBacking::NamedAnonymous(name) => (0, 0, Some([b"[anon:", &name[..], b"]"].concat())),
98        MappingBacking::Special(name) => (0, 0, Some(name.clone())),
99        MappingBacking::File {
100            raw_device,
101            raw_inode,
102            path,
103        } => (*raw_device, *raw_inode, Some(path.clone())),
104        MappingBacking::NamedSharedAnonymous {
105            raw_device,
106            raw_inode,
107            name,
108        } => (
109            *raw_device,
110            *raw_inode,
111            Some([b"[anon_shmem:", &name[..], b"]"].concat()),
112        ),
113        _ => return None,
114    };
115    let prot = |bit: libc::c_int, set: u8| {
116        if mapping.prot & bit as u32 != 0 {
117            set
118        } else {
119            b'-'
120        }
121    };
122    let start = out.len();
123    out.extend_from_slice(
124        format!(
125            "{:08x}-{:08x} {}{}{}{} {:08x} {:02x}:{:02x} {} ",
126            mapping.start,
127            mapping.end,
128            prot(libc::PROT_READ, b'r') as char,
129            prot(libc::PROT_WRITE, b'w') as char,
130            prot(libc::PROT_EXEC, b'x') as char,
131            if mapping.shared { 's' } else { 'p' },
132            mapping.offset,
133            libc::major(device),
134            libc::minor(device),
135            inode,
136        )
137        .as_bytes(),
138    );
139    if let Some(name) = name {
140        let header = out.len() - start;
141        out.extend(std::iter::repeat_n(
142            b' ',
143            NAME_COLUMN.saturating_sub(header) + 1,
144        ));
145        out.extend_from_slice(&name);
146    }
147    out.push(b'\n');
148    Some(())
149}

151/// What a process's `maps`, opened by a task of the process its path names,
152/// is bound to where Detcore may render it (`Detcore::maps_from_memory_map`):
153/// the task its path named, which of the tasks to hold that task's numbers it
154/// was, and the address space it held at the open, the opener's.
155///
156/// Linux's open takes the task's address space (`proc_maps_open`), and each
157/// read first finds the task again by the pid its inode holds (`m_start`):
158/// ESRCH once that task is reaped, whichever task holds its number by then.
159/// Detcore gives a reaped task's number out again
160/// (`Scheduler::forget_previous_process_exit`), and an address space's
161/// identity ([`MmId`]) is a task's number and an exec count that a fork
162/// starts again at 0. So a task holding a reused number may hold an address
163/// space with the bound one's identity, and only the number's incarnation
164/// ([`TaskIncarnation`]) tells the two tasks apart.
165///
166/// That incarnation is the bound task's: [`Binding::held_by`] compares the
167/// reader's address space by its identity alone, which a reused number can
168/// repeat while the bound task lives. On ptrace, once the kernel's pid space
169/// wraps, a task that shares a reaped task's address space (a `CLONE_VM`
170/// child that is not a thread) can open its maps and then fork a child that
171/// gets the reaped task's number: that child's address space has the bound
172/// one's identity, so it reads its own map through the inherited descriptor,
173/// where Linux prints the opener's. Telling the two apart needs the
174/// incarnation of the address space's maker in [`MmId`], which also keys
175/// futexes. LinuxEmu never reuses a task's number.
176#[derive(Clone, Copy, Debug, Eq, PartialEq, Serialize, Deserialize)]
177pub(crate) struct Binding {
178    task: BoundTask,
179    incarnation: TaskIncarnation,
180    mm: MmId,
181}
182
183impl Binding {
184    /// The binding of a `maps` whose path named `task`, a task of the
185    /// opener's process, where the process table answered `answer` about that
186    /// task at the open and the opener's address space is `opener`. `None`
187    /// where the task holds no address space, having exited (Linux's file
188    /// then reads empty, as the kernel's does), or another one.
189    pub(crate) fn at_open(
190        task: BoundTask,
191        answer: &ProcfsTaskAnswer,
192        opener: MmId,
193    ) -> Option<Self> {
194        (answer.exists && !answer.without_mm && answer.mm == Some(opener)).then_some(Self {
195            task,
196            incarnation: answer.incarnation(),
197            mm: opener,
198        })
199    }
200
201    /// The task the path named, which a read asks the process table about.
202    pub(crate) fn task(&self) -> BoundTask {
203        self.task
204    }
205
206    /// Whether a reader whose address space is `reader` holds the bound one:
207    /// not another process that inherited the descriptor, nor the opener's
208    /// process after an `exec`. The reader is then one of the address space's
209    /// users, so Linux's read finds it still in use.
210    pub(crate) fn held_by(&self, reader: MmId) -> bool {
211        self.mm == reader
212    }
213
214    /// Whether `answer`, the process table's answer about the bound task at a
215    /// read, is about that very task, which Linux's read finds: it still
216    /// exists (a zombie too) and no other task has taken its numbers.
217    pub(crate) fn finds_its_task(&self, answer: &ProcfsTaskAnswer) -> bool {
218        answer.still_names(self.incarnation)
219    }
220}

385    /// The process table's answer about a task that holds the numbers
386    /// `incarnation` names and the address space `mm`, or none.
387    fn answer(incarnation: u32, mm: Option<MmId>) -> ProcfsTaskAnswer {
388        ProcfsTaskAnswer {
389            exists: true,
390            without_mm: mm.is_none(),
391            process_incarnation: incarnation,
392            thread_incarnation: None,
393            mm,
394            runtime_private_memory: None,
395        }
396    }

398    /// A `maps` binds the task its path named where that task holds the
399    /// opener's address space at the open, and only there; a read of it is the
400    /// reader's map while the reader holds that address space and the task is
401    /// still there, a zombie too.
402    #[test]
403    fn a_maps_binding_takes_the_task_and_the_address_space_of_the_open() {
404        let process = detcore_model::pid::DetTid::from_raw(5);
405        let task = BoundTask {
406            process: 5,
407            thread: None,
408        };
409        let opener = MmId::initial(process);
410        assert_eq!(Binding::at_open(task, &answer(1, None), opener), None);
411        assert_eq!(
412            Binding::at_open(task, &answer(1, Some(opener.for_exec(process))), opener),
413            None
414        );
415        let gone = ProcfsTaskAnswer::default();
416        assert_eq!(Binding::at_open(task, &gone, opener), None);
417        // Whatever address space an answer names, a task that is not there,
418        // or has released its address space, binds nothing.
419        for (exists, without_mm) in [(false, false), (true, true)] {
420            let inconsistent = ProcfsTaskAnswer {
421                exists,
422                without_mm,
423                ..answer(1, Some(opener))
424            };
425            assert_eq!(Binding::at_open(task, &inconsistent, opener), None);
426        }
427        let binding = Binding::at_open(task, &answer(1, Some(opener)), opener).unwrap();
428        assert_eq!(binding.task(), task);
429        assert!(binding.held_by(opener));
430        assert!(binding.finds_its_task(&answer(1, Some(opener))));
431        // Its leader a zombie, whose address space Linux has released, while
432        // another thread of the process still holds it.
433        assert!(binding.finds_its_task(&answer(1, None)));
434        // Reaped, with no task on its number.
435        assert!(!binding.finds_its_task(&gone));
436        // Another process that inherited the descriptor, and the process
437        // after an exec.
438        assert!(!binding.held_by(MmId::initial(detcore_model::pid::DetTid::from_raw(6))));
439        assert!(!binding.held_by(opener.for_exec(process)));
440    }

442    /// Process 5 opens its `maps`, forks process 6 and exits, and is reaped;
443    /// 6 forks a child that Detcore gives the number 5 again, which inherits
444    /// the descriptor. That child's address space has the identity of the
445    /// opener's (a fork's exec count starts again at 0), so the address
446    /// space alone would render the child's own map where Linux's read
447    /// answers ESRCH for the reaped task: the number's incarnation refuses
448    /// it.
449    #[test]
450    fn a_maps_binding_refuses_a_task_on_a_reused_number() {
451        let reused = detcore_model::pid::DetTid::from_raw(5);
452        let task = BoundTask {
453            process: 5,
454            thread: None,
455        };
456        let opener = MmId::initial(reused);
457        let binding = Binding::at_open(task, &answer(1, Some(opener)), opener).unwrap();
458        let child = MmId::for_clone(
459            MmId::initial(detcore_model::pid::DetTid::from_raw(6)),
460            reused,
461            false,
462        );
463        assert_eq!(child, opener);
464        assert!(binding.held_by(child));
465        assert!(!binding.finds_its_task(&answer(2, Some(child))));
466    }
