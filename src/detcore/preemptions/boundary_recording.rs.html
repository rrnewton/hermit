23/// Keep the preemption history and every existing run setting alongside the
24/// complete boundary stream. No public access returns its legacy component:
25/// doing so would let an editor silently discard the protocol and markers.
26#[derive(Clone, Debug, Eq, PartialEq, Serialize, Deserialize)]
27#[serde(try_from = "BoundaryWire", into = "BoundaryWire")]
28pub struct BoundaryRecording {
29    preemptions: PreemptionRecord,
30    schedule: Schedule,
31}

33#[derive(Serialize, Deserialize)]
34#[serde(deny_unknown_fields)]
35struct BoundaryWire {
36    preemptions: PreemptionRecord,
37    schedule: Schedule,
38}

40/// The wrapper is intentional. Old readers require top-level `per_thread`
41/// and `global`; they therefore refuse this format even for zero markers.
42#[derive(Serialize, Deserialize)]
43#[serde(deny_unknown_fields)]
44struct Envelope {
45    slice_boundary_recording: BoundaryRecording,
46}
47
48#[derive(Deserialize)]
49struct MetadataProbe {
50    // A present null must not look like an absent protocol marker.
51    #[serde(default, deserialize_with = "present_boundary")]
52    slice_boundary_recording: Option<serde_json::Value>,
53}
54
55fn present_boundary<'de, D: serde::Deserializer<'de>>(
56    deserializer: D,
57) -> Result<Option<serde_json::Value>, D::Error> {
58    serde_json::Value::deserialize(deserializer).map(Some)
59}
60
61pub(super) fn read_settings<T: serde::de::DeserializeOwned>(path: &Path) -> Result<T, String> {
62    let bytes = std::fs::read(path)
63        .map_err(|e| format!("cannot read preemption record {}: {}", path.display(), e))?;
64    let probe: MetadataProbe = serde_json::from_slice(&bytes)
65        .map_err(|e| format!("cannot parse preemption record {}: {}", path.display(), e))?;
66    if probe.slice_boundary_recording.is_none() {
67        // Each old metadata reader still deserializes only its own field,
68        // from the original bytes, preserving its legacy acceptance contract.
69        return serde_json::from_slice(&bytes)
70            .map_err(|e| format!("cannot parse preemption record {}: {}", path.display(), e));
71    }
72    let recording = Recording::decode(&bytes)
73        .map_err(|e| format!("cannot parse preemption record {}: {}", path.display(), e))?;
74    serde_json::from_value(serde_json::to_value(recording.metadata()).map_err(|e| e.to_string())?)
75        .map_err(|e| format!("cannot parse preemption record {}: {}", path.display(), e))
76}

78impl BoundaryRecording {
79    /// Combine a validated preemption history with the sole global record
80    /// stream. The legacy history must not contain another global stream.
81    pub fn new(preemptions: PreemptionRecord, schedule: Schedule) -> anyhow::Result<Self> {
82        preemptions.validate().map_err(anyhow::Error::msg)?;
83        if !preemptions.global.is_empty() {
84            bail!("slice-boundary recording has a second, legacy global event stream");
85        }
86        Ok(Self {
87            preemptions,
88            schedule,
89        })
90    }
91
92    /// A read-only copy of every recorded thread history. It contains no
93    /// global stream and cannot flatten or rewrite the boundary protocol.
94    pub fn thread_histories(
95        &self,
96    ) -> std::collections::BTreeMap<crate::types::DetTid, super::ThreadHistory> {
97        self.preemptions.extract_all()
98    }
99
100    /// The complete stored stream, including boundary indices.
101    pub fn schedule(&self) -> &Schedule {
102        &self.schedule
103    }
104}

123/// A recording whose format remains explicit through file transformations.
124#[derive(Clone, Debug, Eq, PartialEq)]
125pub enum Recording {
126    /// The unchanged legacy preemption/event representation.
127    Legacy(PreemptionRecord),
128    /// A versioned boundary stream and its associated preemption history.
129    Boundaries(BoundaryRecording),
130}
131
132impl Recording {
133    /// Require an explicit matching execution policy. The strict verifier must
134    /// not silently fall back to an incomplete legacy stream, and ordinary
135    /// recovery must not silently acquire strict EOF/desync semantics.
136    pub fn require_strict_mode(&self, strict: bool) -> anyhow::Result<()> {
137        match (self, strict) {
138            (Self::Legacy(_), false) | (Self::Boundaries(_), true) => Ok(()),
139            (Self::Legacy(_), true) => {
140                anyhow::bail!("strict schedule boundaries require a complete boundary recording")
141            }
142            (Self::Boundaries(_), false) => {
143                anyhow::bail!("complete boundary recording requires --strict-schedule-boundaries")
144            }
145        }
146    }
147
148    /// Validate and parse either format without a malformed-new-to-legacy
149    /// fallback.
150    pub fn decode(bytes: &[u8]) -> anyhow::Result<Self> {
151        let value: serde_json::Value = serde_json::from_slice(bytes)?;
152        // Inspect only to dispatch. Deserialize the ORIGINAL bytes again so
153        // duplicate fields in the envelope are rejected, not overwritten by
154        // Value. A malformed new recording never falls back to legacy.
155        if value.get("slice_boundary_recording").is_some() {
156            let envelope: Envelope = serde_json::from_slice(bytes)?;
157            Ok(Self::Boundaries(envelope.slice_boundary_recording))
158        } else {
159            let legacy: PreemptionRecord = serde_json::from_slice(bytes)?;
160            legacy.validate().map_err(anyhow::Error::msg)?;
161            Ok(Self::Legacy(legacy))
162        }
163    }
164
165    /// Read a complete recording, preserving errors as failures.
166    pub fn read(path: &Path) -> anyhow::Result<Self> {
167        Self::decode(&std::fs::read(path).with_context(|| format!("reading {}", path.display()))?)
168            .with_context(|| format!("reading schedule {}", path.display()))
169    }
170
171    /// Serialize the same format with the conventional terminal newline.
172    pub fn encode(&self) -> anyhow::Result<Vec<u8>> {
173        let mut bytes = match self {
174            Self::Legacy(record) => serde_json::to_vec(record)?,
175            Self::Boundaries(record) => serde_json::to_vec(&Envelope {
176                slice_boundary_recording: record.clone(),
177            })?,
178        };
179        bytes.push(b'\n');
180        Ok(bytes)
181    }
182
183    /// Write the complete serialized recording, without changing its format.
184    pub fn write(&self, path: &Path) -> anyhow::Result<()> {
185        // Complete validation/serialization precedes opening the destination.
186        let bytes = self.encode()?;
187        std::fs::write(path, bytes).with_context(|| format!("writing {}", path.display()))
188    }
189
190    /// Give a legacy-only operation its input, or refuse a boundary format
191    /// before that operation can discard its protocol or write an output.
192    pub fn into_legacy(self, operation: &'static str) -> anyhow::Result<PreemptionRecord> {
193        match self {
194            Self::Legacy(record) => Ok(record),
195            Self::Boundaries(_) => {
196                Err(detcore_model::schedule::boundary::Error::UnsupportedEdit { operation }.into())
197            }
198        }
199    }
200
201    /// Remove only optional event timestamps, retaining boundary identities,
202    /// RIP observations, preemption history and run settings.
203    pub fn strip_times(&mut self) {
204        match self {
205            Self::Legacy(record) => {
206                record.strip_event_times();
207            }
208            Self::Boundaries(record) => record.schedule.strip_times(),
209        }
210    }
211
212    /// The recorded virtual-time epoch, if the recording states one.
213    pub fn epoch(&self) -> Option<chrono::DateTime<chrono::Utc>> {
214        self.metadata().epoch()
215    }
216
217    /// The recorded pipe-wake latency, preserving `Some(0)` versus off.
218    pub fn pipe_wake_latency(&self) -> Option<u64> {
219        self.metadata().pipe_wake_latency()
220    }
221
222    /// The recorded signal-copy setting, including the legacy unknown case.
223    pub fn signal_state_copy(&self) -> Option<bool> {
224        self.metadata().signal_state_copy()
225    }
226
227    fn metadata(&self) -> &PreemptionRecord {
228        match self {
229            Self::Legacy(record) => record,
230            Self::Boundaries(record) => &record.preemptions,
231        }
232    }
233}

235/// Runtime access to recorded priorities/RCB history, retaining its complete
236/// backing format. This reader cannot export a flattened legacy recording.
237#[derive(Debug)]
238pub(crate) struct HistoryReader(Recording);
239
240impl HistoryReader {
241    pub(crate) fn read(path: &Path) -> anyhow::Result<Self> {
242        Recording::read(path).map(Self)
243    }
244
245    /// History-only replay cannot honor boundary records. Admission must not
246    /// silently project a new-format recording into legacy priorities/RCBs.
247    pub(crate) fn require_legacy_mode(&self) -> anyhow::Result<()> {
248        self.0.require_strict_mode(false)
249    }
250
251    pub(crate) fn thread_initial_priority(
252        &self,
253        tid: &crate::types::DetTid,
254    ) -> Option<crate::scheduler::runqueue::Priority> {
255        self.0
256            .metadata()
257            .per_thread
258            .get(tid)
259            .map(|history| history.initial_priority())
260    }
261
262    pub(crate) fn extract_thread_record(
263        &self,
264        tid: &crate::types::DetTid,
265    ) -> Option<super::ThreadHistory> {
266        self.0.metadata().per_thread.get(tid).cloned()
267    }
268}
269
270impl std::fmt::Display for HistoryReader {
271    fn fmt(&self, output: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
272        let bytes = self.0.encode().map_err(|_| std::fmt::Error)?;
273        output.write_str(std::str::from_utf8(&bytes).map_err(|_| std::fmt::Error)?)
274    }
275}
