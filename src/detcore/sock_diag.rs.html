8//! Determinize socket inode numbers in `NETLINK_SOCK_DIAG` dump replies.
9//!
10//! Tools such as `ss` enumerate sockets over an `AF_NETLINK`/`NETLINK_SOCK_DIAG`
11//! socket and receive a binary, multipart `nlmsghdr`-framed reply. Each
12//! `SOCK_DIAG_BY_FAMILY` message carries a family-specific body whose socket
13//! inode number (`udiag_ino` for `AF_UNIX`, `ndiag_ino` for `AF_NETLINK`) is
14//! assigned by a host-global kernel counter and therefore differs on every run.
15//! `AF_UNIX` bodies additionally carry a `UNIX_DIAG_PEER` attribute holding the
16//! peer socket's inode. Those numbers leak into guest-visible output (for
17//! example `ss -a`), breaking `--strict --verify`.
18//!
19//! detcore already zeroes the same inodes in the procfs *text* interfaces
20//! (`/proc/net/unix`, `/proc/net/netlink`; see `crate::procfs`). This module
21//! applies the identical zeroing to the *binary* `SOCK_DIAG` path so both paths
22//! agree.
23//!
24//! The parser is deliberately pure and fail-open: on any framing inconsistency
25//! it leaves the buffer untouched (mirroring the procfs sanitizers' fail-open
26//! behavior on unknown schemas). It only ever *zeroes* fixed-width fields in
27//! place; it never grows, shrinks, or shifts the buffer, so a partial parse can
28//! never corrupt message boundaries.
29

33/// `sizeof(struct nlmsghdr)`.
34const NLMSG_HDRLEN: usize = 16;
35/// Netlink message alignment (`NLMSG_ALIGNTO`).
36const NLMSG_ALIGNTO: usize = 4;
37/// Routing-attribute header length (`sizeof(struct rtattr)`).
38const RTA_HDRLEN: usize = 4;
39/// Routing-attribute alignment (`RTA_ALIGNTO`).
40const RTA_ALIGNTO: usize = 4;
41/// `nlmsg_type` for a socket-diag payload message.
42const SOCK_DIAG_BY_FAMILY: u16 = 20;
43
44/// Offset of `udiag_ino` within `struct unix_diag_msg`.
45const UNIX_DIAG_INO_OFFSET: usize = 4;
46/// `sizeof(struct unix_diag_msg)` — attributes begin immediately after.
47const UNIX_DIAG_MSG_LEN: usize = 16;
48/// `UNIX_DIAG_PEER` attribute type (payload is the peer's `__u32` inode).
49const UNIX_DIAG_PEER: u16 = 2;
50
51/// Offset of `ndiag_ino` within `struct netlink_diag_msg`.
52const NETLINK_DIAG_INO_OFFSET: usize = 16;
53
54/// Offset of `idiag_inode` within `struct inet_diag_msg`, shared by `AF_INET`
55/// and `AF_INET6`: four leading bytes, then a 48-byte `inet_diag_sockid`, then
56/// `idiag_expires`, `idiag_rqueue`, `idiag_wqueue` and `idiag_uid`.
57///
58/// Confirmed against the installed headers rather than counted by hand --
59/// `offsetof(struct inet_diag_msg, idiag_inode)` is 68 and `sizeof` is 72.
60const INET_DIAG_INO_OFFSET: usize = 68;
61
62/// Zero every socket inode number in a `NETLINK_SOCK_DIAG` multipart reply.
63///
64/// Returns `true` when `buf` was rewritten (at least one non-zero inode was
65/// zeroed) and the stream parsed consistently. Returns `false` — leaving `buf`
66/// unchanged — when the bytes do not match the expected netlink framing
67/// (fail-open) or when there were no non-zero inodes to zero.
68pub fn sanitize_sock_diag_inodes(buf: &mut [u8]) -> bool {
69    match rewritten(buf) {
70        Some(next) => {
71            buf.copy_from_slice(&next);
72            true
73        }
74        None => false,
75    }
76}

78/// Round `value` up to the next multiple of `align` (a power of two).
79fn align_up(value: usize, align: usize) -> usize {
80    (value + align - 1) & !(align - 1)
81}

83/// One exact, contiguous netlink message span (including its alignment
84/// padding), so reordering whole spans preserves the buffer's total length.
85struct Span {
86    nlmsg_type: u16,
87    nlmsg_len: usize,
88    bytes: Vec<u8>,
89}
90
91/// Parse the multipart stream, zeroing inode fields and canonicalizing the
92/// order of the diag messages. Returns `Some(copy)` when it parsed consistently
93/// and changed something, otherwise `None` (fail-open / nothing to do).
94fn rewritten(buf: &[u8]) -> Option<Vec<u8>> {
95    // Split the stream into exact contiguous spans that tile `buf`; reordering
96    // whole spans therefore preserves the total length exactly.
97    let mut spans: Vec<Span> = Vec::new();
98    let mut offset = 0usize;
99    while offset < buf.len() {
100        // A well-formed stream ends on a message boundary; a trailing remnant
101        // too small for a header is a framing error -> fail open.
102        if offset + NLMSG_HDRLEN > buf.len() {
103            return None;
104        }
105        let nlmsg_len = u32::from_ne_bytes(buf[offset..offset + 4].try_into().ok()?) as usize;
106        let nlmsg_type = u16::from_ne_bytes(buf[offset + 4..offset + 6].try_into().ok()?);
107        if nlmsg_len < NLMSG_HDRLEN || offset + nlmsg_len > buf.len() {
108            return None;
109        }
110        let advance = align_up(nlmsg_len, NLMSG_ALIGNTO);
111        if advance == 0 {
112            return None;
113        }
114        // The final message may lack trailing alignment padding.
115        let span_end = (offset + advance).min(buf.len());
116        spans.push(Span {
117            nlmsg_type,
118            nlmsg_len,
119            bytes: buf[offset..span_end].to_vec(),
120        });
121        offset += advance;
122    }
123
124    let mut modified = false;
125
126    // Zero the socket inode numbers inside every diag message.
127    for span in &mut spans {
128        if span.nlmsg_type == SOCK_DIAG_BY_FAMILY {
129            let end = span.nlmsg_len.min(span.bytes.len());
130            if NLMSG_HDRLEN < end {
131                zero_family_inodes(&mut span.bytes, NLMSG_HDRLEN, end, &mut modified);
132            }
133        }
134    }
135
136    // Canonicalize the order of the diag messages, mirroring the row-sort the
137    // procfs text sanitizers apply. Only when the diag messages form a
138    // contiguous prefix terminated by a non-diag trailer (the normal
139    // `[diag*, NLMSG_DONE]` dump shape) — this guarantees every diag message
140    // keeps its full alignment padding, so reordering cannot misalign the
141    // stream. Any other shape is left in original order (fail-open).
142    let diag_count = spans
143        .iter()
144        .filter(|s| s.nlmsg_type == SOCK_DIAG_BY_FAMILY)
145        .count();
146    let sortable = diag_count > 1
147        && diag_count < spans.len()
148        && spans[..diag_count]
149            .iter()
150            .all(|s| s.nlmsg_type == SOCK_DIAG_BY_FAMILY);
151    if sortable {
152        let mut order: Vec<usize> = (0..diag_count).collect();
153        order.sort_by(|&a, &b| spans[a].bytes.cmp(&spans[b].bytes));
154        if order.iter().enumerate().any(|(i, &j)| i != j) {
155            let reordered: Vec<Span> = order
156                .into_iter()
157                .map(|j| Span {
158                    nlmsg_type: spans[j].nlmsg_type,
159                    nlmsg_len: spans[j].nlmsg_len,
160                    bytes: spans[j].bytes.clone(),
161                })
162                .collect();
163            for (slot, span) in spans[..diag_count].iter_mut().zip(reordered) {
164                *slot = span;
165            }
166            modified = true;
167        }
168    }
169
170    if !modified {
171        return None;
172    }
173    let mut out = Vec::with_capacity(buf.len());
174    for span in &spans {
175        out.extend_from_slice(&span.bytes);
176    }
177    // Reordering whole spans must preserve the exact length; bail out otherwise.
178    if out.len() != buf.len() {
179        return None;
180    }
181    Some(out)
182}

184/// Zero the inode field(s) of one `SOCK_DIAG_BY_FAMILY` body, dispatched on the
185/// address family stored in its first byte. Bounds are always re-checked; an
186/// unknown or truncated family is left untouched.
187fn zero_family_inodes(out: &mut [u8], body: usize, end: usize, modified: &mut bool) {
188    let family = out[body] as i32;
189    match family {
190        libc::AF_UNIX => {
191            zero_u32(out, body + UNIX_DIAG_INO_OFFSET, end, modified);
192            zero_unix_peer_attrs(out, body + UNIX_DIAG_MSG_LEN, end, modified);
193        }
194        libc::AF_NETLINK => {
195            zero_u32(out, body + NETLINK_DIAG_INO_OFFSET, end, modified);
196        }
197        // `ss -t`/`ss -u` and anything else reading the socket table. Both
198        // families share `struct inet_diag_msg`, so they share the offset.
199        libc::AF_INET | libc::AF_INET6 => {
200            zero_u32(out, body + INET_DIAG_INO_OFFSET, end, modified);
201        }
202        // AF_VSOCK and AF_XDP also register socket-diag handlers on this
203        // kernel, and their bodies also carry a host-assigned inode
204        // (`vdiag_ino`, `xdiag_ino`). They are deliberately absent: no dump on
205        // this host returns a single message for either, so the offsets could
206        // not be checked against a real reply. This parser is zero-only and
207        // bounds-checked, so a wrong offset would not fault -- it would
208        // silently zero the wrong field and fail open, which is worse than an
209        // acknowledged gap. Add them alongside a populated dump, not before.
210        _ => {}
211    }
212}

214/// Walk the routing attributes trailing a `unix_diag_msg`, zeroing the peer
215/// inode carried by any `UNIX_DIAG_PEER` attribute. Stops on any inconsistency;
216/// because it only zeroes, an early stop is safe.
217fn zero_unix_peer_attrs(out: &mut [u8], mut attr: usize, end: usize, modified: &mut bool) {
218    while attr + RTA_HDRLEN <= end {
219        let rta_len = u16::from_ne_bytes([out[attr], out[attr + 1]]) as usize;
220        let rta_type = u16::from_ne_bytes([out[attr + 2], out[attr + 3]]);
221        if rta_len < RTA_HDRLEN || attr + rta_len > end {
222            break;
223        }
224        if rta_type == UNIX_DIAG_PEER && rta_len >= RTA_HDRLEN + 4 {
225            zero_u32(out, attr + RTA_HDRLEN, end, modified);
226        }
227        let advance = align_up(rta_len, RTA_ALIGNTO);
228        if advance == 0 {
229            break;
230        }
231        attr += advance;
232    }
233}

235/// Zero the `u32` at `at` when it is fully in bounds and currently non-zero.
236fn zero_u32(out: &mut [u8], at: usize, end: usize, modified: &mut bool) {
237    if at + 4 <= end && at + 4 <= out.len() && out[at..at + 4] != [0, 0, 0, 0] {
238        out[at..at + 4].copy_from_slice(&[0, 0, 0, 0]);
239        *modified = true;
240    }
241}

247    /// Append an `nlmsghdr` + body as one message, padded to `NLMSG_ALIGNTO`.
248    fn push_message(buf: &mut Vec<u8>, nlmsg_type: u16, body: &[u8]) {
249        let len = NLMSG_HDRLEN + body.len();
250        buf.extend_from_slice(&(len as u32).to_ne_bytes()); // nlmsg_len
251        buf.extend_from_slice(&nlmsg_type.to_ne_bytes()); // nlmsg_type
252        buf.extend_from_slice(&2u16.to_ne_bytes()); // nlmsg_flags (NLM_F_MULTI)
253        buf.extend_from_slice(&0u32.to_ne_bytes()); // nlmsg_seq
254        buf.extend_from_slice(&0u32.to_ne_bytes()); // nlmsg_pid
255        buf.extend_from_slice(body);
256        while !buf.len().is_multiple_of(NLMSG_ALIGNTO) {
257            buf.push(0);
258        }
259    }

261    /// A `unix_diag_msg` (16 bytes) followed by a `UNIX_DIAG_PEER` attribute.
262    fn unix_diag_body(ino: u32, peer_ino: u32) -> Vec<u8> {
263        let mut body = vec![
264            libc::AF_UNIX as u8, // udiag_family
265            0,                   // udiag_type
266            1,                   // udiag_state
267            0,                   // pad
268        ];
269        body.extend_from_slice(&ino.to_ne_bytes()); // udiag_ino
270        body.extend_from_slice(&[0u8; 8]); // udiag_cookie[2]
271        assert_eq!(body.len(), UNIX_DIAG_MSG_LEN);
272        // UNIX_DIAG_PEER attribute: rta_len, rta_type, u32 payload.
273        let rta_len = (RTA_HDRLEN + 4) as u16;
274        body.extend_from_slice(&rta_len.to_ne_bytes());
275        body.extend_from_slice(&UNIX_DIAG_PEER.to_ne_bytes());
276        body.extend_from_slice(&peer_ino.to_ne_bytes());
277        body
278    }

280    /// A `netlink_diag_msg` with `ndiag_ino` at offset 16.
281    fn netlink_diag_body(ino: u32) -> Vec<u8> {
282        let mut body = vec![
283            libc::AF_NETLINK as u8, // ndiag_family
284            0,                      // ndiag_type
285            0,                      // ndiag_protocol
286            1,                      // ndiag_state
287        ];
288        body.extend_from_slice(&0x4000_0000u32.to_ne_bytes()); // ndiag_portid
289        body.extend_from_slice(&0u32.to_ne_bytes()); // ndiag_dst_portid
290        body.extend_from_slice(&0u32.to_ne_bytes()); // ndiag_dst_group
291        body.extend_from_slice(&ino.to_ne_bytes()); // ndiag_ino
292        body.extend_from_slice(&[0u8; 8]); // ndiag_cookie[2]
293        assert_eq!(body.len(), NETLINK_DIAG_INO_OFFSET + 4 + 8);
294        body
295    }

302    fn zeroes_unix_and_netlink_inodes_and_peer() {
303        let mut buf = Vec::new();
304        push_message(
305            &mut buf,
306            SOCK_DIAG_BY_FAMILY,
307            &unix_diag_body(0x1122_3344, 0x5566_7788),
308        );
309        let unix_body = NLMSG_HDRLEN;
310        let netlink_start = buf.len();
311        push_message(
312            &mut buf,
313            SOCK_DIAG_BY_FAMILY,
314            &netlink_diag_body(0x99aa_bbcc),
315        );
316        // A trailing NLMSG_DONE (type 3) as real dumps emit.
317        push_message(&mut buf, 3, &0i32.to_ne_bytes());
318
319        // Preconditions: inodes are the non-zero host values.
320        assert_eq!(
321            read_u32(&buf, unix_body + UNIX_DIAG_INO_OFFSET),
322            0x1122_3344
323        );
324
325        assert!(sanitize_sock_diag_inodes(&mut buf));
326
327        // unix udiag_ino zeroed.
328        assert_eq!(read_u32(&buf, unix_body + UNIX_DIAG_INO_OFFSET), 0);
329        // UNIX_DIAG_PEER inode zeroed (attr payload after the 16-byte struct + 4-byte rta header).
330        assert_eq!(
331            read_u32(&buf, unix_body + UNIX_DIAG_MSG_LEN + RTA_HDRLEN),
332            0
333        );
334        // netlink ndiag_ino zeroed; portid (offset 4) preserved.
335        assert_eq!(
336            read_u32(&buf, netlink_start + NLMSG_HDRLEN + NETLINK_DIAG_INO_OFFSET),
337            0
338        );
339        assert_eq!(
340            read_u32(&buf, netlink_start + NLMSG_HDRLEN + 4),
341            0x4000_0000
342        );
343    }

345    #[test]
346    fn canonicalizes_diag_message_order() {
347        // Two netlink diag messages in "wrong" order followed by NLMSG_DONE.
348        // ndiag_ino differs so, after zeroing, they still differ by portid and
349        // must sort into a stable order regardless of arrival order.
350        let hi = netlink_diag_body(0x1111_1111); // portid 0x40000000 (fixed in helper)
351        let mut lo = netlink_diag_body(0x2222_2222);
352        // Give `lo` a smaller portid so it sorts first after inode zeroing.
353        lo[4..8].copy_from_slice(&0x1000_0000u32.to_ne_bytes());
354
355        let build = |first: &[u8], second: &[u8]| {
356            let mut b = Vec::new();
357            push_message(&mut b, SOCK_DIAG_BY_FAMILY, first);
358            push_message(&mut b, SOCK_DIAG_BY_FAMILY, second);
359            push_message(&mut b, 3, &0i32.to_ne_bytes()); // NLMSG_DONE
360            b
361        };
362
363        let mut forward = build(&lo, &hi);
364        let mut reverse = build(&hi, &lo);
365        assert!(sanitize_sock_diag_inodes(&mut forward));
366        assert!(sanitize_sock_diag_inodes(&mut reverse));
367        assert_eq!(
368            forward, reverse,
369            "diag messages must be canonicalized to the same order regardless of arrival order"
370        );
371        // Length is preserved exactly.
372        assert_eq!(forward.len(), reverse.len());
373    }
