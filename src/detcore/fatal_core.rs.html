9//! Opt-in core files for guest threads killed by a core-dumping signal.
10//!
11//! The ptrace backend offers each thread of such a process at its exit stop
12//! ([`reverie::GlobalTool::on_fatal_signal_exit`]), after the kernel's fatal
13//! decision. [`capture`] keeps a core only for the one thread that ran the
14//! kernel's core dump step (`exit.dumping`), so the registers in the core are
15//! those of the thread that took the signal. A process the kernel would not
16//! dump, for example one that is not dumpable, gets no core. It writes one ELF
17//! core, compressed as a sequence of zstd frames, into
18//! [`FatalCoreCapture::dir`]. It only reads the stopped guest, through procfs,
19//! so the guest observes nothing and its exit status is already fixed.
20//!
21//! The thread's memory is normally still mapped at the exit stop, but nothing
22//! pins it: it can be reaped before or during the capture. A page that cannot
23//! be read, or reads short, is stored as zeros and the core is kept as a
24//! partial one; when `/proc/<tid>/mem` cannot be opened at all, the core holds
25//! only the notes and the mapping layout. Neither fails the capture.
26//!
27//! A core stores only what the kernel's own dump would (see [`dump_rule`]):
28//! nothing the guest marked `MADV_DONTDUMP`, nothing of an I/O or raw-PFN
29//! mapping, and only the kinds of mapping its `coredump_filter` selects, so
30//! shared file mappings are left out by default.
31//!
32//! Every core gets a name no other core has: the caller's
33//! [`FatalCoreCapture::file_prefix`] is unique to the invocation, and a
34//! repeated guest process ID within a run gets a numbered name.
35//!
36//! The capture is best effort and bounded. One core never exceeds
37//! `max_core_bytes`; the regular files in the directory never exceed
38//! `max_total_bytes` together, counted under a directory lock that every
39//! writer takes. It never runs past `time_limit`, nor past `exit.deadline`
40//! when the run has already failed and the backend must finish cleaning up.
41//! A core that would break either cap, or that takes too long, is retried
42//! with less memory: first only the stack the thread was running on, then
43//! only the notes (registers, signal, process identity, auxiliary vector and
44//! mapped files) and the mapping layout. A core that cannot fit even then is
45//! not written. Once a Full core runs out of time, later captures in the same
46//! hermit process start at the Stack tier. The tracer waits for a capture only
47//! until its end, even when a read or write blocks in the kernel (see
48//! [`capture`]), and a capture deletes temporary files that dead writers left
49//! before it sizes the remaining total. Every failure is returned for the
50//! caller to log.

79/// Uncompressed bytes per zstd frame.
80const CHUNK: usize = 1 << 20;
81const PAGE: u64 = 4096;
82static ZEROS: [u8; CHUNK] = [0; CHUNK];

84/// What one [`capture`] did.
85#[derive(Debug, PartialEq, Eq)]
86pub(crate) enum Outcome {
87    /// A core was written to `path`.
88    Written { path: PathBuf },
89    /// The thread did not run the kernel's core dump step, so it gets no
90    /// core: another thread of its process did, or none did.
91    NotDumping,
92    /// No tier fitted in `budget` bytes before the capture's end.
93    NoRoom { budget: u64 },
94}
95
96/// How much memory a core holds. Every tier keeps only what the kernel's own
97/// dump would: see [`dump_rule`].
98#[derive(Clone, Copy, Debug, PartialEq, Eq)]
99pub(crate) enum Tier {
100    /// Every mapping the kernel would dump whole, and the first page of each
101    /// file mapped from offset 0 (its ELF header).
102    Full,
103    /// Of those, only the mapping holding the stack pointer and the main
104    /// stack, plus the ELF header pages.
105    Stack,
106    /// No memory: notes and the mapping layout only.
107    NotesOnly,
108}

110/// Captures `exit` under `cfg` and reports the outcome on stderr. Never fails
111/// the run.
112///
113/// The report goes to stderr, not to the tracing log, which a `--verify` run
114/// compares. Hermit's stderr can still be the guest's stderr, which `--verify`
115/// also compares, so the report names only the file: the core's size and tier
116/// depend on host timing and memory pressure and would make two identical runs
117/// differ.
118pub(crate) fn capture_and_log(cfg: &FatalCoreCapture, exit: &FatalSignalExit) {
119    let report = match capture(cfg, exit) {
120        Ok(Outcome::Written { path }) => format!(
121            "kept as {}",
122            path.file_name().unwrap_or_default().to_string_lossy()
123        ),
124        Ok(Outcome::NotDumping) => return,
125        Ok(Outcome::NoRoom { .. }) => {
126            "not kept: no core fits within its size and time limits".to_string()
127        }
128        Err(error) => format!("not kept: {error:#}"),
129    };
130    let line = format!(
131        "hermit: fatal core for thread {} ({}): {report}\n",
132        exit.tid, exit.signal
133    );
134    crate::nonwaiting_write::write_without_waiting(libc::STDERR_FILENO, line.as_bytes());
135}

137/// Writes a core for the thread `exit` names, held at its exit stop, if it is
138/// the thread that ran the kernel's core dump step.
139///
140/// The work runs on a helper thread that the tracer waits for only until the
141/// capture's end, plus [`HELPER_SLACK`]. The capture checks its end between
142/// frames, but one read of a file mapping on a hung network filesystem, or a
143/// write into a core directory on one, can block in the kernel past any
144/// check. The tracer then gives up, so the guest waits no longer, and the
145/// helper deletes what it was writing if it ever returns.
146pub(crate) fn capture(cfg: &FatalCoreCapture, exit: &FatalSignalExit) -> anyhow::Result<Outcome> {
147    if !exit.dumping {
148        return Ok(Outcome::NotDumping);
149    }
150    let end = capture_end(Instant::now(), cfg.time_limit, exit.deadline) + HELPER_SLACK;
151    let (cfg, exit) = (cfg.clone(), *exit);
152    run_within(end, move |abandoned| {
153        // A memory that is already gone still leaves the notes worth keeping.
154        let mem = File::open(format!("/proc/{}/mem", exit.tid.as_raw())).ok();
155        capture_abandonable(&cfg, &exit, mem.as_ref(), &FULL_TIER_TIMED_OUT, abandoned)
156    })
157}
158
159/// How long past a capture's own end the tracer still waits for it. A healthy
160/// capture overruns its end by at most one frame: one read of up to 1 MiB and
161/// its compression.
162const HELPER_SLACK: Duration = Duration::from_millis(500);
163
164/// Runs `work` on a helper thread and returns its result if it arrives by
165/// `end`. Otherwise sets the flag `work` was given and returns an error: the
166/// helper is left to finish on its own and must not keep anything once the
167/// flag is set. A panic in `work` is an error too, not a tracer panic.
168///
169/// A helper that finishes in the instant between the timeout and the flag
170/// can still keep its core while the report says it was not kept.
171fn run_within(
172    end: Instant,
173    work: impl FnOnce(&AtomicBool) -> anyhow::Result<Outcome> + Send + 'static,
174) -> anyhow::Result<Outcome> {
175    let abandoned = Arc::new(AtomicBool::new(false));
176    let (send, receive) = std::sync::mpsc::sync_channel(1);
177    let flag = Arc::clone(&abandoned);
178    std::thread::Builder::new()
179        .name("fatal-core".to_string())
180        .spawn(move || {
181            let _ = send.send(work(&flag));
182        })
183        .map_err(|error| anyhow::anyhow!("cannot start the capture thread: {error}"))?;
184    match receive.recv_timeout(end.saturating_duration_since(Instant::now())) {
185        Ok(result) => result,
186        Err(RecvTimeoutError::Timeout) => {
187            abandoned.store(true, Ordering::SeqCst);
188            match receive.try_recv() {
189                Ok(result) => result,
190                Err(_) => anyhow::bail!("gave up: a read or write blocked past the time limit"),
191            }
192        }
193        Err(RecvTimeoutError::Disconnected) => anyhow::bail!("the capture thread panicked"),
194    }
195}

197/// Set once a Full core has run out of time in this hermit process. Every
198/// capture stops the whole guest, so later crashes then go straight to the
199/// smaller tiers instead of each spending most of the time limit again.
200static FULL_TIER_TIMED_OUT: AtomicBool = AtomicBool::new(false);

202/// [`capture`] with the thread's memory, or `None` when it cannot be read at
203/// all, in which case only the notes and the mapping layout are kept.
204/// `full_timed_out` is [`FULL_TIER_TIMED_OUT`] outside tests. Tests call it
205/// directly; [`capture`] goes through [`capture_abandonable`].
206#[cfg(test)]
207fn capture_from(
208    cfg: &FatalCoreCapture,
209    exit: &FatalSignalExit,
210    mem: Option<&File>,
211    full_timed_out: &AtomicBool,
212) -> anyhow::Result<Outcome> {
213    capture_abandonable(cfg, exit, mem, full_timed_out, &AtomicBool::new(false))
214}

216/// [`capture_from`] for a helper the tracer may give up on: once `abandoned`
217/// is set, a finished core is deleted instead of kept.
218fn capture_abandonable(
219    cfg: &FatalCoreCapture,
220    exit: &FatalSignalExit,
221    mem: Option<&File>,
222    full_timed_out: &AtomicBool,
223    abandoned: &AtomicBool,
224) -> anyhow::Result<Outcome> {
225    let start = Instant::now();
226    let end = capture_end(start, cfg.time_limit, exit.deadline);
227    let tid = exit.tid.as_raw();
228    let process = Process::new(tid)?;
229    let snapshot = Snapshot::read(&process, exit)?;
230    fs::create_dir_all(&cfg.dir)?;
231    let _lock = DirLock::acquire(&cfg.dir, end)?;
232    remove_orphaned_temps(&cfg.dir, cfg.time_limit * 2)?;
233    let used = regular_file_bytes(&cfg.dir)?;
234    let budget = cfg
235        .max_core_bytes
236        .min(cfg.max_total_bytes.saturating_sub(used));
237    let stem = format!(
238        "{}core.{}.{tid}.{}",
239        cfg.file_prefix,
240        snapshot.tgid,
241        exit.signal.as_str()
242    );
243    let name = free_name(&cfg.dir, &stem)?;
244    let path = cfg.dir.join(&name);
245    let temp = cfg.dir.join(format!(".{name}.tmp"));
246    let span = end.saturating_duration_since(start);
247    let tiers: &[(Tier, u32)] = match mem {
248        Some(_) if !full_timed_out.load(Ordering::Relaxed) => {
249            &[(Tier::Full, 7), (Tier::Stack, 9), (Tier::NotesOnly, 10)]
250        }
251        Some(_) => &[(Tier::Stack, 9), (Tier::NotesOnly, 10)],
252        None => &[(Tier::NotesOnly, 10)],
253    };
254    for &(tier, tenths) in tiers {
255        let deadline = start + span * tenths / 10;
256        let segments = snapshot.segments(tier);
257        match write_core(&temp, budget, deadline, &snapshot, &segments, mem) {
258            Ok(_) => {
259                if abandoned.load(Ordering::SeqCst) {
260                    let _ = fs::remove_file(&temp);
261                    anyhow::bail!("the tracer gave up on this capture");
262                }
263                fs::rename(&temp, &path)?;
264                return Ok(Outcome::Written { path });
265            }
266            Err(error) => {
267                let _ = fs::remove_file(&temp);
268                match error {
269                    WriteError::Deadline if tier == Tier::Full => {
270                        full_timed_out.store(true, Ordering::Relaxed);
271                    }
272                    WriteError::OverBudget | WriteError::Deadline => {}
273                    WriteError::Io(error) => return Err(error.into()),
274                }
275            }
276        }
277    }
278    Ok(Outcome::NoRoom { budget })
279}

281/// When a capture started at `start` must be over: after `time_limit`, or at
282/// the backend's cleanup `deadline` if that comes first.
283fn capture_end(start: Instant, time_limit: Duration, deadline: Option<Instant>) -> Instant {
284    let limit = start + time_limit;
285    deadline.map_or(limit, |deadline| deadline.min(limit))
286}

288/// An exclusive `flock` on the core directory itself, held for the whole
289/// sizing and writing of one core, so two writers cannot both spend the same
290/// remaining total. Locking the directory leaves no lock file behind.
291struct DirLock {
292    _held: File,
293}
294
295impl DirLock {
296    fn acquire(dir: &Path, deadline: Instant) -> anyhow::Result<Self> {
297        let file = File::open(dir)?;
298        loop {
299            // SAFETY: flock on a descriptor this function owns.
300            if unsafe { libc::flock(file.as_raw_fd(), libc::LOCK_EX | libc::LOCK_NB) } == 0 {
301                return Ok(Self { _held: file });
302            }
303            let error = std::io::Error::last_os_error();
304            if error.raw_os_error() != Some(libc::EWOULDBLOCK) {
305                return Err(error.into());
306            }
307            if Instant::now() >= deadline {
308                anyhow::bail!("timed out waiting for the lock on {}", dir.display());
309            }
310            std::thread::sleep(Duration::from_millis(5));
311        }
312    }
313}

315/// `<stem>.zst`, or `<stem>.<n>.zst` for the first `n` that names no file yet,
316/// so a core never replaces another: a guest process ID can be reused within
317/// a run. Called under [`DirLock`], so two writers cannot pick the same name.
318fn free_name(dir: &Path, stem: &str) -> std::io::Result<String> {
319    let mut name = format!("{stem}.zst");
320    for n in 1u32.. {
321        if !dir.join(&name).try_exists()? {
322            break;
323        }
324        name = format!("{stem}.{n}.zst");
325    }
326    Ok(name)
327}

329/// Removes the temporary files of writers that died mid-core, so they stop
330/// counting against `max_total_bytes`. Every writer creates its temp only while
331/// it holds [`DirLock`], so one found while the caller holds the lock has no
332/// live writer on this host. Only hermit's own core temp names
333/// (`.hermit-*core.*.zst.tmp`) are touched, and only ones last written at
334/// least `min_age` ago, which also covers a writer on another host whose lock
335/// this host cannot see on some network filesystems. A temp that cannot be
336/// removed, such as another user's in a sticky directory, is skipped: it keeps
337/// counting, but it never stops this capture.
338fn remove_orphaned_temps(dir: &Path, min_age: Duration) -> std::io::Result<()> {
339    let now = SystemTime::now();
340    for entry in fs::read_dir(dir)? {
341        let Ok(entry) = entry else {
342            continue;
343        };
344        let name = entry.file_name();
345        let name = name.to_string_lossy();
346        if !(name.starts_with(".hermit-") && name.contains("core.") && name.ends_with(".zst.tmp")) {
347            continue;
348        }
349        let Ok(metadata) = entry.metadata() else {
350            continue;
351        };
352        let old = metadata
353            .modified()
354            .ok()
355            .and_then(|modified| now.duration_since(modified).ok())
356            .is_some_and(|age| age >= min_age);
357        if metadata.is_file() && old {
358            let _ = fs::remove_file(entry.path());
359        }
360    }
361    Ok(())
362}

364/// Bytes held by the regular files directly in `dir`.
365fn regular_file_bytes(dir: &Path) -> std::io::Result<u64> {
366    let mut total = 0u64;
367    for entry in fs::read_dir(dir)? {
368        let metadata = entry?.metadata()?;
369        if metadata.file_type().is_file() {
370            total = total.saturating_add(metadata.len());
371        }
372    }
373    Ok(total)
374}

376/// Everything a core needs except memory contents, read before writing.
377struct Snapshot {
378    tgid: i32,
379    stack_pointer: u64,
380    notes: Vec<u8>,
381    maps: Vec<MemoryMap>,
382    /// What may be stored of each of `maps`, by [`dump_rule`].
383    rules: Vec<Dump>,
384}
385
386impl Snapshot {
387    fn read(process: &Process, exit: &FatalSignalExit) -> anyhow::Result<Self> {
388        let status = process.status()?;
389        let stat = process.stat()?;
390        // smaps carries the kernel's VmFlags, which say what a dump must skip;
391        // without them, keep the layout and store no memory at all.
392        let (maps, flags_known) = match process.smaps() {
393            Ok(maps) => (maps.0, true),
394            Err(_) => (process.maps()?.0, false),
395        };
396        let filter = coredump_filter(exit.tid.as_raw());
397        let rules = maps
398            .iter()
399            .map(|map| {
400                let flags = map.extension.vm_flags;
401                dump_rule(
402                    map,
403                    (flags_known && !flags.is_empty()).then_some(flags),
404                    filter,
405                )
406            })
407            .collect();
408        let cmdline = process.cmdline().unwrap_or_default().join(" ");
409        let auxv = fs::read(format!("/proc/{}/auxv", exit.tid.as_raw())).ok();
410        let identity = Identity {
411            tid: exit.tid.as_raw(),
412            tgid: status.tgid,
413            ppid: stat.ppid,
414            pgrp: stat.pgrp,
415            sid: stat.session,
416            uid: status.ruid,
417            gid: status.rgid,
418            sigpend: status.sigpnd | status.shdpnd,
419            sighold: status.sigblk,
420            state: stat.state,
421            nice: stat.nice,
422            flags: stat.flags,
423            comm: stat.comm,
424            cmdline,
425        };
426        let mut notes = Vec::new();
427        note(
428            &mut notes,
429            NT_PRSTATUS,
430            &arch::prstatus(exit.signal as i32, &identity, &exit.regs),
431        );
432        note(&mut notes, NT_PRPSINFO, &prpsinfo(&identity));
433        if let Some(auxv) = auxv {
434            note(&mut notes, NT_AUXV, &auxv);
435        }
436        note(&mut notes, NT_FILE, &nt_file(&maps));
437        Ok(Self {
438            tgid: status.tgid,
439            stack_pointer: arch::stack_pointer(&exit.regs),
440            notes,
441            maps,
442            rules,
443        })
444    }
445
446    fn segments(&self, tier: Tier) -> Vec<Segment> {
447        self.maps
448            .iter()
449            .zip(&self.rules)
450            .filter(|(map, _)| dumpable(map))
451            .map(|(map, rule)| {
452                let (start, end) = map.address;
453                let len = end - start;
454                let all = rule.whole
455                    && match tier {
456                        Tier::Full => true,
457                        Tier::Stack => {
458                            (start..end).contains(&self.stack_pointer)
459                                || matches!(map.pathname, MMapPath::Stack)
460                        }
461                        Tier::NotesOnly => false,
462                    };
463                let elf_header = tier != Tier::NotesOnly && rule.elf_header;
464                let filesz = if all {
465                    len
466                } else if elf_header {
467                    len.min(PAGE)
468                } else {
469                    0
470                };
471                Segment {
472                    vaddr: start,
473                    memsz: len,
474                    filesz,
475                    flags: segment_flags(map.perms),
476                }
477            })
478            .collect()
479    }
480}

482/// What a core may store of one mapping.
483#[derive(Clone, Copy, Debug, PartialEq, Eq)]
484struct Dump {
485    /// All of it.
486    whole: bool,
487    /// Its first page, the ELF header of a file mapped from offset 0.
488    elf_header: bool,
489}
490
491/// What the kernel's own dump (`vma_dump_size` in fs/coredump.c) would keep of
492/// `map`, given its VmFlags (`None` when unknown, which keeps nothing) and the
493/// process's `coredump_filter`:
494///
495/// - nothing of a mapping the guest excluded with `MADV_DONTDUMP` (`dd`), or
496///   of an I/O or raw-PFN mapping (`io`, `pf`), whose read could touch a
497///   device;
498/// - hugetlb and shared mappings only if the filter bit for their kind is set
499///   (shared file mappings are off by default);
500/// - anonymous memory, and a writable private file mapping, whose written
501///   pages the kernel holds as anonymous, under the anonymous-private bit; a
502///   read-only private file mapping under the file-private bit (off by
503///   default);
504/// - the first page of a file mapped from offset 0 under the ELF-headers bit.
505fn dump_rule(map: &MemoryMap, flags: Option<VmFlags>, filter: CoredumpFlags) -> Dump {
506    const NOTHING: Dump = Dump {
507        whole: false,
508        elf_header: false,
509    };
510    let Some(flags) = flags else {
511        return NOTHING;
512    };
513    if flags.intersects(VmFlags::DD | VmFlags::IO | VmFlags::PF) {
514        return NOTHING;
515    }
516    let file = matches!(map.pathname, MMapPath::Path(_));
517    let shared = flags.contains(VmFlags::SH);
518    let whole = if flags.contains(VmFlags::HT) {
519        filter.contains(if shared {
520            CoredumpFlags::SHARED_HUGEPAGES
521        } else {
522            CoredumpFlags::PROVATE_HUGEPAGES
523        })
524    } else if shared {
525        filter.contains(if file {
526            CoredumpFlags::FILEBACKED_SHARED_MAPPINGS
527        } else {
528            CoredumpFlags::ANONYMOUS_SHARED_MAPPINGS
529        })
530    } else if !file || map.perms.contains(MMPermissions::WRITE) {
531        filter.contains(CoredumpFlags::ANONYMOUS_PRIVATE_MAPPINGS)
532            || (file && filter.contains(CoredumpFlags::FILEBACKED_PRIVATE_MAPPINGS))
533    } else {
534        filter.contains(CoredumpFlags::FILEBACKED_PRIVATE_MAPPINGS)
535    };
536    Dump {
537        whole,
538        elf_header: file && map.offset == 0 && filter.contains(CoredumpFlags::ELF_HEADERS),
539    }
540}

542/// The thread's `/proc/<tid>/coredump_filter`, or the kernel default (0x33:
543/// anonymous private and shared, ELF headers, private hugetlb) when it cannot
544/// be read. Unknown bits are ignored.
545fn coredump_filter(tid: i32) -> CoredumpFlags {
546    fs::read_to_string(format!("/proc/{tid}/coredump_filter"))
547        .ok()
548        .and_then(|text| u32::from_str_radix(text.trim(), 16).ok())
549        .map_or(
550            CoredumpFlags::from_bits_truncate(0x33),
551            CoredumpFlags::from_bits_truncate,
552        )
553}

555/// Whether a mapping can appear in a core: procfs cannot read the vsyscall
556/// page or the vvar pages, and an unreadable mapping has nothing to read.
557fn dumpable(map: &MemoryMap) -> bool {
558    map.perms.contains(MMPermissions::READ)
559        && !matches!(map.pathname, MMapPath::Vvar | MMapPath::Vsyscall)
560        && !matches!(&map.pathname, MMapPath::Other(name) if name.starts_with("vvar"))
561}
562
563fn segment_flags(perms: MMPermissions) -> u32 {
564    let mut flags = 0;
565    if perms.contains(MMPermissions::EXECUTE) {
566        flags |= PF_X;
567    }
568    if perms.contains(MMPermissions::WRITE) {
569        flags |= PF_W;
570    }
571    if perms.contains(MMPermissions::READ) {
572        flags |= PF_R;
573    }
574    flags
575}

577/// One `PT_LOAD`: `filesz` bytes from `vaddr` are stored, the rest of
578/// `memsz` is described but not stored.
579#[derive(Clone, Copy, Debug, PartialEq, Eq)]
580struct Segment {
581    vaddr: u64,
582    memsz: u64,
583    filesz: u64,
584    flags: u32,
585}
586
587const PT_LOAD: u32 = 1;
588const PT_NOTE: u32 = 4;
589const PF_X: u32 = 1;
590const PF_W: u32 = 2;
591const PF_R: u32 = 4;
592const NT_PRSTATUS: u32 = 1;

602    tgid: i32,

617fn note(out: &mut Vec<u8>, kind: u32, desc: &[u8]) {
618    const NAME: &[u8] = b"CORE\0";
619    out.extend_from_slice(&(NAME.len() as u32).to_le_bytes());
620    out.extend_from_slice(&(desc.len() as u32).to_le_bytes());
621    out.extend_from_slice(&kind.to_le_bytes());
622    out.extend_from_slice(NAME);
623    pad4(out);
624    out.extend_from_slice(desc);
625    pad4(out);
626}
627
628fn pad4(out: &mut Vec<u8>) {
629    out.resize(out.len().next_multiple_of(4), 0);
630}

632fn put(out: &mut [u8], offset: usize, bytes: &[u8]) {
633    out[offset..offset + bytes.len()].copy_from_slice(bytes);
634}

636/// `struct elf_prpsinfo` for a 64-bit process (136 bytes).
637fn prpsinfo(id: &Identity) -> Vec<u8> {
638    let mut out = vec![0u8; 136];
639    let state = id.state as u8;
640    out[0] = b"RSDTZW".iter().position(|&c| c == state).unwrap_or(0) as u8;
641    out[1] = state;
642    out[2] = u8::from(state == b'Z');
643    out[3] = id.nice as i8 as u8;
644    put(&mut out, 8, &u64::from(id.flags).to_le_bytes());
645    put(&mut out, 16, &id.uid.to_le_bytes());
646    put(&mut out, 20, &id.gid.to_le_bytes());
647    put(&mut out, 24, &id.tgid.to_le_bytes());
648    put(&mut out, 28, &id.ppid.to_le_bytes());
649    put(&mut out, 32, &id.pgrp.to_le_bytes());
650    put(&mut out, 36, &id.sid.to_le_bytes());
651    let comm = id.comm.as_bytes();
652    put(&mut out, 40, &comm[..comm.len().min(15)]);
653    let args = id.cmdline.as_bytes();
654    put(&mut out, 56, &args[..args.len().min(79)]);
655    out
656}

658/// `NT_FILE`: the file-backed mappings, so a debugger can find the files.
659fn nt_file(maps: &[MemoryMap]) -> Vec<u8> {
660    let files: Vec<(&MemoryMap, &Path)> = maps
661        .iter()
662        .filter_map(|map| match &map.pathname {
663            MMapPath::Path(path) => Some((map, path.as_path())),
664            _ => None,
665        })
666        .collect();
667    let mut out = Vec::new();
668    out.extend_from_slice(&(files.len() as u64).to_le_bytes());
669    out.extend_from_slice(&PAGE.to_le_bytes());
670    for (map, _) in &files {
671        out.extend_from_slice(&map.address.0.to_le_bytes());
672        out.extend_from_slice(&map.address.1.to_le_bytes());
673        out.extend_from_slice(&(map.offset / PAGE).to_le_bytes());
674    }
675    for (_, path) in &files {
676        use std::os::unix::ffi::OsStrExt;
677        out.extend_from_slice(path.as_os_str().as_bytes());
678        out.push(0);
679    }
680    out
681}

729#[derive(Debug)]
730enum WriteError {
731    OverBudget,
732    Deadline,
733    Io(std::io::Error),
734}
735
736impl From<std::io::Error> for WriteError {
737    fn from(error: std::io::Error) -> Self {
738        Self::Io(error)
739    }

742/// Compresses what is pushed into zstd frames of [`CHUNK`] bytes and appends
743/// them to a file, refusing any frame that would pass `budget` or start after
744/// `deadline`.
745struct Compressed {
746    file: File,
747    buf: Vec<u8>,
748    written: u64,
749    budget: u64,
750    deadline: Instant,
751}
752
753impl Compressed {
754    fn push(&mut self, mut data: &[u8]) -> Result<(), WriteError> {
755        while !data.is_empty() {
756            let take = data.len().min(CHUNK - self.buf.len());
757            self.buf.extend_from_slice(&data[..take]);
758            data = &data[take..];
759            if self.buf.len() == CHUNK {
760                self.flush()?;
761            }
762        }
763        Ok(())
764    }
765
766    fn push_zeros(&mut self, mut len: u64) -> Result<(), WriteError> {
767        while len > 0 {
768            let take = len.min(CHUNK as u64) as usize;
769            self.push(&ZEROS[..take])?;
770            len -= take as u64;
771        }
772        Ok(())
773    }
774
775    fn flush(&mut self) -> Result<(), WriteError> {
776        if self.buf.is_empty() {
777            return Ok(());
778        }
779        if Instant::now() >= self.deadline {
780            return Err(WriteError::Deadline);
781        }
782        let frame = ruzstd::encoding::compress_to_vec(&self.buf[..], CompressionLevel::Fastest);
783        self.buf.clear();
784        if self.written + frame.len() as u64 > self.budget {
785            return Err(WriteError::OverBudget);
786        }
787        self.file.write_all(&frame)?;
788        self.written += frame.len() as u64;
789        Ok(())
790    }
791}

793/// The ELF header and program headers for `segments`, with their stored
794/// bytes laid out page-aligned after `notes_len` bytes of notes.
795fn headers(notes_len: u64, segments: &[Segment]) -> (Vec<u8>, u64) {
796    let phnum = 1 + segments.len() as u64;
797    let notes_offset = EHDR_SIZE + PHDR_SIZE * phnum;
798    let data_offset = (notes_offset + notes_len).next_multiple_of(PAGE);
799    let mut out = Vec::with_capacity(notes_offset as usize);
800    out.extend_from_slice(b"\x7fELF\x02\x01\x01");
801    out.resize(16, 0);
802    out.extend_from_slice(&4u16.to_le_bytes()); // e_type: ET_CORE
803    out.extend_from_slice(&arch::MACHINE.to_le_bytes());
804    out.extend_from_slice(&1u32.to_le_bytes()); // e_version
805    out.extend_from_slice(&0u64.to_le_bytes()); // e_entry
806    out.extend_from_slice(&EHDR_SIZE.to_le_bytes()); // e_phoff
807    out.extend_from_slice(&0u64.to_le_bytes()); // e_shoff
808    out.extend_from_slice(&0u32.to_le_bytes()); // e_flags
809    out.extend_from_slice(&(EHDR_SIZE as u16).to_le_bytes());
810    out.extend_from_slice(&(PHDR_SIZE as u16).to_le_bytes());
811    out.extend_from_slice(&(phnum as u16).to_le_bytes());
812    out.extend_from_slice(&[0; 6]); // e_shentsize, e_shnum, e_shstrndx
813    let mut phdr =
814        |kind: u32, flags: u32, offset: u64, vaddr: u64, filesz: u64, memsz: u64, align: u64| {
815            out.extend_from_slice(&kind.to_le_bytes());
816            out.extend_from_slice(&flags.to_le_bytes());
817            out.extend_from_slice(&offset.to_le_bytes());
818            out.extend_from_slice(&vaddr.to_le_bytes());
819            out.extend_from_slice(&0u64.to_le_bytes()); // p_paddr
820            out.extend_from_slice(&filesz.to_le_bytes());
821            out.extend_from_slice(&memsz.to_le_bytes());
822            out.extend_from_slice(&align.to_le_bytes());
823        };
824    phdr(PT_NOTE, 0, notes_offset, 0, notes_len, 0, 4);
825    let mut offset = data_offset;
826    for segment in segments {
827        phdr(
828            PT_LOAD,
829            segment.flags,
830            offset,
831            segment.vaddr,
832            segment.filesz,
833            segment.memsz,
834            PAGE,
835        );
836        offset += segment.filesz.next_multiple_of(PAGE);
837    }
838    (out, data_offset)
839}

841fn write_core(
842    temp: &Path,
843    budget: u64,
844    deadline: Instant,
845    snapshot: &Snapshot,
846    segments: &[Segment],
847    mem: Option<&File>,
848) -> Result<u64, WriteError> {
849    let _ = fs::remove_file(temp);
850    let file = fs::OpenOptions::new()
851        .write(true)
852        .create_new(true)
853        .mode(0o600)
854        .open(temp)?;
855    let mut out = Compressed {
856        file,
857        buf: Vec::with_capacity(CHUNK),
858        written: 0,
859        budget,
860        deadline,
861    };
862    let (headers, data_offset) = headers(snapshot.notes.len() as u64, segments);
863    out.push(&headers)?;
864    out.push(&snapshot.notes)?;
865    out.push_zeros(data_offset - headers.len() as u64 - snapshot.notes.len() as u64)?;
866    let mut page = vec![0u8; CHUNK];
867    for segment in segments {
868        match mem {
869            Some(mem) => copy_memory(&mut out, mem, segment.vaddr, segment.filesz, &mut page)?,
870            None => out.push_zeros(segment.filesz)?,
871        }
872        out.push_zeros(segment.filesz.next_multiple_of(PAGE) - segment.filesz)?;
873    }
874    out.flush()?;
875    Ok(out.written)
876}

878/// Pushes `len` bytes of guest memory from `addr`; a page procfs cannot read
879/// is stored as zeros.
880fn copy_memory(
881    out: &mut Compressed,
882    mem: &File,
883    mut addr: u64,
884    len: u64,
885    buf: &mut [u8],
886) -> Result<(), WriteError> {
887    let end = addr + len;
888    while addr < end {
889        let want = (end - addr).min(buf.len() as u64) as usize;
890        match read_fully(mem, &mut buf[..want], addr) {
891            Ok(()) => out.push(&buf[..want])?,
892            Err(_) => {
893                for page in buf[..want].chunks_mut(PAGE as usize) {
894                    let page_addr = addr;
895                    addr += page.len() as u64;
896                    if read_fully(mem, page, page_addr).is_err() {
897                        page.fill(0);
898                    }
899                }
900                out.push(&buf[..want])?;
901                continue;
902            }
903        }
904        addr += want as u64;
905    }
906    Ok(())
907}
908
909fn read_fully(mem: &File, buf: &mut [u8], addr: u64) -> std::io::Result<()> {
910    let mut done = 0;
911    while done < buf.len() {
912        match mem.read_at(&mut buf[done..], addr + done as u64)? {
913            0 => return Err(std::io::ErrorKind::UnexpectedEof.into()),
914            n => done += n,
915        }
916    }
917    Ok(())
918}
