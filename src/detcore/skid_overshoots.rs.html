9//! Skid overshoots counted as events, not as reports.
10//!
11//! Two sites report a precise PMU timer event that took effect past its
12//! target. Reverie reports it where its timer finds the thread past the
13//! target, usually with a [`reverie::SkidOvershootDetail`] naming the thread
14//! and the target. Detcore's prehook reports it when the thread ran more
15//! branches since its last handler than its timer was armed for
16//! (`record_rcb_overshoot_report`, called by `report_rcb_overshoot`). Both
17//! increment Reverie's process-global count
18//! ([`reverie::record_skid_overshoot`]), so one late event usually counts
19//! twice. That count stays a count of reports: a refusal, and `--verify`'s
20//! attribution of a divergence to skid, need only that it is nonzero, and
21//! Detcore's report is what keeps it nonzero for an overshoot no Reverie site
22//! saw.
23//!
24//! Detcore keys each report by thread and target, the pair a detail names, and
25//! `SkidOvershootCounts::count` counts a Detcore report whose pair matches a
26//! detail once.
27//!
28//! The pair is matched rather than the order of detection, because the two
29//! sites detect in either order. Reverie records a late delivery before the
30//! Tool's timer callback, whose prehook reports it again; it records an event
31//! that another stop overtook only when that stop is settled, after the Tool's
32//! callback for the stop, and so after the prehook's report and after any other
33//! thread the prehook yielded to has run.
34
35use std::collections::HashMap;
36use std::sync::Mutex;
37
38use reverie::SkidOvershootDetail;
39use reverie::Tid;

41/// One skid overshoot Detcore's prehook found: the thread, and the Reverie
42/// clock value at which the timer Detcore had armed for it was due. These are
43/// the `tid` and `rcb_target` of a [`SkidOvershootDetail`] that Reverie records
44/// for the same event.
45#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
46pub(crate) struct RcbOvershootReport {
47    /// The thread whose timer was late.
48    pub(crate) tid: Tid,
49    /// The Reverie clock value at which its timer was due.
50    pub(crate) rcb_target: u64,
51}
52
53/// How many [`RcbOvershootReport`]s are kept between takes; later ones are
54/// counted in [`RcbOvershootReports::unkept`]. Large enough that no run seen so
55/// far comes near it, and bounded, as Reverie bounds its details.
56pub(crate) const RCB_OVERSHOOT_REPORTS_KEPT: usize = 1 << 16;
57
58/// Detcore's reports since the last [`take_rcb_overshoot_reports`].
59#[derive(Debug, Clone, Default, PartialEq, Eq)]
60pub(crate) struct RcbOvershootReports {
61    /// The first [`RCB_OVERSHOOT_REPORTS_KEPT`] reports, in detection order.
62    pub(crate) kept: Vec<RcbOvershootReport>,
63    /// Reports past the first [`RCB_OVERSHOOT_REPORTS_KEPT`].
64    pub(crate) unkept: u64,
65}
66
67impl RcbOvershootReports {
68    /// All of Detcore's reports, kept or not.
69    fn total(&self) -> u64 {
70        u64::try_from(self.kept.len())
71            .unwrap_or(u64::MAX)
72            .saturating_add(self.unkept)
73    }
74}
75
76static DETCORE_STORE: Mutex<RcbOvershootReports> = Mutex::new(RcbOvershootReports {
77    kept: Vec::new(),
78    unkept: 0,
79});
80
81/// Detcore's own reports, since the last [`take_rcb_overshoot_reports`].
82fn detcore_store() -> std::sync::MutexGuard<'static, RcbOvershootReports> {
83    DETCORE_STORE
84        .lock()
85        .unwrap_or_else(|poisoned| poisoned.into_inner())
86}
87
88/// Record a report of Detcore's prehook. Its caller also counts it with
89/// [`reverie::record_skid_overshoot`].
90pub(crate) fn record_rcb_overshoot_report(report: RcbOvershootReport) {
91    let mut store = detcore_store();
92    if store.kept.len() < RCB_OVERSHOOT_REPORTS_KEPT {
93        store.kept.push(report);
94    } else {
95        store.unkept = store.unkept.saturating_add(1);
96    }
97}
98
99/// Take Detcore's reports since the last take and start anew.
100pub(crate) fn take_rcb_overshoot_reports() -> RcbOvershootReports {
101    std::mem::take(&mut *detcore_store())
102}

104/// The skid overshoots of one run.
105#[derive(Debug, Clone, Copy, Default, PartialEq, Eq)]
106pub struct SkidOvershootCounts {
107    /// Distinct overshoot events: a Detcore report and a Reverie report of the
108    /// same thread and target count once. At most `reports`, and zero only
109    /// when `reports` is.
110    pub events: u64,
111    /// Reports of every site, Reverie's process-global count.
112    pub reports: u64,
113}
114
115impl SkidOvershootCounts {
116    /// Count the events among `all_reports`, Reverie's count of every site's
117    /// reports, given the `details` Reverie kept and `detcore`, Detcore's own
118    /// reports, all taken since the same point.
119    ///
120    /// A Detcore report whose thread and target match a detail is that
121    /// detail's event. A Reverie report can lack a detail, since Reverie
122    /// keeps only its first [`reverie::SKID_OVERSHOOT_DETAILS_KEPT`] and its
123    /// block-step guard records none. A Detcore report left unmatched is then
124    /// taken to be such a report's event, while both remain: a Detcore report
125    /// whose event Reverie also reported is the usual case.
126    ///
127    /// So the count is exact whenever every Detcore report has a Reverie
128    /// report with the same thread and target, however many details Reverie
129    /// kept. It counts one event twice only if the two sites name different
130    /// targets for it while Reverie keeps that event's detail, and it
131    /// undercounts only a Detcore report with no Reverie twin that is taken
132    /// for a detail-less Reverie report of another event.
133    pub(crate) fn count(
134        all_reports: u64,
135        details: &[SkidOvershootDetail],
136        detcore: &RcbOvershootReports,
137    ) -> Self {
138        let detcore_reports = detcore.total();
139        let reverie_reports = all_reports.saturating_sub(detcore_reports);
140        let mut unmatched: HashMap<(Tid, u64), u64> = HashMap::new();
141        for detail in details {
142            *unmatched
143                .entry((detail.tid, detail.rcb_target))
144                .or_default() += 1;
145        }
146        let mut matched = 0u64;
147        for report in &detcore.kept {
148            if let Some(left) = unmatched.get_mut(&(report.tid, report.rcb_target))
149                && *left > 0
150            {
151                *left -= 1;
152                matched += 1;
153            }
154        }
155        // A detail with no count of its own (a take that drained one store
156        // and not the other) is no report to merge.
157        let detailed = u64::try_from(details.len())
158            .unwrap_or(u64::MAX)
159            .min(reverie_reports);
160        let matched = matched.min(detailed);
161        let paired = (detcore_reports - matched).min(reverie_reports - detailed);
162        // matched + paired is at most reverie_reports and at most
163        // detcore_reports: less than all_reports while either is nonzero, so
164        // events is zero only when all_reports is.
165        let counts = Self {
166            events: all_reports - matched - paired,
167            reports: all_reports,
168        };
169        debug_assert!(
170            counts.events <= counts.reports && (counts.events > 0) == (counts.reports > 0),
171            "{counts:?}"
172        );
173        counts
174    }
175}

177/// Take every site's skid overshoots since the last take: Reverie's count and
178/// details and Detcore's reports, counted with `SkidOvershootCounts::count`.
179/// Returns the details as well, in detection order.
180pub fn take_skid_overshoots() -> (SkidOvershootCounts, Vec<SkidOvershootDetail>) {
181    let all_reports = reverie::take_skid_overshoot_count();
182    let details = reverie::take_skid_overshoot_details();
183    let detcore = take_rcb_overshoot_reports();
184    (
185        SkidOvershootCounts::count(all_reports, &details, &detcore),
186        details,
187    )
188}
189
190/// [`take_skid_overshoots`]'s counts, read without taking them, as a report
191/// written before the run's final take does. Reverie returns its details only
192/// by taking them, so they are put back in order: unchanged, since this runs
193/// once the run's threads have stopped and no site records in between, and
194/// all of them, since no more than Reverie keeps were taken. A Reverie
195/// accessor that reads the details without taking them, as
196/// `reverie::skid_overshoot_count` reads the count, would replace the take
197/// and re-record; that is the clean follow-up, left out of this change to
198/// keep Reverie's pin.
199pub fn skid_overshoot_counts() -> SkidOvershootCounts {
200    let all_reports = reverie::skid_overshoot_count();
201    let details = reverie::take_skid_overshoot_details();
202    let detcore = detcore_store().clone();
203    let counts = SkidOvershootCounts::count(all_reports, &details, &detcore);
204    for detail in details {
205        reverie::record_skid_overshoot_detail(detail);
206    }
207    counts
208}

211pub(crate) mod tests {
212    use super::*;
213
214    /// Serializes the tests that use the process-global counts and details.
215    static GLOBAL_SKID_OVERSHOOTS: Mutex<()> = Mutex::new(());
216
217    pub(crate) fn lock_global_skid_overshoots() -> std::sync::MutexGuard<'static, ()> {
218        GLOBAL_SKID_OVERSHOOTS
219            .lock()
220            .unwrap_or_else(|poisoned| poisoned.into_inner())
221    }
222
223    fn tid(raw: i32) -> Tid {
224        Tid::from_raw(raw)
225    }
226
227    fn detail(raw_tid: i32, rcb_target: u64) -> SkidOvershootDetail {
228        SkidOvershootDetail {
229            tid: tid(raw_tid),
230            kind: reverie::SkidOvershootKind::DeliveredLate,
231            rcb_actual: rcb_target + 13,
232            rcb_target,
233            overflow_point: Some(rcb_target),
234            overflow_recorded: Some(true),
235            sample_age_ns: Some(400_000),
236        }
237    }
238
239    fn detcore(pairs: &[(i32, u64)]) -> RcbOvershootReports {
240        RcbOvershootReports {
241            kept: pairs
242                .iter()
243                .map(|&(raw_tid, rcb_target)| RcbOvershootReport {
244                    tid: tid(raw_tid),
245                    rcb_target,
246                })
247                .collect(),
248            unkept: 0,
249        }
250    }
251
252    fn count(all_reports: u64, details: &[SkidOvershootDetail], ours: &[(i32, u64)]) -> (u64, u64) {
253        let counts = SkidOvershootCounts::count(all_reports, details, &detcore(ours));
254        (counts.events, counts.reports)
255    }
256
257    #[test]
258    fn one_event_reported_by_both_sites_counts_once() {
259        assert_eq!(count(2, &[detail(5, 1_000)], &[(5, 1_000)]), (1, 2));
260    }
261
262    #[test]
263    fn a_detection_by_one_site_counts_once() {
264        // Detcore's prehook alone: the safety net for an overshoot no Reverie
265        // site counted.
266        assert_eq!(count(1, &[], &[(5, 1_000)]), (1, 1));
267        // Reverie alone.
268        assert_eq!(count(1, &[detail(5, 1_000)], &[]), (1, 1));
269    }
270
271    #[test]
272    fn two_events_count_twice() {
273        let both = [detail(5, 1_000), detail(5, 2_000)];
274        assert_eq!(count(4, &both, &[(5, 1_000), (5, 2_000)]), (2, 4));
275        // The same target on two threads is two events.
276        let threads = [detail(5, 1_000), detail(6, 1_000)];
277        assert_eq!(count(4, &threads, &[(6, 1_000), (5, 1_000)]), (2, 4));
278        // A Detcore report of another target beside a Reverie event.
279        assert_eq!(count(2, &[detail(5, 1_000)], &[(5, 1_500)]), (2, 2));
280        // No reports, no events.
281        assert_eq!(count(0, &[], &[]), (0, 0));
282    }
283
284    #[test]
285    fn reverie_reports_without_a_detail_take_up_unmatched_detcore_reports() {
286        // The block-step guard counts without a detail.
287        assert_eq!(count(2, &[], &[(5, 1_000)]), (1, 2));
288        // Twenty events, each reported by both sites; Reverie kept details of
289        // the first sixteen.
290        let targets: Vec<u64> = (1..=20).map(|n| n * 1_000).collect();
291        let details: Vec<_> = targets
292            .iter()
293            .take(reverie::SKID_OVERSHOOT_DETAILS_KEPT)
294            .map(|&target| detail(5, target))
295            .collect();
296        let ours: Vec<_> = targets.iter().map(|&target| (5, target)).collect();
297        assert_eq!(count(40, &details, &ours), (20, 40));
298        // Detcore reports past those it keeps pair the same way.
299        let mut capped = detcore(&ours[..16]);
300        capped.unkept = 4;
301        let counts = SkidOvershootCounts::count(40, &details, &capped);
302        assert_eq!((counts.events, counts.reports), (20, 40));
303    }

305    #[test]
306    fn inconsistent_inputs_never_lose_every_event() {
307        // A detail with no report to go with it.
308        assert_eq!(count(1, &[detail(5, 1_000)], &[(5, 1_000)]), (1, 1));
309        // More Detcore reports than reports at all.
310        assert_eq!(count(1, &[], &[(5, 1_000), (5, 2_000)]), (1, 1));
311        // Never zero while reports are not.
312        for all_reports in 1..6 {
313            for matching in 0..3 {
314                let details: Vec<_> = (0..matching).map(|n| detail(5, n)).collect();
315                let ours: Vec<_> = (0..matching).map(|n| (5, n)).collect();
316                assert!(count(all_reports, &details, &ours).0 > 0);
317            }
318        }
319    }

321    #[test]
322    fn detcore_keeps_a_bounded_number_of_reports() {
323        let almost_full = RcbOvershootReports {
324            kept: vec![
325                RcbOvershootReport {
326                    tid: tid(5),
327                    rcb_target: 1,
328                };
329                RCB_OVERSHOOT_REPORTS_KEPT - 1
330            ],
331            unkept: 0,
332        };
333        let _lock = lock_global_skid_overshoots();
334        *detcore_store() = almost_full;
335        record_rcb_overshoot_report(RcbOvershootReport {
336            tid: tid(5),
337            rcb_target: 2,
338        });
339        record_rcb_overshoot_report(RcbOvershootReport {
340            tid: tid(5),
341            rcb_target: 3,
342        });
343        let taken = take_rcb_overshoot_reports();
344        assert_eq!(taken.kept.len(), RCB_OVERSHOOT_REPORTS_KEPT);
345        assert_eq!(taken.kept.last().map(|report| report.rcb_target), Some(2));
346        assert_eq!(taken.unkept, 1);
347        assert_eq!(taken.total(), RCB_OVERSHOOT_REPORTS_KEPT as u64 + 1);
348        assert_eq!(take_rcb_overshoot_reports(), RcbOvershootReports::default());
349    }
