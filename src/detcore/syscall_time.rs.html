22/// Returns a deterministic, representative syscall cost in nanoseconds.
23///
24/// These values model syscall overhead rather than host-observed latency. The fallback is
25/// deliberately nonzero so newly intercepted syscalls cannot stop virtual-time progress.
26pub(crate) fn cost_ns(sysno: Sysno) -> u64 {
27    match sysno {
28        Sysno::gettimeofday => GETTIMEOFDAY_NS,
29
30        Sysno::time
31        | Sysno::clock_gettime
32        | Sysno::clock_getres
33        | Sysno::timerfd_gettime
34        | Sysno::timer_gettime
35        | Sysno::timer_getoverrun => CLOCK_NS,
36
37        Sysno::getpid
38        | Sysno::gettid
39        | Sysno::getcpu
40        | Sysno::arch_prctl
41        | Sysno::set_tid_address
42        | Sysno::set_robust_list
43        | Sysno::rt_sigprocmask
44        | Sysno::rt_sigaction
45        | Sysno::sigaltstack => FAST_NS,
46
47        Sysno::stat
48        | Sysno::lstat
49        | Sysno::fstat
50        | Sysno::newfstatat
51        | Sysno::statx
52        | Sysno::uname
53        | Sysno::getrusage
54        | Sysno::sysinfo
55        | Sysno::access
56        | Sysno::readlink
57        | Sysno::readlinkat
58        | Sysno::utime
59        | Sysno::utimes
60        | Sysno::utimensat
61        | Sysno::futimesat
62        | Sysno::prlimit64
63        | Sysno::prctl => METADATA_NS,
64
65        Sysno::futex
66        | Sysno::rseq
67        | Sysno::sched_yield
68        | Sysno::nanosleep
69        | Sysno::clock_nanosleep
70        | Sysno::alarm
71        | Sysno::pause
72        | Sysno::poll
73        | Sysno::epoll_pwait
74        | Sysno::epoll_wait
75        | Sysno::epoll_wait_old
76        | Sysno::rt_sigtimedwait => SYNC_NS,
77
78        Sysno::read
79        | Sysno::pread64
80        | Sysno::write
81        | Sysno::lseek
82        | Sysno::fadvise64
83        | Sysno::fcntl
84        | Sysno::ioctl
85        | Sysno::getrandom
86        | Sysno::getdents
87        | Sysno::getdents64 => IO_NS,
88
89        Sysno::mmap
90        | Sysno::munmap
91        | Sysno::mremap
92        | Sysno::brk
93        | Sysno::mprotect
94        | Sysno::madvise
95        | Sysno::userfaultfd => MEMORY_NS,
96
97        Sysno::open
98        | Sysno::openat
99        | Sysno::creat
100        | Sysno::close
101        | Sysno::dup
102        | Sysno::dup2
103        | Sysno::dup3
104        | Sysno::pipe
105        | Sysno::pipe2
106        | Sysno::eventfd
107        | Sysno::eventfd2
108        | Sysno::signalfd
109        | Sysno::signalfd4
110        | Sysno::timerfd_create
111        | Sysno::timerfd_settime
112        | Sysno::timer_create
113        | Sysno::timer_settime
114        | Sysno::timer_delete
115        | Sysno::inotify_init
116        | Sysno::inotify_init1
117        | Sysno::inotify_add_watch
118        | Sysno::inotify_rm_watch
119        | Sysno::memfd_create
120        | Sysno::io_uring_setup
121        | Sysno::io_uring_enter
122        | Sysno::io_uring_register
123        | Sysno::epoll_create
124        | Sysno::epoll_create1
125        | Sysno::epoll_ctl
126        | Sysno::epoll_ctl_old
127        | Sysno::add_key
128        | Sysno::request_key
129        | Sysno::keyctl => OPEN_CLOSE_NS,
130
131        Sysno::socket
132        | Sysno::socketpair
133        | Sysno::connect
134        | Sysno::bind
135        | Sysno::accept
136        | Sysno::accept4
137        | Sysno::recvfrom
138        | Sysno::recvmsg
139        | Sysno::sendto
140        | Sysno::sendmsg
141        | Sysno::sendmmsg => NETWORK_NS,
142
143        Sysno::clone
144        | Sysno::clone3
145        | Sysno::fork
146        | Sysno::vfork
147        | Sysno::wait4
148        | Sysno::setsid
149        | Sysno::execve
150        | Sysno::execveat
151        | Sysno::exit
152        | Sysno::exit_group
153        | Sysno::sched_getaffinity
154        | Sysno::sched_setaffinity => PROCESS_NS,
155
156        _ => FAST_NS,
157    }
158}

160/// The most syscalls one backend-runtime bootstrap window may leave uncharged on
161/// the bootstrapping thread. Once a window reaches this count, every further
162/// syscall in it is charged its normal cost, so a window cannot stop the thread's
163/// syscall-driven clock indefinitely.
164///
165/// The LiteInst runtime's own constructor makes more syscalls the larger the
166/// image: for each executable mapping it rereads `/proc/self/maps` and
167/// allocates a trampoline arena. Uncharged syscalls per window, measured under
168/// `--backend liteinst --max-timeslice=disabled` with Reverie dbf2b5c8 (lines
169/// of `ldd` output in parentheses): 315 for `tests/c/host_identity.c` and for
170/// each image of `hermit-cli/tests/fixtures/clock_trajectory.c`; 401 for
171/// `python3` (5); 1,589 for `curl` (28); 3,588 for `gdb` (58); 4,752 for
172/// `qemu-system-x86_64` (72); 6,970 for `emacs` (95). The counts vary slightly
173/// with the environment: in another build directory the same commit measured
174/// 316 for `host_identity.c` and 318 for each `clock_trajectory.c` image,
175/// probably because the `/proc/self/maps` lines the runtime rereads name the
176/// mapped paths. Beyond a fixed cost the
177/// count grows faster than the number of libraries, because each reread of
178/// `/proc/self/maps` is longer: about 57 syscalls per `ldd` line for `curl` and
179/// 73 for `emacs`. So the margin below shrinks as images grow. The cap must
180/// stay at least four times the largest such count; 32,768 is 4.7 times
181/// emacs's. A window that reaches the cap says so once, at info level (see
182/// `ThreadState::charge_syscall_time`).
183pub(crate) const MAX_UNCHARGED_BOOTSTRAP_SYSCALLS: u32 = 32768;
184
185/// Returns true for syscalls whose main result is a clock or timer value: the
186/// wall-clock and monotonic clocks, the CPU-time clocks and counters, uptime,
187/// the time left on a timer, and the `time` field an `adjtimex` or
188/// `clock_adjtime` query returns (glibc's `ntp_gettime` reads the clock that
189/// way). Timerfd calls currently pass through to a host timer
190/// (https://github.com/rrnewton/hermit/issues/1923), so `timerfd_gettime`
191/// reports host time left, not virtual time; it is listed because its result
192/// is a timer value.
193///
194/// Inside a backend-runtime bootstrap window these syscalls are always charged,
195/// so two such reads on the bootstrapping thread differ by at least one syscall
196/// cost, exactly as outside the window. `clock_getres` is not listed: it reports
197/// a constant.
198///
199/// Some syscalls give the guest virtual time without it being their main
200/// result. They are not listed, so inside a window they are bounded only by
201/// [`MAX_UNCHARGED_BOOTSTRAP_SYSCALLS`]:
202/// - side outputs: the old value written by `setitimer` or `timer_settime`,
203///   and the seconds `alarm` returns;
204/// - a `read` of a procfs file Detcore renders from virtual time, such as
205///   `/proc/uptime` (whole seconds);
206/// - `utime` and `utimes` with no times given, and every write that updates a
207///   file's modification time (`write`, `writev`, `pwrite64`, `pwritev`,
208///   `pwritev2` and `sendfile`), which stamp the file with the virtual time,
209///   readable back through `stat`;
210/// - socket receive timestamps, which Detcore sets to the virtual time of the
211///   receive.
212pub(crate) fn observes_virtual_time(sysno: Sysno) -> bool {
213    matches!(
214        sysno,
215        Sysno::gettimeofday
216            | Sysno::time
217            | Sysno::clock_gettime
218            | Sysno::sysinfo
219            | Sysno::times
220            | Sysno::getrusage
221            | Sysno::timerfd_gettime
222            | Sysno::timer_gettime
223            | Sysno::getitimer
224            | Sysno::adjtimex
225            | Sysno::clock_adjtime
226    )
227}

229#[cfg(test)]
230mod tests {
231    use detcore_model::time::DetTime;
232
233    use super::*;
234
235    #[test]
236    fn representative_costs_cover_each_category() {
237        assert_eq!(cost_ns(Sysno::clock_gettime), 10_000);
238        assert_eq!(cost_ns(Sysno::gettimeofday), 2_000);
239        assert_eq!(cost_ns(Sysno::getpid), 250);
240        assert_eq!(cost_ns(Sysno::statx), 3_000);
241        assert_eq!(cost_ns(Sysno::futex), 5_000);
242        assert_eq!(cost_ns(Sysno::read), 10_000);
243        assert_eq!(cost_ns(Sysno::mmap), 10_000);
244        assert_eq!(cost_ns(Sysno::openat), 25_000);
245        assert_eq!(cost_ns(Sysno::socket), 50_000);
246        assert_eq!(cost_ns(Sysno::clone), 250_000);
247    }
248
249    #[test]
250    fn unsupported_syscalls_still_advance_time() {
251        assert_eq!(cost_ns(Sysno::restart_syscall), FAST_NS);
252        assert!(cost_ns(Sysno::restart_syscall) > 0);
253    }
254
255    #[test]
256    fn dettime_preserves_legacy_serialized_syscall_counts() {
257        let mut time = DetTime::zero();
258        time.add_syscall();
259        time.add_syscall();
260        let mut serialized = serde_json::to_value(time).unwrap();
261        serialized.as_object_mut().unwrap().remove("syscall_nanos");
262
263        let mut restored: DetTime = serde_json::from_value(serialized).unwrap();
264        assert_eq!(restored.without_starting().as_nanos(), 20_000);
265
266        restored.add_syscall_with_cost(cost_ns(Sysno::clock_gettime));
267        assert_eq!(restored.without_starting().as_nanos(), 30_000);
268    }
269
270    #[test]
271    fn dettime_accumulates_mixed_syscall_costs() {
272        let mut time = DetTime::zero();
273        time.add_syscall_with_cost(cost_ns(Sysno::clock_gettime));
274        time.add_syscall_with_cost(cost_ns(Sysno::read));
275
276        assert_eq!(time.without_starting().as_nanos(), 20_000);
277    }
278
279    #[test]
280    fn dettime_applies_multiplier_to_syscall_costs() {
281        let mut time = DetTime::zero().with_multiplier(2.0);
282        time.add_syscall_with_cost(cost_ns(Sysno::write));
283
284        assert_eq!(time.without_starting().as_nanos(), 20_000);
285    }
