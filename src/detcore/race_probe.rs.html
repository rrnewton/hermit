9//! Race probes: env-gated perturbation points on Hermit's own host-timed
10//! edges, for reproducing rare races in Hermit itself (docs/RACE_PROBES.md).
11//!
12//! Off unless [`RACE_PROBE_ENV`] is set to a non-empty list of
13//! `point=LO[-HI][@P]` items. At `point`, with probability `P` (default 1), a
14//! visit is perturbed: an async point waits a uniform `LO..=HI` milliseconds,
15//! and a synchronous point takes its perturbed branch (its numbers ignored).
16//! [`RACE_PROBE_SEED_ENV`] fixes the choices; without it a seed is drawn from
17//! the host clock. Either way the configuration and seed are printed once to
18//! stderr, so a failing run names its seed.
19//!
20//! `point=hold` makes every visit to an async point wait, without a time
21//! limit, until a file named after the point exists in
22//! [`RACE_PROBE_HOLD_DIR_ENV`]: a test can then act on a task while it is
23//! known to be at the point, and release it after. A held visit prints one
24//! line to stderr naming the file it waits for.
25//!
26//! Unset, every point is one read of an initialized `OnceLock` and a branch;
27//! nothing is logged, and no tracing event is emitted.
28//!
29//! A probe only moves host timing that Hermit already treats as
30//! nondeterministic input (when a backgrounded call returns, and when the
31//! scheduler samples that it has), so an outcome reached under a probe is one
32//! host timing alone could produce. Decisions are drawn in visit order from one
33//! process-wide counter, so a seed replays exactly only when the number of
34//! visits before the deciding one does not itself depend on host timing.
35

44/// The variable that enables race probes and lists their settings.
45pub const RACE_PROBE_ENV: &str = "HERMIT_RACE_PROBE";
46
47/// The variable that fixes the seed of the race probes' choices.
48pub const RACE_PROBE_SEED_ENV: &str = "HERMIT_RACE_PROBE_SEED";
49
50/// The directory whose files release held points (`point=hold`).
51pub const RACE_PROBE_HOLD_DIR_ENV: &str = "HERMIT_RACE_PROBE_HOLD_DIR";

53/// How often a held visit looks for its release file.
54const HOLD_POLL: Duration = Duration::from_millis(10);
55
56/// An async wait just before a backgrounded blocking call runs.
57pub(crate) const BG_PRE_CALL: &str = "bg_pre_call";
58/// An async wait after a backgrounded call returns, before it rejoins.
59pub(crate) const BG_POST_CALL: &str = "bg_post_call";
60/// The scheduler treats a finished backgrounded call as still running on one
61/// of its checks.
62pub(crate) const BG_HARVEST_SKIP: &str = "bg_harvest_skip";
63/// An async wait after a FIFO open's staging page is mapped, before anything
64/// is written to it: a task killed by a host process then leaves the page
65/// mapped and empty, as one killed during the `mmap` does.
66pub(crate) const FIFO_STAGING_MAPPED: &str = "fifo_staging_mapped";
67/// An async wait before a task unmaps a staging page that a killed task left:
68/// a task killed by a host process then leaves the page to the next task.
69pub(crate) const FIFO_STAGING_RECLAIM: &str = "fifo_staging_reclaim";
70/// An async wait in `openat` after Detcore has named the directory a relative
71/// lookup starts from, before the kernel's open: without sequentialized
72/// threads another guest thread can replace that directory's descriptor in
73/// between, as host timing alone can let it.
74pub(crate) const PROCFS_SPELLING_RESOLVED: &str = "procfs_spelling_resolved";

76/// Every probe point, for validating a configuration.
77const POINTS: [&str; 6] = [
78    BG_PRE_CALL,
79    BG_POST_CALL,
80    BG_HARVEST_SKIP,
81    FIFO_STAGING_MAPPED,
82    FIFO_STAGING_RECLAIM,
83    PROCFS_SPELLING_RESOLVED,
84];

86#[derive(Clone, Copy, Debug, PartialEq)]
87struct Spec {
88    lo_ms: u64,
89    hi_ms: u64,
90    probability: f64,
91    /// Every visit waits for the point's release file (`point=hold`).
92    hold: bool,
93}
94
95#[derive(Debug)]
96struct Probes {
97    seed: u64,
98    counter: AtomicU64,
99    points: BTreeMap<String, Spec>,
100    hold_dir: Option<PathBuf>,
101}
102
103/// What a perturbed visit does.
104#[derive(Clone, Debug, PartialEq)]
105enum Perturbation {
106    /// An async point waits this long; a synchronous point takes its branch.
107    Wait(Duration),
108    /// An async point waits until this file exists.
109    Hold(PathBuf),
110}
111
112fn parse_item(item: &str) -> Result<(String, Spec), String> {
113    let (name, rest) = item
114        .split_once('=')
115        .ok_or_else(|| format!("{item:?} is not point=LO[-HI][@P]"))?;
116    if !POINTS.contains(&name) {
117        return Err(format!("unknown point {name:?}; points are {POINTS:?}"));
118    }
119    if rest == "hold" {
120        return Ok((
121            name.to_owned(),
122            Spec {
123                lo_ms: 0,
124                hi_ms: 0,
125                probability: 1.0,
126                hold: true,
127            },
128        ));
129    }
130    let (range, probability) = match rest.split_once('@') {
131        Some((range, p)) => (
132            range,
133            p.parse::<f64>()
134                .ok()
135                .filter(|p| (0.0..=1.0).contains(p))
136                .ok_or_else(|| format!("{item:?}: probability must be in [0, 1]"))?,
137        ),
138        None => (rest, 1.0),
139    };
140    let number = |text: &str| {
141        text.parse::<u64>()
142            .map_err(|_| format!("{item:?}: {text:?} is not a number of milliseconds"))
143    };
144    let (lo_ms, hi_ms) = match range.split_once('-') {
145        Some((lo, hi)) => (number(lo)?, number(hi)?),
146        None => {
147            let ms = number(range)?;
148            (ms, ms)
149        }
150    };
151    if hi_ms < lo_ms {
152        return Err(format!("{item:?}: the range is empty"));
153    }
154    Ok((
155        name.to_owned(),
156        Spec {
157            lo_ms,
158            hi_ms,
159            probability,
160            hold: false,
161        },
162    ))
163}
164
165impl Probes {
166    /// The probes `config`, `seed` and `hold_dir` (the three variables'
167    /// values) describe: `None` when probes are off. A malformed item is
168    /// reported and skipped, as is a hold without a hold directory.
169    fn from_settings(
170        config: Option<&str>,
171        seed: Option<&str>,
172        hold_dir: Option<&str>,
173    ) -> Option<Self> {
174        let config = config?;
175        let hold_dir = hold_dir.filter(|dir| !dir.is_empty()).map(PathBuf::from);
176        // A disabled probe must not create randomized hash state: even an
177        // empty HashMap initializes RandomState's thread-local random keys.
178        // These few exact names need no hashing, and insert still makes the
179        // last valid duplicate win without changing lookup or draw order.
180        let mut points = BTreeMap::new();
181        for item in config.split(',').map(str::trim).filter(|s| !s.is_empty()) {
182            match parse_item(item) {
183                Ok((_, spec)) if spec.hold && hold_dir.is_none() => {
184                    eprintln!(
185                        "race_probe: ignoring {item:?}: {RACE_PROBE_HOLD_DIR_ENV} is not set"
186                    );
187                }
188                Ok((name, spec)) => {
189                    points.insert(name, spec);
190                }
191                Err(error) => eprintln!("race_probe: ignoring {error}"),
192            }
193        }
194        if points.is_empty() {
195            return None;
196        }
197        let seed = match seed.map(str::parse::<u64>) {
198            Some(Ok(seed)) => seed,
199            _ => {
200                std::time::SystemTime::now()
201                    .duration_since(std::time::UNIX_EPOCH)
202                    .map_or(0, |d| d.as_nanos() as u64)
203                    ^ u64::from(std::process::id())
204            }
205        };
206        Some(Probes {
207            seed,
208            counter: AtomicU64::new(0),
209            points,
210            hold_dir,
211        })
212    }
213
214    /// The decision at `point` for this visit: `None` when the point is not
215    /// configured or this visit is not perturbed, otherwise what it does.
216    fn choose(&self, point: &str) -> Option<Perturbation> {
217        let spec = self.points.get(point)?;
218        if spec.hold {
219            // Configured only with a hold directory (`from_settings`).
220            return Some(Perturbation::Hold(self.hold_dir.as_ref()?.join(point)));
221        }
222        let visit = self.counter.fetch_add(1, Ordering::Relaxed);
223        let r = splitmix(self.seed ^ splitmix(visit));
224        let unit = (r >> 11) as f64 / (1u64 << 53) as f64;
225        if unit >= spec.probability {
226            return None;
227        }
228        let span = spec.hi_ms - spec.lo_ms;
229        let ms = spec.lo_ms
230            + if span == 0 {
231                0
232            } else {
233                splitmix(r) % (span + 1)
234            };
235        Some(Perturbation::Wait(Duration::from_millis(ms)))
236    }
237}
238
239fn probes() -> Option<&'static Probes> {
240    static PROBES: OnceLock<Option<Probes>> = OnceLock::new();
241    PROBES
242        .get_or_init(|| {
243            let config = std::env::var(RACE_PROBE_ENV).ok()?;
244            let seed = std::env::var(RACE_PROBE_SEED_ENV).ok();
245            let hold_dir = std::env::var(RACE_PROBE_HOLD_DIR_ENV).ok();
246            let probes =
247                Probes::from_settings(Some(&config), seed.as_deref(), hold_dir.as_deref())?;
248            eprintln!(
249                "race_probe: {RACE_PROBE_ENV}={config} {RACE_PROBE_SEED_ENV}={}",
250                probes.seed
251            );
252            Some(probes)
253        })
254        .as_ref()
255}
256
257fn splitmix(mut z: u64) -> u64 {
258    z = z.wrapping_add(0x9e37_79b9_7f4a_7c15);
259    z = (z ^ (z >> 30)).wrapping_mul(0xbf58_476d_1ce4_e5b9);
260    z = (z ^ (z >> 27)).wrapping_mul(0x94d0_49bb_1331_11eb);
261    z ^ (z >> 31)
262}
263
264fn choose(point: &str) -> Option<Perturbation> {
265    let perturbation = probes()?.choose(point)?;
266    tracing::debug!(target: "race_probe", "race_probe: {point} perturbed ({perturbation:?})");
267    Some(perturbation)
268}
269
270/// An async wait at `point`, when configured.
271pub(crate) async fn delay(point: &str) {
272    match choose(point) {
273        Some(Perturbation::Wait(wait)) if !wait.is_zero() => tokio::time::sleep(wait).await,
274        Some(Perturbation::Hold(release)) => hold(point, &release).await,
275        _ => {}
276    }
277}
278
279/// Waits until `release` exists, saying so on stderr unless it already does.
280async fn hold(point: &str, release: &Path) {
281    if release.exists() {
282        return;
283    }
284    eprintln!(
285        "race_probe: {point} holding until {} exists",
286        release.display()
287    );
288    while !release.exists() {
289        tokio::time::sleep(HOLD_POLL).await;
290    }
291}

293/// Whether a synchronous point takes its perturbed branch on this visit.
294pub(crate) fn perturb(point: &str) -> bool {
295    choose(point).is_some()
296}

302    #[test]
303    fn race_probes_are_off_unless_configured() {
304        // Unset, empty, or naming no valid point: no probes, so every point
305        // returns before drawing a decision.
306        assert!(Probes::from_settings(None, None, None).is_none());
307        assert!(Probes::from_settings(None, Some("7"), None).is_none());
308        assert!(Probes::from_settings(Some(""), Some("7"), None).is_none());
309        assert!(Probes::from_settings(Some(" , "), None, None).is_none());
310        assert!(Probes::from_settings(Some("no_such_point=5"), None, None).is_none());
311        // This test process does not set the variable, so the process-wide
312        // probes are off and no point perturbs.
313        assert!(std::env::var_os(RACE_PROBE_ENV).is_none());
314        assert!(probes().is_none());
315        for point in POINTS {
316            assert!(!perturb(point));
317            assert_eq!(choose(point), None);
318        }
319    }

321    #[test]
322    fn race_probe_settings_parse_as_documented() {
323        assert_eq!(
324            parse_item("bg_pre_call=300"),
325            Ok((
326                BG_PRE_CALL.to_owned(),
327                Spec {
328                    lo_ms: 300,
329                    hi_ms: 300,
330                    probability: 1.0,
331                    hold: false,
332                }
333            ))
334        );
335        assert_eq!(
336            parse_item("bg_post_call=0-50@0.25"),
337            Ok((
338                BG_POST_CALL.to_owned(),
339                Spec {
340                    lo_ms: 0,
341                    hi_ms: 50,
342                    probability: 0.25,
343                    hold: false,
344                }
345            ))
346        );
347        for bad in [
348            "bg_pre_call",
349            "bg_pre_call=x",
350            "bg_pre_call=5-2",
351            "bg_pre_call=1@2",
352            "bg_harvest_skip=0@-0.1",
353            "unknown=1",
354        ] {
355            assert!(parse_item(bad).is_err(), "{bad} parsed");
356        }
357        // A bad item is skipped, the good ones stay.
358        let probes = Probes::from_settings(
359            Some("bg_pre_call=x, bg_harvest_skip=0@0.5"),
360            Some("1"),
361            None,
362        )
363        .unwrap();
364        assert_eq!(probes.points.len(), 1);
365        assert!(probes.points.contains_key(BG_HARVEST_SKIP));
366
367        // A later valid duplicate replaces the earlier setting. A malformed
368        // duplicate is ignored and must not erase that last valid value.
369        let probes = Probes::from_settings(
370            Some("bg_pre_call=10@0,bg_pre_call=20@1,bg_pre_call=x,bg_harvest_skip=0@0.5"),
371            Some("7"),
372            None,
373        )
374        .unwrap();
375        assert_eq!(probes.points.len(), 2);
376        assert_eq!(
377            probes.points.get(BG_PRE_CALL),
378            Some(&Spec {
379                lo_ms: 20,
380                hi_ms: 20,
381                probability: 1.0,
382                hold: false,
383            })
384        );
385        assert_eq!(
386            probes.choose(BG_PRE_CALL),
387            Some(Perturbation::Wait(Duration::from_millis(20)))
388        );
389    }

391    #[test]
392    fn race_probe_choices_follow_probability_range_and_seed() {
393        let draw = |config: &str, seed: &str, point: &str| {
394            let probes = Probes::from_settings(Some(config), Some(seed), None).unwrap();
395            (0..1000).map(|_| probes.choose(point)).collect::<Vec<_>>()
396        };
397        assert!(
398            draw("bg_harvest_skip=0@0", "1", BG_HARVEST_SKIP)
399                .iter()
400                .all(Option::is_none)
401        );
402        assert!(
403            draw("bg_harvest_skip=0@1", "1", BG_HARVEST_SKIP)
404                .iter()
405                .all(|wait| *wait == Some(Perturbation::Wait(Duration::ZERO)))
406        );
407        // A point that is not configured never perturbs.
408        assert!(
409            draw("bg_harvest_skip=0@1", "1", BG_PRE_CALL)
410                .iter()
411                .all(Option::is_none)
412        );
413        let half = draw("bg_pre_call=10-20@0.5", "3", BG_PRE_CALL);
414        let taken = half.iter().flatten().count();
415        assert!((400..600).contains(&taken), "{taken} of 1000 at 0.5");
416        assert!(half.iter().flatten().all(|wait| {
417            matches!(wait, Perturbation::Wait(wait)
418                if (Duration::from_millis(10)..=Duration::from_millis(20)).contains(wait))
419        }));
420        // The same seed draws the same sequence; another seed does not.
421        assert_eq!(half, draw("bg_pre_call=10-20@0.5", "3", BG_PRE_CALL));
422        assert_ne!(half, draw("bg_pre_call=10-20@0.5", "4", BG_PRE_CALL));
423
424        // Missing-point lookups must not consume a visit before the actual
425        // seeded draws, including when configured points are interleaved.
426        let config = Some("bg_pre_call=10-20@0.5,bg_harvest_skip=0@0.5");
427        let with_missing = Probes::from_settings(config, Some("3"), None).unwrap();
428        let direct = Probes::from_settings(config, Some("3"), None).unwrap();
429        for _ in 0..1000 {
430            assert_eq!(with_missing.choose("no_such_point"), None);
431            assert_eq!(with_missing.choose(BG_POST_CALL), None);
432            assert_eq!(with_missing.choose(BG_PRE_CALL), direct.choose(BG_PRE_CALL));
433            assert_eq!(
434                with_missing.choose(BG_HARVEST_SKIP),
435                direct.choose(BG_HARVEST_SKIP)
436            );
437        }
438    }

440    #[test]
441    fn a_hold_needs_a_hold_directory_and_names_its_release_file() {
442        assert_eq!(
443            parse_item("fifo_staging_mapped=hold"),
444            Ok((
445                FIFO_STAGING_MAPPED.to_owned(),
446                Spec {
447                    lo_ms: 0,
448                    hi_ms: 0,
449                    probability: 1.0,
450                    hold: true,
451                }
452            ))
453        );
454        for bad in ["fifo_staging_mapped=hold@0.5", "fifo_staging_mapped=holds"] {
455            assert!(parse_item(bad).is_err(), "{bad} parsed");
456        }
457        // Without a directory the hold is skipped, and the other items stay.
458        let probes = Probes::from_settings(
459            Some("fifo_staging_mapped=hold,bg_harvest_skip=0"),
460            Some("1"),
461            None,
462        )
463        .unwrap();
464        assert!(!probes.points.contains_key(FIFO_STAGING_MAPPED));
465        assert!(Probes::from_settings(Some("fifo_staging_mapped=hold"), None, Some("")).is_none());
466        // With one, every visit holds on the point's own file, and draws no
467        // visit from the seeded sequence of the other points.
468        let config = Some("fifo_staging_mapped=hold,fifo_staging_reclaim=hold,bg_pre_call=1-9@0.5");
469        let holding = Probes::from_settings(config, Some("5"), Some("/hold")).unwrap();
470        let direct = Probes::from_settings(Some("bg_pre_call=1-9@0.5"), Some("5"), None).unwrap();
471        for _ in 0..100 {
472            assert_eq!(
473                holding.choose(FIFO_STAGING_MAPPED),
474                Some(Perturbation::Hold(PathBuf::from(
475                    "/hold/fifo_staging_mapped"
476                )))
477            );
478            assert_eq!(
479                holding.choose(FIFO_STAGING_RECLAIM),
480                Some(Perturbation::Hold(PathBuf::from(
481                    "/hold/fifo_staging_reclaim"
482                )))
483            );
484            assert_eq!(holding.choose(BG_PRE_CALL), direct.choose(BG_PRE_CALL));
485        }
486    }

488    /// A held visit returns only once its release file exists, and at once
489    /// when it already does.
490    #[tokio::test(start_paused = true)]
491    async fn a_held_visit_waits_for_its_release_file() {
492        let dir = tempfile::tempdir().unwrap();
493        let release = dir.path().join(FIFO_STAGING_MAPPED);
494        let mut held = Box::pin(hold(FIFO_STAGING_MAPPED, &release));
495        // A minute of (paused, auto-advancing) time does not release it.
496        tokio::select! {
497            _ = &mut held => panic!("released before its file existed"),
498            _ = tokio::time::sleep(Duration::from_secs(60)) => {}
499        }
500        std::fs::write(&release, b"").unwrap();
501        tokio::time::timeout(Duration::from_secs(1), held)
502            .await
503            .expect("released once its file exists");
504        tokio::time::timeout(Duration::ZERO, hold(FIFO_STAGING_MAPPED, &release))
505            .await
506            .expect("an existing file releases at once");
507    }
