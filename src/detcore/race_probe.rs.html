9//! Race probes: env-gated perturbation points on Hermit's own host-timed
10//! edges, for reproducing rare races in Hermit itself (docs/RACE_PROBES.md).
11//!
12//! Off unless [`RACE_PROBE_ENV`] is set to a non-empty list of
13//! `point=LO[-HI][@P]` items. At `point`, with probability `P` (default 1), a
14//! visit is perturbed: an async point waits a uniform `LO..=HI` milliseconds,
15//! and a synchronous point takes its perturbed branch (its numbers ignored).
16//! [`RACE_PROBE_SEED_ENV`] fixes the choices; without it a seed is drawn from
17//! the host clock. Either way the configuration and seed are printed once to
18//! stderr, so a failing run names its seed.
19//!
20//! Unset, every point is one read of an initialized `OnceLock` and a branch;
21//! nothing is logged, and no tracing event is emitted.
22//!
23//! A probe only moves host timing that Hermit already treats as
24//! nondeterministic input (when a backgrounded call returns, and when the
25//! scheduler samples that it has), so an outcome reached under a probe is one
26//! host timing alone could produce. Decisions are drawn in visit order from one
27//! process-wide counter, so a seed replays exactly only when the number of
28//! visits before the deciding one does not itself depend on host timing.
29
30use std::collections::BTreeMap;
31use std::sync::OnceLock;
32use std::sync::atomic::AtomicU64;
33use std::sync::atomic::Ordering;
34use std::time::Duration;
35
36/// The variable that enables race probes and lists their settings.
37pub const RACE_PROBE_ENV: &str = "HERMIT_RACE_PROBE";
38
39/// The variable that fixes the seed of the race probes' choices.
40pub const RACE_PROBE_SEED_ENV: &str = "HERMIT_RACE_PROBE_SEED";
41
42/// An async wait just before a backgrounded blocking call runs.
43pub(crate) const BG_PRE_CALL: &str = "bg_pre_call";
44/// An async wait after a backgrounded call returns, before it rejoins.
45pub(crate) const BG_POST_CALL: &str = "bg_post_call";
46/// The scheduler treats a finished backgrounded call as still running on one
47/// of its checks.
48pub(crate) const BG_HARVEST_SKIP: &str = "bg_harvest_skip";
49/// An async wait after a FIFO open's staging page is mapped, before anything
50/// is written to it: a task killed by a host process then leaves the page
51/// mapped and empty, as one killed during the `mmap` does.
52pub(crate) const FIFO_STAGING_MAPPED: &str = "fifo_staging_mapped";
53/// An async wait before a task unmaps a staging page that a killed task left:
54/// a task killed by a host process then leaves the page to the next task.
55pub(crate) const FIFO_STAGING_RECLAIM: &str = "fifo_staging_reclaim";
56
57/// Every probe point, for validating a configuration.
58const POINTS: [&str; 5] = [
59    BG_PRE_CALL,
60    BG_POST_CALL,
61    BG_HARVEST_SKIP,
62    FIFO_STAGING_MAPPED,
63    FIFO_STAGING_RECLAIM,
64];
65
66#[derive(Clone, Copy, Debug, PartialEq)]
67struct Spec {
68    lo_ms: u64,
69    hi_ms: u64,
70    probability: f64,
71}
72
73#[derive(Debug)]
74struct Probes {
75    seed: u64,
76    counter: AtomicU64,
77    points: BTreeMap<String, Spec>,
78}
79
80fn parse_item(item: &str) -> Result<(String, Spec), String> {
81    let (name, rest) = item
82        .split_once('=')
83        .ok_or_else(|| format!("{item:?} is not point=LO[-HI][@P]"))?;
84    if !POINTS.contains(&name) {
85        return Err(format!("unknown point {name:?}; points are {POINTS:?}"));
86    }
87    let (range, probability) = match rest.split_once('@') {
88        Some((range, p)) => (
89            range,
90            p.parse::<f64>()
91                .ok()
92                .filter(|p| (0.0..=1.0).contains(p))
93                .ok_or_else(|| format!("{item:?}: probability must be in [0, 1]"))?,
94        ),
95        None => (rest, 1.0),
96    };
97    let number = |text: &str| {
98        text.parse::<u64>()
99            .map_err(|_| format!("{item:?}: {text:?} is not a number of milliseconds"))
100    };
101    let (lo_ms, hi_ms) = match range.split_once('-') {
102        Some((lo, hi)) => (number(lo)?, number(hi)?),
103        None => {
104            let ms = number(range)?;
105            (ms, ms)
106        }
107    };
108    if hi_ms < lo_ms {
109        return Err(format!("{item:?}: the range is empty"));
110    }
111    Ok((
112        name.to_owned(),
113        Spec {
114            lo_ms,
115            hi_ms,
116            probability,
117        },
118    ))
119}

121impl Probes {
122    /// The probes `config` and `seed` (the two variables' values) describe:
123    /// `None` when probes are off. A malformed item is reported and skipped.
124    fn from_settings(config: Option<&str>, seed: Option<&str>) -> Option<Self> {
125        let config = config?;
126        // A disabled probe must not create randomized hash state: even an
127        // empty HashMap initializes RandomState's thread-local random keys.
128        // These few exact names need no hashing, and insert still makes the
129        // last valid duplicate win without changing lookup or draw order.
130        let mut points = BTreeMap::new();
131        for item in config.split(',').map(str::trim).filter(|s| !s.is_empty()) {
132            match parse_item(item) {
133                Ok((name, spec)) => {
134                    points.insert(name, spec);
135                }
136                Err(error) => eprintln!("race_probe: ignoring {error}"),
137            }
138        }
139        if points.is_empty() {
140            return None;
141        }
142        let seed = match seed.map(str::parse::<u64>) {
143            Some(Ok(seed)) => seed,
144            _ => {
145                std::time::SystemTime::now()
146                    .duration_since(std::time::UNIX_EPOCH)
147                    .map_or(0, |d| d.as_nanos() as u64)
148                    ^ u64::from(std::process::id())
149            }
150        };
151        Some(Probes {
152            seed,
153            counter: AtomicU64::new(0),
154            points,
155        })
156    }
157
158    /// The decision at `point` for this visit: `None` when the point is not
159    /// configured or this visit is not perturbed, otherwise the wait.
160    fn choose(&self, point: &str) -> Option<Duration> {
161        let spec = self.points.get(point)?;
162        let visit = self.counter.fetch_add(1, Ordering::Relaxed);
163        let r = splitmix(self.seed ^ splitmix(visit));
164        let unit = (r >> 11) as f64 / (1u64 << 53) as f64;
165        if unit >= spec.probability {
166            return None;
167        }
168        let span = spec.hi_ms - spec.lo_ms;
169        let ms = spec.lo_ms
170            + if span == 0 {
171                0
172            } else {
173                splitmix(r) % (span + 1)
174            };
175        Some(Duration::from_millis(ms))
176    }
177}
178
179fn probes() -> Option<&'static Probes> {
180    static PROBES: OnceLock<Option<Probes>> = OnceLock::new();
181    PROBES
182        .get_or_init(|| {
183            let config = std::env::var(RACE_PROBE_ENV).ok()?;
184            let seed = std::env::var(RACE_PROBE_SEED_ENV).ok();
185            let probes = Probes::from_settings(Some(&config), seed.as_deref())?;
186            eprintln!(
187                "race_probe: {RACE_PROBE_ENV}={config} {RACE_PROBE_SEED_ENV}={}",
188                probes.seed
189            );
190            Some(probes)
191        })
192        .as_ref()
193}
194
195fn splitmix(mut z: u64) -> u64 {
196    z = z.wrapping_add(0x9e37_79b9_7f4a_7c15);
197    z = (z ^ (z >> 30)).wrapping_mul(0xbf58_476d_1ce4_e5b9);
198    z = (z ^ (z >> 27)).wrapping_mul(0x94d0_49bb_1331_11eb);
199    z ^ (z >> 31)
200}
201
202fn choose(point: &str) -> Option<Duration> {
203    let wait = probes()?.choose(point)?;
204    tracing::debug!(target: "race_probe", "race_probe: {point} perturbed ({wait:?})");
205    Some(wait)
206}
207
208/// An async wait at `point`, when configured.
209pub(crate) async fn delay(point: &str) {
210    if let Some(wait) = choose(point)
211        && !wait.is_zero()
212    {
213        tokio::time::sleep(wait).await;
214    }
215}
216
217/// Whether a synchronous point takes its perturbed branch on this visit.
218pub(crate) fn perturb(point: &str) -> bool {
219    choose(point).is_some()
220}

226    #[test]
227    fn race_probes_are_off_unless_configured() {
228        // Unset, empty, or naming no valid point: no probes, so every point
229        // returns before drawing a decision.
230        assert!(Probes::from_settings(None, None).is_none());
231        assert!(Probes::from_settings(None, Some("7")).is_none());
232        assert!(Probes::from_settings(Some(""), Some("7")).is_none());
233        assert!(Probes::from_settings(Some(" , "), None).is_none());
234        assert!(Probes::from_settings(Some("no_such_point=5"), None).is_none());
235        // This test process does not set the variable, so the process-wide
236        // probes are off and no point perturbs.
237        assert!(std::env::var_os(RACE_PROBE_ENV).is_none());
238        assert!(probes().is_none());
239        for point in POINTS {
240            assert!(!perturb(point));
241            assert_eq!(choose(point), None);
242        }
243    }

245    #[test]
246    fn race_probe_settings_parse_as_documented() {
247        assert_eq!(
248            parse_item("bg_pre_call=300"),
249            Ok((
250                BG_PRE_CALL.to_owned(),
251                Spec {
252                    lo_ms: 300,
253                    hi_ms: 300,
254                    probability: 1.0
255                }
256            ))
257        );
258        assert_eq!(
259            parse_item("bg_post_call=0-50@0.25"),
260            Ok((
261                BG_POST_CALL.to_owned(),
262                Spec {
263                    lo_ms: 0,
264                    hi_ms: 50,
265                    probability: 0.25
266                }
267            ))
268        );
269        for bad in [
270            "bg_pre_call",
271            "bg_pre_call=x",
272            "bg_pre_call=5-2",
273            "bg_pre_call=1@2",
274            "bg_harvest_skip=0@-0.1",
275            "unknown=1",
276        ] {
277            assert!(parse_item(bad).is_err(), "{bad} parsed");
278        }
279        // A bad item is skipped, the good ones stay.
280        let probes =
281            Probes::from_settings(Some("bg_pre_call=x, bg_harvest_skip=0@0.5"), Some("1")).unwrap();
282        assert_eq!(probes.points.len(), 1);
283        assert!(probes.points.contains_key(BG_HARVEST_SKIP));
284
285        // A later valid duplicate replaces the earlier setting. A malformed
286        // duplicate is ignored and must not erase that last valid value.
287        let probes = Probes::from_settings(
288            Some("bg_pre_call=10@0,bg_pre_call=20@1,bg_pre_call=x,bg_harvest_skip=0@0.5"),
289            Some("7"),
290        )
291        .unwrap();
292        assert_eq!(probes.points.len(), 2);
293        assert_eq!(
294            probes.points.get(BG_PRE_CALL),
295            Some(&Spec {
296                lo_ms: 20,
297                hi_ms: 20,
298                probability: 1.0,
299            })
300        );
301        assert_eq!(probes.choose(BG_PRE_CALL), Some(Duration::from_millis(20)));
302    }

304    #[test]
305    fn race_probe_choices_follow_probability_range_and_seed() {
306        let draw = |config: &str, seed: &str, point: &str| {
307            let probes = Probes::from_settings(Some(config), Some(seed)).unwrap();
308            (0..1000).map(|_| probes.choose(point)).collect::<Vec<_>>()
309        };
310        assert!(
311            draw("bg_harvest_skip=0@0", "1", BG_HARVEST_SKIP)
312                .iter()
313                .all(Option::is_none)
314        );
315        assert!(
316            draw("bg_harvest_skip=0@1", "1", BG_HARVEST_SKIP)
317                .iter()
318                .all(|wait| *wait == Some(Duration::ZERO))
319        );
320        // A point that is not configured never perturbs.
321        assert!(
322            draw("bg_harvest_skip=0@1", "1", BG_PRE_CALL)
323                .iter()
324                .all(Option::is_none)
325        );
326        let half = draw("bg_pre_call=10-20@0.5", "3", BG_PRE_CALL);
327        let taken = half.iter().flatten().count();
328        assert!((400..600).contains(&taken), "{taken} of 1000 at 0.5");
329        assert!(half.iter().flatten().all(|wait| {
330            (Duration::from_millis(10)..=Duration::from_millis(20)).contains(wait)
331        }));
332        // The same seed draws the same sequence; another seed does not.
333        assert_eq!(half, draw("bg_pre_call=10-20@0.5", "3", BG_PRE_CALL));
334        assert_ne!(half, draw("bg_pre_call=10-20@0.5", "4", BG_PRE_CALL));
335
336        // Missing-point lookups must not consume a visit before the actual
337        // seeded draws, including when configured points are interleaved.
338        let config = Some("bg_pre_call=10-20@0.5,bg_harvest_skip=0@0.5");
339        let with_missing = Probes::from_settings(config, Some("3")).unwrap();
340        let direct = Probes::from_settings(config, Some("3")).unwrap();
341        for _ in 0..1000 {
342            assert_eq!(with_missing.choose("no_such_point"), None);
343            assert_eq!(with_missing.choose(BG_POST_CALL), None);
344            assert_eq!(with_missing.choose(BG_PRE_CALL), direct.choose(BG_PRE_CALL));
345            assert_eq!(
346                with_missing.choose(BG_HARVEST_SKIP),
347                direct.choose(BG_HARVEST_SKIP)
348            );
349        }
350    }
