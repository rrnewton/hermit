9//! Race probes: env-gated perturbation points on Hermit's own host-timed
10//! edges, for reproducing rare races in Hermit itself (docs/RACE_PROBES.md).
11//!
12//! Off unless [`RACE_PROBE_ENV`] is set to a non-empty list of
13//! `point=LO[-HI][@P]` items. At `point`, with probability `P` (default 1), a
14//! visit is perturbed: an async point waits a uniform `LO..=HI` milliseconds,
15//! and a synchronous point takes its perturbed branch (its numbers ignored).
16//! [`RACE_PROBE_SEED_ENV`] fixes the choices; without it a seed is drawn from
17//! the host clock. Either way the configuration and seed are printed once to
18//! stderr, so a failing run names its seed.
19//!
20//! `point=hold` makes every visit to an async point wait, without a time
21//! limit, until a file named after the point exists in
22//! [`RACE_PROBE_HOLD_DIR_ENV`]: a test can then act on a task while it is
23//! known to be at the point, and release it after. A held visit prints one
24//! line to stderr naming the file it waits for. `point=hold-each` holds
25//! every visit the same way and removes the file as it lets the visit go,
26//! so that each visit waits for a release of its own.
27//!
28//! Unset, every point is one read of an initialized `OnceLock` and a branch;
29//! nothing is logged, and no tracing event is emitted.
30//!
31//! A probe only moves host timing that Hermit already treats as
32//! nondeterministic input (when a backgrounded call returns, and when the
33//! scheduler samples that it has), so an outcome reached under a probe is one
34//! host timing alone could produce. Decisions are drawn in visit order from one
35//! process-wide counter, so a seed replays exactly only when the number of
36//! visits before the deciding one does not itself depend on host timing.
37

115    BG_HARVEST_SKIP,

124#[derive(Clone, Copy, Debug, PartialEq)]
125struct Spec {
126    lo_ms: u64,
127    hi_ms: u64,
128    probability: f64,
129    /// Every visit waits for the point's release file (`point=hold`).
130    hold: bool,
131    /// The release is removed as each visit goes on (`point=hold-each`).
132    each: bool,
133}
134
135#[derive(Debug)]
136struct Probes {
137    seed: u64,
138    counter: AtomicU64,
139    points: BTreeMap<String, Spec>,
140    hold_dir: Option<PathBuf>,
141}
142
143/// What a perturbed visit does.
144#[derive(Clone, Debug, PartialEq)]
145enum Perturbation {
146    /// An async point waits this long; a synchronous point takes its branch.
147    Wait(Duration),
148    /// An async point waits until this file exists.
149    Hold(PathBuf),
150    /// As [`Self::Hold`], removing the file as the visit goes on.
151    HoldEach(PathBuf),
152}
153
154fn parse_item(item: &str) -> Result<(String, Spec), String> {
155    let (name, rest) = item
156        .split_once('=')
157        .ok_or_else(|| format!("{item:?} is not point=LO[-HI][@P]"))?;
158    if !POINTS.contains(&name) {
159        return Err(format!("unknown point {name:?}; points are {POINTS:?}"));
160    }
161    if rest == "hold" || rest == "hold-each" {
162        return Ok((
163            name.to_owned(),
164            Spec {
165                lo_ms: 0,
166                hi_ms: 0,
167                probability: 1.0,
168                hold: true,
169                each: rest == "hold-each",
170            },
171        ));
172    }
173    let (range, probability) = match rest.split_once('@') {
174        Some((range, p)) => (
175            range,
176            p.parse::<f64>()
177                .ok()
178                .filter(|p| (0.0..=1.0).contains(p))
179                .ok_or_else(|| format!("{item:?}: probability must be in [0, 1]"))?,
180        ),
181        None => (rest, 1.0),
182    };
183    let number = |text: &str| {
184        text.parse::<u64>()
185            .map_err(|_| format!("{item:?}: {text:?} is not a number of milliseconds"))
186    };
187    let (lo_ms, hi_ms) = match range.split_once('-') {
188        Some((lo, hi)) => (number(lo)?, number(hi)?),
189        None => {
190            let ms = number(range)?;
191            (ms, ms)
192        }
193    };
194    if hi_ms < lo_ms {
195        return Err(format!("{item:?}: the range is empty"));
196    }
197    Ok((
198        name.to_owned(),
199        Spec {
200            lo_ms,
201            hi_ms,
202            probability,
203            hold: false,
204            each: false,
205        },
206    ))
207}

209impl Probes {
210    /// The probes `config`, `seed` and `hold_dir` (the three variables'
211    /// values) describe: `None` when probes are off. A malformed item is
212    /// reported and skipped, as is a hold without a hold directory.
213    fn from_settings(
214        config: Option<&str>,
215        seed: Option<&str>,
216        hold_dir: Option<&Path>,
217    ) -> Option<Self> {
218        let config = config?;
219        let hold_dir = hold_dir
220            .filter(|dir| !dir.as_os_str().is_empty())
221            .map(Path::to_path_buf);
222        // A disabled probe must not create randomized hash state: even an
223        // empty HashMap initializes RandomState's thread-local random keys.
224        // These few exact names need no hashing, and insert still makes the
225        // last valid duplicate win without changing lookup or draw order.
226        let mut points = BTreeMap::new();
227        for item in config.split(',').map(str::trim).filter(|s| !s.is_empty()) {
228            match parse_item(item) {
229                Ok((_, spec)) if spec.hold && hold_dir.is_none() => {
230                    eprintln!(
231                        "race_probe: ignoring {item:?}: {RACE_PROBE_HOLD_DIR_ENV} is not set"
232                    );
233                }
234                Ok((name, spec)) => {
235                    points.insert(name, spec);
236                }
237                Err(error) => eprintln!("race_probe: ignoring {error}"),
238            }
239        }
240        if points.is_empty() {
241            return None;
242        }
243        let seed = match seed.map(str::parse::<u64>) {
244            Some(Ok(seed)) => seed,
245            _ => {
246                std::time::SystemTime::now()
247                    .duration_since(std::time::UNIX_EPOCH)
248                    .map_or(0, |d| d.as_nanos() as u64)
249                    ^ u64::from(std::process::id())
250            }
251        };
252        Some(Probes {
253            seed,
254            counter: AtomicU64::new(0),
255            points,
256            hold_dir,
257        })
258    }
259
260    /// The decision at `point` for this visit: `None` when the point is not
261    /// configured or this visit is not perturbed, otherwise what it does.
262    fn choose(&self, point: &str) -> Option<Perturbation> {
263        let spec = self.points.get(point)?;
264        if spec.hold {
265            // Configured only with a hold directory (`from_settings`).
266            let release = self.hold_dir.as_ref()?.join(point);
267            return Some(if spec.each {
268                Perturbation::HoldEach(release)
269            } else {
270                Perturbation::Hold(release)
271            });
272        }
273        let visit = self.counter.fetch_add(1, Ordering::Relaxed);
274        let r = splitmix64(self.seed ^ splitmix64(visit));
275        let unit = (r >> 11) as f64 / (1u64 << 53) as f64;
276        if unit >= spec.probability {
277            return None;
278        }
279        let span = spec.hi_ms - spec.lo_ms;
280        // One of `span + 1` waits, or of every u64 when the range is all of it.
281        let offset = match span.checked_add(1) {
282            Some(choices) => splitmix64(r) % choices,
283            None => splitmix64(r),
284        };
285        Some(Perturbation::Wait(Duration::from_millis(
286            spec.lo_ms + offset,
287        )))
288    }
289}

291fn probes() -> Option<&'static Probes> {
292    static PROBES: OnceLock<Option<Probes>> = OnceLock::new();
293    PROBES
294        .get_or_init(|| {
295            let config = std::env::var(RACE_PROBE_ENV).ok()?;
296            let seed = std::env::var(RACE_PROBE_SEED_ENV).ok();
297            // The directory's bytes as the environment gives them.
298            let hold_dir = std::env::var_os(RACE_PROBE_HOLD_DIR_ENV);
299            let probes = Probes::from_settings(
300                Some(&config),
301                seed.as_deref(),
302                hold_dir.as_deref().map(Path::new),
303            )?;
304            eprintln!(
305                "race_probe: {RACE_PROBE_ENV}={config} {RACE_PROBE_SEED_ENV}={}",
306                probes.seed
307            );
308            Some(probes)
309        })
310        .as_ref()
311}
312
313fn choose(point: &str) -> Option<Perturbation> {
314    let perturbation = probes()?.choose(point)?;
315    tracing::debug!(target: "race_probe", "race_probe: {point} perturbed ({perturbation:?})");
316    Some(perturbation)
317}

319/// An async wait at `point`, when configured.
320pub(crate) async fn delay(point: &str) {
321    match choose(point) {
322        Some(Perturbation::Wait(wait)) if !wait.is_zero() => tokio::time::sleep(wait).await,
323        Some(Perturbation::Hold(release)) => hold(point, &release, false).await,
324        Some(Perturbation::HoldEach(release)) => hold(point, &release, true).await,
325        _ => {}
326    }
327}

329/// A wait at `point`, when configured, in Detcore code that every backend
330/// runs ([`EXIT_HOOK_TAIL`], [`RECEIPT_DECISION`]). Where the tool runs
331/// inside the guest's process (`in_guest`: the in-guest backends), whose
332/// threads must touch no thread-local state a runtime keeps, it blocks the
333/// calling thread and touches none: the disabled path is only [`choose`]'s
334/// read of an initialized `OnceLock`. Elsewhere it waits on Tokio's timer
335/// where the calling task runs on a Tokio runtime (the ptrace tracer, whose
336/// one thread serves every guest task), so that only that task waits, and
337/// blocks the calling thread where none runs (KVM drives each guest
338/// process's and thread's tool on a thread of its own), which then serves
339/// that task alone.
340#[cfg(debug_assertions)]
341pub(crate) async fn delay_on_any_backend(point: &str, in_guest: bool) {
342    let Some(perturbation) = choose(point) else {
343        return;
344    };
345    let on_tokio = !in_guest && tokio::runtime::Handle::try_current().is_ok();
346    match perturbation {
347        Perturbation::Wait(wait) if wait.is_zero() => {}
348        Perturbation::Wait(wait) if on_tokio => tokio::time::sleep(wait).await,
349        Perturbation::Wait(wait) => std::thread::sleep(wait),
350        Perturbation::Hold(release) if on_tokio => hold(point, &release, false).await,
351        Perturbation::Hold(release) => hold_blocking(point, &release, false, None),
352        Perturbation::HoldEach(release) if on_tokio => hold(point, &release, true).await,
353        Perturbation::HoldEach(release) => hold_blocking(point, &release, true, None),
354    }
355}

357/// Waits until the visit is released ([`released`]), saying so on stderr
358/// unless it is at once.
359async fn hold(point: &str, release: &Path, each: bool) {
360    if released(point, release, each) {
361        return;
362    }
363    eprintln!(
364        "race_probe: {point} holding until {} exists",
365        release.display()
366    );
367    while !released(point, release, each) {
368        tokio::time::sleep(HOLD_POLL).await;
369    }
370}

372/// Whether a held visit may go on: its release file exists (`hold`), or, for
373/// `hold-each`, this visit consumed it by renaming it away. Of two visits that
374/// try to consume one release, one rename succeeds and the other finds the
375/// release gone and goes on waiting, so two visits can never both go on for
376/// one release. (The name it is renamed to need not be unique: in a PID
377/// namespace two processes can share a process ID, and the later rename
378/// simply replaces the earlier consumed file.) A consumption that fails for
379/// any reason but the release being absent fails the visit, naming the point.
380fn released(point: &str, release: &Path, each: bool) -> bool {
381    if !each {
382        return release.exists();
383    }
384    static CONSUMED: AtomicU64 = AtomicU64::new(0);
385    let consumed = release.with_file_name(format!(
386        "{point}.consumed.{}.{}",
387        std::process::id(),
388        CONSUMED.fetch_add(1, Ordering::Relaxed)
389    ));
390    match std::fs::rename(release, &consumed) {
391        Ok(()) => true,
392        Err(error) if error.kind() == std::io::ErrorKind::NotFound => false,
393        Err(error) => panic!(
394            "race_probe: {point} could not consume {}: {error}",
395            release.display()
396        ),
397    }
398}

400/// A hold at `point`, when configured, in Hermit's launcher or a container's
401/// init, outside any guest ([`SETUP_MOUNTS_MADE`]): it blocks the calling
402/// thread. A hold here parks the run itself, so the point is armed only where
403/// [`RACE_PROBE_HOLD_DIR_ENV`] names a directory holding
404/// [`LAUNCHER_HOLD_MARKER`], which only a test makes, and that is decided
405/// first, from that one variable and that one file: an unarmed point reads,
406/// parses and draws nothing else, whatever [`RACE_PROBE_ENV`] says. Armed,
407/// nothing is ignored: `armed_hold` must find a complete, valid hold for the
408/// point, or the visit fails, naming the point and why. The hold fails,
409/// naming the point, if it is not released within `LAUNCHER_HOLD_LIMIT`.
410pub fn delay_in_launcher(point: &str) {
411    launcher_visit(
412        launcher_hold_dir(),
413        || std::env::var_os(RACE_PROBE_ENV),
414        point,
415    );
416}

418/// The hold directory that arms launcher points in this process
419/// ([`armed_by`]), or `None` when they are unarmed. Decided once.
420fn launcher_hold_dir() -> Option<&'static Path> {
421    static ARMED: OnceLock<Option<PathBuf>> = OnceLock::new();
422    ARMED
423        .get_or_init(|| {
424            std::env::var_os(RACE_PROBE_HOLD_DIR_ENV)
425                .map(PathBuf::from)
426                .filter(|dir| armed_by(Some(dir)))
427        })
428        .as_deref()
429}
430
431/// Whether `hold_dir` holds [`LAUNCHER_HOLD_MARKER`], which arms
432/// [`delay_in_launcher`].
433fn armed_by(hold_dir: Option<&Path>) -> bool {
434    hold_dir
435        .is_some_and(|dir| !dir.as_os_str().is_empty() && dir.join(LAUNCHER_HOLD_MARKER).exists())
436}

438/// A [`delay_in_launcher`] visit, armed by `hold_dir`: unless armed, it
439/// returns before `config` is read. Armed, the configuration's hold for the
440/// point ([`armed_hold`]) holds the visit, and anything else fails it.
441fn launcher_visit(
442    hold_dir: Option<&Path>,
443    config: impl FnOnce() -> Option<std::ffi::OsString>,
444    point: &str,
445) {
446    let Some(hold_dir) = hold_dir else {
447        return;
448    };
449    let config = config();
450    let hold = match config.as_deref().map(std::ffi::OsStr::to_str) {
451        Some(None) => Err(format!("{RACE_PROBE_ENV} is not UTF-8")),
452        config => armed_hold(config.flatten(), hold_dir, point),
453    };
454    match hold {
455        Ok(hold) => launcher_hold(point, hold),
456        Err(reason) => {
457            panic!("race_probe: {point} is armed, and {RACE_PROBE_ENV} is refused there: {reason}")
458        }
459    }
460}

462/// The hold an armed launcher point takes under `config`, its release in
463/// `hold_dir`, or why there is none. Nothing is ignored: every item must
464/// parse and name a known point, and the point's own must be `hold` or
465/// `hold-each`; a wait, an item that does not parse, an unknown point, or no
466/// item for the point at all is refused.
467fn armed_hold(config: Option<&str>, hold_dir: &Path, point: &str) -> Result<Perturbation, String> {
468    let config = config.ok_or_else(|| format!("{RACE_PROBE_ENV} is not set"))?;
469    let mut each = None;
470    for item in config.split(',').map(str::trim).filter(|s| !s.is_empty()) {
471        let (name, spec) = parse_item(item)?;
472        if name == point {
473            if !spec.hold {
474                return Err(format!("{item:?} is not a hold (hold or hold-each)"));
475            }
476            each = Some(spec.each);
477        }
478    }
479    let each = each.ok_or_else(|| format!("{config:?} names no hold for {point}"))?;
480    let release = hold_dir.join(point);
481    Ok(if each {
482        Perturbation::HoldEach(release)
483    } else {
484        Perturbation::Hold(release)
485    })
486}
487
488/// An armed launcher hold, failing after [`LAUNCHER_HOLD_LIMIT`].
489fn launcher_hold(point: &str, hold: Perturbation) {
490    match hold {
491        Perturbation::Hold(release) => {
492            hold_blocking(point, &release, false, Some(LAUNCHER_HOLD_LIMIT))
493        }
494        Perturbation::HoldEach(release) => {
495            hold_blocking(point, &release, true, Some(LAUNCHER_HOLD_LIMIT))
496        }
497        Perturbation::Wait(_) => unreachable!("armed_hold returns holds only"),
498    }
499}
500
501/// [`hold`], blocking the calling thread, and failing, naming the point,
502/// after `limit` if one is given.
503fn hold_blocking(point: &str, release: &Path, each: bool, limit: Option<Duration>) {
504    if released(point, release, each) {
505        return;
506    }
507    eprintln!(
508        "race_probe: {point} holding until {} exists",
509        release.display()
510    );
511    let start = std::time::Instant::now();
512    while !released(point, release, each) {
513        if let Some(limit) = limit
514            && start.elapsed() >= limit
515        {
516            panic!(
517                "race_probe: {point} was not released within {limit:?}: {} never appeared",
518                release.display()
519            );
520        }
521        std::thread::sleep(HOLD_POLL);
522    }
523}

525/// Whether a synchronous point takes its perturbed branch on this visit.
526pub(crate) fn perturb(point: &str) -> bool {
527    choose(point).is_some()
528}

626    #[test]
627    fn race_probe_choices_follow_probability_range_and_seed() {
628        let draw = |config: &str, seed: &str, point: &str| {
629            let probes = Probes::from_settings(Some(config), Some(seed), None).unwrap();
630            (0..1000).map(|_| probes.choose(point)).collect::<Vec<_>>()
631        };
632        assert!(
633            draw("bg_harvest_skip=0@0", "1", BG_HARVEST_SKIP)
634                .iter()
635                .all(Option::is_none)
636        );
637        assert!(
638            draw("bg_harvest_skip=0@1", "1", BG_HARVEST_SKIP)
639                .iter()
640                .all(|wait| *wait == Some(Perturbation::Wait(Duration::ZERO)))
641        );
642        // A point that is not configured never perturbs.
643        assert!(
644            draw("bg_harvest_skip=0@1", "1", BG_PRE_CALL)
645                .iter()
646                .all(Option::is_none)
647        );
648        let half = draw("bg_pre_call=10-20@0.5", "3", BG_PRE_CALL);
649        let taken = half.iter().flatten().count();
650        assert!((400..600).contains(&taken), "{taken} of 1000 at 0.5");
651        assert!(half.iter().flatten().all(|wait| {
652            matches!(wait, Perturbation::Wait(wait)
653                if (Duration::from_millis(10)..=Duration::from_millis(20)).contains(wait))
654        }));
655        // The same seed draws the same sequence; another seed does not.
656        assert_eq!(half, draw("bg_pre_call=10-20@0.5", "3", BG_PRE_CALL));
657        assert_ne!(half, draw("bg_pre_call=10-20@0.5", "4", BG_PRE_CALL));
658
659        // Missing-point lookups must not consume a visit before the actual
660        // seeded draws, including when configured points are interleaved.
661        let config = Some("bg_pre_call=10-20@0.5,bg_harvest_skip=0@0.5");
662        let with_missing = Probes::from_settings(config, Some("3"), None).unwrap();
663        let direct = Probes::from_settings(config, Some("3"), None).unwrap();
664        for _ in 0..1000 {
665            assert_eq!(with_missing.choose("no_such_point"), None);
666            assert_eq!(with_missing.choose(BG_POST_CALL), None);
667            assert_eq!(with_missing.choose(BG_PRE_CALL), direct.choose(BG_PRE_CALL));
668            assert_eq!(
669                with_missing.choose(BG_HARVEST_SKIP),
670                direct.choose(BG_HARVEST_SKIP)
671            );
672        }
673    }
