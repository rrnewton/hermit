9//! The post-crash timeslice switch (`Config::post_crash_timeslice`).
10//!
11//! A race that needs fine slices (`--max-timeslice` of 1000 to 10000) often
12//! reaches its crash in seconds and then spends minutes in the crash path:
13//! Android's debuggerd handler starts `crash_dump`, which ptrace-attaches every
14//! thread of the crashed process, reads its registers and memory and unwinds
15//! it, and every branch of that is sliced. Once a guest process has crashed,
16//! the switch raises the slice bound of that process and of every process it
17//! goes on to create to the configured post-crash bound, for the rest of the
18//! run. Other processes (a harness that runs trials as children, a shell)
19//! keep their bound.
20//!
21//! A crash is a fatal signal on its way to ending a process
22//! ([`is_crash_signal`]). Every step below happens at a point the schedule
23//! orders, so it is taken at the same point of every run and replay that
24//! reaches it:
25//! - a crash signal's delivery that takes the default action switches at
26//!   once, for the crashed process;
27//! - one the guest catches arms the switch for that thread. The processes
28//!   that thread's process creates while it is armed (debuggerd's
29//!   pseudothread, and `crash_dump` before its exec) belong to its crash;
30//! - a successful `execve` by a thread of such a process takes the switch for
31//!   an armed crash of its own lineage whose handler is still running:
32//!   Android's crash handler starts `crash_dump` that way, from inside the
33//!   handler. Crashes in unrelated processes keep separate lineages, and each
34//!   crash in a process not yet switched takes a switch of its own. A switch
35//!   applies to its whole process: every crash armed in it switches with it,
36//!   and those crashes' lineages take the bound too;
37//! - an armed thread's handler has returned once the kernel no longer blocks
38//!   its crash signal for it (`signal_blocked`). That is read only where the
39//!   armed thread itself is stopped at a point of its own: each syscall entry,
40//!   before the call runs; each syscall exit or trap, after it completes and
41//!   before the thread can give up its turn there; each end of one of its
42//!   slices; each later signal delivery to it, before the kernel installs that
43//!   handler's mask. A thread gives up its turn only at such a point, and with
44//!   threads serialized no other thread runs in between, so the state known
45//!   at another thread's `execve` is the armed thread's own. A handler that resumes the faulting
46//!   code (ART's and JVMs' implicit null checks) is disarmed at its thread's
47//!   next stop, and never switches.
48//!
49//! The scheduler keeps the [`PostCrashState`]; a thread of a crashed process
50//! learns the bound from the answer to its own step and from every later reply
51//! that resumes it from a grant (`GlobalResponse::WithPostCrashBound`), so it
52//! uses the bound from its first slice after the switch.

69/// One step of the switch, sent by a guest thread
70/// (`GlobalRequest::PostCrash`).
71#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
72pub enum PostCrashEvent {
73    /// A crash signal is being delivered to the sender.
74    Delivered {
75        /// The signal's number.
76        signal: i32,
77        /// Whether the delivery takes the default action (no handler).
78        default_action: bool,
79    },
80    /// The sender's crash handler has returned: at a stop of its own, the
81    /// kernel no longer blocks its armed crash signal.
82    HandlerReturned,
83    /// The sender's `execve` succeeded.
84    Exec,
85}
86
87/// Why the switch was taken.
88#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
89pub enum PostCrashCause {
90    /// The crash signal's delivery took the default action.
91    DefaultAction,
92    /// This thread's `execve` succeeded while the crash signal's handler ran.
93    Exec(DetTid),
94}
95
96/// Where the switch was taken: recorded in a preemption record and reported
97/// in the run summary.
98#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize)]
99pub struct PostCrashSwitch {
100    /// The thread the crash signal was delivered to.
101    pub dtid: DetTid,
102    /// The crash signal's number.
103    pub signal: i32,
104    /// The thread whose `execve` took the switch, or `None` when the
105    /// delivery took the default action.
106    pub exec_by: Option<DetTid>,
107    /// The last committed scheduler turn when the switch was taken.
108    pub turn: u64,
109    /// The bound from then on, in virtual nanoseconds.
110    pub bound: u64,
111}
112
113impl PostCrashSwitch {
114    /// Why it was taken.
115    pub fn cause(&self) -> PostCrashCause {
116        match self.exec_by {
117            Some(dtid) => PostCrashCause::Exec(dtid),
118            None => PostCrashCause::DefaultAction,
119        }
120    }
121}

123/// A thread whose caught crash signal armed the switch.
124#[derive(Debug, Clone, Copy, PartialEq, Eq)]
125struct Armed {
126    /// The crash signal.
127    signal: i32,
128    /// The armed thread's process.
129    process: DetPid,
130    /// The thread that took the crash, named in the switch however the
131    /// surviving task's identity is rekeyed afterwards (a non-leader
132    /// exec keeps the survivor under the leader's tid).
133    crashed: DetTid,
134}

136/// The scheduler's state of the switch.
137#[derive(Debug, Clone, Default)]
138pub struct PostCrashState {
139    /// The post-crash bound in effect for this run, or `None` when the switch
140    /// is off (`Config::post_crash_timeslice_in_effect`).
141    bound: Option<NonZeroU64>,
142    /// Each thread a caught crash signal armed, until its handler returned,
143    /// it exited or its crash took the switch. Ordered, so a switch names the
144    /// same thread in every run.
145    armed: BTreeMap<DetTid, Armed>,
146    /// For each process, the armed crashes whose lineage it is in: the
147    /// armed thread's own process and the processes created from the lineage
148    /// while that thread is armed. An `execve` in a process takes the switch
149    /// for one of its own armed crashes only.
150    lineage: BTreeMap<DetPid, BTreeSet<DetTid>>,
151    /// The switches taken, in order.
152    switches: Vec<PostCrashSwitch>,
153    /// The processes whose threads take the bound: each switched crash's
154    /// lineage, and every process any of them creates after.
155    crashed: BTreeSet<DetPid>,
156    /// Under a replay, the switches its record says were taken, still to
157    /// come: each switch this run takes must be the next of them.
158    expected: Option<std::collections::VecDeque<PostCrashSwitch>>,
159}
160
161impl PostCrashState {
162    /// The state for a run whose post-crash bound in effect is `bound`; a
163    /// replay passes the switches its record holds as `expected`.
164    pub fn new(bound: Option<NonZeroU64>, expected: Option<Vec<PostCrashSwitch>>) -> Self {
165        Self {
166            bound,
167            expected: expected.map(Into::into),
168            ..Self::default()
169        }
170    }
171
172    /// Apply `event` from `dtid`, a thread of `process`, during committed
173    /// turn `turn`. Returns the bound if `process` takes it (now or since an
174    /// earlier switch), and the switch if this step took one; or, under a
175    /// replay, why the switch this step took is not the one its record holds.
176    pub fn step(
177        &mut self,
178        dtid: DetTid,
179        process: DetPid,
180        event: PostCrashEvent,
181        turn: u64,
182    ) -> Result<(Option<NonZeroU64>, Option<PostCrashSwitch>), String> {
183        let Some(bound) = self.bound else {
184            return Ok((None, None));
185        };
186        if self.crashed.contains(&process) {
187            if event == PostCrashEvent::HandlerReturned {
188                self.disarm(dtid);
189            }
190            return Ok((self.bound_for(process), None));
191        }
192        let switch = match event {
193            PostCrashEvent::Delivered {
194                signal,
195                default_action: true,
196            } => {
197                self.switch_process(process);
198                Some((dtid, signal, None))
199            }
200            PostCrashEvent::Delivered {
201                signal,
202                default_action: false,
203            } => {
204                // A crash inside an armed thread's handler keeps the first
205                // arm: the outer handler, whose signal stays blocked until it
206                // returns, is the one still running.
207                self.armed.entry(dtid).or_insert(Armed {
208                    signal,
209                    process,
210                    crashed: dtid,
211                });
212                self.lineage.entry(process).or_default().insert(dtid);
213                None
214            }
215            PostCrashEvent::HandlerReturned => {
216                self.disarm(dtid);
217                None
218            }
219            PostCrashEvent::Exec => {
220                let crash = self
221                    .lineage
222                    .get(&process)
223                    .and_then(|crashes| crashes.iter().next().copied());
224                crash.map(|key| {
225                    let Armed {
226                        signal,
227                        process,
228                        crashed,
229                    } = self.armed[&key];
230                    self.switch_process(process);
231                    (crashed, signal, Some(dtid))
232                })
233            }
234        };
235        let Some((crashed, signal, exec_by)) = switch else {
236            return Ok((None, None));
237        };
238        let taken = PostCrashSwitch {
239            dtid: crashed,
240            signal,
241            exec_by,
242            turn,
243            bound: bound.get(),
244        };
245        if let Some(expected) = &mut self.expected {
246            match expected.pop_front() {
247                Some(recorded) if recorded == taken => {}
248                recorded => {
249                    return Err(format!(
250                        "the replay took the post-crash switch {taken:?}, but its record holds {}; \
251                         the replay has left its record",
252                        recorded.map_or("no further switch".to_owned(), |recorded| format!(
253                            "{recorded:?}"
254                        )),
255                    ));
256                }
257            }
258        }
259        self.switches.push(taken);
260        Ok((self.bound_for(process), Some(taken)))
261    }
262
263    /// `child`, a new process, was created by a thread of `creator`: it
264    /// belongs to its creator's crashes, armed or switched. A genuine
265    /// creation starts with fresh membership: Linux reuses freed process
266    /// IDs, and the ID's previous process may have switched or belonged to
267    /// an armed crash's lineage; the new process inherits only its actual
268    /// creator's state.
269    pub fn process_created(&mut self, creator: DetPid, child: DetPid) {
270        self.crashed.remove(&child);
271        self.lineage.remove(&child);
272        if self.crashed.contains(&creator) {
273            self.crashed.insert(child);
274        }
275        if let Some(crashes) = self.lineage.get(&creator).cloned() {
276            self.lineage.entry(child).or_default().extend(crashes);
277        }
278    }
279
280    /// Makes `process` a crashed one, and with it every crash armed in it:
281    /// each such crash's lineage takes the bound too, and the crash retires.
282    /// So no armed crash outlives its process taking the bound, where its
283    /// handler's return could no longer be followed (the process's threads
284    /// stop reading their masks once they hold the bound).
285    fn switch_process(&mut self, process: DetPid) {
286        self.crashed.insert(process);
287        let in_process: Vec<DetTid> = self
288            .armed
289            .iter()
290            .filter(|(_, armed)| armed.process == process)
291            .map(|(&dtid, _)| dtid)
292            .collect();
293        for crash in in_process {
294            let members: Vec<DetPid> = self
295                .lineage
296                .iter()
297                .filter(|(_, crashes)| crashes.contains(&crash))
298                .map(|(&member, _)| member)
299                .collect();
300            self.crashed.extend(members);
301            self.disarm(crash);
302        }
303    }
304
305    /// The exec'ing thread `former` survives its exec under `new` (the
306    /// leader's tid): an armed crash of its follows it there, as its
307    /// timers and queues do, so the exec's own switch step still finds
308    /// it under the surviving identity.
309    pub fn transfer_arm(&mut self, former: DetTid, new: DetTid) {
310        if let Some(armed) = self.armed.remove(&former) {
311            self.armed.insert(new, armed);
312        }
313        for crashes in self.lineage.values_mut() {
314            if crashes.remove(&former) {
315                crashes.insert(new);
316            }
317        }
318    }
319
320    /// `dtid` exited: it cannot be in a handler any more.
321    pub fn forget(&mut self, dtid: DetTid) {
322        self.disarm(dtid);
323    }
324
325    /// Drops `dtid`'s armed crash and its lineage, leaving every other
326    /// crash's.
327    fn disarm(&mut self, dtid: DetTid) {
328        if self.armed.remove(&dtid).is_some() {
329            self.lineage.retain(|_, crashes| {
330                crashes.remove(&dtid);
331                !crashes.is_empty()
332            });
333        }
334    }
335
336    /// The bound `process`'s threads take, once a switch has made it a
337    /// crashed process.
338    pub fn bound_for(&self, process: DetPid) -> Option<NonZeroU64> {
339        self.bound.filter(|_| self.crashed.contains(&process))
340    }

342    /// The switches taken so far, in order.
343    pub fn switches(&self) -> &[PostCrashSwitch] {
344        &self.switches
345    }
346
347    /// The run summary's account of the switch, or `None` when it is off.
348    pub fn summary(&self) -> Option<detcore_model::summary::PostCrashSummary> {
349        let bound = self.bound?;
350        Some(detcore_model::summary::PostCrashSummary {
351            bound: bound.get(),
352            switches: self
353                .switches
354                .iter()
355                .map(|switch| detcore_model::summary::PostCrashSwitchSummary {
356                    dtid: switch.dtid.as_raw(),
357                    signal: switch.signal,
358                    exec_by: switch.exec_by.map(|exec| exec.as_raw()),
359                    turn: switch.turn,
360                })
361                .collect(),
362            recorded_not_taken: self.expected.as_ref().map_or(0, |expected| expected.len()),
363        })
364    }

367/// Whether `signal`, delivered with `si_code` (`None` when it could not be
368/// read), is a crash: SIGABRT however it was sent (`abort` sends it with
369/// `tgkill`), or SIGSEGV, SIGBUS, SIGILL, SIGFPE or SIGSYS raised by the
370/// kernel (`si_code > 0`; a `kill`, `tgkill` or `sigqueue` sends them with
371/// `si_code <= 0`, and a fault is the only kernel source of these).
372pub fn is_crash_signal(signal: i32, si_code: Option<i32>) -> bool {
373    match Signal::try_from(signal) {
374        Ok(Signal::SIGABRT) => true,
375        Ok(Signal::SIGSEGV | Signal::SIGBUS | Signal::SIGILL | Signal::SIGFPE | Signal::SIGSYS) => {
376            si_code.is_some_and(|code| code > 0)
377        }
378        _ => false,
379    }
380}

382/// `si_code` of the signal a thread is stopped for: the third `int` of its
383/// `siginfo_t` on every architecture Hermit runs on.
384fn si_code<G, T>(guest: &G) -> Option<i32>
385where
386    G: Guest<Detcore<T>>,
387    T: RecordOrReplay,
388{
389    let info = guest.signal_info()?;
390    Some(i32::from_ne_bytes(info.get(8..12)?.try_into().ok()?))
391}

393/// Whether the kernel's signal masks of guest threads are the guests' own and
394/// readable at a thread's own stops: the backend's guest threads are host
395/// tasks whose masks no runtime of its own changes
396/// (`kernel_signal_state_is_guests`: ptrace), threads are serialized (no other
397/// thread runs between an armed thread's stops and another thread's
398/// `execve`), and `/proc` reads are not refused. Elsewhere a caught crash does
399/// not arm the switch.
400fn reads_handler_masks(config: &crate::Config) -> bool {
401    config.sequentialize_threads
402        && config.backend.kernel_signal_state_is_guests
403        && !crate::syscalls::refuses_host_signal_state_reads(&config.backend)
404}
405
406/// The refusal, by name, of a run whose `--post-crash-timeslice` switch is
407/// in effect on a backend that cannot see a caught crash's handler return
408/// (`reads_handler_masks`): the switch would then take the default-action
409/// crashes and silently ignore every caught one, and the same guest would
410/// switch on ptrace and not there. Detcore has no handler-lifetime record of
411/// its own that a backend could share instead, so the run is refused before
412/// the guest starts (`GlobalState::initialize`), as unsupported options are.
413/// `None` when the switch is off or the backend reads the guests' masks.
414pub fn admission_refusal(config: &crate::Config) -> Option<String> {
415    config.post_crash_timeslice_in_effect()?;
416    (!reads_handler_masks(config)).then(|| {
417        "--post-crash-timeslice is unsupported on this backend: the switch follows a \
418         caught crash's handler until it returns, which it reads from the guest \
419         threads' kernel signal masks, and this backend's guest threads do not have \
420         their own kernel masks readable (BackendCapabilities::kernel_signal_state_is_guests, \
421         or host /proc signal state reads are refused); run with --backend=ptrace or \
422         without --post-crash-timeslice"
423            .to_owned()
424    })
425}

427/// Whether the kernel blocks `signal` for the stopped thread `guest`. A crash
428/// handler runs with its signal blocked: Linux adds a caught signal to the
429/// mask while its handler runs (unless the handler was installed with
430/// `SA_NODEFER`, which a crash handler is not: such a handler counts as
431/// returned), and the handler's `rt_sigreturn`, or a `siglongjmp` that
432/// restores the mask, takes it out again; a blocked fault signal is never
433/// delivered to a handler. A thread that cannot be read counts as returned.
434fn signal_blocked<G, T>(guest: &G, signal: i32) -> bool
435where
436    G: Guest<Detcore<T>>,
437    T: RecordOrReplay,
438{
439    let bit = crate::syscalls::kernel_sigset_bit(signal);
440    crate::syscalls::read_guest_thread_signal_state(guest)
441        .is_ok_and(|state| state.blocked & bit != 0)
442}

444/// Records what the scheduler answered a step: the bound, once the thread's
445/// process takes it.
446fn learn(state: &mut crate::tool_local::ThreadState<impl Sized>, bound: Option<NonZeroU64>) {
447    state.post_crash_bound = state.post_crash_bound.or(bound);
448    if state.post_crash_bound.is_some() {
449        state.crash_armed = None;
450    }
451}

453/// At a stop of the thread's own (a syscall entry before the call runs, a
454/// syscall exit or trap after it completes, the end of one of its slices, a
455/// signal delivery before the kernel installs the handler's mask): if the
456/// thread is armed and its crash handler has returned, the step that disarms
457/// it.
458///
459/// Not while a lone thread's skipped slice end is processed
460/// (`ThreadState::replaying_lone_slice_end`): such an end reads nothing from
461/// the guest but its branch count (`Detcore::catch_up_lone_slice_ends`), and
462/// the thread's mask is read where it is now, not at that end. The thread's
463/// next stop of its own, which comes before another thread can run, checks
464/// instead.
465pub(crate) async fn note_stop<G, T>(guest: &mut G)
466where
467    G: Guest<Detcore<T>>,
468    T: RecordOrReplay,
469{
470    if guest.thread_state().replaying_lone_slice_end.is_some() {
471        return;
472    }
473    let Some(signal) = guest.thread_state().crash_armed else {
474        return;
475    };
476    if guest.thread_state().post_crash_bound.is_some() || signal_blocked(guest, signal) {
477        return;
478    }
479    guest.thread_state_mut().crash_armed = None;
480    let bound = tool_global::post_crash_step(guest, PostCrashEvent::HandlerReturned).await;
481    learn(guest.thread_state_mut(), bound);
482}

484/// When `signal` is about to be delivered to the thread, before Detcore's copy
485/// of the dispositions or the kernel applies the delivery (so a one-shot
486/// `SA_RESETHAND` handler still counts as caught): a stop for an armed thread
487/// ([`note_stop`]), then a crash signal's step (`PostCrashEvent::Delivered`).
488pub(crate) async fn note_delivery<G, T>(guest: &mut G, signal: i32)
489where
490    G: Guest<Detcore<T>>,
491    T: RecordOrReplay,
492{
493    if guest.config().post_crash_timeslice_in_effect().is_none()
494        || guest.thread_state().post_crash_bound.is_some()
495    {
496        return;
497    }
498    note_stop(guest).await;
499    if !guest.thread_state().guest_past_first_execve() || !is_crash_signal(signal, si_code(guest)) {
500        return;
501    }
502    let default_action = crate::delivery_takes_default_action(guest, signal);
503    // A crash caught inside an armed thread's handler keeps the first arm;
504    // one that takes the default action switches whatever the thread was
505    // doing.
506    if !default_action
507        && (guest.thread_state().crash_armed.is_some() || !reads_handler_masks(guest.config()))
508    {
509        return;
510    }
511    let bound = tool_global::post_crash_step(
512        guest,
513        PostCrashEvent::Delivered {
514            signal,
515            default_action,
516        },
517    )
518    .await;
519    let state = guest.thread_state_mut();
520    if !default_action && bound.is_none() {
521        state.crash_armed = Some(signal);
522    }
523    learn(state, bound);
524}

526/// After the thread's `execve` succeeded: the step that takes an armed
527/// switch.
528pub(crate) async fn note_exec<G, T>(guest: &mut G)
529where
530    G: Guest<Detcore<T>>,
531    T: RecordOrReplay,
532{
533    if guest.config().post_crash_timeslice_in_effect().is_none()
534        || guest.thread_state().post_crash_bound.is_some()
535    {
536        return;
537    }
538    let bound = tool_global::post_crash_step(guest, PostCrashEvent::Exec).await;
539    learn(guest.thread_state_mut(), bound);
540}

611        turn: u64,
612    ) -> (Option<NonZeroU64>, Option<PostCrashSwitch>) {
613        state.step(tid(dtid), pid(process), event, turn).unwrap()
614    }
