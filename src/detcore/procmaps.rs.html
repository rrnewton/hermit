19fn display_pathname(p: &MMapPath) -> String {
20    match p {
21        MMapPath::Vdso => String::from("[vsdo]"),
22        MMapPath::Stack => String::from("[stack]"),
23        MMapPath::TStack(tid) => format!("[tstack:{}]", tid),
24        MMapPath::Vvar => String::from("[vvar]"),
25        MMapPath::Vsyscall => String::from("[syscalls]"),
26        MMapPath::Heap => String::from("[heap]"),
27        MMapPath::Other(s) => format!("[other: {}]", s),
28        MMapPath::Anonymous => String::from("[annonymous]"),
29        MMapPath::Path(s) => s.display().to_string(),
30        MMapPath::Rollup => String::from("[rollup]"),
31        MMapPath::Vsys(vsys) => vsys.to_string(),
32    }
33}

39/// Render a mapping with an explicit pathname column.
40///
41/// Used to report a region that *is* the guest heap but which the kernel did
42/// not label `[heap]`, so the record is textually comparable with the labelled
43/// one another backend produces. Every other column is the mapping's real
44/// procfs data.
45pub fn display_as(map: &MemoryMap, pathname: &str) -> String {
46    display_range_as(map, map.address.0, map.address.1, pathname)
47}
48
49/// Render a mapping's real procfs columns against an EXPLICIT address range.
50///
51/// The two address columns come from `start`/`end`; every other column is the
52/// mapping's real procfs data. Used where the region being reported is a SUBSET
53/// of the mapping that contains it -- the observed program break inside its
54/// enclosing anonymous mapping -- so the record's extent is the region actually
55/// hashed rather than the arena that happens to hold it.
56pub fn display_range_as(map: &MemoryMap, start: u64, end: u64, pathname: &str) -> String {
57    format!(
58        "{:#x}-{:#x} {:?} {:x} {:x}:{:x} {} {}",
59        start, end, map.perms, map.offset, map.dev.0, map.dev.1, map.inode, pathname
60    )
61}

63/// Reverie ptrace's own pages in every guest process, `[start, end)`: its
64/// trampoline at `0x7100_0000` and the SIGTRAP repairs' scratch page after
65/// it (reverie-ptrace `cp::PRIVATE_PAGE_OFFSET`, `cp::SCRATCH_BASE` and
66/// `cp::SCRATCH_SIZE`, which that crate keeps private; hermit-cli's
67/// `REVERIE_PTRACE_PRIVATE_PAGE`, pinned by a cli test, names the first).
68/// They are private anonymous mappings, but Reverie's state, not the guest's.
69pub const REVERIE_PTRACE_PAGES: (u64, u64) = (0x7100_0000, 0x7100_2000);
70
71/// Whether `map` is one `--detlog-anon` hashes: anonymous (unnamed, named by
72/// `PR_SET_VMA_ANON_NAME` as `[anon:<name>]`, or a thread stack an older
73/// kernel labels `[stack:<tid>]`), private, readable and writable, and not
74/// one of Reverie's own pages ([`REVERIE_PTRACE_PAGES`]). `[heap]` and
75/// `[stack]` are `--detlog-heap`'s and `--detlog-stack`'s; a shared anonymous
76/// mapping (`[anon_shmem:<name>]`, or `/dev/zero (deleted)`) is another
77/// process's memory as much as this one's; and a mapping that cannot be read
78/// (`PROT_NONE`, such as an allocator's reservation) or written holds nothing
79/// the guest changed through it.
80pub fn is_hashed_anonymous(map: &MemoryMap) -> bool {
81    use procfs::process::MMPermissions;
82    let anonymous = match &map.pathname {
83        MMapPath::Anonymous | MMapPath::TStack(_) => true,
84        MMapPath::Other(name) => name.starts_with("anon:"),
85        _ => false,
86    };
87    let reverie =
88        REVERIE_PTRACE_PAGES.0 <= map.address.0 && map.address.1 <= REVERIE_PTRACE_PAGES.1;
89    anonymous
90        && !reverie
91        && map
92            .perms
93            .contains(MMPermissions::READ | MMPermissions::WRITE | MMPermissions::PRIVATE)
94}

96/// The size of the pages `--detlog-anon=pages` digests.
97const DIGESTED_PAGE: usize = 4096;
98
99/// The largest mapping `--detlog-anon` reads. A larger one, such as a
100/// sanitizer's shadow (terabytes, `MAP_NORESERVE`) or a heap a runtime
101/// reserves read-write up front, is recorded by its size alone.
102pub const ANON_HASH_LIMIT: u64 = 256 << 20;
103
104/// How much of a mapping `--detlog-anon` reads at a time, a whole number of
105/// pages, so that page digests never straddle a read.
106const ANON_READ_CHUNK: usize = 1 << 20;
107
108/// The digest of one page for `--detlog-anon=pages`: `<index>:<the first 8
109/// bytes of its SHA-256 in hex>`, or `None` for an all-zero page, which an
110/// untouched anonymous page reads as.
111fn page_digest(index: usize, page: &[u8]) -> Option<String> {
112    page.iter()
113        .any(|byte| *byte != 0)
114        .then(|| format!("{index}:{}", &Digest::new(page).to_string()[..16]))
115}
116
117/// For `--detlog-anon=pages`: each 4 KiB page of `bytes` that is not all zero
118/// ([`page_digest`]), comma-separated, computed at once: the reference
119/// [`hash_in_chunks`] is checked against. A page index counts from the start
120/// of the mapping, so a mapping that merges with a neighbour or moves
121/// (`mremap`) renumbers its pages.
122#[cfg(test)]
123fn page_digests(bytes: &[u8]) -> String {
124    bytes
125        .chunks(DIGESTED_PAGE)
126        .enumerate()
127        .filter_map(|(index, page)| page_digest(index, page))
128        .collect::<Vec<_>>()
129        .join(",")
130}

132/// What `--detlog-anon` records for one mapping.
133pub enum AnonHash {
134    /// Its digest and, where asked for, its page digests (`<index>:<hex>`,
135    /// comma-separated; [`hash_in_chunks`]).
136    Hashed(Digest, Option<String>),
137    /// Larger than [`ANON_HASH_LIMIT`]: its size, unread.
138    TooLarge(u64),
139}
140
141/// The digest of `size` bytes and, where `pages`, their page digests, read
142/// `chunk` bytes at a time (a multiple of the page size) through `read`,
143/// which fills its buffer from the given offset. Equal to hashing all the
144/// bytes at once, which the unit tests check, without holding them at once.
145fn hash_in_chunks(
146    size: u64,
147    chunk: usize,
148    pages: bool,
149    mut read: impl FnMut(u64, &mut [u8]) -> Result<(), reverie::Error>,
150) -> Result<(Digest, Option<String>), reverie::Error> {
151    debug_assert_eq!(chunk % DIGESTED_PAGE, 0);
152    let mut digest = crate::digest::DigestBuilder::default();
153    let mut listed = Vec::new();
154    let mut buf = vec![0; chunk.min(size as usize)];
155    let mut offset = 0u64;
156    while offset < size {
157        let n = (size - offset).min(chunk as u64) as usize;
158        read(offset, &mut buf[..n])?;
159        digest.update(&buf[..n]);
160        if pages {
161            let first = (offset / DIGESTED_PAGE as u64) as usize;
162            listed.extend(
163                buf[..n]
164                    .chunks(DIGESTED_PAGE)
165                    .enumerate()
166                    .filter_map(|(index, page)| page_digest(first + index, page)),
167            );
168        }
169        offset += n as u64;
170    }
171    Ok((digest.finish(), pages.then(|| listed.join(","))))
172}

174/// What `--detlog-anon` records for the guest bytes in `[start, end)`
175/// ([`AnonHash`]), read in [`ANON_READ_CHUNK`]s.
176pub fn compute_hash_and_pages<G, T: Tool>(
177    guest: &mut G,
178    start: u64,
179    end: u64,
180    pages: bool,
181) -> Result<AnonHash, reverie::Error>
182where
183    G: Guest<T>,
184{
185    let size = end.saturating_sub(start);
186    if size > ANON_HASH_LIMIT {
187        return Ok(AnonHash::TooLarge(size));
188    }
189    let memory = guest.memory();
190    let (digest, listed) = hash_in_chunks(size, ANON_READ_CHUNK, pages, |offset, buf| {
191        let address = Addr::<u8>::from_raw((start + offset) as usize).unwrap();
192        memory.read_values(address, buf)?;
193        Ok(())
194    })?;
195    Ok(AnonHash::Hashed(digest, listed))
196}

198fn map_error(err: procfs::ProcError) -> reverie::Error {
199    match err {
200        procfs::ProcError::Io(err, _) => reverie::Error::Io(err),
201        err => reverie::Error::Tool(anyhow::anyhow!(err)),
202    }
203}

205pub fn from_pid<F>(pid: Pid, filter: F) -> Result<Vec<MemoryMap>, reverie::Error>
206where
207    F: Fn(&MemoryMap) -> bool,
208{
209    match procfs::process::Process::new(pid.as_raw()) {
210        Ok(process) => match process.maps() {
211            Ok(mut maps) => {
212                maps.0.retain(filter);
213                Ok(maps.0)
214            }
215            Err(err) => Err(map_error(err)),
216        },
217        Err(err) => Err(map_error(err)),
218    }
219}

221pub fn compute_hash<G, T: Tool>(guest: &mut G, map: &MemoryMap) -> Result<Digest, reverie::Error>
222where
223    G: Guest<T>,
224{
225    compute_hash_range(guest, map.address.0, map.address.1)

228/// Hash the guest bytes in the half-open guest-virtual range `[start, end)`,
229/// read through `guest.memory()`. Used for backend-reported memory regions
230/// ([`reverie::Guest::detlog_memory_regions`]) where the range is a guest
231/// address rather than an entry parsed from a host `/proc/<pid>/maps`.
232pub fn compute_hash_range<G, T: Tool>(
233    guest: &mut G,
234    start: u64,
235    end: u64,
236) -> Result<Digest, reverie::Error>
237where
238    G: Guest<T>,
239{
240    let size = end.saturating_sub(start) as usize;
241    let memory = guest.memory();
242    let mut buf = vec![0; size];
243    let start = Addr::<u8>::from_raw(start as usize).unwrap();
244    memory.read_values(start, buf.as_mut_slice())?;
245    Ok(Digest::new(buf.as_slice()))
246}

266    #[test]
267    fn only_private_writable_anonymous_mappings_are_hashed() {
268        let rw = MMPermissions::READ | MMPermissions::WRITE | MMPermissions::PRIVATE;
269        for pathname in [
270            MMapPath::Anonymous,
271            MMapPath::Other("anon:scudo:primary".to_owned()),
272            MMapPath::TStack(12),
273        ] {
274            assert!(
275                is_hashed_anonymous(&map(rw, pathname.clone())),
276                "{pathname:?}"
277            );
278        }
279        for pathname in [
280            MMapPath::Heap,
281            MMapPath::Stack,
282            MMapPath::Vvar,
283            MMapPath::Other("anon_shmem:x".to_owned()),
284            MMapPath::Path("/usr/lib/libc.so.6".into()),
285        ] {
286            assert!(
287                !is_hashed_anonymous(&map(rw, pathname.clone())),
288                "{pathname:?}"
289            );
290        }
291        // Reverie's trampoline (read-write-execute) and scratch page.
292        let rwx = rw | MMPermissions::EXECUTE;
293        let mut trampoline = map(rwx, MMapPath::Anonymous);
294        trampoline.address = (0x7100_0000, 0x7100_1000);
295        let mut scratch = map(rw, MMapPath::Anonymous);
296        scratch.address = (0x7100_1000, 0x7100_2000);
297        assert!(!is_hashed_anonymous(&trampoline) && !is_hashed_anonymous(&scratch));
298        // A reservation, a read-only mapping, and a shared one.
299        for perms in [
300            MMPermissions::PRIVATE,
301            MMPermissions::READ | MMPermissions::PRIVATE,
302            MMPermissions::READ | MMPermissions::WRITE | MMPermissions::SHARED,
303        ] {
304            assert!(
305                !is_hashed_anonymous(&map(perms, MMapPath::Anonymous)),
306                "{perms:?}"
307            );
308        }
309    }

311    #[test]
312    fn page_digests_list_the_pages_that_are_not_all_zero() {
313        let mut bytes = vec![0u8; 3 * 4096];
314        assert_eq!(page_digests(&bytes), "");
315        bytes[4096 + 17] = 1;
316        let digests = page_digests(&bytes);
317        let expected = &Digest::new(&bytes[4096..8192]).to_string()[..16];
318        assert_eq!(digests, format!("1:{expected}"));
319        bytes[2 * 4096] = 2;
320        assert_eq!(page_digests(&bytes).split(',').count(), 2);
321        // A short last page counts as a page.
322        assert_eq!(page_digests(&[0, 0, 3]).split(':').next(), Some("0"));
323    }
324
325    #[test]
326    fn reading_in_chunks_equals_hashing_at_once() {
327        // Two and a half chunks of two pages each, with a page written in
328        // each chunk and the last, short, chunk.
329        let mut bytes = vec![0u8; 5 * 4096 + 100];
330        for (index, byte) in [(1, 7u8), (2 * 4096 + 9, 8), (5 * 4096 + 50, 9)] {
331            bytes[index] = byte;
332        }
333        let (digest, pages) = hash_in_chunks(bytes.len() as u64, 2 * 4096, true, |offset, buf| {
334            let offset = offset as usize;
335            buf.copy_from_slice(&bytes[offset..offset + buf.len()]);
336            Ok(())
337        })
338        .unwrap();
339        assert_eq!(digest, Digest::new(&bytes));
340        assert_eq!(pages.as_deref(), Some(page_digests(&bytes).as_str()));
341        assert_eq!(pages.unwrap().split(',').count(), 3);
342        let (empty, none) = hash_in_chunks(0, 4096, false, |_, _| Ok(())).unwrap();
343        assert_eq!((empty, none), (Digest::new(&[]), None));
344    }
