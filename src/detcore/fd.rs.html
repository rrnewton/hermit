72/// Deterministic file descriptor
73///
74/// Notice `statbuf` can be cached here, this is because
75/// `stat` is valid as long as fd stays open.
76#[derive(Debug, Clone, Serialize, Deserialize)]
77pub struct DetFd {
78    /// underlying file descriptor
79    pub(crate) fd: RawFd,
80    /// Per-slot descriptor flags, currently only `O_CLOEXEC`.
81    fd_flags: i32,
82    /// State shared by every descriptor referring to the same Linux `struct file`.
83    open_file: Arc<Mutex<OpenFileDescription>>,
84}
85
86#[derive(Debug, Serialize, Deserialize)]
87struct OpenFileDescription {
88    id: OpenFileId,
89    /// fd type
90    ty: FdType,
91    /// Process named by a pidfd created through `pidfd_open`.
92    ///
93    /// This is shared by descriptor aliases just like the kernel pidfd object.
94    /// `None` means either that this is not a pidfd or that Detcore did not
95    /// observe enough provenance to identify its target.
96    #[serde(default)]
97    pidfd_target: Option<DetPid>,
98    /// File status flags shared by dup and fork aliases.
99    status_flags: i32,
100    /// File path associated with fd.
101    /// This cannot be relied upon. Special devices won't have it, for example.
102    path: Option<PathBuf>,
103    /// Cached det/virtual inode.
104    /// This cannot be relied upon. Special devices won't have it, for example.
105    /// However if `ty` indicates a `Regular` file, then there should reliably be an inode.
106    inode: Option<DetInode>,
107    /// inode is dirty
108    dirty: bool,
109
110    /// Irrespective of whether the file descriptor is marked logically blocking by the
111    /// user, this tracks whether Detcore has converted the fd to nonblocking for its own
112    /// purposes.
113    physically_nonblocking: bool,
114
115    /// cached statbuf
116    ///
117    /// This is the RAW stat from the file system, NOT determinized.
118    ///
119    /// Some of these fields will change at runtime. But the following fields will
120    /// be constant when `virtualize_metadata` is on, over the life of the DetFd:
121    ///  - dev, rdev, blksize
122    ///
123    /// This should always be `Some` for regular files, as we eagerly populate it.
124    stat: Option<DetStat>,
125    // AUTONOMOUS-BOT-IMPLEMENTED
126    // TODO-HUMAN-REVIEW(PR-1096): Review canonical random-device cursor sharing.
127    /// Cursor into Hermit's backend-independent random-device byte stream.
128    #[serde(default)]
129    random_device_offset: u64,
130    /// resource
131    resource: Option<ResourceID>,
132    /// Deterministic snapshot state for selected procfs files.
133    procfs: Option<ProcfsFile>,
134    /// Logical timestamp of the last packet delivered through this socket.
135    socket_receive_timestamp: Option<LogicalTime>,
136    /// True when this open file is an `AF_NETLINK`/`NETLINK_SOCK_DIAG` socket,
137    /// whose binary dump replies carry host-assigned socket inode numbers that
138    /// must be determinized (see `crate::sock_diag`).
139    sock_diag: bool,
140    /// Whether this open file is a `NETLINK_ROUTE` socket whose link-dump
141    /// replies carry live interface counters that must be zeroed (see
142    /// `crate::netlink_route`).
143    netlink_route: bool,
144    /// True when this socket connected to an IPv4 or IPv6 loopback peer.
145    #[serde(default)]
146    loopback_peer: bool,
147    // AUTONOMOUS-BOT-IMPLEMENTED
148    // TODO-HUMAN-REVIEW(#2373)
149    /// The `flock(2)` mode this open file description currently holds, as the
150    /// bare `LOCK_SH`/`LOCK_EX` constant, or `None` when it holds no lock.
151    ///
152    /// `flock` locks are a property of the open file description, not of the
153    /// descriptor slot and not of the inode, which is exactly the granularity
154    /// this struct models: `dup`/`fork` aliases share one lock, two separate
155    /// `open`s of the same file contend with each other.
156    ///
157    /// Detcore tracks this only so `handle_flock` can tell a first acquisition
158    /// (where a failed `LOCK_NB` probe changes nothing) from a *conversion* of
159    /// an already-held lock (where Linux drops the old lock before it can fail).
160    /// It is a cache of what Detcore itself granted, never an authority: it is
161    /// written only after the kernel reports success.
162    #[serde(default)]
163    flock_mode: Option<i32>,
164    /// Whether `flock_mode` describes the kernel state. Descriptors created by
165    /// an intercepted syscall start known-unlocked; descriptors discovered
166    /// after entering the guest can already carry a lock.
167    #[serde(default)]
168    flock_mode_known: bool,
169    /// Whether Detcore has EVER known this description's lock state.
170    ///
171    /// This separates two different unknowns that `flock_mode_known == false`
172    /// otherwise collapses. A descriptor Detcore never observed -- stdin,
173    /// stdout, stderr, a live-discovered fd -- has no cached claim at all, so
174    /// there is nothing about it that can be STALE. A descriptor that was known
175    /// and then invalidated by a process copy does have a claim that may now be
176    /// wrong. Only the second is a reason to refuse `vfork`.
177    #[serde(default)]
178    flock_mode_ever_known: bool,
179}

196/// If the flags specify O_NONBLOCK.
197fn oflags_nonblocking(flags: i32) -> bool {
198    let o_nonblock = OFlag::O_NONBLOCK.bits();
199    flags & o_nonblock == o_nonblock
200}

205        let bits = flags.bits();
206        DetFd {

209            open_file: Arc::new(Mutex::new(OpenFileDescription {

213                status_flags: bits & !OFlag::O_CLOEXEC.bits(),
214                path: None,
215                inode: None,

299    /// Update both the logical (guest-visible) and physical (scheduler)
300    /// nonblocking status for every alias of this open file description. Use this
301    /// only when the physical fd genuinely tracks the guest's request; when
302    /// Detcore forces the fd physically nonblocking for the scheduler, update the
303    /// logical view alone via [`Self::set_logical_nonblocking`].
304    pub fn set_nonblocking(&self, enabled: bool) {
305        let mut description = self.description();
306        if enabled {
307            description.status_flags |= OFlag::O_NONBLOCK.bits();
308        } else {
309            description.status_flags &= !OFlag::O_NONBLOCK.bits();
310        }
311        description.physically_nonblocking = enabled;
312    }

314    /// Update only the logical (guest-visible) O_NONBLOCK status flag, leaving
315    /// the physical (scheduler) nonblocking state untouched. This lets a guest
316    /// clear O_NONBLOCK while Detcore keeps the fd physically nonblocking, which
317    /// the scheduler relies on for nonblockize-and-retry.
318    pub fn set_logical_nonblocking(&self, enabled: bool) {
319        let mut description = self.description();
320        if enabled {
321            description.status_flags |= OFlag::O_NONBLOCK.bits();
322        } else {
323            description.status_flags &= !OFlag::O_NONBLOCK.bits();
324        }
325    }

620    /// Mark the kernel lock state unknown without changing the kernel lock.
621    /// This is required for live-discovered and externally received file
622    /// descriptions, whose history Detcore did not observe.
623    pub(crate) fn forget_flock_mode(&self) {
624        let mut description = self.description();
625        description.flock_mode = None;
626        description.flock_mode_known = false;
627    }

629    /// Record that Detcore never observed this description's lock history, as
630    /// opposed to having known it and then invalidated it. Used for descriptors
631    /// that existed before Detcore began observing the guest.
632    pub(crate) fn mark_flock_mode_unobserved(&self) {
633        let mut description = self.description();
634        description.flock_mode = None;
635        description.flock_mode_known = false;
636        description.flock_mode_ever_known = false;
637    }

639    /// True when a cached lock claim existed and may now be wrong.
640    pub(crate) fn flock_mode_may_be_stale(&self) -> bool {
641        let description = self.description();
642        description.flock_mode_ever_known && !description.flock_mode_known
643    }

646impl fmt::Display for DetFd {

711    /// `flock(2)` locks belong to the open file description, so every `dup`
712    /// alias must see one lock, while a separate `open` of the same file is a
713    /// distinct contender with its own state.
714    #[test]
715    fn flock_mode_is_shared_by_dup_aliases_and_private_to_separate_opens() {
716        let owner = DetTid::from_raw(10);
717        let original = DetFd::new(
718            3,
719            OFlag::empty(),
720            FdType::Regular,
721            OpenFileId::new(owner, 0),
722        );
723        let duplicate = original.clone().with_fd(4);
724        let separate_open = DetFd::new(
725            5,
726            OFlag::empty(),
727            FdType::Regular,
728            OpenFileId::new(owner, 1),
729        );
730
731        assert_eq!(original.known_flock_mode(), Some(None));
732        original.set_flock_mode(Some(libc::LOCK_SH));
733        assert_eq!(
734            duplicate.known_flock_mode(),
735            Some(Some(libc::LOCK_SH)),
736            "a dup alias shares the open file description's flock lock"
737        );
738        assert_eq!(
739            separate_open.known_flock_mode(),
740            Some(None),
741            "a separate open of the same file is an independent flock contender"
742        );
743
744        duplicate.set_flock_mode(Some(libc::LOCK_EX));
745        assert_eq!(original.known_flock_mode(), Some(Some(libc::LOCK_EX)));
746        duplicate.set_flock_mode(None);
747        assert_eq!(
748            original.known_flock_mode(),
749            Some(None),
750            "LOCK_UN through one alias releases the whole open file description"
751        );
752
753        duplicate.forget_flock_mode();
754        assert_eq!(
755            original.known_flock_mode(),
756            None,
757            "unknown state must be shared by every alias of the open file description"
758        );
759        original.set_flock_mode(Some(libc::LOCK_SH));
760        assert_eq!(
761            duplicate.known_flock_mode(),
762            None,
763            "a later call cannot make externally mutable lock state reliable again"
764        );
765    }

784    #[test]
785    fn toggling_logical_nonblocking_preserves_physical() {
786        // Models FIONBIO on an fd that Detcore forced physically nonblocking for
787        // the scheduler: the guest-visible flag changes, but the physical state
788        // must survive.
789        let owner = DetTid::from_raw(10);
790        let fd = DetFd::new(3, OFlag::empty(), FdType::Socket, OpenFileId::new(owner, 0));
791        fd.set_physically_nonblocking();
792        fd.set_logical_nonblocking(true);
793        assert!(fd.is_nonblocking());
794        assert!(fd.physically_nonblocking());
795
796        fd.set_logical_nonblocking(false);
797        assert!(
798            !fd.is_nonblocking(),
799            "the guest must observe O_NONBLOCK cleared"
800        );
801        assert!(
802            fd.physically_nonblocking(),
803            "the scheduler's physical nonblocking state must be preserved"
804        );
805
806        fd.set_logical_nonblocking(true);
807        assert!(fd.is_nonblocking());
808        assert!(
809            fd.physically_nonblocking(),
810            "setting the logical flag must not discard physical tracking"
811        );
812
813        // The both-flags setter still tracks physical alongside logical.
814        fd.set_nonblocking(false);
815        assert!(!fd.is_nonblocking());
816        assert!(!fd.physically_nonblocking());
