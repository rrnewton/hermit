31/// Opaque control request for one authenticated mapping observation.
32#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
33pub struct Request(Operation);
34
35#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
36enum Operation {
37    Prepare(MapsTarget, ControllerCaptureFact),
38    Read(MappingCaptureToken),
39    Complete(MappingCaptureToken, Option<String>),
40}
41
42/// Controller reply for the bounded shared mapping capture protocol.
43#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
44pub enum Reply {
45    /// The exact target and launcher containment have been authenticated.
46    Prepared(MappingCaptureToken),
47    /// Complete raw views of the same target during backend exclusion.
48    Captured {
49        /// Echo of the prepared identity.
50        token: MappingCaptureToken,
51        /// Complete Maps extents.
52        maps: Vec<u8>,
53        /// Complete NUMA fields, unchanged.
54        numa: Vec<u8>,
55    },
56    /// Normal restoration was acknowledged.
57    Completed,
58    /// The observation could not be established.
59    Refused(String),
60}
61
62#[derive(Debug)]
63pub(super) struct Pending {
64    token: MappingCaptureToken,
65    target: MapsTarget,
66    fact: ControllerCaptureFact,
67    read: bool,
68}

70impl GlobalState {
71    fn mapping_target_is_current(
72        &self,
73        sched: &Scheduler,
74        from: Tid,
75        mm: MmId,
76        target: &MapsTarget,
77        fact: ControllerCaptureFact,
78    ) -> Option<MappingCaptureIdentity> {
79        let caller = DetTid::from_raw(from.as_raw());
80        if !self.cfg.sequentialize_threads
81            || self.cfg.recordreplay_modes
82            || target.mm != mm
83            || target.task.process != caller.as_raw()
84            || target
85                .task
86                .thread
87                .is_some_and(|thread| thread != caller.as_raw())
88        {
89            return None;
90        }
91        let (process, thread) = sched.controller_capture_task(caller, mm, fact)?;
92        let existence = sched.procfs_task(caller, target.task.thread.map(DetTid::from_raw));
93        let answer = self.inodes.lock().unwrap().procfs_task_answer(
94            caller,
95            target.task.thread.map(DetTid::from_raw),
96            existence,
97        );
98        if !existence.0
99            || existence.1
100            || answer.incarnation() != target.incarnation
101            || sched.runtime_private_memory(mm)
102                != Some(reverie::RuntimePrivateMemory::Ready(target.private.clone()))
103        {
104            return None;
105        }
106        Some(MappingCaptureIdentity {
107            process,
108            thread,
109            address_space: [mm.creator().as_raw() as u64, mm.generation()],
110            private_owner: target.private.owner,
111            private_version: target.private.version,
112        })
113    }
114
115    pub(super) fn mapping_capture_request(&self, from: Tid, mm: MmId, request: &Request) -> Reply {
116        let Some(run) = &self.mapping_capture_run else {
117            return Reply::Refused("enabled launcher containment is unavailable".into());
118        };
119        let refuse_preparation = |reason: &str| {
120            run.reject_preparation(reason);
121            Reply::Refused(reason.into())
122        };
123        if !run.containment_enabled() {
124            return refuse_preparation("enabled launcher containment is unavailable");
125        }
126        // All operations use the same lock order, with no async yield. A Read
127        // retains the scheduler grant/pidfd through closing both proc files.
128        let sched = self.sched.lock().unwrap();
129        let mut pending = self.mapping_capture_pending.lock().unwrap();
130        match &request.0 {
131            Operation::Prepare(target, fact) => {
132                if pending.is_some() {
133                    return refuse_preparation("another mapping capture is pending");
134                }
135                let Some(identity) =
136                    self.mapping_target_is_current(&sched, from, mm, target, *fact)
137                else {
138                    return refuse_preparation(
139                        "opened mapping target is not the current serial caller",
140                    );
141                };
142                let Some(token) = run.prepare(identity) else {
143                    return refuse_preparation(
144                        "mapping capture requires the contained launch root",
145                    );
146                };
147                *pending = Some(Pending {
148                    token,
149                    target: target.clone(),
150                    fact: *fact,
151                    read: false,
152                });
153                Reply::Prepared(token)
154            }
155            Operation::Read(token) => {
156                let Some(view) = pending
157                    .as_mut()
158                    .filter(|view| view.token == *token && !view.read)
159                else {
160                    return Reply::Refused(
161                        "mapping capture request is absent, stale or repeated".into(),
162                    );
163                };
164                view.read = true;
165                if !run.begin_observation(*token)
166                    || self.mapping_target_is_current(&sched, from, mm, &view.target, view.fact)
167                        != Some(token.identity)
168                {
169                    return Reply::Refused("mapping capture authority changed before read".into());
170                }
171                match read_views(token.identity.thread, view.fact) {
172                    Ok((maps, numa))
173                        if self.mapping_target_is_current(
174                            &sched,
175                            from,
176                            mm,
177                            &view.target,
178                            view.fact,
179                        ) == Some(token.identity) =>
180                    {
181                        Reply::Captured {
182                            token: *token,
183                            maps,
184                            numa,
185                        }
186                    }
187                    Ok(_) => Reply::Refused("mapping capture authority changed after read".into()),
188                    Err(reason) => Reply::Refused(reason.into()),
189                }
190            }
191            Operation::Complete(token, rejection) => {
192                let valid = pending.as_ref().is_some_and(|view| {
193                    view.token == *token
194                        && view.read
195                        && self.mapping_target_is_current(&sched, from, mm, &view.target, view.fact)
196                            == Some(token.identity)
197                });
198                let completed = if !valid {
199                    run.complete(*token, false)
200                } else if let Some(reason) = rejection {
201                    run.reject_restored(*token, reason)
202                } else {
203                    run.complete(*token, true)
204                };
205                if completed {
206                    *pending = None;
207                    Reply::Completed
208                } else {
209                    Reply::Refused("mapping capture restoration was not completed".into())
210                }
211            }
212        }
213    }
214}

216const READER_CONTEXT_UNAVAILABLE: &str =
217    "mapping reader root or mount namespace proof is unavailable";
218const READER_CONTEXT_DIFFERS: &str = "mapping reader root or mount namespace differs";
219
220fn open_at(directory: i32, path: &std::ffi::CStr, flags: i32) -> std::io::Result<File> {
221    // SAFETY: the directory and terminated path remain live. A successful
222    // descriptor transfers immediately to File, including on later failures.
223    let raw = unsafe { libc::openat(directory, path.as_ptr(), flags | libc::O_CLOEXEC) };
224    if raw < 0 {
225        Err(std::io::Error::last_os_error())
226    } else {
227        Ok(unsafe { File::from_raw_fd(raw) })
228    }
229}

231#[derive(Clone, Copy, Debug, PartialEq, Eq)]
232struct ReaderContextIdentity {
233    root_device: u64,
234    root_inode: u64,
235    root_mount: u64,
236    namespace_device: u64,
237    namespace_inode: u64,
238}
239
240/// Descriptor pins prevent object/mount identity recycling during the reads.
241/// Reopening the current context afterwards checks changes; merely re-statting
242/// these retained descriptors would only check the old objects.
243struct ReaderContext {
244    _root: File,
245    _namespace: File,
246    identity: ReaderContextIdentity,
247}
248
249impl ReaderContext {
250    fn open(task: &File, reader: bool) -> Result<Self, &'static str> {
251        let root = if reader {
252            open_at(libc::AT_FDCWD, c"/", libc::O_PATH | libc::O_DIRECTORY)
253        } else {
254            open_at(task.as_raw_fd(), c"root", libc::O_PATH | libc::O_DIRECTORY)
255        }
256        .map_err(|_| READER_CONTEXT_UNAVAILABLE)?;
257        let namespace = open_at(task.as_raw_fd(), c"ns/mnt", libc::O_RDONLY)
258            .map_err(|_| READER_CONTEXT_UNAVAILABLE)?;
259        let namespace_metadata = namespace
260            .metadata()
261            .map_err(|_| READER_CONTEXT_UNAVAILABLE)?;
262        let mut filesystem = std::mem::MaybeUninit::<libc::statfs>::uninit();
263        // SAFETY: namespace and the complete output record remain live.
264        if unsafe { libc::fstatfs(namespace.as_raw_fd(), filesystem.as_mut_ptr()) } != 0
265            || unsafe { filesystem.assume_init() }.f_type as u64 != 0x6e736673
266        // NSFS_MAGIC
267        {
268            return Err(READER_CONTEXT_UNAVAILABLE);
269        }
270        use reverie::syscalls::linux_abi;
271        let mut raw = std::mem::MaybeUninit::<linux_abi::Statx>::zeroed();
272        let required = linux_abi::STATX_INO | linux_abi::STATX_MNT_ID;
273        // SAFETY: empty-path statx inspects the pinned root and writes exactly
274        // the declared complete output record. No missing mount-ID fallback.
275        if unsafe {
276            linux_abi::statx(
277                root.as_raw_fd(),
278                c"".as_ptr(),
279                libc::AT_EMPTY_PATH | libc::AT_NO_AUTOMOUNT,
280                required,
281                raw.as_mut_ptr(),
282            )
283        } != 0
284        {
285            return Err(READER_CONTEXT_UNAVAILABLE);
286        }
287        let found = unsafe { raw.assume_init() };
288        if found.stx_mask & required != required {
289            return Err(READER_CONTEXT_UNAVAILABLE);
290        }
291        Ok(Self {
292            _root: root,
293            _namespace: namespace,
294            identity: ReaderContextIdentity {
295                root_device: libc::makedev(found.stx_dev_major, found.stx_dev_minor),
296                root_inode: found.stx_ino,
297                root_mount: found.stx_mnt_id,
298                namespace_device: namespace_metadata.dev(),
299                namespace_inode: namespace_metadata.ino(),
300            },
301        })
302    }
303}

305fn read_views(tid: i32, fact: ControllerCaptureFact) -> Result<(Vec<u8>, Vec<u8>), &'static str> {
306    let root = File::open("/proc").map_err(|_| "controller procfs open failed")?;
307    let metadata = root
308        .metadata()
309        .map_err(|_| "controller procfs identity failed")?;
310    let mut filesystem = std::mem::MaybeUninit::<libc::statfs>::uninit();
311    if !metadata.is_dir() || metadata.dev() != fact.procfs_device || metadata.ino() != fact.procfs_inode
312        // SAFETY: root and the complete output record remain live.
313        || unsafe { libc::fstatfs(root.as_raw_fd(), filesystem.as_mut_ptr()) } != 0
314        || unsafe { filesystem.assume_init() }.f_type as u64
315            != reverie::syscalls::linux_abi::PROC_SUPER_MAGIC as u64
316    {
317        return Err("controller procfs identity changed");
318    }
319    let target_path = std::ffi::CString::new(tid.to_string()).unwrap();
320    let target_task = open_at(
321        root.as_raw_fd(),
322        &target_path,
323        libc::O_PATH | libc::O_DIRECTORY,
324    )
325    .map_err(|_| "controller mapping target directory open failed")?;
326    // thread-self, not self: an OS thread can have its own mount namespace.
327    let reader_task = open_at(
328        root.as_raw_fd(),
329        c"thread-self",
330        libc::O_PATH | libc::O_DIRECTORY,
331    )
332    .map_err(|_| READER_CONTEXT_UNAVAILABLE)?;
333    let target_before = ReaderContext::open(&target_task, false)?;
334    let reader_before = ReaderContext::open(&reader_task, true)?;
335    if target_before.identity != reader_before.identity {
336        return Err(READER_CONTEXT_DIFFERS);
337    }
338    // Linux renders NUMA file= paths relative to the reading task's root.
339    // The normal Hermit controller completes container setup before entering
340    // its current-thread runtime; its subsequent workers do not chroot/setns
341    // or mutate a shared fs_struct. This synchronous, no-yield request retains
342    // the grant/pidfd while native exclusion parks the target's only FS sharer.
343    // That stability is a required controller contract, not an ABA guarantee
344    // inferred from endpoint equality for arbitrary embedding callbacks.
345    let read = |leaf| {
346        let file = open_at(target_task.as_raw_fd(), leaf, libc::O_RDONLY)
347            .map_err(|_| "controller mapping file open failed")?;
348        let mut bytes = Vec::new();
349        file.take((MAX_VIEW_BYTES + 1) as u64)
350            .read_to_end(&mut bytes)
351            .map_err(|_| "controller mapping file read failed")?;
352        if bytes.len() > MAX_VIEW_BYTES {
353            return Err("controller mapping file exceeds fixed capacity");
354        }
355        Ok(bytes)
356    };
357    let maps = read(c"maps")?;
358    let numa = read(c"numa_maps")?;
359    let target_after = ReaderContext::open(&target_task, false)?;
360    let reader_after = ReaderContext::open(&reader_task, true)?;
361    if target_after.identity != target_before.identity
362        || reader_after.identity != reader_before.identity
363    {
364        return Err(READER_CONTEXT_DIFFERS);
365    }
366    Ok((maps, numa))
367}

369/// Validate the controller echo and project raw NUMA bytes only after the
370/// backend has restored its executors. Completion/rejection is owned by capture.
371fn project_captured_reply(
372    token: MappingCaptureToken,
373    response: (Option<LogicalTime>, GlobalResponse),
374    regions: &[reverie::RuntimePrivateMemoryRegion],
375) -> Result<Vec<u8>, String> {
376    match response {
377        (
378            None,
379            GlobalResponse::MappingCapture(Reply::Captured {
380                token: echoed,
381                maps,
382                numa,
383            }),
384        ) if echoed == token => {
385            crate::procfs::private_maps::project_numa(&maps, &numa, regions).map_err(str::to_owned)
386        }
387        (None, GlobalResponse::MappingCapture(Reply::Refused(reason))) => Err(reason),
388        _ => Err("controller reply identity or framing mismatched".into()),
389    }
390}

392pub(crate) fn capture<G, T>(guest: &G, target: &MapsTarget) -> Result<Vec<u8>, Error>
393where
394    G: Guest<Detcore<T>>,
395    T: RecordOrReplay,
396{
397    let fail = |reason| {
398        Error::Tool(anyhow::anyhow!(
399            "runtime-private numa_maps view unavailable: {reason}"
400        ))
401    };
402    let fact = guest
403        .controller_capture_fact()
404        .ok_or_else(|| fail("controller authority is unavailable".to_owned()))?;
405    let state = guest.thread_state();
406    let time = state.thread_logical_time.clone();
407    let mm = state.mm_id;
408    let request = |operation| {
409        (
410            time.clone(),
411            mm,
412            GlobalRequest::MappingCapture(Request(operation)),
413        )
414    };
415    let token = match guest.try_send_rpc_sync(request(Operation::Prepare(target.clone(), fact))) {
416        Ok((None, GlobalResponse::MappingCapture(Reply::Prepared(token)))) => token,
417        Ok((None, GlobalResponse::MappingCapture(Reply::Refused(reason)))) => {
418            return Err(fail(reason));
419        }
420        _ => return Err(fail("controller preparation failed".into())),
421    };
422    let prepared = reverie::PreparedMappingCapture::new(
423        token,
424        target.private.clone(),
425        request(Operation::Read(token)),
426    )
427    .map_err(|error| fail(error.to_string()))?;
428    let response = guest
429        .try_mapping_capture(prepared)
430        .map_err(|error| fail(error.to_string()))?;
431    let projected = project_captured_reply(token, response, &target.private.regions).map_err(fail);
432    // Returning from try_mapping_capture establishes restoration independently
433    // of whether the controller view/projector subsequently accepts the bytes.
434    // A rejected view still poisons completion, with its actual reason.
435    let rejection = projected.as_ref().err().map(ToString::to_string);
436    let completed = guest.try_send_rpc_sync(request(Operation::Complete(token, rejection)));
437    let projected = projected?;
438    if !matches!(
439        completed,
440        Ok((None, GlobalResponse::MappingCapture(Reply::Completed)))
441    ) {
442        return Err(fail(
443            "controller did not accept completed restoration".into(),
444        ));
445    }
446    Ok(projected)
447}

453    const MAPS: &[u8] = b"1000-2000 r--p 00000000 00:00 0 /guest\n\
454        2000-3000 rw-p 00000000 00:00 0 /runtime\n";
455    const GUEST_NUMA: &[u8] = b"1000 default file=/guest unknown=kept N0=7\n";
456    const PRIVATE_NUMA: &[u8] = b"2000 default file=/runtime N0=9\n";
457    const REGIONS: &[reverie::RuntimePrivateMemoryRegion] =
458        &[reverie::RuntimePrivateMemoryRegion {

464    fn token() -> MappingCaptureToken {
465        MappingCaptureToken {
466            nonce: 17,
467            identity: MappingCaptureIdentity {
468                process: 3,
469                thread: 3,

497        let token = token();
498        type TokenMutation = (&'static str, fn(&mut MappingCaptureToken));
499        let mutations: [TokenMutation; 10] = [
500            ("nonce", |token| token.nonce += 1),
501            ("process", |token| token.identity.process += 1),
502            ("thread", |token| token.identity.thread += 1),
503            ("address_space[0]", |token| {
504                token.identity.address_space[0] += 1
505            }),
506            ("address_space[1]", |token| {
507                token.identity.address_space[1] += 1
508            }),
509            ("private_owner[0]", |token| {
510                token.identity.private_owner[0] += 1
511            }),
512            ("private_owner[1]", |token| {
513                token.identity.private_owner[1] += 1
514            }),
515            ("private_owner[2]", |token| {
516                token.identity.private_owner[2] += 1
517            }),
518            ("private_owner[3]", |token| {
519                token.identity.private_owner[3] += 1
520            }),
521            ("private_version", |token| {
522                token.identity.private_version += 1
523            }),
524        ];
525        for (field, mutate) in mutations {
526            let mut echoed = token;
527            mutate(&mut echoed);
528            assert_ne!(echoed, token, "ineffective token mutation: {field}");
529            assert_eq!(
530                project_captured_reply(token, (None, captured(echoed)), REGIONS).unwrap_err(),
