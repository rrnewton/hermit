83impl InGuestSleepSigalrm {
84    /// The snapshot for the guest's `mask` and the signals it ignores and
85    /// catches, with SIGALRM's disposition and blocked bit taken from them.
86    pub(crate) fn from_actions(mask: u64, ignored: u64, caught: u64) -> Self {
87        let alarm = crate::syscalls::kernel_sigset_bit(libc::SIGALRM);
88        let disposition = if caught & alarm != 0 {
89            SigalrmSleepDisposition::Handled
90        } else if ignored & alarm != 0 {
91            SigalrmSleepDisposition::Ignored
92        } else {
93            SigalrmSleepDisposition::Default
94        };
95        Self {
96            disposition,
97            blocked: mask & alarm != 0,
98            mask,
99            ignored,
100            caught,
101        }
102    }
103
104    pub(crate) fn ends_process(self) -> bool {
105        self.disposition == SigalrmSleepDisposition::Default && !self.blocked
106    }

108    /// Whether a `signal` the scheduler sends while the sleep waits leaves it
109    /// asleep, as Linux leaves a `pause` or `nanosleep` for it: blocked, it
110    /// stays pending and wakes nothing (`complete_signal` wakes only a task
111    /// that `wants_signal`, kernel/signal.c:946-948 at v7.1.3); unblocked and
112    /// ignored, `SIG_IGN` or `SIG_DFL` for SIGCHLD, SIGURG, SIGWINCH or
113    /// SIGCONT, it is discarded as it is sent (`prepare_signal` returns
114    /// `!sig_ignored`, :935; `sig_handler_ignored`, :77-82; `sig_ignored`,
115    /// :106-125). Only a caught signal ends the sleep with EINTR; one whose
116    /// default action ends or stops the process is left to those paths.
117    pub(crate) fn send_leaves_asleep(self, signal: i32) -> bool {
118        let bit = crate::syscalls::kernel_sigset_bit(signal);
119        (self.mask | crate::syscalls::ignored_at_generation(self.ignored, self.caught)) & bit != 0
120    }

167/*
168NOTE [Blocking Syscalls via Internal Polling]
169---------------------------------------------
170
171A system like hermit which sequentializes execution must have the ability to detect
172blocking operations, so as not to deadlock the system. (This remains true irrespective of
173whether sequentialization is relaxed to admit some parallelism.)
174
175One option is for hermit to *precisely* model everything that may cause a syscall to
176block, and run the syscall only when it is certain that it can complete.  This is the
177approach with the precise implementation of futexes, for example, but it is tricky in
178general and requires hermit recapitulate a lot of the logic in the Linux kernel.
179
180Hence the need for this quick-and-dirty alternative of polling for internal blocking
181operations. The idea is that many potentially-blocking syscalls have the option to be run
182in a non-blocking fashion. This concept is captured by the `NonBlockableSyscall` trait.
183
184The way the polling strategy works is that the handler inside the guest converts the
185syscall to its nonblocking form (or ensures it runs nonblocking via file descriptor
186configuration).  The handler executes the syscall in a loop, while increasing the
187`poll_attempt` to make it visible to the scheduler that these are retries of a [logically]
188blocking operation, rather than new operations.
189
190The scheduler can then interleave these polling operations into the deterministic
191linearization of all guest side effects, just like any other operations.  However, it is a
192good idea to have some form of "backoff" when polling, because this strategy can
193asymptotically increase the total number of syscalls performed. For example, there are a
194large number of threads polling, and one thread productively running, we don't want to
195poll every thread once for every step of productive work. Conversely, we want to guarantee
196that no poller is starved indefinitely, because that could lead to live lock in the
197presence of any kind of "busy waiting".
198
199Determinism And linearization assumptions
200-----------------------------------------
201
202Why is this strategy deterministic?  How can we guarantee that on two separate executions,
203the same syscall `S` will retry exactly `N` times at times `T_1 ... T_n`?
204
205The assumption we make here is that all syscalls, even across multiple cores, act as
206atomic operations on the kernel state --- from the guest's perspective.  That is, halfway
207through executing a guest, we can describe the prior history of system as the linear
208series of syscalls committed to the kernel. Does this actually hold for Linux?  It doesn't
209seem to be documented anywhere, and Linux doesn't have anything like a formal semantics.
210For most syscalls we treat their man pages as the spec, and they don't answer questions
211like this.
212
213If we execute a guest entirely on one core, then only a weaker form of the assumption
214needs to hold.
215
216The upshot is, because we make this "linearizability of syscalls" assumption, each polling
217attempt for a given syscall is assumed to deterministically succeed or fail based on the
218history of syscalls that committed before it.
219
220The troublesome race
221--------------------
222
223One may wonder why not simply implement polling by *letting* the operation block,
224backgrounding the thread, and then checking in at each polling point to see if the
225operation has completed *without* incurring a user/kernel context switch and executing the
226syscall again. This would indeed be great, but it would require an oracle that can answer
227the question: "has thread X unblocked yet"?
228
229### A non-solution dead-end:
230
231We can recapture control flow immediately after any blocking syscall (in the handler), but
232that "posthook" won't run until (1) the syscall unblocks, and (2) the OS gets around to
233actually scheduling the thread.  Thus, inside the scheduler, we are racing with threads
234like this that are unblocked and in the process of attempting to report for duty.  Such
235threads can signal their readiness by tweaking the scheduler data structures, but those
236updates arrive completely asynchronously.  For example, if a syscall unblocks on scheduler
237turn 100 in one run, how can we know it won't arrive late at turn 101 on the next run?  We
238can't determinize the point that the formerly-blocked thread rejoins the scheduler pool.
239Not unless we do some huge (realtime) wait at each scheduler turn to ensure the OS has had
240a chance to run any recently-unblocked threads!
241
242### Future work: efficient oracle
243
244This nondeterministic-rejoining-the-pool strategy is exactly what we do for *externally*
245blocked syscalls.  But for internally-blocked ones that we want to determinize, then we
246would need a more efficient "has-unblocked?" oracle.  For example, if we had a kernel
247module exposing a safe predicate function that reveals this information to userspace. It's
248possible that the /proc file system exposes enough information to synchronously poll a
249threads status.
250
251But solving this may be extremely difficult, or impossible.  If the syscall that ran
252before us unblocked a thread, how long does that status update take to propogate through
253the kernel itself?  If one futex wake syscall unblocks 1000 other threads, their status
254would have to atomically change immediately by the time the futex wake returns to the
255guest, or at least by the start of the next syscall (or oracle call).
256
257This atomicity property seems hard to guarantee for the OS. Instead, the current polling
258strategy that uses nonblocking syscalls is much more localized.  It doesn't treat the
259kernel state as monolithic and require atomic updates from the guest perspective, but
260rather it only requires atomic treatment of whatever state is relevant to syscalls
261operating on that individual resource (futex, file descriptor etc) --- a much less
262stringent requirement.
263
264*/

444impl ResourceID {
445    /// Whether a granted request for this resource makes its thread's call a
446    /// kernel-run blocking call: the call runs outside the schedule, in the
447    /// kernel, where it may sleep for as long as the kernel decides, while
448    /// the scheduler commits other threads' turns. The thread rejoins with a
449    /// [`ResourceID::BlockedExternalContinue`] request (a failed vfork with
450    /// [`ResourceID::VforkFailed`]), filed at a moment host timing chooses.
451    /// So, in committed state, the call is running from the grant of this
452    /// resource until the grant of that request.
453    ///
454    /// This is the published list of kernel-run blocking calls of the A5
455    /// signal-delivery design (dev-hermit
456    /// `ai_docs/2026-10-16-a5-signal-delivery-design.md`, 16.11). Step 3c.1
457    /// refuses by name a caught or default-fatal signal that a committed send
458    /// makes pending for a thread in one of them, because the in-guest
459    /// runtime's mask keeps the signal pending where Linux would interrupt
460    /// the call. The request names the call: its `fyi` is the syscall's name,
461    /// or `clone_vfork`.
462    ///
463    /// That reason holds for a call that sleeps under the thread's own mask.
464    /// Three forms instead hand the kernel the guest's temporary mask
465    /// (`Resources::blocked_signal_mask`): `rt_sigsuspend` as
466    /// `BlockingRtSigsuspend`, `rt_sigsuspend` with a signal already pending
467    /// as `BlockingExternalIO`, and a masked `pselect6` with `nfds` above 64.
468    /// The kernel installs that mask for the call, so a caught signal it
469    /// unblocks interrupts the call and is delivered on the return into
470    /// runtime code. That is the temporary-mask problem of 16.11 (step 3c.w),
471    /// not a signal kept pending.
472    ///
473    /// A call that may block inside its own turn is not one. The scheduler
474    /// commits no other turn until that thread files its next request, so no
475    /// committed send and no expiry can make a signal pending for it
476    /// meanwhile.
477    ///
478    /// Which calls file these requests is decided by their handlers, from
479    /// Detcore's descriptor model and configuration. The list applies to a
480    /// sequentialized `hermit run`, the configuration in which the in-guest
481    /// backends run (record and replay run on ptrace). Without sequentialized
482    /// threads, `resource_request` sends no request to the scheduler, so
483    /// nothing is granted in the background and there is no committed state
484    /// to key on. In a sequentialized `hermit run` they are:
485    /// - `rt_sigsuspend`, always, except on a backend whose `rt_sigsuspend`
486    ///   never sleeps (`rt_sigsuspend_never_sleeps`, KVM);
487    /// - `vfork`, and `clone` or `clone3` with `CLONE_VFORK` (the in-guest
488    ///   runtime refuses those two before they run);
489    /// - `read` and `readv` of an inherited external input;
490    /// - `read`, `recvfrom`, `recvmsg`, `recvmmsg`, `sendto`, `sendmsg`,
491    ///   `sendmmsg`, `connect`, `accept` and `accept4` on a descriptor that is
492    ///   blocking both logically and physically, or that Detcore cannot
493    ///   classify (`execute_nonblockable_fd_syscall`). Detcore makes the
494    ///   sockets, pipes and eventfds it creates physically nonblocking, and
495    ///   its timerfds while time is virtual, so these are, for example,
496    ///   inotify and signalfd descriptors and sockets Detcore did not create;
497    /// - `select` and `pselect6` with `nfds` above 64 and a timeout that is
498    ///   not zero (a zero timeout runs in the turn first);
499    /// - `futex` under `--debug-futex-mode=external`. (A `restart_syscall`
500    ///   never resumes such a wait: Detcore keeps a futex restart record only
501    ///   in the precise and polling modes, which external mode returns
502    ///   before.)
503    ///
504    /// The scheduler grants exactly these resources in its background arm
505    /// (`Scheduler::block_for_one_resource`, which asserts it). The work-span
506    /// report also counts the continuation (`BlockedExternalContinue`) as a
507    /// wait outside the schedule, though the kernel call has returned by then.
508    ///
509    /// `write`, `writev` and `readv` on a physically blocking descriptor
510    /// (for `readv`, one that is not an inherited input), `preadv2` and
511    /// `pwritev2` at offset -1 on a stream, and a masked
512    /// `epoll_pwait` or `epoll_pwait2` on a backend without
513    /// `supports_blocked_wait_signal_interruption`, run in the turn instead.
514    ///
515    /// Exhaustive, so that a new variant has to be classified.
516    pub(crate) fn is_kernel_run_blocking(&self) -> bool {
517        match self {
518            ResourceID::BlockingExternalIO(_)
519            | ResourceID::BlockingVfork(_)
520            | ResourceID::BlockingRtSigsuspend(_) => true,
521            ResourceID::FileContents(_)
522            | ResourceID::FileMetadata(_)
523            | ResourceID::DirectoryContents(_)
524            | ResourceID::MemAddrSpace(_)
525            | ResourceID::Path(_)
526            | ResourceID::PathsTransitive(_)
527            | ResourceID::Device(_)
528            | ResourceID::Exit { .. }
529            | ResourceID::ParentContinue { .. }
530            | ResourceID::SleepUntil(_)
531            | ResourceID::InternalIOPolling
532            | ResourceID::TraceReplay
533            | ResourceID::FutexWait
534            | ResourceID::VforkFailed(_)
535            | ResourceID::BlockedExternalContinue(_)
536            | ResourceID::PriorityChangePoint(..)
537            | ResourceID::WaitChild { .. }
538            | ResourceID::WaitPhysicalChild(_)
539            | ResourceID::InboundSignal(_)
540            | ResourceID::WaitidSignals(_)
541            | ResourceID::SchedYield
542            | ResourceID::HappensBeforeCheckpoint(_)
543            | ResourceID::HappensBeforeSyscallCheckpoint { .. }
544            | ResourceID::HappensBeforePosthookInterrupted { .. }
545            | ResourceID::InGuestFatalAlarm => false,
546        }
547    }
548}

571/// The map type of [`Resources::resources`]: a `HashMap` whose hasher has
572/// fixed keys.
573///
574/// A `RandomState` map draws its keys from the host's random source the first
575/// time a thread builds one. Where `getrandom` is unavailable, std falls back
576/// to a `/dev/urandom` descriptor that it opens and keeps open for the life of
577/// the process. Detcore builds these requests on the guest's syscall paths, so
578/// in-guest that descriptor would land in the guest's descriptor table, and
579/// only on some hosts. `DefaultHasher::new` has fixed zero keys, so building
580/// this map draws no randomness and opens nothing. Detcore's other
581/// `RandomState` maps are unchanged and can still open that descriptor.
582///
583/// The hash seed is always the same, so the iteration order depends only on
584/// the binary, the map's contents (the `ResourceID` values, including any
585/// host-derived value one carries) and its insertion and removal history. That
586/// order is not sorted or canonical, and it is not promised across toolchains.
587pub type ResourceMap = HashMap<ResourceID, Permission, BuildHasherDefault<DefaultHasher>>;

978    /// The runnable copy of a skipped retry that
979    /// `Scheduler::upgrade_polled_to_runnable` puts in its place:
980    /// `poll_attempt` 0, with the retry number kept in `upgraded_from_attempt`.
981    pub(crate) fn upgraded_to_runnable(&self) -> Resources {
982        assert!(
983            self.slice_boundary.is_none(),
984            "slice-boundary annotation must be consumed before poll upgrade"
985        );
986        let mut runnable = self.clone();
987        runnable.upgraded_from_attempt = runnable.upgraded_from_attempt.max(runnable.poll_attempt);
988        runnable.poll_attempt = 0;
989        runnable
990    }

1030    /// Similar to extend, but union all permissions within the intersection.
1031    /// Union can only be called on resources with matching `tid`.
1032    /// (NB: equivalent to Haskell `Data.Map.unionWith perm_union`)
1033    pub fn union(&mut self, other: &Resources) {
1034        assert_eq!(self.tid, other.tid);
1035        assert!(
1036            self.slice_boundary.is_none() && other.slice_boundary.is_none(),
1037            "slice-boundary protocol refuses merging an annotated operation"
1038        );
1039        // Union describes a combined operation, not the exact wait whose owner
1040        // captured the snapshot. Never promote either operand's authority.
1041        self.in_guest_sleep_sigalrm = None;
1042        for (id, perm2) in other.resources.iter() {
1043            match self.resources.entry(id.clone()) {
1044                Entry::Occupied(mut e) => {
1045                    let perm1 = e.get_mut();
1046                    *perm1 = perm1.union(perm2)
1047                }
1048                Entry::Vacant(e) => {
1049                    e.insert(perm2.clone());
1050                }
1051            }
1052        }
1053        match (self.signal_interrupt_errno, other.signal_interrupt_errno) {
1054            (None, interrupt) => self.signal_interrupt_errno = interrupt,
1055            (Some(left), Some(right)) => assert_eq!(left, right),
1056            (Some(_), None) => {}
1057        }
1058        self.backend_runtime_bootstrap |= other.backend_runtime_bootstrap;
1059        match (self.blocked_signal_mask, other.blocked_signal_mask) {
1060            (None, mask) => self.blocked_signal_mask = mask,
1061            (Some(left), Some(right)) => assert_eq!(left, right),
1062            (Some(_), None) => {}
1063        }
1064        self.blocked_signal_mask_unknown |= other.blocked_signal_mask_unknown;
1065        // Both parts are the same thread's, stopped at the same request.
1066        match (self.guest_signal_mask, other.guest_signal_mask) {
1067            (None, mask) => self.guest_signal_mask = mask,
1068            (Some(left), Some(right)) => assert_eq!(left, right),
1069            (Some(_), None) => {}
1070        }
1071        if self.signal_state_task.is_none() {
1072            self.signal_state_task = other.signal_state_task;
1073        }
1074        self.decides_signal_interruption_itself |= other.decides_signal_interruption_itself;
1075        // A combined operation is a continuation only if both parts are.
1076        self.continues_sigreturn &= other.continues_sigreturn;
1077        self.private_sigchld_decides |= other.private_sigchld_decides;
1078        self.replay_served_from_log |= other.replay_served_from_log;
1079        self.polls_guest_state_only |= other.polls_guest_state_only;
1080        self.polls_a_named_fifo |= other.polls_a_named_fifo;
1081        self.relies_on_guest_held_marks |= other.relies_on_guest_held_marks;
1082        self.passes_descriptors_out |= other.passes_descriptors_out;
1083        self.waits_until_nothing_else_can_run |= other.waits_until_nothing_else_can_run;
1084        // A combined operation is a zero-length sleep only if both parts are.
1085        self.zero_length_sleep &= other.zero_length_sleep;
1086        self.previous_attempt_made_progress |= other.previous_attempt_made_progress;
1087        self.upgraded_from_attempt = self.upgraded_from_attempt.max(other.upgraded_from_attempt);
1088        // A merged request waits for the earlier of two deadlines.
1089        self.poll_deadline = match (self.poll_deadline, other.poll_deadline) {
1090            (Some(left), Some(right)) => Some(left.min(right)),
1091            (left, right) => left.or(right),
1092        };
1093        // Either part expiring on its own is progress the merged wait can
1094        // still make, as for the deadline above.
1095        self.poll_expires |= other.poll_expires;
1096    }
