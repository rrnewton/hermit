9//! The identity of a run's RCB clock, which decides which recorded histories
10//! the run may execute and what identity the histories it writes state.
11//!
12//! A run is prepared in four phases before anything that counts or writes
13//! exists (`prepare_run`):
14//!
15//! 1. qualify: the backend's clock contract, the host's PMU profile and the
16//!    process's installed PMU configuration give the effective identity,
17//!    checked against the host's prepared claim; nothing is changed. Whether
18//!    Detcore observes the counter (a maximum timeslice) or charges it as
19//!    virtual time (`no_rcb_time`) does not qualify or unqualify it;
20//! 2. admit every supplied executable input against it;
21//! 3. pin the qualified clock key once, under the PMU state's lock, so that
22//!    no counter of another key is installed afterwards;
23//! 4. build the scheduler and writer from the admitted inputs and the
24//!    committed identity.
25//!
26//! A refusal in phases 1 to 3 comes before any pin, writer, counter or guest.
27

44/// The identity of the RCB coordinates a run counts.
45#[derive(Clone, Debug, Eq, PartialEq)]
46pub enum ProducerIdentity {
47    /// A qualified clock: its metric and the key it pins.
48    Known {
49        /// What the clock counts.
50        metric: RcbMetric,
51        /// The clock key pinned for the run.
52        key: ClockKey,
53    },
54    /// A backend-local clock stated by its opaque, versioned token. Its
55    /// coordinates are executable only under the same token; Detcore
56    /// compares tokens and never interprets them.
57    BackendLocal(&'static str),
58    /// Not a qualified clock, for the reason given.
59    Unqualified(String),
60}
61
62impl ProducerIdentity {
63    /// The identity a history this run writes states.
64    pub fn history_identity(&self) -> HistoryIdentity {
65        match self {
66            Self::Known { metric, .. } => HistoryIdentity::Known(*metric),
67            Self::BackendLocal(token) => HistoryIdentity::BackendLocal((*token).to_owned()),
68            Self::Unqualified(reason) => HistoryIdentity::Unqualified(reason.clone()),
69        }
70    }
71}

108/// Where phase 1 reads the host and the PMU state. Production reads this
109/// host's CPUID and the process's PMU state; tests supply both.
110pub struct ClockSources<'a> {
111    /// This host's qualification.
112    pub host: Result<QualifiedClock, UnqualifiedReason>,
113    /// The process's PMU state.
114    pub state: &'a PmuState,
115}

117impl ClockSources<'static> {
118    /// This host and this process.
119    pub fn process() -> Self {
120        // Unit tests do not depend on the host running them: unless a test
121        // supplies its own sources, they see a fixed qualified host and a PMU
122        // state of their own, which no real counter reads.
123        #[cfg(test)]
124        return test_sources::current().unwrap_or_else(test_sources::default_sources);
125        #[cfg(not(test))]
126        Self {
127            host: reverie::pmu_config::resolve_host_clock(),
128            state: reverie::pmu_config::pmu_state(),
129        }
130    }

133/// Phase 1. The effective identity comes from the backend's clock contract,
134/// the host's profile and the installed PMU configuration, whether or not
135/// Detcore consumes the counter; `cfg.rcb_clock_claim` is only checked
136/// against it. A `Known` claim the effective clock contradicts is refused; a
137/// run nobody prepared is unqualified.
138pub fn validate_run_clock(
139    cfg: &Config,
140    sources: &ClockSources<'_>,
141) -> Result<ValidatedClock, String> {
142    let effective: Result<(QualifiedClock, PinTicket), String> =
143        match cfg.backend.rcb_clock_contract {
144            None => Err("no-mechanism-contract".to_owned()),
145            Some(_) => match &sources.host {
146                Err(reason) => Err(reason.to_string()),
147                Ok(host) => sources
148                    .state
149                    .qualify(host)
150                    .map(|ticket| (host.clone(), ticket))
151                    .map_err(|refusal| format!("pmu-state: {refusal}")),
152            },
153        };
154    match (cfg.rcb_clock_claim, effective) {
155        (RcbClockClaim::BackendLocal(token), _) if token.trim().is_empty() => Err(
156            "hermit refused the run: it was prepared for an empty backend-local RCB clock \
157             identity"
158                .to_owned(),
159        ),
160        (RcbClockClaim::BackendLocal(token), _) if cfg.backend.rcb_clock_contract.is_some() => {
161            Err(format!(
162                "hermit refused the run: it was prepared for backend-local RCB clock \
163                 {token}, but the backend states a qualified shared RCB clock contract"
164            ))
165        }
166        (RcbClockClaim::BackendLocal(token), _) => Ok(ValidatedClock {
167            producer: ProducerIdentity::BackendLocal(token),
168            ticket: None,
169        }),
170        (RcbClockClaim::Unset, Err(reason)) => Ok(ValidatedClock {
171            producer: ProducerIdentity::Unqualified(format!("claim-unset: {reason}")),
172            ticket: None,
173        }),
174        (RcbClockClaim::Unset, Ok(_)) => Ok(ValidatedClock {
175            producer: ProducerIdentity::Unqualified("claim-unset".to_owned()),
176            ticket: None,
177        }),
178        (RcbClockClaim::Known(claimed), Ok((host, ticket))) if claimed == ticket.key() => {
179            Ok(ValidatedClock {
180                producer: ProducerIdentity::Known {
181                    metric: host.metric,
182                    key: ticket.key(),
183                },
184                ticket: Some(ticket),
185            })
186        }
187        (RcbClockClaim::Known(claimed), Ok((_, ticket))) => Err(format!(
188            "hermit refused the run: it was prepared for RCB clock {claimed:?}, but the \
189             effective clock is {:?}",
190            ticket.key()
191        )),
192        (RcbClockClaim::Known(claimed), Err(reason)) => Err(format!(
193            "hermit refused the run: it was prepared for RCB clock {claimed:?}, but the \
194             effective clock is not qualified ({reason})"
195        )),
196    }
197}

199impl ValidatedClock {
200    /// The effective identity, before it is committed.
201    pub fn producer(&self) -> &ProducerIdentity {
202        &self.producer
203    }
204
205    /// Phase 2: may `recording`, read from `source`, run under this clock?
206    pub fn admit(&self, source: &str, recording: &Recording) -> Result<(), String> {
207        admit_identity(source, recording.identity(), &self.producer)
208    }
209
210    /// Phase 3: pins the qualified key in `state`, rechecking it under the
211    /// lock. Unqualified runs pin nothing.
212    pub fn commit(self, state: &PmuState) -> Result<CommittedClock, String> {
213        if let Some(ticket) = self.ticket {
214            state
215                .commit_pin(ticket)
216                .map_err(|refusal| format!("hermit refused the run: {refusal}"))?;
217        }
218        Ok(CommittedClock {
219            producer: self.producer,
220        })
221    }
222}

224/// May a history stating `recorded`, read from `source`, run under
225/// `consumer`? A qualified history needs a qualified clock of the same
226/// metric (a raw event or model need not match); a backend-local history
227/// needs the same backend-local token. Every other pairing is refused
228/// naming both identities.
229pub fn admit_identity(
230    source: &str,
231    recorded: Option<&HistoryIdentity>,
232    consumer: &ProducerIdentity,
233) -> Result<(), String> {
234    match (recorded, consumer) {
235        (None, _) => Err(format!(
236            "hermit refused to replay {source}: it states no RCB coordinate identity, so the \
237             unit of its recorded coordinates is unknown (it was written before identities were \
238             recorded, or by hand). Record it again with this build, or replay it with the \
239             build that made it."
240        )),
241        (Some(HistoryIdentity::Unqualified(reason)), _) => Err(format!(
242            "hermit refused to replay {source}: it was recorded with an unqualified RCB clock \
243             ({reason}), so its coordinates cannot be executed."
244        )),
245        (
246            Some(HistoryIdentity::BackendLocal(recorded)),
247            ProducerIdentity::BackendLocal(consumer),
248        ) => {
249            if recorded.as_str() == *consumer {
250                Ok(())
251            } else {
252                Err(format!(
253                    "hermit refused to replay {source}: it counts backend-local RCB clock \
254                     {recorded}, and this run counts {consumer}."
255                ))
256            }
257        }
258        (Some(HistoryIdentity::BackendLocal(recorded)), ProducerIdentity::Known { metric, .. }) => {
259            Err(format!(
260                "hermit refused to replay {source}: it counts backend-local RCB clock \
261             {recorded}, and this run counts {}.",
262                metric.id()
263            ))
264        }
265        (Some(HistoryIdentity::Known(recorded)), ProducerIdentity::BackendLocal(consumer)) => {
266            Err(format!(
267                "hermit refused to replay {source}: it counts {}, and this run counts \
268             backend-local RCB clock {consumer}.",
269                recorded.id()
270            ))
271        }
272        (Some(HistoryIdentity::BackendLocal(recorded)), ProducerIdentity::Unqualified(reason)) => {
273            Err(format!(
274                "hermit refused to replay {source} ({recorded}): this run's RCB clock is not \
275             qualified ({reason})."
276            ))
277        }
278        (Some(HistoryIdentity::Known(metric)), ProducerIdentity::Unqualified(reason)) => {
279            Err(format!(
280                "hermit refused to replay {source} ({}): this run's RCB clock is not qualified \
281                 ({reason}).",
282                metric.id()
283            ))
284        }
285        (Some(HistoryIdentity::Known(recorded)), ProducerIdentity::Known { metric, .. }) => {
286            if recorded == metric {
287                Ok(())
288            } else {
289                Err(format!(
290                    "hermit refused to replay {source}: it counts {}, and this run counts {}.",
291                    recorded.id(),
292                    metric.id()
293                ))
294            }
295        }
296    }
297}

299/// The identity a history derived from both `a` and `b` (each a source name
300/// and the identity it states) keeps: theirs, which must be the same. It is
301/// copied from the sources, never taken from the current host, so a
302/// transformation cannot change what its coordinates count.
303pub fn combined_identity(
304    a: (&str, Option<&HistoryIdentity>),
305    b: (&str, Option<&HistoryIdentity>),
306) -> Result<Option<HistoryIdentity>, String> {
307    if a.1 == b.1 {
308        return Ok(a.1.cloned());
309    }
310    let show = |identity: Option<&HistoryIdentity>| match identity {
311        Some(identity) => identity.to_string(),
312        None => "no stated identity".to_owned(),
313    };
314    Err(format!(
315        "hermit refused to combine {} ({}) with {} ({}): their RCB coordinates do not \
316         count the same thing, so no schedule mixing them can be replayed.",
317        a.0,
318        show(a.1),
319        b.0,
320        show(b.1),
321    ))
322}

324/// Phases 1 to 3's outcome, with the admitted inputs phase 4 builds from.
325pub(crate) struct PreparedRun {
326    /// The committed identity.
327    pub(crate) clock: CommittedClock,
328    /// The admitted `replay_schedule_from` recording, with its stack
329    /// selections resolved.
330    pub(crate) schedule: Option<AdmittedSchedule>,
331    /// The admitted `replay_preemptions_from` history.
332    pub(crate) history: Option<HistoryReader>,
333}
334
335/// Phases 1 to 3 against `sources`.
336pub(crate) fn prepare_run_with(
337    cfg: &Config,
338    sources: &ClockSources<'_>,
339) -> Result<PreparedRun, PrepareError> {
340    let validated = validate_run_clock(cfg, sources).map_err(PrepareError::Refused)?;
341    let schedule = match &cfg.replay_schedule_from {
342        None => None,
343        Some(path) => {
344            let recording = Recording::read(path).map_err(|error| {
345                // A format refusal keeps its policy exit, as the scheduler's
346                // loader gave it before preparation existed; only unreadable
347                // or malformed files keep the loader's panic class.
348                if error
349                    .chain()
350                    .any(|cause| cause.is::<crate::preemptions::RecordFormatRefusal>())
351                {
352                    PrepareError::Refused(format!("{error:#}"))
353                } else {
354                    PrepareError::Unreadable(error)
355                }
356            })?;
357            recording
358                .require_strict_mode(cfg.strict_schedule_boundaries)
359                .map_err(|error| PrepareError::Refused(error.to_string()))?;
360            // The same replay-input validation the priority route makes in
361            // `read_for_replay`, before identity admission, so a record that
362            // fails both takes the same refusal on either route. The
363            // scheduler validates the installed record again at construction.
364            recording
365                .validate_replay_inputs(&input_name("--replay-schedule-from", path), cfg.guest_ids)
366                .map_err(|refusal| PrepareError::Refused(refusal.to_string()))?;
367            validated
368                .admit(&input_name("--replay-schedule-from", path), &recording)
369                .map_err(PrepareError::Refused)?;
370            Some(
371                AdmittedSchedule::new(cfg, recording)
372                    .map_err(|refusal| PrepareError::Refused(refusal.to_owned()))?,
373            )
374        }
375    };
376    let history = match &cfg.replay_preemptions_from {
377        None => None,
378        Some(path) => {
379            let history = HistoryReader::read_for_replay(path, cfg.guest_ids).map_err(|error| {
380                if error.chain().any(|cause| {
381                    cause.is::<crate::preemptions::RecordAdmissionRefusal>()
382                        || cause.is::<crate::preemptions::RecordFormatRefusal>()
383                }) {
384                    PrepareError::Refused(format!("{error:#}"))
385                } else {
386                    PrepareError::Unreadable(error)
387                }
388            })?;
389            history
390                .require_legacy_mode()
391                .map_err(|error| PrepareError::Refused(error.to_string()))?;
392            validated
393                .admit(
394                    &input_name("--replay-preemptions-from", path),
395                    history.recording(),
396                )
397                .map_err(PrepareError::Refused)?;
398            Some(history)
399        }
400    };
401    // The priority history's callback checkpoints must agree with the launch
402    // decision; the scheduler installs them from this same object.
403    if let Some(refusal) = crate::scheduler::priority_checkpoint_mode_refusal(
404        cfg.annotated_priority_replay,
405        history.as_ref().map(HistoryReader::callback_checkpoints),
406    ) {
407        return Err(PrepareError::Refused(refusal));
408    }
409    // Faithful replay refuses a departed record before the pin too, unless
410    // the run explicitly explores edited histories; construction repeats
411    // the check on the installed objects.
412    if !cfg.replay_edited_preemptions {
413        if let Some(schedule) = &schedule {
414            schedule
415                .recording
416                .require_faithful_replay()
417                .map_err(|error| PrepareError::Refused(error.to_string()))?;
418        }
419        if let Some(history) = &history {
420            history
421                .recording()
422                .require_faithful_replay()
423                .map_err(|error| PrepareError::Refused(error.to_string()))?;
424        }
425    }
426    let clock = validated
427        .commit(sources.state)
428        .map_err(PrepareError::Refused)?;
429    Ok(PreparedRun {
430        clock,
431        schedule,
432        history,
433    })
434}

436/// Phases 1 to 3 for this process. A refusal ends the run with the policy
437/// exit, before any guest runs; an unreadable file keeps the loader's panic.
438pub(crate) fn prepare_run(cfg: &Config) -> PreparedRun {
439    match prepare_run_with(cfg, &ClockSources::process()) {
440        Ok(prepared) => prepared,
441        Err(PrepareError::Refused(reason)) => crate::scheduler::exit_on_scheduler_refusal(&reason),
442        Err(PrepareError::Unreadable(error)) => panic!("{error:?}"),
443    }
444}

461    /// Runs `f` with `host` and a fresh, leaked PMU state as this thread's
462    /// clock sources.
463    pub(crate) fn with<R>(
464        host: Result<QualifiedClock, UnqualifiedReason>,
465        f: impl FnOnce(&'static PmuState) -> R,
466    ) -> R {
467        let state: &'static PmuState = Box::leak(Box::new(PmuState::new()));
468        SOURCES.with(|slot| *slot.borrow_mut() = Some((host, state)));
469        struct Reset;
470        impl Drop for Reset {
471            fn drop(&mut self) {
472                SOURCES.with(|slot| *slot.borrow_mut() = None);
473            }
474        }
475        let _reset = Reset;
476        f(state)
477    }
