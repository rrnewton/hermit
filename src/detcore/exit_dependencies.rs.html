9//! Descriptors that would make a guest the server another guest's exit waits
10//! for, refused while process exits complete asynchronously.
11//!
12//! With `BackendCapabilities::process_exits_complete_asynchronously`, Detcore
13//! selects no turn between a process's exit grant and its physical exit. A native exit that needs another guest to act therefore
14//! never completes. Coord rulings A and D.1 (dev-hermit
15//! ai_docs/transient/liteinst-inguest-exit-control-design-20261006.md) refuse,
16//! by name, the two descriptors that make a guest such a server:
17//!
18//! - a FUSE or CUSE device (`/dev/fuse` or `/dev/cuse`): the exiting guest's
19//!   close of a file that guest serves sends a synchronous flush to it.
20//!   Recognised by the character device's number (misc major 10, minors 229
21//!   and 203), never by path. A plain file on a FUSE filesystem is not
22//!   refused: when no guest holds the device, the server is outside the guest
23//!   set.
24//! - a seccomp user-notification listener: an exit-time system call the
25//!   listener's filter traps waits for the listener's reply. Recognised by the
26//!   descriptor's anon inode, `anon_inode:seccomp notify`, never by a name.
27//!   Creating one (`SECCOMP_FILTER_FLAG_NEW_LISTENER`) is already refused,
28//!   because guests cannot install filters; an inherited filter with no
29//!   guest-held listener stays allowed.
30//!
31//! A descriptor is checked where it arrives: from an open, through
32//! `SCM_RIGHTS` (`recvmsg`, `recvmmsg`) or through `pidfd_getfd`
33//! (`Detcore::refuse_exit_dependencies`). The in-guest runtime checks the
34//! descriptors a process starts with ([`descriptor_exit_dependency`] on
35//! `/proc/self/fd`). userfaultfd is not refused (ruling D.2): the reference
36//! backend supports it, and a stuck exit is caught by the backend's watchdog.
37//!
38//! Detcore's refusal names the capability and the descriptor, never a backend.
39//! The backend layer, which knows which backend runs and what to try instead,
40//! adds that through [`set_backend_advice`].

57/// What the backend layer adds to a refusal; see [`set_backend_advice`].
58static BACKEND_ADVICE: std::sync::OnceLock<&'static str> = std::sync::OnceLock::new();
59
60/// Sets the sentence a refusal ends with: which backend refused, and what to
61/// run instead. The backend layer calls it once, before Detcore runs; a later
62/// call is ignored.
63pub fn set_backend_advice(advice: &'static str) {
64    let _ = BACKEND_ADVICE.set(advice);
65}
66
67/// The refusal for a guest holding `what` at descriptor `fd`, which arrived as
68/// `how`.
69fn refusal(what: &str, fd: RawFd, how: &str) -> String {
70    let mut refusal = format!(
71        "hermit: refusing a guest that holds {what} (descriptor {fd}, {how}): this backend \
72         completes process exits asynchronously, so no other guest runs until an exit is \
73         complete, and a guest serving that descriptor could make another guest's exit wait \
74         forever."
75    );
76    if let Some(advice) = BACKEND_ADVICE.get() {
77        refusal.push(' ');
78        refusal.push_str(advice);
79    }
80    refusal
81}

83/// The misc character devices' major number.
84const MISC_MAJOR: u64 = 10;
85/// `/dev/fuse` (`FUSE_MINOR` in the kernel's `miscdevice.h`).
86const FUSE_MINOR: u64 = 229;
87/// `/dev/cuse` (`CUSE_MINOR`).
88const CUSE_MINOR: u64 = 203;
89
90/// What `readlink` reports for a seccomp user-notification listener.
91pub const SECCOMP_NOTIFY_LINK: &str = "anon_inode:seccomp notify";
92
93/// The bytes of one received control buffer this module inspects; Linux caps a
94/// control buffer well below this.
95const MAX_CONTROL_BYTES: usize = 64 * 1024;
96
97/// The major and minor numbers of a `dev_t`, as glibc's `major` and `minor`.
98fn device_numbers(rdev: u64) -> (u64, u64) {
99    let major = ((rdev >> 8) & 0xfff) | ((rdev >> 32) & !0xfff);
100    let minor = (rdev & 0xff) | ((rdev >> 12) & !0xff);
101    (major, minor)
102}
103
104/// What a character device numbered `rdev` would make its holder serve, or
105/// `None`.
106pub fn device_exit_dependency(rdev: u64) -> Option<&'static str> {
107    match device_numbers(rdev) {
108        (MISC_MAJOR, FUSE_MINOR) => Some("a FUSE device (/dev/fuse)"),
109        (MISC_MAJOR, CUSE_MINOR) => Some("a CUSE device (/dev/cuse)"),
110        _ => None,
111    }
112}

114/// What the descriptor at `fd_link`, a `/proc/<pid>/fd/<n>` entry, would make
115/// its holder serve, or `None`. A descriptor that cannot be read is not
116/// refused.
117pub fn descriptor_exit_dependency(fd_link: &Path) -> Option<&'static str> {
118    use std::os::unix::ffi::OsStrExt;
119
120    use crate::util::raw_syscall;
121    // Raw readlinkat and newfstatat (see crate::util::raw_syscall): the
122    // in-guest runtime runs this inside the guest before Detcore installs.
123    let link = std::ffi::CString::new(fd_link.as_os_str().as_bytes()).ok()?;
124    let mut target = [0_u8; 64];
125    let length = unsafe {
126        raw_syscall(
127            libc::SYS_readlinkat,
128            [
129                libc::AT_FDCWD as u64,
130                link.as_ptr() as u64,
131                target.as_mut_ptr() as u64,
132                target.len() as u64,
133                0,
134                0,
135            ],
136        )
137    };
138    if length >= 0 && &target[..length as usize] == SECCOMP_NOTIFY_LINK.as_bytes() {
139        return Some("a seccomp user-notification listener");
140    }
141    // Follows the descriptor to the file it refers to.
142    let mut metadata: libc::stat = unsafe { std::mem::zeroed() };
143    let status = unsafe {
144        raw_syscall(
145            libc::SYS_newfstatat,
146            [
147                libc::AT_FDCWD as u64,
148                link.as_ptr() as u64,
149                (&raw mut metadata) as u64,
150                0,
151                0,
152                0,
153            ],
154        )
155    };
156    if status != 0 || metadata.st_mode & libc::S_IFMT != libc::S_IFCHR {
157        return None;
158    }
159    device_exit_dependency(metadata.st_rdev)
160}

162/// The first descriptor listed in `fd_directory` (a `/proc/<pid>/fd`
163/// directory) that would make its holder serve another guest's exit: its name
164/// in the directory and what it is. The in-guest runtime checks
165/// `/proc/self/fd` with it before installing Detcore. A directory that cannot
166/// be listed is an error, never a pass. Listed with
167/// [`crate::util::find_in_directory`]: this runs inside the guest, where
168/// `std::fs::read_dir` would leave the listing in the guest's heap.
169pub fn held_exit_dependency(
170    fd_directory: &Path,
171) -> std::io::Result<Option<(String, &'static str)>> {
172    crate::util::find_in_directory(fd_directory, |name| {
173        descriptor_exit_dependency(&fd_directory.join(name))
174            .map(|what| (name.to_string_lossy().into_owned(), what))
175    })
176}

178/// The descriptors a `struct msghdr` at `address` received through
179/// `SCM_RIGHTS`, read after the call: Linux updates `msg_controllen` to what it
180/// wrote.
181fn received_through<G: MemoryAccess>(memory: &G, header: &libc::msghdr) -> Vec<RawFd> {
182    if header.msg_control.is_null() || header.msg_controllen == 0 {
183        return Vec::new();
184    }
185    let Some(address) = reverie::syscalls::Addr::<u8>::from_raw(header.msg_control as usize) else {
186        return Vec::new();
187    };
188    let mut control = vec![0; header.msg_controllen.min(MAX_CONTROL_BYTES)];
189    if memory.read_exact(address, &mut control).is_err() {
190        return Vec::new();
191    }
192    received_descriptors(&control)
193}

195impl<T: RecordOrReplay> Detcore<T> {
196    /// Ends the run, by name, when `call`, which returned `ret`, gave the guest
197    /// a descriptor that would make it serve another guest's exit; see the
198    /// module documentation. Does nothing unless process exits complete
199    /// asynchronously.
200    pub(crate) async fn refuse_exit_dependencies<G: Guest<Self>>(
201        &self,
202        guest: &mut G,
203        call: &Syscall,
204        ret: i64,
205    ) {
206        if !self.cfg.backend.process_exits_complete_asynchronously || ret < 0 {
207            return;
208        }
209        let (arrived, how): (Vec<RawFd>, &str) = match call {
210            Syscall::Open(_) | Syscall::Openat(_) => (vec![ret as RawFd], "opened"),
211            _ if call.number() == Sysno::openat2 => (vec![ret as RawFd], "opened"),
212            _ if call.number() == Sysno::pidfd_getfd => {
213                (vec![ret as RawFd], "received through pidfd_getfd")
214            }
215            Syscall::Recvmsg(call) => {
216                let Some(address) = call.msg() else { return };
217                let Ok(header): Result<libc::msghdr, _> = guest.memory().read_value(address) else {
218                    return;
219                };
220                (
221                    received_through(&guest.memory(), &header),
222                    "received through SCM_RIGHTS",
223                )
224            }
225            Syscall::Recvmmsg(call) => {
226                let Some(address) = call.mmsg() else { return };
227                let count = (ret as usize).min(libc::UIO_MAXIOV as usize);
228                // SAFETY: `mmsghdr` is a plain C record; all-zero values are
229                // valid staging values that `read_values` overwrites.
230                let mut messages: Vec<libc::mmsghdr> =
231                    (0..count).map(|_| unsafe { std::mem::zeroed() }).collect();
232                if guest
233                    .memory()
234                    .read_values(address.into(), &mut messages)
235                    .is_err()
236                {
237                    return;
238                }
239                let memory = guest.memory();
240                (
241                    messages
242                        .iter()
243                        .flat_map(|message| received_through(&memory, &message.msg_hdr))
244                        .collect(),
245                    "received through SCM_RIGHTS",
246                )
247            }
248            _ => return,
249        };
250        let pid = guest.pid().as_raw();
251        for fd in arrived {
252            let Some(what) = descriptor_exit_dependency(Path::new(&format!("/proc/{pid}/fd/{fd}")))
253            else {
254                continue;
255            };
256            eprintln!("{}", refusal(what, fd, how));
257            unrecoverable_shutdown(guest, HERMIT_POLICY_REFUSAL_EXIT).await
258        }
259    }
260}

268    /// glibc's `makedev`.
269    fn makedev(major: u64, minor: u64) -> u64 {
270        ((major & 0xfff) << 8) | ((major & !0xfff) << 32) | (minor & 0xff) | ((minor & !0xff) << 12)
271    }
272
273    #[test]
274    fn only_the_fuse_and_cuse_devices_are_exit_dependencies() {
275        assert_eq!(
276            device_exit_dependency(makedev(10, 229)),
277            Some("a FUSE device (/dev/fuse)")
278        );
279        assert_eq!(
280            device_exit_dependency(makedev(10, 203)),
281            Some("a CUSE device (/dev/cuse)")
282        );
283        // Other misc devices: /dev/kvm, /dev/net/tun, a dynamic minor such as
284        // /dev/userfaultfd's, and the same minors under another major.
285        for (major, minor) in [
286            (10, 232),
287            (10, 200),
288            (10, 125),
289            (1, 229),
290            (4, 203),
291            (259, 229),
292        ] {
293            assert_eq!(
294                device_exit_dependency(makedev(major, minor)),
295                None,
296                "{major}:{minor}"
297            );
298        }
299        // Large numbers survive the split encoding.
300        assert_eq!(device_numbers(makedev(4095, 1 << 19)), (4095, 1 << 19));
301    }

303    #[test]
304    fn ordinary_descriptors_are_not_exit_dependencies() {
305        let file = tempfile::tempfile().unwrap();
306        let null = std::fs::File::open("/dev/null").unwrap();
307        let (left, _right) = std::os::unix::net::UnixStream::pair().unwrap();
308        for fd in [file.as_raw_fd(), null.as_raw_fd(), left.as_raw_fd()] {
309            assert_eq!(
310                descriptor_exit_dependency(Path::new(&format!("/proc/self/fd/{fd}"))),
311                None,

322    #[test]
323    fn a_received_seccomp_listener_is_recognised_by_its_anon_inode() {
324        // A child installs an allow-everything filter with a listener and sends
325        // the listener over SCM_RIGHTS; the filter stays in the child. Only raw
326        // system calls on buffers prepared before the fork run in the child.
327        const SECCOMP_SET_MODE_FILTER: libc::c_ulong = 1;
328        const SECCOMP_FILTER_FLAG_NEW_LISTENER: libc::c_ulong = 1 << 3;
329        const SECCOMP_RET_ALLOW: u32 = 0x7fff_0000;
330        let (sender, receiver) = std::os::unix::net::UnixDatagram::pair().unwrap();
331        let filter = [libc::sock_filter {
332            code: (libc::BPF_RET | libc::BPF_K) as u16,
333            jt: 0,
334            jf: 0,
335            k: SECCOMP_RET_ALLOW,
336        }];
337        let program = libc::sock_fprog {
338            len: 1,
339            filter: filter.as_ptr().cast_mut(),
340        };
341        let mut payload = [0u8; 1];
342        let mut iov = libc::iovec {
343            iov_base: payload.as_mut_ptr().cast(),
344            iov_len: 1,
345        };
346        let mut control = [0u64; 3];
347        let mut message: libc::msghdr = unsafe { std::mem::zeroed() };
348        message.msg_iov = &mut iov;
349        message.msg_iovlen = 1;
350        message.msg_control = control.as_mut_ptr().cast();
351        message.msg_controllen = 24;
352        // SAFETY: the child makes only raw system calls and writes into
353        // buffers that exist before the fork, then _exits.
354        let child = unsafe { libc::fork() };
355        assert!(child >= 0, "fork: {}", std::io::Error::last_os_error());
356        if child == 0 {
357            unsafe {
358                if libc::prctl(libc::PR_SET_NO_NEW_PRIVS, 1, 0, 0, 0) != 0 {
359                    libc::_exit(2);
360                }
361                let listener = libc::syscall(
362                    libc::SYS_seccomp,
363                    SECCOMP_SET_MODE_FILTER,
364                    SECCOMP_FILTER_FLAG_NEW_LISTENER,
365                    &program as *const libc::sock_fprog,
366                );
367                if listener < 0 {
368                    libc::_exit(3);
369                }
370                let header = control.as_mut_ptr().cast::<libc::cmsghdr>();
371                (*header).cmsg_len = 20;
372                (*header).cmsg_level = libc::SOL_SOCKET;
373                (*header).cmsg_type = libc::SCM_RIGHTS;
374                *control.as_mut_ptr().add(2).cast::<libc::c_int>() = listener as libc::c_int;
375                if libc::sendmsg(sender.as_raw_fd(), &message, 0) != 1 {
376                    libc::_exit(4);
377                }
378                libc::_exit(0);
379            }
380        }
381        let mut status = 0;
382        assert_eq!(unsafe { libc::waitpid(child, &mut status, 0) }, child);
383        assert_eq!(libc::WEXITSTATUS(status), 0, "child status {status:#x}");
384        let mut payload = [0u8; 1];
385        let mut iov = libc::iovec {
386            iov_base: payload.as_mut_ptr().cast(),
387            iov_len: 1,
388        };
389        let mut control = [0u8; 64];
390        let mut message: libc::msghdr = unsafe { std::mem::zeroed() };
391        message.msg_iov = &mut iov;
392        message.msg_iovlen = 1;
393        message.msg_control = control.as_mut_ptr().cast();
394        message.msg_controllen = control.len();
395        let received =
396            unsafe { libc::recvmsg(receiver.as_raw_fd(), &mut message, libc::MSG_CMSG_CLOEXEC) };
397        assert_eq!(received, 1, "recvmsg: {}", std::io::Error::last_os_error());
398        let descriptors = received_descriptors(&control[..message.msg_controllen]);
399        assert_eq!(descriptors.len(), 1, "{descriptors:?}");
400        let listener = unsafe {
401            <std::os::fd::OwnedFd as std::os::fd::FromRawFd>::from_raw_fd(descriptors[0])
402        };
403        assert_eq!(
404            descriptor_exit_dependency(Path::new(&format!(
405                "/proc/self/fd/{}",
406                listener.as_raw_fd()
407            ))),
408            Some("a seccomp user-notification listener")
409        );
410    }
