9//! A set of the guest's virtual CPUs, as the kernel's `struct cpumask`
10//! holds one: a task's affinity, or the CPUs a topology or cache file
11//! names. Bit `n` is CPU `n`; Detcore's masks have
12//! [`VIRTUAL_CPUSET_BYTES`] bytes, so CPUs 0 to 127.
13
14use serde::Deserialize;
15use serde::Serialize;
16
17use crate::syscalls::VIRTUAL_CPUSET_BYTES;
18
19/// A set of virtual CPUs: bit `n` of the 128-bit mask is CPU `n`.
20///
21/// The mask is held as its two 64-bit halves, least significant first,
22/// rather than as a `u128`: a `u128` is 16-byte aligned, and a set held
23/// across an `.await` would raise the alignment, and so the size, of every
24/// syscall future that holds one (`syscall_hot_path_tests` bounds them). It
25/// serializes, prints and hashes as the `u128` it is.
26#[derive(Clone, Copy, Default, Eq, PartialEq)]
27pub struct CpuSet([u64; 2]);
28
29impl CpuSet {
30    const fn from_bits(bits: u128) -> Self {
31        Self([bits as u64, (bits >> 64) as u64])
32    }
33
34    const fn bits(self) -> u128 {
35        ((self.0[1] as u128) << 64) | self.0[0] as u128
36    }
37}
38
39impl std::fmt::Debug for CpuSet {
40    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
41        f.debug_tuple("CpuSet").field(&self.bits()).finish()
42    }
43}
44
45impl std::hash::Hash for CpuSet {
46    fn hash<H: std::hash::Hasher>(&self, state: &mut H) {
47        self.bits().hash(state)
48    }
49}
50
51impl Serialize for CpuSet {
52    fn serialize<S: serde::Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
53        self.bits().serialize(serializer)
54    }
55}
56
57impl<'de> Deserialize<'de> for CpuSet {
58    fn deserialize<D: serde::Deserializer<'de>>(deserializer: D) -> Result<Self, D::Error> {
59        u128::deserialize(deserializer).map(Self::from_bits)
60    }
61}

63const _: () = assert!(VIRTUAL_CPUSET_BYTES == std::mem::size_of::<u128>());
64const _: () = assert!(detcore_model::config::MAX_VIRTUAL_CPUS as usize == VIRTUAL_CPUSET_BYTES * 8);
65
66impl CpuSet {
67    /// How many CPUs a set can name.
68    const BITS: u32 = u128::BITS;
69
70    /// CPUs `0..count`: every CPU of a machine with `count` CPUs.
71    pub fn first(count: u32) -> Self {
72        Self::from_bits(match count {
73            0 => 0,
74            count if count >= Self::BITS => u128::MAX,
75            count => (1u128 << count) - 1,
76        })
77    }
78
79    /// The one CPU `cpu`.
80    pub fn only(cpu: u32) -> Self {
81        Self::from_bits(1u128.checked_shl(cpu).unwrap_or(0))
82    }
83
84    /// The set a `sched_setaffinity` mask of `bytes` names, read as the
85    /// kernel's `get_user_cpu_mask` reads it: bytes past the mask's size are
86    /// ignored and a shorter mask leaves the CPUs past it out.
87    pub fn from_mask_bytes(bytes: &[u8]) -> Self {
88        let mut mask = [0u8; VIRTUAL_CPUSET_BYTES];
89        let copied = bytes.len().min(VIRTUAL_CPUSET_BYTES);
90        mask[..copied].copy_from_slice(&bytes[..copied]);
91        Self::from_bits(u128::from_le_bytes(mask))
92    }
93
94    /// The set as the kernel lays out a `cpumask` in memory.
95    pub fn to_mask_bytes(self) -> [u8; VIRTUAL_CPUSET_BYTES] {
96        self.bits().to_le_bytes()
97    }
98
99    /// The CPUs in both sets.
100    pub fn intersection(self, other: Self) -> Self {
101        Self::from_bits(self.bits() & other.bits())
102    }
103
104    /// Whether `cpu` is in the set.
105    pub fn contains(self, cpu: u32) -> bool {
106        !self.intersection(Self::only(cpu)).is_empty()
107    }
108
109    /// Whether the set has no CPU.
110    pub fn is_empty(self) -> bool {
111        self.bits() == 0
112    }
113
114    /// How many CPUs the set has.
115    pub fn len(self) -> u32 {
116        self.bits().count_ones()
117    }
118
119    /// The set's CPUs, from the lowest.
120    fn cpus(self) -> impl Iterator<Item = u32> {
121        (0..Self::BITS).filter(move |cpu| self.contains(*cpu))
122    }
123
124    /// The set's `index`th CPU, counting from the lowest.
125    pub fn nth(self, index: u32) -> Option<u32> {
126        self.cpus().nth(index as usize)
127    }
128
129    /// The set as a kernel CPU list (`%*pbl`, as `online` and
130    /// `Cpus_allowed_list` print it): ascending ranges, `0-3,5`.
131    pub fn list(self) -> String {
132        let mut ranges: Vec<(u32, u32)> = Vec::new();
133        for cpu in self.cpus() {
134            match ranges.last_mut() {
135                Some((_, end)) if *end + 1 == cpu => *end = cpu,
136                _ => ranges.push((cpu, cpu)),
137            }
138        }
139        let ranges: Vec<String> = ranges
140            .into_iter()
141            .map(|(start, end)| match start == end {
142                true => start.to_string(),
143                false => format!("{start}-{end}"),
144            })
145            .collect();
146        ranges.join(",")
147    }
148
149    /// The set as a kernel CPU mask of `bits` bits (`%*pb`, as `cpumap`,
150    /// `thread_siblings` and `Cpus_allowed` print it, with `nr_cpu_ids`
151    /// bits): hexadecimal groups of 32 bits, the most significant first,
152    /// separated by commas, the first group only as wide as its bits need.
153    pub fn mask(self, bits: u32) -> String {
154        let bits = bits.clamp(1, Self::BITS);
155        let mut groups = Vec::new();
156        let mut low = bits.div_ceil(32) * 32;
157        let mut width = match bits % 32 {
158            0 => 32,
159            partial => partial,
160        };
161        while low > 0 {
162            low -= 32;
163            let value = (self.bits() >> low) as u32 & (u32::MAX >> (32 - width));
164            groups.push(format!(
165                "{value:0digits$x}",
166                digits = width.div_ceil(4) as usize
167            ));
168            width = 32;
169        }
170        groups.join(",")
171    }
172}

174/// Where a task may run and where it is placed among those CPUs, as the
175/// scheduler keeps them (`ThreadTree::cpu_affinities`, `ThreadTree::cpu_homes`).
176#[derive(Clone, Copy, Debug, Default, Eq, PartialEq, Serialize, Deserialize)]
177pub struct TaskCpus {
178    /// The CPUs a `sched_setaffinity` left the task, or `None` where it may
179    /// run on every virtual CPU, as every task starts.
180    pub mask: Option<CpuSet>,
181    /// The number of tasks the run created before it, so that tasks created
182    /// one after another take the CPUs in turn.
183    pub home: u32,
184}
185
186impl TaskCpus {
187    /// The CPUs the task may run on, of a machine with `cpus` CPUs.
188    pub fn allowed(self, cpus: u32) -> CpuSet {
189        self.mask.unwrap_or(CpuSet::first(cpus))
190    }
191
192    /// The CPU the task runs on, of a machine with `cpus` CPUs, always one
193    /// it may run on: CPU home mod `cpus` while it may run there, and
194    /// otherwise its home's place among the CPUs it may run on. Linux moves
195    /// a task only off a CPU its new mask leaves out (`affine_move_task`).
196    pub fn current(self, cpus: u32) -> u32 {
197        let allowed = self.allowed(cpus);
198        let own = self.home % cpus.max(1);
199        match allowed.contains(own) {
200            true => own,
201            false => allowed.nth(self.home % allowed.len().max(1)).unwrap_or(own),
202        }
203    }
204}

206/// A task's CPUs as the scheduler answers for them under `--virtual-cpus`
207/// (`Scheduler::cpu_affinity`).
208#[derive(Clone, Copy, Debug, Eq, PartialEq, Serialize, Deserialize)]
209pub enum TaskAffinity {
210    /// No task Linux would find has the number asked for: ESRCH.
211    NoSuchTask,
212    /// A task: live, or a process that has exited and is not reaped.
213    Task(TaskCpus),
214}
215
216impl TaskAffinity {
217    /// The CPUs the task may run on, of a machine with `cpus` CPUs; ESRCH
218    /// where there is no such task.
219    pub fn cpus(self, cpus: u32) -> Result<CpuSet, reverie::Errno> {
220        match self {
221            Self::NoSuchTask => Err(reverie::Errno::ESRCH),
222            Self::Task(task) => Ok(task.allowed(cpus)),
223        }
224    }
225
226    /// The CPU the task runs on ([`TaskCpus::current`]); ESRCH where there
227    /// is no such task.
228    pub fn current_cpu(self, cpus: u32) -> Result<u32, reverie::Errno> {
229        match self {
230            Self::NoSuchTask => Err(reverie::Errno::ESRCH),
231            Self::Task(task) => Ok(task.current(cpus)),
232        }
233    }
234}

236#[cfg(test)]
237mod tests {
238    use super::*;
239
240    #[test]
241    fn first_and_only_name_the_cpus_they_say() {
242        assert_eq!(CpuSet::first(1), CpuSet::only(0));
243        assert_eq!(CpuSet::first(4).len(), 4);
244        assert_eq!(CpuSet::first(128).len(), 128);
245        assert!(CpuSet::first(0).is_empty());
246        assert!(CpuSet::only(128).is_empty());
247    }
248
249    /// Bytes past the mask are ignored and a short mask names no CPU past
250    /// its end, as `get_user_cpu_mask` reads a user mask.
251    #[test]
252    fn a_mask_reads_as_the_kernel_copies_it() {
253        assert_eq!(CpuSet::from_mask_bytes(&[0b1010]).list(), "1,3");
254        assert!(CpuSet::from_mask_bytes(&[]).is_empty());
255        let mut long = [0xffu8; 32];
256        long[0] = 0b1;
257        assert_eq!(CpuSet::from_mask_bytes(&long).len(), 121);
258        assert_eq!(
259            CpuSet::from_mask_bytes(&CpuSet::first(9).to_mask_bytes()),
260            CpuSet::first(9)
261        );
262    }
263
264    /// A task runs on CPU home mod N while its mask allows it, and
265    /// otherwise on its home's place among the CPUs it may run on.
266    #[test]
267    fn a_task_runs_on_its_home_among_its_cpus() {
268        let task = |mask, home| TaskAffinity::Task(TaskCpus { mask, home });
269        assert_eq!(task(None, 0).current_cpu(4), Ok(0));
270        assert_eq!(task(None, 6).current_cpu(4), Ok(2));
271        let odd = CpuSet::from_mask_bytes(&[0b1010]);
272        assert_eq!(task(Some(odd), 0).current_cpu(4), Ok(1));
273        assert_eq!(task(Some(odd), 5).current_cpu(4), Ok(1));
274        assert_eq!(task(Some(odd), 7).current_cpu(4), Ok(3));
275        assert_eq!(task(Some(odd), 2).current_cpu(4), Ok(1));
276        // A narrower mask that keeps the task's CPU leaves it there.
277        let upper = CpuSet::from_mask_bytes(&[0b0110]);
278        assert_eq!(task(Some(upper), 2).current_cpu(4), Ok(2));
279        assert_eq!(task(Some(odd), 5).cpus(4), Ok(odd));
280        assert_eq!(task(None, 9).current_cpu(1), Ok(0));
281        assert_eq!(
282            TaskAffinity::NoSuchTask.current_cpu(4),
283            Err(reverie::Errno::ESRCH)
284        );
285    }
286
287    #[test]
288    fn nth_counts_from_the_lowest_cpu() {
289        let set = CpuSet::from_mask_bytes(&[0b1011_0000]);
290        assert_eq!(set.nth(0), Some(4));
291        assert_eq!(set.nth(2), Some(7));
292        assert_eq!(set.nth(3), None);
293    }
294
295    /// `%*pbl` and `%*pb` as lib/vsprintf.c's `bitmap_list_string` and
296    /// `bitmap_string` print them.
297    #[test]
298    fn lists_and_masks_print_as_the_kernel_prints_them() {
299        assert_eq!(CpuSet::first(1).list(), "0");
300        assert_eq!(CpuSet::first(4).list(), "0-3");
301        assert_eq!(CpuSet::from_mask_bytes(&[0b0110_1101]).list(), "0,2-3,5-6");
302        assert_eq!(CpuSet::only(127).list(), "127");
303        assert_eq!(CpuSet::first(1).mask(1), "1");
304        assert_eq!(CpuSet::first(4).mask(4), "f");
305        assert_eq!(CpuSet::only(2).mask(4), "4");
306        assert_eq!(CpuSet::first(8).mask(8), "ff");
307        assert_eq!(CpuSet::only(0).mask(40), "00,00000001");
308        assert_eq!(CpuSet::first(32).mask(32), "ffffffff");
309        assert_eq!(
310            CpuSet::only(0).mask(128),
311            "00000000,00000000,00000000,00000001"
312        );
313    }
