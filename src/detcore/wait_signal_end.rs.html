9//! A5 step 3c.1c: which pending signals end a parked wait under the in-guest
10//! delivery gate (`Config::in_guest_signal_delivery`), and what the wait's
11//! end tells the restart checkpoint (design D40, D50; dev-hermit
12//! ai_docs/2026-10-16-a5-signal-delivery-design.md, "3c.1 implementation
13//! plan, revision 4", <https://github.com/rrnewton/hermit/issues/4485>).
14//!
15//! Once the in-guest runtime runs under a mask of its own (step 3c.1e), the
16//! kernel's `SigBlk` for a guest thread is the runtime's, and a signal sent
17//! to a parked thread stays pending in the kernel, held by that mask, until
18//! Detcore ends the wait. So Detcore decides from kernel truth plus the one
19//! fact the guest publishes (design 16.11):
20//!
21//! - the target's pending and action sets from `/proc/<tid>/status`
22//!   (`SigPnd`, `ShdPnd`, `SigCgt`, `SigIgn`);
23//! - the guest's mask as the thread published it at its latest request
24//!   (step 3c.1b), never `SigBlk`.
25//!
26//! The deliverable set is `(SigPnd ∪ ShdPnd) ∖ mask ∩ (SigCgt ∪ (default-fatal
27//! ∖ SigIgn))`: a signal Linux would deliver at the wait's return, a caught one
28//! by running its handler, a default-fatal one by ending the process. A
29//! blocked signal stays pending; an ignored or default-ignored one, or a
30//! default stop, does not end the wait (stop sends are refused under the gate,
31//! D48). The in-guest runtime's reserved signals
32//! (`Config::in_guest_reserved_signals`) are left out: their `SigCgt` bit is
33//! the runtime's handler, not a guest disposition, and a cross-process send of
34//! one is refused (step 3c.0a).
35

40/// What ended a wait for a signal under the in-guest delivery gate: D40's
41/// per-syscall flag, set by the path that ends the wait
42/// (`ThreadState::wait_signal_end`) and taken by the restart checkpoint
43/// straight after the dispatch `match` (`Detcore::wait_end_checkpoint`).
44#[derive(Clone, Copy, Debug, PartialEq, Eq)]
45pub(crate) enum WaitSignalEnd {
46    /// The process dies at the return and the guest never resumes after the
47    /// call: the guest result is `-EINTR`, and the original restart code is
48    /// what the happens-before checkpoint records (IR-2). Either a
49    /// non-core-dumping default-fatal signal is deliverable (the first such
50    /// one Linux would dequeue), or Linux's delivery loop reaches a
51    /// core-dumping default-fatal one that no mask blocks ([`classify`],
52    /// [`DequeueWalk::resolve`]).
53    ///
54    /// The signal is not the exit status for a non-core-dumping end: Linux
55    /// takes that from `group_exit_code`, the first such signal SENT, which
56    /// `complete_signal` records when it starts the group exit
57    /// (kernel/signal.c:1041-1049 at v6.12), not the first one dequeued.
58    /// Step 3c.1e, which takes the death's status, must use the send order.
59    Fatal(i32),
60    /// The first signal Linux would dequeue is caught, and Linux's delivery
61    /// loop reaches no default-fatal signal: either none is deliverable, or
62    /// the masks of the handlers it sets up before reaching one block it,
63    /// and no second instance of a caught signal is dequeued under a default
64    /// action that `SA_RESETHAND` restored. That first signal's handler runs
65    /// at the return, and the call's result goes through the x86 restart
66    /// table with that action's flags (`signal_restart::caught_signal_return`).
67    Caught(i32),
68    /// The first signal Linux would dequeue is caught, and a later delivery
69    /// in the same loop might end the process: a core-dumping default-fatal
70    /// signal is deliverable behind it, or a caught signal has a second
71    /// instance (a standard signal pending on both queues) or may have one (a
72    /// real-time signal), which `SA_RESETHAND` would leave at its default
73    /// action. The end depends on the caught actions (their `sa_mask` and
74    /// `sa_flags`), which the restart checkpoint reads in the thread's own
75    /// turn and resolves ([`DequeueWalk::resolve`]).
76    BehindCaught(DequeueWalk),
77}
78
79/// The parts of a caught signal's action that decide what Linux blocks
80/// while its handler is set up, and the restart table's flags: its
81/// `sa_mask` and `sa_flags`, as a query `rt_sigaction` reads them.
82#[derive(Clone, Copy, Debug, PartialEq, Eq)]
83pub(crate) struct CaughtAction {
84    pub(crate) mask: u64,
85    pub(crate) flags: u64,
86}

88/// A delivery loop the checkpoint cannot decide ([`DequeueWalk::resolve`]),
89/// which it refuses by name (D49), fail closed.
90#[derive(Clone, Copy, Debug, PartialEq, Eq)]
91pub(crate) enum WalkRefusal {
92    /// The action of caught signal `caught` could not be read while a later
93    /// delivery could still end the process; `behind` is the signal that
94    /// could (HERMIT_CAUGHT_ACTION_UNREADABLE).
95    ActionUnknown { caught: i32, behind: i32 },
96    /// `SA_RESETHAND` reset caught signal `signal` to its default action, a
97    /// stop, and a second instance of it is dequeued: the in-guest backend
98    /// cannot stop a guest process on the schedule (D48;
99    /// HERMIT_SIGNAL_RESET_TO_STOP).
100    ResetToStop { signal: i32 },
101    /// A caught real-time signal with `SA_RESETHAND` stays unblocked after its
102    /// delivery, so another queued instance of it would be dequeued at its
103    /// default action, termination; `/proc` shows only whether one instance
104    /// is pending, not how many, and Detcore keeps no count of the
105    /// physically pending ones (HERMIT_REALTIME_INSTANCES_UNKNOWN).
106    RealtimeInstances { signal: i32 },
107}
108
109/// The real-time signals, 32 to 64 in the kernel's numbering (`SIGRTMIN`
110/// is 32 there; C libraries reserve the first few). Any number of instances
111/// of one can be queued (`collect_signal`, kernel/signal.c:566-596 at
112/// v6.12).
113fn realtime() -> u64 {
114    !0 << 31
115}
116
117/// The stop signals, whose default action stops the thread group.
118fn stops() -> u64 {
119    [libc::SIGSTOP, libc::SIGTSTP, libc::SIGTTIN, libc::SIGTTOU]
120        .into_iter()
121        .fold(0, |set, signal| set | kernel_sigset_bit(signal))
122}

124/// The input of Linux's delivery loop at a wait's end, for a
125/// [`WaitSignalEnd::BehindCaught`]: the deliverable signals on the thread's
126/// private queue and on the shared queue (a standard signal on both is two
127/// instances), which of them are default-fatal (the rest are caught), and
128/// the guest's mask they were found under.
129#[derive(Clone, Copy, Debug, PartialEq, Eq)]
130pub(crate) struct DequeueWalk {
131    thread: u64,
132    shared: u64,
133    fatal: u64,
134    mask: u64,
135}
136
137impl DequeueWalk {
138    /// The first instance, in dequeue order, that could still end the
139    /// process: unblocked, and default-fatal, or reset to its default action
140    /// (`reset`), or a caught real-time signal (it may be queued again), or a
141    /// caught standard signal pending on both queues (two instances).
142    fn at_risk(&self, thread: u64, shared: u64, blocked: u64, reset: u64) -> Option<i32> {
143        let unblocked = (thread | shared) & !blocked;
144        let caught = !self.fatal;
145        let risky = unblocked & (self.fatal | reset | (caught & (realtime() | (thread & shared))));
146        queued(thread, shared, risky)
147            .next_dequeued(blocked)
148            .map(|(signal, _)| signal)
149    }
150
151    /// Linux's delivery loop over this walk (v6.12). At the return to user
152    /// mode `exit_to_user_mode_loop` calls `arch_do_signal_or_restart` for as
153    /// long as a signal is pending and unblocked (kernel/entry/common.c:
154    /// 97-130; arch/x86/kernel/signal.c:333-341), before any user
155    /// instruction runs. Each `get_signal` dequeues the next instance in
156    /// order (private queue first, then synchronous signals, then the lowest
157    /// number; kernel/signal.c:634-639, :196-218):
158    ///
159    /// - a caught one: `get_signal` resets the action to `SIG_DFL` when it has
160    ///   `SA_RESETHAND` (`SA_ONESHOT`, :2821-2822), `handle_signal` sets up its
161    ///   frame, and `signal_delivered` (:2941-2954) adds the action's
162    ///   `sa_mask`, and the signal itself unless `SA_NODEFER`, to the blocked
163    ///   mask; the loop goes on;
164    /// - a default-fatal one not blocked by then ends the process
165    ///   (:2878-2906) without any handler running: `Fatal`;
166    /// - a second instance of a caught signal whose action was reset is
167    ///   taken at its default action: termination or a core dump is `Fatal`,
168    ///   a default-ignored one is discarded (the loop goes on), a stop is
169    ///   [`WalkRefusal::ResetToStop`].
170    ///
171    /// So `SA_NODEFER` matters exactly when a second instance of the same
172    /// signal exists and `SA_RESETHAND` reset it: the legacy `signal()`
173    /// installs both (`SA_ONESHOT | SA_NOMASK`, kernel/signal.c:4675). A
174    /// standard signal has at most one instance per queue, so one pending on
175    /// both queues is two. A real-time signal may have any number, which
176    /// `/proc` does not show: a caught one with `SA_RESETHAND` that stays
177    /// unblocked after its delivery is [`WalkRefusal::RealtimeInstances`].
178    /// With every remaining instance blocked or harmless, the end is
179    /// `Caught` with the first caught signal.
180    ///
181    /// `action` gives a caught signal's action. It is asked only while a
182    /// later delivery could still end the process ([`Self::at_risk`], or the
183    /// delivered signal is real-time or has another instance); a `None` then
184    /// is [`WalkRefusal::ActionUnknown`].
185    pub(crate) fn resolve(
186        &self,
187        mut action: impl FnMut(i32) -> Option<CaughtAction>,
188    ) -> Result<WaitSignalEnd, WalkRefusal> {
189        let mut blocked = self.mask;
190        let mut reset = 0;
191        let (mut thread, mut shared) = (self.thread, self.shared);
192        let mut first = None;
193        while let Some((signal, queue)) = queued(thread, shared, !0).next_dequeued(blocked) {
194            let bit = kernel_sigset_bit(signal);
195            match queue {
196                SignalQueue::Thread => thread &= !bit,
197                SignalQueue::Shared => shared &= !bit,
198            }
199            if reset & bit != 0 {
200                if crate::syscalls::default_action_ends_process(0, 0, signal) {
201                    return Ok(WaitSignalEnd::Fatal(signal));
202                }
203                if stops() & bit != 0 {
204                    return Err(WalkRefusal::ResetToStop { signal });
205                }
206                continue;
207            }
208            if self.fatal & bit != 0 {
209                return Ok(WaitSignalEnd::Fatal(signal));
210            }
211            let first = *first.get_or_insert(signal);
212            let again = (thread | shared) & bit != 0 || realtime() & bit != 0;
213            let risk = self.at_risk(thread, shared, blocked, reset);
214            if risk.is_none() && !again {
215                return Ok(WaitSignalEnd::Caught(first));
216            }
217            let Some(caught) = action(signal) else {
218                return Err(WalkRefusal::ActionUnknown {
219                    caught: signal,
220                    behind: risk.unwrap_or(signal),
221                });
222            };
223            blocked |= caught.mask;
224            if caught.flags & libc::SA_NODEFER as u64 == 0 {
225                blocked |= bit;
226            }
227            if caught.flags & libc::SA_RESETHAND as u64 != 0 {
228                reset |= bit;
229                if realtime() & bit != 0 && blocked & bit == 0 {
230                    return Err(WalkRefusal::RealtimeInstances { signal });
231                }
232            }
233        }
234        Ok(WaitSignalEnd::Caught(
235            first.expect("a walk starts with a deliverable caught signal"),
236        ))
237    }
238}

240/// The instances on the private queue `thread` and the shared queue
241/// `shared` among `set`, as a state `next_dequeued` reads.
242fn queued(thread: u64, shared: u64, set: u64) -> KernelSignalState {
243    KernelSignalState {
244        pending: (thread | shared) & set,
245        thread_pending: thread & set,
246        shared_pending: shared & set,
247        ..KernelSignalState::default()
248    }
249}

251/// The signals whose default action ends the process (terminate or dump core)
252/// and whose disposition is still the default: not caught, not ignored, not a
253/// default-ignored signal (`SIGCHLD`, `SIGCONT`, `SIGURG`, `SIGWINCH`), and not
254/// a stop. `SIGKILL` always is.
255pub(crate) fn default_fatal(state: &KernelSignalState) -> u64 {
256    (1..=64).fold(0, |set, signal| {
257        if crate::syscalls::default_action_ends_process(state.ignored, state.caught, signal) {
258            set | kernel_sigset_bit(signal)
259        } else {
260            set
261        }
262    })
263}

265/// The design's deliverable set for a thread whose guest mask is `mask`: the
266/// signals pending for it (`SigPnd ∪ ShdPnd`) that `mask` leaves unblocked and
267/// that are caught, or default-fatal and not ignored, less `reserved`, the
268/// in-guest runtime's reserved signals.
269pub(crate) fn deliverable(state: &KernelSignalState, mask: u64, reserved: u64) -> u64 {
270    state.pending & !mask & (state.caught | default_fatal(state)) & !reserved
271}

273/// What a wait that `deliverable` (a non-empty subset of the design's set for
274/// `state` under the guest's mask `mask`) ends means for the call's result,
275/// or `None` when nothing ends it, in Linux's order (v6.12):
276///
277/// 1. A deliverable default-fatal signal that does not dump core makes it
278///    `Fatal`: `complete_signal` starts the group exit when such a signal is
279///    sent to a thread that wants it, queueing SIGKILL to every thread
280///    (kernel/signal.c:1034-1054, its `!sig_kernel_coredump(sig)` branch), so
281///    the process dies at the return whatever else is pending. The signal
282///    named is the first such one Linux would dequeue (see `Fatal` for the
283///    exit status).
284/// 2. Otherwise the first deliverable signal in `get_signal`'s dequeue order
285///    comes first: `dequeue_signal` takes the thread's private queue before
286///    the shared one (kernel/signal.c:634-639), and `next_signal` the
287///    synchronous signals first, then the lowest number (:196-218); here
288///    `KernelSignalState::next_dequeued`. A core-dumping one is `Fatal`:
289///    `get_signal` ends the process for it (:2881-2905). A caught one is
290///    `Caught` unless a later delivery in the same loop could end the
291///    process: a core-dumping signal deliverable behind it, a deliverable
292///    caught standard signal pending on both queues, or a deliverable caught
293///    real-time signal (the last two through `SA_RESETHAND`). Then the end
294///    depends on the caught actions, so it is `BehindCaught`, which the
295///    restart checkpoint resolves ([`DequeueWalk::resolve`]).
296pub(crate) fn classify(
297    state: &KernelSignalState,
298    deliverable: u64,
299    mask: u64,
300) -> Option<WaitSignalEnd> {
301    if deliverable == 0 {
302        return None;
303    }
304    let fatal = deliverable & default_fatal(state);
305    let among = |set: u64| KernelSignalState {
306        pending: state.pending & set,
307        thread_pending: state.thread_pending & set,
308        shared_pending: state.shared_pending & set,
309        ..*state
310    };
311    let group_exit = fatal & !core_dumping();
312    if group_exit != 0 {
313        let (signal, _) = among(group_exit).next_dequeued(0)?;
314        return Some(WaitSignalEnd::Fatal(signal));
315    }
316    let (signal, _) = among(deliverable).next_dequeued(0)?;
317    let (thread, shared) = (
318        state.thread_pending & deliverable,
319        state.shared_pending & deliverable,
320    );
321    let caught = deliverable & !fatal;
322    if fatal & kernel_sigset_bit(signal) != 0 {
323        Some(WaitSignalEnd::Fatal(signal))
324    } else if fatal != 0 || caught & (realtime() | (thread & shared)) != 0 {
325        Some(WaitSignalEnd::BehindCaught(DequeueWalk {
326            thread,
327            shared,
328            fatal,
329            mask,
330        }))
331    } else {
332        Some(WaitSignalEnd::Caught(signal))
333    }
334}

336/// The signals whose default action dumps core (`SIG_KERNEL_COREDUMP_MASK`,
337/// include/linux/signal.h:426-432 at v6.12).
338pub(crate) fn core_dumping() -> u64 {
339    [
340        libc::SIGQUIT,
341        libc::SIGILL,
342        libc::SIGTRAP,
343        libc::SIGABRT,
344        libc::SIGFPE,
345        libc::SIGSEGV,
346        libc::SIGBUS,
347        libc::SIGSYS,
348        libc::SIGXCPU,
349        libc::SIGXFSZ,
350    ]
351    .into_iter()
352    .fold(0, |set, signal| set | kernel_sigset_bit(signal))
353}

355/// [`deliverable`] then [`classify`].
356pub(crate) fn wait_end(
357    state: &KernelSignalState,
358    mask: u64,
359    reserved: u64,
360) -> Option<WaitSignalEnd> {
361    wait_end_with_virtual(state, mask, reserved, 0)
362}

364/// [`wait_end`] with `virtual_caught`, caught signals the scheduler knows
365/// are pending on the process's shared queue although no physical copy is:
366/// signal phase 1's virtual SIGALRM, which Linux's `ITIMER_REAL` and `alarm`
367/// queue on the shared queue (`it_real_fn`, kernel/time/itimer.c:156-166 at
368/// v6.12) and which the scheduler's ledger delivers only when the guest's
369/// mask admits it. They join the deliverable set and take their place in
370/// Linux's dequeue order ([`classify`]).
371pub(crate) fn wait_end_with_virtual(
372    state: &KernelSignalState,
373    mask: u64,
374    reserved: u64,
375    virtual_caught: u64,
376) -> Option<WaitSignalEnd> {
377    let deliverable = deliverable(state, mask, reserved) | virtual_caught;
378    let state = KernelSignalState {
379        pending: state.pending | virtual_caught,
380        shared_pending: state.shared_pending | virtual_caught,
381        caught: state.caught | virtual_caught,
382        ignored: state.ignored & !virtual_caught,
383        ..*state
384    };
385    classify(&state, deliverable, mask)
386}

388/// D44 (A5 step 3c.1c): whether a send of `signal` to a thread blocked in a
389/// kernel-run call (`ResourceID::is_kernel_run_blocking`), whose guest mask at
390/// that call's request was `mask`, is refused: Linux would deliver it by
391/// interrupting the call, which the kernel runs under the in-guest runtime's
392/// mask, so it would stay pending and the call would never end for it. That
393/// is a signal `mask` leaves unblocked that is caught, or default-fatal and not
394/// ignored. A vfork parent (`vfork`) sleeps killable, and Linux wakes it only
395/// for a fatal signal that is not core-dumping (`wait_for_vfork_done`,
396/// kernel/fork.c:1572-1590; `complete_signal`, kernel/signal.c:1034-1054): a
397/// caught or core-dumping signal waits for the child's release, which keeping
398/// it pending reproduces, so only an unblocked, non-core-dumping default-fatal
399/// signal is refused there. `state` gives the dispositions; its pending set
400/// is not read. SIGKILL and signal 0 are never passed here.
401pub(crate) fn kernel_run_send_refused(
402    state: &KernelSignalState,
403    mask: u64,
404    signal: i32,
405    vfork: bool,
406    reserved: u64,
407) -> bool {
408    if !(1..=64).contains(&signal) {
409        return false;
410    }
411    let bit = kernel_sigset_bit(signal);
412    if mask & bit != 0 || reserved & bit != 0 {
413        return false;
414    }
415    if vfork {
416        default_fatal(state) & bit != 0 && core_dumping() & bit == 0
417    } else {
418        (state.caught | default_fatal(state)) & bit != 0
419    }
420}

422/// Whether `errno` is one of the kernel's internal restart codes
423/// (include/linux/errno.h v6.12 lines 14-18), which a guest never sees: the
424/// kernel either restarts the call or turns the code into `EINTR` at the
425/// return.
426pub(crate) fn is_restart_code(errno: reverie::Errno) -> bool {
427    matches!(
428        errno,
429        reverie::Errno::ERESTARTSYS
430            | reverie::Errno::ERESTARTNOINTR
431            | reverie::Errno::ERESTARTNOHAND
432            | reverie::Errno::ERESTART_RESTARTBLOCK
433    )
434}

571    /// A caught action for the walk's lookups: `mask`, and `SA_NODEFER` or not.
572    fn action(mask: u64, nodefer: bool) -> CaughtAction {
573        CaughtAction {
574            mask,
575            flags: if nodefer { libc::SA_NODEFER as u64 } else { 0 },
576        }

579    /// The walk a `BehindCaught` end carries, for `state` and `mask`.
580    fn walk(state: &KernelSignalState, mask: u64) -> DequeueWalk {
581        match classify(state, deliverable(state, mask, 0), mask) {
582            Some(WaitSignalEnd::BehindCaught(walk)) => walk,
583            other => panic!("not a walk: {other:?}"),
584        }
585    }

587    /// The end of a wait on `state` (guest mask 0), a `BehindCaught` one
588    /// resolved with `action`.
589    fn end_of(
590        state: &KernelSignalState,
591        action: impl FnMut(i32) -> Option<CaughtAction>,
592    ) -> Result<WaitSignalEnd, WalkRefusal> {
593        match classify(state, deliverable(state, 0, 0), 0) {
594            Some(WaitSignalEnd::BehindCaught(walk)) => walk.resolve(action),
595            Some(end) => Ok(end),
596            None => panic!("nothing ends the wait"),
597        }
598    }

600    /// Rule 2 with a caught signal first and a core-dumping one deliverable
601    /// behind it (Claude review round 2): Linux sets up the caught handler and
602    /// goes on delivering under its `sa_mask` (signal_delivered,
603    /// kernel/signal.c:2941-2954), so the core-dumping signal ends the
604    /// process unless that mask, or a later handler's, holds it. Rows: (a) the
605    /// handler's mask blocks it, `Caught` (a private caught signal before a
606    /// shared, lower-numbered SIGQUIT; both private, the caught one
607    /// lower-numbered); (b) it does not, `Fatal`; (c) `SA_NODEFER` with one
608    /// instance of the caught signal, decided by the mask alone (where it does
609    /// matter, with `SA_RESETHAND`:
610    /// `sa_resethand_and_a_second_instance_decide_the_end`); (d) two caught
611    /// signals whose masks together block it,
612    /// `Caught`. Mutations: "the handler's sa_mask ignored" ((a), (d) become
613    /// `Fatal`), "only the first handler's mask" ((d) becomes `Fatal`), "any
614    /// default-fatal signal makes the end fatal" ((a), (d)), "the lowest
615    /// number across both queues" (the first (a) row becomes `Fatal(SIGQUIT)`
616    /// at classification).
617    #[test]
618    fn a_caught_signal_dequeued_before_a_core_dumping_one_is_resolved_by_the_handlers_masks() {
619        let (hup, usr1, usr2) = (bit(libc::SIGHUP), bit(libc::SIGUSR1), bit(libc::SIGUSR2));
620        let (quit, xcpu) = (bit(libc::SIGQUIT), bit(libc::SIGXCPU));
621        type Actions = &'static [(i32, u64, bool)];
622        let rows: [(&str, u64, u64, u64, Actions, WaitSignalEnd); 7] = [
623            (
624                "(a) the private caught signal's mask blocks a shared SIGQUIT",
625                usr2,
626                quit,
627                usr2,
628                &[(libc::SIGUSR2, 1 << (libc::SIGQUIT - 1), false)],
629                WaitSignalEnd::Caught(libc::SIGUSR2),
630            ),
631            (
632                "(a) both private, the caught one's mask blocks SIGQUIT",
633                hup | quit,
634                0,
635                hup,
636                &[(libc::SIGHUP, 1 << (libc::SIGQUIT - 1), false)],
637                WaitSignalEnd::Caught(libc::SIGHUP),
638            ),
639            (
640                "(b) the caught signal's mask does not block SIGQUIT",
641                hup | quit,
642                0,
643                hup,
644                &[(libc::SIGHUP, 1 << (libc::SIGUSR1 - 1), false)],
645                WaitSignalEnd::Fatal(libc::SIGQUIT),
646            ),
647            (
648                "(c) SA_NODEFER, mask blocking SIGQUIT",
649                hup | quit,
650                0,
651                hup,
652                &[(libc::SIGHUP, 1 << (libc::SIGQUIT - 1), true)],
653                WaitSignalEnd::Caught(libc::SIGHUP),
654            ),
655            (
656                "(c) SA_NODEFER, mask not blocking SIGQUIT",
657                hup | quit,
658                0,
659                hup,
660                &[(libc::SIGHUP, 0, true)],
661                WaitSignalEnd::Fatal(libc::SIGQUIT),
662            ),
663            (
664                "(d) two caught signals whose masks together block SIGXCPU",
665                hup | usr1 | xcpu,
666                0,
667                hup | usr1,
668                &[
669                    (libc::SIGHUP, 0, false),
670                    (libc::SIGUSR1, 1 << (libc::SIGXCPU - 1), false),
671                ],
672                WaitSignalEnd::Caught(libc::SIGHUP),
673            ),
674            (
675                "(d) neither mask blocks SIGXCPU",
676                hup | usr1 | xcpu,
677                0,
678                hup | usr1,
679                &[(libc::SIGHUP, 0, false), (libc::SIGUSR1, 0, false)],
680                WaitSignalEnd::Fatal(libc::SIGXCPU),
681            ),
682        ];
683        // Every row is checked before the test fails, so that a mutation's
684        // failure names the rows it changes.
685        let mut wrong = Vec::new();
686        for (name, thread, shared, caught, actions, end) in rows {
687            let s = state(thread, shared, 0, caught);
688            let resolved = end_of(&s, |signal| {
689                actions
690                    .iter()
691                    .find(|(known, _, _)| *known == signal)
692                    .map(|(_, mask, nodefer)| action(*mask, *nodefer))
693            });
694            if resolved != Ok(end) {
695                wrong.push(format!("{name}: {resolved:?}"));
696            }
697        }
698        assert!(wrong.is_empty(), "rows: {wrong:#?}");
699    }

701    /// `SA_RESETHAND` and a second instance (both reviews, round 3): Linux
702    /// resets an `SA_RESETHAND` action to `SIG_DFL` when it dequeues it
703    /// (`get_signal`, kernel/signal.c:2821-2822 at v6.12), so a second
704    /// instance of the same signal that its masks leave unblocked
705    /// (`SA_NODEFER`, the legacy `signal()`'s flags, :4675) is taken at the
706    /// default action. A standard signal pending on both queues is two
707    /// instances; a real-time one may have any number. Rows: SIGUSR1 on both
708    /// queues with `SA_RESETHAND|SA_NODEFER`, `Fatal(SIGUSR1)`; the same
709    /// without `SA_NODEFER`, `Caught`; `SA_RESETHAND` alone with one instance,

712    /// with `SA_RESETHAND|SA_NODEFER`, `RealtimeInstances`; the same without
713    /// `SA_RESETHAND`, `Caught`; and `SA_RESETHAND|SA_NODEFER` with SIGUSR1's
714    /// own mask holding it, `Caught`. Mutations: "SA_NODEFER ignored",
715    /// "SA_RESETHAND ignored", "the second queue's instance ignored" and "the
716    /// real-time refusal removed" each change a row.

722        let resethand = libc::SA_RESETHAND as u64;
723        let nodefer = libc::SA_NODEFER as u64;
724        // name, private, shared, caught, the caught signal's action
725        // (mask, flags), the end
726        type Row = (
727            &'static str,
728            u64,
729            u64,
730            u64,
