9//! The canonical models of open file descriptions that several guest
10//! processes share, and the leases through which a process reaches them.
11//!
12//! A backend that runs Detcore inside each guest process cannot share one
13//! `Arc` between processes, so a description that a fork shares lives here,
14//! in the global state, and each process's handle is `Shared` (see
15//! `crate::fd::OpenFileState`). A `DetFd` method on a shared handle takes the
16//! model, runs its synchronous body on it, and publishes it back, inside one
17//! method call.
18//!
19//! - Only the thread that holds the scheduler's serial grant may take, so
20//!   access is exclusive and happens in canonical order.
21//! - The global state keeps the canonical model while a lease is out and sends
22//!   a copy. A publish replaces the model only when it names the current lease,
23//!   so a late, duplicate or foreign publish is refused. If the holder's
24//!   process dies mid-lease, the lease ends and its unpublished change is
25//!   discarded.
26//! - Models move in chunks of at most [`CHUNK_LEN`] bytes, because a transport
27//!   frame is bounded and a model (a directory or procfs snapshot) is not.
28//!
29//! Every message here is control traffic: it changes no clock, request slot,
30//! scheduler membership or turn.
31

46/// The largest model chunk one message carries. Transport frames are limited
47/// to 16 MiB. The chunk is much smaller because a guest process transfers it
48/// inside the runtime's fixed callback heap, where the transport's own frame
49/// buffers coexist with it.
50pub const CHUNK_LEN: usize = 1 << 20;
51
52/// One control request about a shared open file description.
53#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
54pub enum OpenFileControl {
55    /// Start a lease on `id` and return the model's first chunk.
56    Take {
57        /// The shared open file description.
58        id: OpenFileId,
59    },
60    /// Return the chunk of the leased model that starts at `offset`.
61    TakeChunk {
62        /// The shared open file description.
63        id: OpenFileId,
64        /// The lease, as [`OpenFileControlReply::Taken`] named it.
65        sequence: u64,
66        /// Where the chunk starts in the encoded model.
67        offset: usize,
68    },
69    /// Append `bytes` at `offset` to the model being published under the
70    /// lease. The last chunk commits the model and ends the lease.
71    PublishChunk {
72        /// The shared open file description.
73        id: OpenFileId,
74        /// The lease being published.
75        sequence: u64,
76        /// Where `bytes` start in the encoded model.
77        offset: usize,
78        /// This chunk of the encoded model.
79        bytes: Vec<u8>,
80        /// Whether this is the last chunk.
81        last: bool,
82    },
83}
84
85/// The answer to an [`OpenFileControl`] request.
86#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
87pub enum OpenFileControlReply {
88    /// A lease started: its sequence, the encoded model's length, and its
89    /// first chunk.
90    Taken {
91        /// Names the lease in every later message about it.
92        sequence: u64,
93        /// The encoded model's length in bytes.
94        length: usize,
95        /// The encoded model's first bytes, at most `CHUNK_LEN` (1 MiB).
96        chunk: Vec<u8>,
97    },
98    /// A further chunk of the leased model.
99    Chunk(Vec<u8>),
100    /// A non-final publish chunk was accepted.
101    Accepted,
102    /// The model was committed and the lease ended.
103    Published,
104}
105
106fn encode_model(model: &OpenFileModel) -> Result<Vec<u8>, SharedOpenFileError> {
107    bincode::serde::encode_to_vec(model, bincode::config::legacy())
108        .map_err(|error| SharedOpenFileError(format!("encoding an open file model: {error}")))
109}
110
111fn decode_model(bytes: &[u8]) -> Result<OpenFileModel, SharedOpenFileError> {
112    let (model, used) =
113        bincode::serde::decode_from_slice::<OpenFileModel, _>(bytes, bincode::config::legacy())
114            .map_err(|error| {
115                SharedOpenFileError(format!("decoding an open file model: {error}"))
116            })?;
117    if used != bytes.len() {
118        return Err(SharedOpenFileError(format!(
119            "an open file model left {} undecoded bytes",
120            bytes.len() - used
121        )));
122    }
123    Ok(model)
124}
125
126fn chunk_at(bytes: &[u8], offset: usize) -> Result<Vec<u8>, SharedOpenFileError> {
127    if offset > bytes.len() {
128        return Err(SharedOpenFileError(format!(
129            "chunk offset {offset} is past the model's {} bytes",
130            bytes.len()
131        )));
132    }
133    let end = bytes.len().min(offset + CHUNK_LEN);
134    Ok(bytes[offset..end].to_vec())
135}

137/// Who holds a lease: the process, for settlement at its death, and the
138/// thread that took it.
139#[derive(Debug, Clone, Copy, PartialEq, Eq)]
140struct Holder {
141    process: DetPid,
142    thread: DetTid,
143}
144
145#[derive(Debug)]
146struct Lease {
147    holder: Holder,
148    sequence: u64,
149    /// The encoded copy being taken.
150    outgoing: Vec<u8>,
151    /// The encoded model being published, chunk by chunk.
152    incoming: Vec<u8>,
153}
154
155#[derive(Debug)]
156struct Canonical {
157    model: OpenFileModel,
158    lease: Option<Lease>,
159}

161/// The global store of shared open file descriptions.
162#[derive(Debug, Default)]
163pub struct SharedOpenFileStore {
164    entries: BTreeMap<OpenFileId, Canonical>,
165    next_sequence: u64,
166}
167
168impl SharedOpenFileStore {
169    /// Makes `model` the canonical model of its open file description. The
170    /// share transaction at a fork is the caller.
171    #[cfg_attr(not(test), allow(dead_code))]
172    pub(crate) fn insert(&mut self, model: OpenFileModel) -> Result<(), SharedOpenFileError> {
173        let id = model.id();
174        if self.entries.contains_key(&id) {
175            return Err(SharedOpenFileError(format!("{id:?} is already shared")));
176        }
177        self.entries.insert(id, Canonical { model, lease: None });
178        Ok(())
179    }
180
181    /// Answers one control request from `thread` of `process`. The caller has
182    /// already checked that `thread` holds the scheduler's serial grant.
183    pub(crate) fn handle(
184        &mut self,
185        process: DetPid,
186        thread: DetTid,
187        request: OpenFileControl,
188    ) -> Result<OpenFileControlReply, SharedOpenFileError> {
189        let holder = Holder { process, thread };
190        match request {
191            OpenFileControl::Take { id } => {
192                let sequence = self.next_sequence + 1;
193                let canonical = self.entry(id)?;
194                if let Some(lease) = &canonical.lease {
195                    return Err(SharedOpenFileError(format!(
196                        "{id:?} is already leased to thread {} (lease {})",
197                        lease.holder.thread, lease.sequence
198                    )));
199                }
200                let outgoing = encode_model(&canonical.model)?;
201                let length = outgoing.len();
202                let chunk = chunk_at(&outgoing, 0)?;
203                canonical.lease = Some(Lease {
204                    holder,
205                    sequence,
206                    outgoing,
207                    incoming: Vec::new(),
208                });
209                self.next_sequence = sequence;
210                Ok(OpenFileControlReply::Taken {
211                    sequence,
212                    length,
213                    chunk,
214                })
215            }
216            OpenFileControl::TakeChunk {
217                id,
218                sequence,
219                offset,
220            } => {
221                let lease = Self::current_lease(self.entry(id)?, id, holder, sequence)?;
222                Ok(OpenFileControlReply::Chunk(chunk_at(
223                    &lease.outgoing,
224                    offset,
225                )?))
226            }
227            OpenFileControl::PublishChunk {
228                id,
229                sequence,
230                offset,
231                bytes,
232                last,
233            } => {
234                let canonical = self.entry(id)?;
235                let lease = Self::current_lease(canonical, id, holder, sequence)?;
236                if offset != lease.incoming.len() {
237                    return Err(SharedOpenFileError(format!(
238                        "{id:?}: publish chunk at {offset}, expected {}",
239                        lease.incoming.len()
240                    )));
241                }
242                lease.incoming.extend_from_slice(&bytes);
243                if !last {
244                    return Ok(OpenFileControlReply::Accepted);
245                }
246                let model = decode_model(&lease.incoming)?;
247                if model.id() != id {
248                    return Err(SharedOpenFileError(format!(
249                        "{id:?}: published the model of {:?}",
250                        model.id()
251                    )));
252                }
253                canonical.model = model;
254                canonical.lease = None;
255                Ok(OpenFileControlReply::Published)
256            }
257        }
258    }
259
260    /// Ends every lease held by `process`, discarding their unpublished
261    /// changes. Called when the process is retired.
262    pub(crate) fn end_leases_of(&mut self, process: DetPid) -> usize {
263        let mut ended = 0;
264        for canonical in self.entries.values_mut() {
265            if canonical
266                .lease
267                .as_ref()
268                .is_some_and(|lease| lease.holder.process == process)
269            {
270                canonical.lease = None;
271                ended += 1;
272            }
273        }
274        ended
275    }
276
277    fn entry(&mut self, id: OpenFileId) -> Result<&mut Canonical, SharedOpenFileError> {
278        self.entries
279            .get_mut(&id)
280            .ok_or_else(|| SharedOpenFileError(format!("{id:?} is not shared")))
281    }
282
283    fn current_lease(
284        canonical: &mut Canonical,
285        id: OpenFileId,
286        holder: Holder,
287        sequence: u64,
288    ) -> Result<&mut Lease, SharedOpenFileError> {
289        match &mut canonical.lease {
290            Some(lease) if lease.sequence == sequence && lease.holder == holder => Ok(lease),
291            _ => Err(SharedOpenFileError(format!(
292                "{id:?}: lease {sequence} of thread {} is not the current lease",
293                holder.thread
294            ))),
295        }
296    }
297
298    #[cfg(test)]
299    fn model(&self, id: OpenFileId) -> &OpenFileModel {
300        &self.entries[&id].model
301    }
302
303    #[cfg(test)]
304    pub(crate) fn is_leased(&self, id: OpenFileId) -> bool {
305        self.entries[&id].lease.is_some()
306    }
307}

309/// Carries one [`OpenFileControl`] request to the global state and returns
310/// its answer, synchronously. A backend implements this over its existing
311/// coordinator connection.
312pub trait OpenFileControlTransport: Send + Sync {
313    /// Sends `request` and waits for the answer.
314    fn call(&self, request: OpenFileControl) -> Result<OpenFileControlReply, SharedOpenFileError>;
315}

317/// A [`SharedOpenFileChannel`] that moves models in chunks over an
318/// [`OpenFileControlTransport`].
319///
320/// It streams both ways: a take decodes the model while it fetches the
321/// chunks, and a publish sends each chunk as soon as the encoder fills it. So
322/// the process never holds more than one chunk of the encoding, beside the
323/// model itself and the transport's frame for that chunk. That matters inside a guest process, where the runtime's
324/// callback heap is small and fixed: a model that fits there as a local
325/// description must also fit while it is taken or published.
326pub struct ChunkedOpenFileChannel<T> {
327    transport: T,
328    /// The most encoded bytes a take or publish has held at once.
329    peak_buffered: AtomicUsize,
330}
331
332impl<T: OpenFileControlTransport> ChunkedOpenFileChannel<T> {
333    /// Wraps `transport`.
334    pub fn new(transport: T) -> Self {
335        Self {
336            transport,
337            peak_buffered: AtomicUsize::new(0),
338        }
339    }
340
341    fn note_buffered(&self, bytes: usize) {
342        self.peak_buffered.fetch_max(bytes, Ordering::Relaxed);
343    }
344}
345
346fn unexpected(reply: OpenFileControlReply) -> SharedOpenFileError {
347    SharedOpenFileError(format!("unexpected open file control reply {reply:?}"))
348}

350/// Reads a leased model's encoding one chunk at a time, fetching the next
351/// chunk only when the decoder has consumed the current one.
352struct TakeReader<'a, T> {
353    channel: &'a ChunkedOpenFileChannel<T>,
354    id: OpenFileId,
355    sequence: u64,
356    length: usize,
357    chunk: Vec<u8>,
358    position: usize,
359    /// Encoded bytes received so far, the current chunk included.
360    received: usize,
361    /// The transport's own error, kept so the caller reports it rather than
362    /// the decoder's.
363    failure: Option<SharedOpenFileError>,
364}
365
366impl<T: OpenFileControlTransport> TakeReader<'_, T> {
367    fn fail(&mut self, error: SharedOpenFileError) -> std::io::Error {
368        let io = std::io::Error::other(error.0.clone());
369        self.failure = Some(error);
370        io
371    }
372
373    fn fetch(&mut self) -> std::io::Result<()> {
374        // The consumed chunk is released before the next one arrives, so the
375        // two never coexist.
376        self.chunk = Vec::new();
377        self.position = 0;
378        let reply = self.channel.transport.call(OpenFileControl::TakeChunk {
379            id: self.id,
380            sequence: self.sequence,
381            offset: self.received,
382        });
383        match reply {
384            Ok(OpenFileControlReply::Chunk(chunk))
385                if !chunk.is_empty()
386                    && chunk.len() <= CHUNK_LEN
387                    && self.received + chunk.len() <= self.length =>
388            {
389                self.received += chunk.len();
390                self.chunk = chunk;
391                self.position = 0;
392                self.channel.note_buffered(self.chunk.len());
393                Ok(())
394            }
395            Ok(other) => Err(self.fail(unexpected(other))),
396            Err(error) => Err(self.fail(error)),
397        }
398    }
399
400    /// Whether the decoder consumed exactly the whole encoding.
401    fn consumed_everything(&self) -> bool {
402        self.received == self.length && self.position == self.chunk.len()
403    }
404}
405
406impl<T: OpenFileControlTransport> std::io::Read for TakeReader<'_, T> {
407    fn read(&mut self, out: &mut [u8]) -> std::io::Result<usize> {
408        if self.position == self.chunk.len() {
409            if self.received == self.length {
410                return Ok(0);
411            }
412            self.fetch()?;
413        }
414        let count = out.len().min(self.chunk.len() - self.position);
415        out[..count].copy_from_slice(&self.chunk[self.position..self.position + count]);
416        self.position += count;
417        Ok(count)
418    }
419}

421/// Collects a model's encoding and sends each full chunk as it fills.
422struct PublishWriter<'a, T> {
423    channel: &'a ChunkedOpenFileChannel<T>,
424    id: OpenFileId,
425    sequence: u64,
426    /// Encoded bytes already sent.
427    offset: usize,
428    buffer: Vec<u8>,
429    failure: Option<SharedOpenFileError>,
430}
431
432impl<T: OpenFileControlTransport> PublishWriter<'_, T> {
433    fn send(&mut self, last: bool) -> Result<(), SharedOpenFileError> {
434        let bytes = std::mem::take(&mut self.buffer);
435        let length = bytes.len();
436        let reply = self.channel.transport.call(OpenFileControl::PublishChunk {
437            id: self.id,
438            sequence: self.sequence,
439            offset: self.offset,
440            bytes,
441            last,
442        })?;
443        match (reply, last) {
444            (OpenFileControlReply::Published, true) | (OpenFileControlReply::Accepted, false) => {
445                self.offset += length;
446                Ok(())
447            }
448            (other, _) => Err(unexpected(other)),
449        }
450    }
451}
452
453impl<T: OpenFileControlTransport> std::io::Write for PublishWriter<'_, T> {
454    fn write(&mut self, bytes: &[u8]) -> std::io::Result<usize> {
455        if self.buffer.len() == CHUNK_LEN
456            && let Err(error) = self.send(false)
457        {
458            let io = std::io::Error::other(error.0.clone());
459            self.failure = Some(error);
460            return Err(io);
461        }
462        let count = bytes.len().min(CHUNK_LEN - self.buffer.len());
463        self.buffer.extend_from_slice(&bytes[..count]);
464        self.channel.note_buffered(self.buffer.len());
465        Ok(count)
466    }
467
468    fn flush(&mut self) -> std::io::Result<()> {
469        Ok(())
470    }
471}

473impl<T: OpenFileControlTransport> SharedOpenFileChannel for ChunkedOpenFileChannel<T> {
474    fn take(&self, id: OpenFileId) -> Result<crate::fd::OpenFileLease, SharedOpenFileError> {
475        let (sequence, length, chunk) = match self.transport.call(OpenFileControl::Take { id })? {
476            OpenFileControlReply::Taken {
477                sequence,
478                length,
479                chunk,
480            } if chunk.len() <= CHUNK_LEN.min(length) => (sequence, length, chunk),
481            other => return Err(unexpected(other)),
482        };
483        self.note_buffered(chunk.len());
484        let mut reader = TakeReader {
485            channel: self,
486            id,
487            sequence,
488            length,
489            received: chunk.len(),
490            chunk,
491            position: 0,
492            failure: None,
493        };
494        let decoded = bincode::serde::decode_from_std_read::<OpenFileModel, _, _>(
495            &mut reader,
496            bincode::config::legacy(),
497        );
498        if let Some(error) = reader.failure.take() {
499            return Err(error);
500        }
501        let model = decoded.map_err(|error| {
502            SharedOpenFileError(format!("decoding an open file model: {error}"))
503        })?;
504        if !reader.consumed_everything() {
505            return Err(SharedOpenFileError(format!(
506                "{id:?}: the model did not use exactly its {length} encoded bytes"
507            )));
508        }
509        Ok(crate::fd::OpenFileLease {
510            id,
511            sequence,
512            model,
513        })
514    }
515
516    fn publish(&self, lease: crate::fd::OpenFileLease) -> Result<(), SharedOpenFileError> {
517        let mut writer = PublishWriter {
518            channel: self,
519            id: lease.id,
520            sequence: lease.sequence,
521            offset: 0,
522            buffer: Vec::new(),
523            failure: None,
524        };
525        let encoded = bincode::serde::encode_into_std_write(
526            &lease.model,
527            &mut writer,
528            bincode::config::legacy(),
529        );
530        if let Some(error) = writer.failure.take() {
531            return Err(error);
532        }
533        encoded.map_err(|error| {
534            SharedOpenFileError(format!("encoding an open file model: {error}"))
535        })?;
536        // The last chunk, possibly empty, commits the model.
537        writer.send(true)
538    }
539}

560    /// Calls the store directly, as the global state does after its grant

564    impl OpenFileControlTransport for Direct<'_> {
565        fn call(
566            &self,
567            request: OpenFileControl,

573    #[test]
574    fn take_and_publish_replace_the_canonical_model() {
575        let id = OpenFileId::new(THREAD, 0);
576        let store = Mutex::new(SharedOpenFileStore::default());
577        store.lock().unwrap().insert(model(id, "/old")).unwrap();
578        let channel = ChunkedOpenFileChannel::new(Direct(&store));

591    /// Calls the store directly and records the largest model chunk any one
592    /// message carried.
593    struct Recording<'a> {
594        store: &'a Mutex<SharedOpenFileStore>,
595        largest: Mutex<usize>,
596        messages: Mutex<usize>,
597    }
598
599    impl OpenFileControlTransport for Recording<'_> {
600        fn call(
601            &self,
602            request: OpenFileControl,
603        ) -> Result<OpenFileControlReply, SharedOpenFileError> {
604            if let OpenFileControl::PublishChunk { bytes, .. } = &request {
605                let mut largest = self.largest.lock().unwrap();
606                *largest = (*largest).max(bytes.len());
607            }
608            let reply = self.store.lock().unwrap().handle(PROCESS, THREAD, request);
609            if let Ok(
610                OpenFileControlReply::Taken { chunk, .. } | OpenFileControlReply::Chunk(chunk),
611            ) = &reply
612            {
613                let mut largest = self.largest.lock().unwrap();
614                *largest = (*largest).max(chunk.len());
615            }
616            *self.messages.lock().unwrap() += 1;
617            reply
618        }
619    }

621    /// A model larger than the 16 MiB transport frame moves in chunks of at
622    /// most `CHUNK_LEN`, both ways, and the channel never holds more than one
623    /// chunk of its encoding: it decodes while it fetches and sends while it
624    /// encodes.
625    #[test]
626    fn a_model_larger_than_a_frame_moves_in_bounded_chunks() {
627        let id = OpenFileId::new(THREAD, 1);
628        let long = format!("/{}", "x".repeat((16 << 20) + 17));
629        let store = Mutex::new(SharedOpenFileStore::default());
630        store.lock().unwrap().insert(model(id, &long)).unwrap();
631        let channel = ChunkedOpenFileChannel::new(Recording {
632            store: &store,
633            largest: Mutex::new(0),
634            messages: Mutex::new(0),
635        });
636
637        let lease = channel.take(id).unwrap();
638        assert_eq!(lease.model.path_for_test(), Some(long.clone().into()));
639        channel.publish(lease).unwrap();
640        assert_eq!(
641            store.lock().unwrap().model(id).path_for_test(),
642            Some(long.into())
643        );
644        let transport = &channel.transport;
645        assert!(*transport.largest.lock().unwrap() <= CHUNK_LEN);
646        let peak = channel.peak_buffered.load(Ordering::Relaxed);
647        assert!(
648            peak > 0 && peak <= CHUNK_LEN,
649            "held {peak} encoded bytes at once"
650        );
651        // At least seventeen chunks each way for a model over 16 MiB.
652        assert!(*transport.messages.lock().unwrap() >= 34);
653    }

682        };
683        let new = encode_model(&model(id, "/new")).unwrap();
684        let (head, tail) = new.split_at(new.len() / 2);
685
686        let sequence = take(&mut store);
687        assert_eq!(
688            publish(&mut store, sequence, 0, head, false),
689            Ok(OpenFileControlReply::Accepted)
690        );
691        assert_eq!(store.model(id).path_for_test(), Some("/old".into()));
692        assert!(store.is_leased(id));
693        // A gap or an overlap is refused.
694        assert!(publish(&mut store, sequence, head.len() + 1, tail, true).is_err());
695        assert!(publish(&mut store, sequence, head.len() - 1, tail, true).is_err());
696        assert_eq!(store.model(id).path_for_test(), Some("/old".into()));
697        // The holder dies before the last chunk: its partial publish is gone.
698        assert_eq!(store.end_leases_of(PROCESS), 1);
699        assert_eq!(store.model(id).path_for_test(), Some("/old".into()));
700        assert!(publish(&mut store, sequence, head.len(), tail, true).is_err());
701
702        // A complete publish of another description's model is refused.
703        let other = encode_model(&model(OpenFileId::new(THREAD, 5), "/other")).unwrap();
704        let sequence = take(&mut store);
705        assert!(publish(&mut store, sequence, 0, &other, true).is_err());
706        assert_eq!(store.model(id).path_for_test(), Some("/old".into()));
707    }

710    fn a_second_take_and_a_stale_publish_are_refused() {
711        let id = OpenFileId::new(THREAD, 2);
712        let mut store = SharedOpenFileStore::default();
713        store.insert(model(id, "/a")).unwrap();
714
715        let OpenFileControlReply::Taken { sequence, .. } = store
716            .handle(PROCESS, THREAD, OpenFileControl::Take { id })
717            .unwrap()
718        else {
719            panic!("expected a lease");
720        };
721        assert!(
722            store
723                .handle(PROCESS, THREAD, OpenFileControl::Take { id })
724                .is_err()
725        );
726        let bytes = encode_model(&model(id, "/b")).unwrap();
727        // A publish under another sequence, or from another thread, is refused
728        // and leaves the lease in place.

739                            bytes: bytes.clone(),

741                        },
742                    )
743                    .is_err()

746        }

751    fn the_death_of_the_holder_ends_its_lease_and_keeps_the_model() {
752        let id = OpenFileId::new(THREAD, 3);
753        let mut store = SharedOpenFileStore::default();
754        store.insert(model(id, "/kept")).unwrap();
755        let OpenFileControlReply::Taken { sequence, .. } = store
756            .handle(PROCESS, THREAD, OpenFileControl::Take { id })
757            .unwrap()
758        else {
759            panic!("expected a lease");
760        };
761
762        assert_eq!(store.end_leases_of(DetPid::from_raw(99)), 0);
763        assert_eq!(store.end_leases_of(PROCESS), 1);
764        assert!(!store.is_leased(id));
765        assert_eq!(store.model(id).path_for_test(), Some("/kept".into()));
766        // The dead holder's late publish names no current lease.
767        assert!(
768            store
769                .handle(
770                    PROCESS,
771                    THREAD,
772                    OpenFileControl::PublishChunk {
773                        id,
774                        sequence,
775                        offset: 0,
776                        bytes: encode_model(&model(id, "/late")).unwrap(),
777                        last: true,
778                    },
779                )
780                .is_err()
781        );
782        assert_eq!(store.model(id).path_for_test(), Some("/kept".into()));
