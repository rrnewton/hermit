304/// The module name for `/sys/module/<name>/refcnt`, or `None` for any other
305/// shape. Returning the name (rather than a bare bool) is what lets the sysfs
306/// value be derived from the same module's `/proc/modules` row.
307fn module_refcnt_name(path: &Path) -> Option<String> {
308    let relative = path.strip_prefix("/sys/module").ok()?;
309    let mut components = relative.iter();
310    let module = components.next().and_then(|part| part.to_str())?;
311    if module.is_empty() || components.next().and_then(|part| part.to_str()) != Some("refcnt") {
312        return None;
313    }
314    components.next().is_none().then(|| module.to_owned())
315}

317/// The host `/proc/modules` text, or empty when it cannot be read. An empty
318/// source yields a holder count of zero, which is the same conservative value
319/// the pre-existing behaviour emitted -- so an unreadable source degrades to
320/// the old answer instead of leaking the native counter.
321fn read_host_modules() -> String {
322    std::fs::read_to_string("/proc/modules").unwrap_or_default()
323}

325/// The deterministic holder cardinality `sanitize_modules` publishes for
326/// `module` in `/proc/modules`: the number of comma-separated holder names in
327/// the fourth field, or zero when that field is `-`.
328///
329/// This is the SAME derivation `sanitize_modules` applies, so the two surfaces
330/// cannot disagree. Linux exposes structural dependency holds in both the
331/// `/proc/modules` use count and `/sys/module/<name>/refcnt`; publishing a
332/// normalized 1 in one place and a hard 0 in the other is not a coherent
333/// module view.
334fn deterministic_module_holder_count(modules: &str, module: &str) -> u64 {
335    modules
336        .lines()
337        .find_map(|line| {
338            let mut fields = line.split_whitespace();
339            (fields.next()? == module).then_some(fields)
340        })
341        .and_then(|mut fields| {
342            fields.next()?; // size
343            fields.next()?; // use count (volatile; recomputed below)
344            let holders = fields.next()?;
345            Some(if holders == "-" {
346                0
347            } else {
348                holders.split(',').filter(|h| !h.is_empty()).count() as u64
349            })
350        })
351        .unwrap_or(0)
352}

354fn sysfs_rtc_kind(path: &Path) -> Option<ProcfsKind> {
355    let relative = path.strip_prefix("/sys/class/rtc").ok()?;
356    let mut components = relative.iter();
357    let rtc = components.next()?.to_str()?;
358    let leaf = components.next()?.to_str()?;
359    if components.next().is_some() {
360        return None;
361    }
362    let rtc_index = rtc.strip_prefix("rtc")?;
363    if rtc_index.is_empty() || !rtc_index.bytes().all(|byte| byte.is_ascii_digit()) {
364        return None;
365    }
366
367    match leaf {
368        "date" => Some(ProcfsKind::SysfsRtcDate),
369        "time" => Some(ProcfsKind::SysfsRtcTime),
370        "since_epoch" => Some(ProcfsKind::SysfsRtcEpoch),
371        _ => None,
372    }
373}

409impl MountInfoSnapshot {
410    /// Build namespace-local identities from one kernel snapshot.
411    ///
412    /// Visible mount IDs are assigned in row order before parent-only IDs.
413    /// Thus a forward parent reference maps to the same identity as its later
414    /// row, a self-parent stays self-parent, and a parent outside the visible
415    /// snapshot still receives a stable distinct ID.  Mount and propagation
416    /// IDs are intentionally scoped to this snapshot because Linux scopes both
417    /// to a mount namespace; a run-global raw-ID pool can alias two namespaces.
418    pub(crate) fn new(
419        rows: Vec<MountInfoRow>,
420        mount_id_order: &[u64],
421        virtualize_metadata: bool,
422        mut devices: BTreeMap<u64, u64>,
423        rewrites: BTreeMap<u64, MountInfoRootRewrite>,
424    ) -> Option<Self> {
425        let visible_mount_order = rows.iter().map(|row| row.raw_mount_id).collect::<Vec<_>>();
426        let visible_mounts = visible_mount_order
427            .iter()
428            .copied()
429            .collect::<std::collections::BTreeSet<_>>();
430        let supplied_mounts = mount_id_order
431            .iter()
432            .copied()
433            .collect::<std::collections::BTreeSet<_>>();
434        let rewrites_are_known = if mount_id_order.is_empty() {
435            rewrites
436                .keys()
437                .all(|mount_id| visible_mounts.contains(mount_id))
438        } else {
439            rewrites
440                .keys()
441                .all(|mount_id| supplied_mounts.contains(mount_id))
442        };
443        if visible_mounts.len() != rows.len()
444            || !rewrites_are_known
445            || rewrites.values().any(|rewrite| {
446                rewrite.deterministic_root.first() != Some(&b'/')
447                    || rewrite
448                        .deterministic_root
449                        .iter()
450                        .any(|byte| byte.is_ascii_control() || *byte == b' ')
451                    || match (&rewrite.raw_root_prefix, &rewrite.deterministic_root_prefix) {
452                        (None, None) => false,
453                        (Some(raw), Some(deterministic)) => {
454                            raw.first() != Some(&b'/')
455                                || deterministic.first() != Some(&b'/')
456                                || raw
457                                    .iter()
458                                    .any(|byte| *byte == b' ' || byte.is_ascii_control())
459                                || deterministic
460                                    .iter()
461                                    .any(|byte| *byte == b' ' || byte.is_ascii_control())
462                        }
463                        _ => true,
464                    }
465                    || match (
466                        &rewrite.raw_mountpoint_prefix,
467                        &rewrite.deterministic_mountpoint_prefix,
468                    ) {
469                        (None, None) => false,
470                        (Some(raw), Some(deterministic)) => {
471                            raw.first() != Some(&b'/')
472                                || deterministic.first() != Some(&b'/')
473                                || raw
474                                    .iter()
475                                    .any(|byte| *byte == b' ' || byte.is_ascii_control())
476                                || deterministic
477                                    .iter()
478                                    .any(|byte| *byte == b' ' || byte.is_ascii_control())
479                        }
480                        _ => true,
481                    }
482            })
483        {
484            return None;
485        }
486
487        let raw_devices = rows
488            .iter()
489            .map(|row| row.raw_device)
490            .collect::<std::collections::BTreeSet<_>>();
491        if virtualize_metadata {
492            // The caller obtains these values from the same run-global
493            // DevicePool used by stat/statx. Reading mountinfo therefore
494            // intentionally pre-populates that pool in snapshot row order;
495            // later metadata syscalls reuse the assigned identities. This is
496            // a within-run consistency guarantee, not unconditional
497            // cross-machine reproducibility: a different filesystem layout
498            // can expose a different sequence of device equivalence classes.
499            if devices.len() != raw_devices.len()
500                || !raw_devices.iter().all(|raw| devices.contains_key(raw))
501            {
502                return None;
503            }
504        } else {
505            // `--no-virtualize-metadata` promises native metadata. Keep the
506            // kernel's device numbers so mountinfo agrees with stat/statx and
507            // do not mutate the DevicePool as a hidden side effect.
508            if !devices.is_empty() {
509                return None;
510            }
511            devices = raw_devices.into_iter().map(|raw| (raw, raw)).collect();
512        }
513
514        let expected_mount_ids = rows
515            .iter()
516            .flat_map(|row| [row.raw_mount_id, row.raw_parent_id])
517            .collect::<std::collections::BTreeSet<_>>();
518        let mut derived_order = Vec::with_capacity(expected_mount_ids.len());
519        let mut seen = std::collections::BTreeSet::new();
520        for row in &rows {
521            if seen.insert(row.raw_mount_id) {
522                derived_order.push(row.raw_mount_id);
523            }
524        }
525        for row in &rows {
526            if seen.insert(row.raw_parent_id) {
527                derived_order.push(row.raw_parent_id);
528            }
529        }
530        let mount_ids = if mount_id_order.is_empty() {
531            derived_order
532                .into_iter()
533                .enumerate()
534                .map(|(index, raw)| (raw, index as u64 + 1))
535                .collect()
536        } else {
537            if supplied_mounts.len() != mount_id_order.len()
538                || !mount_ids_are_ordered_subset(&visible_mount_order, mount_id_order)
539                || !expected_mount_ids
540                    .iter()
541                    .all(|raw| supplied_mounts.contains(raw))
542            {
543                return None;
544            }
545            mount_id_order
546                .iter()
547                .copied()
548                .enumerate()
549                .filter(|(_, raw)| expected_mount_ids.contains(raw))
550                .map(|(index, raw)| (raw, index as u64 + 1))
551                .collect()
552        };
553
554        let mut peer_groups = BTreeMap::new();
555        for row in &rows {
556            for &raw in &row.raw_peer_groups {
557                let next = peer_groups.len() as u64 + 1;
558                peer_groups.entry(raw).or_insert(next);
559            }
560        }
561
562        let root_rewrites = rewrites
563            .iter()
564            .filter(|(raw_mount_id, _)| visible_mounts.contains(raw_mount_id))
565            .map(|(raw_mount_id, rewrite)| (*raw_mount_id, rewrite.deterministic_root.clone()))
566            .collect();
567        let mut root_prefix_rewrites = rewrites
568            .values()
569            .cloned()
570            .filter_map(|rewrite| {
571                rewrite
572                    .raw_root_prefix
573                    .zip(rewrite.deterministic_root_prefix)
574            })
575            .collect::<Vec<_>>();
576        root_prefix_rewrites.sort_by_key(|(raw, _)| std::cmp::Reverse(raw.len()));
577        let mut mountpoint_prefix_rewrites = rewrites
578            .into_values()
579            .filter_map(|rewrite| {
580                rewrite
581                    .raw_mountpoint_prefix
582                    .zip(rewrite.deterministic_mountpoint_prefix)
583            })
584            .collect::<Vec<_>>();
585        mountpoint_prefix_rewrites.sort_by_key(|(raw, _)| std::cmp::Reverse(raw.len()));
586
587        Some(Self {
588            rows,
589            mount_ids,
590            devices,
591            peer_groups,
592            root_rewrites,
593            root_prefix_rewrites,
594            mountpoint_prefix_rewrites,
595        })
596    }

603    /// Raw mount IDs in the canonical order assigned by this snapshot.
604    pub(crate) fn raw_mount_id_order(&self) -> Vec<u64> {
605        let mut ordered = self
606            .mount_ids
607            .iter()
608            .map(|(raw, canonical)| (*canonical, *raw))
609            .collect::<Vec<_>>();
610        ordered.sort_unstable();

2376/// Zero the host-assigned identities in `/proc/net/tcp{,6}` and
2377/// `/proc/net/udp{,6}`.
2378///
2379/// Two columns carry host state and both differ on every run. Measured under
2380/// `hermit run --strict`, two runs of a guest binding two listeners:
2381///
2382/// ```text
2383/// run 1  ... 0 3066787052 1 0000000042a27170 100 ...
2384/// run 2  ... 0 3071433840 1 000000005b1c887b 100 ...
2385/// ```
2386///
2387/// `inode` is a host-global counter and `pointer` is the kernel socket address.
2388/// This mirrors `sanitize_netlink_sockets`, which zeroes the same two kinds of
2389/// field, and it is the text half of the `AF_INET`/`AF_INET6` work in
2390/// `crate::sock_diag`: the binary and text views of this table agreed before
2391/// (both leaking) and must still agree after.
2392///
2393/// Zero is in range for `inode` rather than a substitute chosen here. The
2394/// kernel emits it itself for sockets with no owning file -- on the host that
2395/// developed this, `/proc/net/tcp` carried 176 rows of which 12 had inode 0,
2396/// all in state 06 (`TCP_TIME_WAIT`).
2397///
2398/// Row ORDER also varies and is canonicalized here, which zeroing alone does
2399/// not fix. Measured after the two columns above were already zeroed, six runs
2400/// of the same guest binding ports 8000 and 8001: port 8000 appeared in row 0
2401/// on four runs and in row 1 on the other two. The kernel walks a hash table,
2402/// so the order depends on host-side bucket state. Rows are therefore sorted on
2403/// their content with `sl` excluded, and `sl` is then renumbered to match --
2404/// sorting without renumbering would leave a positional index contradicting its
2405/// own position.
2406///
2407/// Fail-open on any unexpected shape, like the other sanitizers.
2408fn sanitize_inet_sockets(contents: &[u8]) -> Vec<u8> {
2409    /// `inode`, counting from zero.
2410    const INODE_FIELD: usize = 9;
2411    /// The kernel socket pointer, printed `%pK` as 16 lower-hex digits.
2412    const POINTER_FIELD: usize = 11;
2413    /// `udp`/`udp6` rows are 13 fields, `tcp`/`tcp6` 17. Both put the two
2414    /// host-state columns at the same index, so only the shorter is required.
2415    const MIN_FIELDS: usize = 13;
2416    const ZERO_POINTER: &str = "0000000000000000";
2417
2418    let Ok(text) = std::str::from_utf8(contents) else {
2419        return contents.to_vec();
2420    };
2421    let Some(body) = text.strip_suffix('\n') else {
2422        return contents.to_vec();
2423    };
2424    let mut lines = body.split('\n');
2425    let Some(header) = lines.next() else {
2426        return contents.to_vec();
2427    };
2428    // The four files have four different headers, so this checks the shape
2429    // rather than an exact string: first column `sl`, and an `inode` column.
2430    let header_fields = header.split_whitespace().collect::<Vec<_>>();
2431    if header_fields.first() != Some(&"sl") || !header_fields.contains(&"inode") {
2432        return contents.to_vec();
2433    }
2434
2435    let mut rows = Vec::new();
2436    for line in lines {
2437        let mut fields = line.split_whitespace().collect::<Vec<_>>();
2438        if fields.len() < MIN_FIELDS {
2439            return contents.to_vec();
2440        }
2441        if parse_decimal(fields[INODE_FIELD]).is_none() || !is_lower_hex(fields[POINTER_FIELD], 16)
2442        {
2443            return contents.to_vec();
2444        }
2445        fields[INODE_FIELD] = "0";
2446        fields[POINTER_FIELD] = ZERO_POINTER;
2447        // Sort on everything but the index, then reassign the index, so the
2448        // ordering key cannot be perturbed by the very column it decides.
2449        rows.push(fields[1..].join(" "));
2450    }
2451    rows.sort_unstable();
2452
2453    let mut normalized = String::with_capacity(text.len());
2454    normalized.push_str(header);
2455    normalized.push('\n');
2456    for (index, row) in rows.iter().enumerate() {
2457        normalized.push_str(&format!("{index}: {row}\n"));
2458    }
2459    normalized.into_bytes()
2460}

3072/// Rewrite the device and inode of one mapping header from the caller-supplied
3073/// table, preserving every other byte of the line and the COLUMN its pathname
3074/// starts in.
3075///
3076/// A raw pair absent from the table is left ALONE rather than guessed at: the
3077/// table is built from this same snapshot, so a miss means the pair was not
3078/// determinizable (an anonymous mapping), and inventing a value here is the
3079/// precise failure this design exists to avoid.
3080///
3081/// ⚠️ PRESERVE THE COLUMN, NOT THE PADDING BYTES. This used to copy the raw
3082/// whitespace run after the inode verbatim, which silently republished the
3083/// HOST inode's digit count as line length. The kernel pads the pathname to a
3084/// fixed column (`seq_setwidth(m, 25 + sizeof(void *) * 6)`, measured at column
3085/// 73 on x86_64), so that run is `73 - len(host_inode_digits)` characters long.
3086/// Substituting a short determinized inode into it therefore left a line whose
3087/// every VISIBLE field was deterministic and whose LENGTH was not.
3088///
3089/// Measured 2026-08-26 at hermit `d741307158`: for a LiteInst guest, whose
3090/// `/memfd:liteinst2-trampoline` mappings take inodes from a host-global
3091/// counter, `cat /proc/self/maps` under `--backend liteinst --strict` produced
3092/// TWO distinct outputs in 20 runs (14x one, 6x the other) differing in nothing
3093/// but one space on each memfd line -- the runs where the host counter had
3094/// crossed a power of ten. With `--detlog-io-buffers` hashing `read()` payloads
3095/// that is a hard `--verify` divergence, and `/usr/bin/tail -n 2 <fixture>`
3096/// under LiteInst failed 10 of 20 verified runs on it.
3097///
3098/// Padding to the column the RAW line used keeps this in step with the kernel
3099/// rather than hard-coding 73, and that column does not depend on the inode.
3100fn rewrite_mapping_header(line: &str, table: &BTreeMap<(u64, u64), (u64, u64)>) -> String {
3101    let Some(raw) = mapping_header_identity(line) else {
3102        return line.to_string();
3103    };
3104    let Some((det_dev, det_inode)) = table.get(&raw).copied() else {
3105        return line.to_string();
3106    };
3107    let mut out = String::with_capacity(line.len());
3108    let mut remaining = line;
3109    // Copy the first three fields (range, perms, offset) verbatim, whitespace
3110    // included, then substitute the fourth and fifth.
3111    for _ in 0..3 {
3112        let field_end = match remaining.find(char::is_whitespace) {
3113            Some(index) => index,
3114            None => return line.to_string(),
3115        };
3116        let gap_end = remaining[field_end..]
3117            .find(|c: char| !c.is_whitespace())
3118            .map_or(remaining.len(), |offset| field_end + offset);
3119        out.push_str(&remaining[..gap_end]);
3120        remaining = &remaining[gap_end..];
3121    }
3122    let dev_end = match remaining.find(char::is_whitespace) {
3123        Some(index) => index,
3124        None => return line.to_string(),
3125    };
3126    let dev_gap = remaining[dev_end..]
3127        .find(|c: char| !c.is_whitespace())
3128        .map_or(remaining.len(), |offset| dev_end + offset);
3129    out.push_str(&format!(
3130        "{:02x}:{:02x}",
3131        libc::major(det_dev),
3132        libc::minor(det_dev)
3133    ));
3134    out.push_str(&remaining[dev_end..dev_gap]);
3135    remaining = &remaining[dev_gap..];
3136    let inode_end = remaining
3137        .find(char::is_whitespace)
3138        .unwrap_or(remaining.len());
3139    let tail = &remaining[inode_end..];
3140    let pad_len = tail
3141        .find(|c: char| !c.is_whitespace())
3142        .unwrap_or(tail.len());
3143    // Column the pathname starts in on the raw line. `line.len() - remaining.len()`
3144    // is how much of the line has already been consumed by the first four fields.
3145    let name_column = (line.len() - remaining.len()) + inode_end + pad_len;
3146    out.push_str(&det_inode.to_string());
3147    let name = &tail[pad_len..];
3148    if name.is_empty() {
3149        // No pathname, so there is no column to preserve and nothing to pad to.
3150        // Trailing whitespace here would be pure host-inode-width leakage.
3151        return out;
3152    }
3153    // A determinized inode can in principle be WIDER than the raw one, which
3154    // leaves no room to pad. One space is what the kernel's `seq_pad` emits in
3155    // that same situation.
3156    if out.len() < name_column {
3157        out.extend(std::iter::repeat_n(' ', name_column - out.len()));
3158    } else {
3159        out.push(' ');
3160    }
3161    out.push_str(name);
3162    out
3163}

6425    /// ROUND-2 REGRESSION GUARD for the scoped fd-resolution fallback.
6426    ///
6427    /// The first version of this fix resolved EVERY open through
6428    /// `/proc/<pid>/fd/<fd>`, which silently reclassified the spelling-defined
6429    /// kinds: `/proc/self/stat` resolves to `/proc/<pid>/stat`, a DIFFERENT
6430    /// kind with a different sanitizer, and the `thread-self` aliases resolve
6431    /// away entirely. The call site now classifies the spelling first and only
6432    /// consults the resolved path when the spelling yields nothing, so the
6433    /// fallback is MONOTONE.
6434    ///
6435    /// This test pins both halves of that property at the classifier level:
6436    /// the spelling-defined kinds keep their own classification (so resolution
6437    /// must never be consulted for them), and the AT_FDCWD/alias shapes have
6438    /// none (so resolution is the only thing that can classify them).
6439    #[test]
6440    fn spelling_defined_kinds_classify_without_resolution() {
6441        // REGRESSION GUARD: each of these classifies from the SPELLING. If any
6442        // stopped classifying, the call site would fall through to the resolved
6443        // numeric path and silently change the sanitizer applied.
6444        let spelling_defined = [
6445            ("/proc/self/stat", ProcfsKind::Stat),
6446            ("/proc/self/status", ProcfsKind::Status),
6447            ("/proc/thread-self/stat", ProcfsKind::ThreadStat),
6448            ("/proc/thread-self/status", ProcfsKind::ThreadStatus),
6449            ("/proc/self/statm", ProcfsKind::Statm),
6450            ("/proc/thread-self/statm", ProcfsKind::Statm),
6451            ("/proc/self/mountinfo", ProcfsKind::Mountinfo),
6452        ];
6453        let mut guarded = 0;
6454        for (path, expected) in spelling_defined {
6455            let kind = ProcfsFile::from_path(Path::new(path))
6456                .unwrap_or_else(|| panic!("{path} must classify from its spelling alone"))
6457                .kind;
6458            assert_eq!(kind, expected, "{path} must keep its spelling-defined kind");
6459            guarded += 1;
6460        }
6461        assert_eq!(guarded, 7, "all seven spelling-defined kinds were guarded");
6462
6463        // And the numeric form these would resolve TO is a DIFFERENT answer --
6464        // which is exactly why unconditional resolution was a regression.
6465        let numeric = ProcfsFile::from_path(Path::new("/proc/1234/stat"))
6466            .expect("numeric process stat still classifies")
6467            .kind;
6468        assert_ne!(
6469            numeric,
6470            ProcfsKind::Stat,
6471            "/proc/<pid>/stat is a different kind from /proc/self/stat; \
6472             resolving the latter into the former is the regression this guards"
6473        );
6474
6475        // POSITIVE (the gap the fallback exists to close): the AT_FDCWD/alias
6476        // spellings classify as NOTHING on their own, so the resolved path is
6477        // the only thing that can classify them -- and it does.
6478        let mut unresolved = 0;
6479        for path in ["refcnt", "kvm/refcnt", "sys/module/kvm/refcnt"] {
6480            assert!(
6481                ProcfsFile::from_path(Path::new(path)).is_none(),
6482                "{path} must not classify lexically; only resolution can"
6483            );
6484            unresolved += 1;
6485        }
6486        assert_eq!(unresolved, 3, "all three relative spellings were checked");
6487        assert_eq!(
6488            ProcfsFile::from_path(Path::new("/sys/module/kvm/refcnt"))
6489                .expect("the resolved form classifies")
6490                .kind,
6491            ProcfsKind::ModuleRefcnt("kvm".to_owned()),
6492            "resolution turns the unclassifiable spelling into the real kind"
6493        );
6494    }
