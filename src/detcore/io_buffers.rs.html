9//! Hash the bytes a syscall moved through a guest buffer, at the syscall
10//! boundary.
11//!
12//! WHAT GAP THIS FILLS. `--detlog-stack` and `--detlog-heap` hash a whole named
13//! mapping, so what they can see is decided by where the guest happened to
14//! ALLOCATE a buffer rather than by what the syscall did. Measured 2026-08-20,
15//! three runs per cell, running the same netlink `RTM_GETLINK` exchange and
16//! changing only the receive buffer's home:
17//!
18//! | buffer home     | `--detlog-stack` | `--detlog-heap` | both   |
19//! |-----------------|------------------|-----------------|--------|
20//! | `[stack]`       | CAUGHT           | MISSED          | CAUGHT |
21//! | `[heap]` (brk)  | MISSED           | CAUGHT          | CAUGHT |
22//! | BSS / static    | MISSED           | MISSED          | MISSED |
23//! | anonymous mmap  | MISSED           | MISSED          | MISSED |
24//!
25//! Two of the four are invisible even with both flags on, and anonymous mmap is
26//! not a corner case: it is where glibc puts any `malloc` above the 128 KiB
27//! `M_MMAP_THRESHOLD`. Taking the address and length from the SYSCALL ARGUMENTS
28//! instead makes all four rows irrelevant by construction.
29//!
30//! WHAT IT CATCHES THAT `--verify` CANNOT. `--verify` compares the INFO record,
31//! and a syscall whose output buffer is typed as a bare pointer in Reverie
32//! prints the address, not the contents (`reverie-syscalls/src/syscalls.rs`
33//! carries standing TODOs saying exactly this for `Read` and `Write`). So a
34//! `recvmsg` that returns a stable `Ok(1468)` while four bytes of its payload
35//! differ produces a character-identical record and `--verify` reports
36//! `bitwise_parity: true`. Measured on one QEMU/Linux boot, 278,824 of 632,228
37//! syscalls (44.1%) move bytes through a buffer whose content the log never
38//! shows.
39//!
40//! COST SHAPE, and it differs from the whole-mapping flags in kind rather than
41//! degree. `--detlog-heap` is `O(syscalls x region_size)` -- it re-reads the
42//! entire heap after every syscall, hashing 10.9 TB per boot to watch a 16.49
43//! MiB heap. This is `O(bytes the syscalls actually returned)`: 139.1 MB per
44//! boot, measured by summing real return values.

59/// One contiguous run of guest bytes a completed syscall moved.
60#[derive(Debug, Clone, Copy, PartialEq, Eq)]
61pub struct BufferExtent {
62    /// Guest virtual address of the first byte.
63    pub addr: u64,
64    /// Number of bytes actually moved, which is bounded by the syscall's return
65    /// value and not by the buffer's declared capacity.
66    pub len: u64,
67}

69/// Bind successful RNG output to the very array the synchronous handler used.
70/// The return count, not the declared capacity, bounds each observed extent.
71pub(crate) fn rng_readv_extents(
72    iovecs: &[crate::iovecs::ImportedIovec],
73    written: usize,
74) -> Result<Vec<BufferExtent>, Error> {
75    let mut output = rng_observation_vec(iovecs.len(), "imported geometry")?;
76    let mut remaining = written;
77    for iov in iovecs {
78        let take = remaining.min(iov.len);
79        remaining -= take;
80        if take > 0 {
81            output.push(BufferExtent {
82                addr: iov.base as u64,
83                len: take as u64,
84            });
85        }
86    }
87    Ok(output)
88}

90/// RNG observation happens after output and any shared-cursor commit. A
91/// recoverable reservation failure must stop the tool, not become guest errno.
92/// This does not promise to catch physical OOM or other logging allocations.
93fn rng_observation_vec<T>(capacity: usize, purpose: &str) -> Result<Vec<T>, Error> {
94    let mut output = Vec::new();
95    output.try_reserve_exact(capacity).map_err(|error| {
96        Error::Tool(anyhow::anyhow!(
97            "RNG vector observation after output commit: cannot reserve {purpose}: {error}"
98        ))
99    })?;
100    Ok(output)
101}

103fn observed_extents(
104    memory: &impl MemoryAccess,
105    call: &Syscall,
106    ret: i64,
107    rng_output: Option<&[BufferExtent]>,
108) -> Result<Vec<BufferExtent>, Error> {
109    if let Some(output) = rng_output {
110        if ret <= 0 {
111            return Ok(Vec::new());
112        }
113        let mut observed = rng_observation_vec(output.len(), "observed geometry")?;
114        let mut remaining = ret as u64;
115        for extent in output {
116            let take = remaining.min(extent.len);
117            remaining -= take;
118            if take > 0 {
119                observed.push(BufferExtent {
120                    addr: extent.addr,
121                    len: take,
122                });
123            }
124        }
125        return Ok(observed);
126    }
127    extents(memory, call, ret)
128}

130/// Direction of travel, recorded so a reader can tell a value the kernel
131/// produced from one the guest produced without knowing every syscall.
132#[derive(Debug, Clone, Copy, PartialEq, Eq)]
133pub enum Direction {
134    /// The kernel filled the buffer: an unvirtualized field here is a leak INTO
135    /// the guest, which is the netlink case.
136    In,
137    /// The guest filled the buffer: a difference here means the guest computed
138    /// different bytes, which is how divergence reaches a file or a device.
139    Out,
140}

142impl Direction {
143    fn as_str(self) -> &'static str {
144        match self {
145            Direction::In => "in",
146            Direction::Out => "out",
147        }
148    }

151/// Clamp a declared buffer to what the syscall says it actually moved.
152///
153/// `capacity` is the argument the guest passed; `moved` is the return value.
154/// Hashing `capacity` would fold in bytes the syscall never wrote, which are
155/// whatever was in the buffer beforehand -- for a stack buffer that is previous
156/// frames, so the hash would report divergence for an unrelated reason.
157fn clamp(addr: Option<u64>, capacity: usize, moved: i64) -> Vec<BufferExtent> {
158    let Some(addr) = addr else { return Vec::new() };
159    let moved = u64::try_from(moved).unwrap_or(0);
160    let len = moved.min(capacity as u64);
161    if len == 0 {
162        return Vec::new();
163    }
164    vec![BufferExtent { addr, len }]
165}

167/// A buffer written wholly in place, whose size does not depend on the return
168/// value (`poll`'s `revents` are rewritten across the whole array regardless of
169/// how many descriptors were ready).
170fn whole(addr: Option<u64>, len: u64) -> Vec<BufferExtent> {
171    match addr {
172        Some(addr) if len > 0 => vec![BufferExtent { addr, len }],
173        _ => Vec::new(),
174    }
175}

177/// Walk an `iovec` array and return the segments the syscall actually filled,
178/// bounded by `moved`.
179///
180/// Mirrors the existing traversal in `crate::syscalls::io`: clamp the count to
181/// `UIO_MAXIOV`, skip null/empty segments, and stop once `moved` bytes are
182/// accounted for. Under `MSG_TRUNC` the returned count can exceed the buffers'
183/// capacity, which is why the running remainder rather than `moved` alone
184/// bounds each segment.
185fn iovec_extents<M: MemoryAccess>(
186    memory: &M,
187    iov_addr: usize,
188    iov_count: usize,
189    moved: i64,
190) -> Result<Vec<BufferExtent>, Error> {
191    let remaining = u64::try_from(moved).unwrap_or(0);
192    if iov_addr == 0 || iov_count == 0 || remaining == 0 {
193        return Ok(Vec::new());
194    }
195    let iov_count = iov_count.min(libc::UIO_MAXIOV as usize);
196    let iov_address: AddrMut<'_, libc::iovec> = AddrMut::from_raw(iov_addr).ok_or(Errno::EFAULT)?;
197    // SAFETY: `iovec` is a plain C record; an all-zero value is a valid staging
198    // value that `read_values` immediately overwrites.
199    let mut iovecs: Vec<libc::iovec> = (0..iov_count)
200        .map(|_| unsafe { std::mem::zeroed() })
201        .collect();
202    memory.read_values(iov_address.into(), &mut iovecs)?;
203    Ok(iovec_extents_from_slice(&iovecs, moved))
204}

206/// Apply the return-value bound after the guest `iovec` array has been read.
207///
208/// This is separate from the memory access so the short-transfer rule can be
209/// tested directly for every syscall family that shares it.
210fn iovec_extents_from_slice(iovecs: &[libc::iovec], moved: i64) -> Vec<BufferExtent> {
211    let mut remaining = u64::try_from(moved).unwrap_or(0);
212
213    let mut out = Vec::new();
214    for iov in iovecs {
215        if remaining == 0 {
216            break;
217        }
218        if iov.iov_base.is_null() || iov.iov_len == 0 {
219            continue;
220        }
221        let take = (iov.iov_len as u64).min(remaining);
222        out.push(BufferExtent {
223            addr: iov.iov_base as u64,
224            len: take,
225        });
226        remaining -= take;
227    }
228    out
229}

231/// Return the guest `iovec` array described by a vectored I/O syscall.
232///
233/// Keep the family in one match so adding a syscall variant cannot update the
234/// log direction without also making its complete, return-value-bounded iovec
235/// prefix available to [`extents`].
236fn iovec_extent_arguments(call: &Syscall) -> Option<(usize, usize)> {
237    match call {
238        Syscall::Readv(call) => Some((call.iov().map_or(0, |p| p.as_raw()), call.len())),
239        Syscall::Preadv(call) => Some((call.iov().map_or(0, |p| p.as_raw()), call.iov_len())),
240        Syscall::Preadv2(call) => Some((
241            call.iov().map_or(0, |p| p.as_raw()),
242            usize::try_from(call.iov_len()).unwrap_or(usize::MAX),
243        )),
244        Syscall::Writev(call) => Some((call.iov().map_or(0, |p| p.as_raw()), call.len())),
245        Syscall::Pwritev(call) => Some((call.iov().map_or(0, |p| p.as_raw()), call.iov_len())),
246        Syscall::Pwritev2(call) => Some((
247            call.iov().map_or(0, |p| p.as_raw()),
248            usize::try_from(call.iov_len()).unwrap_or(usize::MAX),
249        )),
250        _ => None,
251    }
252}

254/// Read a `msghdr` out of the guest and walk the iovecs it points at.
255fn msghdr_extents<M: MemoryAccess>(
256    memory: &M,
257    msg_addr: usize,
258    moved: i64,
259) -> Result<Vec<BufferExtent>, Error> {
260    if msg_addr == 0 {
261        return Ok(Vec::new());
262    }
263    let address: AddrMut<'_, libc::msghdr> = AddrMut::from_raw(msg_addr).ok_or(Errno::EFAULT)?;
264    let message: libc::msghdr = memory.read_value(address)?;
265    iovec_extents(memory, message.msg_iov as usize, message.msg_iovlen, moved)
266}

268/// Number of completed messages whose per-message lengths are meaningful.
269fn completed_mmsghdr_count(vlen: u32, completed: i64) -> usize {
270    usize::try_from(completed)
271        .unwrap_or(0)
272        .min(vlen as usize)
273        .min(libc::UIO_MAXIOV as usize)
274}

276/// Walk the `mmsghdr` array a batch send or receive completed.
277///
278/// `moved` here is a COUNT OF MESSAGES, not a byte count -- which is why
279/// these calls cannot share the `clamp`/`msghdr_extents` path that every other
280/// send or receive uses. Each completed message carries its own byte count in
281/// `msg_len`, so each is walked separately and bounded by that; treating the
282/// batch as one buffer would let one message's length run into the next
283/// message's memory.
284fn mmsghdr_extents<M: MemoryAccess>(
285    memory: &M,
286    mmsg_addr: usize,
287    vlen: u32,
288    delivered: i64,
289) -> Result<Vec<BufferExtent>, Error> {
290    let count = completed_mmsghdr_count(vlen, delivered);
291    if mmsg_addr == 0 || count == 0 {
292        return Ok(Vec::new());
293    }
294    let address: AddrMut<'_, libc::mmsghdr> = AddrMut::from_raw(mmsg_addr).ok_or(Errno::EFAULT)?;
295    // SAFETY: `mmsghdr` is a plain C record; an all-zero value is a valid
296    // staging value that `read_values` immediately overwrites.
297    let mut headers: Vec<libc::mmsghdr> =
298        (0..count).map(|_| unsafe { std::mem::zeroed() }).collect();
299    memory.read_values(address.into(), &mut headers)?;
300
301    let mut out = Vec::new();
302    for header in &headers {
303        out.extend(iovec_extents(
304            memory,
305            header.msg_hdr.msg_iov as usize,
306            header.msg_hdr.msg_iovlen,
307            i64::from(header.msg_len),
308        )?);
309    }
310    Ok(out)
311}

313/// Whether this syscall's return value bounds what it wrote.
314///
315/// True for the ordinary case, where the return IS the byte count, so a return
316/// of 0 or less means nothing was written. FALSE for `poll`/`ppoll`, whose
317/// return is a COUNT OF READY DESCRIPTORS: the kernel rewrites `revents` on
318/// every entry even when it returns 0. Measured -- `poll(timeout=0)` on two
319/// never-ready pipe fds returns 0 and still moves `revents` from a poisoned
320/// 0x7FFF to 0 on both entries.
321///
322/// This exists as a predicate consulted BY the single `ret <= 0` guard, rather
323/// than as poll arms placed above it, so the guard cannot be reordered back
324/// into swallowing the zero-ready case. The original gap was exactly that: the
325/// guard sat above the poll arms, and the unit test called `whole` directly, so
326/// it could not see the short-circuit above the code it exercised.
327/// Every syscall `extents` below returns a non-empty result for.
328///
329/// KEEP THIS IN STEP WITH THE `extents` MATCH -- it is the same set written
330/// twice, once as `Syscall::` arms (which need constructed calls and a `Guest`
331/// to exercise) and once as bare `Sysno`s (which a subscription can be asked
332/// about). The second spelling exists so the relationship between this check
333/// and what Detcore actually intercepts can be ASSERTED rather than assumed;
334/// see `passthru_opt_leaves_io_buffer_hashing_blind_only_for_getcwd` in
335/// `lib.rs`. Adding an arm to `extents` without adding it here does not break
336/// that test, so add both.
337#[cfg(test)]

339    // Bytes the kernel produced.
340    reverie::syscalls::Sysno::read,
341    reverie::syscalls::Sysno::pread64,
342    reverie::syscalls::Sysno::recvfrom,
343    reverie::syscalls::Sysno::getrandom,
344    reverie::syscalls::Sysno::getcwd,
345    reverie::syscalls::Sysno::getdents64,
346    reverie::syscalls::Sysno::readlink,
347    reverie::syscalls::Sysno::readlinkat,
348    reverie::syscalls::Sysno::recvmsg,
349    reverie::syscalls::Sysno::recvmmsg,
350    reverie::syscalls::Sysno::readv,
351    reverie::syscalls::Sysno::preadv,
352    reverie::syscalls::Sysno::preadv2,
353    // Bytes the guest produced.
354    reverie::syscalls::Sysno::write,
355    reverie::syscalls::Sysno::pwrite64,
356    reverie::syscalls::Sysno::sendto,

367fn ret_gates_output(call: &Syscall) -> bool {
368    !matches!(call, Syscall::Poll(_) | Syscall::Ppoll(_))

371/// The extents a completed syscall moved, or `None` when this syscall has no
372/// output buffer worth hashing.
373///
374/// Only syscalls whose buffer CONTENT the INFO record does not already show are
375/// listed. `clock_gettime` and `newfstatat`, for instance, are absent because
376/// Reverie's typed display already dereferences and prints their output.
377fn extents<M: MemoryAccess>(
378    memory: &M,
379    call: &Syscall,
380    ret: i64,
381) -> Result<Vec<BufferExtent>, Error> {
382    // Nothing was written on a failed or empty call -- for every syscall whose
383    // return value is a byte count. `ret_gates_output` is what keeps the poll
384    // family out of this, and it is a predicate rather than an arm placed above
385    // so the exclusion cannot be undone by moving code.
386    if ret <= 0 && ret_gates_output(call) {
387        return Ok(Vec::new());
388    }
389    if let Some((iov_addr, iov_count)) = iovec_extent_arguments(call) {
390        return iovec_extents(memory, iov_addr, iov_count, ret);
391    }
392    let raw = |a: Option<AddrMut<'_, u8>>| a.map(|p| p.as_raw() as u64);
393    Ok(match call {
394        // Bytes the kernel produced.
395        Syscall::Read(c) => clamp(raw(c.buf()), c.len(), ret),
396        Syscall::Pread64(c) => clamp(raw(c.buf()), c.len(), ret),
397        Syscall::Recvfrom(c) => clamp(c.buf().map(|p| p.as_raw() as u64), c.len(), ret),
398        Syscall::Getrandom(c) => clamp(raw(c.buf()), c.buflen(), ret),
399        Syscall::Getcwd(c) => clamp(c.buf().map(|p| p.as_raw() as u64), c.size(), ret),
400        Syscall::Getdents64(c) => clamp(
401            c.dirent().map(|p| p.as_raw() as u64),
402            c.count() as usize,
403            ret,
404        ),
405        Syscall::Readlink(c) => clamp(c.buf().map(|p| p.as_raw() as u64), c.bufsize(), ret),
406        Syscall::Readlinkat(c) => clamp(c.buf().map(|p| p.as_raw() as u64), c.buf_len(), ret),
407        Syscall::Recvmsg(c) => msghdr_extents(memory, c.msg().map_or(0, |p| p.as_raw()), ret)?,
408        // `ret` is a MESSAGE count here, not a byte count; see
409        // `mmsghdr_extents`. recvmmsg is one of the four receive syscalls
410        // that could reach a NETLINK_SOCK_DIAG dump without passing the
411        // sock_diag sanitizer, so leaving it unhashed left this check blind
412        // to exactly the bypass it would otherwise have reported.
413        Syscall::Recvmmsg(c) => {
414            mmsghdr_extents(memory, c.mmsg().map_or(0, |p| p.as_raw()), c.vlen(), ret)?
415        }
416        // Bytes the guest produced. These never reach stdout/stderr for a QEMU
417        // boot -- measured, all 234,872 writes went to fds 7/12/14/11/13/4/8/19/23
418        // and none to fd 1 or 2 -- so `--verify`'s stdout/stderr comparison does
419        // not cover them either.
420        Syscall::Write(c) => clamp(c.buf().map(|p| p.as_raw() as u64), c.len(), ret),
421        Syscall::Pwrite64(c) => clamp(c.buf().map(|p| p.as_raw() as u64), c.len(), ret),
422        Syscall::Sendto(c) => clamp(c.buf().map(|p| p.as_raw() as u64), c.size(), ret),
423        Syscall::Sendmsg(c) => msghdr_extents(memory, c.msg().map_or(0, |p| p.as_raw()), ret)?,
424        // Like recvmmsg, `ret` counts completed messages and each completed
425        // header's `msg_len` bounds the bytes consumed from that message.
426        Syscall::Sendmmsg(c) => {
427            mmsghdr_extents(memory, c.msgvec().map_or(0, |p| p.as_raw()), c.vlen(), ret)?
428        }
429        // Rewritten in place across the WHOLE array: `poll` sets `revents` on
430        // every entry, not just on the `ret` that were ready, so the extent is
431        // the array and not a prefix of it -- and it is reached even when
432        // `ret == 0`, via `ret_gates_output`.
433        Syscall::Poll(c) => whole(
434            c.fds().map(|p| p.as_raw() as u64),
435            c.nfds() * std::mem::size_of::<libc::pollfd>() as u64,
436        ),
437        Syscall::Ppoll(c) => whole(
438            c.fds().map(|p| p.as_raw() as u64),
439            c.nfds() * std::mem::size_of::<libc::pollfd>() as u64,
440        ),
441
442        _ => Vec::new(),
443    })
444}

446/// Which way the bytes travelled, for the record's label.
447fn direction(call: &Syscall) -> Direction {
448    match call {
449        Syscall::Write(_)
450        | Syscall::Pwrite64(_)
451        | Syscall::Sendto(_)
452        | Syscall::Sendmsg(_)
453        | Syscall::Sendmmsg(_)
454        | Syscall::Writev(_)
455        | Syscall::Pwritev(_)
456        | Syscall::Pwritev2(_) => Direction::Out,
457        _ => Direction::In,
458    }

461/// Emit one deterministic record per buffer a completed syscall moved.
462///
463/// Digests that LOCATE a divergence instead of only detecting one.
464///
465/// The whole-extent digest says two buffers differ and nothing else -- not
466/// which bytes, not which field. Comparing two logs of per-chunk digests names
467/// the FIRST DIFFERING CHUNK, which is the offset and a bounded window around
468/// it. That is what a content divergence needs to be classifiable, and it costs
469/// no retained bytes: the guest's data never leaves the guest.
470///
471/// ⚠️ ONE GUEST READ, NOT ONE PER CHUNK. Reading guest memory is the expensive
472/// half of this check -- `compute_hash_range` does a `read_values` per call --
473/// so the extent is read ONCE and every digest is taken from that local copy.
474/// The guest-memory cost is therefore identical to the single-digest version
475/// this replaces; only local hashing and log width grow.
476///
477/// The chunk count is CAPPED so a large buffer cannot produce an unbounded log
478/// line: at most [`CHUNK_CAP`] digests always span the whole extent, so the
479/// window is `max(CHUNK_MIN, ceil(len / CHUNK_CAP))`. A buffer that fits in one
480/// chunk emits no chunk list, because a single chunk repeats what the
481/// whole-extent digest already said.
482const CHUNK_CAP: usize = 8;
483const CHUNK_MIN: usize = 256;
484
485fn extent_digests<G, T>(
486    guest: &mut G,
487    addr: u64,
488    len: u64,
489    rng_output: bool,
490) -> Result<(Digest, usize, Vec<String>), Error>
491where
492    G: Guest<T>,
493    T: Tool,
494{
495    let size = len as usize;
496    let mut buf = if rng_output {
497        let mut buf = rng_observation_vec(size, "digest payload")?;
498        buf.resize(size, 0u8);
499        buf
500    } else {
501        vec![0u8; size]
502    };
503    if size > 0 {
504        let start = Addr::<u8>::from_raw(addr as usize).ok_or(Errno::EFAULT)?;
505        guest.memory().read_values(start, buf.as_mut_slice())?;
506    }
507    let whole = Digest::new(buf.as_slice());
508    let (chunk, chunks) = chunk_digests(buf.as_slice());
509    Ok((whole, chunk, chunks))
510}

512/// The locating half, split out from the guest read so it can be bracketed.
513///
514/// Returns the chunk width and one short digest per chunk, or an empty vector
515/// when the extent fits in a single chunk and the list would only repeat the
516/// whole-extent digest.
517fn chunk_digests(buf: &[u8]) -> (usize, Vec<String>) {
518    let size = buf.len();
519    let chunk = std::cmp::max(CHUNK_MIN, size.div_ceil(CHUNK_CAP));
520    if size <= chunk {
521        return (chunk, Vec::new());
522    }
523    let chunks = buf
524        .chunks(chunk)
525        // Short prefix per chunk: this locates a window, it does not
526        // authenticate one, and eight full SHA-256 digests per buffer would
527        // dominate the line without making the location any sharper. Truncated
528        // explicitly rather than with a `{:.8}` precision, which `Digest`'s
529        // Display does not honour -- it delegates to LowerHex and ignores the
530        // formatter's precision, so the specifier silently printed the full
531        // digest.
532        .map(|c| {
533            Digest::new(c)
534                .to_string()
535                .chars()
536                .take(8)
537                .collect::<String>()
538        })
539        .collect();
540    (chunk, chunks)

543/// ⚠️ THE GUARD IS FIRST AND THAT IS THE POINT. Everything below it touches
544/// guest memory: for `recvmsg` the extents cannot even be computed without
545/// reading a `msghdr` and an `iovec` array out of the guest. That is
546/// preparatory work done BEFORE the `detlog!`, which is exactly the shape that
547/// made `--detlog-stack` and `--detlog-heap` cost 4.36x and 4.76x on a boot
548/// with logging off, producing 123 bytes of log. `detlog_observed!()` is
549/// checked before any of it so the disabled path is genuinely inert, which is
550/// the property a default-on check has to have.
551pub(crate) fn detlog_io_buffers<G, T>(
552    guest: &mut G,
553    call: &Syscall,
554    ret: i64,
555    dettid: DetTid,
556    rng_output: Option<&[BufferExtent]>,
557) -> Result<(), Error>
558where
559    G: Guest<T>,
560    T: Tool,
561{
562    if !crate::detlog_observed!() {
563        return Ok(());
564    }
565    let dir = direction(call).as_str();
566    let name = call.name();
567    // NAME THE DESCRIPTOR THE BYTES CAME THROUGH.
568    //
569    // A content divergence is classified by WHAT the differing bytes are, and
570    // the descriptor is the cheapest available answer to that. Without it,
571    // identifying one real case cost reading backward five syscalls through a
572    // 5.8 MB log to find `socket(16, 524291, 0) = Ok(11)` and recognising
573    // AF_NETLINK by hand; with it, "this is a netlink dump" is on the line that
574    // diverged.
575    //
576    // Safe to add to a byte-for-byte compared surface because the value is
577    // ALREADY compared and already stable: the `[syscall]` lines carry these
578    // same fds, and across two verify pairs measured 2026-08-25 every
579    // fd-bearing syscall line matched with zero differing lines. `-` covers the
580    // calls with no single fd argument (getrandom, getcwd, readlink), which is
581    // a fact about the call rather than a missing lookup.
582    let fd = match crate::syscalls::helpers::get_fd(*call) {
583        Some(fd) => fd.to_string(),
584        None => "-".to_string(),
585    };
586    let moved_extents = {
587        let memory = guest.memory();
588        observed_extents(&memory, call, ret, rng_output)?
589    };
590    for extent in moved_extents {
591        let (whole, chunk, chunks) =
592            extent_digests(guest, extent.addr, extent.len, rng_output.is_some()).map_err(
593                |error| {
594                    if rng_output.is_some() {
595                        Error::Tool(anyhow::Error::new(error).context(format!(
596                            "RNG vector observation after output commit: digest read at {:#x}+{}",
597                            extent.addr, extent.len
598                        )))
599                    } else {
600                        error
601                    }
602                },
603            )?;
604        let located = if chunks.is_empty() {
605            String::new()
606        } else {
607            format!(" chunks={}:{}", chunk, chunks.join(","))
608        };
609        crate::detlog!(
610            "[iobuf][dtid {}] {} {} fd={} {:#x}+{}->{}{}",
611            dettid,
612            name,
613            dir,
614            fd,
615            extent.addr,
616            extent.len,
617            whole,
618            located
619        );
620    }
621    Ok(())
622}

1216    /// (https://github.com/rrnewton/hermit/issues/3338) is counted and fully
1217    /// handled, but charges neither the thread's logical clock nor process CPU
1218    /// time. The first guest syscall after the window resumes from exactly the
1219    /// clock value the window started at: it charges one syscall's cost, the
1220    /// same as a run that never had the window.
1221    #[tokio::test(flavor = "current_thread")]
1222    async fn backend_runtime_bootstrap_syscall_is_handled_but_not_charged_to_guest_time() {
1223        struct Observed {
1224            result: i64,
1225            bytes: Vec<u8>,
1226            syscall_count: u64,
1227            random_offset: u64,
1228            clock_before: LogicalTime,
1229            clock_after: LogicalTime,
1230            system_before: LogicalTime,
1231            system_after: LogicalTime,
1232            process_system_before: LogicalTime,
1233            process_system_after: LogicalTime,
1234        }

1236        async fn one_readv(tool: &Detcore, guest: &mut EventGuest) -> Observed {
1237            let memory = guest.memory.clone();
1238            memory.put_iovec(0, FIRST_DEST, 8);
1239            let clock_before = guest.thread.thread_logical_time.as_nanos();
1240            let system_before = guest.thread.thread_logical_time.system_cpu_time();
1241            let process_system_before = guest.thread.process_cpu_time().system;
1242            let result = tool.handle_syscall_event(guest, readv(1)).await.unwrap();
1243            Observed {
1244                result,
1245                bytes: memory.bytes(FIRST_DEST, 8),
1246                syscall_count: guest.thread.stats.syscall_count,
1247                random_offset: guest
1248                    .thread
1249                    .with_detfd(FD, |fd| fd.random_device_offset())
1250                    .unwrap(),
1251                clock_before,
1252                clock_after: guest.thread.thread_logical_time.as_nanos(),
1253                system_before,
1254                system_after: guest.thread.thread_logical_time.system_cpu_time(),
1255                process_system_before,
1256                process_system_after: guest.thread.process_cpu_time().system,
1257            }
1258        }
1259
1260        // Reference: an ordinary guest syscall is charged.
1261        let (tool, mut plain) = event_guest(FdType::Rng, None);
1262        let guest_call = one_readv(&tool, &mut plain).await;
1263        assert!(
1264            guest_call.clock_after > guest_call.clock_before,
1265            "a guest syscall must advance logical time"
1266        );
1267        assert!(guest_call.system_after > guest_call.system_before);
1268        assert!(guest_call.process_system_after > guest_call.process_system_before);
1269
1270        // The same syscall inside the backend's bootstrap window.
1271        let (tool, mut booting) = event_guest(FdType::Rng, None);

1275        // result, same emulated random bytes written, same RNG cursor advance,

1278        assert_eq!(bootstrap_call.bytes, guest_call.bytes);

1286        );

1288        assert_eq!(
1289            bootstrap_call.process_system_after,
1290            bootstrap_call.process_system_before
1291        );
1292        assert_eq!(bootstrap_call.clock_before, guest_call.clock_before);

1295        // value (no reset, no jump) and is charged one syscall's cost, so the
1296        // clock matches the run that never had the window.
1297        booting.backend_runtime_bootstrap = false;
1298        let after_window = one_readv(&tool, &mut booting).await;
1299        assert_eq!(after_window.syscall_count, 2);
1300        assert_eq!(after_window.clock_before, bootstrap_call.clock_after);
1301        assert_eq!(after_window.clock_after, guest_call.clock_after);
1302        assert_eq!(after_window.system_after, guest_call.system_after);
1303        assert_eq!(
1304            after_window.process_system_after,
1305            guest_call.process_system_after
1306        );
1307        assert!(after_window.clock_after > after_window.clock_before);

1310    /// The bootstrapping thread's clock keeps moving inside a backend-runtime
1311    /// bootstrap window (finding F1 of
1312    /// https://github.com/rrnewton/hermit/pull/3430#issuecomment-5928691696).
1313    /// A syscall that observes virtual time is charged exactly as outside the
1314    /// window, so two reads in a row see different times. Other syscalls are
1315    /// left uncharged only up to `MAX_UNCHARGED_BOOTSTRAP_SYSCALLS` per window;
1316    /// the syscall that reaches the cap logs one info line, the next one is
1317    /// charged its normal cost, and a new window starts counting from zero. The
1318    /// harness's clock is syscall-driven (`max_timeslice: None`),
1319    /// the configuration in which nothing else would advance this thread.
1320    #[tokio::test(flavor = "current_thread")]
1321    async fn backend_runtime_bootstrap_window_charges_time_reads_and_caps_uncharged_syscalls() {
1322        use reverie::syscalls::Sysno;
1323
1324        use crate::syscall_time::MAX_UNCHARGED_BOOTSTRAP_SYSCALLS;
1325        use crate::syscall_time::observes_virtual_time;
1326
1327        const USAGE: usize = RETRY_DEST;
1328
1329        // getrusage(RUSAGE_THREAD) is the time read this harness can execute:
1330        // it reports the thread's own CPU time without a global-clock RPC, and
1331        // every charged syscall adds to that CPU time.
1332        fn thread_rusage() -> Syscall {
1333            reverie::syscalls::Getrusage::new()
1334                .with_who(libc::RUSAGE_THREAD)
1335                .with_usage(AddrMut::from_raw(USAGE))
1336                .into()
1337        }
1338
1339        /// Runs one syscall and returns how far it advanced the thread's
1340        /// logical time, in nanoseconds.
1341        async fn advance(tool: &Detcore, guest: &mut EventGuest, call: Syscall) -> u64 {
1342            guest.memory.put_iovec(0, FIRST_DEST, 8);
1343            let before = guest.thread.thread_logical_time.as_nanos().as_nanos();
1344            tool.handle_syscall_event(guest, call).await.unwrap();
1345            guest.thread.thread_logical_time.as_nanos().as_nanos() - before
1346        }
1347
1348        /// The system CPU time the last getrusage wrote, in nanoseconds.
1349        fn reported_system_ns(guest: &EventGuest) -> u64 {
1350            let usage: libc::rusage = guest
1351                .memory
1352                .read_value(Addr::<libc::rusage>::from_raw(USAGE).unwrap())
1353                .unwrap();
1354            usage.ru_stime.tv_sec as u64 * 1_000_000_000 + usage.ru_stime.tv_usec as u64 * 1_000
1355        }
1356
1357        // Every syscall whose main result is a clock or timer value.
1358        const TIME_READS: [Sysno; 11] = [
1359            Sysno::gettimeofday,
1360            Sysno::time,
1361            Sysno::clock_gettime,
1362            Sysno::sysinfo,
1363            Sysno::times,
1364            Sysno::getrusage,
1365            Sysno::timerfd_gettime,
1366            Sysno::timer_gettime,
1367            Sysno::getitimer,
1368            Sysno::adjtimex,
1369            Sysno::clock_adjtime,
1370        ];
1371        for sysno in TIME_READS {
1372            assert!(observes_virtual_time(sysno), "{sysno} reads virtual time");
1373        }
1374        assert!(!observes_virtual_time(Sysno::readv));
1375        assert!(!observes_virtual_time(Sysno::clock_getres));
1376
1377        // Reference costs outside any window.
1378        let (tool, mut plain) = event_guest(FdType::Rng, None);
1379        let time_read_cost = advance(&tool, &mut plain, thread_rusage()).await;
1380        let first_read_outside = reported_system_ns(&plain);
1381        let readv_cost = advance(&tool, &mut plain, readv(1)).await;
1382        assert!(time_read_cost > 0 && readv_cost > 0);
1383
1384        // (1) Time reads inside the window are charged like time reads outside.
1385        let (tool, mut booting) = event_guest(FdType::Rng, None);
1386        booting.backend_runtime_bootstrap = true;
1387        let first_read_cost = advance(&tool, &mut booting, thread_rusage()).await;
1388        let first_read = reported_system_ns(&booting);
1389        let second_read_cost = advance(&tool, &mut booting, thread_rusage()).await;
1390        let second_read = reported_system_ns(&booting);
1391        assert_eq!(
1392            first_read_cost, time_read_cost,
1393            "a time read inside the window must advance logical time as it does outside"
1394        );
1395        assert_eq!(second_read_cost, time_read_cost);
1396        assert_eq!(first_read, first_read_outside);
1397        assert_ne!(
1398            second_read, first_read,
1399            "two time reads inside the window must observe different times"
1400        );
1401        assert_eq!(second_read - first_read, time_read_cost);
1402        // The other time reads are charged inside the window as well, and
1403        // none of them counts toward the cap.
1404        for sysno in TIME_READS {
1405            assert!(
1406                booting.thread.charge_syscall_time(true, sysno),
1407                "{sysno} inside the window must be charged"
1408            );
1409        }
1410        assert_eq!(booting.thread.uncharged_bootstrap_syscalls, 0);
1411
1412        // (2) The first MAX_UNCHARGED_BOOTSTRAP_SYSCALLS other syscalls of the
1413        // window are uncharged; the next one, and every one after it, is charged.
1414        // The window logs one info line, at the syscall that reaches the cap.
1415        let cap_logs = BufferLog(Arc::default(), "reached its cap");
1416        let _subscriber = tracing::subscriber::set_default(cap_logs.clone());
1417        for index in 0..MAX_UNCHARGED_BOOTSTRAP_SYSCALLS {
1418            if index == MAX_UNCHARGED_BOOTSTRAP_SYSCALLS - 1 {
1419                assert!(
1420                    cap_logs.0.lock().unwrap().is_empty(),
1421                    "the cap line was logged before the window reached the cap"
1422                );
1423            }
1424            assert_eq!(
1425                advance(&tool, &mut booting, readv(1)).await,
1426                0,
1427                "uncharged syscall {index} of the window advanced logical time"
1428            );
1429        }
1430        assert_eq!(
1431            booting.thread.uncharged_bootstrap_syscalls,
1432            MAX_UNCHARGED_BOOTSTRAP_SYSCALLS
1433        );
1434        let cap_line = vec![format!(
1435            "[dtid {}] backend runtime bootstrap window reached its cap of {} uncharged syscalls; every further syscall in this window is charged",
1436            booting.thread.dettid, MAX_UNCHARGED_BOOTSTRAP_SYSCALLS
1437        )];
1438        assert_eq!(
1439            *cap_logs.0.lock().unwrap(),
1440            cap_line,
1441            "reaching the cap must log exactly one info line"
1442        );
1443        assert_eq!(
1444            advance(&tool, &mut booting, readv(1)).await,
1445            readv_cost,
1446            "the first syscall past the cap must be charged one syscall cost"
1447        );
1448        assert_eq!(advance(&tool, &mut booting, readv(1)).await, readv_cost);
1449        assert_eq!(
1450            booting.thread.uncharged_bootstrap_syscalls,
1451            MAX_UNCHARGED_BOOTSTRAP_SYSCALLS
1452        );
1453        assert_eq!(
1454            *cap_logs.0.lock().unwrap(),
1455            cap_line,
1456            "syscalls charged past the cap must not log the cap line again"
1457        );
1458
1459        // (3) Closing the window resets the count, so a reopened window (the
1460        // next exec'd image) starts uncharged again.
1461        booting.backend_runtime_bootstrap = false;
1462        assert_eq!(advance(&tool, &mut booting, readv(1)).await, readv_cost);
1463        assert_eq!(booting.thread.uncharged_bootstrap_syscalls, 0);
1464        booting.backend_runtime_bootstrap = true;
1465        assert_eq!(advance(&tool, &mut booting, readv(1)).await, 0);
1466        assert_eq!(booting.thread.uncharged_bootstrap_syscalls, 1);
1467        assert_eq!(*cap_logs.0.lock().unwrap(), cap_line);
1468
1469        // (4) The reopened window has the whole cap again, and reaching it logs
1470        // the window's own info line: the line is once per window, not once per
1471        // thread or per process.
1472        for index in 1..MAX_UNCHARGED_BOOTSTRAP_SYSCALLS {
1473            if index == MAX_UNCHARGED_BOOTSTRAP_SYSCALLS - 1 {
1474                assert_eq!(
1475                    *cap_logs.0.lock().unwrap(),
1476                    cap_line,
1477                    "the reopened window logged the cap line before reaching the cap"
1478                );
1479            }
1480            assert_eq!(
1481                advance(&tool, &mut booting, readv(1)).await,
1482                0,
1483                "uncharged syscall {index} of the reopened window advanced logical time"
1484            );
1485        }
1486        assert_eq!(
1487            booting.thread.uncharged_bootstrap_syscalls,
1488            MAX_UNCHARGED_BOOTSTRAP_SYSCALLS
1489        );
1490        let two_cap_lines = vec![cap_line[0].clone(), cap_line[0].clone()];
1491        assert_eq!(
1492            *cap_logs.0.lock().unwrap(),
1493            two_cap_lines,
1494            "a reopened window that reaches the cap must log its own info line"
1495        );
1496        assert_eq!(
1497            advance(&tool, &mut booting, readv(1)).await,
1498            readv_cost,
1499            "the first syscall past the reopened window's cap must be charged"
1500        );
1501        assert_eq!(*cap_logs.0.lock().unwrap(), two_cap_lines);
1502    }
