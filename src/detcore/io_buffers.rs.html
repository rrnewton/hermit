9//! Hash the bytes a syscall moved through a guest buffer, at the syscall
10//! boundary.
11//!
12//! WHAT GAP THIS FILLS. `--detlog-stack` and `--detlog-heap` hash a whole named
13//! mapping, so what they can see is decided by where the guest happened to
14//! ALLOCATE a buffer rather than by what the syscall did. Measured 2026-08-20,
15//! three runs per cell, running the same netlink `RTM_GETLINK` exchange and
16//! changing only the receive buffer's home:
17//!
18//! | buffer home     | `--detlog-stack` | `--detlog-heap` | both   |
19//! |-----------------|------------------|-----------------|--------|
20//! | `[stack]`       | CAUGHT           | MISSED          | CAUGHT |
21//! | `[heap]` (brk)  | MISSED           | CAUGHT          | CAUGHT |
22//! | BSS / static    | MISSED           | MISSED          | MISSED |
23//! | anonymous mmap  | MISSED           | MISSED          | MISSED |
24//!
25//! Two of the four are invisible even with both flags on, and anonymous mmap is
26//! not a corner case: it is where glibc puts any `malloc` above the 128 KiB
27//! `M_MMAP_THRESHOLD`. Taking the address and length from the SYSCALL ARGUMENTS
28//! instead makes all four rows irrelevant by construction.
29//!
30//! WHAT IT CATCHES THAT `--verify` CANNOT. `--verify` compares the INFO record,
31//! and a syscall whose output buffer is typed as a bare pointer in Reverie
32//! prints the address, not the contents (`reverie-syscalls/src/syscalls.rs`
33//! carries standing TODOs saying exactly this for `Read` and `Write`). So a
34//! `recvmsg` that returns a stable `Ok(1468)` while four bytes of its payload
35//! differ produces a character-identical record and `--verify` reports
36//! `bitwise_parity: true`. Measured on one QEMU/Linux boot, 278,824 of 632,228
37//! syscalls (44.1%) move bytes through a buffer whose content the log never
38//! shows.
39//!
40//! COST SHAPE, and it differs from the whole-mapping flags in kind rather than
41//! degree. `--detlog-heap` is `O(syscalls x region_size)` -- it re-reads the
42//! entire heap after every syscall, hashing 10.9 TB per boot to watch a 16.49
43//! MiB heap. This is `O(bytes the syscalls actually returned)`: 139.1 MB per
44//! boot, measured by summing real return values.

59/// One contiguous run of guest bytes a completed syscall moved.
60#[derive(Debug, Clone, Copy, PartialEq, Eq)]
61pub struct BufferExtent {
62    /// Guest virtual address of the first byte.
63    pub addr: u64,
64    /// Number of bytes actually moved, which is bounded by the syscall's return
65    /// value and not by the buffer's declared capacity.
66    pub len: u64,
67}
68
69/// Direction of travel, recorded so a reader can tell a value the kernel
70/// produced from one the guest produced without knowing every syscall.
71#[derive(Debug, Clone, Copy, PartialEq, Eq)]
72pub enum Direction {
73    /// The kernel filled the buffer: an unvirtualized field here is a leak INTO
74    /// the guest, which is the netlink case.
75    In,
76    /// The guest filled the buffer: a difference here means the guest computed
77    /// different bytes, which is how divergence reaches a file or a device.
78    Out,
79}
80
81impl Direction {
82    fn as_str(self) -> &'static str {
83        match self {
84            Direction::In => "in",
85            Direction::Out => "out",
86        }
87    }
88}

90/// Clamp a declared buffer to what the syscall says it actually moved.
91///
92/// `capacity` is the argument the guest passed; `moved` is the return value.
93/// Hashing `capacity` would fold in bytes the syscall never wrote, which are
94/// whatever was in the buffer beforehand -- for a stack buffer that is previous
95/// frames, so the hash would report divergence for an unrelated reason.
96fn clamp(addr: Option<u64>, capacity: usize, moved: i64) -> Vec<BufferExtent> {
97    let Some(addr) = addr else { return Vec::new() };
98    let moved = u64::try_from(moved).unwrap_or(0);
99    let len = moved.min(capacity as u64);
100    if len == 0 {
101        return Vec::new();
102    }
103    vec![BufferExtent { addr, len }]
104}

106/// A buffer written wholly in place, whose size does not depend on the return
107/// value (`poll`'s `revents` are rewritten across the whole array regardless of
108/// how many descriptors were ready).
109fn whole(addr: Option<u64>, len: u64) -> Vec<BufferExtent> {
110    match addr {
111        Some(addr) if len > 0 => vec![BufferExtent { addr, len }],
112        _ => Vec::new(),
113    }
114}

116/// Walk an `iovec` array and return the segments the syscall actually filled,
117/// bounded by `moved`.
118///
119/// Mirrors the existing traversal in `crate::syscalls::io`: clamp the count to
120/// `UIO_MAXIOV`, skip null/empty segments, and stop once `moved` bytes are
121/// accounted for. Under `MSG_TRUNC` the returned count can exceed the buffers'
122/// capacity, which is why the running remainder rather than `moved` alone
123/// bounds each segment.
124fn iovec_extents<G, T>(
125    guest: &mut G,
126    iov_addr: usize,
127    iov_count: usize,
128    moved: i64,
129) -> Result<Vec<BufferExtent>, Error>
130where
131    G: Guest<T>,
132    T: Tool,
133{
134    let mut remaining = u64::try_from(moved).unwrap_or(0);
135    if iov_addr == 0 || iov_count == 0 || remaining == 0 {
136        return Ok(Vec::new());
137    }
138    let iov_count = iov_count.min(libc::UIO_MAXIOV as usize);
139    let iov_address: AddrMut<'_, libc::iovec> = AddrMut::from_raw(iov_addr).ok_or(Errno::EFAULT)?;
140    // SAFETY: `iovec` is a plain C record; an all-zero value is a valid staging
141    // value that `read_values` immediately overwrites.
142    let mut iovecs: Vec<libc::iovec> = (0..iov_count)
143        .map(|_| unsafe { std::mem::zeroed() })
144        .collect();
145    guest
146        .memory()
147        .read_values(iov_address.into(), &mut iovecs)?;
148
149    let mut out = Vec::new();
150    for iov in &iovecs {
151        if remaining == 0 {
152            break;
153        }
154        if iov.iov_base.is_null() || iov.iov_len == 0 {
155            continue;
156        }
157        let take = (iov.iov_len as u64).min(remaining);
158        out.push(BufferExtent {
159            addr: iov.iov_base as u64,
160            len: take,
161        });
162        remaining -= take;
163    }
164    Ok(out)
165}

167/// Read a `msghdr` out of the guest and walk the iovecs it points at.
168fn msghdr_extents<G, T>(
169    guest: &mut G,
170    msg_addr: usize,
171    moved: i64,
172) -> Result<Vec<BufferExtent>, Error>
173where
174    G: Guest<T>,
175    T: Tool,
176{
177    if msg_addr == 0 {
178        return Ok(Vec::new());
179    }
180    let address: AddrMut<'_, libc::msghdr> = AddrMut::from_raw(msg_addr).ok_or(Errno::EFAULT)?;
181    let message: libc::msghdr = guest.memory().read_value(address)?;
182    iovec_extents(guest, message.msg_iov as usize, message.msg_iovlen, moved)
183}

185/// Walk the `mmsghdr` array a batch receive filled.
186///
187/// `moved` here is a COUNT OF MESSAGES, not a byte count -- which is why
188/// `recvmmsg` cannot share the `clamp`/`msghdr_extents` path that every other
189/// receive uses. Each delivered message carries its own byte count in
190/// `msg_len`, so each is walked separately and bounded by that; treating the
191/// batch as one buffer would let one message's length run into the next
192/// message's memory.
193fn mmsghdr_extents<G, T>(
194    guest: &mut G,
195    mmsg_addr: usize,
196    vlen: u32,
197    delivered: i64,
198) -> Result<Vec<BufferExtent>, Error>
199where
200    G: Guest<T>,
201    T: Tool,
202{
203    let delivered = usize::try_from(delivered).unwrap_or(0);
204    let count = delivered.min(vlen as usize).min(libc::UIO_MAXIOV as usize);
205    if mmsg_addr == 0 || count == 0 {
206        return Ok(Vec::new());
207    }
208    let address: AddrMut<'_, libc::mmsghdr> = AddrMut::from_raw(mmsg_addr).ok_or(Errno::EFAULT)?;
209    // SAFETY: `mmsghdr` is a plain C record; an all-zero value is a valid
210    // staging value that `read_values` immediately overwrites.
211    let mut headers: Vec<libc::mmsghdr> =
212        (0..count).map(|_| unsafe { std::mem::zeroed() }).collect();
213    guest.memory().read_values(address.into(), &mut headers)?;
214
215    let mut out = Vec::new();
216    for header in &headers {
217        out.extend(iovec_extents(
218            guest,
219            header.msg_hdr.msg_iov as usize,
220            header.msg_hdr.msg_iovlen,
221            i64::from(header.msg_len),
222        )?);
223    }
224    Ok(out)
225}

227/// Whether this syscall's return value bounds what it wrote.
228///
229/// True for the ordinary case, where the return IS the byte count, so a return
230/// of 0 or less means nothing was written. FALSE for `poll`/`ppoll`, whose
231/// return is a COUNT OF READY DESCRIPTORS: the kernel rewrites `revents` on
232/// every entry even when it returns 0. Measured -- `poll(timeout=0)` on two
233/// never-ready pipe fds returns 0 and still moves `revents` from a poisoned
234/// 0x7FFF to 0 on both entries.
235///
236/// This exists as a predicate consulted BY the single `ret <= 0` guard, rather
237/// than as poll arms placed above it, so the guard cannot be reordered back
238/// into swallowing the zero-ready case. The original gap was exactly that: the
239/// guard sat above the poll arms, and the unit test called `whole` directly, so
240/// it could not see the short-circuit above the code it exercised.
241/// Every syscall `extents` below returns a non-empty result for.
242///
243/// KEEP THIS IN STEP WITH THE `extents` MATCH -- it is the same set written
244/// twice, once as `Syscall::` arms (which need constructed calls and a `Guest`
245/// to exercise) and once as bare `Sysno`s (which a subscription can be asked
246/// about). The second spelling exists so the relationship between this check
247/// and what Detcore actually intercepts can be ASSERTED rather than assumed;
248/// see `passthru_opt_leaves_io_buffer_hashing_blind_only_for_getcwd` in
249/// `lib.rs`. Adding an arm to `extents` without adding it here does not break
250/// that test, so add both.
251#[cfg(test)]

278fn ret_gates_output(call: &Syscall) -> bool {
279    !matches!(call, Syscall::Poll(_) | Syscall::Ppoll(_))

282/// The extents a completed syscall moved, or `None` when this syscall has no
283/// output buffer worth hashing.
284///
285/// Only syscalls whose buffer CONTENT the INFO record does not already show are
286/// listed. `clock_gettime` and `newfstatat`, for instance, are absent because
287/// Reverie's typed display already dereferences and prints their output.
288fn extents<G, T>(guest: &mut G, call: &Syscall, ret: i64) -> Result<Vec<BufferExtent>, Error>
289where
290    G: Guest<T>,
291    T: Tool,
292{
293    // Nothing was written on a failed or empty call -- for every syscall whose
294    // return value is a byte count. `ret_gates_output` is what keeps the poll
295    // family out of this, and it is a predicate rather than an arm placed above
296    // so the exclusion cannot be undone by moving code.
297    if ret <= 0 && ret_gates_output(call) {
298        return Ok(Vec::new());
299    }
300    let raw = |a: Option<AddrMut<'_, u8>>| a.map(|p| p.as_raw() as u64);
301    Ok(match call {
302        // Bytes the kernel produced.
303        Syscall::Read(c) => clamp(raw(c.buf()), c.len(), ret),
304        Syscall::Pread64(c) => clamp(raw(c.buf()), c.len(), ret),
305        Syscall::Recvfrom(c) => clamp(c.buf().map(|p| p.as_raw() as u64), c.len(), ret),
306        Syscall::Getrandom(c) => clamp(raw(c.buf()), c.buflen(), ret),
307        Syscall::Getcwd(c) => clamp(c.buf().map(|p| p.as_raw() as u64), c.size(), ret),
308        Syscall::Getdents64(c) => clamp(
309            c.dirent().map(|p| p.as_raw() as u64),
310            c.count() as usize,
311            ret,
312        ),
313        Syscall::Readlink(c) => clamp(c.buf().map(|p| p.as_raw() as u64), c.bufsize(), ret),
314        Syscall::Readlinkat(c) => clamp(c.buf().map(|p| p.as_raw() as u64), c.buf_len(), ret),
315        Syscall::Recvmsg(c) => msghdr_extents(guest, c.msg().map_or(0, |p| p.as_raw()), ret)?,
316        // `ret` is a MESSAGE count here, not a byte count; see
317        // `mmsghdr_extents`. recvmmsg is one of the four receive syscalls
318        // that could reach a NETLINK_SOCK_DIAG dump without passing the
319        // sock_diag sanitizer, so leaving it unhashed left this check blind
320        // to exactly the bypass it would otherwise have reported.
321        Syscall::Recvmmsg(c) => {
322            mmsghdr_extents(guest, c.mmsg().map_or(0, |p| p.as_raw()), c.vlen(), ret)?
323        }
324        Syscall::Readv(c) => iovec_extents(guest, c.iov().map_or(0, |p| p.as_raw()), c.len(), ret)?,
325        Syscall::Preadv(c) => {
326            iovec_extents(guest, c.iov().map_or(0, |p| p.as_raw()), c.iov_len(), ret)?
327        }
328
329        // Bytes the guest produced. These never reach stdout/stderr for a QEMU
330        // boot -- measured, all 234,872 writes went to fds 7/12/14/11/13/4/8/19/23
331        // and none to fd 1 or 2 -- so `--verify`'s stdout/stderr comparison does
332        // not cover them either.
333        Syscall::Write(c) => clamp(c.buf().map(|p| p.as_raw() as u64), c.len(), ret),
334        Syscall::Pwrite64(c) => clamp(c.buf().map(|p| p.as_raw() as u64), c.len(), ret),
335        Syscall::Sendto(c) => clamp(c.buf().map(|p| p.as_raw() as u64), c.size(), ret),
336        Syscall::Sendmsg(c) => msghdr_extents(guest, c.msg().map_or(0, |p| p.as_raw()), ret)?,
337        Syscall::Writev(c) => {
338            iovec_extents(guest, c.iov().map_or(0, |p| p.as_raw()), c.len(), ret)?
339        }
340        Syscall::Pwritev(c) => {
341            iovec_extents(guest, c.iov().map_or(0, |p| p.as_raw()), c.iov_len(), ret)?
342        }
343
344        // Rewritten in place across the WHOLE array: `poll` sets `revents` on
345        // every entry, not just on the `ret` that were ready, so the extent is
346        // the array and not a prefix of it -- and it is reached even when
347        // `ret == 0`, via `ret_gates_output`.
348        Syscall::Poll(c) => whole(
349            c.fds().map(|p| p.as_raw() as u64),
350            c.nfds() * std::mem::size_of::<libc::pollfd>() as u64,
351        ),
352        Syscall::Ppoll(c) => whole(
353            c.fds().map(|p| p.as_raw() as u64),
354            c.nfds() * std::mem::size_of::<libc::pollfd>() as u64,
355        ),
356
357        _ => Vec::new(),
358    })
359}

361/// Which way the bytes travelled, for the record's label.
362fn direction(call: &Syscall) -> Direction {
363    match call {
364        Syscall::Write(_)
365        | Syscall::Pwrite64(_)
366        | Syscall::Sendto(_)
367        | Syscall::Sendmsg(_)
368        | Syscall::Writev(_)
369        | Syscall::Pwritev(_) => Direction::Out,
370        _ => Direction::In,
371    }
372}

374/// Emit one deterministic record per buffer a completed syscall moved.
375///
376/// Digests that LOCATE a divergence instead of only detecting one.
377///
378/// The whole-extent digest says two buffers differ and nothing else -- not
379/// which bytes, not which field. Comparing two logs of per-chunk digests names
380/// the FIRST DIFFERING CHUNK, which is the offset and a bounded window around
381/// it. That is what a content divergence needs to be classifiable, and it costs
382/// no retained bytes: the guest's data never leaves the guest.
383///
384/// ⚠️ ONE GUEST READ, NOT ONE PER CHUNK. Reading guest memory is the expensive
385/// half of this check -- `compute_hash_range` does a `read_values` per call --
386/// so the extent is read ONCE and every digest is taken from that local copy.
387/// The guest-memory cost is therefore identical to the single-digest version
388/// this replaces; only local hashing and log width grow.
389///
390/// The chunk count is CAPPED so a large buffer cannot produce an unbounded log
391/// line: at most [`CHUNK_CAP`] digests always span the whole extent, so the
392/// window is `max(CHUNK_MIN, ceil(len / CHUNK_CAP))`. A buffer that fits in one
393/// chunk emits no chunk list, because a single chunk repeats what the
394/// whole-extent digest already said.
395const CHUNK_CAP: usize = 8;
396const CHUNK_MIN: usize = 256;
397
398fn extent_digests<G, T>(
399    guest: &mut G,
400    addr: u64,
401    len: u64,
402) -> Result<(Digest, usize, Vec<String>), Error>
403where
404    G: Guest<T>,
405    T: Tool,
406{
407    let size = len as usize;
408    let mut buf = vec![0u8; size];
409    if size > 0 {
410        let start = Addr::<u8>::from_raw(addr as usize).ok_or(Errno::EFAULT)?;
411        guest.memory().read_values(start, buf.as_mut_slice())?;
412    }
413    let whole = Digest::new(buf.as_slice());
414    let (chunk, chunks) = chunk_digests(buf.as_slice());
415    Ok((whole, chunk, chunks))
416}
417
418/// The locating half, split out from the guest read so it can be bracketed.
419///
420/// Returns the chunk width and one short digest per chunk, or an empty vector
421/// when the extent fits in a single chunk and the list would only repeat the
422/// whole-extent digest.
423fn chunk_digests(buf: &[u8]) -> (usize, Vec<String>) {
424    let size = buf.len();
425    let chunk = std::cmp::max(CHUNK_MIN, size.div_ceil(CHUNK_CAP));
426    if size <= chunk {
427        return (chunk, Vec::new());
428    }
429    let chunks = buf
430        .chunks(chunk)
431        // Short prefix per chunk: this locates a window, it does not
432        // authenticate one, and eight full SHA-256 digests per buffer would
433        // dominate the line without making the location any sharper. Truncated
434        // explicitly rather than with a `{:.8}` precision, which `Digest`'s
435        // Display does not honour -- it delegates to LowerHex and ignores the
436        // formatter's precision, so the specifier silently printed the full
437        // digest.
438        .map(|c| {
439            Digest::new(c)
440                .to_string()
441                .chars()
442                .take(8)
443                .collect::<String>()
444        })
445        .collect();
446    (chunk, chunks)
447}

449/// ⚠️ THE GUARD IS FIRST AND THAT IS THE POINT. Everything below it touches
450/// guest memory: for `recvmsg` the extents cannot even be computed without
451/// reading a `msghdr` and an `iovec` array out of the guest. That is
452/// preparatory work done BEFORE the `detlog!`, which is exactly the shape that
453/// made `--detlog-stack` and `--detlog-heap` cost 4.36x and 4.76x on a boot
454/// with logging off, producing 123 bytes of log. `detlog_observed!()` is
455/// checked before any of it so the disabled path is genuinely inert, which is
456/// the property a default-on check has to have.
457pub(crate) fn detlog_io_buffers<G, T>(
458    guest: &mut G,
459    call: &Syscall,
460    ret: i64,
461    dettid: DetTid,
462) -> Result<(), Error>
463where
464    G: Guest<T>,
465    T: Tool,
466{
467    if !crate::detlog_observed!() {
468        return Ok(());
469    }
470    let dir = direction(call).as_str();
471    let name = call.name();
472    // NAME THE DESCRIPTOR THE BYTES CAME THROUGH.
473    //
474    // A content divergence is classified by WHAT the differing bytes are, and
475    // the descriptor is the cheapest available answer to that. Without it,
476    // identifying one real case cost reading backward five syscalls through a
477    // 5.8 MB log to find `socket(16, 524291, 0) = Ok(11)` and recognising
478    // AF_NETLINK by hand; with it, "this is a netlink dump" is on the line that
479    // diverged.
480    //
481    // Safe to add to a byte-for-byte compared surface because the value is
482    // ALREADY compared and already stable: the `[syscall]` lines carry these
483    // same fds, and across two verify pairs measured 2026-08-25 every
484    // fd-bearing syscall line matched with zero differing lines. `-` covers the
485    // calls with no single fd argument (getrandom, getcwd, readlink), which is
486    // a fact about the call rather than a missing lookup.
487    let fd = match crate::syscalls::helpers::get_fd(*call) {
488        Some(fd) => fd.to_string(),
489        None => "-".to_string(),
490    };
491    for extent in extents(guest, call, ret)? {
492        let (whole, chunk, chunks) = extent_digests(guest, extent.addr, extent.len)?;
493        let located = if chunks.is_empty() {
494            String::new()
495        } else {
496            format!(" chunks={}:{}", chunk, chunks.join(","))
497        };
498        crate::detlog!(
499            "[iobuf][dtid {}] {} {} fd={} {:#x}+{}->{}{}",
500            dettid,
501            name,
502            dir,
503            fd,
504            extent.addr,
505            extent.len,
506            whole,
507            located
508        );
509    }
510    Ok(())
511}

513#[cfg(test)]
514mod tests {
515    use reverie::syscalls;
516
517    /// A divergence must be LOCATED, not merely detected -- that is the whole
518    /// reason the chunk list exists, so it is asserted rather than assumed.
519    #[test]
520    fn chunk_digests_name_the_first_differing_window() {
521        let len = 1468; // the real netlink dump size
522        let left = vec![0xABu8; len];
523        let mut right = left.clone();
524        let offset = 900;
525        right[offset] ^= 0xFF; // one flipped byte, nothing else
526        let (chunk, l) = chunk_digests(&left);
527        let (_, r) = chunk_digests(&right);
528        assert_eq!(chunk, 256);
529        assert_eq!(l.len(), r.len());
530        let first_diff = l.iter().zip(&r).position(|(a, b)| a != b);
531        assert_eq!(
532            first_diff,
533            Some(offset / chunk),
534            "must name the window holding the flip"
535        );
536        // and ONLY that window: a single-byte change must not smear.
537        assert_eq!(l.iter().zip(&r).filter(|(a, b)| a != b).count(), 1);
538    }
539
540    #[test]
541    fn identical_buffers_produce_identical_chunk_lists() {
542        let buf = vec![7u8; 1468];
543        assert_eq!(chunk_digests(&buf).1, chunk_digests(&buf.clone()).1);

622    /// The gap this file had: `poll(...) = 0` produced no record at all,
623    /// because the `ret <= 0` guard sat above the poll arms. The pre-existing
624    /// test called `whole()` directly, so it passed throughout -- it could not
625    /// see a short-circuit above the code it called.
626    ///
627    /// This drives `ret_gates_output`, which is the predicate the single guard
628    /// consults, so the property is tested where it is decided rather than one
629    /// level away from it.
630    #[test]
631    fn the_poll_family_is_exempt_from_the_return_value_guard() {
632        let nfds = 3;
633        for call in [
634            Syscall::Poll(
635                syscalls::Poll::new()
636                    .with_fds(AddrMut::from_raw(0x2000))
637                    .with_nfds(nfds),
638            ),
639            Syscall::Ppoll(
640                syscalls::Ppoll::new()
641                    .with_fds(AddrMut::from_raw(0x2000))
642                    .with_nfds(nfds),
643            ),
644        ] {
645            assert!(
646                !ret_gates_output(&call),
647                "{} returns a count of READY DESCRIPTORS, not a byte count, so a \
648                 zero return must not suppress its extent",
649                call.name()
650            );
651        }
652
653        // Everything else must stay gated, or a failed or empty call would hash
654        // bytes the kernel never wrote.
655        for call in [
656            Syscall::Read(syscalls::Read::new()),
657            Syscall::Recvmsg(syscalls::Recvmsg::new()),
658            Syscall::Recvmmsg(syscalls::Recvmmsg::new()),
659        ] {
660            assert!(
661                ret_gates_output(&call),
662                "{} returns a byte or message count, so a zero return means \
663                 nothing was written",
664                call.name()
665            );
666        }
667    }
