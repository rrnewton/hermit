9//! Writes a run's host-input records to its `host_input_log` when the run
10//! ends (see `detcore_model::host_input`), sums them up as the run's
11//! [`HostInputSet`] for its summary and its preemption record, and compares
12//! the opens of a schedule replay against the recorded run's set
13//! ([`ReplayHostInputs`]).
14//!
15//! The records are an observation for the verification layer, not Detcore
16//! state: Detcore's openat handler reports each one to the global state
17//! (`tool_global::record_host_input`), which keeps them for the run, and
18//! nothing reads them back, no scheduling decision consults them, and nothing
19//! the guest sees depends on them. They are written once, after the run, so
20//! the log stays empty while any guest runs; a run that does not end normally
21//! writes nothing, which leaves its divergence unexplained rather than
22//! explained by partial evidence.
23//!
24//! Nothing here opens or reads a guest's file while any guest process runs.
25//! Linux raises an inotify or fanotify event for every open, read and close
26//! of a file, whoever makes it (`fsnotify_open`, `fsnotify_access` and
27//! `fsnotify_close`, fs/namei.c, fs/read_write.c and fs/file_table.c), so a
28//! guest watching the file's directory would see Hermit's reads among its
29//! own. While the guest runs, Detcore takes only what raises no event: a
30//! file's status, and the kernel's path for the guest's descriptor (a
31//! `readlink` of `/proc/<tid>/fd/<n>`). The bytes, and a directory's names,
32//! are read when the run has ended ([`content_at_end`]), and only where the
33//! file still has the identity the guest's open found.
34

57/// Write `records` to `log`, one JSON line each, then one
58/// [`HostMutationRecord`] line for each path in `mutated`, followed by a
59/// [`HostInputLogEnd`] line naming how many lines precede it, so that a
60/// reader can tell a complete log from a cut one.
61pub(crate) fn write(log: &Path, records: &[HostInputRecord], mutated: &BTreeSet<String>) {
62    let mut text = Vec::new();
63    let end = HostInputLogEnd {
64        records: (records.len() + mutated.len()) as u64,
65    };
66    let encoded = records
67        .iter()
68        .map(serde_json::to_vec)
69        .chain(mutated.iter().map(|path| {
70            serde_json::to_vec(&HostMutationRecord {
71                mutated: path.clone(),
72            })
73        }))
74        .chain(std::iter::once(serde_json::to_vec(&end)))
75        .try_for_each(|line| {
76            text.extend(line?);
77            text.push(b'\n');
78            Ok::<(), serde_json::Error>(())
79        });
80    let written = encoded
81        .map_err(std::io::Error::other)
82        .and_then(|()| std::fs::write(log, &text));
83    if let Err(error) = written {
84        warn!(
85            "[detcore] cannot write the host-input log {}: {}",
86            log.display(),
87            error
88        );
89    }
90}

92/// The run's [`HostInputSet`]: for each path the guest opened, in path order,
93/// the identity and bytes its first open found, leaving out kernel objects
94/// and paths the guest rebound itself (`mutated`, or a relative path), and
95/// the set's digest (`HostInputSet`).
96pub(crate) fn input_set(records: &[HostInputRecord], mutated: &BTreeSet<String>) -> HostInputSet {
97    let mut first: BTreeMap<&str, &HostInputRecord> = BTreeMap::new();
98    for record in records {
99        first.entry(record.path.as_str()).or_insert(record);
100    }
101    let files: Vec<HostInput> = first
102        .into_iter()
103        .filter(|(path, record)| {
104            !record.kernel_object && !is_kernel_object(path) && !guest_rebound(mutated, path)
105        })
106        .map(|(path, record)| HostInput {
107            path: path.to_owned(),
108            identity: record.identity,
109            content: record.content.clone(),
110        })
111        .collect();
112    // One line per file: its path (which holds no NUL), then its bytes'
113    // digest where hashed, else its identity.
114    let mut hasher = Sha256::new();
115    for file in &files {
116        hasher.update(file.path.as_bytes());
117        hasher.update(b"\0");
118        match &file.content {
119            Some(content) => {
120                hasher.update(b"content ");
121                hasher.update(content.as_bytes());
122            }
123            None => {
124                hasher.update(b"identity ");
125                hasher.update(serde_json::to_vec(&file.identity).expect("an identity serializes"));
126            }
127        }
128        hasher.update(b"\n");
129    }
130    HostInputSet {
131        digest: hex(&hasher.finalize()),
132        files,
133    }
134}

136fn hex(bytes: &[u8]) -> String {
137    bytes.iter().map(|byte| format!("{byte:02x}")).collect()
138}

140/// What a run's end reads for the first open of `path` (the guest's path)
141/// that found `identity`, through `source` (the kernel's path for the
142/// guest's descriptor, or for an exec's image, at that open): the SHA-256 of
143/// a regular file's bytes, when it still has that identity and holds at most
144/// [`HOST_INPUT_CONTENT_LIMIT`] bytes, or of the names in a directory that is
145/// still the same directory (device and inode; its size and times change
146/// with every name made in it) as [`directory_digest`] lists them; `None`
147/// otherwise, and for any other kind of file, which is never opened. Called
148/// only when every guest process has exited (see the module documentation).
149/// Raw syscalls only, as for [`read_digest`].
150pub(crate) fn content_at_end(
151    source: &Path,
152    path: &str,
153    identity: HostFileIdentity,
154    mutated: &BTreeSet<String>,
155) -> Option<String> {
156    let stat = crate::util::raw_stat(source).ok()?;
157    match stat.st_mode & libc::S_IFMT {
158        libc::S_IFREG => {
159            if HostFileIdentity::from_stat(&stat) != identity
160                || u64::try_from(identity.size).map_or(true, |size| size > HOST_INPUT_CONTENT_LIMIT)
161            {
162                return None;
163            }
164            let file = crate::util::RawFd::open(source, libc::O_RDONLY | libc::O_NONBLOCK).ok()?;
165            // The file opened is the one whose status was compared.
166            (HostFileIdentity::from_stat(&file.stat().ok()?) == identity)
167                .then(|| read_digest(&file))
168                .flatten()
169        }
170        libc::S_IFDIR => {
171            if (stat.st_dev, stat.st_ino) != (identity.dev, identity.ino) {
172                return None;
173            }
174            let directory =
175                crate::util::RawFd::open(source, libc::O_RDONLY | libc::O_DIRECTORY).ok()?;
176            let opened = directory.stat().ok()?;
177            ((opened.st_dev, opened.st_ino) == (identity.dev, identity.ino))
178                .then(|| directory_digest(&directory, path, mutated))
179                .flatten()
180        }
181        _ => None,
182    }
183}

185/// The SHA-256 of the file open at `file`, read to its end but no further
186/// than [`HOST_INPUT_CONTENT_LIMIT`] bytes, or `None` when it cannot be read
187/// or is longer than that. Raw syscalls only: in-guest Detcore must not call
188/// libc's wrappers, which the guest may define (`crate::util::raw_syscall`).
189fn read_digest(file: &crate::util::RawFd) -> Option<String> {
190    let mut hasher = Sha256::new();
191    let mut buffer = vec![0_u8; 1 << 16];
192    let mut read = 0_u64;
193    loop {
194        let count = file.read_at(&mut buffer, read).ok()?;
195        if count == 0 {
196            break;
197        }
198        read += count as u64;
199        if read > HOST_INPUT_CONTENT_LIMIT {
200            return None;
201        }
202        hasher.update(&buffer[..count]);
203    }
204    Some(hex(&hasher.finalize()))
205}

207/// The SHA-256 of the names in the directory open at `directory`, which the
208/// guest names `path`: sorted, each followed by a NUL, after `directory\0`,
209/// without `.` and `..` and without each name whose path the guest itself
210/// rebound (`mutated`, [`guest_rebound`]), so that what the guest made or
211/// removed in it (in its private `/tmp`, say) does not count. `None` when
212/// the names are longer in all than [`HOST_INPUT_CONTENT_LIMIT`] bytes or
213/// cannot be read. A directory's identity is not compared: Hermit makes some
214/// afresh for each run (its private `/tmp`), and the guest sees only the
215/// names in one, not its host inode or times.
216fn directory_digest(
217    directory: &crate::util::RawFd,
218    path: &str,
219    mutated: &BTreeSet<String>,
220) -> Option<String> {
221    let mut names = Vec::new();
222    let mut total = 0_u64;
223    let too_long = crate::util::find_in_open_directory(directory, |name| {
224        total += name.len() as u64 + 1;
225        if total > HOST_INPUT_CONTENT_LIMIT {
226            return Some(());
227        }
228        let child = Path::new(path).join(name);
229        if !guest_rebound(mutated, &child.to_string_lossy()) {
230            names.push(name.as_bytes().to_vec());
231        }
232        None
233    })
234    .ok()?;
235    if too_long.is_some() {
236        return None;
237    }
238    names.sort();
239    let mut hasher = Sha256::new();
240    hasher.update(b"directory\0");
241    for name in &names {
242        hasher.update(name);
243        hasher.update(b"\0");
244    }
245    Some(hex(&hasher.finalize()))
246}

248/// A file mapped in a process, as a `/proc/<pid>/maps` line shows it: its
249/// name, the inode number of the object mapped, and the mapping's address
250/// range as the line spells it, which names the mapping's
251/// `/proc/<pid>/map_files` link.
252#[derive(Debug, PartialEq)]
253struct MappedFile {
254    path: PathBuf,
255    ino: u64,
256    range: String,
257}
258
259impl MappedFile {
260    /// Whether `stat`, taken through [`Self::path`], is of the object mapped.
261    ///
262    /// Where the tool may follow the mapping's `map_files` link (`mapped`, a
263    /// stat through it; Linux allows that only to a holder of
264    /// CAP_CHECKPOINT_RESTORE or CAP_SYS_ADMIN in the initial user
265    /// namespace), the two have the same device and inode, which keeps the
266    /// filesystem and, on btrfs, the subvolume.
267    ///
268    /// Otherwise the inode number is all that names the object mapped: a
269    /// maps line shows its superblock's device (`show_map_vma`), while btrfs
270    /// reports a subvolume's own device in a stat (`btrfs_getattr`), so the
271    /// devices differ for the same file. The mapping holds the mapped inode,
272    /// so a file that replaced it at the path has another number on that
273    /// filesystem; what the number cannot rule out is another filesystem or
274    /// subvolume mounted over the path with an inode of the same number. On
275    /// overlayfs a maps line names the underlying file, whose number a stat
276    /// through the overlay need not report, so the object may not be
277    /// identified there.
278    fn identifies(&self, stat: &libc::stat, mapped: Option<&libc::stat>) -> bool {
279        match mapped {
280            Some(mapped) => (mapped.st_dev, mapped.st_ino) == (stat.st_dev, stat.st_ino),
281            None => stat.st_ino == self.ino,
282        }
283    }
284}

286/// The file mapped at `address` in a process whose `/proc/<pid>/maps` is
287/// `maps`: the mapping's pathname field, when it names a file (absolute, and
288/// not deleted since it was mapped), and its inode field. Linux shows the
289/// name from the reader's root, and the inode of the file mapped
290/// (`show_map_vma`, fs/proc/task_mmu.c).
291fn mapped_file(maps: &[u8], address: u64) -> Option<MappedFile> {
292    maps.split(|byte| *byte == b'\n').find_map(|line| {
293        let line = std::str::from_utf8(line).ok()?;
294        // address perms offset dev inode pathname: the pathname is the rest of
295        // the line after the fifth field, and may hold spaces.
296        let mut fields = line.splitn(6, ' ');
297        let range = fields.next()?;
298        let (start, end) = range.split_once('-')?;
299        let (start, end) = (
300            u64::from_str_radix(start, 16).ok()?,
301            u64::from_str_radix(end, 16).ok()?,
302        );
303        if !(start..end).contains(&address) {
304            return None;
305        }
306        let ino = fields.nth(3)?.parse().ok()?;
307        let path = fields.next()?.trim_start();
308        (path.starts_with('/') && !path.ends_with(" (deleted)")).then(|| MappedFile {
309            path: PathBuf::from(path),
310            ino,
311            range: range.to_owned(),
312        })
313    })
314}

316impl<T: crate::record_or_replay::RecordOrReplay> crate::Detcore<T> {
317    /// Report the files the kernel opened for the exec this image came from
318    /// to the global state's host-input records ([`HostInputRecord::exec`]),
319    /// each with its path in the process's root, its identity now, at the
320    /// exec, and where the tool finds it for the run's end to read its bytes
321    /// ([`content_at_end`], which reads only a file that still has that
322    /// identity):
323    /// - the image, through the kernel's own `exe` link; for a `#!` script,
324    ///   the interpreter the kernel finally ran, not the script;
325    /// - its program interpreter, if it has one: the file mapped where the
326    ///   auxiliary vector's `AT_BASE` says the kernel loaded it, by the name
327    ///   `/proc/<tid>/maps` shows ([`mapped_file`]), when a stat through that
328    ///   name finds the object mapped: the device and inode a stat through
329    ///   the mapping's `map_files` link finds, where the tool may follow it,
330    ///   and otherwise the inode number the maps line names
331    ///   ([`MappedFile::identifies`]). Otherwise the mapped object cannot be
332    ///   identified and nothing is recorded for it, rather than a file that
333    ///   replaced it at that name since (or, on overlayfs, an overlay file
334    ///   that does not report the underlying number).
335    ///
336    /// Both identities are the exec's: a file replaced while the run goes on
337    /// keeps the identity the process ran with, as a file the guest opens
338    /// keeps the one its open found. Where the tool may not follow
339    /// `map_files`, only the interpreter's inode number is compared, so an
340    /// inode of the same number that another filesystem or btrfs subvolume
341    /// exposes at the name cannot be told apart. Nothing is opened but the thread's own
342    /// procfs files (the `auxv` and `maps` files, read as
343    /// `Self::check_brk_model` reads `auxv`), and the image and its
344    /// interpreter only have their status taken (see the module
345    /// documentation). Nothing where the tool cannot see the thread's procfs
346    /// directory. Raw syscalls only, as for [`read_digest`].
347    pub(crate) async fn record_exec_host_inputs<G: reverie::Guest<Self>>(&self, guest: &mut G) {
348        let Some(task) = self.guest_task_directory(guest) else {
349            return;
350        };
351        let Ok(root) = crate::util::raw_read_link(&task.join("root")) else {
352            return;
353        };
354        let exe = task.join("exe");
355        let Ok(source) = crate::util::raw_read_link(&exe) else {
356            return;
357        };
358        let Some(image) = source
359            .strip_prefix(&root)
360            .ok()
361            .map(|image| Path::new("/").join(image))
362        else {
363            return;
364        };
365        let Ok(stat) = crate::util::raw_stat(&exe) else {
366            return;
367        };
368        let mut files = vec![(image, source, stat)];
369        if let Some(mapped) = crate::util::raw_read_file(&task.join("auxv"))
370            .ok()
371            .and_then(|auxv| crate::exec_record::auxv_entry(&auxv, libc::AT_BASE))
372            .filter(|base| *base != 0)
373            .and_then(|base| {
374                mapped_file(&crate::util::raw_read_file(&task.join("maps")).ok()?, base)
375            })
376            && let Ok(stat) = crate::util::raw_stat(&mapped.path)
377            && mapped.identifies(
378                &stat,
379                crate::util::raw_stat(&task.join("map_files").join(&mapped.range))
380                    .ok()
381                    .as_ref(),
382            )
383            && let Ok(relative) = mapped.path.strip_prefix(&root)
384        {
385            files.push((Path::new("/").join(relative), mapped.path.clone(), stat));
386        }
387        for (path, source, stat) in files {
388            crate::tool_global::record_host_input(
389                guest,
390                path.to_string_lossy().into_owned(),
391                HostFileIdentity::from_stat(&stat),
392                source.to_str().map(str::to_owned),
393                stat.st_mode & libc::S_IFMT == libc::S_IFREG,
394                true,
395                false,
396            )
397            .await;
398        }
399    }
400}

402/// A schedule replay's comparison of the host files its guest opens against
403/// the recorded run's [`HostInputSet`]. Each path is compared once, for its
404/// first open in this run; a path the recorded set does not hold, a kernel
405/// object, or one the guest has rebound by then is not compared. A change
406/// the open's status already proves is reported at the open
407/// ([`Self::observe`]): another identity where the recording has no content,
408/// another size of a regular file. Any other is decided by the content the
409/// run's end reads ([`Self::finish`]), as [`host_input_changed`] decides.
410#[derive(Debug)]
411pub(crate) struct ReplayHostInputs {
412    recorded: HostInputSet,
413    opened: HashSet<String>,
414    /// First opens left for [`Self::finish`]: the path and the identity the
415    /// open found.
416    pending: Vec<(String, HostFileIdentity)>,
417    changes: Vec<HostInputChange>,
418}
419
420impl ReplayHostInputs {
421    pub(crate) fn new(recorded: HostInputSet) -> Self {
422        let recorded = recorded.normalized();
423        Self {
424            recorded,
425            opened: HashSet::new(),
426            pending: Vec::new(),
427            changes: Vec::new(),
428        }
429    }
430
431    /// Compare an open of `path` that found `identity` (`regular` for a
432    /// regular file; `kernel_object` when the descriptor's own path is a
433    /// kernel object), with `mutated` the paths the guest has rebound so
434    /// far. Returns the change when this first open of the path already
435    /// shows a different file than the recording: where the recording holds
436    /// no content its identity decides, and a regular file of another size
437    /// holds other bytes whatever either run could read. Otherwise the path
438    /// waits for [`Self::finish`].
439    pub(crate) fn observe(
440        &mut self,
441        path: &str,
442        identity: HostFileIdentity,
443        regular: bool,
444        kernel_object: bool,
445        mutated: &BTreeSet<String>,
446    ) -> Option<&HostInputChange> {
447        if !self.opened.insert(path.to_owned())
448            || kernel_object
449            || is_kernel_object(path)
450            || guest_rebound(mutated, path)
451        {
452            return None;
453        }
454        let recorded = self.recorded.file(path)?;
455        let proven = match recorded.content {
456            None => recorded.identity != identity,
457            Some(_) => regular && recorded.identity.size != identity.size,
458        };
459        if !proven {
460            self.pending.push((path.to_owned(), identity));
461            return None;
462        }
463        self.changes.push(HostInputChange {
464            path: path.to_owned(),
465            recorded: recorded.identity,
466            now: identity,
467        });
468        self.changes.last()
469    }
470
471    /// Decide the opens [`Self::observe`] left, once the run has ended, with
472    /// `content` the content the run's end read for a path's first open
473    /// ([`content_at_end`]) and `mutated` every path the guest rebound in
474    /// the run, which the run's own set leaves out too. Returns the changes
475    /// this finds, in the order of their first opens, after those reported
476    /// at the opens.
477    pub(crate) fn finish(
478        &mut self,
479        mut content: impl FnMut(&str) -> Option<String>,
480        mutated: &BTreeSet<String>,
481    ) -> &[HostInputChange] {
482        let found = self.changes.len();
483        for (path, identity) in std::mem::take(&mut self.pending) {
484            if guest_rebound(mutated, &path) {
485                continue;
486            }
487            let Some(recorded) = self.recorded.file(&path) else {
488                continue;
489            };
490            let now = HostInput {
491                content: content(&path),
492                path,
493                identity,
494            };
495            if host_input_changed(recorded, &now) {
496                self.changes.push(HostInputChange {
497                    path: now.path,
498                    recorded: recorded.identity,
499                    now: identity,
500                });
501            }
502        }
503        &self.changes[found..]
504    }
505
506    /// A captured input's private copy could not be checked at shutdown.
507    /// Do not let its stable origin hide that loss of evidence: the existing
508    /// identity-only fallback must compare the physical object instead. Only
509    /// the first open awaiting settlement is changed; changes already found
510    /// by `observe` remain reported.
511    pub(crate) fn restore_snapshot_identity(&mut self, path: &str, physical: HostFileIdentity) {
512        if let Some((_, identity)) = self.pending.iter_mut().find(|(pending, _)| pending == path) {
513            *identity = physical;
514        }
515    }
516
517    /// The changes found so far: those reported at their opens, in the order
518    /// of those opens, then those [`Self::finish`] found.
519    pub(crate) fn changes(&self) -> &[HostInputChange] {
520        &self.changes
521    }
522}

585        assert_ne!(first.digest, input_set(&[], &none).digest);
586        // Where the bytes were hashed they decide, as `host_input_changed`
587        // decides: a new inode with the same bytes keeps the digest, other
588        // bytes under the same inode change it.
589        let hashed = |ino: u64, content: &str| HostInputRecord {
590            content: Some(content.to_owned()),
591            ..open("/b", ino, 2)
592        };
593        let before = input_set(&[open("/a", 1, 1), hashed(2, "aa")], &none);
594        let rewritten = input_set(&[open("/a", 1, 1), hashed(3, "aa")], &none);
595        let edited = input_set(&[open("/a", 1, 1), hashed(2, "bb")], &none);
596        assert_eq!(before.digest, rewritten.digest);
597        assert_ne!(before.digest, edited.digest);
598    }

648            content: Some(content.to_owned()),

654                hashed("/etc/hosts", 10, "cc"),

656                hashed("/etc", 12, "ee"),
657            ],

659        );
660        let mut replay = ReplayHostInputs::new(recorded);
661        let resized = HostFileIdentity {
662            size: 1,
663            ..identity(11)
664        };

714        // is not compared.

717                hashed("/a", 1, "aa"),
718                hashed("/b", 2, "bb"),
719                hashed("/c", 3, "cc"),

740    /// The run's end reads a regular file's bytes only while it has the

745    fn the_end_reads_an_unchanged_file_and_a_directory_by_its_names() {
746        let directory = tempfile::tempdir().unwrap();
747        let path = directory.path().join("F");
748        std::fs::write(&path, "input\n").unwrap();
749        let none = BTreeSet::new();
750        let found = HostFileIdentity::from_stat(&crate::util::raw_stat(&path).unwrap());
751        // `printf 'input\n' | sha256sum`
752        let expected = "7d3f9b6284c6f36e77b425cac882e8fbbcc97a4727ec20790853076d0f463453";
753        assert_eq!(
754            content_at_end(&path, "/F", found, &none).as_deref(),

759            mtime_nsec: found.mtime_nsec ^ 1,

761        };
762        assert_eq!(content_at_end(&path, "/F", other, &none), None);
763        std::fs::write(&path, "other\n").unwrap();
764        let rewritten = HostFileIdentity::from_stat(&crate::util::raw_stat(&path).unwrap());
765        assert_ne!(
766            content_at_end(&path, "/F", rewritten, &none).as_deref(),

769        let large = HostFileIdentity {

771            ..rewritten
772        };
773        assert_eq!(content_at_end(&path, "/F", large, &none), None);
774        assert_eq!(
775            content_at_end(&directory.path().join("missing"), "/missing", found, &none),

777        );

779        let listed = tempfile::tempdir().unwrap();

781        std::fs::write(listed.path().join("a"), "").unwrap();

785            .expect("a directory's names");

837    /// The program interpreter is the file mapped at the auxiliary vector's
838    /// `AT_BASE`: for this test binary, which is dynamically linked, the
839    /// dynamic loader, under the name the mapping shows, and a stat through
840    /// that name identifies the object mapped. A mapping of a deleted file,
841    /// or of no file, names none. Another inode at the name is not the object
842    /// mapped, and nor is an inode of the same number on another device where
843    /// the mapping's `map_files` link can be followed.
844    #[test]
845    fn the_interpreter_is_the_file_mapped_at_the_base_the_auxv_names() {
846        let auxv = crate::util::raw_read_file(Path::new("/proc/self/auxv")).unwrap();
847        let base = crate::exec_record::auxv_entry(&auxv, libc::AT_BASE).unwrap();
848        assert_ne!(base, 0, "the test binary is dynamically linked");
849        let maps = crate::util::raw_read_file(Path::new("/proc/self/maps")).unwrap();
850        let mapped = mapped_file(&maps, base).expect("a file is mapped at AT_BASE");
851        let interpreter = &mapped.path;
852        assert!(interpreter.is_absolute(), "{}", interpreter.display());
853        // The dynamic loader: `ld-linux-<arch>.so.<n>` or `ld-musl-<arch>.so.1`.
854        assert!(
855            interpreter
856                .file_name()
857                .is_some_and(|name| name.to_string_lossy().starts_with("ld-")),
858            "{} is not a dynamic loader",
859            interpreter.display()
860        );
861        // A stat through the name finds the object mapped, except where the
862        // loader is on overlayfs, whose maps lines name the underlying file.
863        let stat = crate::util::raw_stat(interpreter).unwrap();
864        let mut fs = std::mem::MaybeUninit::<libc::statfs>::uninit();
865        let name = std::ffi::CString::new(interpreter.as_os_str().as_bytes()).unwrap();
866        // SAFETY: `name` is NUL-terminated and `fs` is written before it is read.
867        assert_eq!(unsafe { libc::statfs(name.as_ptr(), fs.as_mut_ptr()) }, 0);
868        // SAFETY: statfs returned 0, so it filled `fs`.
869        let overlay = unsafe { fs.assume_init() }.f_type == libc::OVERLAYFS_SUPER_MAGIC;
870        let link = Path::new("/proc/self/map_files").join(&mapped.range);
871        let through_link = crate::util::raw_stat(&link).ok();
872        assert!(
873            mapped.identifies(&stat, through_link.as_ref()) || overlay,
874            "{mapped:?}"
875        );
876        assert!(mapped.identifies(&stat, None) || overlay, "{mapped:?}");
877        // Another inode at the name is not the object mapped, with or without
878        // the link.
879        let mut replaced = stat;
880        replaced.st_ino ^= 1;
881        assert!(!mapped.identifies(&replaced, None));
882        assert!(!mapped.identifies(&replaced, Some(&stat)));
883        // An inode of the same number on another device (another filesystem,
884        // or another btrfs subvolume, mounted over the name) is the object
885        // mapped only by its number: the link's device tells it apart where
886        // the tool may follow the link.
887        let mut covering = stat;
888        covering.st_dev ^= 1;
889        assert!(!mapped.identifies(&covering, Some(&stat)));
890        assert!(mapped.identifies(&covering, None));
891
892        let maps = b"00400000-00401000 r-xp 00000000 08:01 12 /usr/bin/a b\n\
893            7f0000000000-7f0000001000 r--p 00000000 08:01 13 /lib/ld (deleted)\n\
894            7f0000002000-7f0000003000 r--p 00000000 00:00 0 \n\
895            7f0000004000-7f0000005000 r--p 00000000 103:1a 4096 /lib/ld-x.so\n";
896        assert_eq!(
897            mapped_file(maps, 0x400800),
898            Some(MappedFile {
899                path: PathBuf::from("/usr/bin/a b"),
900                ino: 12,
901                range: "00400000-00401000".to_owned(),
902            })
903        );
904        assert_eq!(
905            mapped_file(maps, 0x7f0000004000),
906            Some(MappedFile {
907                path: PathBuf::from("/lib/ld-x.so"),
908                ino: 4096,
909                range: "7f0000004000-7f0000005000".to_owned(),
910            })
911        );
912        assert_eq!(mapped_file(maps, 0x7f0000000000), None);
913        assert_eq!(mapped_file(maps, 0x7f0000002000), None);
914        assert_eq!(mapped_file(maps, 0x500000), None);
915    }
