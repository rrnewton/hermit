9//! Writes a run's host-input records to its `host_input_log` when the run
10//! ends (see `detcore_model::host_input`), sums them up as the run's
11//! [`HostInputSet`] for its summary and its preemption record, and compares
12//! the opens of a schedule replay against the recorded run's set
13//! ([`ReplayHostInputs`]).
14//!
15//! The records are an observation for the verification layer, not Detcore
16//! state: Detcore's openat handler reports each one to the global state
17//! (`tool_global::record_host_input`), which keeps them for the run, and
18//! nothing reads them back, no scheduling decision consults them, and nothing
19//! the guest sees depends on them. They are written once, after the run, so
20//! the log stays empty while any guest runs; a run that does not end normally
21//! writes nothing, which leaves its divergence unexplained rather than
22//! explained by partial evidence.
23//!
24//! Nothing here opens or reads a guest's file while any guest process runs.
25//! Linux raises an inotify or fanotify event for every open, read and close
26//! of a file, whoever makes it (`fsnotify_open`, `fsnotify_access` and
27//! `fsnotify_close`, fs/namei.c, fs/read_write.c and fs/file_table.c), so a
28//! guest watching the file's directory would see Hermit's reads among its
29//! own. While the guest runs, Detcore takes only what raises no event: a
30//! file's status, and the kernel's path for the guest's descriptor (a
31//! `readlink` of `/proc/<tid>/fd/<n>`). The bytes, and a directory's names,
32//! are read when the run has ended ([`content_at_end`]), and only where the
33//! file still has the identity the guest's open found.
34

57/// Write `records` to `log`, one JSON line each, then one
58/// [`HostMutationRecord`] line for each path in `mutated`, followed by a
59/// [`HostInputLogEnd`] line naming how many lines precede it, so that a
60/// reader can tell a complete log from a cut one.
61pub(crate) fn write(log: &Path, records: &[HostInputRecord], mutated: &BTreeSet<String>) {
62    let mut text = Vec::new();
63    let end = HostInputLogEnd {
64        records: (records.len() + mutated.len()) as u64,
65    };
66    let encoded = records
67        .iter()
68        .map(serde_json::to_vec)
69        .chain(mutated.iter().map(|path| {
70            serde_json::to_vec(&HostMutationRecord {
71                mutated: path.clone(),
72            })
73        }))
74        .chain(std::iter::once(serde_json::to_vec(&end)))
75        .try_for_each(|line| {
76            text.extend(line?);
77            text.push(b'\n');
78            Ok::<(), serde_json::Error>(())
79        });
80    let written = encoded
81        .map_err(std::io::Error::other)
82        .and_then(|()| std::fs::write(log, &text));
83    if let Err(error) = written {
84        warn!(
85            "[detcore] cannot write the host-input log {}: {}",
86            log.display(),
87            error
88        );
89    }
90}

92/// The run's [`HostInputSet`]: for each path the guest opened, in path order,
93/// the identity and bytes its first open found, leaving out kernel objects
94/// and paths the guest rebound itself (`mutated`, or a relative path), and
95/// the set's digest (`HostInputSet`).
96pub(crate) fn input_set(records: &[HostInputRecord], mutated: &BTreeSet<String>) -> HostInputSet {
97    let mut first: BTreeMap<&str, &HostInputRecord> = BTreeMap::new();
98    for record in records {
99        first.entry(record.path.as_str()).or_insert(record);
100    }
101    let files: Vec<HostInput> = first
102        .into_iter()
103        .filter(|(path, record)| {
104            // An authenticated snapshot identity (`of_snapshot`, a content
105            // hash Hermit itself computed for its frozen copy) is retained
106            // even though its guest path lies under `/dev`: it names the
107            // snapshot's bytes, not a device. Actual devices and kernel
108            // objects carry no such identity and stay excluded.
109            let snapshot = record.identity.content.is_some();
110            (snapshot || (!record.kernel_object && !is_kernel_object(path)))
111                && !guest_rebound(mutated, path)
112        })
113        .map(|(path, record)| HostInput {
114            path: path.to_owned(),
115            identity: record.identity,
116            content: record.content.clone(),
117        })
118        .collect();
119    // One line per file: its path (which holds no NUL), then its bytes'
120    // digest where hashed, else its identity.
121    let mut hasher = Sha256::new();
122    for file in &files {
123        hasher.update(file.path.as_bytes());
124        hasher.update(b"\0");
125        match &file.content {
126            Some(content) => {
127                hasher.update(b"content ");
128                hasher.update(content.as_bytes());
129            }
130            None => {
131                hasher.update(b"identity ");
132                hasher.update(serde_json::to_vec(&file.identity).expect("an identity serializes"));
133            }
134        }
135        hasher.update(b"\n");
136    }
137    HostInputSet {
138        digest: hex(&hasher.finalize()),
139        files,
140        stdio: None,
141    }
142}

144fn hex(bytes: &[u8]) -> String {
145    bytes.iter().map(|byte| format!("{byte:02x}")).collect()
146}

148/// What a run's end reads for the first open of `path` (the guest's path)
149/// that found `identity`, through `source` (the kernel's path for the
150/// guest's descriptor, or for an exec's image, at that open): the SHA-256 of
151/// a regular file's bytes, when it still has that identity and holds at most
152/// [`HOST_INPUT_CONTENT_LIMIT`] bytes, or of the names in a directory that is
153/// still the same directory (device and inode; its size and times change
154/// with every name made in it) as [`directory_digest`] lists them; `None`
155/// otherwise, and for any other kind of file, which is never opened. Called
156/// only when every guest process has exited (see the module documentation).
157/// Raw syscalls only, as for [`read_digest`].
158pub(crate) fn content_at_end(
159    source: &Path,
160    path: &str,
161    identity: HostFileIdentity,
162    mutated: &BTreeSet<String>,
163) -> Option<String> {
164    let stat = crate::util::raw_stat(source).ok()?;
165    match stat.st_mode & libc::S_IFMT {
166        libc::S_IFREG => {
167            if HostFileIdentity::from_stat(&stat) != identity
168                || u64::try_from(identity.size).map_or(true, |size| size > HOST_INPUT_CONTENT_LIMIT)
169            {
170                return None;
171            }
172            let file = crate::util::RawFd::open(source, libc::O_RDONLY | libc::O_NONBLOCK).ok()?;
173            // The file opened is the one whose status was compared.
174            (HostFileIdentity::from_stat(&file.stat().ok()?) == identity)
175                .then(|| read_digest(&file))
176                .flatten()
177        }
178        libc::S_IFDIR => {
179            if (stat.st_dev, stat.st_ino) != (identity.dev, identity.ino) {
180                return None;
181            }
182            let directory =
183                crate::util::RawFd::open(source, libc::O_RDONLY | libc::O_DIRECTORY).ok()?;
184            let opened = directory.stat().ok()?;
185            ((opened.st_dev, opened.st_ino) == (identity.dev, identity.ino))
186                .then(|| directory_digest(&directory, path, mutated))
187                .flatten()
188        }
189        _ => None,
190    }
191}

193/// The SHA-256 of the file open at `file`, read to its end but no further
194/// than [`HOST_INPUT_CONTENT_LIMIT`] bytes, or `None` when it cannot be read
195/// or is longer than that. Raw syscalls only: in-guest Detcore must not call
196/// libc's wrappers, which the guest may define (`crate::util::raw_syscall`).
197fn read_digest(file: &crate::util::RawFd) -> Option<String> {
198    let mut hasher = Sha256::new();
199    let mut buffer = vec![0_u8; 1 << 16];
200    let mut read = 0_u64;
201    loop {
202        let count = file.read_at(&mut buffer, read).ok()?;
203        if count == 0 {
204            break;
205        }
206        read += count as u64;
207        if read > HOST_INPUT_CONTENT_LIMIT {
208            return None;
209        }
210        hasher.update(&buffer[..count]);
211    }
212    Some(hex(&hasher.finalize()))
213}

215/// The SHA-256 of the names in the directory open at `directory`, which the
216/// guest names `path`: sorted, each followed by a NUL, after `directory\0`,
217/// without `.` and `..` and without each name whose path the guest itself
218/// rebound (`mutated`, [`guest_rebound`]), so that what the guest made or
219/// removed in it (in its private `/tmp`, say) does not count. `None` when
220/// the names are longer in all than [`HOST_INPUT_CONTENT_LIMIT`] bytes or
221/// cannot be read. A directory's identity is not compared: Hermit makes some
222/// afresh for each run (its private `/tmp`), and the guest sees only the
223/// names in one, not its host inode or times.
224fn directory_digest(
225    directory: &crate::util::RawFd,
226    path: &str,
227    mutated: &BTreeSet<String>,
228) -> Option<String> {
229    let mut names = Vec::new();
230    let mut total = 0_u64;
231    let too_long = crate::util::find_in_open_directory(directory, |name| {
232        total += name.len() as u64 + 1;
233        if total > HOST_INPUT_CONTENT_LIMIT {
234            return Some(());
235        }
236        let child = Path::new(path).join(name);
237        if !guest_rebound(mutated, &child.to_string_lossy()) {
238            names.push(name.as_bytes().to_vec());
239        }
240        None
241    })
242    .ok()?;
243    if too_long.is_some() {
244        return None;
245    }
246    names.sort();
247    let mut hasher = Sha256::new();
248    hasher.update(b"directory\0");
249    for name in &names {
250        hasher.update(name);
251        hasher.update(b"\0");
252    }
253    Some(hex(&hasher.finalize()))
254}

256/// A file mapped in a process, as a `/proc/<pid>/maps` line shows it: its
257/// name, the inode number of the object mapped, and the mapping's address
258/// range as the line spells it, which names the mapping's
259/// `/proc/<pid>/map_files` link.
260#[derive(Debug, PartialEq)]
261struct MappedFile {
262    path: PathBuf,
263    ino: u64,
264    range: String,
265}
266
267impl MappedFile {
268    /// Whether `stat`, taken through [`Self::path`], is of the object mapped.
269    ///
270    /// Where the tool may follow the mapping's `map_files` link (`mapped`, a
271    /// stat through it; Linux allows that only to a holder of
272    /// CAP_CHECKPOINT_RESTORE or CAP_SYS_ADMIN in the initial user
273    /// namespace), the two have the same device and inode, which keeps the
274    /// filesystem and, on btrfs, the subvolume.
275    ///
276    /// Otherwise the inode number is all that names the object mapped: a
277    /// maps line shows its superblock's device (`show_map_vma`), while btrfs
278    /// reports a subvolume's own device in a stat (`btrfs_getattr`), so the
279    /// devices differ for the same file. The mapping holds the mapped inode,
280    /// so a file that replaced it at the path has another number on that
281    /// filesystem; what the number cannot rule out is another filesystem or
282    /// subvolume mounted over the path with an inode of the same number. On
283    /// overlayfs a maps line names the underlying file, whose number a stat
284    /// through the overlay need not report, so the object may not be
285    /// identified there.
286    fn identifies(&self, stat: &libc::stat, mapped: Option<&libc::stat>) -> bool {
287        match mapped {
288            Some(mapped) => (mapped.st_dev, mapped.st_ino) == (stat.st_dev, stat.st_ino),
289            None => stat.st_ino == self.ino,
290        }
291    }
292}

294/// The file mapped at `address` in a process whose `/proc/<pid>/maps` is
295/// `maps`: the mapping's pathname field, when it names a file (absolute, and
296/// not deleted since it was mapped), and its inode field. Linux shows the
297/// name from the reader's root, and the inode of the file mapped
298/// (`show_map_vma`, fs/proc/task_mmu.c).
299fn mapped_file(maps: &[u8], address: u64) -> Option<MappedFile> {
300    maps.split(|byte| *byte == b'\n').find_map(|line| {
301        let line = std::str::from_utf8(line).ok()?;
302        // address perms offset dev inode pathname: the pathname is the rest of
303        // the line after the fifth field, and may hold spaces.
304        let mut fields = line.splitn(6, ' ');
305        let range = fields.next()?;
306        let (start, end) = range.split_once('-')?;
307        let (start, end) = (
308            u64::from_str_radix(start, 16).ok()?,
309            u64::from_str_radix(end, 16).ok()?,
310        );
311        if !(start..end).contains(&address) {
312            return None;
313        }
314        let ino = fields.nth(3)?.parse().ok()?;
315        let path = fields.next()?.trim_start();
316        (path.starts_with('/') && !path.ends_with(" (deleted)")).then(|| MappedFile {
317            path: PathBuf::from(path),
318            ino,
319            range: range.to_owned(),
320        })
321    })
322}

324impl<T: crate::record_or_replay::RecordOrReplay> crate::Detcore<T> {
325    /// Report the files the kernel opened for the exec this image came from
326    /// to the global state's host-input records ([`HostInputRecord::exec`]),
327    /// each with its path in the process's root, its identity now, at the
328    /// exec, and where the tool finds it for the run's end to read its bytes
329    /// ([`content_at_end`], which reads only a file that still has that
330    /// identity):
331    /// - the image, through the kernel's own `exe` link; for a `#!` script,
332    ///   the interpreter the kernel finally ran, not the script;
333    /// - its program interpreter, if it has one: the file mapped where the
334    ///   auxiliary vector's `AT_BASE` says the kernel loaded it, by the name
335    ///   `/proc/<tid>/maps` shows ([`mapped_file`]), when a stat through that
336    ///   name finds the object mapped: the device and inode a stat through
337    ///   the mapping's `map_files` link finds, where the tool may follow it,
338    ///   and otherwise the inode number the maps line names
339    ///   ([`MappedFile::identifies`]). Otherwise the mapped object cannot be
340    ///   identified and nothing is recorded for it, rather than a file that
341    ///   replaced it at that name since (or, on overlayfs, an overlay file
342    ///   that does not report the underlying number).
343    ///
344    /// Both identities are the exec's: a file replaced while the run goes on
345    /// keeps the identity the process ran with, as a file the guest opens
346    /// keeps the one its open found. Where the tool may not follow
347    /// `map_files`, only the interpreter's inode number is compared, so an
348    /// inode of the same number that another filesystem or btrfs subvolume
349    /// exposes at the name cannot be told apart. Nothing is opened but the thread's own
350    /// procfs files (the `auxv` and `maps` files, read as
351    /// `Self::check_brk_model` reads `auxv`), and the image and its
352    /// interpreter only have their status taken (see the module
353    /// documentation). Nothing where the tool cannot see the thread's procfs
354    /// directory. Raw syscalls only, as for [`read_digest`].
355    pub(crate) async fn record_exec_host_inputs<G: reverie::Guest<Self>>(&self, guest: &mut G) {
356        let Some(task) = self.guest_task_directory(guest) else {
357            return;
358        };
359        let Ok(root) = crate::util::raw_read_link(&task.join("root")) else {
360            return;
361        };
362        let exe = task.join("exe");
363        let Ok(source) = crate::util::raw_read_link(&exe) else {
364            return;
365        };
366        let Some(image) = source
367            .strip_prefix(&root)
368            .ok()
369            .map(|image| Path::new("/").join(image))
370        else {
371            return;
372        };
373        let Ok(stat) = crate::util::raw_stat(&exe) else {
374            return;
375        };
376        let mut files = vec![(image, source, stat)];
377        if let Some(mapped) = crate::util::raw_read_file(&task.join("auxv"))
378            .ok()
379            .and_then(|auxv| crate::exec_record::auxv_entry(&auxv, libc::AT_BASE))
380            .filter(|base| *base != 0)
381            .and_then(|base| {
382                mapped_file(&crate::util::raw_read_file(&task.join("maps")).ok()?, base)
383            })
384            && let Ok(stat) = crate::util::raw_stat(&mapped.path)
385            && mapped.identifies(
386                &stat,
387                crate::util::raw_stat(&task.join("map_files").join(&mapped.range))
388                    .ok()
389                    .as_ref(),
390            )
391            && let Ok(relative) = mapped.path.strip_prefix(&root)
392        {
393            files.push((Path::new("/").join(relative), mapped.path.clone(), stat));
394        }
395        for (path, source, stat) in files {
396            crate::tool_global::record_host_input(
397                guest,
398                path.to_string_lossy().into_owned(),
399                HostFileIdentity::from_stat(&stat),
400                source.to_str().map(str::to_owned),
401                stat.st_mode & libc::S_IFMT == libc::S_IFREG,
402                true,
403                false,
404            )
405            .await;
406        }
407    }
408}

410/// A schedule replay's comparison of the host files its guest opens against
411/// the recorded run's [`HostInputSet`]. Each path is compared once, for its
412/// first open in this run; a path the recorded set does not hold, a kernel
413/// object, or one the guest has rebound by then is not compared. A change
414/// the open's status already proves is reported at the open
415/// ([`Self::observe`]): another identity where the recording has no content,
416/// another size of a regular file. Any other is decided by the content the
417/// run's end reads ([`Self::finish`]), as [`host_input_changed`] decides.
418#[derive(Debug)]
419pub(crate) struct ReplayHostInputs {
420    recorded: HostInputSet,
421    opened: HashSet<String>,
422    /// First opens left for [`Self::finish`]: the path and the identity the
423    /// open found.
424    pending: Vec<(String, HostFileIdentity)>,
425    changes: Vec<HostInputChange>,
426}
427
428impl ReplayHostInputs {
429    pub(crate) fn new(recorded: HostInputSet) -> Self {
430        let recorded = recorded.normalized();
431        Self {
432            recorded,
433            opened: HashSet::new(),
434            pending: Vec::new(),
435            changes: Vec::new(),
436        }
437    }
438
439    /// Compare an open of `path` that found `identity` (`regular` for a
440    /// regular file; `kernel_object` when the descriptor's own path is a
441    /// kernel object), with `mutated` the paths the guest has rebound so
442    /// far. Returns the change when this first open of the path already
443    /// shows a different file than the recording: where the recording holds
444    /// no content its identity decides, and a regular file of another size
445    /// holds other bytes whatever either run could read. Otherwise the path
446    /// waits for [`Self::finish`].
447    pub(crate) fn observe(
448        &mut self,
449        path: &str,
450        identity: HostFileIdentity,
451        regular: bool,
452        kernel_object: bool,
453        mutated: &BTreeSet<String>,
454    ) -> Option<&HostInputChange> {
455        // A recorded authenticated snapshot identity is compared even
456        // though its path lies under `/dev` (see `input_set`); an actual
457        // device has no recorded entry and stays skipped here.
458        let snapshot = identity.content.is_some()
459            || self
460                .recorded
461                .file(path)
462                .is_some_and(|file| file.identity.content.is_some());
463        if self.opened.contains(path) {
464            return None;
465        }
466        // The descriptor-path `kernel_object` bit is a path heuristic: a
467        // regular snapshot copy under `/dev/shm` (a permitted `TMPDIR`)
468        // gets it from its copy path, yet its authenticated content hash
469        // (`of_snapshot`) is stronger evidence than the path. Mirror
470        // `input_set`: a snapshot identity is compared even when the bit
471        // is set; an actual device has no snapshot identity and stays
472        // excluded by either check.
473        if (!snapshot && (kernel_object || is_kernel_object(path))) || guest_rebound(mutated, path)
474        {
475            return None;
476        }
477        self.opened.insert(path.to_owned());
478        let recorded = self.recorded.file(path)?;
479        let proven = match recorded.content {
480            None => recorded.identity != identity,
481            Some(_) => regular && recorded.identity.size != identity.size,
482        };
483        if !proven {
484            self.pending.push((path.to_owned(), identity));
485            return None;
486        }
487        self.changes.push(HostInputChange {
488            path: path.to_owned(),
489            recorded: recorded.identity,
490            now: identity,
491        });
492        self.changes.last()
493    }
494
495    /// Decide the opens [`Self::observe`] left, once the run has ended, with
496    /// `content` the content the run's end read for a path's first open
497    /// ([`content_at_end`]) and `mutated` every path the guest rebound in
498    /// the run, which the run's own set leaves out too. Returns the changes
499    /// this finds, in the order of their first opens, after those reported
500    /// at the opens.
501    pub(crate) fn finish(
502        &mut self,
503        mut content: impl FnMut(&str) -> Option<String>,
504        mutated: &BTreeSet<String>,
505    ) -> &[HostInputChange] {
506        let found = self.changes.len();
507        for (path, identity) in std::mem::take(&mut self.pending) {
508            if guest_rebound(mutated, &path) {
509                continue;
510            }
511            let Some(recorded) = self.recorded.file(&path) else {
512                continue;
513            };
514            let now = HostInput {
515                content: content(&path),
516                path,
517                identity,
518            };
519            if host_input_changed(recorded, &now) {
520                self.changes.push(HostInputChange {
521                    path: now.path,
522                    recorded: recorded.identity,
523                    now: identity,
524                });
525            }
526        }
527        &self.changes[found..]
528    }
529
530    /// A captured input's private copy could not be checked at shutdown.
531    /// Do not let its stable origin hide that loss of evidence: the existing
532    /// identity-only fallback must compare the physical object instead. Only
533    /// the first open awaiting settlement is changed; changes already found
534    /// by `observe` remain reported.
535    pub(crate) fn restore_snapshot_identity(&mut self, path: &str, physical: HostFileIdentity) {
536        if let Some((_, identity)) = self.pending.iter_mut().find(|(pending, _)| pending == path) {
537            *identity = physical;
538        }
539    }
540
541    /// The changes found so far: those reported at their opens, in the order
542    /// of those opens, then those [`Self::finish`] found.
543    pub(crate) fn changes(&self) -> &[HostInputChange] {
544        &self.changes
545    }
546}

602    /// An authenticated snapshot identity survives `input_set` and replay
603    /// comparison despite its `/dev` guest path: unchanged bytes report no
604    /// change, changed bytes report one, and a real device stays excluded.
605    #[test]
606    fn a_snapshot_identity_survives_the_set_and_replay_comparison() {
607        let none = BTreeSet::new();
608        let snapshot = |hash: u64| HostInputRecord {
609            identity: HostFileIdentity::of_snapshot(64, hash),
610            ..open("/dev/__properties__/properties_serial", 1, 6)
611        };
612        let device = HostInputRecord {
613            kernel_object: true,
614            ..open("/dev/null", 2, 7)
615        };
616        let recorded = input_set(&[snapshot(0xaa), device.clone()], &none);
617        assert_eq!(recorded.files.len(), 1);
618        assert_eq!(
619            recorded.files[0].path,
620            "/dev/__properties__/properties_serial"
621        );
622        assert_eq!(
623            recorded.files[0].identity,
624            HostFileIdentity::of_snapshot(64, 0xaa)
625        );
626        // Different snapshot bytes give a different persisted digest.
627        let changed_set = input_set(&[snapshot(0xbb), device.clone()], &none);
628        assert_ne!(recorded.digest, changed_set.digest);
629        // Unchanged: no change at the open or at the end.
630        let mut replay = ReplayHostInputs::new(recorded.clone());
631        assert_eq!(
632            replay.observe(
633                "/dev/__properties__/properties_serial",
634                HostFileIdentity::of_snapshot(64, 0xaa),
635                true,
636                false,
637                &none,
638            ),
639            None
640        );
641        assert_eq!(replay.finish(|_| None, &none), []);
642        assert!(replay.changes().is_empty());
643        // The `/dev/null` control is neither in the set nor compared.
644        assert_eq!(
645            replay.observe("/dev/null", identity(2), false, true, &none),
646            None
647        );
648        // Changed: reported at the open with both hashes.
649        let mut replay = ReplayHostInputs::new(recorded.clone());
650        let change = replay
651            .observe(
652                "/dev/__properties__/properties_serial",
653                HostFileIdentity::of_snapshot(64, 0xbb),
654                true,
655                false,
656                &none,
657            )
658            .expect("a changed snapshot hash is reported");
659        assert_eq!(change.recorded.content, Some(0xaa));
660        assert_eq!(change.now.content, Some(0xbb));
661        // The producer case: a regular copy under `/dev/shm` (a permitted
662        // `TMPDIR` below `/dev`) carries `kernel_object=true` from its
663        // descriptor path, in the persisted record and at replay. The
664        // authenticated snapshot identity must survive both, while an
665        // actual device with the bit set stays excluded.
666        let shm_snapshot = HostInputRecord {
667            kernel_object: true,
668            ..snapshot(0xaa)
669        };
670        let recorded = input_set(&[shm_snapshot, device.clone()], &none);
671        assert_eq!(recorded.files.len(), 1, "true-bit snapshot is retained");
672        assert_eq!(
673            recorded.files[0].identity,
674            HostFileIdentity::of_snapshot(64, 0xaa)
675        );
676        let mut replay = ReplayHostInputs::new(recorded.clone());
677        assert_eq!(
678            replay.observe(
679                "/dev/__properties__/properties_serial",
680                HostFileIdentity::of_snapshot(64, 0xaa),
681                true,
682                true,
683                &none,
684            ),
685            None,
686            "unchanged true-bit snapshot reports no change"
687        );
688        assert_eq!(replay.finish(|_| None, &none), []);
689        assert!(replay.changes().is_empty());
690        let mut replay = ReplayHostInputs::new(recorded);
691        let change = replay
692            .observe(
693                "/dev/__properties__/properties_serial",
694                HostFileIdentity::of_snapshot(64, 0xbb),
695                true,
696                true,
697                &none,
698            )
699            .expect("a changed true-bit snapshot hash is reported");
700        assert_eq!(change.recorded.content, Some(0xaa));
701        assert_eq!(change.now.content, Some(0xbb));
702    }

724        assert_ne!(before.digest, edited.digest);

964    /// The program interpreter is the file mapped at the auxiliary vector's
965    /// `AT_BASE`: for this test binary, which is dynamically linked, the
966    /// dynamic loader, under the name the mapping shows, and a stat through
967    /// that name identifies the object mapped. A mapping of a deleted file,
968    /// or of no file, names none. Another inode at the name is not the object
969    /// mapped, and nor is an inode of the same number on another device where
970    /// the mapping's `map_files` link can be followed.
971    #[test]
972    fn the_interpreter_is_the_file_mapped_at_the_base_the_auxv_names() {
973        let auxv = crate::util::raw_read_file(Path::new("/proc/self/auxv")).unwrap();
974        let base = crate::exec_record::auxv_entry(&auxv, libc::AT_BASE).unwrap();
975        assert_ne!(base, 0, "the test binary is dynamically linked");
976        let maps = crate::util::raw_read_file(Path::new("/proc/self/maps")).unwrap();
977        let mapped = mapped_file(&maps, base).expect("a file is mapped at AT_BASE");
978        let interpreter = &mapped.path;
979        assert!(interpreter.is_absolute(), "{}", interpreter.display());
980        // The dynamic loader: `ld-linux-<arch>.so.<n>` or `ld-musl-<arch>.so.1`.
981        assert!(
982            interpreter
983                .file_name()
984                .is_some_and(|name| name.to_string_lossy().starts_with("ld-")),
985            "{} is not a dynamic loader",
986            interpreter.display()
987        );
988        // A stat through the name finds the object mapped, except where the
989        // loader is on overlayfs, whose maps lines name the underlying file.
990        let stat = crate::util::raw_stat(interpreter).unwrap();
991        let mut fs = std::mem::MaybeUninit::<libc::statfs>::uninit();
992        let name = std::ffi::CString::new(interpreter.as_os_str().as_bytes()).unwrap();
993        // SAFETY: `name` is NUL-terminated and `fs` is written before it is read.
994        assert_eq!(unsafe { libc::statfs(name.as_ptr(), fs.as_mut_ptr()) }, 0);
995        // SAFETY: statfs returned 0, so it filled `fs`.
996        let overlay =
997            unsafe { fs.assume_init() }.f_type as u64 == libc::OVERLAYFS_SUPER_MAGIC as u64;
998        let link = Path::new("/proc/self/map_files").join(&mapped.range);
999        let through_link = crate::util::raw_stat(&link).ok();
1000        assert!(
1001            mapped.identifies(&stat, through_link.as_ref()) || overlay,
1002            "{mapped:?}"
1003        );
1004        assert!(mapped.identifies(&stat, None) || overlay, "{mapped:?}");
1005        // Another inode at the name is not the object mapped, with or without
1006        // the link.
1007        let mut replaced = stat;
1008        replaced.st_ino ^= 1;
1009        assert!(!mapped.identifies(&replaced, None));
1010        assert!(!mapped.identifies(&replaced, Some(&stat)));
1011        // An inode of the same number on another device (another filesystem,
1012        // or another btrfs subvolume, mounted over the name) is the object
1013        // mapped only by its number: the link's device tells it apart where
1014        // the tool may follow the link.
1015        let mut covering = stat;
1016        covering.st_dev ^= 1;
1017        assert!(!mapped.identifies(&covering, Some(&stat)));
1018        assert!(mapped.identifies(&covering, None));
1019
1020        let maps = b"00400000-00401000 r-xp 00000000 08:01 12 /usr/bin/a b\n\
1021            7f0000000000-7f0000001000 r--p 00000000 08:01 13 /lib/ld (deleted)\n\
1022            7f0000002000-7f0000003000 r--p 00000000 00:00 0 \n\
1023            7f0000004000-7f0000005000 r--p 00000000 103:1a 4096 /lib/ld-x.so\n";
1024        assert_eq!(
1025            mapped_file(maps, 0x400800),
1026            Some(MappedFile {
1027                path: PathBuf::from("/usr/bin/a b"),
1028                ino: 12,
1029                range: "00400000-00401000".to_owned(),
1030            })
1031        );
1032        assert_eq!(
1033            mapped_file(maps, 0x7f0000004000),
1034            Some(MappedFile {
1035                path: PathBuf::from("/lib/ld-x.so"),
1036                ino: 4096,
1037                range: "7f0000004000-7f0000005000".to_owned(),
1038            })
1039        );
1040        assert_eq!(mapped_file(maps, 0x7f0000000000), None);
1041        assert_eq!(mapped_file(maps, 0x7f0000002000), None);
1042        assert_eq!(mapped_file(maps, 0x500000), None);
1043    }
