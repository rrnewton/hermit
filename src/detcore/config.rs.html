14/// [`detcore_model::config::config_wire_fingerprint`], extended with the
15/// revision of Detcore's global RPC (`crate::tool_global::GLOBAL_RPC_REVISION`).
16///
17/// A runtime that embeds Detcore (SaBRe's plugin, the LiteInst runtime that
18/// in-guest-trap also loads, DBT's) sends `GlobalRequest`s to its coordinator
19/// as positional bincode and decodes the `GlobalResponse`s. Both enums live
20/// in this crate, outside the sources the configuration's fingerprint hashes,
21/// so a variant one build had and the other lacked went unnoticed: a newer
22/// runtime's request failed to decode in the middle of a run, and an older
23/// runtime never sent it (and so, for one, kept the host's inotify rename
24/// cookies). Every such runtime compares this fingerprint with its
25/// coordinator's before it decodes anything, so the pair is refused by name.
26/// The revision stands for the enums' source, which, unlike the
27/// configuration's, is not embedded: these runtimes are loaded into the guest,
28/// where every byte they grow by moves the guest's own mappings.
29pub fn config_wire_fingerprint() -> String {
30    // The material is immutable for this loaded image: the model overrides
31    // environment-dependent defaults, and the RPC revision is compiled in.
32    // Reuse it when a forked in-guest tool reconnects on the guest's clone
33    // stack; rebuilding Clap's canonical Config there can exhaust that stack.
34    // SaBRe's parent compatibility check finishes before an intercepted clone
35    // can run, so its child inherits an initialized value. Each connection
36    // still compares the coordinator's fingerprint with this local value.
37    static FINGERPRINT: std::sync::OnceLock<String> = std::sync::OnceLock::new();
38    FINGERPRINT
39        .get_or_init(|| {
40            fingerprint_with_revision(
41                &detcore_model::config::config_wire_fingerprint(),
42                crate::tool_global::GLOBAL_RPC_REVISION,
43            )
44        })
45        .clone()
46}
47
48/// What differs when a coordinator announced `expected` as its
49/// [`config_wire_fingerprint`] and this build's is another, or `None` when
50/// they are equal. The runtimes report this before they decode the
51/// coordinator's configuration (<https://github.com/rrnewton/hermit/issues/3986>).
52pub fn config_fingerprint_mismatch(expected: &std::ffi::OsStr) -> Option<String> {
53    let ours = config_wire_fingerprint();
54    (expected.to_str() != Some(ours.as_str())).then(|| {
55        format!(
56            "its configuration and clock RPC fingerprint is {ours}, Hermit's is {}",
57            expected.to_string_lossy()
58        )
59    })
60}

62/// FNV-1a over `bytes`, each part length-prefixed in a domain of its own,
63/// as `detcore_model`'s fingerprint hashes its material. A mismatch
64/// detector, not a security boundary.
65fn fnv_domains(parts: &[(u8, &[u8])]) -> String {
66    let mut hash: u64 = 0xcbf2_9ce4_8422_2325;
67    for (domain, bytes) in parts {
68        for byte in std::iter::once(domain)
69            .chain((bytes.len() as u64).to_le_bytes().iter())
70            .chain(bytes.iter())
71        {
72            hash ^= u64::from(*byte);
73            hash = hash.wrapping_mul(0x1000_0000_01b3);
74        }
75    }
76    format!("{hash:016x}")
77}
78
79/// The fingerprint of configuration fingerprint `base` at global RPC
80/// revision `revision`.
81fn fingerprint_with_revision(base: &str, revision: u32) -> String {
82    fnv_domains(&[(0, base.as_bytes()), (1, &revision.to_le_bytes())])
83}

145    /// The lines of code, without comments or blank lines, of the type that
146    /// `source` declares on the line `header`, through its closing brace.
147    fn definition(source: &str, header: &str) -> Vec<String> {
148        let mut lines = source.lines().skip_while(|line| *line != header);
149        assert_eq!(lines.next(), Some(header), "{header} is declared");
150        let mut code: Vec<String> = lines
151            .take_while(|line| *line != "}")
152            .map(str::trim)
153            .filter(|line| !line.is_empty() && !line.starts_with("//"))
154            .map(str::to_owned)
155            .collect();
156        code.insert(0, header.to_owned());
157        code
158    }

160    /// A digest of the code of `GlobalRequest` and `GlobalResponse` in
161    /// `source`: their variants, payload types, order and attributes, not
162    /// their documentation.
163    fn definition_digest(source: &str) -> String {
164        let request = definition(source, "pub enum GlobalRequest {").join("\n");
165        let response = definition(source, "pub enum GlobalResponse {").join("\n");
166        fnv_domains(&[(0, request.as_bytes()), (1, response.as_bytes())])
167    }
168
169    fn resources_definition_digest(source: &str) -> String {
170        let resources = definition(source, "pub struct Resources {").join("\n");
171        fnv_domains(&[(2, resources.as_bytes())])
172    }

174    /// The global RPC's revision names its enums and Resources as they are:
175    /// a variant or field added, removed, reordered or retyped changes the digest,
176    /// and the test then
177    /// fails until a new revision is recorded, which changes the fingerprint
178    /// every runtime compares with its coordinator's.
179    #[test]
180    fn the_global_rpc_revision_matches_its_definitions() {
181        let source = include_str!("tool_global.rs");
182        let digest = definition_digest(source);
183        let resources = include_str!("resources.rs");
184        let resources_digest = resources_definition_digest(resources);
185        let revision = crate::tool_global::GLOBAL_RPC_REVISION;
186        let next = GLOBAL_RPC_REVISIONS
187            .last()
188            .map_or(1, |(last, _, _)| last + 1);
189        assert_eq!(
190            GLOBAL_RPC_REVISIONS.last(),
191            Some(&(revision, digest.as_str(), resources_digest.as_str())),
192            "GlobalRequest, GlobalResponse or Resources changed: append ({next}, \
193             \"{digest}\", \"{resources_digest}\") to GLOBAL_RPC_REVISIONS and set \
194             GLOBAL_RPC_REVISION to {next}, so \
195             that runtimes and coordinators from either side of the change refuse each other"
196        );
197        for pair in GLOBAL_RPC_REVISIONS.windows(2) {
198            assert_eq!(pair[1].0, pair[0].0 + 1, "revisions count up by one");
199            assert_ne!(
200                (pair[1].1, pair[1].2),
201                (pair[0].1, pair[0].2),
202                "each revision changed the wire definitions"
203            );
204        }
205
206        // What the digest sees, and what it does not.
207        let documented = source.replacen(
208            "pub enum GlobalRequest {\n",
209            "pub enum GlobalRequest {\n    /// Documentation only.\n\n",
210            1,
211        );
212        assert_eq!(definition_digest(&documented), digest);
213        let added = source.replacen(
214            "pub enum GlobalRequest {\n",
215            "pub enum GlobalRequest {\n    Added(u32),\n",
216            1,
217        );
218        assert_ne!(definition_digest(&added), digest);
219        // Retype GlobalRequest's VirtualInotifyCookies wherever it sits: later
220        // revisions append variants after it.
221        let retyped = source.replacen(
222            "    VirtualInotifyCookies(Vec<u32>),\n",
223            "    VirtualInotifyCookies(Vec<u64>),\n",
224            1,
225        );
226        assert_ne!(retyped, source);
227        assert_ne!(definition_digest(&retyped), digest);
228
229        let documented = resources.replacen(
230            "pub struct Resources {\n",
231            "pub struct Resources {\n    /// Documentation only.\n\n",
232            1,
233        );
234        assert_ne!(documented, resources);
235        assert_eq!(resources_definition_digest(&documented), resources_digest);
236        let removed = resources.replacen("    pub(crate) priority_row: Option<u64>,\n", "", 1);
237        assert_ne!(removed, resources);
238        assert_ne!(resources_definition_digest(&removed), resources_digest);
239        let retyped =
240            resources.replacen("priority_row: Option<u64>", "priority_row: Option<u32>", 1);
241        assert_ne!(retyped, resources);
242        assert_ne!(resources_definition_digest(&retyped), resources_digest);
243    }

245    /// The published fingerprint is the configuration's extended with the
246    /// revision, so another revision is a mismatch, named with both values.
247    #[test]
248    fn the_fingerprint_carries_the_global_rpc_revision() {
249        let model = detcore_model::config::config_wire_fingerprint();
250        let ours = config_wire_fingerprint();
251        let revision = crate::tool_global::GLOBAL_RPC_REVISION;
252        assert_eq!(ours.len(), 16);
253        assert_eq!(ours, fingerprint_with_revision(&model, revision));
254        assert_ne!(ours, model);
255        let other = fingerprint_with_revision(&model, revision + 1);
256        assert_ne!(other, ours);
257        assert_eq!(
258            config_fingerprint_mismatch(std::ffi::OsStr::new(&ours)),
259            None
260        );
261        let mismatch = config_fingerprint_mismatch(std::ffi::OsStr::new(&other))
262            .expect("another revision is a mismatch");
263        assert!(
264            mismatch.contains(&ours) && mismatch.contains(&other),
265            "{mismatch}"
266        );
267    }
