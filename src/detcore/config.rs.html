14/// [`detcore_model::config::config_wire_fingerprint`], extended with the
15/// revision of Detcore's global RPC (`crate::tool_global::GLOBAL_RPC_REVISION`).
16///
17/// A runtime that embeds Detcore (SaBRe's plugin, the LiteInst runtime that
18/// in-guest-trap also loads, DBT's) sends `GlobalRequest`s to its coordinator
19/// as positional bincode and decodes the `GlobalResponse`s. Both enums live
20/// in this crate, outside the sources the configuration's fingerprint hashes,
21/// so a variant one build had and the other lacked went unnoticed: a newer
22/// runtime's request failed to decode in the middle of a run, and an older
23/// runtime never sent it (and so, for one, kept the host's inotify rename
24/// cookies). Every such runtime compares this fingerprint with its
25/// coordinator's before it decodes anything, so the pair is refused by name.
26/// The revision stands for the enums' source, which, unlike the
27/// configuration's, is not embedded: these runtimes are loaded into the guest,
28/// where every byte they grow by moves the guest's own mappings.
29pub fn config_wire_fingerprint() -> String {
30    // The material is immutable for this loaded image: the model overrides
31    // environment-dependent defaults, and the RPC revision is compiled in.
32    // Reuse it when a forked in-guest tool reconnects on the guest's clone
33    // stack; rebuilding Clap's canonical Config there can exhaust that stack.
34    // SaBRe's parent compatibility check finishes before an intercepted clone
35    // can run, so its child inherits an initialized value. Each connection
36    // still compares the coordinator's fingerprint with this local value.
37    static FINGERPRINT: std::sync::OnceLock<String> = std::sync::OnceLock::new();
38    FINGERPRINT
39        .get_or_init(|| {
40            fingerprint_with_revision(
41                &detcore_model::config::config_wire_fingerprint(),
42                crate::tool_global::GLOBAL_RPC_REVISION,
43            )
44        })
45        .clone()
46}

48/// What differs when a coordinator announced `expected` as its
49/// [`config_wire_fingerprint`] and this build's is another, or `None` when
50/// they are equal. The runtimes report this before they decode the
51/// coordinator's configuration (<https://github.com/rrnewton/hermit/issues/3986>).
52pub fn config_fingerprint_mismatch(expected: &std::ffi::OsStr) -> Option<String> {
53    let ours = config_wire_fingerprint();
54    (expected.to_str() != Some(ours.as_str())).then(|| {
55        format!(
56            "its configuration and clock RPC fingerprint is {ours}, Hermit's is {}",
57            expected.to_string_lossy()
58        )
59    })
60}

62/// FNV-1a over `bytes`, each part length-prefixed in a domain of its own,
63/// as `detcore_model`'s fingerprint hashes its material. A mismatch
64/// detector, not a security boundary.
65fn fnv_domains(parts: &[(u8, &[u8])]) -> String {
66    let mut hash: u64 = 0xcbf2_9ce4_8422_2325;
67    for (domain, bytes) in parts {
68        for byte in std::iter::once(domain)
69            .chain((bytes.len() as u64).to_le_bytes().iter())
70            .chain(bytes.iter())
71        {
72            hash ^= u64::from(*byte);
73            hash = hash.wrapping_mul(0x1000_0000_01b3);
74        }
75    }
76    format!("{hash:016x}")
77}
78
79/// The fingerprint of configuration fingerprint `base` at global RPC
80/// revision `revision`.
81fn fingerprint_with_revision(base: &str, revision: u32) -> String {
82    fnv_domains(&[(0, base.as_bytes()), (1, &revision.to_le_bytes())])
83}

459    /// The lines of code, without comments or blank lines, of the type that
460    /// `source` declares on the line `header`, through its closing brace.
461    fn definition(source: &str, header: &str) -> Vec<String> {
462        let mut lines = source.lines().skip_while(|line| *line != header);
463        assert_eq!(lines.next(), Some(header), "{header} is declared");
464        let mut code: Vec<String> = lines
465            .take_while(|line| *line != "}")
466            .map(str::trim)
467            .filter(|line| !line.is_empty() && !line.starts_with("//"))
468            .map(str::to_owned)
469            .collect();
470        code.insert(0, header.to_owned());
471        code
472    }

485    /// The lines of code, as [`definition`] gives them, of the type that
486    /// `source` declares, indented, on the line `header`, through the brace
487    /// at that indentation that closes it.
488    fn nested_definition(source: &str, header: &str) -> Vec<String> {
489        let mut lines = source.lines().skip_while(|line| line.trim() != header);
490        let first = lines
491            .next()
492            .unwrap_or_else(|| panic!("{header} is declared"));
493        let close = format!("{}}}", &first[..first.len() - first.trim_start().len()]);
494        let mut code: Vec<String> = lines
495            .take_while(|line| *line != close)
496            .map(str::trim)
497            .filter(|line| !line.is_empty() && !line.starts_with("//"))
498            .map(str::to_owned)
499            .collect();
500        code.insert(0, header.to_owned());
501        code
502    }

548    /// The lines of code of the item `source` declares, unindented, on the
549    /// line `header`: the attributes just above it (its derives and serde
550    /// attributes decide its encoding), the header, and its body through
551    /// the closing brace without comments or blank lines.
552    fn wire_item(source: &str, header: &str) -> Vec<String> {
553        let lines: Vec<&str> = source.lines().collect();
554        let at = lines
555            .iter()
556            .position(|line| *line == header)
557            .unwrap_or_else(|| panic!("{header} is declared"));
558        let attributes = lines[..at]
559            .iter()
560            .rev()
561            .take_while(|line| line.starts_with("#["))
562            .count();
563        let mut code: Vec<String> = lines[at - attributes..=at]
564            .iter()
565            .map(|line| (*line).to_owned())
566            .collect();
567        code.extend(
568            lines[at + 1..]
569                .iter()
570                .take_while(|line| **line != "}")
571                .map(|line| line.trim())
572                .filter(|line| !line.is_empty() && !line.starts_with("//"))
573                .map(str::to_owned),
574        );
575        code
576    }

578    /// The wire items `headers` of `source`, one after another.
579    fn wire_items(source: &str, headers: &[&str]) -> String {
580        headers
581            .iter()
582            .map(|header| wire_item(source, header).join("\n"))
583            .collect::<Vec<_>>()
584            .join("\n")
585    }

587    /// A digest of the code of `GlobalRequest` and `GlobalResponse` in
588    /// `source`, of the StoreExclusive payload's types in `payload`, of the
589    /// scheduled event's wire items in `schedule`, of the trace reply's
590    /// wire items in `source` and `replayer`, of the nested wire types
591    /// in `host_input` (`HOST_INPUT_WIRE`), of 2b-1's payload types in
592    /// `source` and `task_stat`, of 2b-3's in `source`, and of the SIGALRM
593    /// control in `source` (`SIGALRM_CONTROL_WIRE`): their variants, fields,
594    /// payload types, order and attributes, not their documentation.
595    fn definition_digest(
596        source: &str,
597        payload: &str,
598        schedule: &str,
599        replayer: &str,
600        host_input: &str,
601        task_stat: &str,
602    ) -> String {
603        let request = definition(source, "pub enum GlobalRequest {").join("\n");
604        let response = definition(source, "pub enum GlobalResponse {").join("\n");
605        let payload = STORE_EXCLUSIVE_PAYLOAD
606            .iter()
607            .map(|header| nested_definition(payload, header).join("\n"))
608            .collect::<Vec<_>>()
609            .join("\n");
610        let event = wire_items(schedule, &SCHED_EVENT_WIRE);
611        let reply = [
612            wire_items(source, &TRACE_REPLY_WIRE),
613            wire_items(replayer, &REPLAYED_SLICE_END_WIRE),
614        ]
615        .join("\n");
616        let nested = wire_items(host_input, &HOST_INPUT_WIRE);
617        let procfs = [
618            wire_items(source, &PROCFS_PAYLOAD_WIRE),
619            wire_items(task_stat, &TASK_CPU_TIMES_WIRE),
620        ]
621        .join("\n");
622        let exec_strings = wire_items(source, &EXEC_STRINGS_WIRE);
623        let sigalrm_control = wire_items(source, &SIGALRM_CONTROL_WIRE);
624        fnv_domains(&[
625            (0, request.as_bytes()),
626            (1, response.as_bytes()),
627            (3, payload.as_bytes()),
628            (4, event.as_bytes()),
629            (5, reply.as_bytes()),
630            (6, nested.as_bytes()),
631            (7, procfs.as_bytes()),
632            (8, exec_strings.as_bytes()),
633            (10, sigalrm_control.as_bytes()),
634        ])
635    }

1095    /// The published fingerprint is the configuration's extended with the
1096    /// revision, so another revision is a mismatch, named with both values.
1097    #[test]
1098    fn the_fingerprint_carries_the_global_rpc_revision() {
1099        let model = detcore_model::config::config_wire_fingerprint();
1100        let ours = config_wire_fingerprint();
1101        let revision = crate::tool_global::GLOBAL_RPC_REVISION;
1102        assert_eq!(ours.len(), 16);
1103        assert_eq!(ours, fingerprint_with_revision(&model, revision));
1104        assert_ne!(ours, model);
1105        let other = fingerprint_with_revision(&model, revision + 1);
1106        assert_ne!(other, ours);
1107        // A build at the previous revision, from before the newest wire
1108        // change, is refused too.
1109        let previous = fingerprint_with_revision(&model, revision - 1);
1110        assert!(
1111            config_fingerprint_mismatch(std::ffi::OsStr::new(&previous)).is_some(),
1112            "the previous revision is a mismatch"
1113        );
1114        assert_eq!(
1115            config_fingerprint_mismatch(std::ffi::OsStr::new(&ours)),
1116            None
1117        );
1118        let mismatch = config_fingerprint_mismatch(std::ffi::OsStr::new(&other))
1119            .expect("another revision is a mismatch");
1120        assert!(
1121            mismatch.contains(&ours) && mismatch.contains(&other),
1122            "{mismatch}"
1123        );
1124    }

1143                .unwrap_or_else(|| panic!("a revision-{revision} runtime is refused"));
