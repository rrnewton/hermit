9//! Handing the turn to a thread polling a pipe or FIFO once another thread's
10//! action has readied it (`--pipe-wake-latency`,
11//! <https://github.com/rrnewton/hermit/issues/4380>).
12//!
13//! A polled read of a scheduler-managed pipe retries in polling turns that
14//! back off in priority. A write that gives it data takes no turn, so without
15//! this the reader waits until the writer's slice ends (200 ms of virtual
16//! time by default), or for ever where no slice end is set. With the option
17//! on:
18//!
19//! 1. A poll names the channel it waits on and what it wants
20//!    (`Resources::polls_channel`).
21//! 2. An action that can satisfy such a poll, checked on each attempt's own
22//!    result while the acting thread holds the grant, reports the channel to
23//!    the scheduler (`ReadyChannel`), which restores the priority of every
24//!    queued poller of that channel and want, skipping doomed ones, and
25//!    remembers them for the actor.
26//! 3. Once the actor's own time is the latency past its first readying
27//!    action, at a syscall boundary, the scheduler moves the actor behind
28//!    those pollers (`HandOff`) and the actor yields, so they run first.
29//!
30//! Every input is guest state read in committed turns, so the hand-offs
31//! happen at the same turns in every run and in a replay made with the same
32//! option. With the option unset none of this runs: no poll names a channel,
33//! no description carries one, and no RPC is sent.
34
35use detcore_model::fd::OpenFileId;
36use serde::Deserialize;
37use serde::Serialize;
38
39/// The pipe or FIFO a descriptor is an end of.
40#[derive(
41    Debug,
42    Clone,
43    Copy,
44    PartialEq,
45    Eq,
46    Hash,
47    PartialOrd,
48    Ord,
49    Serialize,
50    Deserialize
51)]
52pub enum ChannelId {
53    /// A pipe made by `pipe2`, named by its read end's open file description,
54    /// which `pipe2` stores on both ends. Carried across a rediscovering exec
55    /// (`ExecFdState::channels`).
56    Pipe(OpenFileId),
57    /// A FIFO, where the guest's descriptors are visible to the tool
58    /// (`fifo_peer::guest_descriptors_are_visible`), named by the ordinal
59    /// `fifo_peer::fifo_channel` gives its host `(dev, ino)`: the host numbers
60    /// differ between runs, and polls log their channel.
61    Fifo(u64),
62}
63
64/// One end of a channel, as an open file description holds it.
65#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize)]
66pub struct ChannelEnd {
67    /// The pipe or FIFO.
68    pub channel: ChannelId,
69    /// Whether this end reads (a pipe's read end, a FIFO opened for reading
70    /// or for both).
71    pub reads: bool,
72    /// Whether this end writes.
73    pub writes: bool,
74}
75
76impl ChannelEnd {
77    /// What closing this end readies: the readers of the channel (end of
78    /// file) when it writes, and its writers (`EPIPE`) when it reads. It
79    /// need not be the last end; a false hand-off changes no result.
80    pub fn close_readies(&self) -> impl Iterator<Item = Wants> {
81        [(self.writes, Wants::Data), (self.reads, Wants::Space)]
82            .into_iter()
83            .filter_map(|(applies, wants)| applies.then_some(wants))
84    }
85}
86
87/// What a poll of a channel waits for, and so which action readies it.
88#[derive(
89    Debug,
90    Clone,
91    Copy,
92    PartialEq,
93    Eq,
94    Hash,
95    PartialOrd,
96    Ord,
97    Serialize,
98    Deserialize
99)]
100pub enum Wants {
101    /// A read: data, or the end of file a write end's close gives.
102    Data,
103    /// A write: room, or the `EPIPE` a read end's close gives.
104    Space,
105    /// A blocking FIFO open for reading, waiting for a writer.
106    PeerWriter,
107    /// A blocking FIFO open for writing, waiting for a reader.
108    PeerReader,
109}
110
111/// What the scheduler answers a readying report (`Scheduler::ready_channel`).
112#[derive(Debug, Clone, Copy, PartialEq, Eq, Default, Serialize, Deserialize)]
113pub struct ReadyOutcome {
114    /// The acting thread now has pollers to hand the turn to.
115    pub has_pending: bool,
116    /// This report found the first of them since the acting thread's last
117    /// committed turn, so its latency starts now. The scheduler forgets a
118    /// thread's pollers at every committed turn of its own, whichever request
119    /// that turn granted (a futex wait as well as a resource request), so the
120    /// window is the scheduler's, and the thread keeps no copy of it.
121    pub starts_window: bool,
122}
123
124/// The descriptor a polled `call` waits on, and what for: a read waits for
125/// data, a write for room.
126pub(crate) fn poll_of(
127    call: &reverie::syscalls::Syscall,
128) -> Option<(std::os::unix::io::RawFd, Wants)> {
129    progress_of(call).map(|(fd, readies)| {
130        (
131            fd,
132            match readies {
133                Wants::Data => Wants::Space,
134                _ => Wants::Data,
135            },
136        )
137    })
138}
139
140/// The descriptor a successful `call` moved bytes through, and what that
141/// readies on its channel: a write gives the readers data, a read gives the
142/// writers room.
143pub(crate) fn progress_of(
144    call: &reverie::syscalls::Syscall,
145) -> Option<(std::os::unix::io::RawFd, Wants)> {
146    use reverie::syscalls::Syscall;
147    match call {
148        Syscall::Write(write) => Some((write.fd(), Wants::Data)),
149        Syscall::Writev(writev) => Some((writev.fd(), Wants::Data)),
150        // Only at the current position can these move bytes through a pipe
151        // (an explicit offset fails with ESPIPE there), where Detcore runs
152        // them as `writev` and `readv`.
153        Syscall::Pwritev2(pwritev2) => Some((pwritev2.fd(), Wants::Data)),
154        Syscall::Read(read) => Some((read.fd(), Wants::Space)),
155        Syscall::Readv(readv) => Some((readv.fd(), Wants::Space)),
156        Syscall::Preadv2(preadv2) => Some((preadv2.fd(), Wants::Space)),
157        _ => None,
158    }
159}

161impl<T: crate::record_or_replay::RecordOrReplay> crate::Detcore<T> {
162    /// The channel end behind the guest descriptor `fd`, when
163    /// `--pipe-wake-latency` is in effect and Detcore knows it.
164    pub(crate) fn pipe_channel_end<G: reverie::Guest<Self>>(
165        &self,
166        guest: &G,
167        fd: std::os::unix::io::RawFd,
168    ) -> Option<ChannelEnd> {
169        self.cfg.pipe_wake_latency_in_effect()?;
170        guest
171            .thread_state()
172            .with_detfd(fd, |detfd| detfd.channel())
173            .ok()
174            .flatten()
175    }
176
177    /// An action of this thread may satisfy the pollers of `channel` that
178    /// want `wants`: report it to the scheduler, and if the report starts a
179    /// window of pollers to hand off to, start the latency toward this
180    /// thread's hand-off.
181    pub(crate) async fn note_pipe_readied<G: reverie::Guest<Self>>(
182        &self,
183        guest: &mut G,
184        channel: ChannelId,
185        wants: Wants,
186    ) {
187        let Some(latency) = self.cfg.pipe_wake_latency_in_effect() else {
188            return;
189        };
190        let outcome = crate::tool_global::ready_channel(guest, channel, wants).await;
191        if outcome.has_pending
192            && (outcome.starts_window || guest.thread_state().pipe_handoff_due.is_none())
193        {
194            let due = guest.thread_state().thread_logical_time.as_nanos()
195                + std::time::Duration::from_nanos(latency);
196            guest.thread_state_mut().pipe_handoff_due = Some(due);
197        }
198    }
199
200    /// [`Self::note_pipe_readied`] for an action on the guest descriptor
201    /// `fd`, if it is an end of a known channel.
202    pub(crate) async fn note_fd_readied<G: reverie::Guest<Self>>(
203        &self,
204        guest: &mut G,
205        fd: std::os::unix::io::RawFd,
206        wants: Wants,
207    ) {
208        if let Some(end) = self.pipe_channel_end(guest, fd) {
209            self.note_pipe_readied(guest, end.channel, wants).await;
210        }
211    }
212
213    /// At a syscall boundary: if this thread's hand-off is due, let the
214    /// scheduler move it behind the pollers it readied, and yield to them.
215    /// Deferred while the thread holds its turn for a signal to its own
216    /// process (`own_signal_turn_hold`). The yield is a turn of its own,
217    /// outside the timeslice bookkeeping, as the futex wake's is.
218    pub(crate) async fn pipe_hand_off_if_due<G: reverie::Guest<Self>>(&self, guest: &mut G) {
219        let Some(due) = guest.thread_state().pipe_handoff_due else {
220            return;
221        };
222        if guest.thread_state().own_signal_turn_hold.is_some()
223            || guest.thread_state().thread_logical_time.as_nanos() < due
224        {
225            return;
226        }
227        guest.thread_state_mut().pipe_handoff_due = None;
228        if crate::tool_global::hand_off_turn(guest).await {
229            tracing::trace!(
230                "[dtid {}] handing the turn to the pollers it readied",
231                guest.thread_state().dettid
232            );
233            let request = Self::yield_request(guest);
234            crate::tool_global::resource_request(guest, request).await;
235        }
236    }
237
238    /// The channel ends among the guest descriptors `first..=last`, in
239    /// descriptor order, when `--pipe-wake-latency` is in effect.
240    pub(crate) fn pipe_channel_ends_in<G: reverie::Guest<Self>>(
241        &self,
242        guest: &G,
243        first: u32,
244        last: u32,
245    ) -> Vec<ChannelEnd> {
246        if self.cfg.pipe_wake_latency_in_effect().is_none() || first > last {
247            return Vec::new();
248        }
249        guest
250            .thread_state()
251            .open_fds()
252            .into_iter()
253            .filter(|fd| u32::try_from(*fd).is_ok_and(|fd| (first..=last).contains(&fd)))
254            .filter_map(|fd| self.pipe_channel_end(guest, fd))
255            .collect()
256    }
257
258    /// This thread closed `end` (explicitly: `close`, `close_range`, or a
259    /// `dup2` or `dup3` over it): ready what that closing gives the other
260    /// side. An implicit close at exit or exec readies nothing.
261    pub(crate) async fn note_channel_end_closed<G: reverie::Guest<Self>>(
262        &self,
263        guest: &mut G,
264        end: Option<ChannelEnd>,
265    ) {
266        let Some(end) = end else {
267            return;
268        };
269        for wants in end.close_readies() {
270            self.note_pipe_readied(guest, end.channel, wants).await;
271        }
272    }
