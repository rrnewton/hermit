9//! Actual callback identity. Expectations cannot create tokens. A syscall's
10//! token lives in its handler future, so nested callbacks cannot overwrite the
11//! outer syscall's occurrence. An early-returned handler simply drops its
12//! token; there is no pending-map entry to leak or reuse.
13

26#[derive(Clone, Copy, Debug, Eq, PartialEq, Hash, Serialize, Deserialize)]
27pub(crate) enum Callback {
28    Syscall(Sysno),
29    Cpuid,
30    Rdtsc,
31    PmuTimer,
32    NoRcbTimer,
33    ThreadStart,
34    Exec,
35    Signal { signal: SigWrapper, announced: bool },
36}
37
38/// An owned token, deliberately not Copy. Only `begin` creates one; prehook
39/// publication mutates it and posthook publication consumes it. Serialization
40/// is for the enclosing thread state, whose native-cut path already refuses
41/// active boundary observations. It is not a public observation input.
42#[derive(Clone, Debug, Eq, PartialEq, Serialize, Deserialize)]
43#[serde(deny_unknown_fields)]
44pub(crate) struct Token {
45    owner: Owner,
46    entry: Entry,
47    prehook_seen: bool,
48}
49
50impl Observations {
51    fn begin_callback<T>(
52        &mut self,
53        thread: &ThreadState<T>,
54        callback: Callback,
55    ) -> anyhow::Result<Token> {
56        self.require_owner(thread)?;
57        let entry = match callback {
58            Callback::Syscall(syscall) => {
59                let occurrence = self.syscall_occurrence.checked_add(1).ok_or_else(|| {
60                    anyhow::anyhow!("slice-boundary protocol refusal: syscall occurrence overflow")
61                })?;
62                self.syscall_occurrence = occurrence;
63                Entry::Syscall {
64                    syscall,
65                    occurrence,
66                }
67            }
68            Callback::Cpuid => Entry::Cpuid,
69            Callback::Rdtsc => Entry::Rdtsc,
70            Callback::PmuTimer => Entry::PmuTimer,
71            Callback::NoRcbTimer => Entry::NoRcbTimer,
72            Callback::ThreadStart => Entry::ThreadStart,
73            Callback::Exec => Entry::Exec,
74            Callback::Signal { signal, announced } => {
75                if !(1..=64).contains(&signal.raw()) {
76                    bail!("slice-boundary protocol refusal: invalid actual signal entry");
77                }
78                let occurrence = self.signal_entry_occurrence.checked_add(1).ok_or_else(|| {
79                    anyhow::anyhow!(
80                        "slice-boundary protocol refusal: signal entry occurrence overflow"
81                    )
82                })?;
83                self.signal_entry_occurrence = occurrence;
84                Entry::SignalEntry {
85                    signal,
86                    occurrence,
87                    announced,
88                }
89            }
90        };
91        Ok(Token {
92            owner: self.owner,
93            entry,
94            prehook_seen: false,
95        })
96    }
97
98    fn publish_callback<T>(
99        &mut self,
100        thread: &ThreadState<T>,
101        token: &mut Token,
102        phase: Phase,
103    ) -> anyhow::Result<()> {
104        if token.owner != self.owner {
105            bail!("slice-boundary protocol refusal: callback token belongs to another owner");
106        }
107        if token.prehook_seen != (phase == Phase::Posthook) {
108            bail!("slice-boundary protocol refusal: callback phase is repeated or out of order");
109        }
110        if phase == Phase::Posthook && matches!(token.entry, Entry::SignalEntry { .. }) {
111            bail!("slice-boundary protocol refusal: signal posthook precedes actual delivery");
112        }
113        self.publish(thread, token.entry, phase)?;
114        token.prehook_seen = true;
115        Ok(())
116    }
117}

119fn with_observations<T, R>(
120    thread: &mut ThreadState<T>,
121    action: impl FnOnce(&mut Observations, &ThreadState<T>) -> anyhow::Result<R>,
122) -> anyhow::Result<Option<R>> {
123    let Some(mut observed) = thread.slice_boundary.take() else {
124        return Ok(None);
125    };
126    let answer = action(&mut observed, thread);
127    thread.slice_boundary = Some(observed);
128    answer.map(Some)
129}
130
131pub(crate) fn begin<T>(
132    thread: &mut ThreadState<T>,
133    callback: Option<Callback>,
134) -> anyhow::Result<Option<Token>> {
135    with_observations(thread, |observed, thread| {
136        let callback = callback.ok_or_else(|| {
137            anyhow::anyhow!(
138                "slice-boundary protocol refusal: callback has no actual semantic identity"
139            )
140        })?;
141        observed.begin_callback(thread, callback)
142    })
143}
144
145pub(crate) fn pre<T>(thread: &mut ThreadState<T>, token: &mut Option<Token>) -> anyhow::Result<()> {
146    with_observations(thread, |observed, thread| {
147        let token = token.as_mut().ok_or_else(|| {
148            anyhow::anyhow!("slice-boundary protocol refusal: prehook lost its callback token")
149        })?;
150        observed.publish_callback(thread, token, Phase::Prehook)
151    })?;
152    Ok(())
153}
154
155pub(crate) fn post<T>(thread: &mut ThreadState<T>, mut token: Option<Token>) -> anyhow::Result<()> {
156    with_observations(thread, |observed, thread| {
157        let token = token.as_mut().ok_or_else(|| {
158            anyhow::anyhow!("slice-boundary protocol refusal: posthook lost its callback token")
159        })?;
160        observed.publish_callback(thread, token, Phase::Posthook)
161    })?;
162    Ok(())
163}

165/// Link this already-entered callback to the actual delivery ordinal chosen
166/// by the unchanged signal code. No entry/check-in/turn is allocated here.
167/// An announcement withdrawn before delivery simply drops its retained token;
168/// its earlier semantic check-in and any completed boundary remain observed.
169pub(crate) fn delivered_signal(
170    token: &mut Option<Token>,
171    signal: SigWrapper,
172    zero_based_delivery: u64,
173) -> anyhow::Result<()> {
174    let Some(token) = token else { return Ok(()) };
175    let Entry::SignalEntry {
176        signal: entered,
177        occurrence,
178        announced,
179    } = token.entry
180    else {
181        bail!("slice-boundary protocol refusal: delivery has no unsettled signal entry");
182    };
183    if !token.prehook_seen || entered != signal {
184        bail!("slice-boundary protocol refusal: delivery disagrees with its actual entry");
185    }
186    let delivery = zero_based_delivery.checked_add(1).ok_or_else(|| {
187        anyhow::anyhow!("slice-boundary protocol refusal: signal delivery ordinal overflow")
188    })?;
189    token.entry = Entry::Signal {
190        signal,
191        occurrence,
192        announced,
193        delivery,
194    };
195    Ok(())
196}

211    #[test]
212    fn nested_syscall_tokens_keep_the_outer_actual_occurrence() {
213        let mut thread = thread();
214        let mut outer = begin(&mut thread, Some(Callback::Syscall(Sysno::read))).unwrap();
215        pre(&mut thread, &mut outer).unwrap();
216        let mut inner = begin(&mut thread, Some(Callback::Syscall(Sysno::getpid))).unwrap();
217        pre(&mut thread, &mut inner).unwrap();
218        post(&mut thread, inner).unwrap();
219        post(&mut thread, outer).unwrap();
220        let observed = thread.slice_boundary.as_ref().unwrap();
221        assert_eq!(
222            observed.current.unwrap().0,
223            Entry::Syscall {
224                syscall: Sysno::read,
225                occurrence: 1
226            }
227        );
228        assert_eq!(observed.checkin, 4);
229        assert_eq!(observed.syscall_occurrence, 2);
230        // An early return does not prevent the next same-site invocation.
231        let _abandoned = begin(&mut thread, Some(Callback::Syscall(Sysno::read))).unwrap();
232        let next = begin(&mut thread, Some(Callback::Syscall(Sysno::read)))
233            .unwrap()
234            .unwrap();
235        assert_eq!(
236            next.entry,
237            Entry::Syscall {
238                syscall: Sysno::read,
239                occurrence: 4
240            }
241        );
242    }

244    #[test]
245    fn wrong_phase_missing_identity_stale_owner_and_overflow_refuse_without_publish() {
246        let mut thread = thread();
247        assert!(begin(&mut thread, None).is_err());
248        let mut token = begin(&mut thread, Some(Callback::Cpuid)).unwrap();
249        assert!(post(&mut thread, token.clone()).is_err());
250        pre(&mut thread, &mut token).unwrap();
251        let before = thread.slice_boundary.clone();
252        assert!(pre(&mut thread, &mut token).is_err());
253        assert_eq!(thread.slice_boundary, before);
254        thread.mm_id = thread.mm_id.for_exec(thread.dettid);
255        assert!(post(&mut thread, token).is_err());
256        assert_eq!(thread.slice_boundary, before);
257        thread.slice_boundary = Some(Observations::new(&thread, 0));
258        thread.slice_boundary.as_mut().unwrap().syscall_occurrence = u64::MAX;
259        let before = thread.slice_boundary.clone();
260        assert!(begin(&mut thread, Some(Callback::Syscall(Sysno::getpid))).is_err());
261        assert_eq!(thread.slice_boundary, before);
262    }

264    #[test]
265    fn all_actual_callback_kinds_have_distinct_identity_and_legacy_stays_inert() {
266        let mut thread = thread();
267        for (callback, expected) in [
268            (Callback::Cpuid, Entry::Cpuid),
269            (Callback::Rdtsc, Entry::Rdtsc),
270            (Callback::PmuTimer, Entry::PmuTimer),
271            (Callback::NoRcbTimer, Entry::NoRcbTimer),
272            (Callback::ThreadStart, Entry::ThreadStart),
273            (Callback::Exec, Entry::Exec),
274        ] {
275            let mut token = begin(&mut thread, Some(callback)).unwrap();
276            pre(&mut thread, &mut token).unwrap();
277            assert_eq!(
278                thread.slice_boundary.as_ref().unwrap().current.unwrap().0,
279                expected
280            );
281            post(&mut thread, token).unwrap();
282        }
283        thread.slice_boundary = None;
284        assert!(begin(&mut thread, None).unwrap().is_none());
285        pre(&mut thread, &mut None).unwrap();
286        post(&mut thread, None).unwrap();

288    #[test]
289    fn actual_signal_entry_linkage_refuses_guessing_repetition_and_overflow() {
290        let mut thread = thread();
291        let signal = SigWrapper(libc::SIGUSR1);
292        let mut token = begin(
293            &mut thread,
294            Some(Callback::Signal {
295                signal,
296                announced: true,
297            }),
298        )
299        .unwrap();
300        assert!(delivered_signal(&mut token, signal, 0).is_err());
301        pre(&mut thread, &mut token).unwrap();
302        assert!(post(&mut thread, token.clone()).is_err());
303        let before = token.clone();
304        assert!(delivered_signal(&mut token, SigWrapper(libc::SIGUSR2), 0).is_err());
305        assert!(delivered_signal(&mut token, signal, u64::MAX).is_err());
306        assert_eq!(token, before);
307        let state = thread.slice_boundary.clone();
308        delivered_signal(&mut token, signal, 7).unwrap();
309        assert_eq!(
310            thread.slice_boundary, state,
311            "linking does not observe a second entry"
312        );
313        assert!(delivered_signal(&mut token, signal, 7).is_err());
314        post(&mut thread, token).unwrap();
315        assert_eq!(
316            thread.slice_boundary.as_ref().unwrap().current.unwrap().0,
317            Entry::Signal {
318                signal,
319                occurrence: 1,
320                announced: true,
321                delivery: 8
322            }
323        );
324        thread
325            .slice_boundary
326            .as_mut()
327            .unwrap()
328            .signal_entry_occurrence = u64::MAX;
329        let before = thread.slice_boundary.clone();
330        assert!(
331            begin(
332                &mut thread,
333                Some(Callback::Signal {
334                    signal,
335                    announced: false
336                })
337            )
338            .is_err()
339        );
340        assert_eq!(thread.slice_boundary, before);
341    }
