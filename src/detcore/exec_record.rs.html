9//! The image a process executes, which its `/proc/<pid>/exe` link names.
10//!
11//! Detcore records it at each exec, on every backend, and checks it against
12//! the kernel's own link when the exec completes (`tool_global`). Where the
13//! backend's own procfs is not the guest's kernel procfs (KVM), the caller's
14//! own link (`/proc/self/exe` and its other spellings) is answered from the
15//! record while it still names exactly the image the process executes
16//! ([`ExecRecord::names_path_exactly`]; `Detcore::readlink_own_task_link`,
17//! package K slice E1); otherwise it is answered as before. Where the
18//! kernel's link names a launcher in the guest's place (SaBRe's loader,
19//! DynamoRIO's library; `Detcore::launched_exe`), the record answers every
20//! process's link while the image is at its path, as before. The kernel's
21//! link stands elsewhere.
22//!
23//! Only the path is recorded. Where the image ends, which a program-break
24//! model would need, is known only once the image is loaded, and nothing
25//! reads it yet.

38/// The bytes of a file the kernel reads to find its format
39/// (`BINPRM_BUF_SIZE`, include/linux/binfmts.h).
40const BINPRM_BUF_SIZE: usize = 256;

42/// The interpreter levels the kernel follows before it refuses an exec with
43/// `ELOOP` (`exec_binprm`'s `depth > 5`, fs/exec.c).
44const MAX_INTERPRETER_DEPTH: usize = 5;

46/// The image a process executes.
47#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
48pub struct ExecRecord {
49    /// The image's canonical path, as the process's root directory shows it:
50    /// what `readlink("/proc/self/exe")` returns while the file is there.
51    pub exe: PathBuf,
52    /// The image file's device and inode when it was exec'd, which tell
53    /// whether the file is still at `exe` ([`Self::is_present`]).
54    dev: u64,
55    ino: u64,
56    /// Its link count and inode change time then: a rename, link or unlink
57    /// of the file since changes both, even where the same inode is put back
58    /// at `exe`, which Linux's link would no longer name.
59    nlink: u64,
60    ctime: (i64, i64),
61    /// The device and inode of each directory above the image, from its
62    /// parent up to the process's root directory, when it was exec'd
63    /// ([`Self::names_path_exactly`]).
64    directories: Vec<(u64, u64)>,
65}
66
67/// The device and inode of each directory above `path`, a canonical path
68/// below `root`, from its parent up to `root`, not following a symbolic link
69/// at any of them, read with raw syscalls (in-guest Detcore must not call
70/// libc's wrappers: `crate::util::raw_syscall`). `None` where one cannot be
71/// read or is no directory ([`directory_sample`]).
72fn directory_chain(root: &Path, path: &Path) -> Option<Vec<(u64, u64)>> {
73    path.ancestors()
74        .skip(1)
75        .take_while(|directory| directory.starts_with(root))
76        .map(|directory| directory_sample(&crate::util::raw_lstat(directory).ok()?))
77        .collect()
78}
79
80/// The device and inode of a directory, from its unfollowed status: `None`
81/// for anything but a directory, so that another kind of file put in a
82/// directory's place never matches it, even where it reuses the freed
83/// directory's inode number (a symbolic link to where the directory went).
84fn directory_sample(stat: &libc::stat) -> Option<(u64, u64)> {
85    (stat.st_mode & libc::S_IFMT == libc::S_IFDIR).then_some((stat.st_dev, stat.st_ino))
86}

88impl ExecRecord {
89    /// The image an `execve` of `path` runs, as the kernel finds it: `path`
90    /// taken from the working directory `cwd`, every symbolic link followed,
91    /// and for a script the interpreter its `#!` line names, level by level,
92    /// as `/proc/<pid>/exe` names the interpreter. `root` is the process's
93    /// root directory as the caller sees it, and `cwd` is the working
94    /// directory below it, absolute.
95    ///
96    /// `None` where the kernel would refuse the exec (no such file, a
97    /// malformed `#!` line, too many interpreter levels), where the caller
98    /// cannot look the path up, and for a file the caller cannot read, which
99    /// the kernel may still run (an execute-only binary, or a script whose
100    /// `#!` line the kernel reads without a permission check): the link is
101    /// then the kernel's to answer. A resolution can still be wrong where
102    /// the kernel runs a file differently (a `binfmt_misc` handler) or the
103    /// caller sees another filesystem; `tool_global` checks every record
104    /// against the kernel's own link after the exec.
105    pub fn resolve(root: &Path, cwd: &Path, path: &Path) -> Option<Self> {
106        Self::resolve_from(root, Some(cwd), path)
107    }
108
109    /// [`Self::resolve`] for a caller whose working directory may be unknown
110    /// (`cwd` is `None`): then `None` wherever the kernel would look a
111    /// relative path up from it (a relative `path`, or a script whose `#!`
112    /// line names a relative interpreter).
113    pub fn resolve_from(root: &Path, cwd: Option<&Path>, path: &Path) -> Option<Self> {
114        let from_cwd = |path: &Path| {
115            if path.is_absolute() {
116                Some(path.to_path_buf())
117            } else {
118                Some(cwd?.join(path))
119            }
120        };
121        let mut spelled = from_cwd(path)?;
122        for _ in 0..=MAX_INTERPRETER_DEPTH {
123            let found = crate::util::canonicalize(&below(root, &spelled)?).ok()?;
124            // Linux execs only a regular file, and refuses anything else
125            // (EACCES) before it opens it.
126            if !fs::metadata(&found).ok()?.is_file() {
127                return None;
128            }
129            match script_interpreter(&found) {
130                Script::Not => return Self::at(root, &found),
131                // The kernel opens the interpreter from the caller's working
132                // directory, as it opened the script.
133                Script::Interpreter(interpreter) => spelled = from_cwd(&interpreter)?,
134                Script::Declined | Script::Unread => return None,
135            }
136        }
137        None
138    }
139
140    /// The record of the image at `found`, a canonical path the caller sees
141    /// below `root`, as that file is now.
142    pub fn at(root: &Path, found: &Path) -> Option<Self> {
143        let metadata = fs::metadata(found).ok()?;
144        Some(Self {
145            exe: Path::new("/").join(found.strip_prefix(root).ok()?),
146            dev: metadata.dev(),
147            ino: metadata.ino(),
148            nlink: metadata.nlink(),
149            ctime: (metadata.ctime(), metadata.ctime_nsec()),
150            directories: directory_chain(root, found).unwrap_or_default(),
151        })
152    }
153
154    /// The image file's device and inode when it was exec'd.
155    pub fn identity(&self) -> (u64, u64) {
156        (self.dev, self.ino)
157    }
158
159    /// Whether the image's file is still at its path, untouched, for a
160    /// process whose root directory the caller sees at `root`: the same
161    /// inode, with the link count and change time it had at the exec. A
162    /// rename or link of the image since, or a change of its mode, reads as
163    /// absent, so the caller refuses rather than name a path Linux's link
164    /// might not.
165    pub fn is_present(&self, root: &Path) -> bool {
166        below(root, &self.exe)
167            .and_then(|path| fs::symlink_metadata(path).ok())
168            .is_some_and(|now| {
169                (
170                    now.dev(),
171                    now.ino(),
172                    now.nlink(),
173                    (now.ctime(), now.ctime_nsec()),
174                ) == (self.dev, self.ino, self.nlink, self.ctime)
175            })
176    }
177
178    /// Whether the image's path still names exactly the file the process
179    /// executes, as Linux's `exe` link renders it (`d_path` of the image's
180    /// dentry), for a process whose root directory the caller sees at
181    /// `root`: the file is untouched at its path ([`Self::is_present`]), and
182    /// every directory above it is the one that was there at the exec (its
183    /// device and inode, not following a symbolic link). A directory has one
184    /// dentry, so the same directory inode at a path is that path's dentry,
185    /// and the same untouched file in the same directory under the same name
186    /// is the same dentry: a rename of the image or of any directory above
187    /// it, another directory or a symbolic link put at one of their paths,
188    /// and a hard link of the image put back at its path all read as not
189    /// exact. (Mounts are refused in the guest.)
190    pub fn names_path_exactly(&self, root: &Path) -> bool {
191        self.is_present(root)
192            && !self.directories.is_empty()
193            && below(root, &self.exe)
194                .and_then(|path| directory_chain(root, &path))
195                .is_some_and(|now| now == self.directories)
196    }
197}

199/// What decides where Linux starts a process's program break after it execs
200/// an ELF image (`load_elf_binary`, fs/binfmt_elf.c), from the image's file
201/// header and program headers.
202#[derive(Clone, Copy, Debug, PartialEq, Eq)]
203pub struct ElfLayout {
204    /// Whether the image is `ET_DYN` (position-independent) rather than
205    /// `ET_EXEC`.
206    pub position_independent: bool,
207    /// Whether the image names a program interpreter (`PT_INTERP`).
208    pub interpreter: bool,
209    /// The virtual address of the image's program headers before the load
210    /// bias, as Linux finds it for `AT_PHDR` (since Linux 5.18): `e_phoff` in
211    /// the last `PT_LOAD` segment whose file range holds it, or 0 where none
212    /// does (`AT_PHDR` is then the bias itself).
213    pub phdr: u64,
214    /// The end of the highest `PT_LOAD` segment in memory, `p_vaddr +
215    /// p_memsz`, before the load bias: the end of its bss.
216    pub end: u64,
217    /// The first executable `PT_LOAD` segment's file offset and virtual
218    /// address, each page-aligned down: where its text mapping starts in the
219    /// file and, before the load bias, in memory. `None` without one.
220    pub text: Option<(u64, u64)>,
221    /// `mm->end_data - mm->start_data` as Linux sets them: the highest
222    /// segment end in the file (`p_vaddr + p_filesz`) less the highest
223    /// segment start, which `RLIMIT_DATA` counts with the heap
224    /// (`check_data_rlimit`).
225    pub data_span: u64,
226    /// The page size Linux maps the segments with and aligns the program
227    /// break to: `ELF_MIN_ALIGN`, the larger of `ELF_EXEC_PAGESIZE` and
228    /// `PAGE_SIZE` (fs/binfmt_elf.c), which is the host's page size, since
229    /// `ELF_EXEC_PAGESIZE` is `PAGE_SIZE` on arm64 and 4096 on x86_64.
230    pub page: u64,
231}

233impl ElfLayout {
234    /// The layout of the ELF64 image at `path` (`ET_EXEC` or `ET_DYN`), or
235    /// `None` for any other file, one without a loadable segment, and one the
236    /// caller cannot read.
237    pub fn read(path: &Path) -> Option<Self> {
238        Self::read_file(&crate::util::RawFd::open(path, libc::O_RDONLY).ok()?)
239    }
240
241    /// [`Self::read`] of an open file, read with raw syscalls: in-guest
242    /// Detcore must not call libc's wrappers (`crate::util::raw_syscall`).
243    pub fn read_file(file: &crate::util::RawFd) -> Option<Self> {
244        let read_exact = |offset: u64, buffer: &mut [u8]| {
245            (file.read_at(buffer, offset).ok()? == buffer.len()).then_some(())
246        };
247        let mut header = [0_u8; 64];
248        read_exact(0, &mut header)?;
249        Self::parse(&header, read_exact, crate::page::page_size())
250    }
251
252    /// [`Self::read`] of an image whose 64-byte file header is `header`,
253    /// reading its program headers through `read_at(offset, buffer)`, on a
254    /// host whose pages are `page` bytes.
255    fn parse(
256        header: &[u8; 64],
257        mut read_at: impl FnMut(u64, &mut [u8]) -> Option<()>,
258        page: u64,
259    ) -> Option<Self> {
260        let u16_at = |at: usize| u16::from_le_bytes([header[at], header[at + 1]]);
261        let u64_at = |at: usize| u64::from_le_bytes(header[at..at + 8].try_into().unwrap());
262        // ELF magic, 64-bit class, little-endian data, ET_EXEC or ET_DYN.
263        if header[..4] != *b"\x7fELF" || header[4] != 2 || header[5] != 1 {
264            return None;
265        }
266        let position_independent = match u16_at(16) {
267            2 => false, // ET_EXEC
268            3 => true,  // ET_DYN
269            _ => return None,
270        };
271        let (phoff, phentsize, phnum) = (u64_at(32), usize::from(u16_at(54)), u16_at(56));
272        if phentsize < 56 {
273            return None;
274        }
275        let mut entry = vec![0_u8; phentsize];
276        let (mut phdr, mut end, mut interpreter) = (0, None::<u64>, false);
277        let (mut text, mut start_data, mut end_data) = (None::<(u64, u64)>, 0_u64, 0_u64);
278        for index in 0..u64::from(phnum) {
279            read_at(phoff.checked_add(index * phentsize as u64)?, &mut entry)?;
280            let field = |at: usize| u64::from_le_bytes(entry[at..at + 8].try_into().unwrap());
281            match u32::from_le_bytes(entry[..4].try_into().unwrap()) {
282                1 => {} // PT_LOAD
283                3 => {
284                    // PT_INTERP
285                    interpreter = true;
286                    continue;
287                }
288                _ => continue,
289            }
290            let (offset, vaddr, filesz, memsz) = (field(8), field(16), field(32), field(40));
291            // PF_X.
292            if text.is_none() && u32::from_le_bytes(entry[4..8].try_into().unwrap()) & 1 != 0 {
293                text = Some((offset & !(page - 1), vaddr & !(page - 1)));
294            }
295            if offset <= phoff && phoff < offset.checked_add(filesz)? {
296                phdr = (phoff - offset).checked_add(vaddr)?;
297            }
298            start_data = start_data.max(vaddr);
299            end_data = end_data.max(vaddr.checked_add(filesz)?);
300            let segment_end = vaddr.checked_add(memsz)?;
301            end = Some(end.map_or(segment_end, |end| end.max(segment_end)));
302        }
303        Some(Self {
304            position_independent,
305            interpreter,
306            phdr,
307            end: end?,
308            text,
309            data_span: end_data.saturating_sub(start_data),
310            page,
311        })
312    }

314    /// Where Linux starts the program break of a process that exec'd this
315    /// image, given the `AT_PHDR` its exec reported, with address
316    /// randomization off: the end of the highest loaded segment at the load
317    /// bias, page-aligned up (`mm->start_brk = mm->brk =
318    /// ELF_PAGEALIGN(elf_brk)`), or why there is no answer.
319    pub fn start_brk(&self, at_phdr: u64) -> Result<u64, &'static str> {
320        let bias = at_phdr
321            .checked_sub(self.phdr)
322            .ok_or("AT_PHDR below the image's program headers")?;
323        self.start_brk_at_bias(bias)
324    }
325
326    /// [`Self::start_brk`] for an image loaded at load bias `bias`.
327    pub fn start_brk_at_bias(&self, bias: u64) -> Result<u64, &'static str> {
328        if self.position_independent && !self.interpreter {
329            // Recent kernels (6.17's load_elf_binary) move it to
330            // ELF_ET_DYN_BASE even without randomization, unless built with
331            // CONFIG_COMPAT_BRK.
332            return Err(
333                "a position-independent image without an interpreter, whose break depends on the kernel's build",
334            );
335        }
336        bias.checked_add(self.end)
337            .and_then(|end| end.checked_add(self.page - 1))
338            .map(|end| end & !(self.page - 1))
339            .ok_or("an image ending past the address space")
340    }
341}

343/// What an x86-64 ELF image declares it needs of the CPU: the
344/// `GNU_PROPERTY_X86_ISA_1_NEEDED` property of its `NT_GNU_PROPERTY_TYPE_0`
345/// note (elf.h; the x86-64 psABI's GNU property section), which glibc's
346/// loader checks against the CPUID it sees.
347#[derive(Clone, Copy, Debug, PartialEq, Eq)]
348pub enum IsaNeeds {
349    /// No such property. The image declares nothing, which does not prove it
350    /// runs on any CPU: a library built for a higher level need not say so.
351    Unmarked,
352    /// The highest x86-64 microarchitecture level its needed bits name:
353    /// 1 baseline, 2 x86-64-v2, 3 x86-64-v3, 4 x86-64-v4; 0 for an empty set.
354    /// A level above 4 is a bit this Hermit does not know, a level above
355    /// x86-64-v4.
356    Level(u8),
357    /// A declaration that cannot be read as written: a truncated or
358    /// misaligned note, or a repeated property.
359    Malformed(&'static str),
360}
361
362/// `PT_GNU_PROPERTY` (elf.h), which holds the image's
363/// `NT_GNU_PROPERTY_TYPE_0` note.
364const PT_GNU_PROPERTY: u32 = 0x6474_e553;
365/// `PT_NOTE`, read for that note where no `PT_GNU_PROPERTY` segment exists.
366const PT_NOTE: u32 = 4;
367/// `NT_GNU_PROPERTY_TYPE_0`.
368const NT_GNU_PROPERTY_TYPE_0: u32 = 5;
369/// `GNU_PROPERTY_X86_ISA_1_NEEDED`.
370const GNU_PROPERTY_X86_ISA_1_NEEDED: u32 = 0xc000_8002;
371/// `EM_X86_64`.
372const EM_X86_64: u16 = 62;
373/// The largest `NT_GNU_PROPERTY_TYPE_0` description read; a larger one
374/// leaves the image's declaration unknown rather than read without bound.
375/// Other notes' descriptions are skipped, not read, whatever their size.
376const MAX_PROPERTY_DESCRIPTION: usize = 64 * 1024;
377/// The most notes walked in one segment, each at least 12 bytes; past it
378/// the rest of the segment is unknown.
379const MAX_NOTES: usize = 4096;
380
381/// What one image's note segments showed: the declaration found, with the
382/// file offset of its note; whether two different notes declared one; and
383/// the first reason some other part could not be read.
384#[derive(Default)]
385struct NoteWalk {
386    found: Option<(u64, IsaNeeds)>,
387    conflict: bool,
388    unknown: Option<&'static str>,
389}
390
391impl NoteWalk {
392    fn declare(&mut self, at: u64, level: IsaNeeds) {
393        match self.found {
394            // One note seen through two segments is one declaration.
395            Some((seen, _)) if seen == at => {}
396            Some(_) => self.conflict = true,
397            None => self.found = Some((at, level)),
398        }
399    }
400
401    fn unknown(&mut self, why: &'static str) {
402        self.unknown.get_or_insert(why);
403    }
404
405    /// A declaration read is kept whatever else could not be read: an
406    /// unrelated note's failure says nothing about it. Only where none was
407    /// read does a failure make the whole image's declaration unknown.
408    fn result(self) -> IsaNeeds {
409        match (self.found, self.unknown) {
410            _ if self.conflict => IsaNeeds::Malformed("more than one x86 ISA needed property"),
411            (Some((_, level)), _) => level,
412            (None, Some(why)) => IsaNeeds::Malformed(why),
413            (None, None) => IsaNeeds::Unmarked,
414        }
415    }
416
417    /// [`Self::result`] for a reader that must refuse incomplete evidence:
418    /// a declaration found beside a later scan state the reader could not
419    /// decide (a truncated or unreadable property) is itself malformed
420    /// there, because the later state could hide a higher need. The known
421    /// declaration stays the lower bound [`Self::result`] reports for a
422    /// refusal; completeness is kept separately, here.
423    fn result_strict(self) -> IsaNeeds {
424        match (self.found, self.unknown) {
425            _ if self.conflict => IsaNeeds::Malformed("more than one x86 ISA needed property"),
426            (Some(_), Some(why)) => IsaNeeds::Malformed(why),
427            (Some((_, level)), None) => level,
428            (None, Some(why)) => IsaNeeds::Malformed(why),
429            (None, None) => IsaNeeds::Unmarked,
430        }
431    }
432}

434impl IsaNeeds {
435    /// What the x86-64 ELF64 image open as `file` declares, read with raw
436    /// syscalls as [`ElfLayout::read_file`] reads. `None` for a file that is
437    /// not a little-endian ELF64 `EM_X86_64` image (nothing for this policy
438    /// to read) and for one the caller cannot read.
439    pub fn read_file(file: &crate::util::RawFd) -> Option<Self> {

451    }

453    /// [`Self::read_file`], refusing incomplete scan evidence: a found
454    /// declaration beside a scan state the reader could not decide (a
455    /// truncated or unreadable later property) reads as
456    /// [`Self::Malformed`], so an automatic consumer cannot select a
457    /// profile from a lower bound alone. An explicit consumer that only
458    /// needs the lower bound for its refusal keeps [`Self::read_file`].
459    pub fn read_file_strict(file: &crate::util::RawFd) -> Option<Self> {

462        };
463        let mut header = [0_u8; 64];
464        read_exact(0, &mut header)?;
465        Self::parse_inner(&header, None, read_exact, true)
466    }
467
468    /// [`Self::read_file_strict`] of the file at `path`.
469    pub fn read_path_strict(path: &Path) -> Option<Self> {
470        Self::read_file_strict(&crate::util::RawFd::open(path, libc::O_RDONLY).ok()?)
471    }
472
473    /// What the image a process has mapped at load bias `bias`, with its
474    /// file header at `header` (where its segment holding file offset 0 is
475    /// mapped: `bias` plus that segment's address), declares, read through
476    /// `read_memory(address, buffer)` from the process's memory: the bytes
477    /// the loader itself reads its properties from. The program headers must
478    /// show a `PT_LOAD` mapping file offset 0 at `header - bias` and holding
479    /// them, so that what was read is known to be this image's. `None` for
480    /// an image mapped otherwise, one that is not x86-64 ELF64, and one whose
481    /// header or program headers are unreadable.
482    pub fn read_mapped(
483        bias: u64,
484        header: u64,
485        mut read_memory: impl FnMut(u64, &mut [u8]) -> Option<()>,
486    ) -> Option<Self> {
487        let header_vaddr = header.checked_sub(bias)?;
488        let mut bytes = [0_u8; 64];
489        read_memory(header, &mut bytes)?;
490        Self::parse(&bytes, Some((bias, header_vaddr)), read_memory)
491    }
492
493    /// [`Self::read_file`] of an image whose file header is `header`, reading
494    /// the rest through `read_at(address, buffer)`: at file offsets, or where
495    /// `mapped` gives the load bias and the address its file header is
496    /// mapped at before the bias, at the addresses the mapped image has them
497    /// ([`Self::read_mapped`]).
498    fn parse(
499        header: &[u8; 64],
500        mapped: Option<(u64, u64)>,
501        read_at: impl FnMut(u64, &mut [u8]) -> Option<()>,
502    ) -> Option<Self> {
503        Self::parse_inner(header, mapped, read_at, false)
504    }
505
506    fn parse_inner(
507        header: &[u8; 64],
508        mapped: Option<(u64, u64)>,
509        mut read_at: impl FnMut(u64, &mut [u8]) -> Option<()>,
510        strict: bool,
511    ) -> Option<Self> {
512        let u16_at = |at: usize| u16::from_le_bytes([header[at], header[at + 1]]);
513        let u64_at = |at: usize| u64::from_le_bytes(header[at..at + 8].try_into().unwrap());
514        if header[..4] != *b"\x7fELF" || header[4] != 2 || header[5] != 1 || u16_at(18) != EM_X86_64
515        {
516            return None;
517        }
518        let (phoff, phentsize, phnum) = (u64_at(32), usize::from(u16_at(54)), u16_at(56));
519        if phentsize < 56 {
520            return None;
521        }
522        let headers_end = (phentsize as u64)
523            .checked_mul(u64::from(phnum))
524            .and_then(|size| size.checked_add(phoff))?;
525        // Where file offset `offset` of the image (in its first segment) and
526        // address `vaddr` are read.
527        let at_offset = |offset: u64| match mapped {
528            None => Some(offset),
529            Some((bias, header_vaddr)) => bias.checked_add(header_vaddr)?.checked_add(offset),
530        };
531        let at_vaddr = |offset: u64, vaddr: u64| match mapped {
532            None => Some(offset),
533            Some((bias, _)) => bias.checked_add(vaddr),
534        };
535        // Every property and note segment, each `(offset, vaddr, size,
536        // align)`, and whether the file header is mapped where it was read.
537        let (mut segments, mut properties, mut header_mapped) = (Vec::new(), 0, false);
538        let mut entry = vec![0_u8; phentsize];
539        for index in 0..u64::from(phnum) {
540            let offset = phoff.checked_add(index * phentsize as u64)?;
541            read_at(at_offset(offset)?, &mut entry)?;
542            let field = |at: usize| u64::from_le_bytes(entry[at..at + 8].try_into().unwrap());
543            let (offset, vaddr, filesz) = (field(8), field(16), field(32));
544            match u32::from_le_bytes(entry[..4].try_into().unwrap()) {
545                // PT_LOAD: the program headers are where they were read only
546                // if a segment maps them, from file offset 0, at the address
547                // the header was read at.
548                1 => {
549                    header_mapped |= offset == 0
550                        && mapped.is_some_and(|(_, header_vaddr)| vaddr == header_vaddr)
551                        && headers_end <= filesz
552                }
553                // Offset, address, size and alignment (p_align), which pads
554                // every note's name and description in the segment (4, or 8
555                // for the ELF64 property notes).
556                kind @ (PT_GNU_PROPERTY | PT_NOTE) => {
557                    properties += usize::from(kind == PT_GNU_PROPERTY);
558                    segments.push((offset, vaddr, filesz, field(48)));
559                }
560                _ => {}
561            }
562        }
563        if mapped.is_some() && !header_mapped {
564            return None;
565        }
566        if properties > 1 {
567            return Some(Self::Malformed("more than one PT_GNU_PROPERTY segment"));
568        }
569        // A linker gives the property note both a PT_GNU_PROPERTY segment
570        // and a PT_NOTE one (or one PT_NOTE with the image's other notes):
571        // every segment is read, so that neither hides a declaration from
572        // the other, and a note is told apart by its place in the file, so
573        // that one declaration seen twice is not two.
574        let mut walk = NoteWalk::default();
575        for (offset, vaddr, size, align) in segments {
576            let align = match align {
577                // An unaligned note segment is laid out as a 4-aligned one.
578                0..=4 => 4,
579                8 => 8,
580                _ => {
581                    walk.unknown("a note segment aligned to neither 4 nor 8");
582                    continue;
583                }
584            };
585            match at_vaddr(offset, vaddr) {
586                Some(address) => {
587                    Self::walk_notes(&mut walk, address, offset, size, align, &mut read_at)
588                }
589                None => walk.unknown("a note segment past the address space"),
590            }
591        }
592        Some(if strict {
593            walk.result_strict()
594        } else {
595            walk.result()
596        })
597    }
598
599    /// Walk the notes of one segment of `size` bytes, read at `address` and
600    /// at file offset `offset`, their names and descriptions each padded to
601    /// `align` (as glibc's ELF_NOTE_DESC_OFFSET and ELF_NOTE_NEXT_OFFSET pad
602    /// them: the description starts, and the next note follows it, at the
603    /// next multiple of `align` from the note's start). Only a GNU property
604    /// note's description is read; every other note is skipped by its sizes,
605    /// whatever they are, so an unrelated note cannot hide a declaration.
606    fn walk_notes(
607        walk: &mut NoteWalk,
608        address: u64,
609        offset: u64,
610        size: u64,
611        align: u64,
612        read_at: &mut impl FnMut(u64, &mut [u8]) -> Option<()>,
613    ) {
614        let mut cursor = 0_u64;
615        for _ in 0..MAX_NOTES {
616            if cursor == size {
617                return;
618            }
619            if size - cursor < 12 {
620                return walk.unknown("a truncated note header");
621            }
622            let mut head = [0_u8; 12];
623            if address
624                .checked_add(cursor)
625                .and_then(|at| read_at(at, &mut head))
626                .is_none()
627            {
628                return walk.unknown("a note past the end of the file or unmapped");
629            }
630            let word =
631                |at: usize| u64::from(u32::from_le_bytes(head[at..at + 4].try_into().unwrap()));
632            let (namesz, descsz, kind) = (word(0), word(4), word(8) as u32);
633            let ends = namesz
634                .checked_add(12)
635                .and_then(|name_end| name_end.checked_next_multiple_of(align))
636                .and_then(|desc_start| {
637                    let desc_end = desc_start.checked_add(descsz)?;
638                    Some((desc_start, desc_end.checked_next_multiple_of(align)?))
639                });
640            let Some((desc_start, next)) = ends.filter(|&(_, next)| next <= size - cursor) else {
641                return walk.unknown("a truncated note");
642            };
643            if kind == NT_GNU_PROPERTY_TYPE_0 && namesz == 4 {
644                let mut name = [0_u8; 4];
645                let read = address
646                    .checked_add(cursor + 12)
647                    .and_then(|at| read_at(at, &mut name));
648                if read.is_none() {
649                    return walk.unknown("a note past the end of the file or unmapped");
650                }
651                if name == *b"GNU\0" {
652                    match usize::try_from(descsz) {
653                        Ok(length) if length <= MAX_PROPERTY_DESCRIPTION => {
654                            let mut description = vec![0_u8; length];
655                            let read = address
656                                .checked_add(cursor + desc_start)
657                                .and_then(|at| read_at(at, &mut description));
658                            if read.is_none() {
659                                return walk.unknown("a note past the end of the file or unmapped");
660                            }
661                            let (level, unknown) = Self::from_properties(&description);
662                            if let Some(level) = level {
663                                walk.declare(offset + cursor, level);
664                            }
665                            if let Some(why) = unknown {
666                                walk.unknown(why);
667                            }
668                        }
669                        _ => walk.unknown("a GNU property note larger than 64 KiB"),
670                    }
671                }
672            }
673            cursor += next;
674        }
675        if cursor < size {
676            walk.unknown("a note segment of more than 4096 notes");
677        }
678    }
679
680    /// The declaration in one `NT_GNU_PROPERTY_TYPE_0` description, if it
681    /// has one, and why some of it could not be read, if so. A declaration
682    /// read before an unreadable property is kept.
683    fn from_properties(mut properties: &[u8]) -> (Option<Self>, Option<&'static str>) {
684        let mut found = None;
685        while !properties.is_empty() {
686            if properties.len() < 8 {
687                return (found, Some("a truncated GNU property header"));
688            }
689            let word = |at: usize| u32::from_le_bytes(properties[at..at + 4].try_into().unwrap());
690            let (kind, size) = (word(0), word(4) as usize);
691            let end = size
692                .checked_next_multiple_of(8)
693                .and_then(|data| data.checked_add(8));
694            let Some(end) = end.filter(|&end| end <= properties.len()) else {
695                return (found, Some("a truncated GNU property"));
696            };
697            if kind == GNU_PROPERTY_X86_ISA_1_NEEDED {
698                if found.is_some() {
699                    return (
700                        Some(Self::Malformed("more than one x86 ISA needed property")),
701                        None,
702                    );
703                }
704                if size != 4 {
705                    return (None, Some("an x86 ISA needed property that is not 4 bytes"));
706                }
707                // The highest set bit: 1 baseline, 2 v2, 4 v3, 8 v4, and
708                // above that, levels this Hermit does not know.
709                found = Some(Self::Level(32 - word(8).leading_zeros() as u8));
710            }
711            properties = &properties[end..];
712        }
713        (found, None)
714    }
715}

717/// The start and pathname of the lowest file mapping at or above `bias` in
718/// `maps` (a process's `/proc/<pid>/maps`) that maps its file from offset 0:
719/// where the kernel put the first segment of an image it loaded at load
720/// bias `bias`, and with it the image's file header. The kernel maps that
721/// segment at the bias plus its address, which need not be 0.
722pub fn first_image_mapping(maps: &[u8], bias: u64) -> Option<(u64, Vec<u8>)> {
723    maps.split(|&byte| byte == b'\n')
724        .filter_map(|line| {
725            let mut fields = line.splitn(6, |&byte| byte == b' ');
726            let range = std::str::from_utf8(fields.next()?).ok()?;
727            let (start, _) = range.split_once('-')?;
728            let start = u64::from_str_radix(start, 16).ok()?;
729            let _permissions = fields.next()?;
730            let offset = u64::from_str_radix(std::str::from_utf8(fields.next()?).ok()?, 16).ok()?;
731            let _device = fields.next()?;
732            let inode: u64 = std::str::from_utf8(fields.next()?).ok()?.parse().ok()?;
733            let path = fields
734                .next()
735                .unwrap_or_default()
736                .trim_ascii_start()
737                .to_vec();
738            (start >= bias && offset == 0 && inode != 0).then_some((start, path))
739        })
740        .min_by_key(|&(start, _)| start)
741}

743/// The extent of the image whose first segment [`first_image_mapping`] finds
744/// at or above `bias` in `maps`, with its pathname: from that mapping's start
745/// to the end of the last mapping of the same file (the same inode and
746/// pathname, as [`first_image_mapping`] matches it) at or above that start.
747/// Every instruction the image's code executes lies in this range. Its
748/// anonymous mappings (`.bss`) are not counted: no code runs there.
749pub fn image_mapping_range(maps: &[u8], bias: u64) -> Option<(u64, u64, Vec<u8>)> {
750    let mappings: Vec<(u64, u64, u64, u64, Vec<u8>)> = maps
751        .split(|&byte| byte == b'\n')
752        .filter_map(|line| {
753            let mut fields = line.splitn(6, |&byte| byte == b' ');
754            let range = std::str::from_utf8(fields.next()?).ok()?;
755            let (start, end) = range.split_once('-')?;
756            let start = u64::from_str_radix(start, 16).ok()?;
757            let end = u64::from_str_radix(end, 16).ok()?;
758            let _permissions = fields.next()?;
759            let offset = u64::from_str_radix(std::str::from_utf8(fields.next()?).ok()?, 16).ok()?;
760            let _device = fields.next()?;
761            let inode: u64 = std::str::from_utf8(fields.next()?).ok()?.parse().ok()?;
762            let path = fields
763                .next()
764                .unwrap_or_default()
765                .trim_ascii_start()
766                .to_vec();
767            (inode != 0).then_some((start, end, offset, inode, path))
768        })
769        .collect();
770    let (first, _, _, inode, path) = mappings
771        .iter()
772        .filter(|&&(start, _, offset, _, _)| start >= bias && offset == 0)
773        .min_by_key(|&&(start, ..)| start)?
774        .clone();
775    let end = mappings
776        .iter()
777        .filter(|(start, _, _, other_inode, other_path)| {
778            *start >= first && *other_inode == inode && *other_path == path
779        })
780        .map(|&(_, end, ..)| end)
781        .max()?;
782    Some((first, end, path))
783}

817/// The kernel's page size on x86-64, the most a `PT_INTERP` name can be.
818const PAGE_SIZE: u64 = 4096;
819
820/// The program interpreter (`PT_INTERP`) the x86-64 ELF64 image at `path`
821/// names: `Ok(None)` for an image naming none, an error for a file that is
822/// not one or whose `PT_INTERP` is unreadable. Read as the kernel reads it
823/// (`load_elf_binary`): at most a page, at least two bytes, the last of
824/// them NUL, and the name is the string up to its first NUL.
825pub fn program_interpreter(path: &Path) -> Result<Option<PathBuf>, &'static str> {
826    program_interpreter_of(path, EM_X86_64)
827}

829/// [`program_interpreter`] of an ELF64 image built for ELF machine
830/// `machine` (`e_machine`) instead of x86-64: the `PT_INTERP` it names is
831/// read the same way on every little-endian ELF64 machine.
832pub fn program_interpreter_of(path: &Path, machine: u16) -> Result<Option<PathBuf>, &'static str> {
833    use std::os::unix::ffi::OsStrExt;
834    let file = crate::util::RawFd::open(path, libc::O_RDONLY).map_err(|_| "unreadable")?;
835    let read_exact = |offset: u64, buffer: &mut [u8]| {
836        (file.read_at(buffer, offset).ok()? == buffer.len()).then_some(())
837    };
838    let mut header = [0_u8; 64];
839    read_exact(0, &mut header).ok_or("shorter than an ELF header")?;
840    let u16_at = |at: usize| u16::from_le_bytes([header[at], header[at + 1]]);
841    let u64_at = |at: usize| u64::from_le_bytes(header[at..at + 8].try_into().unwrap());
842    if header[..4] != *b"\x7fELF" || header[4] != 2 || header[5] != 1 || u16_at(18) != machine {
843        return Err(if machine == EM_X86_64 {
844            "not an x86-64 ELF64 image"
845        } else {
846            "not an ELF64 image for this machine"
847        });
848    }
849    let (phoff, phentsize, phnum) = (u64_at(32), usize::from(u16_at(54)), u16_at(56));
850    if phentsize < 56 {
851        return Err("program headers too small");
852    }
853    let mut entry = vec![0_u8; phentsize];
854    for index in 0..u64::from(phnum) {
855        let offset = (index * phentsize as u64)
856            .checked_add(phoff)
857            .ok_or("program headers past the address space")?;
858        read_exact(offset, &mut entry).ok_or("program headers past the end of the file")?;
859        if u32::from_le_bytes(entry[..4].try_into().unwrap()) != 3 {
860            continue; // Not PT_INTERP.
861        }
862        let field = |at: usize| u64::from_le_bytes(entry[at..at + 8].try_into().unwrap());
863        let size = field(32);
864        if !(2..=PAGE_SIZE).contains(&size) {
865            return Err("a PT_INTERP shorter than two bytes or larger than a page");
866        }
867        let mut name = vec![0_u8; size as usize];
868        read_exact(field(8), &mut name).ok_or("a PT_INTERP past the end of the file")?;
869        if name.last() != Some(&0) {
870            return Err("a PT_INTERP without a terminating NUL");
871        }
872        let end = name
873            .iter()
874            .position(|&byte| byte == 0)
875            .expect("the last byte is NUL");
876        return Ok(Some(PathBuf::from(std::ffi::OsStr::from_bytes(
877            &name[..end],
878        ))));
879    }
880    Ok(None)
881}

883/// The named profile `--cpu-profile auto` resolves to for an exec of
884/// `objects`, each `(what, declaration)`: the lowest presenting the highest
885/// level any declares, x86-64-v2 where none declares one, with why. An
886/// object unread or malformed, or a level no profile presents, refuses with
887/// why: auto never guesses.
888pub fn auto_profile(
889    objects: &[(String, Option<IsaNeeds>)],
890) -> Result<(detcore_model::cpu_profile::CpuProfile, String), String> {
891    use detcore_model::cpu_profile::CpuProfile;
892    let mut highest: Option<(&str, u8)> = None;
893    for (what, needs) in objects {
894        match needs {
895            None => return Err(format!("{what} is unreadable or not an x86-64 ELF64 image")),
896            Some(IsaNeeds::Malformed(why)) => {
897                return Err(format!(
898                    "{what} declares its x86 ISA level unreadably ({why})"
899                ));
900            }
901            Some(IsaNeeds::Unmarked) => {}
902            Some(IsaNeeds::Level(level)) => {
903                if highest.is_none_or(|(_, highest)| *level > highest) {
904                    highest = Some((what, *level));
905                }
906            }
907        }
908    }
909    let level = highest.map_or(0, |(_, level)| level);
910    let Some(profile) = CpuProfile::lowest_presenting(level) else {
911        let (what, _) = highest.expect("a level above 3 was declared");
912        return Err(format!(
913            "{what} declares that it needs {}, which no CPU profile presents",
914            x86_64_level_name(level)
915        ));
916    };
917    let why = match highest {
918        Some((what, level)) if level > 2 => {
919            format!("{what} declares that it needs {}", x86_64_level_name(level))
920        }
921        _ => "nothing it execs first declares more than x86-64-v2".to_owned(),
922    };
923    Ok((profile, why))
924}

935/// The load bias of the image file at `path` with inode `ino`, in a
936/// process whose `/proc/<pid>/maps` reads `maps`: the start of the file's
937/// executable mapping at the text segment's offset less the text segment's
938/// address ([`ElfLayout::text`]), or why there is none. The executable
939/// mapping, because a launcher may also map the file to read it (SaBRe's
940/// rewriter maps it read-only at offset 0) and may not map the image's first
941/// page at all. The file is matched by inode and its whole pathname as the
942/// reader sees it, not by device: maps shows the superblock's device, which
943/// on btrfs and overlay is not the `st_dev` a `stat` of the file reports.
944/// Where several such mappings give different biases, there is no answer.
945/// This is
946/// how Detcore finds an image's bias where the kernel exec'd a launcher,
947/// whose `AT_PHDR` the process's auxiliary vector holds.
948pub fn load_bias(
949    maps: &[u8],
950    path: &Path,
951    ino: u64,
952    (text_offset, text_page): (u64, u64),
953) -> Result<u64, &'static str> {
954    // Bytes, not text: a pathname in the mappings is whatever bytes the file
955    // was named with.
956    fn text(field: &[u8]) -> Option<&str> {
957        std::str::from_utf8(field).ok()
958    }
959    // As the mappings show it: Linux writes a newline in a pathname as
960    // `\012` (`show_map_vma`, `seq_file_path(m, file, "\n")`).
961    let mut displayed = Vec::new();
962    for &byte in path.as_os_str().as_bytes() {
963        if byte == b'\n' {
964            displayed.extend_from_slice(b"\\012");
965        } else {
966            displayed.push(byte);
967        }
968    }
969    let mut biases = maps.split(|&byte| byte == b'\n').filter_map(|line| {
970        let mut fields = line.splitn(6, |byte| byte.is_ascii_whitespace());
971        let range = text(fields.next()?)?;
972        let permissions = fields.next()?;
973        let offset = u64::from_str_radix(text(fields.next()?)?, 16).ok()?;
974        let _device = fields.next()?;
975        let map_ino: u64 = text(fields.next()?)?.parse().ok()?;
976        let pathname = fields.next()?.trim_ascii_start();
977        if !permissions.contains(&b'x')
978            || offset != text_offset
979            || map_ino != ino
980            || pathname != displayed
981        {
982            return None;
983        }
984        let start = u64::from_str_radix(range.split_once('-')?.0, 16).ok()?;
985        start.checked_sub(text_page)
986    });
987    let bias = biases
988        .next()
989        .ok_or("no executable mapping of the image's text")?;
990    if biases.any(|other| other != bias) {
991        return Err("executable mappings of the image's text at different biases");
992    }
993    Ok(bias)
994}

1006/// What Detcore found when it checked its model of a process's program break
1007/// ([`ElfLayout::start_brk`]) against the kernel's `start_brk`.
1008#[derive(Clone, Copy, Debug, PartialEq, Eq)]
1009pub enum BrkCheck {
1010    /// The model equals the kernel's start.
1011    Equal(u64),
1012    /// The model differs from the kernel's start: a Detcore defect.
1013    Differs {
1014        /// Detcore's start.
1015        model: u64,
1016        /// The kernel's start.
1017        kernel: u64,
1018    },
1019    /// No model, for the reason given.
1020    Unmodeled(&'static str),
1021    /// Not compared: the kernel randomized the break, which Linux's rule then
1022    /// does not give.
1023    Randomized(u64),
1024    /// Not compared: whether the kernel randomized the break is unknown.
1025    RandomizationUnknown(u64),
1026    /// Not compared: the kernel's start cannot be read.
1027    KernelUnread(u64),
1028}
1029
1030impl BrkCheck {
1031    /// The outcome for a model (or why there is none), whether the kernel
1032    /// randomized the break (`None`: unknown), and the kernel's start (`None`:
1033    /// unread).
1034    pub fn of(
1035        model: Result<u64, &'static str>,
1036        randomized: Option<bool>,
1037        kernel: Option<u64>,
1038    ) -> Self {
1039        let model = match model {
1040            Ok(model) => model,
1041            Err(reason) => return Self::Unmodeled(reason),
1042        };
1043        match (randomized, kernel) {
1044            (Some(true), _) => Self::Randomized(model),
1045            (None, _) => Self::RandomizationUnknown(model),
1046            (Some(false), None) => Self::KernelUnread(model),
1047            (Some(false), Some(kernel)) if kernel == model => Self::Equal(model),
1048            (Some(false), Some(kernel)) => Self::Differs { model, kernel },
1049        }
1050    }
1051}

1098/// What the kernel's script handler makes of a file.
1099#[derive(Debug, PartialEq, Eq)]
1100pub enum Script {
1101    /// Not a script.
1102    Not,
1103    /// A file the caller may not read, which the kernel may still execute.
1104    Unread,
1105    /// A script; the kernel executes this interpreter path.
1106    Interpreter(PathBuf),
1107    /// A `#!` line that names no interpreter the kernel would run.
1108    Declined,
1109}
1110
1111/// What the kernel's script handler makes of the file at `path`.
1112pub fn script_interpreter(path: &Path) -> Script {
1113    let mut header = Vec::with_capacity(BINPRM_BUF_SIZE);
1114    // Nonblocking and without a controlling terminal, so that a file
1115    // replaced by a FIFO or device since it was checked cannot wait.
1116    let read = fs::OpenOptions::new()
1117        .read(true)
1118        .custom_flags(libc::O_NONBLOCK | libc::O_NOCTTY)
1119        .open(path)
1120        .and_then(|file| file.take(BINPRM_BUF_SIZE as u64).read_to_end(&mut header));
1121    match read {
1122        Ok(_) => parse_script(&header),
1123        Err(_) => Script::Unread,
1124    }
1125}
1126
1127/// Source of truth: `load_script()` in fs/binfmt_script.c. The kernel reads
1128/// the first [`BINPRM_BUF_SIZE`] bytes into a zeroed buffer. The line ends at
1129/// the first newline, and the interpreter path at a space, tab or NUL (so a
1130/// short file with no newline ends at the zeroes). A line with no newline
1131/// whose interpreter path runs to the end of the buffer may be truncated, and
1132/// the kernel declines it.
1133fn parse_script(header: &[u8]) -> Script {
1134    let mut buffer = [0_u8; BINPRM_BUF_SIZE];
1135    let read = header.len().min(BINPRM_BUF_SIZE);
1136    buffer[..read].copy_from_slice(&header[..read]);
1137    let Some(line) = buffer.strip_prefix(b"#!") else {
1138        return Script::Not;
1139    };
1140    let (line, ended) = match line.iter().position(|&byte| byte == b'\n') {
1141        Some(end) => (&line[..end], true),
1142        None => (line, false),
1143    };
1144    let start = line
1145        .iter()
1146        .position(|&byte| byte != b' ' && byte != b'\t')
1147        .unwrap_or(line.len());
1148    let rest = &line[start..];
1149    let end = rest
1150        .iter()
1151        .position(|&byte| matches!(byte, b' ' | b'\t' | 0));
1152    if end.is_none() && !ended {
1153        return Script::Declined;
1154    }
1155    let interpreter = &rest[..end.unwrap_or(rest.len())];
1156    if interpreter.is_empty() {
1157        return Script::Declined;
1158    }
1159    Script::Interpreter(PathBuf::from(std::ffi::OsStr::from_bytes(interpreter)))
1160}
