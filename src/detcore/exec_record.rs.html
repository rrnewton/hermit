9//! The image a process executes, which its `/proc/<pid>/exe` link names.
10//!
11//! Detcore records it at each exec, on every backend, and checks it against
12//! the kernel's own link when the exec completes (`tool_global`). The
13//! kernel's answer is the file it exec'd, and a backend whose launcher loads
14//! the guest itself (SaBRe's loader) has the kernel exec the launcher, so the
15//! kernel names the launcher; only there does Detcore answer the link from
16//! the record (`Detcore::launched_exe`). Everywhere else the kernel's answer
17//! stands.
18//!
19//! Only the path is recorded. Where the image ends, which a program-break
20//! model would need, is known only once the image is loaded, and nothing
21//! reads it yet.
22

34/// The bytes of a file the kernel reads to find its format
35/// (`BINPRM_BUF_SIZE`, include/linux/binfmts.h).
36const BINPRM_BUF_SIZE: usize = 256;
37
38/// The interpreter levels the kernel follows before it refuses an exec with
39/// `ELOOP` (`exec_binprm`'s `depth > 5`, fs/exec.c).
40const MAX_INTERPRETER_DEPTH: usize = 5;
41
42/// The image a process executes.
43#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
44pub struct ExecRecord {
45    /// The image's canonical path, as the process's root directory shows it:
46    /// what `readlink("/proc/self/exe")` returns while the file is there.
47    pub exe: PathBuf,
48    /// The image file's device and inode when it was exec'd, which tell
49    /// whether the file is still at `exe` ([`Self::is_present`]).
50    dev: u64,
51    ino: u64,
52    /// Its link count and inode change time then: a rename, link or unlink
53    /// of the file since changes both, even where the same inode is put back
54    /// at `exe`, which Linux's link would no longer name.
55    nlink: u64,
56    ctime: (i64, i64),
57}
58
59impl ExecRecord {
60    /// The image an `execve` of `path` runs, as the kernel finds it: `path`
61    /// taken from the working directory `cwd`, every symbolic link followed,
62    /// and for a script the interpreter its `#!` line names, level by level,
63    /// as `/proc/<pid>/exe` names the interpreter. `root` is the process's
64    /// root directory as the caller sees it, and `cwd` is the working
65    /// directory below it, absolute.
66    ///
67    /// `None` where the kernel would refuse the exec (no such file, a
68    /// malformed `#!` line, too many interpreter levels), where the caller
69    /// cannot look the path up, and for a file the caller cannot read, which
70    /// the kernel may still run (an execute-only binary, or a script whose
71    /// `#!` line the kernel reads without a permission check): the link is
72    /// then the kernel's to answer. A resolution can still be wrong where
73    /// the kernel runs a file differently (a `binfmt_misc` handler) or the
74    /// caller sees another filesystem; `tool_global` checks every record
75    /// against the kernel's own link after the exec.
76    pub fn resolve(root: &Path, cwd: &Path, path: &Path) -> Option<Self> {
77        let mut spelled = cwd.join(path);
78        for _ in 0..=MAX_INTERPRETER_DEPTH {
79            let found = fs::canonicalize(below(root, &spelled)?).ok()?;
80            // Linux execs only a regular file, and refuses anything else
81            // (EACCES) before it opens it.
82            if !fs::metadata(&found).ok()?.is_file() {
83                return None;
84            }
85            match script_interpreter(&found) {
86                Script::Not => return Self::at(root, &found),
87                // The kernel opens the interpreter from the caller's working
88                // directory, as it opened the script.
89                Script::Interpreter(interpreter) => spelled = cwd.join(interpreter),
90                Script::Declined | Script::Unread => return None,
91            }
92        }
93        None
94    }
95
96    /// The record of the image at `found`, a canonical path the caller sees
97    /// below `root`, as that file is now.
98    pub fn at(root: &Path, found: &Path) -> Option<Self> {
99        let metadata = fs::metadata(found).ok()?;
100        Some(Self {
101            exe: Path::new("/").join(found.strip_prefix(root).ok()?),
102            dev: metadata.dev(),
103            ino: metadata.ino(),
104            nlink: metadata.nlink(),
105            ctime: (metadata.ctime(), metadata.ctime_nsec()),
106        })
107    }
108
109    /// The image file's device and inode when it was exec'd.
110    pub fn identity(&self) -> (u64, u64) {
111        (self.dev, self.ino)
112    }
113
114    /// Whether the image's file is still at its path, untouched, for a
115    /// process whose root directory the caller sees at `root`: the same
116    /// inode, with the link count and change time it had at the exec. A
117    /// rename or link of the image since, or a change of its mode, reads as
118    /// absent, so the caller refuses rather than name a path Linux's link
119    /// might not.
120    pub fn is_present(&self, root: &Path) -> bool {
121        below(root, &self.exe)
122            .and_then(|path| fs::symlink_metadata(path).ok())
123            .is_some_and(|now| {
124                (
125                    now.dev(),
126                    now.ino(),
127                    now.nlink(),
128                    (now.ctime(), now.ctime_nsec()),
129                ) == (self.dev, self.ino, self.nlink, self.ctime)
130            })
131    }
132}

135/// x86_64, where Hermit runs.
136const PAGE_SIZE: u64 = 4096;

138/// What decides where Linux starts a process's program break after it execs
139/// an ELF image (`load_elf_binary`, fs/binfmt_elf.c), from the image's file
140/// header and program headers.
141#[derive(Clone, Copy, Debug, PartialEq, Eq)]
142pub struct ElfLayout {
143    /// Whether the image is `ET_DYN` (position-independent) rather than
144    /// `ET_EXEC`.
145    pub position_independent: bool,
146    /// Whether the image names a program interpreter (`PT_INTERP`).
147    pub interpreter: bool,
148    /// The virtual address of the image's program headers before the load
149    /// bias, as Linux finds it for `AT_PHDR` (since Linux 5.18): `e_phoff` in
150    /// the last `PT_LOAD` segment whose file range holds it, or 0 where none
151    /// does (`AT_PHDR` is then the bias itself).
152    pub phdr: u64,
153    /// The end of the highest `PT_LOAD` segment in memory, `p_vaddr +
154    /// p_memsz`, before the load bias: the end of its bss.
155    pub end: u64,
156}
157
158impl ElfLayout {
159    /// The layout of the ELF64 image at `path` (`ET_EXEC` or `ET_DYN`), or
160    /// `None` for any other file, one without a loadable segment, and one the
161    /// caller cannot read.
162    pub fn read(path: &Path) -> Option<Self> {
163        Self::read_file(&crate::util::RawFd::open(path, libc::O_RDONLY).ok()?)
164    }
165
166    /// [`Self::read`] of an open file, read with raw syscalls: in-guest
167    /// Detcore must not call libc's wrappers (`crate::util::raw_syscall`).
168    pub fn read_file(file: &crate::util::RawFd) -> Option<Self> {
169        let read_exact = |offset: u64, buffer: &mut [u8]| {
170            (file.read_at(buffer, offset).ok()? == buffer.len()).then_some(())
171        };
172        let mut header = [0_u8; 64];
173        read_exact(0, &mut header)?;
174        Self::parse(&header, read_exact)
175    }
176
177    /// [`Self::read`] of an image whose 64-byte file header is `header`,
178    /// reading its program headers through `read_at(offset, buffer)`.
179    fn parse(
180        header: &[u8; 64],
181        mut read_at: impl FnMut(u64, &mut [u8]) -> Option<()>,
182    ) -> Option<Self> {
183        let u16_at = |at: usize| u16::from_le_bytes([header[at], header[at + 1]]);
184        let u64_at = |at: usize| u64::from_le_bytes(header[at..at + 8].try_into().unwrap());
185        // ELF magic, 64-bit class, little-endian data, ET_EXEC or ET_DYN.
186        if header[..4] != *b"\x7fELF" || header[4] != 2 || header[5] != 1 {
187            return None;
188        }
189        let position_independent = match u16_at(16) {
190            2 => false, // ET_EXEC
191            3 => true,  // ET_DYN
192            _ => return None,
193        };
194        let (phoff, phentsize, phnum) = (u64_at(32), usize::from(u16_at(54)), u16_at(56));
195        if phentsize < 56 {
196            return None;
197        }
198        let mut entry = vec![0_u8; phentsize];
199        let (mut phdr, mut end, mut interpreter) = (0, None::<u64>, false);
200        for index in 0..u64::from(phnum) {
201            read_at(phoff.checked_add(index * phentsize as u64)?, &mut entry)?;
202            let field = |at: usize| u64::from_le_bytes(entry[at..at + 8].try_into().unwrap());
203            match u32::from_le_bytes(entry[..4].try_into().unwrap()) {
204                1 => {} // PT_LOAD
205                3 => {
206                    // PT_INTERP
207                    interpreter = true;
208                    continue;
209                }
210                _ => continue,
211            }
212            let (offset, vaddr, filesz, memsz) = (field(8), field(16), field(32), field(40));
213            if offset <= phoff && phoff < offset.checked_add(filesz)? {
214                phdr = (phoff - offset).checked_add(vaddr)?;
215            }
216            let segment_end = vaddr.checked_add(memsz)?;
217            end = Some(end.map_or(segment_end, |end| end.max(segment_end)));
218        }
219        Some(Self {
220            position_independent,
221            interpreter,
222            phdr,
223            end: end?,
224        })
225    }
226
227    /// Where Linux starts the program break of a process that exec'd this
228    /// image, given the `AT_PHDR` its exec reported, with address
229    /// randomization off: the end of the highest loaded segment at the load
230    /// bias, page-aligned up (`mm->start_brk = mm->brk =
231    /// ELF_PAGEALIGN(elf_brk)`), or why there is no answer.
232    pub fn start_brk(&self, at_phdr: u64) -> Result<u64, &'static str> {
233        if self.position_independent && !self.interpreter {
234            // Recent kernels (6.17's load_elf_binary) move it to
235            // ELF_ET_DYN_BASE even without randomization, unless built with
236            // CONFIG_COMPAT_BRK.
237            return Err(
238                "a position-independent image without an interpreter, whose break depends on the kernel's build",
239            );
240        }
241        let bias = at_phdr
242            .checked_sub(self.phdr)
243            .ok_or("AT_PHDR below the image's program headers")?;
244        bias.checked_add(self.end)
245            .and_then(|end| end.checked_add(PAGE_SIZE - 1))
246            .map(|end| end & !(PAGE_SIZE - 1))
247            .ok_or("an image ending past the address space")
248    }
249}

251/// The kernel's `start_brk` in a `/proc/<pid>/stat` line: field 47, counted
252/// past the command, which may hold spaces and parentheses.
253pub fn stat_start_brk(stat: &str) -> Option<u64> {
254    stat[stat.rfind(')')? + 1..]
255        .split_ascii_whitespace()
256        .nth(47 - 3)?
257        .parse()
258        .ok()
259}

261/// What Detcore found when it checked its model of a process's program break
262/// ([`ElfLayout::start_brk`]) against the kernel's `start_brk`.
263#[derive(Clone, Copy, Debug, PartialEq, Eq)]
264pub enum BrkCheck {
265    /// The model equals the kernel's start.
266    Equal(u64),
267    /// The model differs from the kernel's start: a Detcore defect.
268    Differs {
269        /// Detcore's start.
270        model: u64,
271        /// The kernel's start.
272        kernel: u64,
273    },
274    /// No model, for the reason given.
275    Unmodeled(&'static str),
276    /// Not compared: the kernel randomized the break, which Linux's rule then
277    /// does not give.
278    Randomized(u64),
279    /// Not compared: whether the kernel randomized the break is unknown.
280    RandomizationUnknown(u64),
281    /// Not compared: the kernel's start cannot be read.
282    KernelUnread(u64),
283}
284
285impl BrkCheck {
286    /// The outcome for a model (or why there is none), whether the kernel
287    /// randomized the break (`None`: unknown), and the kernel's start (`None`:
288    /// unread).
289    pub fn of(
290        model: Result<u64, &'static str>,
291        randomized: Option<bool>,
292        kernel: Option<u64>,
293    ) -> Self {
294        let model = match model {
295            Ok(model) => model,
296            Err(reason) => return Self::Unmodeled(reason),
297        };
298        match (randomized, kernel) {
299            (Some(true), _) => Self::Randomized(model),
300            (None, _) => Self::RandomizationUnknown(model),
301            (Some(false), None) => Self::KernelUnread(model),
302            (Some(false), Some(kernel)) if kernel == model => Self::Equal(model),
303            (Some(false), Some(kernel)) => Self::Differs { model, kernel },
304        }
305    }
306}
307
308impl std::fmt::Display for BrkCheck {
309    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
310        match self {
311            Self::Equal(model) => write!(f, "model {model:#x} equals the kernel's start_brk"),
312            Self::Differs { model, kernel } => write!(
313                f,
314                "model {model:#x} differs from the kernel's start_brk {kernel:#x}"
315            ),
316            Self::Unmodeled(reason) => write!(f, "not modeled: {reason}"),
317            Self::Randomized(model) => write!(
318                f,
319                "model {model:#x} not compared: the kernel randomized the break"
320            ),
321            Self::RandomizationUnknown(model) => write!(
322                f,
323                "model {model:#x} not compared: whether the kernel randomized the break is unknown"
324            ),
325            Self::KernelUnread(model) => write!(
326                f,
327                "model {model:#x} not compared: the kernel's start_brk is unreadable"
328            ),
329        }
330    }
331}

333/// The value of entry `key` in an auxiliary vector read from
334/// `/proc/<pid>/auxv` as `auxv`, native-endian pairs ended by `AT_NULL`.
335pub fn auxv_entry(auxv: &[u8], key: u64) -> Option<u64> {
336    auxv.as_chunks::<16>()
337        .0
338        .iter()
339        .map(|pair| {
340            let word = |at: usize| u64::from_ne_bytes(pair[at..at + 8].try_into().unwrap());
341            (word(0), word(8))
342        })
343        .take_while(|&(entry, _)| entry != 0)
344        .find_map(|(entry, value)| (entry == key).then_some(value))
345}

353/// What the kernel's script handler makes of a file.
354#[derive(Debug, PartialEq, Eq)]
355enum Script {
356    /// Not a script.
357    Not,
358    /// A file the caller may not read, which the kernel may still execute.
359    Unread,
360    /// A script; the kernel executes this interpreter path.
361    Interpreter(PathBuf),
362    /// A `#!` line that names no interpreter the kernel would run.
363    Declined,
364}
365
366/// What the kernel's script handler makes of the file at `path`.
367fn script_interpreter(path: &Path) -> Script {
368    let mut header = Vec::with_capacity(BINPRM_BUF_SIZE);
369    // Nonblocking and without a controlling terminal, so that a file
370    // replaced by a FIFO or device since it was checked cannot wait.
371    let read = fs::OpenOptions::new()
372        .read(true)
373        .custom_flags(libc::O_NONBLOCK | libc::O_NOCTTY)
374        .open(path)
375        .and_then(|file| file.take(BINPRM_BUF_SIZE as u64).read_to_end(&mut header));
376    match read {
377        Ok(_) => parse_script(&header),
378        Err(_) => Script::Unread,
379    }
380}
381
382/// Source of truth: `load_script()` in fs/binfmt_script.c. The kernel reads
383/// the first [`BINPRM_BUF_SIZE`] bytes into a zeroed buffer. The line ends at
384/// the first newline, and the interpreter path at a space, tab or NUL (so a
385/// short file with no newline ends at the zeroes). A line with no newline
386/// whose interpreter path runs to the end of the buffer may be truncated, and
387/// the kernel declines it.
388fn parse_script(header: &[u8]) -> Script {
389    let mut buffer = [0_u8; BINPRM_BUF_SIZE];
390    let read = header.len().min(BINPRM_BUF_SIZE);
391    buffer[..read].copy_from_slice(&header[..read]);
392    let Some(line) = buffer.strip_prefix(b"#!") else {
393        return Script::Not;
394    };
395    let (line, ended) = match line.iter().position(|&byte| byte == b'\n') {
396        Some(end) => (&line[..end], true),
397        None => (line, false),
398    };
399    let start = line
400        .iter()
401        .position(|&byte| byte != b' ' && byte != b'\t')
402        .unwrap_or(line.len());
403    let rest = &line[start..];
404    let end = rest
405        .iter()
406        .position(|&byte| matches!(byte, b' ' | b'\t' | 0));
407    if end.is_none() && !ended {
408        return Script::Declined;
409    }
410    let interpreter = &rest[..end.unwrap_or(rest.len())];
411    if interpreter.is_empty() {
412        return Script::Declined;
413    }
414    Script::Interpreter(PathBuf::from(std::ffi::OsStr::from_bytes(interpreter)))
415}

424    fn an_elf_layout_gives_linuxs_start_of_the_program_break() {

426        let layout = ElfLayout::read(&std::env::current_exe().unwrap()).unwrap();
427        assert!(layout.end > layout.phdr);

431        header[..4].copy_from_slice(b"\x7fELF");
432        header[4] = 1;
433        header[5] = 1;
434        assert_eq!(ElfLayout::parse(&header, |_, _| None), None);
435
436        // A synthetic ET_EXEC image with two loads: text at 0x400000 holding
437        // the program headers at file offset 64, and data at 0x403e10 whose
438        // bss runs to 0x4050b0.
439        let mut header = [0_u8; 64];
440        header[..4].copy_from_slice(b"\x7fELF");
441        header[4] = 2;
442        header[5] = 1;
443        header[16..18].copy_from_slice(&2_u16.to_le_bytes());
444        header[32..40].copy_from_slice(&64_u64.to_le_bytes());
445        header[54..56].copy_from_slice(&56_u16.to_le_bytes());
446        header[56..58].copy_from_slice(&3_u16.to_le_bytes());
447        let program_header = |kind: u32, offset: u64, vaddr: u64, filesz: u64, memsz: u64| {
448            let mut entry = [0_u8; 56];
449            entry[..4].copy_from_slice(&kind.to_le_bytes());
450            entry[8..16].copy_from_slice(&offset.to_le_bytes());
451            entry[16..24].copy_from_slice(&vaddr.to_le_bytes());
452            entry[32..40].copy_from_slice(&filesz.to_le_bytes());
453            entry[40..48].copy_from_slice(&memsz.to_le_bytes());
454            entry
455        };
456        let entries = [
457            program_header(6, 64, 0x400040, 0xa8, 0xa8), // PT_PHDR, not loaded
458            program_header(1, 0, 0x400000, 0x1000, 0x1000),
459            program_header(1, 0x2e10, 0x403e10, 0x290, 0x12a0),
460        ];
461        let read_at = |offset: u64, buffer: &mut [u8]| {
462            let index = usize::try_from((offset - 64) / 56).ok()?;
463            buffer.copy_from_slice(entries.get(index)?);
464            Some(())
465        };
466        let layout = ElfLayout::parse(&header, read_at).unwrap();
467        assert_eq!(

476        assert_eq!(layout.start_brk(0x400040), Ok(0x406000));

479        let aligned = ElfLayout {

482        };
483        assert_eq!(aligned.start_brk(0x400040), Ok(0x406000));
484        // A position-independent image with an interpreter at its bias.
485        let dynamic = ElfLayout {
486            position_independent: true,
487            interpreter: true,

489        };
490        assert_eq!(
491            dynamic.start_brk(0x5555_5555_4000 + 0x400040),
492            Ok(0x5555_5595_a000)
493        );
494        // Without an interpreter (a static PIE, or the dynamic loader run
495        // directly), the kernel's build decides: no model.

500        assert!(static_pie.start_brk(0x5555_5555_4000 + 0x400040).is_err());
501        // Program headers outside every load: Linux's AT_PHDR is the bias.
502        let unloaded = ElfLayout { phdr: 0, ..layout };
503        assert_eq!(unloaded.start_brk(0), Ok(0x406000));
504        // A PT_INTERP entry and ET_DYN are read from the headers.
505        header[16..18].copy_from_slice(&3_u16.to_le_bytes());
506        header[56..58].copy_from_slice(&4_u16.to_le_bytes());
507        let with_interp = [
508            entries[0],
509            program_header(3, 0x2a8, 0x4002a8, 0x1c, 0x1c), // PT_INTERP
510            entries[1],

514            let index = usize::try_from((offset - 64) / 56).ok()?;
515            buffer.copy_from_slice(with_interp.get(index)?);

530    #[test]
531    fn the_kernels_start_of_the_break_is_what_the_layout_gives() {

533        let mut command = std::process::Command::new("/bin/sh");

535            .args(["-c", "read line"])

541                    return Err(std::io::Error::last_os_error());

633    #[test]
634    fn a_script_resolves_to_its_interpreter_and_a_link_to_its_file() {
635        let directory = tempfile::tempdir().unwrap();
636        let root = fs::canonicalize(directory.path()).unwrap();
637        let interpreter = std::env::current_exe().unwrap();
638        let interpreter = fs::canonicalize(interpreter).unwrap();
639        let script = root.join("script");
640        fs::write(&script, format!("#!{}\n", interpreter.display())).unwrap();
641        fs::set_permissions(&script, fs::Permissions::from_mode(0o755)).unwrap();
642        std::os::unix::fs::symlink("script", root.join("link")).unwrap();
643
644        let record = ExecRecord::resolve(Path::new("/"), &root, Path::new("link")).unwrap();
645        assert_eq!(record.exe, interpreter);
646        assert!(record.is_present(Path::new("/")));
647
648        // The same file seen below another root is named from that root.
649        let image = root.join("image");
650        fs::copy(&interpreter, &image).unwrap();
651        let record = ExecRecord::resolve(&root, Path::new("/"), Path::new("/image")).unwrap();
652        assert_eq!(record.exe, Path::new("/image"));
653        assert!(record.is_present(&root));
654
655        // A file removed or replaced at the path is no longer the image.
656        fs::remove_file(&image).unwrap();
657        assert!(!record.is_present(&root));
658        fs::copy(&interpreter, &image).unwrap();
659        assert!(!record.is_present(&root));
660
661        // The same inode put back at the path after a rename is not the
662        // image's link any more: Linux's link follows the renamed name.
663        let image = root.join("image");
664        let record = ExecRecord::resolve(&root, Path::new("/"), Path::new("/image")).unwrap();
665        fs::hard_link(&image, root.join("alias")).unwrap();
666        fs::rename(&image, root.join("moved")).unwrap();
667        fs::hard_link(root.join("alias"), &image).unwrap();
668        assert!(!record.is_present(&root));
669
670        // With its link count back where it was, a rename still changed
671        // the inode's change time.
672        let image = root.join("relinked");
673        fs::copy(&interpreter, &image).unwrap();
674        fs::hard_link(&image, root.join("second")).unwrap();
675        let record = ExecRecord::at(&root, &image).unwrap();
676        fs::rename(&image, root.join("away")).unwrap();
677        fs::rename(root.join("second"), &image).unwrap();
678        assert_eq!(fs::metadata(&image).unwrap().nlink(), 2);
679        assert!(!record.is_present(&root));
680
681        // Linux refuses to exec anything but a regular file (EACCES) before
682        // it opens it, and so does the resolution: an open of a FIFO with no
683        // writer would wait forever.
684        let fifo = root.join("fifo");
685        let fifo_name = std::ffi::CString::new(fifo.as_os_str().as_bytes()).unwrap();
686        assert_eq!(unsafe { libc::mkfifo(fifo_name.as_ptr(), 0o755) }, 0);
687        assert_eq!(
688            ExecRecord::resolve(&root, Path::new("/"), Path::new("/fifo")),
689            None
690        );
691        assert_eq!(
692            ExecRecord::resolve(Path::new("/"), Path::new("/"), Path::new("/dev/null")),
693            None
694        );
695
696        assert_eq!(
697            ExecRecord::resolve(&root, Path::new("/"), Path::new("/missing")),
698            None
699        );
700        // A file whose header the caller cannot read is the kernel's to
701        // resolve (Linux runs an execute-only script's interpreter): its
702        // header reads as unread, never as not a script.
703        assert_eq!(script_interpreter(&root), Script::Unread);
704    }
