9//! A5 step 3c.1d, design D37 (post-facto-human-review trigger 3): the ignored
10//! signals the in-guest runtime's mask holds pending, under the in-guest
11//! delivery gate (`Config::in_guest_signal_delivery`; dev-hermit
12//! ai_docs/2026-10-16-a5-signal-delivery-design.md, "3c.1 implementation plan,
13//! revision 4", D37 and IR-7, <https://github.com/rrnewton/hermit/issues/4485>).
14//!
15//! Linux discards an ignored signal when it is sent unless it is blocked
16//! (`sig_ignored`, kernel/signal.c:104-120 at v6.12: "Blocked signals are never
17//! ignored"), and ignored means `SIG_IGN`, or `SIG_DFL` for a signal whose
18//! default is to ignore it (`sig_handler_ignored`, :75-80;
19//! `SIG_KERNEL_IGNORE_MASK`, include/linux/signal.h:434-436: SIGCONT,
20//! SIGCHLD, SIGWINCH, SIGURG). One that is blocked is queued, and
21//! `get_signal` discards it once a mask lets it be dequeued (:2815, :2830).
22//! Once the in-guest runtime runs under a mask of its own (step 3c.1e), every
23//! signal sent to a guest thread is blocked when it arrives, so ignored ones
24//! that Linux discarded at send are queued too. This module names which of
25//! them Linux would not hold at a given point, so Detcore can remove them
26//! and keep them out of `/proc`:
27//!
28//! - `I` = the pending signals that are ignored, under the actions in force;
29//! - `M_req` = the guest mask in force when an instance was queued;
30//! - `M_next` = the mask that governs next;
31//! - remove `I ∖ (M_req ∩ M_next)`: an instance outside `M_req` was discarded
32//!   by Linux when it was sent, and one outside `M_next` is discarded by Linux
33//!   when that mask lets it be dequeued (at the return, or at the transparent
34//!   restart of `pselect6`, `ppoll` and `rt_sigsuspend`, which return
35//!   `-ERESTARTNOHAND` for it, kernel/signal.c:157-166 counting it as pending
36//!   whatever its action, F21).
37//!
38//! The one place where Linux shows the guest something for such a signal is
39//! a masked `epoll_pwait` that would wait: `ep_poll` returns `-EINTR` for any
40//! pending signal the temporary mask leaves unblocked (fs/eventpoll.c:
41//! 2017-2018), and keeps that mask until the signal is dequeued, and discarded
42//! (:2495). That is [`epoll_interrupting`].
43
44use crate::syscalls::KernelSignalState;
45use crate::syscalls::ignored_at_generation;
46
47/// `I` of design D37: the signals pending for the thread (`SigPnd ∪ ShdPnd`)
48/// whose action is ignored (`SigIgn`, or the default for SIGCHLD, SIGURG,
49/// SIGWINCH or SIGCONT with no handler: `ignored_at_generation`), less
50/// `reserved`, the in-guest runtime's reserved signals, whose action is the
51/// runtime's and which a Detcore removal must never dequeue.
52pub(crate) fn pending_ignored(state: &KernelSignalState, reserved: u64) -> u64 {
53    state.pending & ignored_at_generation(state.ignored, state.caught) & !reserved
54}
55
56/// D37's removal at a point whose instances were queued under `m_req` and
57/// which `m_next` governs next: `I ∖ (M_req ∩ M_next)`.
58pub(crate) fn removable(state: &KernelSignalState, m_req: u64, m_next: u64, reserved: u64) -> u64 {
59    pending_ignored(state, reserved) & !(m_req & m_next)
60}
61
62/// D37's exception at the park of a masked `epoll_pwait` or `epoll_pwait2`
63/// with a nonzero timeout and nothing ready at its first probe (F21, IR-3):
64/// the ignored signals pending at the call's entry that the guest's own mask
65/// `m_req` blocked and its temporary mask `m_temp` lets in, `I ∩ M_req ∖
66/// M_temp`. Linux returns `-EINTR` for them at once and discards them as the
67/// call returns; Detcore removes them and returns `EINTR`. The ignored ones
68/// outside `m_req` were discarded by Linux when they were sent, so they end
69/// nothing (they are in [`removable`] all the same).
70pub(crate) fn epoll_interrupting(
71    state: &KernelSignalState,
72    m_req: u64,
73    m_temp: u64,
74    reserved: u64,
75) -> u64 {
76    pending_ignored(state, reserved) & m_req & !m_temp
77}
78
79/// The bits a `/proc` status under the gate leaves out of `SigPnd` and
80/// `ShdPnd`: the pending ignored signals the thread's published guest mask
81/// `m_req` does not block, `I ∖ M_req`, which Linux discarded when they were
82/// sent (D37's visibility rule). `pending` is the field's own set; `sig_ign`
83/// and `sig_cgt` are the status's `SigIgn` and `SigCgt`, the actions in force
84/// (fs/proc/array.c:264-300 at v6.12 renders all five from one lock of the
85/// thread's signal state). A reserved signal of the in-guest runtime is never
86/// in it: its `SigCgt` bit is the runtime's handler.
87pub(crate) fn hidden_from_status(pending: u64, sig_ign: u64, sig_cgt: u64, m_req: u64) -> u64 {
88    pending & ignored_at_generation(sig_ign, sig_cgt) & !m_req
89}

91#[cfg(test)]
92mod tests {
93    use super::*;
94
95    fn bit(signal: i32) -> u64 {
96        1_u64 << (signal - 1)
97    }
98
99    fn state(pending: u64, ignored: u64, caught: u64) -> KernelSignalState {
100        KernelSignalState {
101            pending,
102            thread_pending: pending,
103            shared_pending: 0,
104            blocked: !0,
105            ignored,
106            caught,
107        }
108    }
109
110    /// `I` is `SIG_IGN`, or the default of a default-ignored signal with no
111    /// handler; a caught default-ignored signal, a default-fatal one and a
112    /// reserved one are not in it. Mutation "default-ignored counted even
113    /// when caught": the caught SIGCHLD row fails.
114    #[test]
115    fn pending_ignored_is_sig_ign_or_an_uncaught_default_ignored_signal() {
116        let urg = bit(libc::SIGURG);
117        let chld = bit(libc::SIGCHLD);
118        let term = bit(libc::SIGTERM);
119        let usr1 = bit(libc::SIGUSR1);
120        let s = state(urg | chld | term | usr1, usr1, chld);
121        assert_eq!(pending_ignored(&s, 0), urg | usr1);
122        assert_eq!(pending_ignored(&s, usr1), urg);
123    }
124
125    /// `I ∖ (M_req ∩ M_next)`: kept only when both masks block it. Rows of
126    /// design D37 and IR-7: an instance pending at entry under the own mask
127    /// and a temporary mask that unblocks it (removed: the transparent
128    /// restart discards it); one both block (kept); one only the temporary
129    /// mask blocks, sent during the park (`M_req` is then the temporary
130    /// mask, kept); one neither blocks (removed: discarded at send).
131    /// Mutation "(i) using only `M_next`": the first row keeps it.
132    #[test]
133    fn removable_is_i_minus_the_intersection_of_the_two_masks() {
134        let urg = bit(libc::SIGURG);
135        let s = state(urg, 0, 0);
136        assert_eq!(
137            removable(&s, urg, 0, 0),
138            urg,
139            "own blocks, temporary lets in"
140        );
141        assert_eq!(removable(&s, urg, urg, 0), 0, "both block");
142        assert_eq!(removable(&s, 0, urg, 0), urg, "the own mask lets it in");
143        assert_eq!(removable(&s, 0, 0, 0), urg, "neither blocks");
144    }
145
146    /// The `epoll` exception: only an ignored instance the own mask blocks
147    /// and the temporary mask lets in. Mutation "every removable instance
148    /// interrupts": the own-mask-unblocked row returns EINTR where Linux,
149    /// which discarded it at send, waits.
150    #[test]
151    fn the_epoll_exception_is_i_within_the_own_mask_outside_the_temporary_one() {
152        let urg = bit(libc::SIGURG);
153        let s = state(urg, 0, 0);
154        assert_eq!(epoll_interrupting(&s, urg, 0, 0), urg);
155        assert_eq!(epoll_interrupting(&s, 0, 0, 0), 0);
156        assert_eq!(epoll_interrupting(&s, urg, urg, 0), 0);
157    }
158
159    /// The status subtraction leaves a blocked ignored signal and every
160    /// signal that is not ignored. Mutation "the subtraction ignores
161    /// `M_req`": the blocked SIGURG row disappears.
162    #[test]
163    fn the_status_hides_only_ignored_signals_the_published_mask_lets_in() {
164        let urg = bit(libc::SIGURG);
165        let term = bit(libc::SIGTERM);
166        assert_eq!(hidden_from_status(urg | term, 0, 0, 0), urg);
167        assert_eq!(hidden_from_status(urg | term, 0, 0, urg), 0);
168        assert_eq!(hidden_from_status(term, term, 0, 0), term);
169        assert_eq!(hidden_from_status(urg, 0, urg, 0), 0);
170    }
171}
