69/// Take a single-valued selection flag, refusing a second occurrence.
70///
71/// ⚠️ LAST-VALUE-WINS SILENTLY UNDID THIS FILE'S OWN GUARD, WHICH IS WHY THIS EXISTS.
72/// `plan` refuses a `--test` naming no known id. With plain assignment a SECOND
73/// `--test` overwrote the first, so the unknown one was never looked up at all:
74///
75/// ```text
76/// plan --lane portable --test no-such-test-xyz                             rc=2
77/// plan --lane portable --test no-such-test-xyz --test applications/...     rc=0   []
78/// plan --lane portable --test applications/... --test no-such-test-xyz     rc=2
79/// ```
80///
81/// Measured 2026-08-26 at `979a50b17a75` by `agent(codex-rev-2686)` and confirmed
82/// independently by `agent(hermit-012)` and by me. The asymmetry is the tell: the
83/// same two ids in the other order refuse, because only the LAST occurrence is ever
84/// examined. A bisection driver reading rc=0 there sees "nothing failed" for a list
85/// containing an id that does not exist -- the exact silent green this guard was
86/// added to remove, reappearing one layer up in the argument parser.
87///
88/// `--jobs` already refused a repeat; the selection flags did not. Refusing is right
89/// rather than taking the first or the last, because a repeated selector has no
90/// defensible meaning: the caller asked for two different things and we cannot serve
91/// both from one field.
92fn set_once(slot: &mut Option<String>, values: &mut impl Iterator<Item = String>, flag: &str) {
93    let value = required_value(values, flag);
94    if slot.replace(value).is_some() {
95        fail(format!(
96            "{flag} may be specified only once; a repeat silently overwrote the first \
97             value, so an earlier id was never validated"
98        ));
99    }
100}

102fn parse(mut values: impl Iterator<Item = String>) -> Args {
103    let mut args = Args {
104        format: "text".into(),
105        ..Args::default()
106    };
107    while let Some(value) = values.next() {
108        match value.as_str() {
109            "--lane" => set_once(&mut args.selection.lane, &mut values, "--lane"),
110            "--category" => set_once(&mut args.selection.category, &mut values, "--category"),
111            "--test" => set_once(&mut args.selection.test, &mut values, "--test"),
112            "--mode" => set_once(&mut args.selection.mode, &mut values, "--mode"),
113            "--backend" => set_once(&mut args.selection.backend, &mut values, "--backend"),
114            "--ci-only" => {
115                args.ci_only = true;
116                args.selection.population = Some(Population::Required);
117            }
118            "--include-occasional" => args.selection.include_occasional = true,
119            "--include-manual" => args.selection.include_manual = true,
120            "--probe-disabled" => {
121                args.probe_disabled = true;
122                args.selection.population = Some(Population::Disabled);
123            }
124            "--prebuilt" => args.prebuilt = true,
125            "--allow-empty" => args.allow_empty = true,
126            "--results" => {
127                args.results = Some(PathBuf::from(required_value(&mut values, "--results")))
128            }
129            "--junit" => args.junit = Some(PathBuf::from(required_value(&mut values, "--junit"))),
130            "--format" => args.format = required_value(&mut values, "--format"),
131            "--jobs" => {
132                let value = required_value(&mut values, "--jobs");
133                let jobs = value
134                    .parse::<usize>()
135                    .ok()
136                    .filter(|jobs| *jobs > 0)
137                    .unwrap_or_else(|| fail("--jobs requires a positive integer"));
138                if args.jobs.replace(jobs).is_some() {
139                    fail("--jobs may be specified only once");
140                }
141            }
142            other => fail(format!("unknown option {other}")),
143        }
144    }
145    args
146}

196fn host_inapplicable_reason(
197    requires: &[String],
198    verdicts: &HostCapabilities,
199) -> Option<(Vec<String>, String)> {
200    let mut absent = requires
201        .iter()
202        .filter_map(|token| requires_capability(token).ok().flatten())
203        .filter_map(|capability| {
204            verdicts
205                .get(&capability)
206                .filter(|verdict| !verdict.present)
207                .map(|verdict| (capability.value().to_string(), verdict.evidence.clone()))
208        })
209        .collect::<Vec<_>>();
210    absent.sort();
211    absent.dedup();
212    if absent.is_empty() {
213        return None;
214    }
215    let capabilities = absent
216        .iter()
217        .map(|(capability, _)| capability.clone())
218        .collect::<Vec<_>>();
219    let reason = format!(
220        "NOT RUN, NOT a pass, no coverage: this machine lacks {}",
221        absent
222            .iter()
223            .map(|(capability, evidence)| format!("{capability} ({evidence})"))
224            .collect::<Vec<_>>()
225            .join(", ")
226    );
227    Some((capabilities, reason))
228}

244fn validate_args(command: &str, args: &Args) {
245    if !matches!(args.format.as_str(), "text" | "json") {
246        fail(format!("invalid format {}", args.format));
247    }
248    if args
249        .selection
250        .lane
251        .as_deref()
252        .is_some_and(|lane| !matches!(lane, "portable" | "privileged"))
253    {
254        fail("--lane must be portable or privileged");
255    }
256    if args
257        .selection
258        .mode
259        .as_deref()
260        .is_some_and(|mode| !matches!(mode, "verify" | "chaos" | "replay" | "naked" | "custom"))
261    {
262        fail("--mode must be verify, chaos, replay, naked, or custom");
263    }
264    if args
265        .selection
266        .backend
267        .as_deref()
268        .is_some_and(|backend| !matches!(backend, "ptrace" | "dbt" | "kvm" | "sabre" | "liteinst"))
269    {
270        fail("--backend must name a Hermit backend");
271    }
272    if command == "build" && args.prebuilt {
273        fail("build does not accept --prebuilt");
274    }
275    if command != "run" && args.jobs.is_some() {
276        fail("--jobs is accepted by run only");
277    }
278    if args.selection.include_manual
279        && (args.selection.test.is_none() || args.selection.mode.is_none())
280    {
281        fail("--include-manual requires exact --test and --mode filters");
282    }
283    if args.probe_disabled {
284        if command != "run" {
285            fail("--probe-disabled is accepted by run only");
286        }
287        if args.selection.test.is_none()
288            || args.selection.mode.is_none()
289            || args.selection.backend.is_none()
290        {
291            fail("--probe-disabled requires exact --test, --mode, and --backend filters");
292        }
293        if args.selection.include_manual || args.ci_only {
294            fail("--probe-disabled is mutually exclusive with --include-manual and --ci-only");
295        }
296    }
297    if args.allow_empty {
298        if !args.ci_only {
299            fail("--allow-empty requires --ci-only");
300        }
301        match command {
302            "build" if args.selection.lane.is_some() || args.selection.category.is_some() => {}
303            "run" if args.selection.category.is_some() => {}
304            "build" => fail("build --allow-empty requires an explicit --lane or --category"),
305            "run" => fail("run --allow-empty requires an explicit --category"),
306            _ => fail("--allow-empty is accepted by build and run only"),
307        }
308    }
309}

311fn main() -> ExitCode {
312    let mut values = std::env::args().skip(1);
313    let command = values.next().unwrap_or_else(|| fail("missing command"));
314    let values = values.collect::<Vec<_>>();
315    if command == "expected-plan" && !values.is_empty() {
316        fail("expected-plan accepts no options");
317    }
318    let args = parse(values.into_iter());
319    validate_args(&command, &args);
320    let root = root();
321    let manifests = ManifestSet::load(&root).unwrap_or_else(|error| fail(error));
322    // One front-door schema/inventory authority governs every command, not
323    // only the metadata gate. This prevents a direct/manual run from accepting
324    // a recipe that the canonical manifest planner would refuse.
325    run_manifest_plan(&root);
326    match command.as_str() {
327        "validate" => validate(&root, &manifests),
328        "plan" => print_plan(&manifests, &args, Population::Required),
329        "expected-plan" => print_expected_plan(&root, &manifests),
330        "audit-gaps" => print_plan(&manifests, &args, Population::Disabled),
331        "audit-inventory" | "audit-test-binary-registration" => ExitCode::SUCCESS,
332        "audit-test-footprints" => {
333            run_audit(
334                &root,
335                &root.join("target/debug/generate-test-footprints"),
336                &["--check"],
337            );
338            ExitCode::SUCCESS
339        }
340        "audit-ci" => {
341            audit_dag_correspondence(&root, &manifests).unwrap_or_else(|error| fail(error));
342            audit_budget_ordering(&root).unwrap_or_else(|error| fail(error));
343            audit_expected_plan(&root, &manifests);
344            ExitCode::SUCCESS
345        }
346        "build" => build(&root, &manifests, &args),
347        "audit-compile" => audit_compile(&root, &manifests, &args),
348        "run" => run(&root, &manifests, &args),
349        other => fail(format!("unknown command {other}")),
350    }
351}

353fn validate(root: &Path, manifests: &ManifestSet) -> ExitCode {
354    // Keep the existing Rust manifest-plan front door as the authority for
355    // inventory, schema, lane, workflow, and DAG consistency.  The cell
356    // runner owns execution; it must not silently narrow `validate` to only
357    // the expected-plan comparison during the shell removal.
358    audit_dag_correspondence(root, manifests).unwrap_or_else(|error| fail(error));
359    audit_budget_ordering(root).unwrap_or_else(|error| fail(error));
360    run_audit(
361        root,
362        &root.join("target/debug/generate-test-footprints"),
363        &["--check"],
364    );
365    run_audit(
366        root,
367        &root.join("tests/backend-parity/split_asymmetric_pr.py"),
368        &["--self-test"],
369    );
370    audit_determinism_stress_evidence(root);
371    run_audit(root, &root.join("tests/manifest-cli.rs"), &["self-test"]);
372    // The DBT budget wrapper gates roughly twenty portable nodes and fails
373    // CLOSED on a pin it is not calibrated for. Nothing else notices: a
374    // truncated node reads like a fast one. This asserts end to end that the
375    // wrapper still REACHES its wrapped command at the recorded pin.
376    run_audit(
377        root,
378        &root.join("ci/run-with-reverie-dbt-budget-test.sh"),
379        &[],
380    );
381    run_audit(
382        root,
383        &root.join("ci/compat-envelope/scorecard.rs"),
384        &["self-test-and-check"],
385    );
386    run_audit(
387        root,
388        &root.join("ci/compat-envelope/pressure-test.rs"),
389        &["self-test"],
390    );
391    // The removed shell front door accumulated plan/scheduler/receipt guards
392    // that now belong to the Rust validate driver.  Exercise those brackets
393    // here without executing the validation DAG; otherwise deleting the shell
394    // would also silently delete its protection against incomplete plans.
395    run_audit(root, &root.join("scripts/validate.rs"), &["--self-test"]);
396    audit_cli_brackets(root);
397    let cells = audit_expected_plan(root, manifests);
398    println!(
399        "PASS: {} YAML manifests, {} required cells",
400        manifests.documents.len(),
401        cells
402    );
403    ExitCode::SUCCESS
404}

406fn audit_cli_brackets(root: &Path) {
407    let executable = std::env::current_exe().unwrap_or_else(|error| fail(error));
408    for option in [
409        "--lane",
410        "--category",
411        "--test",
412        "--mode",
413        "--backend",
414        "--results",
415        "--junit",
416        "--format",
417        "--jobs",
418    ] {
419        let status = Command::new(&executable)
420            .args(["plan", option])
421            .current_dir(root)
422            .stdout(Stdio::null())
423            .stderr(Stdio::null())
424            .status()
425            .unwrap_or_else(|error| fail(error));
426        if status.success() {
427            fail(format!("missing value for {option} was accepted"));
428        }
429        let status = Command::new(&executable)
430            .args(["plan", option, ""])
431            .current_dir(root)
432            .stdout(Stdio::null())
433            .stderr(Stdio::null())
434            .status()
435            .unwrap_or_else(|error| fail(error));
436        if status.success() {
437            fail(format!("empty value for {option} was accepted"));
438        }
439    }
440    let output = Command::new(&executable)
441        .args([
442            "plan",
443            "--lane",
444            "portable",
445            "--ci-only",
446            "--format",
447            "json",
448        ])
449        .current_dir(root)
450        .output()
451        .unwrap_or_else(|error| fail(error));
452    let cells = serde_json::from_slice::<Vec<JsonValue>>(&output.stdout).unwrap_or_default();
453    if !output.status.success() || cells.is_empty() {
454        fail("complete CLI control was refused or selected no cells");
455    }
456    for argv in [
457        vec!["run", "--jobs", "0"],
458        vec!["run", "--jobs", "not-a-number"],
459        vec!["run", "--jobs", "2", "--jobs", "3"],
460        vec!["plan", "--jobs", "2"],
461    ] {
462        let status = Command::new(&executable)
463            .args(argv)
464            .current_dir(root)
465            .stdout(Stdio::null())
466            .stderr(Stdio::null())
467            .status()
468            .unwrap_or_else(|error| fail(error));
469        if status.success() {
470            fail("invalid --jobs control was accepted");
471        }
472    }
473}

536fn unique_plan_rows(label: &str, rows: Vec<JsonValue>) -> Result<BTreeSet<String>, String> {
537    let physical = rows.len();
538    let mut identities = BTreeSet::new();
539    let mut duplicates = BTreeSet::new();
540    let mut normalized = BTreeSet::new();
541    for row in rows {
542        let identity = PlanCellIdentity::from_json(&row)?;
543        if !identities.insert(identity.clone()) {
544            duplicates.insert(identity);
545        }
546        normalized.insert(serde_json::to_string(&row).map_err(|error| error.to_string())?);
547    }
548    if !duplicates.is_empty() {
549        let names = duplicates
550            .iter()
551            .map(PlanCellIdentity::display)
552            .collect::<Vec<_>>()
553            .join(", ");
554        return Err(format!(
555            "{label} contains {physical} physical rows but only {} unique identities; duplicate identities: {names}",
556            identities.len()
557        ));
558    }
559    Ok(normalized)
560}

593    });
594    (cells.len(), actual)

597fn expected_plan_document(root: &Path, manifests: &ManifestSet) -> JsonValue {
598    let (_, cells) = required_plan_rows(manifests);
599    let mut remaining = cells
600        .into_iter()
601        .map(|row| {
602            let identity = PlanCellIdentity::from_json(&row)
603                .expect("required plan rows have complete identities");
604            (identity, row)
605        })
606        .collect::<BTreeMap<_, _>>();
607    let mut cells = Vec::with_capacity(remaining.len());
608    let path = root.join("ci/expected-e2e-plan.json");
609    if let Ok(source) = fs::read(&path) {
610        let current: JsonValue = serde_json::from_slice(&source)
611            .unwrap_or_else(|error| fail(format!("cannot parse {}: {error}", path.display())));
612        let current = current["cells"]
613            .as_array()
614            .unwrap_or_else(|| fail(format!("{} has no cells array", path.display())));
615        let mut seen = BTreeSet::new();
616        for row in current {
617            let identity = PlanCellIdentity::from_json(row)
618                .unwrap_or_else(|error| fail(format!("{}: {error}", path.display())));
619            if !seen.insert(identity.clone()) {
620                fail(format!(
621                    "{} contains duplicate identity {}",
622                    path.display(),
623                    identity.display()
624                ));
625            }
626            if let Some(row) = remaining.remove(&identity) {
627                cells.push(row);
628            }
629        }
630    }
631    cells.extend(remaining.into_values());
632    serde_json::json!({
633        "schema": EXPECTED_PLAN_SCHEMA,
634        "cells": cells,
635    })
636}

669fn run_audit(root: &Path, program: &Path, args: &[&str]) {
670    let status = Command::new(program)
671        .args(args)
672        .current_dir(root)
673        .status()
674        .unwrap_or_else(|error| fail(format!("cannot execute {}: {error}", program.display())));
675    if !status.success() {
676        // 127 IS AN ENVIRONMENT FAULT, NOT A FAILED AUDIT, AND SAYING "failed"
677        // FOR BOTH COSTS A CI RUN ITS WHOLE E2E COVERAGE. These programs carry
678        // `#!/usr/bin/env -S rust-script --force`; when that interpreter is
679        // absent the kernel never runs the script and the shell reports 127.
680        // The old message -- "tests/manifest-cli.rs self-test failed" -- reads
681        // as the self-test having run and found a defect. Measured on hermit
682        // run 32512027583: it had not run at all, and because build-debug gates
683        // every e2e shard, a missing tool was read as a product break.
684        if status.code() == Some(127) {
685            fail(format!(
686                "cannot run {}: exited 127, which means its interpreter was not found, \
687                 not that the audit failed. This program runs under \
688                 `#!/usr/bin/env -S rust-script --force`; install it with \
689                 `cargo install rust-script` or put it on PATH (on a dev box it is \
690                 usually ~/.cargo/bin, which a non-login shell does not inherit).",
691                program.display()
692            ));
693        }
694        fail(format!("{} {} failed", program.display(), args.join(" ")));
695    }

719fn command_jobs(command: &str) -> Result<Option<i64>, String> {
720    let words = command.split_whitespace().collect::<Vec<_>>();
721    let mut jobs = None;
722    let mut index = 0;
723    while index < words.len() {
724        if words[index] == "--jobs" {
725            let value = words
726                .get(index + 1)
727                .ok_or_else(|| "manifest command has --jobs without a value".to_string())?
728                .parse::<i64>()
729                .ok()
730                .filter(|value| *value > 0)
731                .ok_or_else(|| "manifest command has invalid --jobs value".to_string())?;
732            if jobs.replace(value).is_some() {
733                return Err("manifest command repeats --jobs".into());
734            }
735            index += 1;
736        }
737        index += 1;
738    }
739    Ok(jobs)
740}

742fn audit_dag_correspondence(root: &Path, manifests: &ManifestSet) -> Result<(), String> {
743    for lane in ["portable", "privileged"] {
744        let path = root.join(format!("ci/dag/{lane}.json"));
745        let dag = read_dag(&path)?;
746        if dag
747            .steps
748            .iter()
749            .any(|step| step.cmd.contains("test_harness.sh"))
750        {
751            return Err(format!(
752                "{} still invokes the removed shell harness",
753                path.display()
754            ));
755        }
756        let mut ids = BTreeSet::new();
757        for step in &dag.steps {
758            let id = format!("{}.{}", step.group, step.job);
759            if !ids.insert(id.clone()) {
760                return Err(format!("{} contains duplicate node {id}", path.display()));
761            }
762        }
763        for step in &dag.steps {
764            for dependency in &step.deps {
765                if !ids.contains(dependency) {
766                    return Err(format!(
767                        "{} node {}.{} names missing dependency {dependency}",
768                        path.display(),
769                        step.group,
770                        step.job
771                    ));
772                }
773            }
774        }
775        if dag
776            .steps
777            .iter()
778            .filter(|step| step.cmd == "target/debug/test-harness validate")
779            .count()
780            != 1
781        {
782            return Err(format!(
783                "{} must contain exactly one Rust metadata validation node",
784                path.display()
785            ));
786        }
787        let build =
788            format!("target/debug/test-harness build --lane {lane} --ci-only --allow-empty");
789        if dag.steps.iter().filter(|step| step.cmd == build).count() != 1 {
790            return Err(format!(
791                "{} must contain exactly one Rust manifest build node",
792                path.display()
793            ));
794        }
795        let expected = manifests
796            .documents
797            .iter()
798            .filter(|document| document.test.iter().any(|test| test.lane == lane))
799            .map(|document| document.bucket.clone())
800            .collect::<BTreeSet<_>>();
801        let mut actual = BTreeSet::new();
802        for step in dag
803            .steps
804            .iter()
805            .filter(|step| step.group == "e2e" && step.job.starts_with("manifest_"))
806        {
807            let manifest = step.manifest.as_ref().ok_or_else(|| {
808                format!("{}.{} lacks typed manifest identity", step.group, step.job)
809            })?;
810            let dagrun::DagManifest {
811                lane: manifest_lane,
812                category,
813            } = manifest;
814            if manifest_lane != lane {
815                return Err(format!(
816                    "{}.{} records lane {} in the {lane} DAG",
817                    step.group, step.job, manifest_lane
818                ));
819            }
820            let selector = format!(
821                "target/debug/test-harness run --lane {lane} --category {} --ci-only --allow-empty --prebuilt",
822                category
823            );
824            if !step.cmd.contains(&selector) {
825                return Err(format!(
826                    "{}.{} does not execute its typed selector literally",
827                    step.group, step.job
828                ));
829            }
830            if let Some(jobs) = command_jobs(&step.cmd)? {
831                let demand = step
832                    .hint
833                    .resources
834                    .get("manifest_guest")
835                    .copied()
836                    .unwrap_or(0);
837                let cap = dag
838                    .resource_caps
839                    .get("manifest_guest")
840                    .copied()
841                    .unwrap_or(0);
842                if demand != jobs
843                    || cap < jobs
844                    || step.hint.preferred_inner_jobs != Some(jobs)
845                    || step.jobs_flag.as_deref() != Some("")
846                {
847                    return Err(format!(
848                        "{}.{} runs --jobs {jobs} but declares manifest_guest={demand}, cap={cap}, preferred_inner_jobs={:?}, jobs_flag={:?}",
849                        step.group, step.job, step.hint.preferred_inner_jobs, step.jobs_flag
850                    ));
851                }
852            }
853            if !actual.insert(category.clone()) {
854                return Err(format!(
855                    "{} has duplicate manifest bucket {}",
856                    path.display(),
857                    category
858                ));
859            }
860        }
861        if actual != expected {
862            return Err(format!(
863                "{} manifest buckets differ: expected={expected:?} actual={actual:?}",
864                path.display()
865            ));
866        }
867    }
868    Ok(())

873    let privileged = read_dag(&root.join("ci/dag/privileged.json"))?;

875        for step in &dag.steps {

878                    "{lane} node {}.{} has no derivable wall budget",

882            if lane == "portable" {

888                    .next()

898                        "{lane} node {}.{} declares CARGO_BUILD_JOBS={jobs} without matching preferred_inner_jobs",

900                    ));
901                }
902            }
903        }
904    }
905
906    let portable_workflow = parse_yaml(&root.join(".github/workflows/ci-portable.yml"))?;
907    let debug_bound = workflow_job_timeout(&portable_workflow, "test-debug")? * 60;

920        ("debug_shards", "test-debug", debug_bound),

924            .as_array()
925            .into_iter()
926            .flatten()
927            .flat_map(|shard| shard["nodes"].as_array().into_iter().flatten())
928        {
929            let node = node
930                .as_str()
931                .ok_or_else(|| format!("{key} contains a non-string node"))?;
932            let step = portable_steps
933                .get(node)
934                .ok_or_else(|| format!("portable shard names missing DAG node {node}"))?;
935            let timeout = u64::try_from(step.timeout).map_err(|_| {
936                format!("portable node {node} has invalid timeout {}", step.timeout)
937            })?;
938            if timeout >= bound {
939                current.insert(format!(
940                    "{node} {timeout}s >= {bound}s (job {job} timeout-minutes)"
941                ));
942            }
943        }
944    }

946    let privileged_workflow = fs::read_to_string(root.join(".github/workflows/ci-privileged.yml"))
947        .map_err(|e| e.to_string())?;

950        .matches("continue-on-error: true")

952        != 1
953    {
954        return Err(
955            "privileged workflow must contain exactly one diagnostic continue-on-error".into(),
956        );
957    }

959        .lines()
960        .find(|line| line.contains("ci/run-dag.sh privileged"))
961        .and_then(|line| {
962            let words = line.split_whitespace().collect::<Vec<_>>();

964                .iter()
965                .position(|word| *word == "env")
966                .and_then(|index| index.checked_sub(1))
967                .and_then(|index| words[index].strip_suffix('s'))
968                .and_then(|value| value.parse::<u64>().ok())
969        })
970        .ok_or_else(|| "cannot derive privileged launcher timeout".to_string())?;
971    let privileged_yaml = parse_yaml(&root.join(".github/workflows/ci-privileged.yml"))?;
972    let privileged_job_bound = workflow_job_timeout(&privileged_yaml, "privileged")? * 60;
973    let declared_step_budgets = workflow_step_timeout_sum(&privileged_yaml, "privileged")?;

975        return Err(format!(
976            "privileged job {privileged_job_bound}s must exceed {declared_step_budgets}s of explicit inner step budgets"
977        ));
978    }
979    let critical_path = dag_critical_path(&privileged)?;

981        return Err(format!(
982            "privileged launcher {launcher_bound}s must exceed {critical_path}s DAG critical path plus 30s runner overhead"
983        ));
984    }
985    for step in &privileged.steps {
986        let timeout = u64::try_from(step.timeout).map_err(|_| {
987            format!(
988                "privileged node {}.{} has invalid timeout {}",
989                step.group, step.job, step.timeout
990            )
991        })?;
992        if timeout >= launcher_bound {
993            current.insert(format!(
994                "{}.{} {timeout}s >= {launcher_bound}s (privileged launcher wrapper)",
995                step.group, step.job
996            ));
997        }
998    }
999
1000    let expected = fs::read_to_string(root.join("ci/budget-inversions-baseline.txt"))
1001        .map_err(|e| e.to_string())?
1002        .lines()
1003        .map(str::trim)
1004        .filter(|line| !line.is_empty() && !line.starts_with('#'))
1005        .map(str::to_string)
1006        .collect::<BTreeSet<_>>();
1007    if current != expected {
1008        let new = current.difference(&expected).cloned().collect::<Vec<_>>();
1009        let fixed = expected.difference(&current).cloned().collect::<Vec<_>>();
1010        return Err(format!(
1011            "budget-inversion baseline drifted: new={new:?} fixed-but-listed={fixed:?}"
1012        ));
1013    }
1014    println!(
1015        "budget ordering: {} baseline inversion(s), {} portable sharded + {} privileged nodes checked",
1016        current.len(),
1017        shards["debug_shards"]
1018            .as_array()
1019            .into_iter()
1020            .flatten()
1021            .chain(shards["release_shards"].as_array().into_iter().flatten())
1022            .map(|shard| shard["nodes"].as_array().map_or(0, Vec::len))
1023            .sum::<usize>(),
1024        privileged.steps.len()
1025    );
1026    Ok(())
1027}

1029fn audit_privileged_unboxed_guard(workflow: &str) -> Result<(), String> {
1030    const ACTIONS_GUARD: &str = "        if [[ ${GITHUB_ACTIONS:-} != true ]]; then";
1031    const REFUSAL: &str = "          echo 'privileged DAG: refusing explicit unboxed execution outside GitHub Actions' >&2";
1032    const FAIL_CLOSED: &str = "          exit 2";
1033
1034    for (line, description) in [
1035        (ACTIONS_GUARD, "exact GitHub Actions context guard"),
1036        (REFUSAL, "explicit outside-Actions refusal"),
1037        (FAIL_CLOSED, "nonzero outside-Actions exit"),
1038    ] {
1039        if workflow
1040            .lines()
1041            .filter(|candidate| *candidate == line)
1042            .count()
1043            != 1
1044        {
1045            return Err(format!(
1046                "privileged workflow must contain exactly one {description}"
1047            ));
1048        }
1049    }
1050
1051    if workflow.matches("--unsafe-no-cgroups").count() != 1 {
1052        return Err(
1053            "privileged workflow must select explicit unboxed execution exactly once".into(),
1054        );
1055    }
1056    if workflow
1057        .lines()
1058        .filter(|line| !line.trim_start().starts_with('#'))
1059        .any(|line| line.contains("--allow-cgroup-failure"))
1060    {
1061        return Err(
1062            "privileged workflow must not execute with broad --allow-cgroup-failure".into(),
1063        );
1064    }
1065    Ok(())
1066}

1068fn dag_critical_path(dag: &dagrun::DagConfig) -> Result<u64, String> {
1069    let steps = dag
1070        .steps
1071        .iter()
1072        .map(|step| (format!("{}.{}", step.group, step.job), step))
1073        .collect::<std::collections::BTreeMap<_, _>>();
1074    fn visit(
1075        id: &str,
1076        steps: &std::collections::BTreeMap<String, &dagrun::Step>,
1077        active: &mut BTreeSet<String>,
1078        memo: &mut std::collections::BTreeMap<String, u64>,
1079    ) -> Result<u64, String> {
1080        if let Some(value) = memo.get(id) {
1081            return Ok(*value);
1082        }
1083        if !active.insert(id.to_string()) {
1084            return Err(format!("DAG dependency cycle reaches {id}"));
1085        }
1086        let step = steps
1087            .get(id)
1088            .ok_or_else(|| format!("DAG critical path references missing node {id}"))?;
1089        let predecessor = step
1090            .deps
1091            .iter()
1092            .map(|dependency| visit(dependency, steps, active, memo))
1093            .collect::<Result<Vec<_>, _>>()?
1094            .into_iter()
1095            .max()
1096            .unwrap_or(0);
1097        active.remove(id);
1098        let timeout = u64::try_from(step.timeout)
1099            .map_err(|_| format!("DAG node {id} has invalid timeout {}", step.timeout))?;
1100        let value = predecessor
1101            .checked_add(timeout)
1102            .ok_or_else(|| format!("DAG critical path overflows at {id}"))?;
1103        memo.insert(id.to_string(), value);
1104        Ok(value)
1105    }
1106    let mut memo = std::collections::BTreeMap::new();
1107    let mut maximum = 0;
1108    for id in steps.keys() {
1109        maximum = maximum.max(visit(id, &steps, &mut BTreeSet::new(), &mut memo)?);
1110    }
1111    Ok(maximum)
1112}

1126    let steps = workflow["jobs"][job]["steps"]

1153fn print_plan(manifests: &ManifestSet, args: &Args, population: Population) -> ExitCode {
1154    let mut selection = args.selection.clone();
1155    selection.population = Some(population);
1156
1157    // ⚠️ AN UNKNOWN TEST ID IS A REFUSAL HERE, AND THIS IS THE ONLY SUBCOMMAND THAT
1158    // NEEDED IT. `run` and `build` already fail closed on an empty selection
1159    // (`filters selected no cells`), but `plan` printed an empty list and exited 0 --
1160    // measured 2026-08-26: `plan --lane portable --test no-such-test-xyz` is rc=0, and
1161    // so is the same command with a REAL id, so its exit code carried no information in
1162    // either direction. Anything driving a bisection off `plan` therefore reads a typo
1163    // as "nothing failed here" and converges, confidently, on the wrong commit.
1164    //
1165    // ⚠️ AND THE CHECK IS "UNKNOWN ID", NOT "EMPTY RESULT", WHICH IS NOT THE SAME FIX.
1166    // `print_plan` also serves `audit-gaps` (Population::Disabled), where an empty
1167    // answer legitimately means NO GAPS. Mirroring run's `cells.is_empty()` guard here
1168    // would turn that good answer into a failure. Asking whether the named id exists at
1169    // all separates the two: a real id with no cells in this population still prints
1170    // nothing and exits 0.
1171    if let Some(id) = selection.test.as_deref() {
1172        if !manifests.knows_test(id) {
1173            fail(format!(
1174                "unknown test id {id:?}: it is not in any manifest. An empty plan for a \
1175                 real id means that population has no cells; an empty plan for an id \
1176                 that does not exist means the filter is wrong, and refusing is what \
1177                 stops a bisection reading a typo as a pass."
1178            ));
1179        }
1180    }
1181
1182    let cells = manifests.select(&selection).unwrap_or_else(|e| fail(e));
1183    if args.format == "json" {
1184        println!("{}", serde_json::to_string(&cells.iter().map(|c| {
1185            let backend = if population == Population::Disabled && c.id.mode == "naked" {
1186                Some("native")
1187            } else {
1188                c.id.backend.as_deref()
1189            };
1190            serde_json::json!({"test":c.id.test,"category":c.category,"lane":c.test.lane,"mode":c.id.mode,"backend":backend})
1191        }).collect::<Vec<_>>()).unwrap());
1192    } else {
1193        for cell in cells {
1194            let backend = if population == Population::Disabled && cell.id.mode == "naked" {
1195                "native"
1196            } else {
1197                cell.id.backend.as_deref().unwrap_or("-")
1198            };
1199            println!(
1200                "{}\t{}\t{}\t{}\t{}",
1201                cell.test.lane, cell.category, cell.id.test, cell.id.mode, backend
1202            );
1203        }
1204    }
1205    ExitCode::SUCCESS
1206}

1208fn build(root: &Path, manifests: &ManifestSet, args: &Args) -> ExitCode {
1209    let mut selection = args.selection.clone();
1210    if selection.population.is_none() {
1211        selection.population = Some(if selection.include_manual {
1212            Population::Enabled
1213        } else {
1214            Population::Required
1215        });
1216    }
1217    let cells = manifests.select(&selection).unwrap_or_else(|e| fail(e));
1218    if cells.is_empty() && !args.allow_empty {
1219        fail("filters selected no cells");
1220    }
1221    let context = RunContext::from_env(root.to_path_buf(), false).unwrap_or_else(|e| fail(e));
1222    let mut seen = BTreeSet::new();

1233                failed = true;
1234            }
1235        }
1236    }
1237    if failed {
1238        ExitCode::FAILURE
1239    } else {
1240        ExitCode::SUCCESS
1241    }
1242}

1325/// Execute `count` independent items with at most `jobs` workers, delivering
1326/// each emitted value to `consume` immediately and waiting for its
1327/// acknowledgement before the worker may continue.
1328///
1329/// The consumer stays on the calling thread so durable publication is
1330/// serialized even while the expensive cell executions overlap. This is
1331/// deliberately not a collect-then-publish helper: an outer bucket timeout
1332/// must not discard rows that completed before the timeout, and a retry must
1333/// not start before the prior attempt is flushed.
1334fn for_each_parallel<T: Send>(
1335    count: usize,
1336    capacity: ScheduledWorkerCapacity,
1337    execute: impl Fn(usize, &mut dyn FnMut(T, bool) -> bool) + Sync,
1338    mut consume: impl FnMut(usize, T, bool) -> bool,
1339) {
1340    if count == 0 {
1341        return;
1342    }
1343    let workers = capacity.workers_for(count);
1344    let next = AtomicUsize::new(0);
1345    let (sender, receiver) = mpsc::channel::<(usize, T, bool, mpsc::SyncSender<bool>)>();
1346    thread::scope(|scope| {
1347        for _ in 0..workers {
1348            let sender = sender.clone();
1349            let execute = &execute;
1350            let next = &next;
1351            scope.spawn(move || {
1352                loop {
1353                    let index = next.fetch_add(1, Ordering::Relaxed);
1354                    if index >= count {
1355                        break;
1356                    }
1357                    let mut emit = |value, will_retry| {
1358                        let (ack_sender, ack_receiver) = mpsc::sync_channel(0);
1359                        if sender.send((index, value, will_retry, ack_sender)).is_err() {
1360                            return false;
1361                        }
1362                        ack_receiver.recv().unwrap_or(false)
1363                    };
1364                    execute(index, &mut emit);
1365                }
1366            });
1367        }
1368        drop(sender);
1369        for (index, value, will_retry, ack_sender) in receiver {
1370            let acknowledged = consume(index, value, will_retry);
1371            let _ = ack_sender.send(acknowledged);
1372        }
1373    });
1374}

1376fn run_with_retry<T>(
1377    first_attempt: u64,
1378    mut execute: impl FnMut(u64) -> T,
1379    mut retryable: impl FnMut(&T) -> bool,
1380    mut emit: impl FnMut(T, bool) -> bool,
1381) {
1382    assert!(
1383        (1..=MAX_ATTEMPTS_PER_CELL).contains(&first_attempt),
1384        "first cell attempt must be within the shared attempt cap"
1385    );
1386    for attempt in first_attempt..=MAX_ATTEMPTS_PER_CELL {
1387        let result = execute(attempt);
1388        let will_retry = retryable(&result) && attempt < MAX_ATTEMPTS_PER_CELL;
1389        if !emit(result, will_retry) || !will_retry {
1390            break;
1391        }
1392    }
1393}

1395fn run(root: &Path, manifests: &ManifestSet, args: &Args) -> ExitCode {
1396    let mut selection = args.selection.clone();
1397    if selection.population.is_none() {
1398        selection.population = Some(if selection.include_manual {
1399            Population::Enabled
1400        } else {
1401            Population::Required
1402        });
1403    }
1404    let cells = manifests.select(&selection).unwrap_or_else(|e| fail(e));
1405    if cells.is_empty() && !args.allow_empty {
1406        fail("filters selected no cells");
1407    }
1408    let capacity = scheduled_worker_capacity(args);
1409    let context = RunContext::from_env(root.to_path_buf(), args.prebuilt)
1410        .unwrap_or_else(|e| fail(e))
1411        .with_scheduled_worker_capacity(capacity);
1412    for (capability, verdict) in &context.host_capabilities {
1413        eprintln!(
1414            "Host capability {}: {} — {}",
1415            capability.value(),
1416            if verdict.present { "PRESENT" } else { "ABSENT" },
1417            verdict.evidence
1418        );
1419    }
1420    let results_path = args.results.clone().unwrap_or_else(|| {
1421        context
1422            .result_root
1423            .join(&context.run_id)
1424            .join("results.jsonl")
1425    });
1426    let junit = args
1427        .junit
1428        .clone()
1429        .unwrap_or_else(|| context.result_root.join(&context.run_id).join("junit.xml"));
1430    prepare_result_path(&results_path).unwrap_or_else(|error| {
1431        fail(format!(
1432            "cannot prepare result path {}: {error}",
1433            results_path.display()
1434        ))
1435    });
1436    let mut indexed_results = Vec::new();
1437    let mut attempt_results = vec![Vec::new(); cells.len()];
1438    let mut failed = false;
1439    // Sum the producer-owned CPU measurement from EVERY executed observation,
1440    // including a failed row that is retried. This is specifically cell CPU;
1441    // the harness process itself remains in the enclosing DAG cgroup.
1442    let mut cell_cpu_usage_usec = Some(0u64);
1443    let mut cpu_measurements = 0usize;
1444    let expected = cells.len();
1445    for_each_parallel(
1446        expected,
1447        capacity,
1448        |index, emit| {
1449            let cell = &cells[index];
1450            if let Some((_, reason)) =
1451                host_inapplicable_reason(&cell.test.requires, &context.host_capabilities)
1452            {
1453                let _ = emit(host_inapplicable_result(&context, cell, reason), false);
1454                return;
1455            }
1456
1457            run_with_retry(
1458                context.attempt,
1459                |attempt| {
1460                    let attempt_context = context.with_attempt(attempt);
1461                    match run_cell(&attempt_context, cell) {
1462                        Ok(result) => result,
1463                        Err(error) => infrastructure_error_result(&attempt_context, cell, error),
1464                    }
1465                },
1466                |result| !matches!(result.outcome.as_str(), "PASS" | "HOST-INAPPLICABLE"),
1467                emit,
1468            );
1469        },
1470        |index, mut result: CellResult, will_retry| {
1471            accumulate_cell_cpu_usage(
1472                &mut cell_cpu_usage_usec,
1473                &mut cpu_measurements,
1474                &result.outcome,
1475                result.cpu_usage_usec,
1476            );
1477            // Publish before announcing the outcome. After a visible PASS line,
1478            // the complete typed row is already present even if the containing
1479            // bucket is killed before its JUnit/summary epilogue. The worker
1480            // waits for this acknowledgement before starting a retry.
1481            let published = if let Err(error) = append_result(&results_path, &result) {
1482                eprintln!(
1483                    "ERROR {} ({}/{}): completed cell result could not be published: {error}",
1484                    result.test,
1485                    result.mode,
1486                    result.backend.as_deref().unwrap_or("native")
1487                );
1488                result.outcome = "ERROR".into();
1489                result.result = None;
1490                result.failure_class = Some(FailureClass::UnderstoodInfrastructureFailure);
1491                result.error_kind = Some("result-publication".into());
1492                result.reason = Some(format!(
1493                    "completed cell result could not be published: {error}"
1494                ));
1495                false
1496            } else {
1497                true
1498            };
1499
1500            if result.outcome == "ERROR" {
1501                eprintln!(
1502                    "ERROR {} ({}/{}): {}",
1503                    result.test,
1504                    result.mode,
1505                    result.backend.as_deref().unwrap_or("native"),
1506                    result.reason.as_deref().unwrap_or("infrastructure error")
1507                );
1508            }
1509            // A FAILURE MUST SAY ENOUGH TO BE CLASSIFIED, NOT JUST COUNTED.
1510            let located = if result.outcome == "PASS" {
1511                String::new()
1512            } else if result.outcome == "HOST-INAPPLICABLE" {
1513                format!(
1514                    " {}",
1515                    result.reason.as_deref().unwrap_or("host-inapplicable")
1516                )
1517            } else {
1518                let coords = [
1519                    ("turn", result.first_divergent_scheduler_turn),
1520                    ("vns", result.first_divergent_virtual_nanoseconds),
1521                    ("rec", result.first_divergent_record),
1522                    ("sys", result.first_divergent_syscall),
1523                ]
1524                .iter()
1525                .filter_map(|(key, value)| value.map(|value| format!("{key}={value}")))
1526                .collect::<Vec<_>>();
1527                let mut suffix = String::new();
1528                if !coords.is_empty() {
1529                    suffix.push_str(&format!(" [{}]", coords.join(" ")));
1530                }
1531                if let Some(reason) = result.reason.as_deref() {
1532                    suffix.push_str(&format!(" {reason}"));
1533                }
1534                suffix.push_str(&format!("\n    evidence: {}", result.artifact_dir));
1535                suffix
1536            };
1537            let effective_will_retry = published && will_retry;
1538            let retry_note = if effective_will_retry {
1539                format!(
1540                    " [attempt {} of at most {}; retrying this cell only]",
1541                    result.attempt, MAX_ATTEMPTS_PER_CELL
1542                )
1543            } else {
1544                String::new()
1545            };
1546            println!(
1547                "{} {} ({}/{}){}{}",
1548                result.outcome,
1549                result.test,
1550                result.mode,
1551                result.backend.as_deref().unwrap_or("native"),
1552                retry_note,
1553                located
1554            );
1555
1556            attempt_results[index].push(result);
1557            if !effective_will_retry {
1558                let result = match cell_result_after_retries(&attempt_results[index]) {
1559                    Ok(result) => result.clone(),
1560                    Err(error) => {
1561                        let mut result = attempt_results[index]
1562                            .last()
1563                            .expect("the current attempt was retained before reporting")
1564                            .clone();
1565                        result.outcome = "ERROR".into();
1566                        result.error_kind = Some("result-history".into());
1567                        result.reason = Some(error);
1568                        result
1569                    }
1570                };
1571                failed |= matches!(result.outcome.as_str(), "FAIL" | "ERROR");
1572                indexed_results.push((index, result));
1573            }
1574            published
1575        },
1576    );
1577    if indexed_results.len() != expected {
1578        eprintln!(
1579            "test-harness: only {} of {expected} selected cells returned a result",
1580            indexed_results.len()
1581        );
1582        failed = true;
1583    }
1584    indexed_results.sort_by_key(|(index, _)| *index);
1585    let results = indexed_results
1586        .into_iter()
1587        .map(|(_, result)| result)
1588        .collect::<Vec<_>>();
1589    if expected > 0 {
1590        println!(
1591            "test-harness: completed {} cell(s) with up to {} concurrent worker(s)",
1592            results.len(),
1593            capacity.workers_for(expected)
1594        );
1595    }
1596    let host_inapplicable = results
1597        .iter()
1598        .filter(|result| result.outcome == "HOST-INAPPLICABLE")
1599        .count();
1600    let cell_cpu_usage_usec = (cpu_measurements > 0)
1601        .then_some(cell_cpu_usage_usec)
1602        .flatten();
1603    if let Some(path) = std::env::var_os("DAGRUN_TEST_COUNTS_PATH") {
1604        let path = PathBuf::from(path);
1605        if let Err(error) =
1606            structured_test_results(&attempt_results).and_then(|report| report.write_current(&path))
1607        {
1608            eprintln!("test-harness: {error}");
1609            failed = true;
1610        }
1611    }
1612    if expected > 0 && host_inapplicable == expected {
1613        eprintln!(
1614            "test-harness: every one of the {expected} selected cell(s) was host-inapplicable; \
1615             a run that executed no cell is not a pass"
1616        );
1617        failed = true;
1618    }
1619    write_junit(&junit, &results).unwrap();
1620    let summary = serde_json::json!({
1621        "schema": 1,
1622        "cells": results.len(),
1623        "passed": results.iter().filter(|result| result.outcome == "PASS").count(),
1624        "failed": results.iter().filter(|result| result.outcome == "FAIL").count(),
1625        "errors": results.iter().filter(|result| result.outcome == "ERROR").count(),
1626        "host_inapplicable": host_inapplicable,
1627        "cell_cpu_usage_usec": cell_cpu_usage_usec,
1628        "host_inapplicable_cells": results
1629            .iter()
1630            .filter(|result| result.outcome == "HOST-INAPPLICABLE")
1631            .map(|result| serde_json::json!({
1632                "test": result.test,
1633                "mode": result.mode,
1634                "backend": result.backend,
1635                "reason": result.reason,
1636            }))
1637            .collect::<Vec<_>>(),
1638    });
1639    fs::write(
1640        results_path.parent().unwrap().join("summary.json"),
1641        serde_json::to_vec_pretty(&summary).unwrap(),
1642    )
1643    .unwrap();
1644    if failed {
1645        ExitCode::FAILURE
1646    } else {
1647        ExitCode::SUCCESS
1648    }
1649}
