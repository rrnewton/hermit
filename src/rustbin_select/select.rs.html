9//! Regression guest for determinized `select(2)` (the `timeval` sibling of
10//! `pselect6`). Before it was reclassified Determinized in Detcore, running this
11//! under `--strict` aborted with an unsupported-syscall error. It exercises both
12//! the ready path (a pipe with pending data reports its read end readable) and
13//! the timeout path (an empty pipe times out and `select` returns 0). A pipe is
14//! used deliberately so the guest does not depend on socket support.
15//!
16//! NOTE: on x86-64 glibc's `select()` wrapper is implemented on top of the
17//! `pselect6` syscall, so this guest issues the raw `select` syscall directly to
18//! actually cover Detcore's `handle_select` path rather than `handle_pselect6`.
19
20use std::os::fd::AsRawFd;
21use std::os::fd::OwnedFd;
22
23use nix::unistd::pipe;
24use nix::unistd::write;
25
26/// Invoke the raw `select(2)` syscall (bypassing glibc's pselect6-based wrapper)
27/// and return its raw result plus the (possibly updated) timeout that Linux
28/// writes back with the time not slept.
29fn raw_select(read_fd: i32, timeout: &mut libc::timeval) -> i64 {
30    // SAFETY: fd_set is a plain bitmap; we zero it and set a single valid fd.
31    let mut readfds: libc::fd_set = unsafe { std::mem::zeroed() };
32    unsafe { libc::FD_ZERO(&mut readfds) };
33    unsafe { libc::FD_SET(read_fd, &mut readfds) };
34
35    // SAFETY: readfds and timeout are valid; writefds/exceptfds are NULL.
36    let ret = unsafe {
37        libc::syscall(
38            libc::SYS_select,
39            read_fd + 1,
40            &mut readfds as *mut libc::fd_set,
41            std::ptr::null_mut::<libc::fd_set>(),
42            std::ptr::null_mut::<libc::fd_set>(),
43            timeout as *mut libc::timeval,
44        )
45    };
46    if ret > 0 {
47        assert!(
48            unsafe { libc::FD_ISSET(read_fd, &readfds) },
49            "select reported a ready fd but the read end bit was clear"
50        );
51    }
52    ret
53}
54
55fn main() {
56    // Ready path: data queued in the pipe makes the read end selectable.
57    let (ready_r, ready_w): (OwnedFd, OwnedFd) = pipe().unwrap();
58    write(&ready_w, b"ping").unwrap();
59    let mut timeout = libc::timeval {
60        tv_sec: 5,
61        tv_usec: 0,
62    };
63    let ret = raw_select(ready_r.as_raw_fd(), &mut timeout);
64    assert_eq!(ret, 1, "read end with pending data should be readable");
65
66    // Timeout path: an empty pipe blocks until the timeout elapses, then returns 0.
67    let (empty_r, _empty_w): (OwnedFd, OwnedFd) = pipe().unwrap();
68    let mut timeout = libc::timeval {
69        tv_sec: 0,
70        tv_usec: 10_000,
71    };
72    let ret = raw_select(empty_r.as_raw_fd(), &mut timeout);
73    assert_eq!(ret, 0, "empty pipe select should time out with 0 ready fds");
74
75    println!("select ready+timeout paths OK. Test complete.");
76}
