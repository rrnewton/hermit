38/// One line of `/proc/<pid>/maps`.
39#[derive(Debug, Clone, Copy, PartialEq, Eq)]
40struct Mapping {
41    start: usize,
42    end: usize,
43    /// `PROT_*` bits from the permission column.
44    prot: i32,
45    private: bool,
46    /// Whether the mapping has a backing inode. Replay represents recorded
47    /// file mappings as anonymous memory, so these are the ranges whose
48    /// contents can differ after advice that drops pages.
49    file_backed: bool,
50    /// Whether this is shared anonymous memory (`MAP_SHARED|MAP_ANONYMOUS` or
51    /// `/dev/zero`, which Linux backs with a deleted `/dev/zero` inode, or
52    /// System V shared memory). Their pages outlive advice that drops a
53    /// mapping's pages, as the pages of replay's stand-ins do (live shared
54    /// memory, or a shared view of a backing file), so they need no refill.
55    anonymous_shmem: bool,
56}
57
58/// `(major, minor)` of a device as `/proc/<pid>/maps` prints it.
59type Device = (u32, u32);
60
61/// The device of the kernel's internal shmem mount, found from a memfd,
62/// which Linux creates on that mount.
63///
64/// Shared anonymous memory and System V segments live on the same mount, and
65/// their names there (`/dev/zero`, `/SYSV<key>`) are chosen by the kernel:
66/// nothing can create a named file on it. A file elsewhere with one of those
67/// names is on another device. `None` if no memfd could be made, in which
68/// case every file-backed mapping is refilled.
69fn shmem_device() -> Option<Device> {
70    static DEVICE: OnceLock<Option<Device>> = OnceLock::new();
71    *DEVICE.get_or_init(|| {
72        // SAFETY: the name is a valid C string; the result is checked.
73        let fd = unsafe { libc::memfd_create(c"hermit-shmem-device".as_ptr(), libc::MFD_CLOEXEC) };
74        if fd < 0 {
75            return None;
76        }
77        // SAFETY: memfd_create returned a new descriptor that nothing else owns.
78        let fd = unsafe { OwnedFd::from_raw_fd(fd) };
79        let dev = std::fs::metadata(format!("/proc/self/fd/{}", fd.as_raw_fd()))
80            .ok()?
81            .dev();
82        Some((libc::major(dev), libc::minor(dev)))
83    })
84}

86fn parse_device(field: &str) -> Option<Device> {
87    let (major, minor) = field.split_once(':')?;
88    Some((
89        u32::from_str_radix(major, 16).ok()?,
90        u32::from_str_radix(minor, 16).ok()?,
91    ))
92}

94fn parse_maps(text: &str, shmem_device: Option<Device>) -> Vec<Mapping> {
95    text.lines()
96        .filter_map(|line| {
97            let mut fields = line.split_ascii_whitespace();
98            let (start, end) = fields.next()?.split_once('-')?;
99            let perms = fields.next()?.as_bytes();
100            let _offset = fields.next()?;
101            let device = parse_device(fields.next()?);
102            let inode = fields.next()?;
103            let path = fields.collect::<Vec<_>>().join(" ");
104            if perms.len() < 4 {
105                return None;
106            }
107            let mut prot = libc::PROT_NONE;
108            for (flag, bit) in [
109                (b'r', libc::PROT_READ),
110                (b'w', libc::PROT_WRITE),
111                (b'x', libc::PROT_EXEC),
112            ] {
113                if perms[..3].contains(&flag) {
114                    prot |= bit;
115                }
116            }
117            Some(Mapping {
118                start: usize::from_str_radix(start, 16).ok()?,
119                end: usize::from_str_radix(end, 16).ok()?,
120                prot,
121                private: perms[3] == b'p',
122                file_backed: inode != "0",
123                anonymous_shmem: shmem_device.is_some()
124                    && device == shmem_device
125                    && (path == "/dev/zero (deleted)" || path.starts_with("/SYSV")),
126            })
127        })
128        .collect()
129}

131/// Reads the guest's mappings. Recording cannot continue without them: an
132/// error returned to the guest would leave no event for replay to consume.
133fn read_mappings(path: &str) -> Vec<Mapping> {
134    let maps = std::fs::read(path)
135        .unwrap_or_else(|error| panic!("Cannot record madvise: {path}: {error}"));
136    parse_maps(&String::from_utf8_lossy(&maps), shmem_device())
137}

139/// The parts of `[start, end)` covered by mappings that replay represents
140/// with an anonymous stand-in, each with the protection of its mapping.
141///
142/// After their pages are dropped these can read back differently in replay:
143/// a private file mapping refaults from the file here but reads zeros there,
144/// and a shared file mapping refaults from the page cache here, including
145/// writes made through a descriptor since it was mapped, which replay's
146/// stand-in never saw. Shared anonymous memory is mapped live in replay and
147/// keeps its contents in both.
148fn refill_ranges(mappings: &[Mapping], start: usize, end: usize) -> Vec<(usize, usize, i32)> {
149    mappings
150        .iter()
151        .filter(|mapping| mapping.file_backed && !mapping.anonymous_shmem)
152        .filter_map(|mapping| {
153            let lo = mapping.start.max(start);
154            let hi = mapping.end.min(end);
155            (lo < hi).then_some((lo, hi, mapping.prot))
156        })
157        .collect()
158}

160/// Offset of the first private file-backed mapping in `[start, end)`.
161///
162/// Linux refuses `MADV_WIPEONFORK` with EINVAL at the first VMA that has a
163/// file, after applying the advice to the VMAs before it. Replay's anonymous
164/// replacement would accept the advice, so replay applies it only to this
165/// prefix. Shared mappings stay shared in replay and fail there too.
166fn wipeonfork_live_len(mappings: &[Mapping], start: usize, end: usize) -> Option<usize> {
167    mappings
168        .iter()
169        .find(|mapping| {
170            mapping.file_backed && mapping.private && mapping.start < end && start < mapping.end
171        })
172        .map(|mapping| mapping.start.max(start) - start)
173}

175/// Reads `[start, end)` of the guest through `/proc/<tid>/mem`, which also
176/// reads mappings without `PROT_READ`. Pages that cannot be read (guard pages,
177/// pages beyond the end of the file) are skipped; the readable runs are
178/// returned with their addresses.
179fn read_readable_runs(mem: &std::fs::File, start: usize, end: usize) -> Vec<(usize, Vec<u8>)> {
180    let mut runs: Vec<(usize, Vec<u8>)> = Vec::new();
181    let mut pos = start;
182    let mut run_open = false;
183    while pos < end {
184        let mut buf = vec![0u8; end - pos];
185        match mem.read_at(&mut buf, pos as u64) {
186            Ok(n) if n > 0 => {
187                buf.truncate(n);
188                if run_open {
189                    runs.last_mut().unwrap().1.extend_from_slice(&buf);
190                } else {
191                    runs.push((pos, buf));
192                    run_open = true;
193                }
194                pos += n;
195            }
196            _ => {
197                run_open = false;
198                pos = (pos & !(PAGE_SIZE - 1)) + PAGE_SIZE;
199            }
200        }
201    }
202    runs
203}

209/// The name of the shmem object Linux makes for a shared mapping of
210/// `/dev/zero` or a `MAP_SHARED | MAP_ANONYMOUS` one (`shmem_zero_setup`), as

215/// File systems whose regular files can have an `mmap` of their own, which
216/// maps kernel or device memory rather than file pages: a PCI device's
217/// `resourceN` in sysfs or its `/proc/bus/pci` file, and debugfs and tracefs

219const OWN_MMAP_FILE_SYSTEMS: [(i64, &str); 4] = [
220    (0x6265_6572, "sysfs"),

226/// What a `MAP_SHARED` mapping shares, as recording sees it before the call.
227#[derive(Debug, PartialEq, Eq)]
228enum Sharing {
229    /// The file whose pages the mapping maps, for replay to share them again.
230    File(SharedMapping),
231    /// `/dev/zero` at offset 0 through a descriptor opened with `access`:
232    /// the call maps a new object of its own, which `/proc/<tid>/maps` names
233    /// once it is made ([`object_at`]).
234    Zero { access: i32 },
235    /// Memory replay cannot reproduce, and why. A successful mapping of it
236    /// marks the recording as one replay refuses
237    /// (`detcore::detlog::record_replay_refusal`), and the guest runs on.
238    Refused(String),
239}

241/// What a `MAP_SHARED` mapping of descriptor `fd` of thread `tid`, at
242/// `offset`, shares: for a regular file, the file and the descriptor's access
243/// mode. `/proc/<tid>` shows that thread's own descriptor table, which is not
244/// the leader's for a thread cloned without `CLONE_FILES`.
245fn shared_mapping(flags: MapFlags, tid: i32, fd: i32, offset: i64) -> io::Result<Option<Sharing>> {
246    shared_mapping_with(flags, tid, fd, offset, |path| std::fs::read_to_string(path))
247}

249fn shared_mapping_with(
250    flags: MapFlags,
251    tid: i32,
252    fd: i32,
253    offset: i64,
254    read_fdinfo: impl FnOnce(&str) -> io::Result<String>,
255) -> io::Result<Option<Sharing>> {
256    if !flags.contains(MapFlags::MAP_SHARED) {
257        return Ok(None);
258    }
259    let path = format!("/proc/{tid}/fd/{fd}");
260    let metadata = std::fs::metadata(&path)?;
261    let file = MappedFileFacts {
262        mode: metadata.mode(),
263        rdev: metadata.rdev(),
264        device: metadata.dev(),
265        inode: metadata.ino(),
266        fs_type: file_system_type(&path)?,
267        target: std::fs::read_link(&path)?,
268    };
269    let offset = u64::try_from(offset)
270        .map_err(|error| io::Error::new(io::ErrorKind::InvalidInput, error))?;
271    let read_access = move || {
272        let fdinfo = read_fdinfo(&format!("/proc/{tid}/fdinfo/{fd}"))?;
273        descriptor_access(&fdinfo).ok_or_else(|| {
274            io::Error::new(
275                io::ErrorKind::InvalidData,
276                "fdinfo has no valid access flags",
277            )
278        })
279    };
280    let refused = |what: &str| {
281        Sharing::Refused(format!(
282            "a MAP_SHARED mapping of {} ({what}): replay reproduces the memory of \
283             shared mappings of regular files and of /dev/zero at offset 0 only",
284            file.target.display()
285        ))
286    };
287    Ok(Some(match mapped_file(&file, shmem_device()) {
288        MappedFile::Regular => Sharing::File(SharedMapping {
289            device: file.device,
290            inode: file.inode,
291            offset,
292            access: read_access()?,
293        }),
294        // The new object is as long as the mapping, and its pages at or past
295        // that length fault SIGBUS: at offset 0 the mapping has none, as a
296        // replay mapping of a backing at least as long has none.
297        MappedFile::Zero if offset == 0 => Sharing::Zero {
298            access: read_access()?,
299        },
300        MappedFile::Zero => refused(&format!(
301            "at offset {offset}, whose pages past the mapping's length fault SIGBUS"
302        )),
303        MappedFile::Other(what) => refused(&what),
304    }))
305}

307/// The `statfs` magic of the file system that holds the file `path` names.
308fn file_system_type(path: &str) -> io::Result<i64> {
309    let path = std::ffi::CString::new(path).expect("a /proc path holds no NUL");
310    // SAFETY: `statfs` is plain old data, valid when zeroed.
311    let mut buf: libc::statfs = unsafe { std::mem::zeroed() };
312    // SAFETY: the path is a valid C string and `buf` is writable.
313    if unsafe { libc::statfs(path.as_ptr(), &mut buf) } != 0 {
314        return Err(io::Error::last_os_error());
315    }
316    Ok(buf.f_type)
317}

319/// What `stat`, `statfs` and `/proc/<tid>/fd` say of a mapped file.
320#[derive(Debug)]
321struct MappedFileFacts {
322    /// `st_mode`.
323    mode: u32,
324    /// `st_rdev`.
325    rdev: u64,
326    /// `st_dev`.
327    device: u64,
328    /// `st_ino`.
329    inode: u64,
330    /// `f_type` of its file system.
331    fs_type: i64,
332    /// The name `/proc/<tid>/fd` gives it.
333    target: std::path::PathBuf,
334}

336/// What a `MAP_SHARED` mapping maps, by its file's type and device number.
337#[derive(Debug, PartialEq, Eq)]
338enum MappedFile {
339    /// A regular file, whose pages every shared mapping of it maps. The shmem
340    /// object of a shared anonymous or `/dev/zero` mapping is not one:
341    /// [`mapped_file`] refuses it.
342    Regular,
343    /// `/dev/zero`, of which every shared mapping is a new shmem object.
344    Zero,
345    /// Any other file, described.
346    Other(String),
347}
348
349/// What `file` is to a `MAP_SHARED` mapping. Linux opens a character device
350/// by its number, whatever its name, so `/dev/zero` is any node of
351/// [`DEV_ZERO`]. `shmem_device` is the kernel's internal shmem mount.
352fn mapped_file(file: &MappedFileFacts, shmem_device: Option<Device>) -> MappedFile {
353    // The files `anon_inode_getfile` makes share one inode, so its identity
354    // names no file of its own. Linux reports an anonymous inode with no type
355    // (`anon_inode_getattr`); its name says what it is whatever its reported
356    // type.
357    let anonymous = file
358        .target
359        .as_os_str()
360        .as_encoded_bytes()
361        .starts_with(b"anon_inode:");
362    let device = (libc::major(file.rdev), libc::minor(file.rdev));
363    match file.mode & libc::S_IFMT {
364        _ if anonymous => {
365            MappedFile::Other("an anonymous inode, which such files share".to_owned())
366        }
367        libc::S_IFREG => {
368            if let Some((_, name)) = OWN_MMAP_FILE_SYSTEMS
369                .iter()
370                .find(|(magic, _)| *magic == file.fs_type)
371            {
372                return MappedFile::Other(format!(
373                    "a file on {name}, whose mmap can map kernel or device memory"
374                ));
375            }
376            let on_shmem_mount = shmem_device.is_some()
377                && shmem_device == Some((libc::major(file.device), libc::minor(file.device)));
378            if on_shmem_mount && file.target == Path::new(ZERO_OBJECT_NAME) {
379                // The object of a MAP_SHARED | MAP_ANONYMOUS mapping, which
380                // runs live in record and replay with no event, or of a
381                // /dev/zero mapping, which replay backs by the mapping and not
382                // by this file. Only /proc/<pid>/map_files gives a descriptor
383                // for it, and opening that needs CAP_CHECKPOINT_RESTORE or
384                // CAP_SYS_ADMIN in the initial user namespace
385                // (proc_map_files_get_link). No guest of the hermit command
386                // has either: it runs in a user namespace of its own, and
387                // Hermit's /proc has no map_files. The refusal also covers a
388                // descriptor the guest inherited.
389                return MappedFile::Other(
390                    "the shmem object of a MAP_SHARED | MAP_ANONYMOUS or /dev/zero \
391                     mapping, whose memory replay does not back as a file"
392                        .to_owned(),
393                );
394            }
395            MappedFile::Regular
396        }
397        libc::S_IFCHR if device == DEV_ZERO => MappedFile::Zero,
398        libc::S_IFCHR => MappedFile::Other(format!(
399            "a character device {}:{}, whose driver maps memory of its own",
400            device.0, device.1
401        )),
402        libc::S_IFBLK => MappedFile::Other("a block device".to_owned()),
403        0 => MappedFile::Other("a pseudo-file with no file type".to_owned()),
404        kind => MappedFile::Other(format!("a file of type {kind:o}")),
405    }
406}

408/// Names the new `/dev/zero` mapping at `address` of thread `tid`, mapped
409/// through a descriptor opened with `access`, by the object Linux made for it
410/// as `/proc/<tid>/maps` shows it. The mapping exists, so a listing without
411/// it cannot be recorded.
412fn name_zero_mapping(tid: i32, address: usize, access: i32) -> SharedMapping {
413    let maps_path = format!("/proc/{tid}/maps");
414    let (device, inode) = std::fs::read_to_string(&maps_path)
415        .ok()
416        .and_then(|maps| object_at(&maps, address))
417        .unwrap_or_else(|| {
418            panic!(
419                "Cannot record MAP_SHARED mmap of /dev/zero: \
420                 {maps_path} names no object at {address:#x}"
421            )
422        });
423    SharedMapping {
424        device,
425        inode,
426        offset: 0,
427        access,
428    }
429}

431/// The file a `/proc/<pid>/maps` listing shows mapped at `address`, the start
432/// of a mapping, by device and inode.
433fn object_at(maps: &str, address: usize) -> Option<(u64, u64)> {
434    maps.lines().find_map(|line| {
435        let mut fields = line.split_ascii_whitespace();
436        let (start, _) = fields.next()?.split_once('-')?;
437        if usize::from_str_radix(start, 16).ok()? != address {
438            return None;
439        }
440        let (_perms, _offset) = (fields.next()?, fields.next()?);
441        let (major, minor) = parse_device(fields.next()?)?;
442        let inode = fields.next()?.parse().ok()?;
443        Some((libc::makedev(major, minor), inode))
444    })
445}

447/// A failed mmap needs no sharing identity. A successful one cannot be
448/// recorded without it: returning the capture error to the guest would hide
449/// its live mapping, and writing `None` would let replay replace shared pages
450/// with a private copy.
451fn captured_mmap_result(
452    result: Result<i64, Errno>,
453    shared: io::Result<Option<Sharing>>,
454) -> Result<(i64, Option<Sharing>), Errno> {
455    let addr = result?;
456    let shared = shared.unwrap_or_else(|error| {
457        panic!(
458            "Cannot record MAP_SHARED mmap: cannot capture file identity and access mode: {error}"
459        )
460    });
461    Ok((addr, shared))
462}

464/// The access mode in a `/proc/<pid>/fdinfo/<fd>` listing: the `O_ACCMODE`
465/// bits of its octal `flags:` line.
466fn descriptor_access(fdinfo: &str) -> Option<i32> {
467    let flags = fdinfo
468        .lines()
469        .find_map(|line| line.strip_prefix("flags:"))?
470        .trim();
471    Some(i32::from_str_radix(flags, 8).ok()? & libc::O_ACCMODE)
472}

474impl Recorder {
475    pub(super) async fn handle_mmap<G: Guest<Self>>(
476        &self,
477        guest: &mut G,
478        syscall: Mmap,
479    ) -> Result<i64, Errno> {
480        // Let anonymous mappings through. This should already be deterministic
481        // because ASLR is disabled.
482        if syscall.flags().contains(MapFlags::MAP_ANONYMOUS) {
483            return guest.inject(syscall).await;
484        }
485
486        let len = syscall.len();
487        // Read before the call: the descriptor names the same file then, and a
488        // sibling thread could not have closed it, as the call holds the turn.
489        let shared = shared_mapping(
490            syscall.flags(),
491            guest.tid().as_raw(),
492            syscall.fd(),
493            syscall.offset(),
494        );
495
496        // Do the injection. We need to record the pointer the mapping is at and
497        // all of the bytes contained in the mapping. This is very inefficient
498        // for large mappings and should be replaced by just letting the mmap
499        // through to the real file.
500        let result = guest.inject(syscall).await;
501
502        self.record_event(
503            guest,
504            match captured_mmap_result(result, shared) {
505                Ok((addr, sharing)) => {
506                    let addr = Addr::from_raw(addr as usize).ok_or(Errno::EINVAL)?;
507                    // Replay fills a file's backing from recorded bytes. A
508                    // new /dev/zero object holds zeros, as a new backing does,
509                    // and a refused mapping is never replayed, so their bytes
510                    // are not read: reading them through /proc/<pid>/mem
511                    // would fault in the new object's pages, or call a
512                    // device driver's fault or access handler, during the
513                    // recorded run.
514                    let (shared, read_bytes) = match sharing {
515                        None => (None, true),
516                        Some(Sharing::File(shared)) => (Some(shared), true),
517                        Some(Sharing::Zero { access }) => (
518                            Some(name_zero_mapping(
519                                guest.tid().as_raw(),
520                                addr.as_raw(),
521                                access,
522                            )),
523                            false,
524                        ),
525                        Some(Sharing::Refused(reason)) => {
526                            tracing::error!("Cannot replay this recording: {reason}");
527                            detcore::detlog::record_replay_refusal(&reason);
528                            (None, false)
529                        }
530                    };
531                    let buf = if read_bytes {
532                        // NOTE: We can't use `read_exact` here to slurp up the
533                        // memory map bytes. The memory size may be larger than
534                        // the physical file size for dynamic libraries. The
535                        // following `read` will read up to the physical file
536                        // length, not the length specified on the memory map.
537                        // The left over bytes that extend past the end of the
538                        // physical file should be set to zeros when we replay
539                        // this `mmap`.
540                        let mut buf = vec![0u8; len];
541                        let physical_length = guest.memory().read(addr, &mut buf)?;
542
543                        // Don't store more bytes than we need to. When we
544                        // create the anonymous map later, the extra bytes will
545                        // be initialized to zero automatically.
546                        buf.truncate(physical_length);
547                        buf
548                    } else {
549                        Vec::new()
550                    };
551
552                    Ok(SyscallEvent::Mmap(MmapEvent {
553                        addr: addr.as_raw(),
554                        buf,
555                        shared,
556                    }))
557                }
558                Err(errno) => Err(errno),
559            },
560        );
561
562        result
563    }

565    /// Records a guest-semantic `madvise`.
566    ///
567    /// The advice runs live in both recording and replay. Replay's anonymous
568    /// stand-ins for file mappings differ in two guest-visible ways, which the
569    /// event carries: pages dropped from a file mapping read back as file
570    /// contents here but as zeros in replay, and `MADV_WIPEONFORK` fails at a
571    /// private file mapping here but would succeed in replay.
572    // AUTONOMOUS-BOT-IMPLEMENTED
573    // TODO-HUMAN-REVIEW(https://github.com/rrnewton/hermit/pull/3589): Audit the madvise refill and WIPEONFORK prefix.
574    pub(super) async fn handle_madvise<G: Guest<Self>>(
575        &self,
576        guest: &mut G,
577        syscall: Madvise,
578    ) -> Result<i64, Errno> {
579        let advice = syscall.advice();
580        let start = syscall.addr().map(AddrMut::as_raw).unwrap_or(0);
581        let len = syscall.len();
582        // Detcore has already rejected unaligned or overflowing ranges.
583        let end = start.saturating_add(len).next_multiple_of(PAGE_SIZE);
584
585        let drops_pages = matches!(
586            advice,
587            libc::MADV_DONTNEED | libc::MADV_DONTNEED_LOCKED | MADV_GUARD_REMOVE
588        );
589        // The thread's own maps: the thread-group leader's is empty once it has
590        // exited.
591        let maps_path = format!("/proc/{}/maps", guest.tid().as_raw());
592
593        let wipeonfork_prefix = if advice == libc::MADV_WIPEONFORK {
594            let mappings = read_mappings(&maps_path);
595            wipeonfork_live_len(&mappings, start, end)
596        } else {
597            None
598        };
599
600        let result = guest.inject(syscall).await;
601
602        // Linux reports EINVAL at the first private file mapping and returns
603        // without visiting the rest of the range. Replay reproduces that by
604        // applying the advice only up to that mapping. Any other outcome
605        // means Linux accepted or failed independently of the file mappings,
606        // so replay applies the whole range and must reproduce the result.
607        let live_len = match (wipeonfork_prefix, result) {
608            (Some(prefix), Err(Errno::EINVAL)) => prefix,
609            _ => len,
610        };
611
612        let mut refills = Vec::new();
613        if drops_pages {
614            let mappings = read_mappings(&maps_path);
615            let ranges = refill_ranges(&mappings, start, end);
616            if !ranges.is_empty() {
617                let mem_path = format!("/proc/{}/mem", guest.tid().as_raw());
618                // The advice has already taken effect, so failing the guest
619                // call now would leave the recording without its event.
620                let mem = std::fs::File::open(&mem_path)
621                    .unwrap_or_else(|error| panic!("Cannot record madvise: {mem_path}: {error}"));
622                for (lo, hi, prot) in ranges {
623                    for (addr, bytes) in read_readable_runs(&mem, lo, hi) {
624                        refills.push(MadviseRefill { addr, bytes, prot });
625                    }
626                }
627            }
628        }
629
630        self.record_event(
631            guest,
632            Ok(SyscallEvent::Madvise(MadviseEvent {
633                result,
634                live_len,
635                refills,
636            })),
637        );
638
639        result
640    }
641}
