17/// `event`, a branch event, as two: its first half of the branches, ending
18/// with no RIP, then the rest, ending where `event` ended, with its RIP. A
19/// schedule replay stops a thread at a branch event's count when the event
20/// has a RIP (a timer stop), or when another branch event or another thread's
21/// event follows it (`Replayer::resolve_time_to_run`). The first half is
22/// always followed by the second, so it stays a stop. The second keeps
23/// `event`'s RIP, so it is a stop exactly where `event` was, even when the
24/// thread's own OtherInstructions follows it, as `hermit analyze`'s fragments
25/// keep theirs.
26fn split_one(event: SchedEvent) -> Vec<SchedEvent> {
27    let first_count = event.count / 2;
28    let second_count = event.count - first_count;
29
30    let first_end_time = event.end_time.map(|t| {
31        LogicalTime::from_nanos((t.as_nanos() as i64 - 10i64 * second_count as i64) as u64)
32    });
33
34    vec![
35        SchedEvent {
36            count: first_count,
37            op: Op::Branch,
38            end_time: first_end_time,
39            start_rip: None,
40            end_rip: None,
41            // A new cut: no recorded end, taken without a stop or not, is here.
42            stopless: false,
43            // The offset names where the original batch ended, which only
44            // the second fragment does.
45            instruction_offset: None,
46            ..event
47        },
48        SchedEvent {
49            count: second_count,
50            op: Op::Branch,
51            start_rip: None,
52            end_rip: event.end_rip,
53            ..event
54        },
55    ]
56}
57
58fn split_branches(
59    event: SchedEvent,
60    cursor: &ReplayCursor<SchedEvent>,
61) -> impl IntoIterator<Item = SchedEvent> + use<> {
62    match (&event, cursor.peek_nth(0), cursor.peek_nth(1)) {
63        (
64            SchedEvent {
65                op: Op::Branch,
66                count,
67                ..
68            },
69            Some(SchedEvent {
70                op: Op::OtherInstructions,
71                ..
72            }),
73            Some(SchedEvent { op: Op::Branch, .. }),
74        )
75        | (
76            SchedEvent {
77                op: Op::Branch,
78                count,
79                ..
80            },
81            Some(SchedEvent { op: Op::Branch, .. }),
82            _,
83        ) if *count > 1 && event.instruction_offset.is_none() => split_one(event),
84        _ => vec![event],
85    }
86}

88/// Split the batches of branches in the preemption record at `from_file`.
89/// The record is read fallibly: a format this Hermit does not read is a
90/// typed refusal (`detcore::preemptions::RecordFormatRefusal`) and a
91/// malformed one an error, never a panic.
92pub fn split_branches_in_file(from_file: impl AsRef<Path>, backup: bool) -> anyhow::Result<()> {
93    let mut preemptions = PreemptionReader::try_new(from_file.as_ref())?.into_inner();
94    preemptions.split_map(split_branches);
95
96    if backup {
97        let mut new_name = from_file.as_ref().to_path_buf();
98        new_name.set_extension("bak");
99        std::fs::rename(&from_file, &new_name)?;
100    }
101
102    preemptions
103        .write_to_disk(from_file.as_ref())
104        .map_err(anyhow::Error::msg)?;
105    Ok(())
106}

118    #[test]
119    pub fn test_split_one() {
120        let event = SchedEvent {
121            count: 3,
122            dettid: DetPid::from_raw(3),
123            end_time: Some(LogicalTime::from_nanos(10000)),
124            stopless: false,
125            instruction_offset: None,
126            op: Op::Branch,
127            start_rip: None,
128            end_rip: None,
129        };
130
131        let counts: Vec<_> = split_one(event)
132            .into_iter()
133            .map(|x| (x.op, x.count, x.end_time.unwrap().as_nanos()))
134            .collect();
135
136        assert_eq!(counts, vec![(Op::Branch, 1, 9980), (Op::Branch, 2, 10000)]);
137    }

139    /// The second half of a split timer stop keeps its RIP, so a schedule
140    /// replay still stops the thread there when the thread's own
141    /// OtherInstructions follows; the first half, which the second follows,
142    /// needs none.
143    #[test]
144    fn a_split_timer_stop_keeps_its_rip_on_the_second_half() {
145        let rip = std::num::NonZeroUsize::new(0x0040_110b).unwrap();
146        let event = SchedEvent {
147            count: 7,
148            dettid: DetPid::from_raw(3),
149            end_time: Some(LogicalTime::from_nanos(10000)),
150            op: Op::Branch,
151            start_rip: None,
152            end_rip: Some(rip),
153            stopless: false,
154            instruction_offset: None,
155        };
156        let rips: Vec<_> = split_one(event)
157            .into_iter()
158            .map(|x| (x.count, x.end_rip))
159            .collect();
160        assert_eq!(rips, vec![(3, None), (4, Some(rip))]);
161    }

163    /// A batch's instruction offset names where the batch ended, so only the
164    /// fragment that ends there keeps it.
165    #[test]
166    pub fn a_split_keeps_the_instruction_offset_on_the_fragment_that_ends_the_batch() {
167        let offset = detcore::preemptions::InstructionOffset {
168            rcb_target: 40,
169            instructions: 5,
170        };
171        let event =
172            SchedEvent::branches(DetPid::from_raw(3), 3).with_instruction_offset(Some(offset));
173        let offsets: Vec<_> = split_one(event)
174            .into_iter()
175            .map(|fragment| fragment.instruction_offset)
176            .collect();
177        assert_eq!(offsets, vec![None, Some(offset)]);
178    }

180    /// Through the file the splitter edits: a batch of branches that an
181    /// instruction-offset timer ended stays one unchanged event (its last
182    /// branches can lie past the timer's target), while the same batch
183    /// without an offset is split. A record of a newer format is refused,
184    /// typed, without a panic.
185    #[test]
186    pub fn split_branches_in_file_leaves_an_offset_batch_whole() {
187        let offset = detcore::preemptions::InstructionOffset {
188            rcb_target: 40,
189            instructions: 5,
190        };
191        let dir = tempfile::tempdir().unwrap();
192        let tid = DetPid::from_raw(3);
193        let schedule = |offset| {
194            vec![
195                SchedEvent::branches(tid, 10).with_instruction_offset(offset),
196                SchedEvent {
197                    op: Op::OtherInstructions,
198                    count: 1,
199                    ..SchedEvent::branches(tid, 1)
200                        .with_end_rip(std::num::NonZeroUsize::new(0x1000).unwrap())
201                },
202                SchedEvent::branches(tid, 5),
203            ]
204        };
205        let split = |name: &str, events: Vec<SchedEvent>| {
206            let path = dir.path().join(name);
207            let mut writer = detcore::preemptions::PreemptionWriter::new(None);
208            for event in events {
209                writer.insert_schedevent(event);
210            }
211            std::fs::write(&path, writer.into_string()).unwrap();
212            split_branches_in_file(&path, false).unwrap();
213            PreemptionReader::try_new(&path)
214                .unwrap()
215                .into_inner()
216                .into_global()
217        };
218        let with = schedule(Some(offset));
219        assert_eq!(split("with.json", with.clone()), with);
220        let without = split("without.json", schedule(None));
221        assert_eq!(without.len(), 4, "{without:?}");
222        assert_eq!((without[0].count, without[1].count), (5, 5), "{without:?}");
223
224        let newer = dir.path().join("newer.json");
225        std::fs::write(
226            &newer,
227            serde_json::json!({"per_thread": {}, "global": [], "format_version": 99}).to_string(),
228        )
229        .unwrap();
230        let error = split_branches_in_file(&newer, false).unwrap_err();
231        assert!(
232            error
233                .chain()
234                .any(|cause| cause.is::<detcore::preemptions::RecordFormatRefusal>()),
235            "{error:#}"
236        );
237    }
