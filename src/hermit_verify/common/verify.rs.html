23/// options to pass for log diff tool
24#[derive(Default)]
25pub struct LogDiffOptions {
26    /// Number of syscalls to display when failure
27    pub syscall_history: usize,
28    /// Whether to skip commits for determinism check
29    pub skip_commits: bool,
30    /// Whether to skip syscalls for determinism check
31    pub skip_detlog_syscalls: bool,
32    /// Whether to skip syscall results for determinism check
33    pub skip_detlog_syscall_results: bool,
34    /// Whether to skip the rest of DETLOGs for determinism check
35    pub skip_detlog_others: bool,
36    /// Which lines to ignore for DETLOGs
37    pub ignore_lines: Vec<String>,
38}
39
40impl LogDiffOptions {
41    fn into_args(self) -> Vec<String> {
42        // hermit-verify compares ordinary Hermit logs with the all-records
43        // envelope, which is the standalone command's default. It is NOT named
44        // explicitly here: `hermit_bin` defaults to whatever `HERMIT_BIN` or
45        // the PATH supplies, which can be a published bundle built without the
46        // flag, and passing it there is `error: unexpected argument`, exit 2.
47        let mut result: Vec<String> = Vec::new();
48        if self.skip_detlog_others && self.skip_detlog_syscalls && self.skip_detlog_syscall_results
49        {
50            result.push("--skip-detlog".to_owned());
51        } else {
52            if !self.skip_detlog_others {
53                result.push("--include-detlogs=other".to_owned());
54            }
55            if !self.skip_detlog_syscall_results {
56                result.push("--include-detlogs=syscallresult".to_owned());
57            }
58            if !self.skip_detlog_syscalls {
59                result.push("--include-detlogs=syscall".to_owned());
60            }
61        }
62
63        if self.skip_commits {
64            result.push("--skip-commit".to_owned())
65        }
66
67        result.push(format!("--syscall-history={}", self.syscall_history));
68
69        for ignore_lines in self.ignore_lines {
70            result.push(format!("--ignore-lines={}", ignore_lines));
71        }
72
73        result
74    }
75}

77impl<P: AsRef<OsStr>> Verify<P> {
78    pub fn new(hermit_bin: P) -> Self {
79        Self { hermit_bin }

82    /// Whether the child hermit's `log-diff` offers
83    /// `--canonicalize-host-addresses`. `hermit_bin` comes from `HERMIT_BIN` or
84    /// the PATH, so it can be a published bundle that predates the flag, where
85    /// passing it is `error: unexpected argument` (see
86    /// `log_diff_command_does_not_pin_a_record_envelope_the_child_may_lack`).
87    fn child_offers_host_address_canonicalization(&self) -> bool {
88        std::process::Command::new(&self.hermit_bin)
89            .args(["log-diff", "--help"])
90            .stdin(std::process::Stdio::null())
91            .stderr(std::process::Stdio::null())
92            .output()
93            .is_ok_and(|output| {
94                output.status.success()
95                    && String::from_utf8_lossy(&output.stdout)
96                        .contains("--canonicalize-host-addresses")
97            })
98    }

100    fn verify_lines<S1: AsRef<str>, S2: AsRef<str>>(left: S1, right: S2) -> anyhow::Result<bool> {
101        let result = similar::TextDiff::configure()
102            .algorithm(similar::Algorithm::Myers)
103            .diff_lines(left.as_ref(), right.as_ref());
104
105        if result.ratio() == 1.0 {
106            Ok(true)
107        } else {
108            for c in result.iter_all_changes() {
109                match c.tag() {
110                    similar::ChangeTag::Equal => print!("{}", c),
111                    similar::ChangeTag::Delete => {
112                        print!("{}", format!("-{}", c).red().bold())
113                    }
114                    similar::ChangeTag::Insert => {
115                        print!("{}", format!("+{}", c).green().bold())
116                    }
117                }
118            }
119            Ok(false)
120        }
121    }
122
123    fn verify_files(left: &Path, right: &Path) -> anyhow::Result<bool> {
124        let left = std::fs::read_to_string(left)?;
125        let right = std::fs::read_to_string(right)?;
126        Self::verify_lines(left, right)
127    }
128
129    fn format_json<TPath: AsRef<Path>>(path: TPath) -> anyhow::Result<String> {
130        let value: serde_json::Value =
131            serde_json::from_reader(std::fs::File::open(path.as_ref())?)?;
132        let schedules = match &value {
133            serde_json::Value::Object(sched_file) => sched_file.get("global").ok_or_else(|| {
134                anyhow::Error::msg("expecting \"global\" key in the target json file")
135            }),
136            _ => Err(anyhow::Error::msg(format!(
137                "{} has unexpected format",
138                path.as_ref().display()
139            ))),
140        };
141        Ok(serde_json::to_string_pretty(schedules?)?)
142    }

183    pub fn verify_desync(&self, right: &RunEnvironment) -> anyhow::Result<bool> {
184        //FIXME: extract log-diff out of hermit and handle DESYNCs there (T135657122 + some extra work)
185        println!("{}", "::  Looking for desync events:".bold());
186        let buffer = BufReader::new(std::fs::File::open(&right.log_file_path)?);
187        for line in buffer.lines() {
188            if line?.contains("DESYNC") {
189                println!("{}", "WARNING: DESYNCs found".red());
190                return Ok(false);
191            }
192        }
193        println!("DESYNC events not found");
194        Ok(true)
195    }

197    /// The `log-diff` argument vector. `canonicalize_host_addresses` says
198    /// whether the child hermit offers `--canonicalize-host-addresses`.
199    ///
200    /// The two logs come from two hermit processes, so ASLR moves every marked
201    /// launcher pointer (`<hostaddr 0x...>`) between them; compared by raw
202    /// value they always differ. The flag compares them by per-run ordinal
203    /// instead, keeping identity, order and aliasing
204    /// (<https://github.com/rrnewton/hermit/issues/3412>).
205    fn build_command_args(
206        &self,
207        left: &RunEnvironment,
208        right: &RunEnvironment,
209        options: LogDiffOptions,
210        canonicalize_host_addresses: bool,
211    ) -> Vec<String> {
212        let mut result = vec![];
213        result.push(String::from("log-diff"));
214        result.append(&mut options.into_args());
215        if canonicalize_host_addresses {
216            result.push("--canonicalize-host-addresses".to_owned());
217        }
218        result.push(format!("{}", left.log_file_path.display()));
219        result.push(format!("{}", right.log_file_path.display()));
220
221        result
222    }
223
224    pub fn verify_logs(
225        &self,
226        left: &RunEnvironment,
227        right: &RunEnvironment,
228        options: LogDiffOptions,
229    ) -> anyhow::Result<bool> {
230        println!("{}", "::  Comparing log files".bold());
231        let canonicalize_host_addresses = self.child_offers_host_address_canonicalization();
232        if !canonicalize_host_addresses {
233            println!(
234                "    (this hermit's log-diff has no --canonicalize-host-addresses; \
235                 marked host addresses are compared by raw value)"
236            );
237        }
238        let mut command = std::process::Command::new(&self.hermit_bin);
239        command.args(self.build_command_args(left, right, options, canonicalize_host_addresses));
240
241        println!("{}", format!("    {}", display_cmd(&command)).bold());
242        Ok(command.status()?.success())
243    }
244
245    pub fn verify_exit_statuses(
246        &self,
247        left: &RunEnvironment,
248        right: &RunEnvironment,
249    ) -> anyhow::Result<bool> {
250        println!("{}", "::  Comparing exit codes".bold());
251        Self::verify_files(
252            left.exit_status_file_path.as_path(),
253            right.exit_status_file_path.as_path(),
254        )
255    }

309    /// hermit-verify must NOT name a record envelope on the `log-diff` it
310    /// spawns. `hermit_bin` defaults to `HERMIT_BIN` or the PATH, so the child
311    /// can be a published bundle built without the flag, where passing it is
312    /// `error: unexpected argument` and exit 2. The all-records envelope this
313    /// callsite wants is that command's default, so omitting it is both
314    /// correct and compatible.
315    #[test]
316    fn log_diff_command_does_not_pin_a_record_envelope_the_child_may_lack() -> anyhow::Result<()> {
317        let env = TemporaryEnvironmentBuilder::new().run_count(2).build()?;
318        let verify = Verify::new(PathBuf::from("hermit"));
319        for canonicalize_host_addresses in [false, true] {
320            let args = verify.build_command_args(
321                &env.runs()[0],
322                &env.runs()[1],
323                LogDiffOptions::default(),
324                canonicalize_host_addresses,
325            );
326            assert!(
327                !args.iter().any(|arg| arg.starts_with("--record-envelope")),
328                "hermit-verify must not pass --record-envelope to a hermit it did not build: \
329                 {args:?}"
330            );
331        }
332        Ok(())
333    }

335    /// <https://github.com/rrnewton/hermit/issues/3412>: a child that offers the
336    /// flag is asked to canonicalize marked host addresses; one that does not
337    /// is never passed an argument it would reject.
338    #[test]
339    fn log_diff_canonicalizes_host_addresses_only_when_the_child_offers_it() -> anyhow::Result<()> {
340        let env = TemporaryEnvironmentBuilder::new().run_count(2).build()?;
341        let verify = Verify::new(PathBuf::from("hermit"));
342        let args = |offered| {
343            verify.build_command_args(
344                &env.runs()[0],
345                &env.runs()[1],
346                LogDiffOptions::default(),
347                offered,
348            )
349        };
350        let flag = "--canonicalize-host-addresses".to_owned();
351        assert!(args(true).contains(&flag));
352        assert!(!args(false).contains(&flag));
353
354        // The probe itself: a child whose log-diff help lacks the flag, and a
355        // child that cannot run at all, both read as "not offered".
356        let directory = tempfile::tempdir()?;
357        let old = directory.path().join("old-hermit");
358        std::fs::write(&old, "#!/bin/sh\necho 'Usage: hermit log-diff [OPTIONS]'\n")?;
359        let new = directory.path().join("new-hermit");
360        std::fs::write(
361            &new,
362            "#!/bin/sh\necho '      --canonicalize-host-addresses'\n",
363        )?;
364        for script in [&old, &new] {
365            use std::os::unix::fs::PermissionsExt;
366            std::fs::set_permissions(script, std::fs::Permissions::from_mode(0o755))?;
367        }
368        assert!(!Verify::new(&old).child_offers_host_address_canonicalization());
369        assert!(Verify::new(&new).child_offers_host_address_canonicalization());
370        assert!(
371            !Verify::new(directory.path().join("absent"))
372                .child_offers_host_address_canonicalization()
373        );
374        Ok(())
375    }
