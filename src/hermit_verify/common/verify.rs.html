23/// options to pass for log diff tool
24#[derive(Default)]
25pub struct LogDiffOptions {
26    /// Number of syscalls to display when failure
27    pub syscall_history: usize,
28    /// Whether to skip commits for determinism check
29    pub skip_commits: bool,
30    /// Whether to skip syscalls for determinism check
31    pub skip_detlog_syscalls: bool,
32    /// Whether to skip syscall results for determinism check
33    pub skip_detlog_syscall_results: bool,
34    /// Whether to skip the rest of DETLOGs for determinism check
35    pub skip_detlog_others: bool,
36    /// Which lines to ignore for DETLOGs
37    pub ignore_lines: Vec<String>,
38}
39
40impl LogDiffOptions {
41    fn into_args(self) -> Vec<String> {
42        // hermit-verify compares ordinary Hermit logs with the all-records
43        // envelope, which is the standalone command's default. It is NOT named
44        // explicitly here: `hermit_bin` defaults to whatever `HERMIT_BIN` or
45        // the PATH supplies, which can be a published bundle built without the
46        // flag, and passing it there is `error: unexpected argument`, exit 2.
47        let mut result: Vec<String> = Vec::new();
48        if self.skip_detlog_others && self.skip_detlog_syscalls && self.skip_detlog_syscall_results
49        {
50            result.push("--skip-detlog".to_owned());
51        } else {
52            if !self.skip_detlog_others {
53                result.push("--include-detlogs=other".to_owned());
54            }
55            if !self.skip_detlog_syscall_results {
56                result.push("--include-detlogs=syscallresult".to_owned());
57            }
58            if !self.skip_detlog_syscalls {
59                result.push("--include-detlogs=syscall".to_owned());
60            }
61        }
62
63        if self.skip_commits {
64            result.push("--skip-commit".to_owned())
65        }
66
67        result.push(format!("--syscall-history={}", self.syscall_history));
68
69        for ignore_lines in self.ignore_lines {
70            result.push(format!("--ignore-lines={}", ignore_lines));
71        }
72
73        result
74    }
75}

82    /// Whether the child hermit's `log-diff` offers
83    /// `--canonicalize-host-addresses`. `hermit_bin` comes from `HERMIT_BIN` or
84    /// the PATH, so it can be a published bundle that predates the flag, where
85    /// passing it is `error: unexpected argument` (see
86    /// `log_diff_command_does_not_pin_a_record_envelope_the_child_may_lack`).
87    fn child_offers_host_address_canonicalization(&self) -> bool {
88        std::process::Command::new(&self.hermit_bin)
89            .args(["log-diff", "--help"])
90            .stdin(std::process::Stdio::null())
91            .stderr(std::process::Stdio::null())
92            .output()
93            .is_ok_and(|output| {
94                output.status.success()
95                    && String::from_utf8_lossy(&output.stdout)
96                        .contains("--canonicalize-host-addresses")
97            })
98    }

100    fn verify_lines<S1: AsRef<str>, S2: AsRef<str>>(left: S1, right: S2) -> anyhow::Result<bool> {
101        let result = similar::TextDiff::configure()
102            .algorithm(similar::Algorithm::Myers)
103            .diff_lines(left.as_ref(), right.as_ref());
104
105        if result.ratio() == 1.0 {
106            Ok(true)
107        } else {
108            for c in result.iter_all_changes() {
109                match c.tag() {
110                    similar::ChangeTag::Equal => print!("{}", c),
111                    similar::ChangeTag::Delete => {
112                        print!("{}", format!("-{}", c).red().bold())
113                    }
114                    similar::ChangeTag::Insert => {
115                        print!("{}", format!("+{}", c).green().bold())
116                    }
117                }
118            }
119            Ok(false)
120        }
121    }

123    fn verify_files(left: &Path, right: &Path) -> anyhow::Result<bool> {

125        let right = std::fs::read_to_string(right)?;
126        Self::verify_lines(left, right)

129    /// Every difference between two schedule files in what priority-history
130    /// replay must reproduce beside the global events: each per-thread
131    /// history, whole, and each callback checkpoint. A key present on one
132    /// side only is a difference; absent on both is equal. Nothing is
133    /// projected, reordered or normalized.
134    fn priority_history_differences(
135        left: &serde_json::Value,
136        right: &serde_json::Value,
137    ) -> Vec<String> {
138        let mut differences = Vec::new();
139        for key in ["per_thread", "callback_checkpoints"] {
140            match (left.get(key), right.get(key)) {
141                (None, None) => {}
142                (Some(_), None) => differences.push(format!("{key}: in the first run only")),
143                (None, Some(_)) => differences.push(format!("{key}: in the second run only")),
144                (Some(left), Some(right)) => {
145                    Self::value_differences(key.to_owned(), left, right, &mut differences)
146                }
147            }
148        }
149        differences
150    }
151
152    /// The number of per-thread histories, their preemption rows, and the
153    /// callback checkpoints in one schedule file.
154    fn priority_history_extent(schedule: &serde_json::Value) -> (usize, usize, usize) {
155        let threads = schedule.get("per_thread").and_then(|t| t.as_object());
156        let rows = threads.map_or(0, |threads| {
157            threads
158                .values()
159                .filter_map(|t| t.get("preemption_rcbs")?.as_array().map(Vec::len))
160                .sum()
161        });
162        let checkpoints = schedule
163            .get("callback_checkpoints")
164            .and_then(|c| c.as_object())
165            .map_or(0, |c| {
166                c.values().filter_map(|t| t.as_array().map(Vec::len)).sum()
167            });
168        (threads.map_or(0, |t| t.len()), rows, checkpoints)
169    }

171    fn value_differences(
172        path: String,
173        left: &serde_json::Value,
174        right: &serde_json::Value,
175        differences: &mut Vec<String>,
176    ) {
177        use serde_json::Value;
178        if left == right {
179            return;
180        }
181        match (left, right) {
182            (Value::Object(left), Value::Object(right)) => {
183                let keys: std::collections::BTreeSet<&String> =
184                    left.keys().chain(right.keys()).collect();
185                for key in keys {
186                    match (left.get(key), right.get(key)) {
187                        (Some(l), Some(r)) => {
188                            Self::value_differences(format!("{path}.{key}"), l, r, differences)
189                        }
190                        (Some(_), None) => {
191                            differences.push(format!("{path}.{key}: in the first run only"))
192                        }
193                        (None, Some(_)) => {
194                            differences.push(format!("{path}.{key}: in the second run only"))
195                        }
196                        (None, None) => unreachable!(),
197                    }
198                }
199            }
200            (Value::Array(l), Value::Array(r)) => {
201                if l.len() != r.len() {
202                    differences.push(format!("{path}: {} entries vs {}", l.len(), r.len()));
203                }
204                for (index, (l, r)) in l.iter().zip(r).enumerate() {
205                    Self::value_differences(format!("{path}[{index}]"), l, r, differences);
206                }
207            }
208            _ => differences.push(format!("{path}: {left} vs {right}")),
209        }
210    }

212    /// The fixed point of priority-history replay, which re-records what it
213    /// replays: the two schedule files must agree exactly on every
214    /// per-thread priority history and every callback checkpoint.
215    /// `verify_schedules`, which compares only the global events, is
216    /// unchanged and checked separately.
217    pub fn verify_priority_histories(
218        &self,
219        left: &RunEnvironment,
220        right: &RunEnvironment,
221    ) -> anyhow::Result<bool> {
222        println!(
223            "{}",
224            "::  Checking that per-thread priority histories and callback checkpoints match".bold()
225        );
226        let read = |path: &Path| -> anyhow::Result<serde_json::Value> {
227            Ok(serde_json::from_reader(BufReader::new(
228                std::fs::File::open(path)?,
229            ))?)
230        };
231        let (left, right) = (read(&left.schedule_file)?, read(&right.schedule_file)?);
232        // Name what was compared, so an exact agreement of two empty
233        // histories is never mistaken for coverage.
234        let (threads, rows, checkpoints) = Self::priority_history_extent(&left);
235        println!(
236            "   first run: {threads} per-thread histories, {rows} preemption rows, \
237             {checkpoints} callback checkpoints"
238        );
239        let differences = Self::priority_history_differences(&left, &right);
240        const SHOWN: usize = 20;
241        for difference in differences.iter().take(SHOWN) {
242            println!("{}", difference.red().bold());
243        }
244        if differences.len() > SHOWN {
245            println!("... and {} more differences", differences.len() - SHOWN);
246        }
247        Ok(differences.is_empty())
248    }

250    fn format_json<TPath: AsRef<Path>>(path: TPath) -> anyhow::Result<String> {
251        let value: serde_json::Value =
252            serde_json::from_reader(BufReader::new(std::fs::File::open(path.as_ref())?))?;
253        let schedules = match &value {
254            serde_json::Value::Object(sched_file) => sched_file.get("global").ok_or_else(|| {
255                anyhow::Error::msg("expecting \"global\" key in the target json file")
256            }),
257            _ => Err(anyhow::Error::msg(format!(
258                "{} has unexpected format",
259                path.as_ref().display()
260            ))),
261        };
262        Ok(serde_json::to_string_pretty(schedules?)?)
263    }

318    /// The `log-diff` argument vector. `canonicalize_host_addresses` says
319    /// whether the child hermit offers `--canonicalize-host-addresses`.
320    ///
321    /// The two logs come from two hermit processes, so ASLR moves every marked
322    /// launcher pointer (`<hostaddr 0x...>`) between them; compared by raw
323    /// value they always differ. The flag compares them by per-run ordinal
324    /// instead, keeping identity, order and aliasing
325    /// (<https://github.com/rrnewton/hermit/issues/3412>).
326    fn build_command_args(
327        &self,
328        left: &RunEnvironment,
329        right: &RunEnvironment,
330        options: LogDiffOptions,
331        canonicalize_host_addresses: bool,
332    ) -> Vec<String> {
333        let mut result = vec![];
334        result.push(String::from("log-diff"));
335        result.append(&mut options.into_args());
336        if canonicalize_host_addresses {
337            result.push("--canonicalize-host-addresses".to_owned());
338        }
339        result.push(format!("{}", left.log_file_path.display()));
340        result.push(format!("{}", right.log_file_path.display()));
341
342        result
343    }

345    pub fn verify_logs(
346        &self,
347        left: &RunEnvironment,
348        right: &RunEnvironment,
349        options: LogDiffOptions,
350    ) -> anyhow::Result<bool> {
351        println!("{}", "::  Comparing log files".bold());
352        let canonicalize_host_addresses = self.child_offers_host_address_canonicalization();
353        if !canonicalize_host_addresses {
354            println!(
355                "    (this hermit's log-diff has no --canonicalize-host-addresses; \
356                 marked host addresses are compared by raw value)"
357            );
358        }
359        let mut command = std::process::Command::new(&self.hermit_bin);
360        command.args(self.build_command_args(left, right, options, canonicalize_host_addresses));
361
362        println!("{}", format!("    {}", display_cmd(&command)).bold());
363        Ok(command.status()?.success())
364    }

420    /// The priority-history fixed point passes only on exact agreement of
421    /// every per-thread history and every checkpoint: one changed field, a
422    /// missing checkpoint map, a removed checkpoint and an added row each
423    /// fail, named by path, while the global events are left to
424    /// `verify_schedules`.
425    #[test]
426    fn priority_histories_must_agree_exactly() {
427        let left = priority_schedule();
428        assert!(Verify::<PathBuf>::priority_history_differences(&left, &left).is_empty());
429        assert_eq!(Verify::<PathBuf>::priority_history_extent(&left), (1, 1, 1));
430        assert_eq!(
431            Verify::<PathBuf>::priority_history_extent(&serde_json::json!({"global": []})),
432            (0, 0, 0)
433        );
434        let mut global_only = left.clone();
435        global_only["global"] = serde_json::json!([]);
436        assert!(
437            Verify::<PathBuf>::priority_history_differences(&left, &global_only).is_empty(),
438            "global events are verify_schedules' to compare"
439        );
440
441        let mut site = left.clone();
442        site["callback_checkpoints"]["3"][0]["origin"]["site"] = serde_json::json!("Prehook");
443        assert_eq!(
444            Verify::<PathBuf>::priority_history_differences(&left, &site),
445            vec!["callback_checkpoints.3[0].origin.site: \"Posthook\" vs \"Prehook\"".to_owned()]
446        );
447
448        let mut absent = left.clone();
449        absent
450            .as_object_mut()
451            .unwrap()
452            .remove("callback_checkpoints");
453        assert_eq!(
454            Verify::<PathBuf>::priority_history_differences(&left, &absent),
455            vec!["callback_checkpoints: in the first run only".to_owned()]
456        );
457
458        let mut removed = left.clone();
459        removed["callback_checkpoints"]["3"] = serde_json::json!([]);
460        assert_eq!(
461            Verify::<PathBuf>::priority_history_differences(&left, &removed),
462            vec!["callback_checkpoints.3: 1 entries vs 0".to_owned()]
463        );
464
465        let mut added = left.clone();
466        added["per_thread"]["3"]["preemption_rcbs"] = serde_json::json!([1253548, 2000000]);
467        assert_eq!(
468            Verify::<PathBuf>::priority_history_differences(&left, &added),
469            vec!["per_thread.3.preemption_rcbs: 1 entries vs 2".to_owned()]
470        );
471
472        let mut thread = left.clone();
473        thread["per_thread"]["5"] = left["per_thread"]["3"].clone();
474        assert_eq!(
475            Verify::<PathBuf>::priority_history_differences(&left, &thread),
476            vec!["per_thread.5: in the second run only".to_owned()]
477        );
478    }

579        assert!(Verify::new(&new).child_offers_host_address_canonicalization());
