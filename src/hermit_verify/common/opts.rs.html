9use std::ffi::OsString;
10use std::path::PathBuf;
11use std::time::SystemTime;
12
13use clap::Parser;
14
15#[derive(Parser, Debug)]
16pub struct CommonOpts {
17    /// Specify hermit binary path.
18    #[clap(long, env = "HERMIT_BIN", default_value = "hermit")]
19    pub hermit_bin: PathBuf,
20
21    // In test environment allows to specify root directory with produced artifacts
22    #[clap(long, env = "TEST_RESULT_ARTIFACTS_DIR")]
23    pub test_result_artifact_dir: Option<PathBuf>,
24
25    // Allows to ignore TEST_RESULT_ARTIFACTS_DIR when running in test environment
26    #[clap(long, default_value = "false")]
27    pub ignore_test_result_artifacts_dir: bool,
28
29    // Specify root directory with produced artifacts
30    #[clap(long)]
31    pub temp_dir_path: Option<PathBuf>,
32
33    /// Allow the run itself to exit with a nonzero code, but still count the
34    /// verification as successful if all runs match each other.
35    #[clap(long)]
36    pub allow_nonzero_exit: bool,
37
38    /// Print extra information while running.
39    #[clap(long, short = 'v')]
40    pub verbose: bool,
41
42    /// The virtual-time epoch given to every hermit run of this verification.
43    ///
44    /// `hermit run` samples the host clock for its epoch when none is supplied,
45    /// so two separate runs would otherwise start their virtual clocks at two
46    /// different instants and every compared time would differ
47    /// (<https://github.com/rrnewton/hermit/issues/3411>). `None` when
48    /// `HERMIT_EPOCH` is set: each run then inherits that one explicit value.
49    #[clap(skip)]
50    pub comparison_epoch: Option<String>,
51}
52
53impl CommonOpts {
54    /// Resolve [`Self::comparison_epoch`] once, before the first run.
55    pub fn resolve_comparison_epoch(
56        &mut self,
57        environment_epoch: Option<OsString>,
58        capture_now: impl FnOnce() -> SystemTime,
59    ) {
60        self.comparison_epoch = match environment_epoch {
61            Some(_) => None,
62            None => Some(detcore_model::config::epoch_from_host_time(capture_now()).to_rfc3339()),
63        };
64    }
65
66    /// Give a `hermit ... run ...` argument vector this verification's epoch,
67    /// unless the caller's own `--hermit-arg` already chose one.
68    pub fn pin_comparison_epoch(&self, mut hermit_args: Vec<String>) -> Vec<String> {
69        let Some(epoch) = &self.comparison_epoch else {
70            return hermit_args;
71        };
72        let Some(run) = hermit_args.iter().position(|arg| arg == "run") else {
73            return hermit_args;
74        };
75        let run_options = hermit_args[run + 1..]
76            .iter()
77            .take_while(|arg| arg.as_str() != "--");
78        if run_options
79            .clone()
80            .any(|arg| arg == "--epoch" || arg.starts_with("--epoch="))
81        {
82            return hermit_args;
83        }
84        hermit_args.insert(run + 1, format!("--epoch={epoch}"));
85        hermit_args
86    }
87}

89#[cfg(test)]
90mod tests {
91    use std::time::Duration;
92    use std::time::UNIX_EPOCH;
93
94    use super::*;
95
96    fn opts() -> CommonOpts {
97        CommonOpts::parse_from(["hermit-verify"])
98    }
99
100    fn strings(args: &[&str]) -> Vec<String> {
101        args.iter().map(|arg| (*arg).to_owned()).collect()
102    }
103
104    #[test]
105    fn one_captured_epoch_is_given_to_every_run() {
106        let mut opts = opts();
107        opts.resolve_comparison_epoch(None, || {
108            UNIX_EPOCH + Duration::new(978_307_199, 123_456_789)
109        });
110        let args = strings(&["--log=trace", "run", "--chaos", "--", "/bin/true", "run"]);
111        let expected = strings(&[
112            "--log=trace",
113            "run",
114            "--epoch=2000-12-31T23:59:59.123456789+00:00",
115            "--chaos",
116            "--",
117            "/bin/true",
118            "run",
119        ]);
120        assert_eq!(opts.pin_comparison_epoch(args.clone()), expected);
121        assert_eq!(opts.pin_comparison_epoch(args), expected);
122    }
123
124    #[test]
125    fn an_explicit_epoch_is_never_replaced_or_duplicated() {
126        let mut opts = opts();
127        opts.resolve_comparison_epoch(None, || UNIX_EPOCH);
128        for explicit in [
129            strings(&["run", "--epoch=2026-01-01T00:00:00Z", "--", "/bin/true"]),
130            strings(&["run", "--epoch", "2026-01-01T00:00:00Z", "--", "/bin/true"]),
131        ] {
132            assert_eq!(opts.pin_comparison_epoch(explicit.clone()), explicit);
133        }
134        // A guest argument spelled like the option does not count.
135        assert_eq!(
136            opts.pin_comparison_epoch(strings(&["run", "--", "/bin/echo", "--epoch=x"])),
137            strings(&[
138                "run",
139                "--epoch=1970-01-01T00:00:00+00:00",
140                "--",
141                "/bin/echo",
142                "--epoch=x"
143            ]),
144        );
145    }
146
147    #[test]
148    fn an_environment_epoch_is_inherited_without_reading_the_clock() {
149        let mut opts = opts();
150        opts.resolve_comparison_epoch(Some("2026-01-01T00:00:00Z".into()), || {
151            panic!("an explicit environment epoch must not sample the clock")
152        });
153        assert_eq!(opts.comparison_epoch, None);
154        let args = strings(&["run", "--", "/bin/true"]);
155        assert_eq!(opts.pin_comparison_epoch(args.clone()), args);
156    }
157}
