25/// Attempts at one iteration while Hermit refuses its runs for PMU skid
26/// overshoot.
27const SKID_ATTEMPTS: u32 = 4;
28
29/// The prefix of the line Hermit prints when it refuses a run for PMU skid
30/// overshoot (`classify_failure` in hermit-cli/src/bin/hermit/main.rs).
31const SKID_OVERSHOOT_REFUSAL_PREFIX: &str =
32    "HERMIT_POLICY_REFUSAL class=policy-refusal cause=skid-overshoot count=";
33
34/// Hermit's own outcomes that no guest produced, each the exit status from the
35/// allocation table in hermit-cli/src/lib.rs with the stderr marker that tells
36/// it apart from a guest choosing the same status. A run that ends in one of
37/// these says nothing about the guest.
38const HERMIT_OUTCOMES: [(i32, &str); 5] = [
39    (HERMIT_POLICY_REFUSAL_EXIT, "HERMIT_POLICY_REFUSAL "),
40    (123, "HERMIT_LOG_CAP "),
41    (124, "HERMIT_RUN_TIMEOUT "),
42    (124, "HERMIT_RUN_TIMEOUT_FALLBACK"),
43    (125, "HERMIT_INTERNAL_FAILURE "),
44];
45
46/// Verification utility for replaying preemptions under hermit
47/// This utility runs a guest program under "hermit run" and records schedules and preemptions. On the second run those recorded schedules are replayed via "hermit run" subcommand
48/// The outputs (stdout, stderr, log file, schedules) are captured and placed in a temporary directory allowing further inspection (if --keep-temp-dir is provided)
49#[derive(Parser, Debug)]
50pub struct ChaosStressOpts {
51    // Maximum number of iterations performed during the test
52    #[clap(long, default_value = "100")]
53    max_iterations_count: i32,
54    /// Path to a guest program
55    #[clap(value_name = "PROGRAM")]
56    guest_program: PathBuf,
57    /// Additional argument for hermit run subcommand
58    #[clap(long)]
59    hermit_arg: Vec<String>,
60    /// Arguments for a guest program
61    #[clap(value_name = "ARGS")]
62    args: Vec<String>,
63}
64
65/// What one chaos run showed about the guest.
66#[derive(Debug, PartialEq, Eq)]
67enum RunOutcome {
68    /// The guest exited 0.
69    Passed,
70    /// The guest exited nonzero or died of a signal.
71    Failed,
72    /// Hermit refused the run for PMU skid overshoot, which measures the
73    /// host's interrupt latency rather than the guest
74    /// (<https://github.com/rrnewton/hermit/issues/1845>). The refusal line.
75    SkidRefused(String),
76}
77
78/// Classify one finished `hermit run` by its exit status and stderr. Only the
79/// guest's own outcome counts: a skid refusal is returned for a retry, and any
80/// other outcome of Hermit's own ([`HERMIT_OUTCOMES`]) is an error, because
81/// counting it as the guest failing would report a race nobody found
82/// (<https://github.com/rrnewton/hermit/issues/4490>).
83///
84/// Hermit raises a guest's fatal signal on itself, so a signal death is the
85/// guest failing.
86fn classify(status: ExitStatus, stderr: &str) -> anyhow::Result<RunOutcome> {
87    let Some(code) = status.code() else {
88        return Ok(RunOutcome::Failed);
89    };
90    let marker = |line: &str| {
91        HERMIT_OUTCOMES
92            .iter()
93            .any(|(status, prefix)| *status == code && line.starts_with(prefix))
94    };
95    let hermit_lines: Vec<&str> = stderr
96        .lines()
97        .map(str::trim_end)
98        .filter(|line| marker(line))
99        .collect();
100    match hermit_lines.as_slice() {
101        [] if code == 0 => Ok(RunOutcome::Passed),
102        [] => Ok(RunOutcome::Failed),
103        [refusal] if is_skid_refusal(refusal, stderr) => {
104            Ok(RunOutcome::SkidRefused(refusal.to_string()))
105        }
106        lines => anyhow::bail!(
107            "Hermit itself ended the run with status {code}, so it says nothing about the guest:\n{}",
108            lines.join("\n")
109        ),
110    }
111}

113/// Whether REFUSAL is the typed skid-overshoot refusal: a positive count, and
114/// Hermit's `HERMIT_SKID_OVERSHOOT` report of the late interrupt beside it
115/// with no panic.
116fn is_skid_refusal(refusal: &str, stderr: &str) -> bool {
117    let counted = refusal
118        .strip_prefix(SKID_OVERSHOOT_REFUSAL_PREFIX)
119        .is_some_and(|count| {
120            count.bytes().all(|byte| byte.is_ascii_digit())
121                && count.parse::<u64>().is_ok_and(|count| count > 0)
122        });
123    counted
124        && stderr
125            .lines()
126            .any(|line| line.starts_with("HERMIT_SKID_OVERSHOOT "))
127        && !stderr.contains("panicked at")
128}

130impl ChaosStressOpts {
131    /// One `hermit run` of the guest under a fresh chaos seed, its stderr
132    /// passed through as it arrives and kept for [`classify`].
133    fn run_hermit(&self, common_args: &CommonOpts) -> anyhow::Result<RunOutcome> {
134        let mut rng = rand::rng();
135        let random_seed: u16 = rng.random();
136        let hermit_cli_args = Hermit::new()
137            .run(self.guest_program.clone(), self.args.clone())
138            .hermit_args(self.hermit_arg.clone())
139            .seed(i32::from(random_seed))
140            .log_level(tracing::Level::WARN)
141            .into_args();
142
143        let mut command = std::process::Command::new(&common_args.hermit_bin);
144        command.args(hermit_cli_args).stderr(Stdio::piped());
145        println!("   {}", display_cmd(&command).white().bold());
146        let started = Instant::now();
147        let mut child = command.spawn()?;
148        let mut stderr = String::new();
149        let mut reader = std::io::BufReader::new(child.stderr.take().expect("stderr is piped"));
150        let mut line = Vec::new();
151        while reader.read_until(b'\n', &mut line)? > 0 {
152            std::io::stderr().write_all(&line)?;
153            stderr.push_str(&String::from_utf8_lossy(&line));
154            line.clear();
155        }
156        let status = child.wait()?;
157        let status_text = match status.code() {
158            Some(code) => code.to_string(),
159            None => status.to_string(),
160        };
161        println!(
162            "Executed command with status: {status_text} in {:.1?}",
163            started.elapsed()
164        );
165        classify(status, &stderr)
166    }
167
168    pub fn run(&self, common_args: &CommonOpts) -> anyhow::Result<bool> {
169        println!("Run at max {} iterations", self.max_iterations_count);
170        let mut success_count = 0;
171        let mut fail_count = 0;
172        let mut skid_retries = 0;
173
174        for i in 1..=self.max_iterations_count {
175            println!("{}", format!(":: Run #{}...", i).yellow().bold());
176            let mut attempt = 1;
177            let outcome = loop {
178                match self.run_hermit(common_args)? {
179                    RunOutcome::SkidRefused(refusal) if attempt < SKID_ATTEMPTS => {
180                        eprintln!("SKID-RETRY run #{i} attempt {attempt}: {refusal}");
181                        skid_retries += 1;
182                        attempt += 1;
183                    }
184                    RunOutcome::SkidRefused(refusal) => anyhow::bail!(
185                        "Hermit refused run #{i} for PMU skid overshoot on all {SKID_ATTEMPTS} attempts: {refusal}"
186                    ),
187                    outcome => break outcome,
188                }
189            };
190            match outcome {
191                RunOutcome::Passed => success_count += 1,
192                RunOutcome::Failed => fail_count += 1,
193                RunOutcome::SkidRefused(_) => unreachable!("retried above"),
194            }
195
196            if success_count > 0 && fail_count > 0 {
197                println!("Found a passing and a failing run; {skid_retries} skid refusals retried");
198                return Ok(true);
199            }
200        }
201
202        println!(
203            "{success_count} passing and {fail_count} failing runs; {skid_retries} skid refusals retried"
204        );
205        Ok(false)
206    }
207}

209#[cfg(test)]
210mod tests {
211    use std::os::unix::process::ExitStatusExt as _;
212
213    use super::*;
214
215    const SKID_REFUSAL: &str = "HERMIT_SKID_OVERSHOOT rcb_actual=240935 rcb_target=240682 skid_margin=1000 overshoot=253\n\
216        HERMIT_POLICY_REFUSAL class=policy-refusal cause=skid-overshoot count=1\n";
217
218    fn exited(code: i32) -> ExitStatus {
219        ExitStatus::from_raw(code << 8)
220    }
221
222    #[test]
223    fn the_guest_status_decides_a_run() {
224        assert_eq!(classify(exited(0), "").unwrap(), RunOutcome::Passed);
225        assert_eq!(classify(exited(1), "").unwrap(), RunOutcome::Failed);
226        // SIGABRT, raised on Hermit for the guest's own abort.
227        assert_eq!(
228            classify(ExitStatus::from_raw(6), "").unwrap(),
229            RunOutcome::Failed
230        );
231        // A guest may exit 122 or 125 itself, and may print a marker under
232        // another status.
233        assert_eq!(classify(exited(122), "").unwrap(), RunOutcome::Failed);
234        assert_eq!(
235            classify(exited(1), "HERMIT_INTERNAL_FAILURE class=cli-error\n").unwrap(),
236            RunOutcome::Failed
237        );
238    }
239
240    #[test]
241    fn a_skid_refusal_is_retried() {
242        assert_eq!(
243            classify(exited(122), SKID_REFUSAL).unwrap(),
244            RunOutcome::SkidRefused(
245                "HERMIT_POLICY_REFUSAL class=policy-refusal cause=skid-overshoot count=1".into()
246            )
247        );
248        // Without Hermit's overshoot report, or beside a panic, it is a refusal
249        // like any other.
250        let unreported =
251            "HERMIT_POLICY_REFUSAL class=policy-refusal cause=skid-overshoot count=1\n";
252        assert!(classify(exited(122), unreported).is_err());
253        let panicked = format!("thread 'main' panicked at x.rs:1:1\n{SKID_REFUSAL}");
254        assert!(classify(exited(122), &panicked).is_err());
255    }
256
257    #[test]
258    fn hermit_s_own_outcomes_end_the_search() {
259        for (code, stderr) in [
260            (122, "HERMIT_POLICY_REFUSAL class=policy-refusal\n"),
261            (123, "HERMIT_LOG_CAP class=log-cap\n"),
262            (124, "HERMIT_RUN_TIMEOUT class=run-timeout\n"),
263            (125, "HERMIT_INTERNAL_FAILURE class=container-child-panic\n"),
264        ] {
265            let error = classify(exited(code), stderr).unwrap_err().to_string();
266            assert!(error.contains(stderr.trim_end()), "{error}");
267        }
268    }
